```python
    best_t = (lo + hi) / 2
    best_obj = compute_obj((1-best_t)*a + best_t*g)
    return best_t, best_obj

def coordinate_descent_fast(a, time_budget=60.0, seed=42):
    """Fast coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a

            # for k = 2i, conv[k] += x^2 - v^2
            k_min = i  # minimum k where a[i] appears (k = i + 0, but we need k >= i)
            # Actually: a[i] appears in conv[k] when k-i is a valid index of a
            # conv[k] = sum_{j} a[j]*a[k-j], so a[i] appears when j=i or k-j=i
            # i.e., k-i is a valid index (for the j=i term) or i is a valid index (for k-j=i)
            # For j=i: need 0 <= k-i < n, i.e., i <= k <= i+n-1
            # For k-j=i: need 0 <= j < n and j = k-i, same condition
            # So a[i] appears in conv[k] for k in [i, i+n-1] (capped at [0, 2n-2])
            
            k_lo = max(0, i)
            k_hi = min(2*n-2, i+n-1)
            
            # For each such k, the coefficient of x (new value) in conv[k] is:
            # If k != 2i: coefficient is 2*a[k-i] (from the cross terms)
            # If k == 2i: there's also x^2 term
            ks = np.arange(k_lo, k_hi+1)
            # conv[k] = (contribution from a[i]) + (rest)
            # Contribution of a[i] to conv[k]:
            #   if k != 2i: a[i] * a[k-i] + a[k-i] * a[i] = 2*a[i]*a[k-i] (if k-i != i)
            #   if k == 2i: a[i] * a[i] = a[i]^2
            # Wait, more carefully:
            # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # The terms involving a[i] are:
            #   j=i: a[i]*a[k-i] (valid if 0<=k-i<n)
            #   k-j=i => j=k-i: a[k-i]*a[i] (valid if 0<=k-i<n)
            # If i != k-i (i.e., k != 2i): both terms present, total = 2*a[i]*a[k-i]
            # If i == k-i (i.e., k == 2i): only one term, total = a[i]^2

                # There's a quadratic term at k=2i
                mask_quad = (ks == 2*i)
                mask_lin = ~mask_quad
                a_lin = a[ks[mask_lin] - i]  # a[k-i] for linear terms
                c_lin = conv[ks[mask_lin]] - 2*v*a_lin  # current conv minus a[i]'s contribution
                # Wait, this is getting complicated. Let me just evaluate f(x) directly.
                pass
            
            # Simpler approach: just evaluate the objective at a few points + golden section
            # f(x) = 2n * max(conv_after) / (S - v + x)^2
            # where conv_after is conv with a[i] changed from v to x
            
            def f_1d(x):
                if x < 0:
                    return 1e30
                # Update conv
                new_conv = conv.copy()
                for k in range(k_lo, k_hi+1):
                    j = k - i
                    if 0 <= j < n:

                        else:

                new_S = S - v + x
                if new_S < 1e-15:
                    return 1e30
                return 2.0 * n * float(np.max(new_conv)) / (new_S * new_S)

            # Better to precompute and use vectorized ops.
            
            # Actually for n=800, the number of k values is at most n=800, 
            # and we do maybe 30 evaluations per coordinate, so 800*30*800 = 19.2M ops per sweep.
            # That's too slow for 800 coordinates.
            
            # Let me use a vectorized approach:
            # Precompute the coefficients
            j_vals = ks - i  # these are the other indices
            valid = (j_vals >= 0) & (j_vals < n)
            ks_valid = ks[valid]
            j_valid = j_vals[valid]
            a_j = a[j_valid]
            
            is_quad = (ks_valid == 2*i)

            # We can write: conv_new[k] = conv[k] + (x-v) * coeff_lin[k] + (x^2-v^2) * is_quad[k]
            coeff = 2.0 * a_j
            coeff[is_quad] = 0  # For quad term, the linear part is different
            
            # Actually: for k=2i, the change is x^2 - v^2 (not 2*(x-v)*a[i])
            # For k != 2i, the change is 2*(x-v)*a[k-i]
            # So: conv_new[k] = conv[k] + 2*(x-v)*a_j[k]*(1-is_quad[k]) + (x^2-v^2)*is_quad[k]
            
            base = conv[ks_valid].copy()
            lin_coeff = 2.0 * a_j * (~is_quad).astype(float)
            quad_mask = is_quad.astype(float)
            
            # f(x) = 2n * max(base + lin_coeff*(x-v) + quad_mask*(x^2-v^2)) / (S-v+x)^2
            # = 2n * max(base + lin_coeff*(x-v) + quad_mask*(x^2-v^2)) / (S-v+x)^2
            
            # Let's define h(x) = max_k [base[k] + lin_coeff[k]*(x-v) + quad_mask[k]*(x^2-v^2)]
            # This is the max of linear functions (for non-quad) and one quadratic (for quad)
            # We can evaluate this at specific x values
            
            def f_1d_vec(x, base=base, lin_coeff=lin_coeff, quad_mask=quad_mask,
                        v=v, S=S, n=n):
                vals = base + lin_coeff * (x - v) + quad_mask * (x*x - v*v)
                max_val = float(np.max(vals))
                new_S = S - v + x
                if new_S < 1e-15:
                    return 1e30
                return 2.0 * n * max_val / (new_S * new_S)
            
            # Golden section search
            scale = max(v, 1.0/n, 1e-6)
            x_lo = 0.0
            x_hi = v + 2.0 * scale
            
            # First try a few candidate points
            candidates = [0.0, v*0.5, v*0.8, v, v*1.2, v*1.5, 2.0/n, 3.0/n]
            candidates = [max(0.0, min(c, x_hi)) for c in candidates]
            
            best_x = v
            best_f = f_1d_vec(v)
            for xc in candidates:
                fc = f_1d_vec(xc)
                if fc < best_f:
                    best_f = fc
                    best_x = xc
            
            # Golden section around best_x
            s2 = max(best_x, 1.0/n, 1e-6) * 0.5
            x_lo = max(0.0, best_x - s2)
            x_hi = best_x + s2
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d_vec(c_pt)
            fd = f_1d_vec(d_pt)
            
            for _ in range(30):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d_vec(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d_vec(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue
            
            # Update a and conv
            delta = x_opt - v
            a[i] = x_opt
            for idx, k in enumerate(ks_valid):
                conv[k] += lin_coeff[idx] * delta + quad_mask[idx] * (x_opt*x_opt - v*v)
            S += delta
            
            obj = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a
```

Actually, let me reconsider the LP formulation. The standard LP is:
min M
s.t. (a*g)[k] <= M for all k
     sum(g) = 1
     g >= 0

The optimal M is the minimum possible peak of a*g over all probability vectors g. This is a lower bound on what we can achieve in one step (since we're mixing a and g, not replacing a with g).

A better formulation might be:
min M
s.t. ((1-t)*a + t*g) * ((1-t)*a + t*g))[k] <= M for all k
     sum(g) = 1
     g >= 0

But this is non-linear (quadratic in t and g). For fixed t, it's:
min M
s.t. (1-t)^2 (a*a)[k] + 2t(1-t)(a*g)[k] + t^2(g*g)[k] <= M

This is non-linear in g (the t^2(g*g) term). So we can't make it a simple LP.

The linearization approach (ignoring the g*g term) gives:
min M
s.t. (1-t)^2 (a*a)[k] + 2t(1-t)(a*g)[k] <= M + something

Which simplifies to the standard LP when we're just looking for a direction.

OK let me just go with the standard LP approach but make it faster and iterate more.

For n=800, the LP has:
- 800 variables (g)
- 1 extra variable (M)
- 2*800-1 = 1599 inequality constraints (a*g)[k] <= M
- 1 equality constraint: sum(g) = 1
- Non-negativity: g >= 0

This is a small LP. With MOSEK or CLARABEL, it should solve in < 1 second.

```python
"""
Optimizes step function coefficients using LP-based sequential optimization
with coordinate descent polishing. Key improvements over previous approach:
1. Heavier LP focus with proper line search
2. Efficient convolution matrix construction
3. Adaptive restart on stall
4. Multi-n exploration
"""
import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix B (2n-1 x n) where B @ g = conv(a, g)."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = a
    return B


def solve_lp(a, time_budget=15.0):
    """
    Solve: min M s.t. B(a) @ g <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    B = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [B @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_budget:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, n_evals=25):
    """Find optimal t for a_new = (1-t)*a + t*g using golden section."""
    a = a / float(np.sum(a))
    g = g / float(np.sum(g))
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj((1 - c) * a + c * g)
    fd = compute_obj((1 - d) * a + d * g)
    
    for _ in range(n_evals):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = compute_obj((1 - c) * a + c * g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = compute_obj((1 - d) * a + d * g)
    
    t_opt = (lo + hi) / 2.0
    obj_opt = compute_obj((1 - t_opt) * a + t_opt * g)
    # Also check endpoints
    obj_0 = compute_obj(a)
    obj_1 = compute_obj(g)
    if obj_0 < obj_opt:
        t_opt = 0.0
        obj_opt = obj_0
    if obj_1 < obj_opt:
        t_opt = 1.0
        obj_opt = obj_1
    return t_opt, obj_opt


def coordinate_descent(a, time_budget=60.0, seed=42):
    """Efficient coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_lo, k_hi + 1)
            j_vals = ks - i
            a_j = a[j_vals]
            is_quad = (ks == 2 * i)
            
            base = conv[ks].copy()
            lin_coeff = np.zeros_like(ks, dtype=float)
            lin_coeff[~is_quad] = 2.0 * a_j[~is_quad]
            quad_mask = is_quad.astype(float)
            
            S_other = S - v
            
            def f_1d(x, base=base, lin_coeff=lin_coeff, quad_mask=quad_mask,
                     v=v, S_other=S_other, n=n):
                vals = base + lin_coeff * (x - v) + quad_mask * (x * x - v * v)
                max_val = float(np.max(vals))
                new_S = S_other + x
                if new_S < 1e-15:
                    return 1e30
                return 2.0 * n * max_val / (new_S * new_S)
            
            # Candidates
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.6, v, v * 1.4, v * 1.8,
                          1.0 / n, 2.0 / n, 3.0 / n]
            candidates = sorted(set(max(0.0, min(c, 3.0 * scale)) for c in candidates))
            
            best_x = v
            best_f = f_1d(v)
            for xc in candidates:
                fc = f_1d(xc)
                if fc < best_f:
                    best_f = fc
                    best_x = xc
            
            # Golden section
            s2 = max(best_x, 1.0 / n, 1e-6) * 0.5
            x_lo = max(0.0, best_x - s2)
            x_hi = best_x + s2 * 1.5
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            
            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue
            
            delta = x_opt - v
            a[i] = x_opt
            conv[ks] += lin_coeff * delta + quad_mask * (x_opt * x_opt - v * v)
            S += delta
            
            obj = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a


def lp_iteration(a, time_budget=15.0):
    """One LP iteration: solve LP, line search, return new a and improvement."""
    g, M = solve_lp(a, time_budget=time_budget)
    if g is None:
        return a, 0.0, None
    
    t, obj_new = line_search_blend(a, g)
    if t <= 0:
        return a, 0.0, g
    
    a_new = (1 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a, 0.0, g
    a_new = a_new / s
    improvement = compute_obj(a) - obj_new
    return a_new, improvement, g


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a += 0.1
        a = a / a.sum()
    elif mode == 4:
        # Gaussian
        x = np.arange(n) - n / 2
        a = np.exp(-x**2 / (2 * (n/4)**2))
        a = a / a.sum()
    elif mode == 5:
        # Two peaks
        x = np.arange(n)
        a = np.exp(-((x - n//4)**2) / (2*(n//8)**2)) + np.exp(-((x - 3*n//4)**2) / (2*(n//8)**2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0) / max(float(np.sum(np.maximum(a, 0))), 1e-10)
```

```python
"""
Optimizes step function coefficients using LP-based sequential optimization
with coordinate descent polishing. Key improvements:
1. Heavier LP focus with proper golden-section line search
2. Efficient convolution matrix via numpy slicing
3. Active-set LP for speed (top K constraints)
4. Adaptive perturbation on stall
5. Multi-n exploration with resampling
6. Smoothed gradient multiplicative updates as alternative direction

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix B (2n-1 x n) where B @ g computes conv(a, g)."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = a
    return B


def solve_lp_full(a, time_budget=15.0):
    """Solve: min M s.t. B(a) @ g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    B = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [B @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_budget:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_active(a, K, time_budget=12.0):
    """Solve LP with only top-K active constraints for speed."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)

    # Get top K lag indices
    top_k = np.argsort(conv)[-K:]
    top_k = sorted(top_k.tolist())

    # Build sub-matrix
    B_full = build_conv_matrix(a_norm)
    B_sub = B_full[top_k, :]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [B_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_budget:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, max_iter=30):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g using golden section."""
    a = a / float(np.sum(a))
    g = g / float(np.sum(g))

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj((1.0 - c) * a + c * g)
    fd = compute_obj((1.0 - d) * a + d * g)

    for _ in range(max_iter):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = compute_obj((1.0 - c) * a + c * g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = compute_obj((1.0 - d) * a + d * g)

    t_opt = (lo + hi) / 2.0
    obj_opt = compute_obj((1.0 - t_opt) * a + t_opt * g)

    # Check endpoints
    obj_0 = compute_obj(a)
    if obj_0 < obj_opt:
        t_opt = 0.0
        obj_opt = obj_0
    obj_1 = compute_obj(g)
    if obj_1 < obj_opt:
        t_opt = 1.0
        obj_opt = obj_1

    return t_opt, obj_opt


def coordinate_descent(a, time_budget=60.0, seed=42):
    """Efficient coordinate descent with golden section 1D minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_lo, k_hi + 1)
            j_vals = ks - i
            a_j = a[j_vals]
            is_quad = (ks == 2 * i)

            base = conv[ks].copy()
            lin_coeff = np.zeros(len(ks), dtype=np.float64)
            lin_coeff[~is_quad] = 2.0 * a_j[~is_quad]
            quad_mask = is_quad.astype(np.float64)

            S_other = S - v

            # 1D function: f(x) = 2n * max(base + lin*(x-v) + quad*(x^2-v^2)) / (S_other+x)^2

            # Golden section search
            scale = max(v, 1.0 / n, 1e-6)
            x_lo = 0.0
            x_hi = v + 2.0 * scale

            # Quick candidate evaluation
            candidates = [0.0, v * 0.4, v * 0.7, v, v * 1.3, v * 1.7, 1.5 / n, 2.5 / n]
            candidates = [max(0.0, min(c, x_hi)) for c in candidates]

            best_x = v
            best_f = 2.0 * n * float(np.max(base)) / (S * S)  # current value
            for xc in candidates:
                vals = base + lin_coeff * (xc - v) + quad_mask * (xc * xc - v * v)
                max_val = float(np.max(vals))
                ns = S_other + xc
                if ns < 1e-15:
                    continue
                fx = 2.0 * n * max_val / (ns * ns)
                if fx < best_f:
                    best_f = fx
                    best_x = xc

            # Golden section refine
            s2 = max(best_x, 1.0 / n, 1e-6) * 0.5
            x_lo = max(0.0, best_x - s2)
            x_hi = best_x + s2 * 1.5
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)

            vals_c = base + lin_coeff * (c_pt - v) + quad_mask * (c_pt * c_pt - v * v)
            ns_c = S_other + c_pt
            fc = 2.0 * n * float(np.max(vals_c)) / (ns_c * ns_c) if ns_c > 1e-15 else 1e30

            vals_d = base + lin_coeff * (d_pt - v) + quad_mask * (d_pt * d_pt - v * v)
            ns_d = S_other + d_pt
            fd = 2.0 * n * float(np.max(vals_d)) / (ns_d * ns_d) if ns_d > 1e-15 else 1e30

            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    vals_c = base + lin_coeff * (c_pt - v) + quad_mask * (c_pt * c_pt - v * v)
                    ns_c = S_other + c_pt
                    fc = 2.0 * n * float(np.max(vals_c)) / (ns_c * ns_c) if ns_c > 1e-15 else 1e30
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    vals_d = base + lin_coeff * (d_pt - v) + quad_mask * (d_pt * d_pt - v * v)
                    ns_d = S_other + d_pt
                    fd = 2.0 * n * float(np.max(vals_d)) / (ns_d * ns_d) if ns_d > 1e-15 else 1e30

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            conv[ks] += lin_coeff * delta + quad_mask * (x_opt * x_opt - v * v)
            S += delta

            obj = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def smoothed_gradient_update(a, time_budget=30.0, tau=50.0):
    """Multiplicative update using smoothed gradient of max(conv)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr = 5.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Softmax weights over lags
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (approx)
        # Use FFT for cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        grad_full = ifft(P * np.conj(A)).real
        grad = 2.0 * grad_full[:n]

        # Multiplicative update
        update = np.exp(-lr * grad)
        update = np.clip(update, 0.01, 50.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(a_new.sum())

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        # Increase tau gradually to sharpen
        tau *= 1.05
        lr *= 0.995

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 0.01)
        a = a / a.sum()
    elif mode == 3:
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n)) + 0.1
        a = a / a.sum()
    elif mode == 4:
        x = np.arange(n) - n / 2
        a = np.exp(-x ** 2 / (2 * (n / 4) ** 2))
        a = a / a.sum()
    elif mode == 5:
        x = np.arange(n)
        a = np.exp(-((x - n // 4) ** 2) / (2 * (n // 8) ** 2)) + \
            np.exp(-((x - 3 * n // 4) ** 2) / (2 * (n // 8) ** 2))
        a = a / a.sum()
    elif mode == 6:
        a = rng.dirichlet(np.ones(n) * 0.5)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = float(a.sum())
    return a / s if s > 1e-10 else np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
```
