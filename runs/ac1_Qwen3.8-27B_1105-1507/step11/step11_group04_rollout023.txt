import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:

                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    grad1 = P_matrix @ a  # (n,)
    
    # For grad2: row j is p[max(0,j-n+1):j+1] reversed, dot a[0:min(j+1,n)]
    # This is trickier with variable length. Let me use:
    # grad2[j] = sum_{m=0}^{n-1} p[j-m] * a[m] (where j-m >= 0)
    # = sum_{m=0}^{min(j, n-1)} p[j-m] * a[m]
    
    # Reverse p and use the same trick
    p_rev = p[::-1].copy()
    # grad2[j] = sum_{m=0}^{n-1} p[j-m]*a[m]
    # Let k = j-m, so m = j-k: sum_{k} p[k]*a[j-k] where 0<=j-k<n and k>=0
    # = (p * a)[j] in the convolution sense (cross-correlation)
    # This is the first n elements of convolve(p, a)
    
    # convolve(p, a)[j] = sum_k p[k]*a[j-k] for valid k, j-k
    # For j=0: p[0]*a[0]
    # For j=1: p[0]*a[1] + p[1]*a[0]
    # etc.
    
    # This is exactly what we want for grad2!
    # convolve(p, a) has length (2n-1) + n - 1 = 3n-2
    # We want the first n elements
    
    # Use FFT:
    size2 = 1
    while size2 < (2 * n - 1) + n - 1:
        size2 <<= 1
    P_fft = rfft(p, size2)
    A_fft = rfft(a, size2)
    conv_pa = irfft(P_fft * A_fft, size2)[:n]
    grad2 = conv_pa
    
    grad = grad1 + grad2
    return obj, grad


def smooth_gradient_descent(a, time_budget, seed=42, verbose=True, tau_init=50.0, tau_max=500.0):
    """
    Momentum-based gradient descent on the smoothed objective.
    Works in a-space (simplex constraint handled by projection).
    Periodically performs LP jumps.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    tau = tau_init
    momentum = 0.85
    step_size = 2.0
    velocity = np.zeros(n)
    
    iter_count = 0
    stall_count = 0
    prev_obj = best_obj
    best_iter = 0
    lp_interval = 200  # do LP every 200 iterations
    last_lp_iter = 0
    prev_g = None
    
    while time.monotonic() - start < time_budget:
        # Periodic LP jump
        if iter_count - last_lp_iter >= lp_interval and time.monotonic() - start < time_budget - 5:
            try:
                g_lp, M_lp = solve_lp(best_a, warm_start=prev_g)
                if g_lp is not None:
                    prev_g = g_lp
                    # Try line search
                    best_local = best_obj
                    best_local_a = best_a.copy()
                    for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1.0 - t) * best_a + t * g_lp
                        a_try = np.maximum(a_try, 0)
                        st = float(np.sum(a_try))
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        obj_try = compute_obj_norm(a_try)
                        if obj_try < best_local - 1e-14:
                            best_local = obj_try
                            best_local_a = a_try.copy()
                    if best_local < best_obj - 1e-13:
                        best_obj = best_local
                        best_a = best_local_a
                        a = best_a.copy()
                        velocity = np.zeros(n)
                        stall_count = 0
                last_lp_iter = iter_count
            except Exception:
                pass
        
        # Compute smoothed objective and gradient
        obj_smooth, grad = smoothed_obj_and_grad(a, tau)
        
        # Momentum update
        velocity = momentum * velocity - step_size * grad
        a_new = a + velocity
        
        # Project to simplex (clip and normalize)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))
        
        # Check improvement
        obj_new = compute_obj_norm(a_new)
        
        if obj_new < best_obj - 1e-12:
            a = a_new
            best_obj = obj_new
            best_a = a_new.copy()
            best_iter = iter_count
            stall_count = 0
            # Slightly increase step size if improving
            step_size = min(step_size * 1.02, 10.0)
            # Increase tau gradually
            tau = min(tau * 1.005, tau_max)
        else:
            stall_count += 1
            step_size *= 0.9
            # If stalled a lot, restart velocity and try different tau
            if stall_count > 50:
                velocity = np.zeros(n)
                tau = max(tau * 0.5, tau_init)
                step_size = 2.0
                # Perturb
                strength = 0.003 * min(stall_count // 10, 5)
                pert = rng.standard_normal(n) * strength
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 1e-15)
                a = a / float(np.sum(a))
                stall_count = 0
            elif stall_count > 20:
                # Reduce tau to smooth more
                tau = max(tau * 0.8, tau_init * 0.5)
        
        iter_count += 1
        
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD iter {iter_count}: obj={best_obj:.10f}, tau={tau:.1f}, step={step_size:.3f}, stall={stall_count}, time={elapsed:.0f}s")
        
        if stall_count > 200:
            break
    
    if verbose:
        print(f"    SGD done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with adaptive step size and warm starts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        t0 = time.monotonic()
        try:
            g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        except Exception:
            break
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        best_local_obj = best_obj
        best_local_a = a.copy()
        best_t = 0.0

        t_values = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 6:
                strength = 0.004 + 0.002 * min(stall, 10)
                pert = rng.standard_normal(n) * strength
                a = best_a + pert * best_a
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """
    Vectorized coordinate descent on normalized sequence (sum=1).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 5
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            candidates = np.array([0.0, mean_val * 0.1, mean_val * 0.3, mean_val * 0.5,
                                    mean_val * 0.7, mean_val, mean_val * 1.3, mean_val * 1.7,
                                    mean_val * 2.5, mean_val * 4, mean_val * 7])
            if a_j > 1e-12:
                candidates = np.concatenate([candidates, [a_j * 0.3, a_j * 0.5, a_j * 0.7,
                                                          a_j * 1.1, a_j * 1.4, a_j * 2.0]])
            candidates = np.unique(candidates[(candidates >= 0) & (candidates <= 0.3)])

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue

                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                affected = conv[k_start:k_end + 1].copy()
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_unscaled = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_unscaled / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 12):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_structured_init(n, rng):
    """Create a random structured initial sequence."""
    # Mix of uniform, periodic, and random components
    base = np.ones(n) / n
    # Add periodic component
    period = rng.integers(max(2, n // 20), max(3, n // 5))
    periodic = np.zeros(n)
    for j in range(n):
        periodic[j] = 0.5 + 0.5 * np.cos(2 * np.pi * j / period)
    periodic = periodic / periodic.sum() * n
    
    # Add noise
    noise = rng.standard_normal(n) * 0.3
    
    alpha = rng.uniform(0.3, 0.7)
    beta = rng.uniform(0.2, 0.5)
    a = (1 - alpha - beta) * base + alpha * periodic / n + beta * noise
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
