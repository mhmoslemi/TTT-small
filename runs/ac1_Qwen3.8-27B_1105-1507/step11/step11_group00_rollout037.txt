```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def aspg_optimize(a, deadline, rng):

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    tau = 1.0 / (10 * n)  # Initial smoothing parameter
    momentum = 0.9
    v = np.zeros(n)
    
    it = 0
    while time.monotonic() < deadline:
        # Compute autocorrelation
        conv = compute_conv(a)
        
        # Choose active set: top-k lags with exponential weighting
        # Use temperature-based smoothing
        max_conv = conv.max()
        # Weight: exp((conv - max_conv) / tau_scale)
        tau_scale = max_conv * 0.1 + 1e-10
        weights = np.exp((conv - max_conv) / tau_scale)
        # Only keep top-k
        k = min(n, max(10, int(0.1 * n)))
        top_idx = np.argsort(conv)[-k:]
        w_full = np.zeros(2*n-1)
        w_full[top_idx] = weights[top_idx]
        w_full = w_full / w_full.sum()
        
        # Compute gradient via FFT
        grad = compute_guided_gradient(a, w_full)
        
        # Nesterov momentum
        v = momentum * v - grad
        a_candidate = a + v
        
        # Project to simplex
        a_candidate = project_simplex(a_candidate)
        
        # Backtracking line search

            best_a = a_candidate.copy()
        
        a = a_candidate
        it += 1
        
        if it % 100 == 0:
```

```python
"""
Active-Set Projected Gradient (ASPG) with FFT-based gradient for minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. Exact FFT-based gradient for the active-set QP (O(n log n) per iteration)
2. Projected gradient with Nesterov momentum and backtracking line search
3. Adaptive active set with temperature-based smoothing
4. Shape-aware initialization (U-shape, raised cosine, parent)
5. Multi-scale exploration with diverse restarts
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_guided_gradient(a, w_full):
    """
    Compute gradient of sum_k w_full[k]*conv(a,a)[k] wrt a.
    Uses FFT cross-correlation: grad[j] = 2 * sum_m w_full[j+m]*a[m].
    w_full has length 2n-1.
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(w_full, N)
    A = fft(a, N)
    cross = ifft(W * np.conj(A), N).real[:n]
    return 2.0 * cross


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s > 0:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def build_weights_topk(conv, k, alpha=1.0):
    """Build weight vector from top-k lags with power weighting."""
    L = len(conv)
    top_idx = np.argsort(conv)[-k:]
    w_full = np.zeros(L)
    vals = conv[top_idx]
    if vals.max() > 0:
        w_full[top_idx] = np.power(vals / vals.max(), alpha)
        w_full = w_full / w_full.sum()
    else:
        w_full[top_idx] = 1.0 / k
    return w_full


def build_weights_smoothed(conv, tau):
    """Build weight vector with exponential smoothing over all lags."""
    max_c = conv.max()
    if max_c < 1e-30:
        return np.ones(len(conv)) / len(conv)
    shifted = (conv - max_c) / tau
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        idx = np.argmax(conv)
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w = w / s
    return w


def aspg_core(a, time_budget, tau_frac=0.05, k_frac=0.1, alpha=1.5,
              momentum=0.85, lr_init=0.1, verbose=False, verbose_interval=50):
    """
    Active-Set Projected Gradient core optimizer.
    Minimizes 2n*max(conv(a,a))/(sum(a))^2 via iterative active-set QP.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum_buf = np.zeros(n)
    lr = lr_init
    it = 0
    stall = 0
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.1:
            break

        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))

        # Compute autocorrelation
        conv = compute_conv(a)

        # Build weights: blend top-k with smoothed
        k = max(5, int(k_frac * n))
        tau = tau_frac * conv.max() * (1.0 + 2.0 * progress)

        # Blend: early on use more top-k, later use more smoothed
        blend = 0.3 + 0.5 * progress
        w_topk = build_weights_topk(conv, k, alpha=alpha)
        w_smooth = build_weights_smoothed(conv, tau)
        w_full = blend * w_topk + (1.0 - blend) * w_smooth

        # Compute gradient via FFT
        grad = compute_guided_gradient(a, w_full)

        # Nesterov momentum
        momentum_buf = momentum * momentum_buf - lr * grad
        a_candidate = a + momentum_buf

        # Project to simplex
        a_candidate = project_simplex(a_candidate)

        # Evaluate and backtracking
        obj_candidate = compute_obj(a_candidate)

        if obj_candidate > best_obj:
            # Backtracking: reduce step
            for bt in range(5):
                lr_try = lr * (0.5 ** (bt + 1))
                momentum_buf_bt = momentum * momentum_buf - lr_try * grad
                a_bt = project_simplex(a + momentum_buf_bt)
                obj_bt = compute_obj(a_bt)
                if obj_bt < best_obj:
                    a_candidate = a_bt
                    obj_candidate = obj_bt
                    momentum_buf = momentum_buf_bt
                    lr = lr_try
                    break
            else:
                # All backtracking failed, try direct step
                lr = lr * 0.5
                a_candidate = project_simplex(a - lr * grad)
                obj_candidate = compute_obj(a_candidate)

        if obj_candidate < best_obj - 1e-14:
            best_obj = obj_candidate
            best_a = a_candidate.copy()
            no_improve_count = 0
            # Increase LR slightly on success
            lr = min(lr * 1.05, 1.0)
        else:
            no_improve_count += 1
            lr = lr * 0.95
            if no_improve_count > 100:
                # Reset momentum and try perturbation
                momentum_buf = np.zeros(n)
                lr = lr_init * 0.5
                noise = 0.01 * np.random.default_rng(it).standard_normal(n)
                a = project_simplex(a + noise)
                no_improve_count = 0

        a = a_candidate
        it += 1

        if verbose and (it % verbose_interval == 0 or it == 1):
            print(f"    ASPG it={it}, obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"tau_frac={tau_frac*(1+2*progress):.4f}, time={time.monotonic()-start:.1f}s")

    return best_a


def nesterov_fft_optimize(a, time_budget, tau_init=0.05, tau_end=0.5,
                          momentum=0.9, verbose=False, verbose_interval=100):
    """
    Nesterov accelerated gradient on smoothed max using FFT gradients.
    Uses LSE approximation of max for smoothness.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    lr = 0.1 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.1:
            break

        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv_y = compute_conv(y)
        max_c = conv_y.max()
        tau_scale = tau * max_c + 1e-10
        w_full = np.exp((conv_y - max_c) / tau_scale)
        w_full = w_full / w_full.sum()

        grad = compute_guided_gradient(y, w_full)

        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            lr_adapt = 0.1 / (grad_norm + 1e-10)
            lr = 0.8 * lr + 0.2 * lr_adapt
            lr = min(lr, 1.0 / max(grad_norm, 1e-10))

        # Update
        v = momentum * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and (it % verbose_interval == 0):
            print(f"    NFFT it={it}, obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"tau={tau:.4f}, time={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent_polish(a, time_budget, verbose=False):

    This is piecewise linear in a[i], so we can find the optimal a[i] efficiently.
    """

    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    passes = 0
    while time.monotonic() - start < time_budget:
        # Shuffle order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget - 0.5:
                break
            # Try perturbing a[i]
            best_local = a[i]
            best_local_obj = best_obj
            # Try several values
            for amt in [0.5, 2.0, 0.1, 0.05]:
                for sign in [-1, 1]:
                    a_try = a.copy()
                    a_try[i] = max(0, a[i] + sign * amt / n)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_local_obj:
                        best_local_obj = obj_try
                        best_local = a_try[i]
            a[i] = best_local

        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
            a = best_a.copy()

        passes += 1
        if verbose and passes % 5 == 0:
            print(f"    CD pass={passes}, obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_u_shape(n, eps=0.3):
    """Create a U-shaped initialization (large at edges, small at center)."""
    j = np.arange(n)
    t = 1.0 - 2.0 * np.minimum(j, n - 1 - j) / max(n - 1, 1)
    a = 1.0 + eps * t
    return np.maximum(a, 0.01)


def make_raised_cosine(n, eps=0.3):
    """Raised cosine shape."""
    j = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * j / n)
    return np.maximum(a, 0.01)


def make_v_shape(n, eps=0.3):
    """V-shape (small at edges, large at center) - opposite of U."""
    j = np.arange(n)
    t = 2.0 * np.minimum(j, n - 1 - j) / max(n - 1, 1) - 1.0
    a = 1.0 + eps * t
    return np.maximum(a, 0.01)


def symmetric_mirror(a):
    """Average with reverse to enforce approximate symmetry."""
    return (a + a[::-1]) / 2.0


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: ASPG + Nesterov FFT + coordinate descent + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)
    best_a = None
    best_score = np.inf

    # === Initialization ===
    inits = {}

    # From parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(parent_arr.sum())
            if s > 0.01:
                inits['parent'] = parent_arr / s
                best_score = compute_obj(inits['parent'])
                best_a = inits['parent'].copy()
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    # Shape-based inits
    n_base = len(best_a) if best_a is not None else 800
    inits['u_shape'] = make_u_shape(n_base, 0.4) / make_u_shape(n_base, 0.4).sum()
    inits['v_shape'] = make_v_shape(n_base, 0.4) / make_v_shape(n_base, 0.4).sum()
    inits['raised_cos'] = make_raised_cosine(n_base, 0.4) / make_raised_cosine(n_base, 0.4).sum()
    inits['sym_parent'] = symmetric_mirror(inits.get('parent', np.ones(n_base)))
    inits['sym_parent'] = inits['sym_parent'] / inits['sym_parent'].sum()

    # Evaluate all inits
    for name, a_init in inits.items():
        obj = compute_obj(a_init)
        print(f"  Init '{name}': obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_a = a_init.copy()

    # === Phase 1: ASPG on best init (40% budget) ===
    print(f"\n=== Phase 1: ASPG (n={len(best_a)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.4, 500.0)
    if p1_budget > 30:
        a_p1 = aspg_core(best_a, time_budget=p1_budget,
                         tau_frac=0.05, k_frac=0.1, alpha=1.5,
                         momentum=0.85, lr_init=0.1,
                         verbose=True, verbose_interval=100)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_a = a_p1.copy()
            print(f"  ASPG: obj={best_score:.8f} **")
        else:
            print(f"  ASPG: obj={obj_p1:.8f} (no improvement)")

    # === Phase 2: Nesterov FFT (20% budget) ===
    print(f"\n=== Phase 2: Nesterov FFT (n={len(best_a)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 20:
        a_p2 = nesterov_fft_optimize(best_a, time_budget=p2_budget,
                                     tau_init=0.05, tau_end=0.8,
                                     momentum=0.9, verbose=True, verbose_interval=200)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_a = a_p2.copy()
            print(f"  NFFT: obj={best_score:.8f} **")
        else:
            print(f"  NFFT: obj={obj_p2:.8f} (no improvement)")

    # === Phase 3: Second ASPG with different params (15% budget) ===
    print(f"\n=== Phase 3: ASPG v2 (different params) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 180.0)
    if p3_budget > 20:
        # Use different hyperparameters
        a_p3 = aspg_core(best_a, time_budget=p3_budget,
                         tau_frac=0.1, k_frac=0.05, alpha=2.0,
                         momentum=0.8, lr_init=0.05,
                         verbose=True, verbose_interval=100)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_score:
            best_score = obj_p3
            best_a = a_p3.copy()
            print(f"  ASPG2: obj={best_score:.8f} **")
        else:
            print(f"  ASPG2: obj={obj_p3:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration (15% budget) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    n_options = []
    for delta in [-100, -200, 100, 200, 400]:
        n_try = n_best + delta
        if 400 <= n_try <= 3000:
            n_options.append(n_try)
    for n_abs in [600, 1000, 1200, 1500, 2000]:
        if n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))[:4]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_a, n_try)

        # Quick ASPG
        alloc = min(remaining * 0.25, 60.0)
        if alloc > 10:
            a_try = aspg_core(a_try, time_budget=alloc,
                              tau_frac=0.05, k_frac=0.1, alpha=1.5,
                              momentum=0.85, lr_init=0.1, verbose=False)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_a = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts (15% budget) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_a)

    restart_strategies = [
        ('u_blend', lambda: best_a * 0.5 + make_u_shape(n_cur, 0.5) * 0.5),
        ('v_blend', lambda: best_a * 0.5 + make_v_shape(n_cur, 0.5) * 0.5),
        ('fourier', lambda: best_a * np.exp(0.05 * np.sin(2 * np.pi * np.arange(n_cur) / n_cur * rng.integers(1, 5)) + rng.uniform(0, 2 * np.pi))),
        ('noise', lambda: best_a + 0.02 * rng.standard_normal(n_cur)),
    ]

    for name, gen_fn in restart_strategies:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        try:
            a_r = np.maximum(gen_fn(), 0)
            s_r = a_r.sum()
            if s_r < 1e-10:
                continue
            a_r = a_r / s_r
        except Exception:
            continue

        alloc_md = min(remaining * 0.3, 50.0)
        if alloc_md > 10:
            a_r = aspg_core(a_r, time_budget=alloc_md,
                            tau_frac=0.05, k_frac=0.1, alpha=1.5,
                            momentum=0.85, lr_init=0.1, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_nf = min(remaining * 0.15, 30.0)
        if alloc_nf > 8:
            a_r = nesterov_fft_optimize(a_r, time_budget=alloc_nf,
                                        tau_init=0.05, tau_end=0.5, momentum=0.9)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_a = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 6: Coordinate descent polish (10% budget) ===
    print(f"\n=== Phase 6: Coordinate Descent Polish ===")
    remaining = deadline - time.monotonic()
    cd_budget = min(remaining * 0.1, 120.0)
    if cd_budget > 15:
        # Only do CD if n is small enough
        if len(best_a) <= 1500:
            a_cd = coordinate_descent_polish(best_a, time_budget=cd_budget, verbose=False)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_a = a_cd.copy()
                print(f"  CD: obj={best_score:.8f} **")
            else:
                print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final ASPG with very fine parameters
    if remaining > 40:
        alloc = min(remaining * 0.5, 150.0)
        a_final = aspg_core(best_a, time_budget=alloc,
                            tau_frac=0.02, k_frac=0.05, alpha=3.0,
                            momentum=0.9, lr_init=0.03,
                            verbose=True, verbose_interval=200)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_a = a_final.copy()
            print(f"  Final ASPG: obj={best_score:.8f} **")

    # Final Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_final2 = nesterov_fft_optimize(best_a, time_budget=alloc,
                                         tau_init=0.02, tau_end=0.3, momentum=0.92)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_a = a_final2.copy()
            print(f"  Final NFFT: obj={best_score:.8f} **")

    # === Validation ===
    if best_a is None:
        best_a = np.ones(200)
        best_score = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
