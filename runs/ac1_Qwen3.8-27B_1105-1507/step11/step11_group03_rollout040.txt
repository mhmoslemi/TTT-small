      - `d(c[k])/d(a[i]) = a[k-i]` (for valid k-i)
      - So to reduce `c[k]`, we should decrease `a[i]` in proportion to `a[k-i]`
      - This gives a specific direction to move in

   4. **Multiple worst peaks**: If there are multiple near-maximal peaks, we can average their gradients to get a direction that reduces all of them simultaneously.

import numpy as np

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve(a, solver_list=None):
    n = len(a)
    # Build constraint matrix A where (A @ g)[k] = conv(a, g)[k]
    # conv(a, g)[k] = sum_i a[i] * g[k-i] for valid indices
    # This is a Toeplitz-like structure
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in conv[k] for k = j, j+1, ..., j+n-1
        # conv[j+i] += a[i] * g[j] for i = 0, ..., n-1
        # So A[j+i, j] = a[i] for i = 0, ..., n-1
        A[j:j+n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.array(g_var.value).flatten()
                g_val = np.maximum(g_val, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None

def lp_fixed_point_v2(a, time_budget, verbose=False, seed=42, max_iters=1000):
    """
    LP fixed-point with parabolic line search and smarter perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t0 = time.monotonic()
        g_lp, M_lp = lp_solve(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            stall_count += 2
            if stall_count > 20:
                break
            continue

        # Direction: g_lp - a (normalized to sum zero for the step)
        direction = g_lp - a
        # Adjust so that a + t*direction maintains sum = 1
        # (since sum(g_lp) = 1 and sum(a) = 1, sum(direction) = 0, so it's fine)

        # Line search
        best_t = 0.0
        best_f = best_obj
        t_grid = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for t in t_grid:
            a_try = (1.0 - t) * a + t * g_lp

            st = float(np.sum(a_try))
            if st < 1e-10:
                continue

            f = compute_obj(a_try)
            if f < best_f:
                best_f = f
                best_t = t

        if best_t > 0 and best_f < best_obj - 1e-12:
            best_obj = best_f
            best_a = ((1.0 - best_t) * a + best_t * g_lp)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / best_a.sum()
            a = best_a
            stall_count = 0
            if verbose and it % 20 == 0:
                print(f"    LP2 it{it:3d}: obj={best_obj:.10f}, t={best_t:.3f}, M_lp={M_lp:.6f}")
        else:
            stall_count += 1

            strategy = stall_count % 5

                conv = compute_conv(a)

                threshold = max_val * 0.995
                max_lags = np.where(conv >= threshold)[0]

            elif strategy == 2:

                a = a + 0.01 * rng.standard_normal(n)

            a = np.maximum(a, 0)

            if sa > 1e-10:
                a = a / sa
            else:

            if stall_count > 50:

                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:

def freq_domain_raking(a, time_budget, verbose=False, seed=42, max_iters=500):
    """
    Frequency-domain error reduction (Gerchberg-Saxton style).
    Alternates between:
    1. Time domain: project onto non-negative, sum=1
    2. Frequency domain: reduce peaks in the convolution
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    damping = 0.3  # Start gentle

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        # Frequency domain step: modify A to reduce max of IFFT(A^2)
        A = fft(a, N)
        C = A * A  # This gives the DFT of the circular convolution
        c_circ = ifft(C).real  # Circular convolution (length N)
        # The linear convolution is c_circ[:2n-1] (if N >= 2n-1)
        c_lin = c_circ[:2 * n - 1]
        max_c = float(np.max(c_lin))

        # Find where c exceeds a threshold
        threshold = max_c * 0.95
        # In frequency domain, we want to reduce the energy at frequencies that
        # contribute to the peaks
        
        # Simple approach: reduce |A| at frequencies where |C| is large
        # and corresponds to peak regions
        C_abs = np.abs(C)
        C_mean = np.mean(C_abs)
        # Scale down the largest frequency components
        mask = C_abs > C_mean * 1.5
        A_new = A.copy()
        A_new[mask] *= (1.0 - damping)
        # Also boost small ones
        mask_small = C_abs < C_mean * 0.5
        A_new[mask_small] *= (1.0 + damping * 0.5)

        a_new = ifft(A_new).real[:n]

        # Time domain projection
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            continue
        a_new = a_new / sn

        # Blend
        blend = 0.3 + 0.5 * (it / max_iters)  # Gradually increase blending
        a_try = (1 - blend) * a + blend * a_new
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            continue
        a_try = a_try / st

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            # Increase damping as we make progress
            damping = min(0.8, damping + 0.005)
            if verbose and it % 50 == 0:
                print(f"    Freq it{it:3d}: obj={best_obj:.10f}, damping={damping:.3f}")
        else:
            # Try smaller step
            a_try2 = 0.95 * a + 0.05 * a_new
            a_try2 = np.maximum(a_try2, 0)
            a_try2 = a_try2 / a_try2.sum()
            obj_try2 = compute_obj(a_try2)
            if obj_try2 < best_obj - 1e-12:
                best_obj = obj_try2
                best_a = a_try2.copy()
                a = a_try2
            else:
                # Random restart in frequency domain
                A2 = fft(a, N)
                noise = 0.02 * rng.standard_normal(N)
                A2 = A2 + noise * np.abs(A2)
                a = ifft(A2).real[:n]
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa

    return best_a


def anti_peak_refine(a, time_budget, verbose=False, seed=42, max_iters=500):
    """
    Targeted refinement: specifically reduce the worst convolution peaks.
    Uses the gradient of each peak to determine which a[i] to reduce.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Find all near-max peaks
        threshold = max_val * (1.0 - 0.001)  # Within 0.1% of max
        max_lags = np.where(conv >= threshold)[0]

        if len(max_lags) == 0:
            max_lags = np.array([int(np.argmax(conv))])

        # Compute the "anti-peak" direction: decrease a[i] proportionally to
        # the contribution of a[i] to the worst peaks
        grad = np.zeros(n)
        for lag in max_lags[:10]:  # Limit to 10 worst
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    grad[i] += a[i] * a[j]  # Contribution to c[lag]
        
        # Normalize
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        # Try multiple step sizes
        improved = False
        for step in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Alternate: try increasing some elements (to maintain sum)
            # The anti-peak direction decreases elements that contribute to peaks
            # We should also try increasing elements that DON'T contribute to peaks
            contributions = np.zeros(n)
            for lag in max_lags[:10]:
                for i in range(n):
                    j = lag - i
                    if 0 <= j < n:
                        contributions[i] += a[i] * a[j]
            
            # Find low-contribution indices
            low_contrib = contributions < np.mean(contributions) * 0.5
            if np.any(low_contrib):
                # Increase low-contribution elements
                increase_dir = np.zeros(n)
                increase_dir[low_contrib] = 1.0
                increase_dir = increase_dir - np.mean(increase_dir)  # Zero mean
                gn2 = np.linalg.norm(increase_dir)
                if gn2 > 1e-15:
                    for step in [0.002, 0.005, 0.01]:
                        a_try = a + step * increase_dir / gn2
                        a_try = np.maximum(a_try, 0)
                        st = float(np.sum(a_try))
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break

            if not improved:
                # Random perturbation
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                else:
                    a = best_a.copy()

    return best_a


def gcd_smoothed(a, time_budget, verbose=False, seed=42, max_iters=10000):
    """
    Gradient descent with log-sum-exp smoothing of the max.
    Uses Nesterov acceleration and adaptive tau.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Start with moderate tau and decrease
    taus = [50, 20, 5, 2, 1, 0.5, 0.2, 0.1, 0.05]
    beta = 0.9
    N = 1
    while N < 2 * n + n:
        N <<= 1

    for tau in taus:
        if time.monotonic() - start >= time_budget:
            break
        lr = 2.0 / np.sqrt(n)
        v = np.zeros(n)
        tau_start = time.monotonic()
        max_tau_time = time_budget * 0.08  # 8% of budget per tau level

        while time.monotonic() - tau_start < max_tau_time:
            if time.monotonic() - start >= time_budget:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute smoothed gradient of max(conv(y,y))
            conv = compute_conv(y)
            exact_max = float(np.max(conv))
            shifted = tau * (conv - exact_max)
            shifted = np.clip(shifted, -500, 500)
            exp_vals = np.exp(shifted)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            # Gradient of sum p_k * c_k with respect to a
            # c_k = sum_i a_i * a[k-i]
            # d(c_k)/d(a_i) = a[k-i] (for valid k-i)
            # So grad_obj[a_i] = sum_k p_k * a[k-i]
            # This is a cross-correlation of p with a
            P = fft(p, N)
            A = fft(y, N)
            # We need sum_k p_k * a[k-i], which is (p cross-correlate a)[i]
            # = (reverse(p) * a) in convolution = ifft(conj(FFT(p)) * FFT(a))
            cross = ifft(np.conj(P) * A).real[:n]
            g = 2.0 * cross  # Factor of 2 because c_k = sum a_i a_{k-i} and both terms depend on a_i

            # Project gradient onto the hyperplane sum = 0
            g = g - np.mean(g)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            lr = max(1e-8, lr * 0.9999)

        if verbose:
            print(f"    GD tau={tau:6.2f}: obj={best_obj:.10f}")

    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n <= 1:
        return np.repeat(a, new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point with line search, frequency-domain error reduction,
anti-peak refinement, and smoothed gradient descent with Nesterov acceleration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve(a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.array(g_var.value).flatten().astype(np.float64)
                g_val = np.maximum(g_val, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None

    beta = 0.9
    N = 1
    while N < 3 * n:
        N <<= 1

    for tau in tau_list:
        if time.monotonic() - start >= time_budget:
            break
        lr = 2.0 / np.sqrt(n)
        v = np.zeros(n)
        tau_start = time.monotonic()
        max_tau_time = min(time_budget * 0.1, (time_budget - (time.monotonic() - start)) / max(1, len(tau_list)))

        while time.monotonic() - tau_start < max_tau_time:
            if time.monotonic() - start >= time_budget:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute smoothed gradient
            conv = compute_conv(y)
            exact_max = float(np.max(conv))
            shifted = tau * (conv - exact_max)
            shifted = np.clip(shifted, -500, 500)
            exp_vals = np.exp(shifted)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[int(np.argmax(conv))] = 1.0
            else:
                p = exp_vals / sum_exp

            # Gradient: 2 * cross_correlation(p, y)
            P = fft(p, N)
            A = fft(y, N)
            cross = ifft(np.conj(P) * A).real[:n]
            g = 2.0 * cross
            g = g - np.mean(g)  # Project onto sum=0 hyperplane
            gnorm = float(np.linalg.norm(g))
            if gnorm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            lr = max(1e-8, lr * 0.99995)

        if verbose:
            print(f"    GD tau={tau:6.2f}: obj={best_obj:.10f}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n <= 1:
        return np.full(new_n, float(a[0]) if n == 1 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s
