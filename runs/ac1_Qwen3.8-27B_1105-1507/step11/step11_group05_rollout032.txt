import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_conv(a_norm):
    """Compute 2n*max(conv) for already-normalized a (sum=1)."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    return proj


def lse_smoothed_obj_and_grad(a, T):
    """
    Compute LSE-smoothed objective and gradient.
    Objective: 2n * LSE_T(conv(a,a)) where sum(a)=1.
    Returns (objective, gradient) for the USEF (unconstrained on simplex) problem.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    # Numerical stability
    shifted = T * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    lse_val = cmax + np.log(np.sum(exp_vals)) / T
    obj = 2.0 * n * lse_val
    
    # Gradient: d/d a[i] of 2n * LSE
    # = 2n * sum_k w[k] * d(conv[k])/d a[i]
    # d(conv[k])/d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    grad = np.zeros(n)
    for i in range(n):
        # Term 1: sum over k where 0 <= k-i < n, i.e., k in [i, i+n-1]
        lo = max(0, i)
        hi = min(2*n-2, i+n-1)
        if lo <= hi:
            grad[i] += np.dot(w[lo:hi+1], a[lo-i:hi-i+1])
        # Term 2: sum over k where 0 <= i-k < n, i.e., k in [i-n+1, i]
        lo2 = max(0, i-n+1)
        hi2 = min(2*n-2, i)
        if lo2 <= hi2:
            grad[i] += np.dot(w[lo2:hi2+1], a[i-hi2:i-lo2+1][::-1])
    
    grad *= 2.0 * n
    return obj, grad


def lse_scipy_obj(x, T, n):
    """Wrapper for scipy minimize. x is flat, we project to simplex."""
    a = np.maximum(x, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    a = a / s
    obj, grad = lse_smoothed_obj_and_grad(a, T)
    return obj, grad


def lse_lbfgs_optimize(a, T, time_budget, maxiter=200):
    """
    Optimize the LSE-smoothed objective using L-BFGS-B on the simplex.
    We work in the unconstrained space and project.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    
    # Use the LSE objective as a proxy, minimize over the simplex
    # Strategy: minimize LSE in the space of a >= 0, sum(a) = 1
    # We parameterize as a = exp(t) / sum(exp(t)) (softmax) to handle constraints
    
    # Actually, let's just use the projected gradient approach with L-BFGS
    # by working in unconstrained space and projecting
    
    # Simpler approach: just iterate projected LSE gradient with adaptive step
    a_curr = a.copy()
    lr = 1.0 / (T * n)  # Initial learning rate
    
    for it in range(maxiter):
        if time.monotonic() - start > time_budget:
            break
        
        obj, grad = lse_smoothed_obj_and_grad(a_curr, T)
        
        # Line search
        a_old = a_curr.copy()
        t = lr
        improved = False
        for _ in range(20):
            a_try = project_simplex(a_old - t * grad)
            obj_try, _ = lse_smoothed_obj_and_grad(a_try, T)
            if obj_try < obj - 1e-15:
                a_curr = a_try
                improved = True
                break
            t *= 0.5
        if not improved:
            break
        
        obj_true = compute_obj_conv(a_curr)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a_curr.copy()
        
        lr = t * 1.5
    
    return best_a


def full_lp_solve(a, solver_list=None):
    """
    Solve the full LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses all 2n-1 convolution constraints.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_len = 2 * n - 1
    
    # Build the constraint matrix A (conv_len x n)
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid j, k-j
    # Coefficient of g[m] in (a*g)[k] is a[k-m] if 0 <= k-m < n
    A = np.zeros((conv_len, n))
    for m in range(n):
        # g[m] appears in (a*g)[k] when k-m is in [0, n-1], i.e., k in [m, m+n-1]
        lo = m
        hi = min(conv_len - 1, m + n - 1)
        for k in range(lo, hi + 1):
            A[k, m] = a_norm[k - m]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, np.inf


def sparse_lp_solve(a, top_m, solver_list=None):
    """
    Solve LP with only top_m most active constraints.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    # Select top_m peaks
    top_indices = np.argsort(conv)[::-1][:top_m]
    conv_len = 2 * n - 1
    
    A = np.zeros((len(top_indices), n))
    for idx, k in enumerate(top_indices):
        k = int(k)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[idx, m] = a_norm[j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def frank_wolfe_lp(a, time_budget, mode='full', top_m=50, verbose=False):
    """
    Frank-Wolfe style alternating with LP.
    mode='full': use all constraints
    mode='sparse': use top_m constraints
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        if mode == 'full':
            g_lp, lp_val = full_lp_solve(best_a)
            if g_lp is None:
                break
        else:
            g_lp = sparse_lp_solve(best_a, top_m)
            if g_lp is None:
                break
        
        # Find best blend
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj_conv(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t
        
        # Refine t
        if best_t > 0:
            for dt in [-0.1, -0.05, 0.05, 0.1, 0.2]:
                t_try = best_t + dt
                if 0 < t_try < 1.5:
                    a_try = (1 - t_try) * best_a + t_try * g_lp
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s < 0.01:
                        continue
                    a_try = a_try / s
                    obj_try = compute_obj_conv(a_try)
                    if obj_try < best_blend_obj:
                        best_blend_obj = obj_try
                        best_t = t_try
        
        if best_t > 0 and best_blend_obj < best_obj - 1e-14:
            a_new = (1 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            a_new = a_new / s
            best_a = a_new
            best_obj = best_blend_obj
            stall = 0
        else:
            stall += 1
        
        it += 1
        if verbose and it % 3 == 0:
            print(f"    FW-LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 5:
            # Perturb
            rng_local = np.random.default_rng(it * 31 + 7)
            noise = rng_local.normal(0, 0.003, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj_conv(best_a)
            stall = 0
    
    return best_a

Actually, for the L-BFGS-B approach, I need to be careful. The LSE-smoothed objective is:
`F(a) = 2n * LSE_T(conv(a,a))`

where `a` is on the simplex. To use L-BFGS-B, I can parameterize `a = softmax(t)` for some unconstrained `t`, and minimize `F(softmax(t))`.

The gradient with respect to `t` is:
`dF/dt[i] = sum_j (dF/da[j]) * (da[j]/dt[i])`
`da[j]/dt[i] = a[j] * (delta_{ij} - a[i])`

So `dF/dt[i] = a[i] * (dF/da[i] - sum_j a[j] * dF/da[j])`

This is the standard softmax gradient chain rule.

However, the LSE objective `F(a)` is only a smooth approximation to `2n * max(conv(a,a))`. As T increases, the approximation gets better but the optimization becomes harder (the smoothed function becomes more peaked).

Strategy: Start with small T (very smooth, easy to optimize), then gradually increase T while re-optimizing. This is like "simulated annealing" for the temperature.

def lse_anneal_optimize(a, time_budget, T_start=3, T_end=100, verbose=False):
    """
    LSE-annealed optimization using L-BFGS-B on softmax parameterization.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    
    # Current temperature
    T = T_start
    
    while time.monotonic() - start < time_budget:
        # Get current a in softmax parameterization
        # a = exp(t) / sum(exp(t)) => t = log(a) (roughly)
        # We'll optimize t directly
        
        def softmax(t):
            t_shifted = t - t.max()
            e = np.exp(t_shifted)
            return e / e.sum()
        
        def objective(t):
            a_t = softmax(t)
            return compute_obj_conv(a_t)
        
        def lse_objective(t):
            a_t = softmax(t)
            obj, grad_a = lse_smoothed_obj_and_grad(a_t, T)
            # Chain rule through softmax
            # dF/dt[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])
            mean_grad = np.dot(a_t, grad_a)
            grad_t = a_t * (grad_a - mean_grad)
            return obj, grad_t
        
        # Initial t from current a
        t_init = np.log(np.maximum(a, 1e-15))
        
        # L-BFGS-B
        result = minimize(lse_objective, t_init, method='L-BFGS-B',
                         jac=True, options={'maxiter': 50, 'maxfun': 200})
        
        t_opt = result.x
        a_new = softmax(t_opt)
        obj_new = compute_obj_conv(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        
        # Update temperature
        elapsed_frac = (time.monotonic() - start) / time_budget
        if elapsed_frac < 0.5:
            T = T_start * (T_end / T_start) ** (2 * elapsed_frac)

        if time.monotonic() - start > time_budget * 0.8:
            # Final iterations at high T
            T = T_end
    
    return best_a

"""
Optimized step function coefficient search using:
1. Asymmetric parametric + spectral-motivated initialization
2. LSE-smoothed L-BFGS-B optimization with temperature annealing
3. Full-constraint LP Frank-Wolfe alternating (sparse matrix)
4. Adaptive LP/LSE interleaving with multi-direction exploration
5. Coordinate descent polish
6. Multi-n exploration with interpolation and re-optimization

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_conv(a_norm):
    """Compute 2n*max(conv) for normalized a (sum=1)."""
    n = len(a_norm)
    if n == 0:
        return np.inf
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    return proj


def softmax_fn(t):
    """Compute softmax of vector t."""
    t_shifted = t - t.max()
    e = np.exp(t_shifted)
    return e / e.sum()


def lse_smoothed_obj_grad(a, T):
    """
    Compute LSE-smoothed objective and gradient w.r.t. a.
    Objective: 2n * LSE_T(conv(a*a)) for a with sum=1.
    Returns (obj, grad).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = T * (conv - cmax)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()

    lse_val = cmax + np.log(exp_vals.sum()) / T
    obj = 2.0 * n * lse_val

    # Gradient: d obj / d a[i] = 2n * sum_k w[k] * (a[k-i] + a[i-k]) [with bounds]
    grad = np.zeros(n)
    for i in range(n):
        # Term 1: k in [i, i+n-1] intersect [0, 2n-2]
        lo1 = i
        hi1 = min(2 * n - 2, i + n - 1)
        if lo1 <= hi1:
            grad[i] += np.dot(w[lo1:hi1 + 1], a[lo1 - i:hi1 - i + 1])
        # Term 2: k in [i-n+1, i] intersect [0, 2n-2]
        lo2 = max(0, i - n + 1)
        hi2 = i
        if lo2 <= hi2:
            grad[i] += np.dot(w[lo2:hi2 + 1], a[i - hi2:i - lo2 + 1][::-1])

    grad *= 2.0 * n
    return obj, grad


def lse_lbfgs_round(a, T, maxiter=100, time_limit=30.0, verbose=False):
    """
    One round of L-BFGS-B optimization on the LSE-smoothed objective.
    Uses softmax parameterization to stay on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj_conv(a)

    # Initial t from a
    t_init = np.log(np.maximum(a, 1e-15))

    def fun_and_jac(t):
        a_t = softmax_fn(t)
        obj, grad_a = lse_smoothed_obj_grad(a_t, T)
        # Chain rule: dF/dt[i] = a[i] * (grad_a[i] - dot(a, grad_a))
        mean_g = np.dot(a_t, grad_a)
        grad_t = a_t * (grad_a - mean_g)
        return obj, grad_t

    try:
        result = scipy_minimize(fun_and_jac, t_init, method='L-BFGS-B',
                                jac=True, options={'maxiter': maxiter, 'maxfun': maxiter * 3})
        t_opt = result.x
        a_new = softmax_fn(t_opt)
        obj_new = compute_obj_conv(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    except Exception:
        pass

    if verbose:
        print(f"    LSE-LBFGS (T={T:.1f}): obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def lse_anneal_optimize(a, time_budget, T_start=3.0, T_end=150.0, rounds=8, verbose=False):
    """
    LSE-annealed optimization: multiple rounds with increasing T.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj_conv(a)

    for r in range(rounds):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Anneal T
        frac = r / max(1, rounds - 1)
        T = T_start * (T_end / T_start) ** frac

        alloc = min(remaining * 0.8 / (rounds - r), 30.0)
        if alloc < 2:
            break

        a_curr = lse_lbfgs_round(best_a, T, maxiter=50, time_limit=alloc, verbose=verbose)
        obj_curr = compute_obj_conv(a_curr)

        if obj_curr < best_obj:
            best_obj = obj_curr
            best_a = a_curr.copy()
            if verbose:
                print(f"    LSE-anneal round {r}: T={T:.1f}, obj={best_obj:.8f}, "
                      f"elapsed={time.monotonic()-start:.1f}s")

    return best_a


def build_full_lp_sparse(a_norm):
    """Build sparse constraint matrix for full LP: (a*g)[k] for all k."""
    n = len(a_norm)
    conv_len = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for m in range(n):
        lo = m
        hi = min(conv_len - 1, m + n - 1)
        for k in range(lo, hi + 1):
            rows[count] = k
            cols[count] = m
            vals[count] = a_norm[k - m]
            count += 1
    A = csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(conv_len, n))
    return A


def build_topm_lp_sparse(a_norm, top_m):
    """Build sparse constraint matrix for top-m LP."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    top_indices = np.argsort(conv)[::-1][:top_m]
    conv_len = 2 * n - 1

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_indices):
        k = int(k)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                rows.append(idx)
                cols.append(m)
                vals.append(a_norm[j])

    if len(rows) == 0:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n))
    return A


def full_lp_solve(a, solver_list=None):
    """Solve full LP: min M s.t. (a*g)[k] <= M all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_full_lp_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, np.inf


def topm_lp_solve(a, top_m, solver_list=None):
    """Solve LP with top_m constraints."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_topm_lp_sparse(a_norm, top_m)
    if A is None:
        return None

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def find_best_blend(a, direction, n_evals=15):
    """Find best blend factor t in (1-t)*a + t*direction."""
    best_t = 0.0
    best_obj = compute_obj_conv(a)

    # Coarse search
    for t in np.linspace(0.05, 1.0, n_evals):
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 0.01:
            continue
        a_try = a_try / s
        obj_try = compute_obj_conv(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    # Refine around best_t
    if best_t > 0:
        for dt in np.linspace(-0.15, 0.15, 7):
            t_try = best_t + dt
            if 0 < t_try < 1.5:
                a_try = (1 - t_try) * a + t_try * direction
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 0.01:
                    continue
                a_try = a_try / s
                obj_try = compute_obj_conv(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_t = t_try

    return best_t, best_obj


def frank_wolfe_lp(a, time_budget, mode='auto', top_m=40, verbose=False):
    """
    Frank-Wolfe alternating with LP.
    mode='full': all constraints
    mode='topm': top_m constraints
    mode='auto': try full first, fallback to topm
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    it = 0
    stall = 0
    use_full = (mode == 'full' or (mode == 'auto' and n <= 600))

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Solve LP
        g_lp = None
        if use_full and remaining > 10:
            g_lp, _ = full_lp_solve(best_a)
            if g_lp is None:
                use_full = False
        if g_lp is None:
            g_lp = topm_lp_solve(best_a, top_m)
            if g_lp is None:
                break

        # Find best blend
        t_opt, obj_opt = find_best_blend(best_a, g_lp)

        if t_opt > 0 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * best_a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            a_new = a_new / s
            best_a = a_new
            best_obj = obj_opt
            stall = 0
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"    FW-LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, "
                  f"mode={'full' if use_full else 'topm'}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 6:
            # Perturb to escape
            rng_local = np.random.default_rng(it * 7919 + 3)
            noise_scale = 0.002 * (1 + stall * 0.5)
            noise = rng_local.normal(0, noise_scale, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj_conv(best_a)
            stall = 0

    return best_a


def multi_direction_lp(a, time_budget, n_dirs=4, top_m=40, verbose=False):
    """
    Solve LP from multiple perturbed directions and pick best.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()

    rng = np.random.default_rng(12345)

    for d in range(n_dirs):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Create perturbed version
        if d == 0:
            a_pert = best_a.copy()
        else:
            noise = rng.normal(0, 0.005, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s > 0.01:
                a_pert = a_pert / s
            else:
                a_pert = best_a.copy()

        # Solve LP from perturbed point
        g_lp = topm_lp_solve(a_pert, top_m)
        if g_lp is None:
            continue

        # Try blending best_a with g_lp
        t_opt, obj_opt = find_best_blend(best_a, g_lp)
        if t_opt > 0 and obj_opt < best_obj:
            a_new = (1 - t_opt) * best_a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_a = a_new
            best_obj = obj_opt
            if verbose:
                print(f"    MultiDir {d}: obj={best_obj:.8f}")

    return best_a


def coordinate_descent(a, time_budget, max_sweeps=5, verbose=False):
    """
    Coordinate descent: for each coordinate, try several values and pick best.
    Uses incremental convolution update for speed.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break

            ai_old = a[i]
            if ai_old < 1e-12:
                continue

            # Try several multipliers
            candidates = [0.0, 0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0]
            best_val = ai_old
            best_local = best_obj

            for mult in candidates:
                a[i] = ai_old * mult
                s_new = a.sum()
                if s_new < 0.01:
                    a[i] = ai_old
                    continue
                a_try = a / s_new
                obj_try = compute_obj_conv(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = ai_old * mult

            if best_val != ai_old:
                a[i] = best_val
                s_new = a.sum()
                a = a / s_new
                sweep_improved = True

        if time.monotonic() - start >= time_budget:
            break

        obj_now = compute_obj_conv(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()

        if verbose:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")

        if not sweep_improved:
            break

    return best_a


def asymmetric_init(n, mode=0):
    """
    Generate asymmetric initializations motivated by spectral flatness.
    Key: asymmetric shapes can beat the symmetric lower bound of 2.
    """
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)

    if mode == 0:
        # Asymmetric power law: sharp on left, smooth on right
        a = (1 - x) ** 2.0 + 0.01
        a = a / (a + (x * 0.5) ** 1.5 + 0.01)
    elif mode == 1:
        # Skewed Gaussian
        center = 0.35
        width_l = 0.3
        width_r = 0.45
        a = np.where(x < center,
                     np.exp(-((x - center) ** 2) / (2 * width_l ** 2)),
                     np.exp(-((x - center) ** 2) / (2 * width_r ** 2))) + 0.01
    elif mode == 2:
        # Two asymmetric peaks
        a = (np.exp(-((x - 0.25) ** 2) / (2 * 0.15 ** 2)) * 1.0 +
             np.exp(-((x - 0.7) ** 2) / (2 * 0.2 ** 2)) * 0.6) + 0.01
    elif mode == 3:
        # Irational scaling: a[i] ~ 1/(i+1)^alpha on one side
        alpha = 0.7
        a = 1.0 / (idx + 1) ** alpha
        a = a + 0.01
    elif mode == 4:
        # Sinc-like (zero at edges)
        a = np.sin(np.pi * x) ** 1.5 + 0.05
    elif mode == 5:
        # Asymmetric raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.3))) * (1 - x * 0.5) + 0.02
    elif mode == 6:
        # Power with asymmetry
        a = x ** 0.6 * (1 - x) ** 1.2 + 0.01
    elif mode == 7:
        # Very skewed: heavy on one side
        a = (1 - x) ** 3.0 + 0.02
    elif mode == 8:
        # Modulated: decaying oscillation
        a = np.exp(-2 * x) * (1 + 0.3 * np.cos(4 * np.pi * x)) + 0.01
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def spectral_init(n):
    """
    Initialize using spectral reasoning: try to make |FFT(a)|^2 as flat as possible.
    This corresponds to a raised-cosine or Wagner window type shape.
    """
    x = np.linspace(0, 1, n)
    # Wagner window (optimal for maximizing flatness of power spectrum)
    # Approximation: product of sinc functions
    a = np.ones(n)
    for k in range(1, 6):
        a *= np.sinc((x - 0.5) * k)
    a = np.abs(a) + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s


def multi_scale_optimize(n_final, time_budget, rng, verbose=False):
    """
    Multi-scale: start with small n, optimize, interpolate to larger n.
    """
    n_small = max(50, n_final // 4)
    a_small = asymmetric_init(n_small, mode=1)
    a_small = lse_anneal_optimize(a_small, time_budget=min(20, time_budget * 0.1),
                                   T_start=3, T_end=50, rounds=3)
    obj_small = compute_obj_conv(a_small)

    # Interpolate to n_final
    x_old = np.linspace(0, 1, n_small)
    x_new = np.linspace(0, 1, n_final)
    a_full = np.interp(x_new, x_old, a_small)
    a_full = np.maximum(a_full, 0)
    s = a_full.sum()
    if s < 0.01:
        a_full = asymmetric_init(n_final, mode=1)
        s = a_full.sum()
    a_full = a_full / s

    if verbose:
        print(f"    Multi-scale: small_n={n_small}, obj_small={obj_small:.6f}, "
              f"interp_obj={compute_obj_conv(a_full):.6f}")

    return a_full


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0  # Safety margin
    rng = np.random.default_rng(seed)

    # === Initialization ===
    n_current = 800

    # Try to use parent
    best_seq = None
    best_score = np.inf
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj_conv(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("Could not load parent, using fresh initialization")

    # Try multiple initializations
    if best_seq is None or best_score > 1.51:
        inits = []
        for mode in range(9):
            a_init = asymmetric_init(n_current, mode=mode)
            inits.append((f"asym_{mode}", a_init))
        inits.append(("spectral", spectral_init(n_current)))
        inits.append(("uniform", np.ones(n_current) / n_current))

        for name, a_init in inits:
            obj_init = compute_obj_conv(a_init)
            if best_seq is None or obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
                if verbose or obj_init < 1.8:
                    print(f"  Init '{name}': obj={obj_init:.8f} **")
            elif obj_init < 1.7:
                print(f"  Init '{name}': obj={obj_init:.8f}")

    print(f"\nBest initial (n={len(best_seq)}): obj={best_score:.8f}")

    # === Phase 1: LSE-annealed L-BFGS-B ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 1: LSE-Annealed L-BFGS-B ===")
        alloc = min(120.0, remaining * 0.15)
        a_p1 = lse_anneal_optimize(best_seq, time_budget=alloc,
                                    T_start=3.0, T_end=80.0, rounds=6, verbose=True)
        obj_p1 = compute_obj_conv(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f} (no improvement)")

    # === Phase 2: Full LP Frank-Wolfe ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 2: Full LP Frank-Wolfe ===")
        n_cur = len(best_seq)
        if n_cur <= 600:
            alloc = min(200.0, remaining * 0.25)
            a_p2 = frank_wolfe_lp(best_seq, time_budget=alloc, mode='full', verbose=True)
        else:
            alloc = min(150.0, remaining * 0.2)
            a_p2 = frank_wolfe_lp(best_seq, time_budget=alloc, mode='topm', top_m=50, verbose=True)

        obj_p2 = compute_obj_conv(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_p2:.8f} (no improvement)")

    # === Phase 3: LSE polish after LP ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 3: LSE Polish ===")
        alloc = min(60.0, remaining * 0.08)
        a_p3 = lse_anneal_optimize(best_seq, time_budget=alloc,
                                    T_start=20.0, T_end=200.0, rounds=4, verbose=True)
        obj_p3 = compute_obj_conv(a_p3)
        if obj_p3 < best_score:
            best_score = obj_p3
            best_seq = a_p3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_p3:.8f}")

    # === Phase 4: Multi-direction LP ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 4: Multi-Direction LP ===")
        alloc = min(120.0, remaining * 0.15)
        a_p4 = multi_direction_lp(best_seq, time_budget=alloc, n_dirs=5, top_m=50, verbose=True)
        obj_p4 = compute_obj_conv(a_p4)
        if obj_p4 < best_score:
            best_score = obj_p4
            best_seq = a_p4.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_p4:.8f}")

    # === Phase 5: LP + LSE interleave ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 5: LP+LSE Interleave ===")
        n_inner = 12
        alloc = remaining * 0.18
        for it in range(n_inner):
            if time.monotonic() > deadline:
                break
            chunk = alloc / n_inner
            # LP step
            if len(best_seq) <= 500:
                a_step = frank_wolfe_lp(best_seq, time_budget=chunk * 0.6, mode='auto')
            else:
                a_step = frank_wolfe_lp(best_seq, time_budget=chunk * 0.6, mode='topm', top_m=35)
            obj_step = compute_obj_conv(a_step)
            if obj_step < best_score:
                best_score = obj_step
                best_seq = a_step.copy()
            # LSE step
            a_step2 = lse_lbfgs_round(best_seq, T=50.0, maxiter=30, time_limit=chunk * 0.35)
            obj_step2 = compute_obj_conv(a_step2)
            if obj_step2 < best_score:
                best_score = obj_step2
                best_seq = a_step2.copy()
            if it % 4 == 0:
                print(f"    Interleave it{it}: obj={best_score:.8f}")

    # === Phase 6: Coordinate Descent ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 6: Coordinate Descent ===")
        alloc = min(80.0, remaining * 0.1)
        a_p6 = coordinate_descent(best_seq, time_budget=alloc, max_sweeps=4, verbose=True)
        obj_p6 = compute_obj_conv(a_p6)
        if obj_p6 < best_score:
            best_score = obj_p6
            best_seq = a_p6.copy()
            print(f"  Phase 6: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6: obj={obj_p6:.8f}")

    # === Phase 7: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Phase 7: Multi-n Exploration ===")
        n_best = len(best_seq)
        n_options = []
        for delta in [-200, -100, -50, 50, 100, 200, 300]:
            n_try = n_best + delta
            if 200 <= n_try <= 2000 and n_try != n_best:
                n_options.append(n_try)
        n_options = sorted(set(n_options))[:5]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            # Interpolate
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 0.01:
                a_try = asymmetric_init(n_try, mode=1)
                s = a_try.sum()
            a_try = a_try / s

            # Quick optimize
            t1 = min(20.0, remaining * 0.25)
            a_try = lse_anneal_optimize(a_try, time_budget=t1, T_start=10, T_end=80, rounds=3)
            t2 = min(25.0, remaining * 0.3)
            a_try = frank_wolfe_lp(a_try, time_budget=t2, mode='topm', top_m=30)

            obj_try = compute_obj_conv(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 8: Final Polish ===")
        # LSE at very high T
        alloc1 = min(remaining * 0.25, 40.0)
        if alloc1 > 10:
            a_f = lse_anneal_optimize(best_seq, time_budget=alloc1,
                                       T_start=50.0, T_end=300.0, rounds=3)
            obj_f = compute_obj_conv(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LSE: obj={best_score:.8f} **")

        # LP
        alloc2 = min(remaining * 0.35, 50.0)
        if alloc2 > 10:
            a_f2 = frank_wolfe_lp(best_seq, time_budget=alloc2,
                                   mode='auto', top_m=40)
            obj_f2 = compute_obj_conv(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # CD
        alloc3 = min(remaining * 0.3, 30.0)
        if alloc3 > 8:
            a_f3 = coordinate_descent(best_seq, time_budget=alloc3, max_sweeps=3)
            obj_f3 = compute_obj_conv(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # === Phase 9: One last shot with diverse restart ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 9: Diverse Restart ===")
        n_cur = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            mode = (r * 3 + 1) % 9
            a_r = asymmetric_init(n_cur, mode=mode)
            blend = 0.3
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            t_arg = min(15.0, remaining * 0.3)
            a_r = lse_lbfgs_round(a_r, T=20.0, maxiter=30, time_limit=t_arg)
            t_lp = min(15.0, remaining * 0.3)
            if t_lp > 5:
                a_r = frank_wolfe_lp(a_r, time_budget=t_lp, mode='topm', top_m=30)

            obj_r = compute_obj_conv(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_conv(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
