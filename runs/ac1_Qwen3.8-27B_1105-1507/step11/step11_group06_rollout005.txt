    """Minimize a^T C a subject to a >= 0, sum(a) = 1"""
    a = np.ones(n) / n
    for it in range(max_iters):
        # Gradient: 2 * C @ a
        grad = 2 * C @ a
        # Projected gradient step
        lr = 1.0 / (2.0 * np.abs(C).max() * n)  # conservative step size

        # Project onto simplex

            break
        a = a_new
    return a

def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}"""
    n = len(v)
    # Use the algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_conv_fast(a):
    """Compute linear convolution a*a using FFT"""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def gradient_at_lag(a, k):
    """Compute gradient of conv(a,a)[k] with respect to a"""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    return g

def projected_gradient_step(a, k, lr):
    """One projected gradient step to minimize conv(a,a)[k]"""
    g = gradient_at_lag(a, k)
    a_new = a - lr * g
    return project_simplex(a_new)

"""
Optimized step function coefficient search combining LP-based directions,
golden-section quadratic line search, Nesterov-accelerated subgradient,
and symmetric constraint exploitation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.PILE, cp.SCS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def golden_section_line_search(a, g_dir, t_max=1.5, tol=1e-6):
    """Minimize f(t) = obj((1-t)*a + t*g_dir) using golden section search.
    The function is convex in t (max of quadratics)."""
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    a_t = a.copy()
    g_d = g_dir.copy()

    def f_of_t(t):
        x = (1.0 - t) * a_t + t * g_d
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 0.01:
            return np.inf
        x = x / s
        return compute_obj(x)

    lo = 0.0
    hi = t_max
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)

    best_t = 0.0
    best_f = f_of_t(0.0)
    if fc < best_f:
        best_f = fc
        best_t = c
    if fd < best_f:
        best_f = fd
        best_t = d

    while (hi - lo) > tol:
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = f_of_t(d)
        if fc < best_f:
            best_f = fc
            best_t = c
        if fd < best_f:
            best_f = fd
            best_t = d

    return best_t, best_f


def gradient_at_lag(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a"""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    return g


def nesterov_subgradient(a, time_budget=20.0, n_active=5):
    """Nesterov-accelerated subgradient method for min max_k conv(a,a)[k]"""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Nesterov state
    x = a.copy()  # main iterate
    y = a.copy()  # Nesterov lookahead
    beta = 1.0  # initial momentum parameter

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        # Evaluate at y (lookahead point)
        conv_y = compute_conv(y)
        max_val = float(np.max(conv_y))

        # Find active lags (top n_active)
        sorted_idx = np.argsort(conv_y)[::-1]
        active_lags = sorted_idx[:n_active]
        active_vals = conv_y[active_lags]
        # Soft weights: proportional to values
        w = active_vals / float(np.sum(active_vals))

        # Compute gradient at y for the weighted combination
        grad = np.zeros(n)
        for k_idx, wk in zip(active_lags, w):
            grad += wk * gradient_at_lag(y, int(k_idx))

        # Step size: use decreasing schedule
        t = 1.0 / (np.sqrt(it + 1) + 1.0)
        lr = t / (np.linalg.norm(grad) + 1e-10)
        lr = min(lr, 2.0)  # cap step size

        # Update
        x_new = y - lr * grad
        x_new = np.maximum(x_new, 0)
        s_new = float(np.sum(x_new))
        if s_new < 1e-10:
            x_new = best_a.copy()
        else:
            x_new = x_new / s_new

        # Nesterov update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = (beta_new / (beta_new + 1.0)) * x_new + (beta / (beta_new + 1.0)) * y
        y_new = np.maximum(y_new, 0)
        s_y = float(np.sum(y_new))
        if s_y < 1e-10:
            y_new = x_new.copy()
        else:
            y_new = y_new / s_y

        x = x_new
        y = y_new
        beta = beta_new

        # Evaluate
        obj = compute_obj(x)
        if obj < best_obj:
            best_obj = obj
            best_a = x.copy()

        it += 1
        # Reset if stuck for too long
        if it % 500 == 0:
            obj_y = compute_obj(y)
            if obj_y < best_obj:
                best_obj = obj_y
                best_a = y.copy()

    return best_a, best_obj


def lp_with_golden(a, time_budget=100.0, max_iters=80):
    """LP-based optimization with golden section line search"""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)

def diverse_init(n, rng, mode=0):

    elif mode == 3:
        a = np.exp(6.0 * (2 * x - 1) ** 2)

    elif mode == 5:

        a = a ** 3 + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:

    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s
