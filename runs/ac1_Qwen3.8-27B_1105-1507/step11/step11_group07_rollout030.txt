def soft_max_gradient_descent(a, time_budget=200.0, beta_start=10.0, beta_end=50.0,
                              lr_start=0.008, momentum=0.9, verbose=True, seed=42):
    """Soft-max gradient descent with Nesterov momentum."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    lr = lr_start
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.3 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient
        x_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A = rfft(x_rev, size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 100:
                v = np.zeros(n)
                stall = 0
                pert = rng.normal(0, 0.002, n)
                a = project_simplex(best_a + pert)
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    SmaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
        
        if it - best_it > 5000 and it > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            it = best_it
            lr *= 0.7
    
    return best_a, best_obj

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # For large n, this LP has 2n-1 constraints and n variables
    # Let's only use it for n <= 1500
    if n > 2000:
        return None, None
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def peak_shave_optimize(a, time_budget=100.0, top_k=10, verbose=False):
    """
    New approach: Iterative peak shaving.
    At each step, identify the top peaks in conv(a,a), compute for each position
    its contribution to those peaks, and reduce those positions.
    """

    a = project_simplex(a.astype(np.float64))

        conv = compute_conv(a)
        # Find top peaks
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[sorted_idx]
        
        # For each position i, compute its "contribution score" to the top peaks
        # Contribution of a[i] to conv[k] is: a[i]*a[k-i] + a[i]*a[i-k] (symmetric)
        # Actually: d_conv_k/d_a_i = 2 * a[k-i] if 0 <= k-i < n
        # So the "harshness" of position i is: sum over top peaks k of 2*a[k-i]
        
        harshness = np.zeros(n)
        for k in sorted_idx:
            weight = peak_vals[sorted_idx[k] if isinstance(k, int) else k]  # hmm
            # For peak at position k in conv, the gradient contribution to a[i] is 2*a[k-i]
            # if 0 <= k-i < n
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                idx_range = np.arange(lo, hi + 1)
                conv_idx = k - idx_range  # k - i for i in idx_range
                harshness[idx_range] += 2.0 * a[conv_idx]
        
        # Normalize harshness
        h_max = np.max(harshness)
        if h_max > 0:
            harshness /= h_max
        
        # Create update: reduce positions with high harshness
        # a_new = a * (1 - eta * harshness) then project
        eta = 0.05  # step size
        a_new = a * (1.0 - eta * harshness)
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1

                a = project_simplex(best_a + noise)

"""
Multi-phase step function optimization combining L-BFGS-B with softmax reparameterization,
peak shaving heuristic, soft-max gradient descent, and LP polishing.
Key innovations: quasi-Newton optimization via scipy L-BFGS-B on log-sum-exp smooth approximation,
iterative peak-shaving based on convolution gradient attribution, and diverse restarts.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lsexp_obj_and_grad(x, beta):
    """
    Log-sum-exp objective and gradient for x with sum(x)=1.
    obj = (1/beta) * log(sum_k exp(beta * conv(x,x)_k))
    grad_i = 2 * sum_k p_k * x_{k-i} where p is softmax of conv values.
    Uses FFT cross-correlation for efficiency.
    """
    n = len(x)
    conv = compute_conv(x)
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    p = exp_vals / float(np.sum(exp_vals))
    obj = (c_max + float(np.log(np.sum(exp_vals)))) / beta

    # Gradient via cross-correlation
    x_rev = x[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    X = rfft(x_rev, size)
    P = rfft(p, size)
    full = irfft(X * P, size)
    q = full[n - 1:n - 1 + n]
    grad = 2.0 * q

    return obj, grad


def softmax_from_w(w):
    """Convert log-weights to probabilities via softmax."""
    w = w - np.max(w)
    e = np.exp(w)
    return e / np.sum(e)

        w_full = np.append(w, 0.0)
        a_trial = softmax_from_w(w_full)
        obj, _ = lsexp_obj_and_grad(a_trial, beta)
        return obj

    def jac_w(w):
        w_full = np.append(w, 0.0)
        a_trial = softmax_from_w(w_full)
        obj, grad_a = lsexp_obj_and_grad(a_trial, beta)
        # Chain rule: grad_w_i = sum_j grad_a_j * da_j/dw_i
        # da_j/dw_i = a_j * (delta_ji - a_i)
        dot = float(np.dot(grad_a, a_trial))
        grad_w_full = a_trial * (grad_a - dot)
        return grad_w_full[:-1]

    try:
        result = minimize(
            fun_w, w_init, jac=jac_w, method='L-BFGS-B',
            options={'maxiter': maxiter, 'maxfun': 15000, 'gtol': 1e-9, 'iprint': -1}
        )
        w_result = result.x
        w_full = np.append(w_result, 0.0)
        a_result = softmax_from_w(w_full)
        a_result = project_simplex(np.maximum(a_result, 0))
        obj_result = compute_obj(a_result)

        if obj_result < best_obj:
            best_obj = obj_result
            best_a = a_result
    except RuntimeError:
        pass
    except Exception as e:
        print(f"  L-BFGS-B warning: {e}")

    return best_a, best_obj, call_count[0]


def soft_max_gradient_descent(a, time_budget=200.0, beta_start=10.0, beta_end=50.0,
                              lr_start=0.008, momentum=0.9, verbose=True, seed=42):
    """Soft-max gradient descent with momentum on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    lr = lr_start
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.3 * progress)

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))

        x_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A = rfft(x_rev, size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q

        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 150:
                v = np.zeros(n)
                stall = 0
                pert = rng.normal(0, 0.002, n)
                a = project_simplex(best_a + pert)

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    SmaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")

        if it - best_it > 5000 and it > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            it = best_it
            lr *= 0.7

    return best_a, best_obj


def peak_shave_optimize(a, time_budget=100.0, top_k=15, verbose=False):
    """
    Iterative peak shaving: identify top peaks in conv(a,a), compute each position's
    contribution to those peaks, and reduce high-contribution positions.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    it = 0
    stall = 0
    eta = 0.08

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top peaks
        peak_indices = np.argsort(conv)[::-1][:top_k]
        peak_values = conv[peak_indices]

        # Compute harshness: for each position i, sum of 2*a[k-i] over top peaks k
        harshness = np.zeros(n)
        for idx in range(len(peak_indices)):
            k = peak_indices[idx]
            weight = peak_values[idx]
            # d_conv_k / da_i = 2 * a[k-i] if 0 <= k-i < n
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                i_range = np.arange(lo, hi + 1)
                a_idx = k - i_range
                harshness[i_range] += 2.0 * weight * a[a_idx]

        h_max = float(np.max(harshness))
        if h_max > 0:
            harshness /= h_max

        # Multiplicative reduction
        a_new = a * (1.0 - eta * harshness)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Decay eta slowly
            eta *= 0.999
        else:
            stall += 1
            if stall > 80:
                noise = rng.normal(0, 0.01, n)
                a = project_simplex(best_a + noise)
                stall = 0
                eta = 0.08

        it += 1
        if verbose and it % 200 == 0:
            print(f"    PeakShave it{it}: obj={best_obj:.8f}, eta={eta:.5f}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def multiplicative_update_optimize(a, time_budget=100.0, verbose=False):
    """
    Alternative: multiplicative weight update based on peak attribution.
    At each step, compute the ratio of target (flat) conv to actual conv,
    and update a multiplicatively to reduce peaks.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(456)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        conv_max = float(np.max(conv))
        # Target: uniform conv at level conv_max * 0.8 (try to reduce peaks by 20%)
        # For each position i, penalize proportional to its contribution to peaks
        # Use the top 5% of conv values as "peaks"
        threshold = np.percentile(conv, 95)
        peak_mask = conv > threshold

        # Compute gradient of "excess peak" objective
        excess = np.maximum(conv - threshold, 0)
        # Weighted gradient
        x_rev = a[::-1]
        size = 1
        while size < n + len(excess) - 1:
            size <<= 1
        A = rfft(x_rev, size)
        E = rfft(excess, size)
        full = irfft(A * E, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q

        # Multiplicative update: a_i *= exp(-eta * grad_i)
        grad_max = float(np.max(np.abs(grad)))
        if grad_max > 0:
            grad = grad / grad_max
        eta = 0.3
        a_new = a * np.exp(-eta * grad)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 60:
                noise = rng.normal(0, 0.008, n)
                a = project_simplex(best_a + noise)
                stall = 0

        it += 1
        if verbose and it % 200 == 0:
            print(f"    MultUpd it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 1500:
        return None, None
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_lp, time_budget=3.0):
    """Golden section search for optimal blend factor."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5

    def obj_at(t):
        t = max(0, t)
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(25):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.2, 1.4]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best


def lp_polish(a, time_budget=120.0, max_iters=60, verbose=False):
    """LP-based polishing: iteratively find best LP direction and blend."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.05))

        if t_opt > 0 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 4:
            if stall <= 7:
                noise = rng.normal(0, 0.005 * (stall - 3), n)
                a = project_simplex(best_a + noise)
            else:
                x = np.linspace(0, 1, n)
                a_div = 1.0 / (0.05 + 4.0 * (2 * x - 1) ** 2)
                a = project_simplex(0.5 * a_div + 0.5 * best_a)
            stall = 0

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.005, 0.995, n)
    a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2 + 1e-10)
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations based on mode."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        if n >= 7:
            kernel = np.ones(7) / 7
            a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        if n > 3:
            a[1:3] = 0.3
            a[n-3:n-1] = 0.3
            if n > 6:
                a[3:n-3] = 0.05
    elif mode == 8:
        a = 1.0 / (0.01 + 8.0 * (2 * x - 1) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy:
    Phase 1: L-BFGS-B optimization (quasi-Newton, fast convergence)
    Phase 2: Soft-max gradient descent with momentum
    Phase 3: Peak shaving + multiplicative update
    Phase 4: LP polishing
    Phase 5: Diverse restarts
    Phase 6: Length exploration
    Phase 7: Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except (NameError, Exception):
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # Try arcsine initialization
    a_arc = arcsine_init(n_current)
    obj_arc = compute_obj(a_arc)
    if obj_arc < best_score:
        best_score = obj_arc
        best_seq = a_arc.copy()
        print(f"Arcsine init: obj={best_score:.8f}")

    # Try sharp U-shape
    a_sharp = u_shaped_init(n_current, sharpness=8.0)
    obj_sharp = compute_obj(a_sharp)
    if obj_sharp < best_score:
        best_score = obj_sharp
        best_seq = a_sharp.copy()
        print(f"Sharp U init: obj={best_score:.8f}")

    # === Phase 1: L-BFGS-B optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: L-BFGS-B (beta=15), time={remaining:.0f}s ===")
        alloc = min(remaining * 0.20, 200.0)
        a1, obj1, n_calls = lbfgs_optimize(best_seq, time_budget=alloc, beta=15.0, maxiter=4000)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  L-BFGS-B beta=15: obj={best_score:.8f} ** ({n_calls} evals)")
        else:
            print(f"  L-BFGS-B beta=15: obj={obj1:.8f} ({n_calls} evals)")

        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"  L-BFGS-B (beta=40), time={remaining:.0f}s remaining")
            alloc = min(remaining * 0.18, 170.0)
            a1b, obj1b, n_calls = lbfgs_optimize(best_seq, time_budget=alloc, beta=40.0, maxiter=3000)
            if obj1b < best_score:
                best_score = obj1b
                best_seq = a1b.copy()
                print(f"  L-BFGS-B beta=40: obj={best_score:.8f} ** ({n_calls} evals)")
            else:
                print(f"  L-BFGS-B beta=40: obj={obj1b:.8f} ({n_calls} evals)")

    # === Phase 2: Soft-max gradient descent ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 2: SoftMax GD, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.18, 180.0)
        a2, obj2 = soft_max_gradient_descent(
            best_seq, time_budget=alloc,
            beta_start=8.0, beta_end=40.0,
            lr_start=0.01, momentum=0.9,
            verbose=True, seed=seed
        )
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 SmaxGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 SmaxGD: obj={obj2:.8f}")

    # === Phase 3: Peak shaving + multiplicative update ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: Peak Shaving + MultUpdate, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.12, 80.0)
        a3, obj3 = peak_shave_optimize(best_seq, time_budget=alloc, top_k=20, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  PeakShave: obj={best_score:.8f} **")
        else:
            print(f"  PeakShave: obj={obj3:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.10, 60.0)
            a3b, obj3b = multiplicative_update_optimize(best_seq, time_budget=alloc, verbose=True)
            if obj3b < best_score:
                best_score = obj3b
                best_seq = a3b.copy()
                print(f"  MultUpdate: obj={best_score:.8f} **")
            else:
                print(f"  MultUpdate: obj={obj3b:.8f}")

    # === Phase 4: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 4: LP Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.20, 180.0)
        a4, obj4 = lp_polish(best_seq, time_budget=alloc, max_iters=50, verbose=True)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj4:.8f}")

    # === Phase 5: L-BFGS-B again on polished result ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: L-BFGS-B (beta=80) on polished, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.20, 120.0)
        a5, obj5, n_calls = lbfgs_optimize(best_seq, time_budget=alloc, beta=80.0, maxiter=3000)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  L-BFGS-B beta=80: obj={best_score:.8f} ** ({n_calls} evals)")
        else:
            print(f"  L-BFGS-B beta=80: obj={obj5:.8f} ({n_calls} evals)")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 6: Diverse Restarts, time={remaining:.0f}s ===")
        n_cur = len(best_seq)

        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            mode = r
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.5
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)

            # Quick L-BFGS-B
            alloc_lbfgs = min(remaining * 0.25, 50.0)
            if alloc_lbfgs > 10:
                a_r, obj_r, _ = lbfgs_optimize(a_r, time_budget=alloc_lbfgs, beta=20.0, maxiter=2000)

            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.2, 30.0)
                a_r, obj_r = lp_polish(a_r, time_budget=alloc_lp, max_iters=15, verbose=False)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 7: Length Exploration, time={remaining:.0f}s ===")
        n_best = len(best_seq)

        n_options = sorted(set([
            max(400, n_best - 200), max(400, n_best - 100),
            n_best + 100, n_best + 200, n_best + 400,
            600, 900, 1200, 1500
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_best) >= 50]
        n_options = n_options[:4]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))

            alloc_sgd = min(remaining * 0.4, 40.0)
            if alloc_sgd > 10:
                a_try, obj_try = soft_max_gradient_descent(
                    a_try, time_budget=alloc_sgd,
                    beta_start=10.0, beta_end=50.0,
                    lr_start=0.008, momentum=0.9,
                    verbose=False, seed=seed + 400
                )

            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.3, 25.0)
                a_try, obj_try = lp_polish(a_try, time_budget=alloc_lp, max_iters=12, verbose=False)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Final Polish, time={remaining:.0f}s ===")

        # Peak shaving with many peaks
        alloc = min(remaining * 0.25, 50.0)
        if alloc > 10:
            a_f, obj_f = peak_shave_optimize(best_seq, time_budget=alloc, top_k=40, verbose=False)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final PeakShave: obj={best_score:.8f} **")

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 25:
            alloc = min(remaining * 0.35, 60.0)
            a_f2, obj_f2 = lp_polish(best_seq, time_budget=alloc, max_iters=25, verbose=True)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # One more L-BFGS-B
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 40.0)
            a_f3, obj_f3, _ = lbfgs_optimize(best_seq, time_budget=alloc, beta=100.0, maxiter=2000)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final L-BFGS: obj={best_score:.8f} **")

    # === Phase 9: Last-ditch random perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            noise = rng.normal(0, 0.003, len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.35, 15.0)
            if alloc > 5:
                a_p, obj_p = lp_polish(a_p, time_budget=alloc, max_iters=8, verbose=False)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Last-ditch {attempt}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
