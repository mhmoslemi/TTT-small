def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def compute_smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv, p


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):

    Exact coordinate descent with gradient-informed bracketing.
    Uses the smoothed gradient to provide better initial brackets for golden section.
    """

        return a

        # Recompute conv periodically
        if sweep % 4 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        grad = None
        if sweep % 8 == 1:
            try:
                grad, _, _ = compute_smoothed_gradient(a / float(np.sum(a)), 50.0)
            except Exception:
                grad = None

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)

                    max_lin = 0.0

                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

                              v, max(0.0, v + 0.5 * step * np.sign(-grad[i])),
                              max(0.0, v + step * np.sign(-grad[i])),
                              v * 0.5, v * 1.5, 1.0 / n]
            else:
                scale = max(v, 1.0 / n, 1e-6)
                candidates = [0.0, v * 0.5, v * 0.7, v * 1.3, v * 1.5,

            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            scale = max(best_x, 1.0 / n, 1e-6)
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.5
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break

                    x_hi = d_pt
                    d_pt = c_pt

                else:

                    c_pt = d_pt
                    fc = fd

                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

def multiplicative_descent(a, time_budget=80.0, tau_init=50.0, tau_end=500.0, verbose=False):
    """Multiplicative gradient descent with adaptive learning rate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr_base = 8.0 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        lr = lr_base * (1.0 - 0.7 * progress)

        try:
            grad, conv, p = compute_smoothed_gradient(a, tau)
        except Exception:
            break

        # Apply multiplicative update with clipping
        update = np.exp(-lr * grad)
        update = np.clip(update, 0.01, 100.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 1000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def annealed_search(a, time_budget=100.0, T_init=0.01, T_end=0.001, seed=42):
    """
    Simulated annealing with structured perturbations.
    Perturbations are drawn from a mixture of:
    - Small Gaussian noise
    - Coordinate swaps
    - Block rescaling
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    cur_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    accept_count = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        T = T_init * (T_end / T_init) ** progress

        # Choose perturbation type
        r = rng.random()
        a_try = a.copy()

        if r < 0.5:
            # Gaussian noise
            scale = 0.02 * (1.0 - 0.5 * progress)
            a_try = a * np.exp(rng.normal(0, scale, n))
        elif r < 0.75:
            # Block rescaling
            bs = rng.integers(2, max(3, n // 20))
            i0 = rng.integers(0, n - bs)
            a_try[i0:i0+bs] *= np.exp(rng.normal(0, 0.05))
        else:
            # Coordinate swap-like: scale up one coord, scale down another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                factor = np.exp(rng.normal(0, 0.1))
                a_try[i] *= factor
                a_try[j] /= factor

        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj_normalized(a_try)
        delta = obj_try - cur_obj

        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_try
            cur_obj = obj_try
            accept_count += 1
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
        else:
            # Occasionally accept large perturbations to escape
            if rng.random() < 0.01:
                a = a_try
                cur_obj = obj_try

        it += 1

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-4, 4)
        a = a / a.sum()
    elif mode == 6:
        # Triangular with random peaks
        a = 1.0 - np.abs(2 * np.linspace(0, 1, n) - 1)
        a += 0.2 * rng.uniform(0, 1, n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=30.0, tau_end=400.0, verbose=False):
    """Nesterov-accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        lr_eff = lr * (1.0 - 0.5 * progress)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        try:
            grad, _, _ = compute_smoothed_gradient(y, tau)
        except Exception:
            break

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr_eff = min(max(lr_eff * 0.995 + 0.005 / (g_norm + 1e-10),
                            0.05 * lr), lr)

        v = beta * v - lr_eff * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    NSGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")
