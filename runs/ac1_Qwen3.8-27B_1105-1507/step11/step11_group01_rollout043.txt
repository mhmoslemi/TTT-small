def compute_conv(a):

    n = len(a)
    if n == 0:

    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-30:

    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g) (linear)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """

    Returns (g, M) or (None, None) on failure.

    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_g = None
    best_M = np.inf
    
    for solver_name in ['MOSEK', 'CLARABEL', 'SCS']:

                prob.solve(solver=cp.CLARABEL)
            else:
                prob.solve(solver=cp.SCS)

                gs = float(np.sum(g_val))

        except Exception:

def compute_conv(a):

    n = len(a)

        return np.array([])
    size = 1
    while size < 2 * n - 1:

    """Full linear convolution a*b via FFT."""

            kq = 2 * i
            if k_lo <= kq <= k_hi:
                qi = np.where(ks == kq)[0][0]
                cq = float(cv[qi])
                lks = np.delete(ks, qi)
                lav = np.delete(av, qi)
                lcv = np.delete(cv, qi)
                hq = True
            else:
                lks = ks
                lav = av
                lcv = cv
                cq = 0.0
                hq = False

            # Compute unaffected max (critical!)
            nconv = 2 * n - 1
            mask = np.zeros(nconv, dtype=bool)
            mask[ks] = True
            if not mask.all():
                unaff_max = float(np.max(conv[~mask]))
            else:
                unaff_max = 0.0

            def f1(x, v=v, lcv=lcv, lav=lav, cq=cq, hq=hq, unaff_max=unaff_max, n=n):
                if len(lav) > 0:
                    ml = float(np.max(lcv + 2.0 * (x - v) * lav))
                else:
                    ml = 0.0
                if hq:
                    mq = cq + x * x - v * v
                else:
                    mq = 0.0
                mv = max(ml, mq, unaff_max)
                d = 1.0 - v + x
                if d < 1e-30:
                    return 1e30
                return 2.0 * n * mv / (d * d)

            # Candidate evaluation
            cands = [0.0, v, v * 0.5, v * 1.5, 1.0 / n, 2.0 / n, 0.5 / n]
            cands = sorted(set(max(0.0, float(c)) for c in cands))
            bx, bf = v, f1(v)
            for xc in cands:
                fc = f1(xc)
                if fc < bf:
                    bf, bx = fc, xc

            # Golden section search
            sc = max(v, 1.0 / n, 1e-8)
            lo = max(0.0, bx - sc)
            hi = bx + 2.0 * sc
            if hi <= lo:
                hi = lo + 1e-10
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc, fd = f1(c), f1(d)
            for _ in range(45):
                if hi - lo < 1e-14 * max(1.0, hi):
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = f1(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = f1(d)

            xo = max(0.0, (lo + hi) / 2.0)
            if abs(xo - v) > 1e-14:
                a[i] = xo
                if hq:
                    conv[kq] += xo * xo - v * v
                if len(lav) > 0:
                    conv[lks] += 2.0 * (xo - v) * lav
                s2 = float(np.sum(a))
                if s2 > 1e-10:
                    o2 = 2.0 * n * float(np.max(conv)) / (s2 * s2)
                    if o2 < best_obj:
                        best_obj = o2
                        best_a = (a / s2).copy()

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences (normalized, sum=1)."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 6
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Semicircle (arcsine distribution)
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x * x)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1)
        a = x ** -0.5
        a = a / a.sum()
    elif mode == 6:
        # Flat middle, tapered edges
        a = np.ones(n)
        taper = n // 8
        if taper > 0:
            a[:taper] = np.linspace(0.1, 1, taper)
            a[-taper:] = np.linspace(1, 0.1, taper)
        a = a / a.sum()
    elif mode == 7:
        # Three bumps
        for c in [n // 6, n // 2, 5 * n // 6]:
            w = n // 10
            a = a + np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2)) if n > 0 else a
        if 'a' not in dir() or len(a) != n:
            a = np.ones(n)
            for c in [n // 6, n // 2, 5 * n // 6]:
                w = n // 10
                a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def lp_iteration(a, t0_lp, budget_lp):
    """
    One LP iteration: solve LP, find optimal blend, update a.
    Returns (new_a, new_obj, improved).
    """
    n = len(a)
    g, M_lp = solve_lp(a)
    if g is None:
        return a, obj_norm(a), False

    best_t, best_obj = blend_search(a, g)

    if best_t < 1e-6:
        return a, obj_norm(a), False

    a_new = (1.0 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a, obj_norm(a), False
    a_new /= s

    obj_new = obj_norm(a_new)
    obj_old = obj_norm(a)
    improved = obj_new < obj_old - 1e-15
    return a_new, obj_new, improved


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP minimax + coordinate descent + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    # --- Initialize ---
    n_init = 800
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        pa = np.clip(pa, 0, None)
        if pa.size > 0 and np.isfinite(pa).all():
            ps = float(np.sum(pa))
            if ps > 0.01:
                parent = pa / ps
    except:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_n = len(parent)
        best_score = obj_norm(best_a)
        print(f"Parent loaded: n={best_n}, obj={best_score:.8f}")
    else:
        best_a = np.ones(n_init) / n_init
        best_n = n_init
        best_score = obj_norm(best_a)
        print(f"Default init: n={best_n}, obj={best_score:.8f}")

    # --- Phase 1: Initial CD polish ---
    print("\n=== Phase 1: Initial Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 120.0)
    if p1_budget > 20:
        a1 = cd_fast(best_a, p1_budget, seed=seed, verbose=True)
        o1 = obj_norm(a1)
        if o1 < best_score:
            best_score = o1
            best_a = a1
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={o1:.8f}")

    # --- Phase 2: LP Iterations (main workhorse) ---
    print(f"\n=== Phase 2: LP Minimax Iterations (n={len(best_a)}) ===")
    p2_budget = min((deadline - time.monotonic()) * 0.4, 600.0)
    a = best_a.copy()
    cur_obj = best_score
    lp_count = 0
    stall = 0
    t0_p2 = time.monotonic()

    while time.monotonic() < deadline and (deadline - time.monotonic()) > 30:
        elapsed_p2 = time.monotonic() - t0_p2
        if elapsed_p2 > p2_budget:
            break
        remaining_p2 = p2_budget - elapsed_p2
        if remaining_p2 < 15:
            break

        # LP solve + blend
        a_new, obj_new, improved = lp_iteration(a, time.monotonic(), remaining_p2)
        lp_count += 1

        if improved:
            a = a_new
            cur_obj = obj_new
            if cur_obj < best_score:
                best_score = cur_obj
                best_a = a.copy()
                print(f"  LP{lp_count}: obj={best_score:.8f} ** (t budget used: {elapsed_p2:.0f}s)")
            stall = 0
        else:
            stall += 1

        # Periodic CD polish every 3 LP iterations
        if lp_count % 3 == 0 and time.monotonic() < deadline - 60:
            cd_alloc = min(20.0, (deadline - time.monotonic()) * 0.05)
            if cd_alloc > 10:
                a_cd = cd_fast(a, cd_alloc, seed=lp_count * 100)
                o_cd = obj_norm(a_cd)
                if o_cd < cur_obj:
                    a = a_cd
                    cur_obj = o_cd
                    if o_cd < best_score:
                        best_score = o_cd
                        best_a = a.copy()
                        print(f"  CD{lp_count}: obj={best_score:.8f} **")

        # Escape from stall: perturb and try again
        if stall >= 4:
            print(f"  Stall at {lp_count} LPs, obj={cur_obj:.8f}. Perturbing...")
            pert = rng.normal(0, 0.03, len(a))
            a_p = a * np.exp(pert)
            a_p = np.maximum(a_p, 0)
            s_p = float(np.sum(a_p))
            if s_p > 1e-10:
                a_p /= s_p
                o_p = obj_norm(a_p)
                if o_p < cur_obj:
                    a = a_p
                    cur_obj = o_p
                else:
                    # Try stronger perturbation
                    pert2 = rng.normal(0, 0.08, len(a))
                    a_p2 = a * np.exp(pert2)
                    a_p2 = np.maximum(a_p2, 0)
                    s_p2 = float(np.sum(a_p2))
                    if s_p2 > 1e-10:
                        a_p2 /= s_p2
                        o_p2 = obj_norm(a_p2)
                        if o_p2 < cur_obj:
                            a = a_p2
                            cur_obj = o_p2
            stall = 0

    print(f"  Phase 2 done: {lp_count} LPs, best={best_score:.8f}, time={time.monotonic()-t0_p2:.0f}s")

    # --- Phase 3: Multi-n Exploration ---
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    p3_budget = min((deadline - time.monotonic()) * 0.3, 350.0)
    n_cur = len(best_a)

    # Generate candidate n values
    n_cands = sorted(set([
        max(200, n_cur - 200), max(200, n_cur - 100),
        n_cur + 100, n_cur + 300, n_cur + 600,
        500, 700, 900, 1100, 1300, 1600, 2000, 2500
    ]))
    # Filter
    n_filtered = []
    for nc in n_cands:
        if 300 <= nc <= 3000:
            if all(abs(nc - nf) > 100 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:8]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == len(best_a):
            continue

        # Resample
        a_try = resample_seq(best_a, n_try)
        s_t = float(np.sum(a_try))
        if s_t < 1e-10:
            continue
        a_try /= s_t

        # Quick CD
        cd_b = min(30.0, remaining * 0.15)
        if cd_b > 10:
            a_try = cd_fast(a_try, cd_b, seed=n_try)

        o_try = obj_norm(a_try)
        if o_try < best_score:
            best_score = o_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={o_try:.8f}")

            # If close, try LP on it
            if o_try < best_score + 0.01 and remaining > 60:
                for _ in range(2):
                    a_try, o_try2, imp = lp_iteration(a_try, time.monotonic(), 30)
                    if imp and o_try2 < o_try:
                        a_try = a_try / np.sum(a_try)
                        o_try = o_try2
                    if o_try < best_score:
                        best_score = o_try
                        best_a = a_try.copy()
                        print(f"    LP on n={n_try}: obj={best_score:.8f} **")
                    remaining = deadline - time.monotonic()
                    if remaining < 30:
                        break

    # --- Phase 4: Final Polish ---
    print(f"\n=== Phase 4: Final Polish (n={len(best_a)}) ===")
    remaining = deadline - time.monotonic()
    a = best_a.copy()

    for rnd in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # LP step
        a_new, obj_new, improved = lp_iteration(a, time.monotonic(), remaining)
        if improved:
            a = a_new
            if obj_new < best_score:
                best_score = obj_new
                best_a = a.copy()
                print(f"  Final LP{rnd}: obj={best_score:.8f} **")
        else:
            # Try perturbation + LP
            pert = rng.normal(0, 0.02, len(a))
            a_p = a * np.exp(pert)
            a_p = np.maximum(a_p, 0)
            s_p = float(np.sum(a_p))
            if s_p > 1e-10:
                a_p /= s_p
                a_new2, obj_new2, imp2 = lp_iteration(a_p, time.monotonic(), remaining)
                if imp2 and obj_new2 < best_score:
                    best_score = obj_new2
                    best_a = a_new2.copy()
                    a = a_new2
                    print(f"  Pert+LP{rnd}: obj={best_score:.8f} **")

        # CD step
        remaining = deadline - time.monotonic()
        if remaining > 30:
            cd_b = min(remaining * 0.3, 80.0)
            a_cd = cd_fast(a, cd_b, seed=10000 + rnd * 100, verbose=True)
            o_cd = obj_norm(a_cd)
            if o_cd < best_score:
                best_score = o_cd
                best_a = a_cd.copy()
                a = a_cd
                print(f"  Final CD{rnd}: obj={best_score:.8f} **")

    # --- Phase 5: Try diverse restarts if time permits ---
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Diverse Restarts ===")
        n_cur = len(best_a)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = r + 1
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with best
            blend = 0.4
            a_r = (1 - blend) * a_r + blend * best_a
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r < 1e-10:
                continue
            a_r /= s_r

            cd_b = min(remaining * 0.15, 40.0)
            if cd_b > 10:
                a_r = cd_fast(a_r, cd_b, seed=20000 + r * 200)
            o_r = obj_norm(a_r)

            # Try LP on top
            if o_r < best_score + 0.05 and remaining > 30:
                a_r2, o_r2, imp = lp_iteration(a_r, time.monotonic(), 30)
                if imp:
                    o_r = o_r2
                    a_r = a_r2

            if o_r < best_score:
                best_score = o_r
                best_a = a_r.copy()
                print(f"  Restart{r} (mode{mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart{r} (mode{mode}): obj={o_r:.8f}")

    # --- Final output ---
    print(f"\n=== FINAL: n={len(best_a)}, score={best_score:.8f}, time={time.monotonic()-start_time:.1f}s ===")

    # Ensure valid output for evaluate_sequence
    best_a = np.maximum(best_a, 0.0)
    result = [float(min(1000.0, x)) for x in best_a]

    # Validate
    val = evaluate_sequence(result)
    if val == np.inf or val >= best_score + 0.01:
        # Fallback
        result = [1.0] * 200
        val = evaluate_sequence(result)
        print(f"  Fallback used: score={val}")

    return result
