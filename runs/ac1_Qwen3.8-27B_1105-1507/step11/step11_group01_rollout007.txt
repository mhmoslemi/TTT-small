import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2n*max(conv)/(sum)^2 for sequence a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_active(a, active_indices, all_indices=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for k in active_indices, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    A_active = A[active_indices, :]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_active @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)

                    conv_check = A[all_indices, :] @ g_val
                    M_actual = float(np.max(conv_check))
                    return g_val, M_actual
        except Exception:
            continue
    return None, None

    """Evaluate 1D subproblem objective at x."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def block_cd(a, time_budget=100.0, block_size=3, verbose=False, seed=42):
    """Block coordinate descent: optimize blocks of adjacent coordinates jointly."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    b = min(block_size, n)

    while time.monotonic() - start < time_budget:
        # Choose random block positions
        num_blocks = max(1, n // (b * 2))
        starts = rng.integers(0, n - b + 1, size=num_blocks)

        for s in starts:
            if time.monotonic() - start >= time_budget:
                break
            idx = list(range(s, s + b))
            v_block = a[idx].copy()
            S_other = float(np.sum(a)) - float(np.sum(v_block))

            # Nelder-Mead style in the block
            best_block = v_block.copy()
            best_f = _block_objective(v_block, a, idx, s, n, S_other)

            # Try several random perturbations
            for attempt in range(30):
                scale = max(float(np.mean(v_block)), 1.0 / n)
                pert = rng.normal(0, scale * 0.3, size=b)
                new_block = np.maximum(v_block + pert, 0)
                f_new = _block_objective(new_block, a, idx, s, n, S_other)
                if f_new < best_f:
                    best_f = f_new
                    best_block = new_block.copy()
                    v_block = new_block.copy()

            # Local refinement with smaller steps
            for _ in range(15):
                scale = max(float(np.mean(v_block)), 1.0 / n) * 0.1
                pert = rng.normal(0, scale, size=b)
                new_block = np.maximum(v_block + pert, 0)
                f_new = _block_objective(new_block, a, idx, s, n, S_other)
                if f_new < best_f:
                    best_f = f_new
                    best_block = new_block.copy()
                    v_block = new_block.copy()

            # Apply update
            if np.any(best_block != v_block):
                a[idx] = best_block
                conv = compute_conv(a)
                S_now = float(np.sum(a))
                if S_now > 1e-10:
                    obj = 2.0 * n * float(np.max(conv)) / (S_now * S_now)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    BlockCD: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def _block_objective(block_vals, a_full, idx, start_pos, n, S_other):
    """Compute objective for a block update."""
    a_try = a_full.copy()
    a_try[idx] = block_vals
    S_try = S_other + float(np.sum(block_vals))
    if S_try < 1e-10:
        return 1e30
    conv = compute_conv(a_try)
    return 2.0 * n * float(np.max(conv)) / (S_try * S_try)


def active_set_lp(a, time_budget=100.0, max_iter=20, verbose=False):
    """Iterative active set LP: grow active set, solve LP, blend."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - start >= time_budget:
            break
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        # Active set: all lags within 1% of max
        threshold = max_conv * 0.995
        active = np.where(conv >= threshold)[0]

        if len(active) < 3:
            active = np.argsort(conv)[-10:]

        g_lp, M_lp = solve_lp_active(a, active, all_indices=np.arange(2 * n - 1))
        if g_lp is None:
            break

        # Try blending
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose:
                print(f"    AS-LP it{it}: obj={best_obj:.8f} (active={len(active)})")
        else:
            # Try with wider active set
            threshold = max_conv * 0.95
            active_wide = np.where(conv >= threshold)[0]
            if len(active_wide) > len(active):
                g_lp2, _ = solve_lp_active(a, active_wide, all_indices=np.arange(2 * n - 1))
                if g_lp2 is not None:
                    for t in [0.1, 0.2, 0.3, 0.5]:
                        a_try = (1.0 - t) * a + t * g_lp2
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            break
                    else:
                        break
                else:
                    break
            else:
                break

    return best_a


def simulated_annealing(a, time_budget=200.0, verbose=False, seed=42):
    """Simulated annealing with block perturbations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T_init = 0.001
    T_end = 1e-7
    block_sizes = [2, 3, 4, 5, 7, 10]

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        T = T_init * (T_end / T_init) ** progress

        b = int(rng.choice(block_sizes))
        b = min(b, n)
        s_pos = int(rng.integers(0, n - b + 1))
        idx = np.arange(s_pos, s_pos + b)

        v = a[idx].copy()
        S_other = float(np.sum(a)) - float(np.sum(v))

        # Perturb the block
        scale = max(float(np.mean(v)), 1.0 / n)
        if b <= 4:
            pert = rng.normal(0, 0.1 * scale, size=b)
        else:
            pert = rng.normal(0, 0.05 * scale, size=b)
        new_block = np.maximum(v + pert, 0)

        # Compute new objective
        a_try = a.copy()
        a_try[idx] = new_block
        S_try = S_other + float(np.sum(new_block))
        if S_try < 1e-10:
            continue
        conv_try = compute_conv(a_try)
        obj_try = 2.0 * n * float(np.max(conv_try)) / (S_try * S_try)

        delta = obj_try - current_obj
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-15)):
            a = a_try
            current_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()

        if verbose and int(elapsed) % 30 == 0:
            print(f"    SA: T={T:.2e}, obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def spectral_reshape(a, time_budget=60.0, verbose=False, seed=42):
    """Spectral peak-shaping: iteratively adjust spectrum to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    N = 1
    while N < 2 * n:
        N <<= 1

    alpha = 0.5
    rng = np.random.default_rng(seed)

    for it in range(1000):
        if time.monotonic() - start >= time_budget:
            break

        A = fft(a, N)
        power = np.abs(A) ** 2
        c = irfft(fft(power, N) * np.conj(fft(a, N)), N)
        # Actually, the autocorrelation in freq domain: IFFT(|A|^2) gives circular conv
        # Let's use the direct relation: c[k] corresponds to IFFT of |A|^2
        c_full = np.fft.ifft(np.abs(A) ** 2).real
        c = c_full[:2 * n - 1]
        # Reorder to get linear autocorrelation
        c_lin = np.concatenate([c_full[n - 1:], c_full[:n - 1][::-1]]) if n > 1 else c_full[:1]

        max_c = float(np.max(c_full[:2 * n - 1]))
        # Damp peaks in power spectrum
        peak_mask = c_full[:2 * n - 1] > max_c * 0.9
        # Reduce power at frequencies contributing to peaks
        A_damped = A.copy()
        A_damped[peak_mask[:N]] *= (1.0 - alpha * 0.1)

        a_new = np.fft.ifft(A_damped).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            alpha = max(alpha * 0.99, 0.05)
        else:
            # Add noise to escape
            pert = rng.normal(0, 0.01 * alpha, n)
            a_pert = a * np.exp(pert)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
            a = a_pert

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def get_parent_sequence():
    """Try to load the parent sequence from global variable."""
    try:
        if 'height_sequence_1' in globals():
            arr = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            arr = np.clip(arr, 0, None)
            if arr.size > 0 and np.isfinite(arr).all():
                s = float(np.sum(arr))
                if s > 0.01:
                    return arr / s
    except Exception:
        pass
    return None


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Gaussian bump
        x = np.arange(n) / n
        a = np.exp(-4 * (x - 0.5) ** 2)
        a = a / a.sum()
    elif mode == 3:
        # Beta-like distribution
        x = np.linspace(0.01, 0.99, n)
        a = x ** 0.5 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        # Triangular
        x = np.arange(n) / n
        a = 1 - 2 * np.abs(x - 0.5)
        a = a / a.sum()
    elif mode == 5:
        # Sine squared
        x = np.arange(n) / n
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        a = np.zeros(n)
        w = n // 3
        center1 = n // 4
        center2 = 3 * n // 4
        a[center1 - w // 2:center1 + w // 2] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a[center2 - w // 2:center2 + w // 2] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining CD, block CD, active-set LP,
    simulated annealing, spectral reshaping, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    rng = np.random.default_rng(seed)

    # Load parent
    parent = get_parent_sequence()
    if parent is not None:
        best_seq = parent.copy()
        best_score = compute_obj(best_seq)
        n_best = len(best_seq)
        print(f"Parent (n={n_best}): obj={best_score:.8f}")
    else:
        n_best = 800
        best_seq = np.ones(n_best) / n_best
        best_score = 2.0
        print(f"Default init (n={n_best}): obj={best_score:.8f}")

    # === Phase 1: Initial CD on parent ===
    print(f"\n=== Phase 1: CD on parent (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 250.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 2: Block CD on best ===
    print(f"\n=== Phase 2: Block CD (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 120.0)
    if p2_budget > 20:
        a_bc = block_cd(best_seq, time_budget=p2_budget, block_size=4, verbose=False, seed=seed + 1)
        obj_bc = compute_obj(a_bc)
        if obj_bc < best_score:
            best_score = obj_bc
            best_seq = a_bc.copy()
            print(f"  BlockCD: obj={best_score:.8f} **")
        else:
            print(f"  BlockCD: obj={obj_bc:.8f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_candidates = sorted(set([
        n_cur - 100, n_cur - 50, n_cur + 50, n_cur + 100, n_cur + 200,
        n_cur - 200, n_cur + 300,
        700, 750, 850, 900, 950, 1000, 1100
    ]))
    n_filtered = []
    for nc in n_candidates:
        if 500 <= nc <= 1300:
            if all(abs(nc - nf) > 30 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:8]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.08, 40.0)

        if alloc > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Active Set LP ===
    print(f"\n=== Phase 4: Active Set LP ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 20:
        a_lp = active_set_lp(best_seq, time_budget=p4_budget, max_iter=30, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  AS-LP: obj={best_score:.8f} **")
        else:
            print(f"  AS-LP: obj={obj_lp:.8f}")

        # Follow-up CD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_cd2 = coordinate_descent(best_seq, time_budget=min(60, remaining * 0.1),
                                       verbose=False, seed=seed + 10)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD after LP: obj={best_score:.8f} **")

    # === Phase 5: Simulated Annealing ===
    print(f"\n=== Phase 5: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 200.0)
    if p5_budget > 30:
        a_sa = simulated_annealing(best_seq, time_budget=p5_budget, verbose=False, seed=seed + 20)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # === Phase 6: Diverse restarts with CD ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r % 7
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.4 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.1, 50.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False, seed=5000 + r * 100)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 7: Spectral reshaping ===
    print(f"\n=== Phase 7: Spectral Reshaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_spec = spectral_reshape(best_seq, time_budget=min(remaining * 0.1, 80.0), seed=seed + 30)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Spectral: obj={obj_spec:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Round 1: CD
        cd_alloc = min(remaining * 0.3, 150.0)
        if cd_alloc > 15:
            a_f = coordinate_descent(best_seq, time_budget=cd_alloc, verbose=True, seed=9000)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        # Round 2: LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            lp_alloc = min(remaining * 0.2, 60.0)
            if lp_alloc > 10:
                a_f2 = active_set_lp(best_seq, time_budget=lp_alloc, max_iter=15, verbose=False)
                obj_f2 = compute_obj(a_f2)
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  Final LP: obj={best_score:.8f} **")

        # Round 3: Block CD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            bc_alloc = min(remaining * 0.2, 60.0)
            if bc_alloc > 10:
                a_f3 = block_cd(best_seq, time_budget=bc_alloc, block_size=5, verbose=False, seed=9500)
                obj_f3 = compute_obj(a_f3)
                if obj_f3 < best_score:
                    best_score = obj_f3
                    best_seq = a_f3.copy()
                    print(f"  Final BlockCD: obj={best_score:.8f} **")

        # Round 4: Final CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            cd_alloc2 = min(remaining * 0.4, 80.0)
            if cd_alloc2 > 10:
                a_f4 = coordinate_descent(best_seq, time_budget=cd_alloc2, verbose=False, seed=9999)
                obj_f4 = compute_obj(a_f4)
                if obj_f4 < best_score:
                    best_score = obj_f4
                    best_seq = a_f4.copy()
                    print(f"  Final CD2: obj={best_score:.8f} **")

    # === Phase 9: Try a few more n values with best structure ===
    print(f"\n=== Phase 9: Final n exploration ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    n_final_candidates = [n_final - 25, n_final + 25, n_final - 50, n_final + 50,
                          n_final - 75, n_final + 75]
    n_final_candidates = [nc for nc in set(n_final_candidates) if 500 <= nc <= 1400]

    for n_try in n_final_candidates[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 30.0)
        if alloc > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False, seed=n_try + 100)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Final n={n_try}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
