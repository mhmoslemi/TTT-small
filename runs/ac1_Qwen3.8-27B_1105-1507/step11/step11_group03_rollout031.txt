def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

    n = len(a)

        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def gradient_descent_smoothed(a, time_budget, tau_levels=None, verbose=False):
    """Nesterov accelerated gradient descent with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    beta = 0.9
    
    for tau in tau_levels:
        lr = 1.0 / np.sqrt(n)
        v = np.zeros(n)
        
        while time.monotonic() - start < time_budget:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            # Compute smoothed gradient
            conv = compute_conv(y)
            exact_max = float(np.max(conv))
            shifted = tau * (conv - exact_max)
            shifted = np.clip(shifted, -700, 700)
            exp_vals = np.exp(shifted)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp
            
            # Gradient of smooth max of conv(y,y) w.r.t. y
            # d/dy_i sum_k p_k * conv(y,y)[k] = 2 * sum_k p_k * conv(e_i, y)[k] (approximately)
            # Actually: conv(y,y)[k] = sum_j y[j]*y[k-j]
            # d/dy_i conv(y,y)[k] = y[k-i] + y[i-k] (for valid indices)
            # So gradient[i] = sum_k p_k * (y[k-i] + y[i-k]) for valid k
            # This is 2 * (p * y) circular-like but for linear conv
            # More precisely: grad[i] = sum_k p[k] * (y[k-i] (if valid) + y[i-k] (if valid))
            
            N = 1
            while N < 2 * n:
                N <<= 1
            P = fft(p, N)
            Y = fft(y, N)
            cross = ifft(P * Y).real  # This gives sum_k p[k]*y[k-i] (circular)
            # For linear conv, we need to be more careful

            # = sum_k p[k] * y[k-i] (for valid k-i) + sum_k p[k] * y[i-k] (for valid i-k)
            # The first term is (p * y)[i] (cross-correlation)
            # The second term is (p * y)[i] as well (by symmetry of the sum)
            # Actually for linear conv: grad[i] = 2 * sum_k p[k] * y[k-i] (clamped)
            
            # Use FFT for efficiency (circular approximation, good for large n)
            grad = 2.0 * cross[:n]
            
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break
            
            v = beta * v - lr * grad / gn
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
                continue
            a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP with warm starts, adaptive step refinement, diverse perturbations,
multi-n exploration, random restarts, and smoothed gradient polishing.

import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """Build constraint matrix for LP: row i corresponds to lag lags[i].
    A[i, j] = a[lag_i - j] if 0 <= lag_i - j < n, else 0.
    So that A @ g gives conv(a, g) at the specified lags.
    """
    n = len(a)
    m = len(lags)
    A = np.zeros((m, n))
    for i, lag in enumerate(lags):
        # conv(a,g)[lag] = sum_j a[j]*g[lag-j]
        # coeff of g[j] is a[lag-j] if valid
        for j in range(n):
            a_idx = lag - j
            if 0 <= a_idx < n:
                A[i, j] = a[a_idx]
    return A


def lp_solve(a, lags, warm_start=None):
    """Solve min M s.t. conv(a,g)[lag] <= M for lag in lags, g>=0, sum(g)=1."""
    n = len(a)
    m = len(lags)
    if m == 0:
        return None, None
    
    A = build_conv_matrix(a, lags)
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    
    if warm_start is not None:
        g_var.value = warm_start
        M_var.value = float(np.max(A @ warm_start))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    
    return None, None


def lp_active(a, top_k=15, thresh_frac=0.99, warm_start=None):
    """Solve LP with active set (top lags of conv(a,a))."""
    n = len(a)
    conv_aa = compute_conv(a)
    max_val = float(np.max(conv_aa))
    
    # Get lags above threshold
    thresh = max_val * thresh_frac
    active_lags = np.where(conv_aa >= thresh)[0]
    
    # Ensure we have at least 5 and at most top_k*2 constraints
    if len(active_lags) < 5:
        top_idx = np.argsort(conv_aa)[::-1][:top_k]
        active_lags = top_idx
    elif len(active_lags) > top_k * 2:
        # Keep only the top top_k*2
        sorted_lags = active_lags[np.argsort(conv_aa[active_lags])[::-1][:top_k * 2]]
        active_lags = sorted_lags
    
    active_lags = sorted(set(active_lags.astype(int)))
    return lp_solve(a, active_lags, warm_start=warm_start)


def adaptive_line_search(a, direction, best_obj, n_tries=20):
    """Find best step size along direction using adaptive strategy."""
    n = len(a)
    best_t = 0.0
    best_a = a.copy()
    best_local = best_obj
    
    # Coarse search
    candidates = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 1.0]
    
    for t in candidates:
        a_try = a + t * direction
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            continue
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_local - 1e-13:
            best_local = obj_try
            best_t = t
            best_a = a_try.copy()
    
    if best_t > 0:
        # Fine search around best_t
        t_lo = max(0.005, best_t * 0.6)
        t_hi = min(1.0, best_t * 1.5)
        fine_candidates = np.linspace(t_lo, t_hi, n_tries - len(candidates) + 1)
        for t in fine_candidates:
            if t <= 0:
                continue
            a_try = a + t * direction
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local - 1e-13:
                best_local = obj_try
                best_t = t
                best_a = a_try.copy()
    
    return best_a, best_local, best_t


def perturb_strategy(a, rng, level=1):
    """Apply diverse perturbations, return best perturbed version."""
    n = len(a)
    x = np.arange(n)
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    
    scale = 0.008 * (1.0 + 0.3 * level)
    
    # Strategy 1: Low-frequency sinusoidal
    for trial in range(2):
        f = rng.integers(1, 6)
        p = rng.uniform(0, 2 * np.pi)
        a_try = a + scale * np.sin(2 * np.pi * f * x / n + p)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            a_try = a_try / s
            o = compute_obj(a_try)
            if o < best_obj:
                best_obj = o
                best_a = a_try.copy()
    
    # Strategy 2: Mid-frequency
    for trial in range(2):
        f = rng.integers(n // 8, n // 2 + 1)
        p = rng.uniform(0, 2 * np.pi)
        a_try = a + scale * 0.7 * np.sin(2 * np.pi * f * x / n + p)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            a_try = a_try / s
            o = compute_obj(a_try)
            if o < best_obj:
                best_obj = o
                best_a = a_try.copy()
    
    # Strategy 3: Random additive
    a_try = a + scale * rng.standard_normal(n)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s > 1e-10:
        a_try = a_try / s
        o = compute_obj(a_try)
        if o < best_obj:
            best_obj = o
            best_a = a_try.copy()
    
    # Strategy 4: Multiplicative (log-normal)
    a_try = a * np.exp(scale * 0.5 * rng.standard_normal(n))
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s > 1e-10:
        a_try = a_try / s
        o = compute_obj(a_try)
        if o < best_obj:
            best_obj = o
            best_a = a_try.copy()
    
    # Strategy 5: Block modification
    block = rng.integers(n // 8, n // 3)
    start = rng.integers(0, n - block)
    a_try = a.copy()
    a_try[start:start + block] *= (1.0 + scale * rng.standard_normal())
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s > 1e-10:
        a_try = a_try / s
        o = compute_obj(a_try)
        if o < best_obj:
            best_obj = o
            best_a = a_try.copy()
    
    return best_a, best_obj


def lp_fixed_point(a, time_budget, verbose=False, seed=42):
    """
    Main LP fixed-point iteration with:
    - Active set LP (adaptive number of constraints)
    - Warm starts
    - Adaptive line search
    - Diverse perturbations on stall
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall_count = 0
    it = 0
    prev_g = None
    top_k = 12
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        it += 1
        
        # Gradually increase active set size
        top_k = min(30, 10 + it // 15)
        thresh = max(0.99, 1.0 - 0.01 * min(1.0, it / 50.0))
        
        g_lp, M_lp = lp_active(a, top_k=top_k, thresh_frac=thresh, warm_start=prev_g)
        
        if g_lp is None:
            stall_count += 2
            # Try with fewer constraints
            g_lp, M_lp = lp_active(a, top_k=max(5, top_k // 2), thresh_frac=0.98)
            if g_lp is None:
                continue
        prev_g = g_lp.copy()
        
        direction = g_lp - a
        a_new, obj_new, t_used = adaptive_line_search(a, direction, best_obj)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and it % 30 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it:4d}: obj={best_obj:.10f}, t={t_used:.4f}, M={M_lp:.8f}, k={top_k}, time={elapsed:.0f}s")
        else:
            stall_count += 1
            if stall_count >= 4:
                a, obj_p = perturb_strategy(a, rng, level=stall_count // 4)
                if obj_p <= best_obj + 0.001:
                    a = a.copy()
                else:
                    a = best_a.copy()
                stall_count = 0
                if verbose:
                    print(f"    Perturb it{it}: a_obj={compute_obj(a):.10f}")
            elif stall_count >= 2:
                # Small nudge
                a = a + 0.003 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a


def smoothed_gd(a, time_budget, tau_levels=None, verbose=False):
    """Nesterov accelerated gradient descent with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000]
    
    beta = 0.9
    N = 1
    while N < 2 * n:
        N <<= 1
    
    for tau in tau_levels:
        lr = 0.8 / np.sqrt(n)
        v = np.zeros(n)
        stagnant = 0
        
        while time.monotonic() - start < time_budget:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                continue
            y = y / sy
            
            conv = compute_conv(y)
            exact_max = float(np.max(conv))
            shifted = tau * (conv - exact_max)
            shifted = np.clip(shifted, -700, 700)
            exp_vals = np.exp(shifted)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp
            
            # Gradient via FFT cross-correlation
            P = fft(p, N)
            Y = fft(y, N)
            cross = ifft(P * Y).real
            grad = 2.0 * cross[:n]
            
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break
            
            v = beta * v - lr * grad / gn
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
                continue
            a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                stagnant = 0
            else:
                stagnant += 1
            
            a = a_new
            lr = max(1e-9, lr * 0.99997)
            
            if stagnant > 500:
                break
        
        if verbose:
            print(f"    tau={tau:5d}: obj={best_obj:.10f}")
    
    return best_a


def active_set_polish(a, time_budget, verbose=False, seed=42):
    """Targeted reduction of peak lags using subgradient + LP hybrid."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        it += 1
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Get top lags
        top_lags = np.argsort(conv)[::-1][:min(15, 2 * n - 1)]
        
        # Compute subgradient at active lags (weighted by proximity to max)
        weights = np.zeros(len(conv))
        for lag in top_lags:
            w = conv[lag] / max_val
            weights[lag] = w ** 4  # Sharpen weights toward the peaks
        
        subgrad = np.zeros(n)
        for lag in top_lags:
            w = weights[lag]
            if w < 1e-10:
                continue
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    subgrad[i] += w * 2.0 * a[j]
        
        wsum = float(np.sum(weights[top_lags]))
        if wsum > 1e-15:
            subgrad /= wsum
        
        gn = np.linalg.norm(subgrad)
        if gn < 1e-15:
            break
        subgrad = subgrad / gn
        
        improved = False
        # Try subgradient descent
        for step in [0.0003, 0.0008, 0.002, 0.005, 0.01, 0.02]:
            a_try = a - step * subgrad
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            # Try LP direction
            g_lp, _ = lp_active(a, top_k=10, thresh_frac=0.995)
            if g_lp is not None:
                for t in [0.03, 0.08, 0.15, 0.25, 0.4]:
                    a_try = (1 - t) * a + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    if a_try.sum() < 1e-10:
                        continue
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
            
            if not improved:
                # Perturb
                a, _ = perturb_strategy(a, rng, level=1)
                
                if it > 30 and it % 20 == 0:
                    if compute_obj(a) > best_obj * 1.005:
                        a = best_a.copy()
        
        if verbose and it % 50 == 0:
            print(f"    Polish it{it}: obj={best_obj:.10f}")
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.005)
        return a / a.sum()
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(-2.5, 2.5, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 5:
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        taper = max(1, int(0.15 * n))
        a = np.ones(n)
        mask = d < taper
        a[mask] = 0.5 * (1 + np.cos(np.pi * d[mask] / taper))
        return a / a.sum()
    elif mode == 6:
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    else:
        a = rng.uniform(0.2, 1.0, n)
        return a / a.sum()
