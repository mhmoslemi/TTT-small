import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_conv(a_norm):
    """Compute 2n*max(conv) for normalized a (sum=1)."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

    weights = weights / w_sum

    # d conv[k] / da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n, i != k-i)

        g = 0.0
        # Term 1: k - i = j, 0 <= j < n => k = i + j, 0 <= j < n
        # conv[i+j] weight * a[j]

        if lo <= hi:

            g += np.dot(weights[i + k_range], a[k_range])
        # Term 2: i - k = j, 0 <= j < n => k = i - j, 0 <= j < n, k >= 0

        if lo2 <= hi2:
            j_range = np.arange(lo2, hi2 + 1)
            g += np.dot(weights[i - j_range], a[j_range])
        
        grad[i] = g

    obj_val = np.sum(np.maximum(conv_masked, eps) ** p) ** (1.0 / p)
    scale = p * obj_val ** (1 - p)

    grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])

- Multi-scale n exploration
Key improvements: adaptive p-schedule, full-constraint LP (not truncated),
swap-based local search, and structured spectral perturbations for restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_conv(a_norm):
    """Compute 2n*max(conv) for normalized a (sum=1)."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def compute_grad_fft(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv(a,a)[k].
    grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) with valid index handling.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv_len = 2 * n - 1

    size = 1
    while size < conv_len + n - 1:
        size <<= 1

    a_pad = np.zeros(size)
    a_pad[:n] = a

    w_pad = np.zeros(size)
    w_pad[:conv_len] = weights

    w_rev = np.zeros(size)
    w_rev[:conv_len] = weights[::-1]

    A = rfft(a_pad, size)
    W = rfft(w_pad, size)
    W_rev = rfft(w_rev, size)

    # term2[i] = conv(a_pad, w_pad)[i] for i=0..n-1
    term2 = irfft(A * W, size)[:n]

    # term1[i] = conv(a_pad, w_rev)[2n-2-i] for i=0..n-1
    conv_aw = irfft(A * W_rev, size)
    term1 = conv_aw[n - 1:2 * n - 1][::-1]

    return term1 + term2


def lp_smoothed_gradient(a, p, top_k=None):
    """Compute gradient of (sum_{top peaks} conv[k]^p)^{1/p} at a."""
    n = len(a)
    conv = compute_conv(a)

    if top_k is not None and top_k < len(conv):
        top_idx = np.argsort(conv)[::-1][:top_k]
        w = np.zeros_like(conv)
        w[top_idx] = np.maximum(conv[top_idx], 1e-30) ** (p - 1)
    else:
        w = np.maximum(conv, 1e-30) ** (p - 1)

    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum

    grad = compute_grad_fft(a, w)
    obj_val = np.sum(w * conv) ** (1.0 / p) if np.sum(w * conv) > 0 else 1.0
    # Actually the objective is (sum conv[k]^p)^{1/p} for active peaks
    # Gradient = (1/p) * (sum conv^p)^{1/p - 1} * p * conv^{p-1} * dconv/da
    # = (sum conv^p)^{1/p - 1} * sum_k conv[k]^{p-1} * dconv[k]/da[i]
    # With normalized weights w[k] = conv[k]^{p-1} / sum(conv^{p-1}),
    # grad_unscaled = compute_grad_fft(a, w)
    # True grad = (sum conv^p)^{1/p-1} * sum_k conv[k]^{p-1} * dconv/da
    # = (sum conv^p)^{1/p-1} * sum_k conv[k]^{p-1} * dconv/da
    # = (sum conv^p)^{1/p-1} * sum_k (sum(conv^{p-1})) * w[k] * dconv/da
    # = (sum conv^p)^{1/p-1} * sum(conv^{p-1}) * grad_unscaled

    conv_powers = np.where(w > 0, conv, 0)
    sum_conv_p = np.sum(conv_powers ** p)
    sum_conv_pm1 = np.sum(conv_powers ** (p - 1))
    if sum_conv_p > 1e-30:
        scale = sum_conv_p ** (1.0 / p - 1.0) * sum_conv_pm1
    else:
        scale = 1.0

    return scale * grad


def lp_gradient_descent(a, time_budget, p_start=4.0, p_end=80.0, top_k=None,
                        lr_init=None, verbose=False):
    """
    Lp-norm smoothed gradient descent with Nesterov momentum and backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()

    if lr_init is None:
        lr = 0.02 / np.sqrt(n)
    else:
        lr = lr_init

    beta = 0.8
    v = np.zeros(n)  # momentum
    it = 0
    best_it = 0
    patience = max(200, int(time_budget * 10))
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        p = p_start * (p_end / p_start) ** frac
        # Decay momentum over time
        beta_current = beta * (1.0 - 0.5 * frac)

        grad = lp_smoothed_gradient(a, p, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Nesterov momentum update
        v = beta_current * v - lr * grad
        a_new = project_simplex(a + v)

        # Backtracking line search
        obj_new = compute_obj_conv(a_new)
        if obj_new >= best_obj * (1 - 1e-10):
            improved = False
            for shrink in [0.5, 0.25, 0.125, 0.0625]:
                v_try = beta_current * v - lr * shrink * grad
                a_try = project_simplex(a + v_try)
                obj_try = compute_obj_conv(a_try)
                if obj_try < best_obj * (1 - 1e-10):
                    a = a_try
                    v = v_try
                    improved = True
                    break
            if not improved:
                lr = lr * 0.5
                v *= 0.5
                no_improve_count += 1
            else:
                no_improve_count = 0
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a.copy()
                    best_it = it
                    lr = min(lr * 1.1, 0.5 / np.sqrt(n))
        else:
            a = a_new
            v = v
            no_improve_count = 0
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                best_it = it
                lr = min(lr * 1.05, 0.5 / np.sqrt(n))
            else:
                # Moved but not better than best; keep going
                pass

        it += 1

        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = max(lr * 0.3, 1e-12)
            patience = max(50, patience // 2)
            best_it = it
            no_improve_count = 0

        if no_improve_count > 20:
            # Restart with perturbation
            break

        if verbose and it % 500 == 0:
            print(f"      LpGD it{it}: obj={best_obj:.8f}, p={p:.1f}, "
                  f"lr={lr:.2e}, beta={beta_current:.2f}, t={time.monotonic()-start:.1f}s")

    return best_a


def lp_linearization_solve(a, use_all_constraints=True, top_m=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k (or top_m), sum(g)=1, g>=0.
    Returns optimal g.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_len = 2 * n - 1

    if use_all_constraints or top_m is None:
        num_constr = conv_len
    else:
        conv_a = compute_conv(a_norm)
        num_constr = min(top_m, conv_len)
        top_idx = np.argsort(conv_a)[::-1][:num_constr]

    # Build constraint matrix
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid indices
    # Coefficient of g[m] in (a*g)[k] is a[k-m] if 0 <= k-m < n
    A = np.zeros((num_constr, n))

    if use_all_constraints or top_m is None:
        for k in range(conv_len):
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    A[k, m] = a_norm[j]
    else:
        for idx, k in enumerate(top_idx):
            k = int(k)
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    A[idx, m] = a_norm[j]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value) if M.value is not None else np.inf
        except Exception:
            continue
    return None, np.inf


def lp_iterative_polish(a, time_budget, verbose=False):
    """
    Iterative LP polishing: linearize at current a, solve, blend, repeat.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    it = 0
    stall = 0

    # Start with fewer constraints, increase over time
    max_top_m = min(2 * n - 1, 400)

    while time.monotonic() - start < time_budget and stall < 8:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        frac = min(1.0, it / 50.0)
        # Use all constraints if n is small, else top_m
        if n <= 300:
            top_m = None
            use_all = True
        else:
            top_m = int(max_top_m * (0.3 + 0.7 * frac))
            use_all = False

        g_lp, m_lp = lp_linearization_solve(best_a, use_all_constraints=use_all, top_m=top_m)
        if g_lp is None:
            stall += 2
            it += 1
            continue

        # Find best blend
        best_t = 0.0
        best_blend_obj = best_obj
        for t in np.concatenate([np.linspace(0.1, 1.0, 10), np.linspace(0.01, 0.1, 5)]):
            a_try = (1 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj_conv(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        if best_t > 0 and best_blend_obj < best_obj - 1e-12:
            a_new = (1 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj_conv(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                  f"top_m={top_m}, elapsed={time.monotonic()-start:.1f}s")

        # Perturb if stalled
        if stall in (3, 5, 7):
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.normal(0, 0.002 * stall, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj_conv(best_a)
            stall = 0

    return best_a


def coordinate_descent(a, time_budget, max_sweeps=10, verbose=False):
    """Coordinate descent: try scaling each coordinate and pick best."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        sweep_improved = False

        # Randomize coordinate order for better mixing
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            ai_old = a[i]
            if ai_old < 1e-12:
                continue

            # Try several multipliers
            best_val = ai_old
            best_local = best_obj

            for mult in [0.0, 0.3, 0.6, 0.8, 1.2, 1.5, 2.0]:
                cv = ai_old * mult
                a[i] = cv
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    continue
                a_try = a / s_new
                obj_try = compute_obj_conv(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv

            a[i] = best_val
            if best_val != ai_old:
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    sweep_improved = True

                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()

        if verbose:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        if not sweep_improved:
            break

    return best_a


def swap_based_local_search(a, time_budget, verbose=False):
    """
    Local search that tries to reduce the top peaks by swapping mass
    between coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:3]
        peak_idx = int(top_k[0])

        # For the peak at peak_idx, find which coordinate to reduce
        # and which to increase
        # Contribution of a[i] to conv[peak_idx]:
        # a[peak_idx - i] (if 0 <= peak_idx-i < n) + a[i - peak_idx] (if 0 <= i-peak_idx < n)
        contrib = np.zeros(n)
        for i in range(n):
            c = 0.0
            j1 = peak_idx - i
            if 0 <= j1 < n:
                c += a[j1]
            j2 = i - peak_idx
            if 0 <= j2 < n:
                c += a[j2]
            contrib[i] = c

        # Try to move mass from high-contribution to low-contribution positions
        # But also check effect on other peaks
        high_contrib = np.argsort(contrib)[::-1]
        low_contrib = np.argsort(contrib)

        improved = False
        for _ in range(20):
            # Pick a high-contribution source and low-contribution target
            src = high_contrib[rng.integers(0, min(50, n))]
            dst = low_contrib[rng.integers(0, min(50, n))]
            if src == dst or a[src] < 1e-8:
                continue

            # Try different amounts of transfer
            for frac in [0.1, 0.3, 0.5]:
                delta = a[src] * frac
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj_conv(a_try)
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        it += 1
        if not improved and it % 50 == 0:
            # Perturb
            idx = rng.integers(0, n)
            a[idx] += rng.normal(0, 0.005)
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and it % 100 == 0:
            print(f"      SwapLS it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def diverse_init(n, rng, mode):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    t = 2 * x - 1  # in [-1, 1]

    if mode == 0:
        # U-shape (inverse quadratic)
        a = 1.0 / (0.01 + t ** 2)
    elif mode == 1:
        # Edge cosine
        a = np.cos(np.pi * t / 2) ** 4 + 0.01
    elif mode == 2:
        # Power law edge
        a = np.maximum(np.abs(t), 1.0 / n) ** (-1.5) + 0.01
    elif mode == 3:
        # Two-sided Gaussian at edges
        a = np.exp(-((idx - 0) ** 2) / (2 * (n / 8) ** 2)) + \
            np.exp(-((idx - (n - 1)) ** 2) / (2 * (n / 8) ** 2)) + 0.01
    elif mode == 4:
        # Periodic edge concentration
        a = 1.0 / (0.02 + (np.sin(np.pi * x) ** 2) ** 3)
    elif mode == 5:
        # Sine power
        a = (np.abs(np.sin(np.pi * x))) ** 6 + 0.005
    elif mode == 6:
        # Arcsine-like
        a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - t ** 2 + 0.01)) + 0.01
    elif mode == 7:
        # Mixed: strong edges + weak center
        a = np.exp(-8 * t ** 2) + 0.3 * np.exp(-0.5 * t ** 2) + 0.01
    elif mode == 8:
        # Polynomial edge
        a = (1 + t ** 2) + 0.1 * (1 + t ** 4)
    else:
        # Random smooth
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        ampls = rng.uniform(0.5, 1.0, size=4)
        a = 0.1
        for f, ph, amp in zip(freqs, phases, ampls):
            a += amp * (1 + np.cos(2 * np.pi * f * x + ph)) / 2

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_n_change(best_seq, n_target, time_budget):
    """Try a different sequence length via interpolation + quick optimization."""
    n_old = len(best_seq)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_target)
    a_try = np.interp(x_new, x_old, best_seq)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick Lp gradient descent
    t1 = min(20.0, time_budget * 0.4)
    a_try = lp_gradient_descent(a_try, time_budget=t1, p_start=6.0, p_end=40.0)

    # Quick LP
    t2 = min(20.0, time_budget * 0.4)
    if t2 > 5:
        a_try = lp_iterative_polish(a_try, time_budget=t2)

    obj_try = compute_obj_conv(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method optimization for step function coefficients.
    Combines Lp-smoothed gradient descent, LP linearization polishing,
    swap-based local search, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # === Load parent if available ===
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj_conv(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Loaded parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("Could not load parent, generating from scratch")

    # === Initialization phase: try many diverse starts ===
    print("\n=== Initialization ===")
    n_default = 800

    if best_seq is None:
        best_seq = diverse_init(n_default, rng, mode=0)
        best_score = compute_obj_conv(best_seq)

    n_fixed = len(best_seq)

    # Try diverse initializations at current n
    inits_to_try = [2, 3, 4, 5, 6, 7, 8]
    for mode in inits_to_try:
        a_init = diverse_init(n_fixed, rng, mode)
        obj_init = compute_obj_conv(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init mode={mode}: obj={obj_init:.8f} **")
        else:
            print(f"  Init mode={mode}: obj={obj_init:.8f}")

    print(f"Best init: obj={best_score:.8f}, n={n_fixed}")

    # === Phase 1: Lp Gradient Descent (global) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: Lp Gradient Descent (budget={min(120, remaining*0.15):.0f}s) ===")
        alloc = min(120.0, remaining * 0.15)
        a_p1 = lp_gradient_descent(best_seq, time_budget=alloc,
                                    p_start=4.0, p_end=60.0, top_k=10, verbose=True)
        obj_p1 = compute_obj_conv(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 result: obj={obj_p1:.8f}")

    # === Phase 2: LP Polishing (main workhorse) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 2: LP Polishing (budget={min(400, remaining*0.35):.0f}s) ===")
        alloc = min(400.0, remaining * 0.35)
        a_p2 = lp_iterative_polish(best_seq, time_budget=alloc, verbose=True)
        obj_p2 = compute_obj_conv(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  Phase 2 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 result: obj={obj_p2:.8f}")

    # === Phase 3: Lp GD Round 2 (refinement) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: Lp GD Refinement (budget={min(80, remaining*0.1):.0f}s) ===")
        alloc = min(80.0, remaining * 0.1)
        a_p3 = lp_gradient_descent(best_seq, time_budget=alloc,
                                    p_start=20.0, p_end=200.0, top_k=5,
                                    lr_init=0.01 / np.sqrt(len(best_seq)), verbose=False)
        obj_p3 = compute_obj_conv(a_p3)
        if obj_p3 < best_score:
            best_score = obj_p3
            best_seq = a_p3.copy()
            print(f"  Phase 3 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.8f}")

    # === Phase 4: LP Polishing Round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Polish Round 2 (budget={min(200, remaining*0.2):.0f}s) ===")
        alloc = min(200.0, remaining * 0.2)
        a_p4 = lp_iterative_polish(best_seq, time_budget=alloc, verbose=True)
        obj_p4 = compute_obj_conv(a_p4)
        if obj_p4 < best_score:
            best_score = obj_p4
            best_seq = a_p4.copy()
            print(f"  Phase 4 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 result: obj={obj_p4:.8f}")

    # === Phase 5: Swap-based local search ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Swap Local Search (budget={min(100, remaining*0.1):.0f}s) ===")
        alloc = min(100.0, remaining * 0.1)
        a_p5 = swap_based_local_search(best_seq, time_budget=alloc, verbose=True)
        obj_p5 = compute_obj_conv(a_p5)
        if obj_p5 < best_score:
            best_score = obj_p5
            best_seq = a_p5.copy()
            print(f"  Phase 5 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5 result: obj={obj_p5:.8f}")

    # === Phase 6: Coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Coordinate Descent (budget={min(80, remaining*0.1):.0f}s) ===")
        alloc = min(80.0, remaining * 0.1)
        a_p6 = coordinate_descent(best_seq, time_budget=alloc, max_sweeps=5, verbose=True)
        obj_p6 = compute_obj_conv(a_p6)
        if obj_p6 < best_score:
            best_score = obj_p6
            best_seq = a_p6.copy()
            print(f"  Phase 6 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6 result: obj={obj_p6:.8f}")

    # === Phase 7: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 7: n Exploration ===")
        n_best = len(best_seq)
        # Try nearby n values
        n_options = []
        for delta in [-100, -50, 50, 100, 150, 200, -200, 300]:
            n_try = n_best + delta
            if 400 <= n_try <= 2500 and n_try not in n_options:
                n_options.append(n_try)
        # Also try some specific values
        for n_specific in [600, 700, 900, 1000, 1200]:
            if n_specific not in n_options and 400 <= n_specific <= 2500:
                n_options.append(n_specific)

        print(f"  Candidate n values: {n_options[:8]}")

        for n_try in n_options[:6]:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            t_alloc = min(60.0, remaining * 0.4)
            a_try, obj_try = try_n_change(best_seq, n_try, t_alloc)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 8: Final Polish ===")

        # 8a: LP
        alloc_lp = min(remaining * 0.4, 100.0)
        if alloc_lp > 20:
            a_f = lp_iterative_polish(best_seq, time_budget=alloc_lp, verbose=False)
            obj_f = compute_obj_conv(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # 8b: Lp GD with high p
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.3, 50.0)
        if alloc_gd > 15:
            a_f = lp_gradient_descent(best_seq, time_budget=alloc_gd,
                                       p_start=50.0, p_end=500.0, top_k=3,
                                       lr_init=0.005 / np.sqrt(len(best_seq)))
            obj_f = compute_obj_conv(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LpGD: obj={best_score:.8f} **")

        # 8c: CD
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.3, 40.0)
        if alloc_cd > 10:
            a_f = coordinate_descent(best_seq, time_budget=alloc_cd, max_sweeps=3)
            obj_f = compute_obj_conv(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # === Phase 9: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 9: Diverse Restarts ===")
        n_cur = len(best_seq)

        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            mode = (r * 3 + 1) % 9
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best
            blend = 0.3
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            # Quick optimization
            t_gd = min(20.0, remaining * 0.3)
            a_r = lp_gradient_descent(a_r, time_budget=t_gd, p_start=6.0, p_end=40.0)
            t_lp = min(20.0, remaining * 0.3)
            if t_lp > 5:
                a_r = lp_iterative_polish(a_r, time_budget=t_lp)

            obj_r = compute_obj_conv(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_conv(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
