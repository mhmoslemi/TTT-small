import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

def full_lp_solve(a, solver_list=None):
    """Solve full LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    
    # Build constraint matrix A where (A@g)[k] = (a*g)[k]
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid indices
    # Coefficient of g[m] in (a*g)[k] is a[k-m] if 0 <= k-m < n
    
    # A is (2n-1) x n
    # A[k, m] = a[k-m] if 0 <= k-m < n, else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[k, m] = a[j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:

    for solver in solver_list:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, M.value
        except Exception as e:
            continue
    return None, None

"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import scipy.sparse as sp
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def build_conv_matrix_sparse(a):
    """Build sparse matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for m in range(n):
        # (a*g)[k] has coefficient a[k-m] for g[m], valid when 0<=k-m<n
        # So k ranges from m to m+n-1 (clipped to 0..2n-2)
        k_start = m
        k_end = min(m + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            j = k - m
            rows.append(k)
            cols.append(m)
            vals.append(a[j])
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_highs(a):
    """Solve the full LP using HiGHS via scipy."""
    n = len(a)
    a = a / float(np.sum(a))
    
    A_conv = build_conv_matrix_sparse(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality: A_conv @ g - M <= 0
    # [A_conv | -e] @ [g; M] <= 0
    one_col = sp.csr_matrix(np.ones((2 * n - 1, 1)) * (-1.0))
    A_ub = sp.hstack([A_conv, one_col], format='csc')
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = np.maximum(result.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(result.x[n])
    return None, None


def solve_lp_cvxpy(a, top_m=None):
    """Solve LP using cvxpy (fallback). If top_m given, use truncated."""
    n = len(a)
    a = a / float(np.sum(a))
    
    if top_m is not None and top_m < 2 * n - 1:
        conv_a = compute_conv(a)
        top_indices = np.argsort(conv_a)[::-1][:top_m]
    else:
        top_indices = list(range(2 * n - 1))
    
    # Build A matrix for selected constraints
    A = np.zeros((len(top_indices), n))
    for idx, k in enumerate(top_indices):
        k = int(k)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[idx, m] = a[j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

        except Exception:

    return None


def golden_section_1d(f, a, b, tol=1e-8, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    
    x_opt = (a + b) / 2
    return x_opt, f(x_opt)

                a[i] = t_opt
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    best_obj = obj_opt
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = ai_old
        
        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
        
        if verbose:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, t={time.monotonic()-start:.1f}s")
        
        if not improved:
            break
    
    return best_a


def peak_reduction_descent(a, time_budget, verbose=True):
    """Descent that specifically targets the peaks of the convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / n
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top peaks
        top_k = np.argsort(conv)[::-1][:5]
        peak_vals = conv[top_k]
        weights = peak_vals ** 4  # Emphasize the highest peaks
        weights /= weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for ki, w in zip(top_k, weights):
            ki = int(ki)
            # d/d(a[i]) (a*a)[ki] = 2*a[ki-i] if 0 <= ki-i < n
            valid = (ki - np.arange(n) >= 0) & (ki - np.arange(n) < n)
            indices = np.nonzero(valid)[0]
            for i in indices:
                grad[i] += 2.0 * w * a[ki - i]
        
        # Project gradient onto tangent space of simplex
        grad_mean = np.mean(grad)
        grad_tangent = grad - grad_mean
        
        # Gradient step with backtracking
        a_old = a.copy()
        obj_old = best_obj
        t = lr
        improved = False
        for _ in range(10):
            a_try = a_old - t * grad_tangent
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < obj_old - 1e-14:
                a = a_try
                improved = True
                break
            t *= 0.5
        
        if improved:
            obj_now = compute_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
            lr = t * 1.05
        else:
            lr *= 0.8
            if lr < 1e-10:
                break
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    PRD it{it}: obj={best_obj:.10f}, lr={lr:.2e}, t={time.monotonic()-start:.1f}s")
    
    return best_a


def u_shaped_init(n, sharpness):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    x = np.linspace(0.005, 0.995, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def power_init(n, alpha):
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.maximum(d, 1.0 / (2 * n)) ** (-alpha) + 0.01
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def symmetric_enforce(a):
    """Enforce a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = float(np.sum(a_sym))
    if s > 0.01:
        return a_sym / s
    return a
