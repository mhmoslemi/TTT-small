def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2 assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def compute_smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via log-sum-exp trick."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv, p


def init_uniform(n):
    return np.ones(n) / n


def init_raised_cosine(n):
    x = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x + 0.5) / n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_hamming(n):
    x = np.arange(n)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_blackman(n):
    x = np.arange(n)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_thue_morse(n):
    """Generate Thue-Morse based sequence, smoothed."""
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Smooth the 0/1 sequence into a positive weight
    from scipy.ndimage import uniform_filter1d
    a = uniform_filter1d(tm.astype(float), size=max(3, n // 50))
    a = a * 0.5 + 0.5  # shift to [0.5, 1.5]
    a = a / a.sum()
    return a


def init_logistic(n, r=3.9, rho=0.5):
    """Use logistic map to generate a chaotic-like sequence."""
    x = 0.5
    a = np.zeros(n)
    for i in range(n):
        x = r * x * (1.0 - x)
        a[i] = x
    a = a * 0.7 + 0.3  # ensure positive
    return a / a.sum()


def init_two_hump(n, centers=None, width=None):
    """Two Gaussian humps."""
    if centers is None:
        centers = [0.3, 0.7]
    if width is None:
        width = n // 8
    x = np.arange(n)
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-((x - c * (n - 1)) ** 2) / (2.0 * width ** 2))
    a += 0.01  # floor
    return a / a.sum()


def init_irregular(n, seed=42):
    """Random initialization with structure."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.3, 1.0, n)
    # Add some smooth structure
    from scipy.ndimage import gaussian_filter1d
    a = gaussian_filter1d(a, sigma=max(2, n // 40))
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def fast_coordinate_descent(a, time_budget=100.0, verbose=False, seed=42,
                            max_sweeps=5000, warm_start=True):
    """
    Fast coordinate descent with:
    - Precomputed convolution that gets incrementally updated
    - Gradient-informed coordinate ordering
    - Warm-starting from previous sweep's optimum
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    # Previous optima for warm-starting
    prev_optima = a.copy()

    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        sweep += 1
        S = float(np.sum(a))

        # Compute gradient for ordering (every 4 sweeps)
        if sweep % 4 == 1:
            try:
                grad, _, _ = compute_smoothed_gradient(a, 80.0)
                # Sort by |gradient| descending - update high-gradient coords first
                order = np.argsort(-np.abs(grad))
            except Exception:
                order = rng.permutation(n)
        else:
            # Perturb the order slightly for diversity
            order = np.arange(n)
            rng.shuffle(order)

        improved = False

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            # Determine affected convolution positions
            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)

            # a[k-i] values for the linear terms
            a_ki = a[ks - i]
            c_k = conv[ks]

            # Separate the quadratic term (k = 2i)
            k_quad = 2 * i
            if 0 <= k_quad <= 2 * n - 2 and k_quad >= k_min and k_quad <= k_max:
                mask = (ks != k_quad)
                ks_lin = ks[mask]
                a_ki_lin = a_ki[mask]
                c_k_lin = c_k[mask]
                c_2i = float(conv[k_quad])
            else:
                ks_lin = ks
                a_ki_lin = a_ki
                c_k_lin = c_k
                c_2i = -1e30  # No quadratic term in range

            # Unchanged conv max (for reference)
            if 0 <= k_quad <= 2 * n - 2:
                unchanged_mask = np.ones(2 * n - 1, dtype=bool)
                unchanged_mask[ks] = False
                if unchanged_mask.any():
                    c_unchanged_max = float(np.max(conv[unchanged_mask]))
                else:
                    c_unchanged_max = -1e30
            else:
                c_unchanged_max = float(np.max(conv))

            def eval_1d(x, v=v, S_other=S_other, c_k_lin=c_k_lin,
                        a_ki_lin=a_ki_lin, c_2i=c_2i, c_unchanged_max=c_unchanged_max):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = -1e30
                quad_val = c_2i + x * x - v * v if c_2i > -1e29 else -1e30
                max_val = max(max_lin, quad_val, c_unchanged_max)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Warm start: use previous optimum as center
            if warm_start and prev_optima[i] > 0:
                center = prev_optima[i]
            else:
                center = v

            # Evaluate at a few strategic points
            scale = max(center, 1.0 / n, 1e-6)
            candidates = [0.0, center * 0.5, center * 0.7, center,
                          center * 1.3, center * 1.5,
                          1.0 / n, 2.0 / n, 3.0 / n]
            candidates = sorted(set(max(0.0, c) for c in candidates))

            best_x = v
            best_f = eval_1d(v)
            for x in candidates:
                fx = eval_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section search
            lo = max(0.0, best_x - scale)
            hi = best_x + scale * 2.0
            if hi - lo < 1e-12:
                hi = lo + 1e-8

            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = eval_1d(c1)
            f2 = eval_1d(c2)

            for _ in range(30):
                if hi - lo < 1e-12 * max(1.0, hi):
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_1d(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_1d(c2)

            x_opt = max(0.0, (lo + hi) / 2.0)
            prev_optima[i] = x_opt

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            improved = True

            # Update convolution
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

        # Early stopping if no improvement in this sweep
        if not improved:
            # Try one more sweep with perturbed order, then stop
            break

    return best_a


def peak_focused_perturbation(a, rng, intensity=0.05):
    """
    Perturb coordinates that contribute to the peak lag.
    Reduces coordinates that are "paired" at the peak.
    """
    n = len(a)
    conv = compute_conv(a)
    peak_k = int(np.argmax(conv))

    # Find pairs (i, j) where i+j = peak_k and both in [0, n-1]
    # These contribute to conv[peak_k]
    # Reduce the larger of each pair
    a_new = a.copy()
    for i in range(max(0, peak_k - n + 1), min(n, peak_k + 1)):
        j = peak_k - i
        if 0 <= j < n and j != i:
            if a_new[i] > a_new[j]:
                a_new[i] *= (1.0 - intensity * rng.uniform(0, 1))
            else:
                a_new[j] *= (1.0 - intensity * rng.uniform(0, 1))
        elif j == i:
            a_new[i] *= (1.0 - intensity * 0.5 * rng.uniform(0, 1))

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s


def multiplicative_update(a, time_budget=60.0, tau=100.0, lr_scale=5.0):
    """Multiplicative weight update with smoothed gradient."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    lr = lr_scale / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        try:
            grad, conv, p = compute_smoothed_gradient(a, tau)
        except Exception:
            break

        update = np.exp(-lr * grad)
        update = np.clip(update, 0.05, 20.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1
        tau *= 1.05  # Increase sharpness over time

    return best_a


def block_cd(a, block_size=2, time_budget=60.0, seed=42):
    """
    Block coordinate descent: update pairs of coordinates simultaneously.
    For a pair (i,j), we optimize over (a_i, a_j) jointly.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        # Pick a random pair
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        v_i, v_j = a[i], a[j]
        S_rest = S - v_i - v_j
        if S_rest < 1e-10:
            continue

        # For the 2D subproblem, we'll use a simple grid + refine approach
        # The affected conv positions are those involving i or j
        # This is complex, so we do a simple 2D grid search
        scale_i = max(v_i, 1.0 / n, 1e-6)
        scale_j = max(v_j, 1.0 / n, 1e-6)

        best_xi, best_xj = v_i, v_j
        best_f = 2.0 * n * float(np.max(conv)) / (S * S)

        # Coarse grid
        for xi_frac in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
            for xj_frac in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
                xi = xi_frac * v_i
                xj = xj_frac * v_j
                # Quick evaluation: update conv temporarily
                # This is expensive, so we approximate
                # Just check: does reducing the larger help?
                if xi + xj > S:
                    continue
                # Approximate: if both are at the peak lag, reducing both helps
                # For now, just do a simple heuristic
                pass

        # Actually, block CD is too expensive to do properly here.
        # Fall back to sequential CD on the pair.
        a = fast_coordinate_descent(a, time_budget=min(5.0, time_budget - (time.monotonic() - start)),
                                     seed=seed + i * 7 + j)
        obj = compute_obj_normalized(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    1. Diverse initialization exploration
    2. Fast coordinate descent (main workhorse)
    3. Peak-focused perturbation + re-optimize
    4. Multiplicative updates
    5. Size exploration
    6. Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_base = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_seq = parent.copy()
                best_score = compute_obj_normalized(parent)
                n_base = len(parent)
                print(f"Loaded parent (n={n_base}): obj={best_score:.8f}")
    except Exception:
        best_seq = init_uniform(n_base)
        best_score = compute_obj_normalized(best_seq)
        print(f"Default uniform (n={n_base}): obj={best_score:.8f}")

    # === Phase 1: Diverse Initialization Exploration ===
    print("\n=== Phase 1: Diverse Initialization ===")
    inits = []
    inits.append(("parent", best_seq.copy()))
    inits.append(("uniform", init_uniform(n_base)))
    inits.append(("raised_cos", init_raised_cosine(n_base)))
    inits.append(("hamming", init_hamming(n_base)))
    inits.append(("blackman", init_blackman(n_base)))
    inits.append(("thue_morse", init_thue_morse(n_base)))
    inits.append(("logistic", init_logistic(n_base)))
    inits.append(("two_hump", init_two_hump(n_base)))
    inits.append(("irregular", init_irregular(n_base, seed=123)))
    inits.append(("irregular2", init_irregular(n_base, seed=456)))

    # Blend some with parent for diversity
    for name, init_a in list(inits):
        if name not in ("parent", "uniform"):
            blend = 0.5
            blended = (1 - blend) * init_a + blend * best_seq
            blended = np.maximum(blended, 0)
            blended = blended / blended.sum()
            inits.append((name + "_blend", blended))

    for name, init_a in inits:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.15, 30.0)
        if alloc < 10:
            continue
        a_opt = fast_coordinate_descent(init_a, time_budget=alloc, seed=seed + hash(name) % 10000)
        obj = compute_obj_normalized(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  {name}: obj={best_score:.8f} **")
        else:
            print(f"  {name}: obj={obj:.8f}")

    # === Phase 2: Extended Coordinate Descent ===
    print(f"\n=== Phase 2: Extended CD (current best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 400.0)
    if p2_budget > 30:
        a_cd = fast_coordinate_descent(best_seq, time_budget=p2_budget,
                                       verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 3: Peak-Focused Perturbation ===
    print(f"\n=== Phase 3: Peak-Focused Perturbation (best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    for attempt in range(5):
        if remaining < 30:
            break
        alloc_pert = min(15.0, remaining * 0.1)
        a_pert = peak_focused_perturbation(best_seq, rng, intensity=0.03 + 0.02 * attempt)
        # Also try general perturbation
        a_gen = best_seq * np.exp(rng.normal(0, 0.05 * (attempt + 1), n_base))
        a_gen = np.maximum(a_gen, 0) / float(np.sum(a_gen))

        for label, a_try in [("pert", a_pert), ("gen", a_gen)]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.2, 40.0)
            if alloc < 10:
                continue
            a_opt = fast_coordinate_descent(a_try, time_budget=alloc, seed=seed + 10000 + attempt * 13)
            obj_opt = compute_obj_normalized(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  {label} attempt {attempt}: obj={best_score:.8f} **")
            break  # Only try first perturbation type per attempt to save time

    # === Phase 4: Multiplicative Updates ===
    print(f"\n=== Phase 4: Multiplicative Updates (best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_mu = multiplicative_update(best_seq, time_budget=alloc, tau=50.0, lr_scale=4.0)
        obj_mu = compute_obj_normalized(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            # Follow up with CD
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd2 = fast_coordinate_descent(best_seq, time_budget=min(remaining * 0.3, 60.0),
                                                seed=seed + 20000)
                obj_cd2 = compute_obj_normalized(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  MU+CD: obj={best_score:.8f} **")
            print(f"  MU: obj={best_score:.8f} **")
        else:
            print(f"  MU: obj={obj_mu:.8f}")

    # === Phase 5: Size Exploration ===
    print(f"\n=== Phase 5: Size Exploration (best={best_score:.8f}, n={len(best_seq)}) ===")
    n_current = len(best_seq)
    n_options = []
    for delta in [-100, -50, 50, 100, 150, 200, 300]:
        n_try = n_current + delta
        if 400 <= n_try <= 2000:
            n_options.append(n_try)
    # Also try some round numbers
    for n_try in [500, 600, 700, 900, 1000, 1200, 1500]:
        if abs(n_try - n_current) > 100 and 400 <= n_try <= 2000:
            n_options.append(n_try)
    n_options = sorted(set(n_options))[:6]

    print(f"  Trying sizes: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_resampled = resample_sequence(best_seq, n_try)
        alloc_cd = min(remaining * 0.2, 45.0)
        if alloc_cd < 15:
            continue
        a_opt = fast_coordinate_descent(a_resampled, time_budget=alloc_cd, seed=n_try)
        obj_opt = compute_obj_normalized(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Second CD Pass on Best ===
    print(f"\n=== Phase 6: Final CD (best={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 120.0)
        a_final = fast_coordinate_descent(best_seq, time_budget=alloc,
                                          verbose=True, seed=seed + 30000)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # === Phase 7: One more perturbation + CD ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        for pert_scale in [0.02, 0.01, 0.005]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_p = best_seq * np.exp(rng.normal(0, pert_scale, len(best_seq)))
            a_p = np.maximum(a_p, 0) / float(np.sum(a_p))
            alloc = min(remaining * 0.4, 40.0)
            a_opt = fast_coordinate_descent(a_p, time_budget=alloc, seed=seed + 40000 + int(pert_scale * 10000))
            obj_opt = compute_obj_normalized(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Pert({pert_scale})+CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
