"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Multi-phase optimization combining fast exact coordinate descent,
LP-based linearization with adaptive stepping, spectrally-informed initialization,
and multi-length exploration. Key improvements: vectorized CD, full-constraint LP,
symmetry exploitation, and adaptive time allocation.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute full objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix_direct(a):
    """Build matrix B (2n-1 x n) such that B @ g = conv(a, g) element-wise.
    B[k, j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        B[lo:hi, j] = a[:hi - lo]
    return B


def solve_lp_full(g_current, time_budget=20.0):
    """
    Solve LP: min t s.t. conv(g_cur, g_new)[k] <= t for all k, sum(g_new)=1, g_new>=0.
    This is the linearization of the quadratic around g_current.
    Returns (g_new, t) or (None, None).
    """
    n = len(g_current)
    a_norm = g_current / float(np.sum(g_current))
    B = build_conv_matrix_direct(a_norm)

    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [B @ g <= t, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(np.max(B @ (g_val / s)))
        except Exception:
            continue
    return None, None


def solve_lp_topk(g_current, topk=200, time_budget=15.0):
    """LP with only top-k most active constraints for speed."""
    n = len(g_current)
    a_norm = g_current / float(np.sum(g_current))
    conv_cur = compute_conv(a_norm)
    max_conv = float(np.max(conv_cur))

    # Select top-k lags with highest current values
    k_indices = np.argsort(conv_cur)[::-1][:topk]

    B_full = build_conv_matrix_direct(a_norm)
    B = B_full[k_indices, :]

    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [B @ g <= t, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(np.max(B @ g_val))
        except Exception:
            continue
    return None, None


def coordinate_descent_fast(a, time_budget=300.0, seed=42, verbose=False):
    """
    Fast exact coordinate descent for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses the 1D structure of each coordinate update.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    no_improve_streak = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        S = float(np.sum(a))
        obj_start_sweep = 2.0 * n * float(np.max(conv)) / (S * S)
        improved_this_sweep = False

        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            # Determine which conv indices are affected
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)

            # For each affected lag k, the contribution of a[i] is:
            # If k == 2*i: a[i]^2 (quadratic)
            # Else: 2 * a[i] * a[k-i] (linear in a[i])
            k_quad = 2 * i
            mask_lin = (ks != k_quad)
            ks_lin = ks[mask_lin]
            # For linear terms: coefficient is 2*a[k-i]
            lin_coeffs = np.zeros(len(ks_lin))
            for idx, k in enumerate(ks_lin):
                j = k - i
                if 0 <= j < n:
                    lin_coeffs[idx] = 2.0 * a[j]
            # Base values (without a[i] contribution)
            base_lin = conv[ks_lin].copy() - lin_coeffs * v
            # Quadratic base
            c_quad_base = float(conv[k_quad]) - v * v if 0 <= k_quad <= 2 * n - 2 else 0.0
            if 0 <= k_quad <= 2 * n - 2:
                # Remove the a[i]*a[k-i] terms that are also in linear
                pass  # Already handled since k_quad is excluded from ks_lin

            def f_1d(x):
                # Compute max conv value given a[i] = x
                if len(ks_lin) > 0:
                    lin_vals = base_lin + lin_coeffs * x
                    max_lin = float(np.max(lin_vals)) if len(lin_vals) > 0 else 0.0
                else:
                    max_lin = 0.0
                quad_val = c_quad_base + x * x
                max_val = max(max_lin, quad_val)
                # Also consider unaffected conv values
                if k_min > 0 or k_max < 2 * n - 2:
                    # Some conv values are unaffected
                    unaffected_mask = np.zeros(2 * n - 1, dtype=bool)
                    unaffected_mask[:k_min] = True
                    unaffected_mask[k_max + 1:] = True
                    if np.any(unaffected_mask):
                        max_val = max(max_val, float(np.max(conv[unaffected_mask])))
                S_new = S_other + x
                if S_new < 1e-15:
                    return 1e30
                return 2.0 * n * max_val / (S_new * S_new)

            # Golden section search on [0, upper_bound]
            # Upper bound: the value can't be too large
            upper = max(2.0 * v, 4.0 / n, 0.1)
            x_lo = 0.0
            x_hi = upper

            # Quick scan to narrow bracket
            n_scan = 10
            xs = np.linspace(0, upper, n_scan + 1)
            vals = np.array([f_1d(float(x)) for x in xs])
            best_idx = int(np.argmin(vals))
            if best_idx == 0:
                x_hi = xs[1]
                x_lo = 0.0
            elif best_idx == n_scan:
                x_lo = xs[n_scan - 1]
                x_hi = upper
            else:
                x_lo = xs[best_idx - 1]
                x_hi = xs[best_idx + 1]

            # Golden section
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            # Update conv
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += lin_coeffs * delta
            S += delta

            if S > 1e-10:
                obj_new = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved_this_sweep = True

        if not improved_this_sweep:
            no_improve_streak += 1
            if no_improve_streak >= 5:
                # Perturb to escape
                pert_scale = 0.05 * no_improve_streak
                pert = rng.normal(0, pert_scale, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                S = float(np.sum(a))
                if S > 1e-10:
                    a = a / S
                    conv = compute_conv(a)
                    best_obj = min(best_obj, 2.0 * n * float(np.max(conv)))
                no_improve_streak = 0
        else:
            no_improve_streak = 0

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a


def lp_iteration(a, n_iters=20, step_decay=0.9, time_budget=100.0, seed=42):
    """
    LP-based iterative refinement.
    At each step, linearize around current point, solve LP, blend.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    t_step = 0.3
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start >= time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Choose LP variant
        if n <= 600 and remaining > 30:
            g_lp, _ = solve_lp_full(a, time_budget=min(remaining * 0.3, 15.0))
        else:
            g_lp, _ = solve_lp_topk(a, topk=min(200, 2 * n), time_budget=min(remaining * 0.3, 15.0))

        if g_lp is None:
            break

        # Line search over blend factor
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            t_step = min(0.5, t_step * 1.1)
        else:
            stall += 1
            t_step *= step_decay
            if stall > 5:
                # Perturb and restart
                pert = rng.normal(0, 0.02 * stall, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0
                t_step = 0.2

    return best_a


def make_hann_init(n):
    """Hann window initialized sequence."""
    k = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * k / (n - 1))) if n > 1 else np.array([1.0])
    return a / a.sum()


def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian centered sequence."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    return a / a.sum()


def make_cosine_taper_init(n, power=2.0):
    """Cosine taper: a[k] = cos^power(pi * k / (n-1))."""
    k = np.arange(n, dtype=float)
    if n > 1:
        a = np.cos(np.pi * k / (n - 1)) ** power
    else:
        a = np.array([1.0])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def make_slepian_approx(n, bandwidth=0.3):
    """Approximate prolate spheroidal wave function using iterative spectral concentration."""
    k = np.arange(n)
    a = np.ones(n)

    # Iteratively project: time-domain windowing + frequency-domain windowing
    W = 1 << (len(a) * 2)  # FFT size
    for _ in range(50):
        # Frequency-domain: keep only low frequencies
        A = fft(a, W)
        freq_mask = np.zeros(W)
        center = W // 2
        half_bw = int(bandwidth * W / 2)
        freq_mask[center - half_bw:center + half_bw + 1] = 1
        freq_mask[:half_bw] = 1
        freq_mask[W - half_bw:] = 1
        A *= freq_mask
        a = ifft(A).real[:n]
        # Time-domain: keep only center
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def harmonic_descent(a, time_budget=100.0, seed=42):
    """
    Harmonic (multiplicative) gradient descent using smoothed max.
    Computes gradient of smooth approximation to max(conv) via log-sum-exp.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    tau = 100.0
    lr = 5.0 / np.sqrt(n)
    momentum = np.zeros(n)
    beta = 0.9
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        tau_eff = tau * (1 + 3 * progress)
        lr_eff = lr * (1 - 0.6 * progress)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Smoothed gradient via log-sum-exp
        exp_vals = np.exp(tau_eff * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient of max(conv) w.r.t. a[i] is 2*sum_k p[k]*a[k-i] (for valid k-i)
        # Compute via FFT: grad = 2 * ifft(fft(p, N) * conj(fft(a, N)))[:n]
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_ft = fft(a, N)
        grad_full = ifft(P * np.conj(A_ft)).real
        grad = 2.0 * grad_full[:n]

        # Also subtract the "self" term for the quadratic part
        # Actually the gradient of conv[k] w.r.t. a[i] is:
        # a[k-i] + a[i-(k-i)] if i != k-i, or 2*a[i] if i == k-i
        # This is exactly what the FFT gives us above.

        # Momentum update
        momentum = beta * momentum - lr_eff * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Reject update, reduce learning rate
            momentum = 0.9 * momentum
            lr_eff *= 0.7
            a_new = a

        a = a_new
        it += 1

        if it % 5000 == 0:
            print(f"    HD it{it}: obj={best_obj:.10f}, tau={tau_eff:.0f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining coordinate descent,
    LP-based refinement, harmonic descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                n_parent = len(parent)
                print(f"Parent (n={n_parent}): obj={best_score:.10f}")
    except Exception:
        n_parent = 800
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.10f}")

    def time_left():
        return deadline - time.monotonic()

    # === Phase 1: Initialization exploration (30s) ===
    print("\n=== Phase 1: Initialization ===")
    if time_left() > 30:
        init_budget = min(30.0, time_left())
        n_try = n_parent
        inits = {
            'parent': parent.copy(),
            'hann': make_hann_init(n_try),
            'gauss': make_gaussian_init(n_try, 0.25),
            'cos_taper': make_cosine_taper_init(n_try, 2.0),
            'slepian': make_slepian_approx(n_try, 0.25),
        }
        for name, a_init in inits.items():
            a_init = np.maximum(a_init, 0)
            s = float(np.sum(a_init))
            if s < 1e-10:
                continue
            a_init = a_init / s
            obj_init = compute_obj_normalized(a_init)
            print(f"  {name}: obj={obj_init:.10f}")
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()

    # === Phase 2: Coordinate Descent (main workhorse, ~50% of time) ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    p2_budget = min(time_left() * 0.45, 500.0)
    if p2_budget > 30:
        print(f"  Budget: {p2_budget:.0f}s")
        a_cd = coordinate_descent_fast(best_seq, time_budget=p2_budget,
                                       seed=seed, verbose=True)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD result: obj={best_score:.10f} **")
        else:
            print(f"  CD result: obj={obj_cd:.10f} (no improvement)")

    # === Phase 3: LP Refinement (~20% of time) ===
    print(f"\n=== Phase 3: LP Refinement ===")
    p3_budget = min(time_left() * 0.2, 200.0)
    if p3_budget > 20:
        print(f"  Budget: {p3_budget:.0f}s")
        for rnd in range(4):
            if time_left() < 15:
                break
            alloc = min(p3_budget / (4 - rnd), time_left() * 0.4)
            if alloc < 10:
                break
            a_lp = lp_iteration(best_seq, n_iters=30,
                                time_budget=alloc, seed=5000 + rnd * 100)
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.10f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.10f}")

            # Follow up with brief CD
            if time_left() > 20:
                a_cd2 = coordinate_descent_fast(best_seq, time_budget=min(30.0, time_left() * 0.1),
                                                seed=6000 + rnd)
                obj_cd2 = compute_obj_normalized(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"    CD after LP{rnd}: obj={best_score:.10f} **")

    # === Phase 4: Harmonic Descent (~10% of time) ===
    print(f"\n=== Phase 4: Harmonic Descent ===")
    if time_left() > 30:
        alloc = min(time_left() * 0.1, 80.0)
        a_hd = harmonic_descent(best_seq, time_budget=alloc, seed=seed + 999)
        obj_hd = compute_obj_normalized(a_hd)
        if obj_hd < best_score:
            best_score = obj_hd
            best_seq = a_hd.copy()
            print(f"  HD: obj={best_score:.10f} **")
        else:
            print(f"  HD: obj={obj_hd:.10f}")
            # Try from perturbed start
            if time_left() > 30:
                pert = rng.normal(0, 0.03, len(best_seq))
                a_pert = best_seq * np.exp(pert)
                a_pert = np.maximum(a_pert, 0) / float(a_pert.sum())
                a_hd2 = harmonic_descent(a_pert, time_budget=alloc * 0.5, seed=seed + 888)
                obj_hd2 = compute_obj_normalized(a_hd2)
                if obj_hd2 < best_score:
                    best_score = obj_hd2
                    best_seq = a_hd2.copy()
                    print(f"  HD (perturbed): obj={best_score:.10f} **")

    # === Phase 5: Multi-n Exploration (~20% of time) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    p5_budget = min(time_left() * 0.25, 250.0)

    n_options = sorted(set([
        max(400, n_best - 100), n_best - 50,
        n_best + 50, n_best + 100, n_best + 150, n_best + 200,
        900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 500 <= n <= 2000 and abs(n - n_best) >= 40]
    n_options = n_options[:6]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        if time_left() < 40 or p5_budget <= 0:
            break

        # Resample from best sequence
        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Also try fresh initializations at this n
        a_hann = make_hann_init(n_try)
        a_gauss = make_gaussian_init(n_try, 0.25)

        for a_init in [a_try, a_hann, a_gauss]:
            if time_left() < 30 or p5_budget <= 0:
                break
            alloc_cd = min(p5_budget * 0.4, time_left() * 0.3, 60.0)
            if alloc_cd < 15:
                continue
            a_opt = coordinate_descent_fast(a_init, time_budget=alloc_cd,
                                            seed=n_try, verbose=False)
            obj_try = compute_obj_normalized(a_opt)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try} (init obj={compute_obj_normalized(a_init):.6f}): best={obj_try:.10f}")

            # Brief LP if promising
            if obj_try < best_score * 1.01 and time_left() > 20:
                alloc_lp = min(20.0, time_left() * 0.2)
                if alloc_lp > 8:
                    a_lp = lp_iteration(a_opt, n_iters=10, time_budget=alloc_lp,
                                        seed=n_try + 700)
                    obj_lp = compute_obj_normalized(a_lp)
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        print(f"  ** n={n_try} LP: obj={best_score:.10f} **")

    # === Phase 6: Final Polish (~10% remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    if time_left() > 20:
        alloc = min(time_left() * 0.4, 100.0)
        a_final = coordinate_descent_fast(best_seq, time_budget=alloc,
                                          seed=seed + 77777, verbose=True)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.10f} **")
        else:
            print(f"  Final CD: obj={obj_final:.10f}")

    if time_left() > 15:
        alloc = min(time_left() * 0.4, 50.0)
        a_lp_final = lp_iteration(best_seq, n_iters=15, time_budget=alloc,
                                  seed=88888)
        obj_lp_final = compute_obj_normalized(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_lp_final:.10f}")

    # One more CD pass
    if time_left() > 15:
        a_cd_final = coordinate_descent_fast(best_seq, time_budget=min(time_left() * 0.5, 40.0),
                                             seed=seed + 99999)
        obj_cd_final = compute_obj_normalized(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD2: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
