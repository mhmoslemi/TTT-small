import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2 for normalized a (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x : sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v.copy()

    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_gradient_l2(p):
    """
    Compute gradient of 2n * sum(conv[p,p]^2) w.r.t. p.
    This is 4n * sum_k conv[k] * p[k-i] = 4n * cross_correlation(conv, p)[i]
    """
    n = len(p)
    conv = compute_conv(p)
    N = 1
    while N < 3 * n:
        N <<= 1
    Q = fft(conv, N)
    P = fft(p, N)
    corr = ifft(np.conj(P) * Q).real[:n]
    grad = 4.0 * n * corr
    obj_val = 2.0 * n * float(np.sum(conv ** 2))
    return grad, obj_val


def fista_optimize(p, n, time_budget, tau_start, tau_end, use_l2_phase=True, 
                   l2_ratio=0.3, verbose=False):
    """
    FISTA optimization on the simplex to minimize the (smoothed) objective.
    Phase 1: minimize l2 norm of convolution (smoother)
    Phase 2: minimize smoothed l-infinity (log-sum-exp)
    """
    start = time.monotonic()
    p = p.copy()
    s = float(np.sum(p))
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_obj = compute_obj_normalized(best_p)

    # Phase 1: L2 minimization
    if use_l2_phase:
        l2_budget = time_budget * l2_ratio
        lr = 1.0 / (8.0 * n)
        t_k = 1.0
        y = p.copy()
        it = 0
        while time.monotonic() - start < l2_budget:
            grad_l2, obj_l2 = compute_gradient_l2(y)
            # Step size
            step = lr / (1.0 + it * 0.001)
            z = y - step * grad_l2
            x_new = project_simplex(z)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - p)
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y > 0:
                y = y / s_y
            p = x_new
            t_k = t_new
            it += 1

            if it % 500 == 0:
                obj_cur = compute_obj_normalized(p)
                if obj_cur < best_obj:
                    best_obj = obj_cur
                    best_p = p.copy()
                if verbose:
                    print(f"      L2 phase it{it}: obj={best_obj:.8f}")

    # Phase 2: Log-sum-exp (smoothed max) minimization
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        n_start = max(1, int(l2_ratio * time_budget * 5000 / max(n, 1)))
        p = best_p.copy()
        s = float(np.sum(p))
        if s > 0:
            p = p / s

        t_k = 1.0
        y = p.copy()
        it = 0
        lr_base = 1.0 / (4.0 * n)

        while time.monotonic() - start < time_budget - 2:
            elapsed = time.monotonic() - start
            progress = min(1.0, elapsed / max(time_budget, 1e-10))
            tau = tau_start + (tau_end - tau_start) * (progress ** 0.7)
            lr = lr_base * (1.0 + 0.5 * (1.0 - progress))

            grad, conv = compute_gradient_max_smoothed(y, tau)
            g_norm = float(np.linalg.norm(grad))
            if g_norm < 1e-15:
                break
            # Backtracking-like step size
            step = lr / max(g_norm / (lr_base + 1e-10), 1.0)
            step = min(step, 2.0 / n)

            z = y - step * grad
            x_new = project_simplex(z)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - p)
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                y = p.copy()
                t_k = 1.0
            else:
                y = y / s_y
            p = x_new
            t_k = t_new
            it += 1

            if it % 200 == 0:
                obj_cur = compute_obj_normalized(p)
                if obj_cur < best_obj:
                    best_obj = obj_cur
                    best_p = p.copy()
                if verbose:
                    print(f"      LS phase it{it}: tau={tau:.1f}, obj={best_obj:.8f}")

    # Final check
    obj_final = compute_obj_normalized(p)
    if obj_final < best_obj:
        best_obj = obj_final
        best_p = p.copy()

    return best_p, best_obj


def coordinate_descent(a, time_budget=120.0, verbose=False, seed=42):
    """Exact coordinate descent with golden section search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.6, v * 0.8, v, v * 1.2, v * 1.5,
                          v * 2.0, 1.0 / n, 2.0 / n, 0.5 / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 3.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(50):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def chirp_init(n, freq_start=0.2, freq_end=1.8):
    """Create a chirp-like asymmetric initialization."""
    x = np.arange(n) / max(n - 1, 1)
    phase = 2 * np.pi * (freq_start * x + (freq_end - freq_start) / 2 * x ** 2)
    # Make it non-negative using absolute value + offset
    a = 0.5 + 0.5 * np.cos(phase)
    a = np.maximum(a, 0.01)
    # Add asymmetry
    a *= (0.5 + 0.5 * x)
    a = a / a.sum()
    return a


def asymmetric_init(n, skew=0.3):
    """Create an asymmetric initialization that biases mass toward one side."""
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 + skew * (2 * x - 1)
    a = np.maximum(a, 0.05)
    a = a / a.sum()
    return a


def gaussian_init(n, center_frac=0.4, width_frac=0.2):
    """Gaussian-shaped initialization (off-center for asymmetry)."""
    x = np.arange(n)
    center = center_frac * n
    width = width_frac * n
    a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def blended_init(n, base, other, alpha=0.5):
    """Blend two sequences."""
    a = alpha * base + (1 - alpha) * other
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return base.copy()
    return a / s

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = chirp_init(n_default)
        best_score = compute_obj_normalized(best_seq)
        print(f"Default chirp init (n={n_default}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Multi-n quick exploration ===
    print(f"\n=== Phase 1: Multi-n Exploration ===")
    n_options = sorted(set([
        max(400, n_best - 100),
        n_best,
        n_best + 100,
        n_best + 200,
        600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000]
    n_options = n_options[:8]

    n_candidates = {}
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Generate a few init types
        inits = [
            chirp_init(n_try, freq_start=0.1 + 0.1 * rng.random(),
                       freq_end=1.0 + 0.5 * rng.random()),
            asymmetric_init(n_try, skew=0.2 + 0.3 * rng.random()),
            gaussian_init(n_try, center_frac=0.3 + 0.3 * rng.random(),
                         width_frac=0.15 + 0.1 * rng.random()),
            np.ones(n_try) / n_try,
        ]
        # Also resample parent if available
        if parent is not None and n_try != len(parent):
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_try)
            parent_resampled = np.interp(x_new, x_old, parent)
            parent_resampled = np.maximum(parent_resampled, 0)
            s_r = parent_resampled.sum()
            if s_r > 1e-10:
                inits.append(parent_resampled / s_r)

        best_for_n = None
        best_obj_for_n = np.inf
        per_init_budget = min(remaining / len(n_options) / 4.0, 8.0)

        for init_i, a_init in enumerate(inits):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(per_init_budget, remaining * 0.4)
            if alloc < 3:
                continue
            a_try, obj_try = fista_optimize(a_init, n_try, time_budget=alloc,
                                            tau_start=50, tau_end=200,
                                            use_l2_phase=True, l2_ratio=0.4)
            if obj_try < best_obj_for_n:
                best_obj_for_n = obj_try
                best_for_n = a_try.copy()

        if best_for_n is not None:
            n_candidates[n_try] = (best_for_n, best_obj_for_n)
            if best_obj_for_n < best_score:
                best_score = best_obj_for_n
                best_seq = best_for_n.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: best obj={best_obj_for_n:.8f}")

    # === Phase 2: Main FISTA optimization on best n ===
    print(f"\n=== Phase 2: Main FISTA (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    # Use best available start
    if n_best in n_candidates:
        start_p = n_candidates[n_best][0].copy()
    else:
        start_p = best_seq.copy()

    # Multiple FISTA runs with different tau schedules
    fista_configs = [
        (30, 300, 0.3),   # tau_start, tau_end, l2_ratio
        (50, 500, 0.2),
        (20, 200, 0.4),
        (40, 400, 0.25),
    ]

    for cfg_i, (ts, te, lr2) in enumerate(fista_configs):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(remaining * 0.25, 250.0)
        if alloc < 30:
            break

        a_fista, obj_fista = fista_optimize(start_p, n_best, time_budget=alloc,
                                            tau_start=ts, tau_end=te,
                                            use_l2_phase=True, l2_ratio=lr2,
                                            verbose=True)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  FISTA cfg{cfg_i}: obj={best_score:.8f} **")
        else:
            print(f"  FISTA cfg{cfg_i}: obj={obj_fista:.8f} (no improvement)")

        # Use this as start for next
        if obj_fista < compute_obj_normalized(start_p):
            start_p = a_fista.copy()

    # === Phase 3: Coordinate Descent Polish ===
    print(f"\n=== Phase 3: Coordinate Descent Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 200.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True,
                                  seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 4: Second FISTA round from CD result ===
    print(f"\n=== Phase 4: FISTA Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.2, 150.0)
        # Perturb slightly
        pert = rng.normal(0, 0.02, n_best)
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        a_f2, obj_f2 = fista_optimize(a_pert, n_best, time_budget=alloc,
                                      tau_start=100, tau_end=800,
                                      use_l2_phase=False, l2_ratio=0.0)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  FISTA2: obj={best_score:.8f} **")
        else:
            print(f"  FISTA2: obj={obj_f2:.8f}")

    # === Phase 5: Final CD polish ===
    print(f"\n=== Phase 5: Final CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 120.0)
        a_cd2 = coordinate_descent(best_seq, time_budget=alloc, verbose=False,
                                   seed=seed + 999)
        obj_cd2 = compute_obj_normalized(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_cd2:.8f}")

    # === Phase 6: Try neighboring n values with full optimization ===
    print(f"\n=== Phase 6: Neighboring n ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        for dn in [-50, 50, -100, 100, 150, -150]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            n_try = n_best + dn
            if n_try < 300 or n_try > 2500:
                continue
            # Resample
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

            alloc = min(remaining * 0.2, 60.0)
            a_try, obj_try = fista_optimize(a_try, n_try, time_budget=alloc,
                                            tau_start=50, tau_end=300,
                                            use_l2_phase=True, l2_ratio=0.3)
            # Quick CD
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try = coordinate_descent(a_try, time_budget=min(remaining * 0.15, 30.0),
                                           seed=n_try)
                obj_try = compute_obj_normalized(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
