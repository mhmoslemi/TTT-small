import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def compute_cross_corr(a, w):
    """
    Compute q_i = sum_k w[k] * a[k-i] for i in [0, n-1].
    This is the cross-correlation needed for power-mean gradient.
    Uses FFT for O(n log n) efficiency.
    """
    n = len(a)
    nw = len(w)
    size = 1
    while size < nw + n - 1:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    # IFFT(conj(W) * A) gives sum_m w[m+i] * a[m] = sum_k w[k] * a[k-i]
    q = irfft(np.conj(W) * A, size)[:n]
    return q


def power_mean_gradient(a, p):
    """
    Compute gradient of M_p(conv(a,a)) = (mean(conv^p))^(1/p) w.r.t. a.
    Assumes a is on the simplex (sum=1, a>=0).
    """
    n = len(a)
    conv = compute_conv(a)
    K = 2 * n - 1
    c_max = float(np.max(conv))
    if c_max <= 1e-30:
        return np.zeros(n)
    vals = np.maximum(conv, 1e-30)
    M_p = float(np.mean(vals ** p)) ** (1.0 / p)
    w = vals ** (p - 1.0)
    q = compute_cross_corr(a, w)
    factor = 2.0 * M_p ** (1.0 - p) / K
    return factor * q


def logsumexp_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*conv_k)) w.r.t. a.
    """
    n = len(a)
    conv = compute_conv(a)
    K = 2 * n - 1
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    p = exp_vals / float(np.sum(exp_vals))
    q = compute_cross_corr(a, p)
    return 2.0 * q / K

def exact_max_subgradient_descent(a, time_budget, eps_frac=0.01,

    Subgradient descent using the exact max of the convolution.
    When multiple lags are near the max, average their subgradients.
    """

    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        lr_t = lr * (1.0 - 0.5 * progress)
        mu = momentum + 0.05 * progress

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        threshold = c_max * (1.0 - eps_frac)

        active = np.where(conv >= threshold)[0]

            grad[idx] += a[k - idx]
        grad = 2.0 * grad / len(active)

        a_new = a - lr_t * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:

            if stall > 150:

        if verbose and it % 200 == 0:
            print(f"      EMax it{it}: obj={best_obj:.8f}, n_active={len(active)}, "
                  f"lr={lr_t:.5f}, t={elapsed:.1f}s")

        if it - best_it > 3000 and it > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = lr_t * 0.7

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, max_M=None):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_dir, max_time=2.0):
    """Find best blend factor t in (0, 1.5) for (1-t)*a + t*g_dir."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5

    def obj_at(t):
        t = max(0, t)
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(25):
        if time.monotonic() - start >= max_time:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    # Try some fixed points too
    for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.2, 1.4]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best


def lp_polish(a, time_budget, max_iters=50, verbose=False):
    """LP-based iterative polishing."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 4:
            break

        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, max_time=min(2.0, remaining * 0.05))

        if t_opt > 1e-6 and obj_opt < best_obj - 1e-10:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")

        if stall >= 3:
            if stall <= 6:
                noise = rng.normal(0, 0.004 * (stall - 2), n)
                a = project_simplex(best_a + noise)
            else:
                # Aggressive restart
                x = np.linspace(0, 1, n)
                a_div = 1.0 / (0.03 + 5.0 * (2 * x - 1) ** 2)
                a = project_simplex(0.4 * a_div + 0.6 * best_a)
            stall = 0

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization with tunable sharpness."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.03 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2 + 1e-10)
    return project_simplex(a)


def bimodal_init(n, separation=0.2, width=0.08):
    """Bimodal initialization with two peaks."""
    x = np.linspace(0, 1, n)
    a = (np.exp(-((x - (0.5 - separation)) ** 2) / (2 * width ** 2)) +
         np.exp(-((x - (0.5 + separation)) ** 2) / (2 * width ** 2)) + 0.01)
    return project_simplex(a)


def edge_concentrated_init(n, power=2.0):
    """Mass concentrated at edges with polynomial falloff."""
    x = np.linspace(0, 1, n)
    a = (2 * x - 1) ** (2 * power) + 0.02
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(6 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / np.cosh(4.0 * (2 * x - 1))
    elif mode == 6:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:3] = 0.4
        a[max(0, n-3):n-1] = 0.4
        if n > 4:
            a[1:-1] += 0.03
    elif mode == 7:
        a = 1.0 / (0.01 + 8.0 * (2 * x - 1) ** 4)
    elif mode == 8:
        a = np.abs(np.cos(np.pi * x)) ** 0.5 + 0.02
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def spectral_equality_init(n, rng, seed=0):
    """
    Initialize by trying to equalize the power spectrum.
    Start with a random phase spectrum and take the non-negative part.
    """
    rng2 = np.random.default_rng(seed)
    # Target: flat autocorrelation -> delta spectrum (only at freq 0)
    # But we can't achieve that. Instead, try to make |F| as flat as possible
    # subject to F(0) = 1 (sum constraint).
    # Use iterative spectral projection (like PROJONC but simplified).
    a = project_simplex(u_shaped_init(n, 3.0))
    for _ in range(10):
        F = rfft(a, 2 * n)
        log_F = np.log(np.abs(F) + 1e-15)
        # Nudge towards flat spectrum (except at DC)
        log_F[1:] -= 0.3 * (log_F[1:] - np.mean(log_F[1:]))
        a_new = irfft(np.exp(log_F), 2 * n)[:n]
        a = project_simplex(np.maximum(a_new, 0))
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase hybrid optimization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, 3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # Try alternative initializations
    for name, init_fn in [
        ("arcsine", lambda: arcsine_init(n_current)),
        ("bimodal", lambda: bimodal_init(n_current)),
        ("edge", lambda: edge_concentrated_init(n_current, 2.5)),
        ("u_sharp", lambda: u_shaped_init(n_current, 5.0)),
        ("spectral", lambda: spectral_equality_init(n_current, rng, seed)),
    ]:
        a_try = init_fn()
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"{name} init: obj={best_score:.8f} **")

    # === Phase 1: Power-mean GD (low to medium p) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: PowerMean GD (p=3->30), {remaining:.0f}s ===")
