import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1 << (2 * n - 1).bit_length()
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def make_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[j]
    return A


def lp_solve(a, matrix_cache=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M, A)."""
    n = len(a)
    if matrix_cache is not None:
        A = matrix_cache
    else:
        A = make_conv_matrix(a)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value), A

    return None, None, A

            beta = min(0.7, beta + 0.02)  # gradually increase momentum
            if verbose and it % 5 == 0:
                print(f"    LP-mom it{it:3d}: obj={best_obj:.10f}, t={best_local_t:.3f}, beta={beta:.2f}")
        else:
            stall_count += 1
            # Perturb to escape
            if stall_count % 5 == 0:
                pert_type = stall_count % 4
                if pert_type == 1:
                    a = a + 0.01 * rng.standard_normal(n)
                elif pert_type == 2:
                    # Target max lags for reduction
                    conv = compute_conv(a)
                    max_val = float(np.max(conv))
                    max_lags = np.where(conv >= max_val * 0.999)[0]
                    grad = np.zeros(n)
                    for lag in max_lags[:5]:
                        lo = max(0, lag - n + 1)
                        hi = min(lag, n - 1)
                        for i in range(lo, hi + 1):
                            grad[i] += a[lag - i]
                    gn = np.linalg.norm(grad)
                    if gn > 1e-15:
                        a = a - 0.005 * grad / gn
                elif pert_type == 3:
                    # Smooth perturbation
                    x = np.arange(n)
                    f = rng.integers(1, 10)
                    p = rng.uniform(0, 2 * np.pi)
                    a = a + 0.008 * np.sin(2 * np.pi * f * x / n + p)
                else:
                    a = a + 0.005 * rng.standard_normal(n)

                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                else:
                    a = best_a.copy()

                v = np.zeros(n)
                beta = 0.3

            if stall_count > 50:
                # Major restart
                a = best_a.copy()
                strength = min(0.1, 0.02 + 0.01 * (stall_count // 10))
                a = a + strength * rng.standard_normal(n)
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                v = np.zeros(n)
                beta = 0.3
                stall_count = 0

    return best_a


def coordinate_descent(a, time_budget, verbose=False, seed=42, max_passes=50):
    """
    LP dual-symmetric coordinate descent.
    For coordinate j: M_j(f) = (c_j*f + v_j) / (S+f)^2
    Optimal f: if c_j > 0: f = sqrt(v_j/c_j) - S (if positive), else push to 0.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0  # normalized

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Precompute: M_j = a[j]*a[j], c_j = 2*(1-a[j]) (for normalized a, S=1)
    # v_j[k] = (a*a)[k] - a[j]*a[k-j] for k != 2j
    #          (a*a)[k] - a[j]^2 for k = 2j

    for passe in range(max_passes):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        S = float(np.sum(a))
        order = rng.permutation(n)

        for j in order:
            if time.time() > time.monotonic() + time_budget:
                break
            aj = a[j]
            # c_j = 2*(S - a[j]) = 2*(1 - a[j]) for normalized
            c_j = 2.0 * (S - aj)
            if c_j < 1e-15:
                continue

            # v_j[k] = conv[k] - aj*a[k-j] (for k-j in [0,n-1])
            #         conv[k] (for k-j not in [0,n-1])
            # The max of v_j is at the center: v_j[n-1] = sum(a^2) - aj^2
            # But we need the actual max of (c_j*f + v_j[k]) / (S+f)^2
            # For the "peak" approximation, the worst k has the largest v_j[k]
            # v_j[k] = conv[k] - aj*a[k-j] if k-j in [0,n-1], else conv[k]
            # The max v_j[k] is at most conv_max (when a[k-j] is small)
            # For a proper calculation, find max v_j[k]:
            # v_j[k] = conv[k] - aj * (a[k-j] if 0<=k-j<n else 0)

            # Compute v_j max efficiently:
            # v_j[k] = conv[k] - aj * a[k-j] for valid k-j
            # The maximum is at the center k=n-1: v_j[n-1] = conv[n-1] - aj*a[n-1-j]
            # But the actual worst case might be elsewhere.
            # For efficiency, use the upper bound: max(v_j) <= conv_max
            # And the lower bound on the "critical" value: v_j[n-1] = sum(a^2) - aj^2

            # Use the center value as the critical constraint (usually the binding one)
            v_center = float(np.sum(a * a)) - aj * aj

            # Optimal f for the center constraint:
            # minimize (c_j*f + v_center) / (S+f)^2
            # d/df = c_j*(S+f)^2 - 2*(c_j*f+v_center)*(S+f) = 0
            # (S+f)[c_j*(S+f) - 2*(c_j*f+v_center)] = 0
            # c_j*S + c_j*f - 2*c_j*f - 2*v_center = 0
            # c_j*S - c_j*f - 2*v_center = 0
            # f = S - 2*v_center/c_j

            f_opt = S - 2.0 * v_center / c_j

            # Clamp to [0, S] (since a[j] + f should be in [0, S] for non-negativity)
            # Actually a[j] new value = aj + f, should be in [0, S] (since total sum stays S)
            # Wait: if we change a[j] to a[j]+f, the sum changes by f.
            # But we're working with normalized sum=1, so we need to renormalize.
            # Let's work with unnormalized: new a[j] = f_new, sum becomes S + (f_new - aj)
            # Objective: 2n * max(conv_new) / (S_new)^2
            # This is getting complex. Let's just try f=0 (no change) and f=f_opt and a few others.

            # Simpler approach: try a few candidate new values for a[j]
            candidates = [0.0, aj * 0.5, aj, aj * 1.5]
            # Add the optimal from the center constraint
            f_candidate = max(0.0, min(S, f_opt))
            candidates.append(f_candidate)
            # Add values based on the actual max lag
            conv_max = float(np.max(conv))
            if c_j > 1e-15 and conv_max > 1e-15:
                # f that makes (c_j*f + v_j_at_max) / (S+f)^2 = current obj / (2n)
                # This is approximate; let's just scan
                for mult in [0.01, 0.1, 0.5, 1.0, 2.0, 5.0]:
                    candidates.append(aj * mult)

            best_f = aj
            best_local_obj = compute_obj(a)  # current
            # Actually, let's compute the obj for each candidate more efficiently
            # For candidate value w (new a[j]):
            # new_sum = S + (w - aj)
            # new_conv[k] = conv[k] + (w-aj)*(a[k-j] + a[k-j]) for k != 2j
            #               conv[k] + 2*(w-aj)*a[j] + (w-aj)^2 for k = 2j
            # = conv[k] + (w-aj)*(a[k-j]+a[j]) for k != 2j (using old a[j])
            # Actually: new_conv[k] = conv[k] + delta*(a[k-j] + a[j]) for k != 2j
            # where delta = w - aj, and for k=2j: + delta*2*a[j] + delta^2

            delta_candidates = [w - aj for w in candidates]
            # Deduplicate
            seen = set()
            unique_deltas = []
            for d in delta_candidates:
                key = round(d, 8)
                if key not in seen and abs(d) > 1e-10:
                    seen.add(key)
                    unique_deltas.append(d)
            unique_deltas.append(0.0)  # no change

            best_delta = 0.0
            best_obj_cand = best_local_obj

            for delta in unique_deltas:
                w = aj + delta
                if w < -1e-15:
                    continue
                w = max(0.0, w)
                new_S = S + (w - aj)
                if new_S < 1e-10:
                    continue

                # Compute new max conv efficiently
                # new_conv[k] = conv[k] + delta * (a[k-j] + a[j]) for k != 2j
                #             conv[k] + delta * 2*a[j] + delta^2 for k = 2j
                # The max is at the argmax of (conv[k] + delta * adj[k]) where adj adjusts
                # For efficiency, just compute the full new conv (O(n) with the formula)
                # Actually for n=800, computing full conv via FFT takes ~50us,
                # but we're doing this for each coordinate (800 times per pass),
                # so 800 * 50us = 40ms per pass. That's fine.

                a_new = a.copy()
                a_new[j] = w
                # Quick objective without full conv:
                # We can upper-bound: new_max <= conv_max + |delta| * max(a) * 2
                # For exact: compute new conv
                if abs(delta) > 1e-8:
                    new_conv = compute_conv(a_new)
                    new_max = float(np.max(new_conv))
                    obj_cand = 2.0 * n * new_max / (new_S ** 2)
                else:
                    obj_cand = best_local_obj

                if obj_cand < best_obj_cand - 1e-14:
                    best_obj_cand = obj_cand
                    best_delta = delta

            if best_delta != 0.0:
                a[j] = max(0.0, a[j] + best_delta)
                S = float(np.sum(a))
                if S > 1e-10:
                    a = a / S
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()

        if verbose and passe % 5 == 0:
            print(f"    CD pass {passe:3d}: obj={best_obj:.10f}")

    return best_a


def smoothed_grad(a, tau):
    """Compute gradient of smoothed max of conv(a,a) with log-sum-exp smoothing."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=1.5):
    """Nesterov accelerated gradient descent with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000, 8000]

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 3000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            tau_iters += 1
            lr = max(1e-8, lr * 0.99998)

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def active_set_targeting(a, time_budget, verbose=False, seed=42, max_iters=100):
    """Targeted reduction of the peak autocorrelation lags."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.998
        max_lags = np.where(conv >= threshold)[0]

        # Compute averaged gradient over active lags
        subgrad = np.zeros(n)
        for lag in max_lags:
            lo = max(0, lag - n + 1)
            hi = min(lag, n - 1)
            for i in range(lo, hi + 1):
                subgrad[i] += a[lag - i]
        subgrad /= len(max_lags)

        gn = np.linalg.norm(subgrad)
        if gn < 1e-15:
            break
        subgrad_norm = subgrad / gn

        improved = False
        for step in [0.001, 0.003, 0.008, 0.02, 0.05]:
            a_try = a - step * subgrad_norm
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            if it % 5 == 0:
                # LP step
                g_lp, _, _ = lp_solve(a)
                if g_lp is not None:
                    for t in [0.1, 0.2, 0.3, 0.5]:
                        a_try = (1 - t) * a + t * g_lp
                        a_try = np.maximum(a_try, 0)
                        st = float(np.sum(a_try))
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
            if not improved:
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: param-scan multi-start, momentum LP fixed-point,
    coordinate descent, active-set targeting, multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # --- Load parent ---
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={n_current}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_current = 800
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.10f}")

    # === Phase 1: Parameter scan for diverse initializations ===
    print(f"\n=== Phase 1: Parameter Scan (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.06, 45.0)
        a_scan, obj_scan = target_param_scan(n_current, start_time + alloc, rng)
        if obj_scan < best_score:
            best_score = obj_scan
            best_seq = a_scan.copy()
            print(f"  Param scan: obj={best_score:.10f} **")
        else:
            print(f"  Param scan: obj={obj_scan:.10f}")

    # === Phase 2: Quick GD warm-up ===
    print(f"\n=== Phase 2: GD Warm-up ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.05, 40.0)
        a_gd = staircase_gd(best_seq, time_budget=alloc,
                            tau_levels=[5, 30, 150], verbose=False)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD warm-up: obj={best_score:.10f} **")
        else:
            print(f"  GD warm-up: obj={obj_gd:.10f}")

    # === Phase 3: Main LP fixed-point with momentum (heaviest) ===
    print(f"\n=== Phase 3: Momentum LP Fixed-Point ===")
    remaining = deadline - time.monotonic()
    if remaining > 150:
        lp_budget = min(remaining * 0.35, 420.0)
        a_lp = lp_fixed_point_momentum(best_seq, time_budget=lp_budget,
                                       verbose=True, seed=42, max_iters=300)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP momentum: obj={best_score:.10f} **")
        else:
            print(f"  LP momentum: obj={obj_lp:.10f}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.12, 100.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc,
                                  verbose=True, seed=77, max_passes=30)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord desc: obj={best_score:.10f} **")
        else:
            print(f"  Coord desc: obj={obj_cd:.10f}")

    # === Phase 5: Active-set targeting ===
    print(f"\n=== Phase 5: Active-Set Targeting ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.10, 80.0)
        a_as = active_set_targeting(best_seq, time_budget=alloc,
                                    verbose=False, seed=314, max_iters=80)
        obj_as = compute_obj(a_as)
        if obj_as < best_score:
            best_score = obj_as
            best_seq = a_as.copy()
            print(f"  Active set: obj={best_score:.10f} **")
        else:
            print(f"  Active set: obj={obj_as:.10f}")

    # === Phase 6: Second LP round (different seed) ===
    print(f"\n=== Phase 6: Second LP Round ===")
    remaining = deadline - time.monotonic()
    if remaining > 100 and len(best_seq) <= 1500:
        lp_budget2 = min(remaining * 0.15, 150.0)
        a_lp2 = lp_fixed_point_momentum(best_seq, time_budget=lp_budget2,
                                        verbose=True, seed=999, max_iters=150)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP round2: obj={best_score:.10f} **")
        else:
            print(f"  LP round2: obj={obj_lp2:.10f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_current = len(best_seq)
        n_options = sorted(set([
            n_current + 50, n_current + 100, n_current + 200,
            n_current - 50, n_current - 100,
            600, 700, 800, 900, 1000, 1100, 1200, 1500
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_current]
        n_options.sort(key=lambda x: abs(x - n_current))
        n_options = n_options[:4]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            a_try = resample(best_seq, n_try)
            # Quick param scan for this n
            alloc_scan = min(remaining * 0.08, 15.0)
            if alloc_scan > 5:
                a_scan_try, obj_scan_try = target_param_scan(n_try,
                                                              time.monotonic() + alloc_scan, rng)
                if obj_scan_try < compute_obj(a_try):
                    a_try = a_scan_try
            # Brief LP
            remaining = deadline - time.monotonic()
            if n_try <= 1200:
                alloc_lp = min(remaining * 0.15, 40.0)
                if alloc_lp > 10:
                    a_try = lp_fixed_point_momentum(a_try, time_budget=alloc_lp,
                                                    verbose=False, seed=n_try, max_
