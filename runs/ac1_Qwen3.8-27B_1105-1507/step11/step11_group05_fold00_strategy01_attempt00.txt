def project_simplex(v):
    """Project v onto the simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u*np.arange(1, n+1) > cssv)[0]
    if not rho.any():
        return np.ones(n) / n
    rho = rho[-1] + 1
    theta = (np.sum(u[:rho]) - 1) / rho
    proj = np.maximum(v - theta, 0)
    return proj / proj.sum()
