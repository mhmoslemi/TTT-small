def gd_smoothed(a, time_budget, tau_list=None, verbose=False):
    """Nesterov GD with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_list is None:
        tau_list = [5, 20, 60, 200, 600, 1500]
    
    for tau in tau_list:
        lr = 1.0 / np.sqrt(n)
        v = np.zeros(n)
        beta = 0.9
        iters = 0
        
        while time.monotonic() - start < time_budget:
            if iters >= 3000:
                break
            y = np.maximum(a + beta*v, 0)
            sy = y.sum()
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            conv = compute_conv(y)
            mx = conv.max()
            shifted = tau * (conv - mx)
            shifted = np.clip(shifted, -500, 500)
            expv = np.exp(shifted)
            pe = expv / expv.sum()
            
            # Gradient: d/dy [sum_k pe_k * conv_k]
            # conv_k = sum_i y_i * y_{k-i}
            # d(conv_k)/dy_j = y_{k-j} + y_{j-k} (if valid)
            # Use FFT for efficiency
            N = 1
            while N < 2*n + len(conv):
                N <<= 1
            from scipy.fft import fft, ifft
            P = fft(pe, N)
            Y = fft(y, N)
            cross = ifft(P * np.conj(Y)).real
            g = 2.0 * cross[:n]
            
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            
            v = beta * v - lr * g / gn
            a_new = np.maximum(y + v, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / sn
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
            iters += 1
            lr *= 0.9999
        
        if verbose:
            print(f"  GD tau={tau}: obj={best_obj:.10f}, iters={iters}")
    
    return best_a

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP fixed-point with geometric mean blending, momentum acceleration,
dual-guided perturbation, Nesterov GD with log-sum-exp smoothing, and multi-n exploration.
Key novelties over previous: geometric blending, momentum on LP direction, 
asymmetric initializations, and adaptive coordinate perturbation.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    an = a / s
    conv = compute_conv(an)
    return 2.0 * n * float(np.max(conv))


def build_lp_matrix(a):
    """Build constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, prefer_solver=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M, dual)."""
    n = len(a)
    A = build_lp_matrix(a)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    conv_constraints = A @ g_var <= M_var
    sum_constraint = cp.sum(g_var) == 1
    constraints = [conv_constraints, sum_constraint]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = []
    if prefer_solver:
        solvers.append(prefer_solver)
    for s in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        if s not in solvers:
            solvers.append(s)

    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_norm = g_val / sg
                    # Try to extract dual variables for the convolution constraints
                    dual_vals = None
                    try:
                        dual_vals = np.array(conv_constraints.dual_value).flatten()
                        dual_vals = np.maximum(dual_vals, 0)
                    except Exception:
                        dual_vals = None
                    return g_norm, float(M_var.value), dual_vals
        except Exception:
            continue
    return None, None, None

                # Random coordinate perturbation
                idx = rng.integers(0, n, size=max(5, n // 20))
                a[idx] += 0.02 * rng.standard_normal(len(idx))
            elif strategy == 7:
                # Exponential taper perturbation (asymmetric)
                taper = np.exp(-np.linspace(0, 2, n))
                a = a + 0.01 * taper * rng.standard_normal(n)
            elif strategy == 8:
                # Nesterov-style lookahead perturbation
                a = a + 0.008 * (a - momentum_g) + 0.01 * rng.standard_normal(n)
            elif strategy == 9:
                # Frequency-domain perturbation
                Af = fft(a, n)
                Af_mag = np.abs(Af)
                Af_phase = np.angle(Af)
                # Boost a few random low frequencies
                for _ in range(2):
                    f = rng.integers(1, n // 4)
                    Af_mag[f] *= 1.05
                    Af_mag[n - f] *= 1.05
                a = ifft(Af_mag * np.exp(1j * Af_phase)).real[:n]

            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()

        if stall > 35:
            # Major restart
            strength = min(0.06, 0.01 * (stall // 7))
            a = best_a + strength * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            t_current = 0.1
            momentum_g = a.copy()

    return best_a


def gd_smoothed(a, time_budget, tau_list=None, verbose=False):
    """Nesterov accelerated gradient descent with log-sum-exp smoothed max."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_list is None:
        tau_list = [5, 20, 60, 200, 600, 1500]

    N_fft = 1
    while N_fft < 2 * n + 2 * n - 1:
        N_fft <<= 1

    for tau in tau_list:
        lr = 1.5 / np.sqrt(n)
        v = np.zeros(n)
        beta = 0.85
        iters = 0
        max_iters = 5000

        while time.monotonic() - start < time_budget:
            if iters >= max_iters:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute smoothed gradient
            conv = compute_conv(y)
            mx = float(np.max(conv))
            shifted = tau * (conv - mx)
            shifted = np.clip(shifted, -500, 500)
            expv = np.exp(shifted)
            se = float(np.sum(expv))
            if se < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = expv / se

            # Gradient via FFT: d/dy_j [sum_k p_k * conv_k] = 2 * (p conv y)_j
            P = fft(p, N_fft)
            Y = fft(y, N_fft)
            cross = ifft(P * np.conj(Y)).real
            g = 2.0 * cross[:n]

            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break

            v = beta * v - lr * g / gn
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            sn = float(np.sum(a_new))
            if sn < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / sn

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            iters += 1
            lr *= 0.99995

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={iters}")

    return best_a


def active_set_direct(a, time_budget, seed=42, max_iters=300):
    """
    Directly target the maximum lags with subgradient steps,
    combined with LP blending for escape.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        rem = time_budget - (time.monotonic() - start)
        if rem < 2:
            break

        conv = compute_conv(a)
        mx = float(np.max(conv))
        # Find active lags (within 0.1% of max)
        threshold = mx * 0.999
        active = np.where(conv >= threshold)[0]

        # Compute subgradient: average over active lags
        sg = np.zeros(n)
        for lag in active:
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    sg[i] += 2.0 * a[j]
        sg /= len(active)

        sgn = float(np.linalg.norm(sg))
        if sgn < 1e-15:
            break
        sg = sg / sgn

        # Try decreasing steps
        improved = False
        for step in [0.0003, 0.0007, 0.0015, 0.003, 0.006, 0.012]:
            a_try = a - step * sg
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Try LP escape every few iterations
            if it % 5 == 0:
                g_lp, _, _ = lp_solve(a)
                if g_lp is not None:
                    for t in [0.1, 0.2, 0.3, 0.5]:
                        a_try = (1 - t) * a + t * g_lp
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
            if not improved:
                a += 0.008 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    sv = float(np.sum(a_new))
    if sv < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / sv


def make_init(n, rng, mode):
    """Create structured initializations with emphasis on asymmetric shapes."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        # Exponentially decaying (strongly asymmetric)
        x = np.arange(n)
        a = np.exp(-0.7 * x / (n / 3))
        return a / a.sum()
    elif mode == 6:
        # Bimodal asymmetric (front-heavy)
        x = np.linspace(0, 1, n)
        a = np.exp(-25 * (x - 0.15) ** 2) + 0.4 * np.exp(-40 * (x - 0.65) ** 2)
        return a / a.sum()
    elif mode == 7:
        # Power-law decay
        x = np.arange(1, n + 1)
        a = 1.0 / x ** 0.8
        return a / a.sum()
    elif mode == 8:
        # Asymmetric multi-lobe
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 1.3 * x - 0.5) + 0.2 * np.cos(2 * np.pi * 2.7 * x + 0.3)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 9:
        # Triangular but skewed
        x = np.arange(n)
        peak = int(0.3 * n)
        a = np.maximum(1.0 - np.abs(x - peak) / (n - peak), 0.01)
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining improved LP fixed-point (with geometric blending and momentum),
    Nesterov GD, active-set refinement, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            s = float(np.sum(parent))
            if s > 0.01:
                best_seq = parent / s
                best_score = compute_obj(best_seq)
                print(f"Parent: n={best_seq.size}, obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default uniform: n=800, obj={best_score:.10f}")

    n_cur = len(best_seq)

    # === Phase 1: Quick screening of diverse inits ===
    print("\n=== Phase 1: Init Screening ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        for mode in [5, 6, 7, 9, 8, 1, 2, 4]:
            rem = deadline - time.monotonic()
            if rem < 12:
                break
            a_init = make_init(n_cur, rng, mode)
            alloc = min(rem * 0.025, 6.0)
            if alloc < 3:
                break
            a_s = gd_smoothed(a_init, time_budget=alloc, tau_list=[10, 50])
            obj_s = compute_obj(a_s)
            if obj_s < best_score:
                best_score = obj_s

                print(f"  Init {mode}: obj={best_score:.10f} **")
            else:
                print(f"  Init {mode}: obj={obj_s:.6f}")

    # === Phase 2: GD warmup ===
    print("\n=== Phase 2: GD Warmup ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        alloc = min(rem * 0.06, 40.0)
        a2 = gd_smoothed(best_seq, time_budget=alloc, tau_list=[5, 20, 60, 200], verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  GD warmup: obj={best_score:.10f} **")

    # === Phase 3: Main LP fixed-point ===
    print("\n=== Phase 3: Main LP Fixed-Point ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        lp_budget = min(rem * 0.32, 420.0)
        a3 = lp_fixed_point_v2(best_seq, time_budget=lp_budget, seed=42, max_iters=500, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Main LP: obj={best_score:.10f} **")
        else:
            print(f"  Main LP: obj={obj3:.10f}")

    # === Phase 4: Active-set refinement ===
    print("\n=== Phase 4: Active-Set Refinement ===")
    rem = deadline - time.monotonic()
    if rem > 50 and best_seq.size <= 1500:
        alloc = min(rem * 0.08, 70.0)
        a4 = active_set_direct(best_seq, time_budget=alloc, seed=123, max_iters=300)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Active-set: obj={best_score:.10f} **")
        else:
            print(f"  Active-set: obj={obj4:.10f}")

    # === Phase 5: Second LP round (different seed) ===
    print("\n=== Phase 5: Second LP Round ===")
    rem = deadline - time.monotonic()
    if rem > 80 and best_seq.size <= 1500:
        assign = min(rem * 0.15, 120.0)
        a5 = lp_fixed_point_v2(best_seq, time_budget=assign, seed=555, max_iters=300, verbose=True)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  LP round 2: obj={best_score:.10f} **")
        else:
            print(f"  LP round 2: obj={obj5:.10f}")

    # === Phase 6: Multi-n exploration ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    rem = deadline - time.monotonic()
    if rem > 70:
        n_options = sorted(set([
            n_cur - 50, n_cur - 100, n_cur + 50, n_cur + 100, n_cur + 150,
            600, 700, 900, 1000, 1100, 1200
        ]))
        n_options = [x for x in n_options if 400 <= x <= 1300 and x != n_cur]
        n_options.sort(key=lambda x: abs(x - n_cur))
        n_options = n_options[:4]
        if n_options:
            print(f"  Candidates: {n_options}")

        for n_try in n_options:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            a_try = resample_seq(best_seq, n_try)
            alloc1 = min(rem * 0.08, 20.0)
            if alloc1 > 8:
                a_try = gd_smoothed(a_try, time_budget=alloc1, tau_list=[10, 50, 200])
            rem = deadline - time.monotonic()
            if n_try <= 1200 and rem > 35:
                alloc2 = min(rem * 0.08, 30.0)
                if alloc2 > 10:
                    a_try = lp_fixed_point_v2(a_try, time_budget=alloc2, seed=n_try, max_iters=80)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_cur = n_try
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:

    # === Phase 7: Final intensive LP polish ===

        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj7:.10f}")

    # === Phase 8: Final high-tau GD ===
    print("\n=== Phase 8: Final GD Polish ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        alloc = min(rem * 0.50, 50.0)
        a8 = gd_smoothed(best_seq, time_budget=alloc, tau_list=[100, 400, 1000, 2500, 5000])
        obj8 = compute_obj(a8)
        if obj8 < best_score:
            best_score = obj8
            best_seq = a8.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Final validation and return
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isinf(val):
        result = [1.0] * 100
        best_score = 2.0
    elif val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
