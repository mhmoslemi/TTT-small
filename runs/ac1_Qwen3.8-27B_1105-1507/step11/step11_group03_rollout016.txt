    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    N = 1
    while N < 2 * n:
        N <<= 1
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        mean_c = float(np.mean(conv[n//4:n*3//4]))  # middle portion
        
        if max_c <= mean_c * 1.01:
            break  # already fairly flat
        
        # Find the worst lags (above median)
        median_c = np.median(conv)
        bad_lags = np.where(conv > median_c * 1.1)[0]
        
        if len(bad_lags) == 0:
            break
        
        # For each bad lag, create a "reduction" direction
        # We want to reduce a[i]*a[i+k] for bad lag k
        # Direction: reduce a[i] proportional to a[i+k] for bad k
        direction = np.zeros(n)
        for k in bad_lags[:20]:  # limit to 20 worst lags
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                direction[j] += a[k - j]  # gradient contribution for this lag
        
        direction = direction / (np.linalg.norm(direction) + 1e-10)
        
        # Try to move in negative gradient direction
        for step in [0.001, 0.005, 0.01, 0.02, 0.05]:
            a_try = project_normalize(a - step * direction)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
        
        if it % 100 == 0 and verbose:
            print(f"      EQ it{it}: obj={best_obj:.10f}, max/mean={max_c/mean_c:.4f}")
    
    return best_a

def resample_seq(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_normalize(a_new)

def make_initial(n, mode=0):
    """Create initial sequences."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(0.5 * x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n-1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(1, n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-0.5 * x * x)
        return a / a.sum()
    elif mode == 5:
        # Optimal low-autocorrelation: raised cosine squared
        x = np.linspace(0, 1, n)
        a = (0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))) ** 2
        return a / a.sum()
    elif mode == 6:
        # Flat top with strong taper
        t = 0.15
        a = np.ones(n)
        for i in range(n):
            d = min(i, n - 1 - i) / n
            if d < t:
                a[i] = np.cos(np.pi * (t - d) / (2 * t)) ** 2
        return a / a.sum()
    else:
        rng = np.random.default_rng(mode)
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()

        parent = np.clip(parent, 0, None)

            s = float(np.sum(parent))
            if s > 0.01:
                parent = parent / s
