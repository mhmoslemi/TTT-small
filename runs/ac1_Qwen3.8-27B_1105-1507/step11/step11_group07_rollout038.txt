"""
L-BFGS with softmax parameterization + LP polishing for step function optimization.
Uses the identity that the objective depends on a only through conv(a,a),
and the softmax reparameterization to handle the simplex constraint smoothly.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_true(a):
    """Compute true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def soft_max_value_and_grad_conv(a, beta):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. a.
    Returns (soft_max_val, grad_conv) where grad_conv = d(S)/d(a) with S = soft_max(conv(a,a)).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    exp_sum = float(np.sum(exp_vals))
    p = exp_vals / exp_sum
    soft_max_val = c_max + np.log(exp_sum) / beta
    
    # Gradient of S w.r.t. a[i] = 2 * sum_m a[m] * p[m+i] (cross-correlation)
    a_rev = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    grad_conv = 2.0 * q
    
    return soft_max_val, grad_conv


def softmax_obj_grad(x, beta):
    """
    Objective and gradient for L-BFGS.
    x: log-parameter (shape n), a = softmax(x)
    Returns (objective_value, gradient_wrt_x)
    """
    a = np.exp(x - x.max())
    a = a / a.sum()
    n = len(a)
    sm_val, grad_conv = soft_max_value_and_grad_conv(a, beta)
    # Chain rule through softmax:
    # df/dx[j] = 2n * a[j] * (grad_conv[j] - dot(grad_conv, a))
    mean_gc = float(np.dot(grad_conv, a))
    dx = 2.0 * n * a * (grad_conv - mean_gc)
    obj = 2.0 * n * sm_val
    return obj, dx


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=120.0, max_iters=40, verbose=False):
    """LP-based polishing: iteratively find best direction and blend."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj_true(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try multiple blend factors
        best_t = 0.0
        best_obj_try = best_obj
        for t_try in [0.3, 0.5, 0.7, 0.9, 1.0, 1.2]:
            a_try = (1 - t_try) * a + t_try * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj_true(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t_try
        
        if best_t > 0 and best_obj_try < best_obj - 1e-14:
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            best_obj = best_obj_try
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 5:
            # Perturb and restart
            if stall <= 9:
                noise = rng.normal(0, 0.003 * (stall - 4), n)
                a = project_simplex(best_a + noise)
            else:
                x = np.linspace(0, 1, n)
                a_div = 1.0 / (0.05 + 4.0 * (2 * x - 1) ** 2)
                a = project_simplex(0.5 * a_div + 0.5 * best_a)
            stall = 0
    
    return best_a, best_obj


def lbfgs_optimize(a_init, beta, time_budget=60.0, verbose=False):
    """
    Run L-BFGS on the softmax-parameterized objective.
    a_init: initial point on the simplex (sum=1, non-negative)
    beta: soft-max parameter
    Returns: (best_a, best_obj)
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    
    # Convert to log-space
    x0 = np.log(a_init)
    
    start = time.monotonic()
    
    # Custom callback to check time
    state = {'done': False}
    
    def callback(xk):
        if time.monotonic() - start >= time_budget:
            state['done'] = True
        return state['done']
    
    result = minimize(
        softmax_obj_grad, x0, args=(beta,),
        method='L-BFGS-B',
        jac=True,
        callback=callback,
        options={'maxiter': 2000, 'maxfun': 5000, 'ftol': 1e-14, 'gtol': 1e-10,
                 'maxls': 50, 'maxcor': 50}
    )
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    
    obj = compute_obj_true(a_opt)
    
    if verbose:
        elapsed = time.monotonic() - start
        print(f"    L-BFGS (beta={beta}): obj={obj:.8f}, "
              f"iters={result.nit}, time={elapsed:.1f}s, success={result.success}")
    
    return a_opt, obj

def u_shaped_init(n, sharpness=3.0):

"""
L-BFGS with softmax parameterization + LP polishing for step function optimization.
Uses continuation in beta (low to high) for the soft-max approximation of the peak,
with FFT-based gradient computation. Combines second-order optimization with
LP-based direction finding for final refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_true(a):
    """Compute true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def soft_max_value_and_grad_conv(a, beta):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. a.
    Returns (soft_max_val, grad_conv) where grad_conv[i] = d(S)/d(a[i]).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    exp_sum = float(np.sum(exp_vals))
    p = exp_vals / exp_sum
    soft_max_val = c_max + np.log(exp_sum) / beta

    # Gradient: dS/da[i] = 2 * sum_m a[m] * p[m+i] (cross-correlation via FFT)
    a_rev = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    grad_conv = 2.0 * q

    return soft_max_val, grad_conv


def softmax_obj_grad(x, beta):
    """
    Objective and gradient for L-BFGS.
    x: log-parameter (shape n), a = softmax(x)
    Returns (objective_value, gradient_wrt_x)
    """
    a = np.exp(x - x.max())
    a = a / a.sum()
    n = len(a)
    sm_val, grad_conv = soft_max_value_and_grad_conv(a, beta)
    # Chain rule through softmax:
    # df/dx[j] = 2n * a[j] * (grad_conv[j] - dot(grad_conv, a))
    mean_gc = float(np.dot(grad_conv, a))
    dx = 2.0 * n * a * (grad_conv - mean_gc)
    obj = 2.0 * n * sm_val
    return obj, dx


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=120.0, max_iters=40, verbose=False):

    n = len(a)

    best_a = a.copy()

def lbfgs_optimize(a_init, beta, time_budget=60.0, verbose=False):
    """
    Run L-BFGS on the softmax-parameterized objective.
    a_init: initial point on the simplex (sum=1, non-negative)
    beta: soft-max parameter
    Returns: (best_a, best_obj)
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()

    x0 = np.log(a_init)
    start = time.monotonic()
    state = {'done': False}

    def callback(xk):
        if time.monotonic() - start >= time_budget:
            state['done'] = True
        return state['done']

    try:
        result = minimize(
            softmax_obj_grad, x0, args=(beta,),
            method='L-BFGS-B',
            jac=True,
            callback=callback,
            options={'maxiter': 3000, 'maxfun': 8000, 'ftol': 1e-15,
                     'gtol': 1e-12, 'maxls': 50, 'maxcor': 60}
        )
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")
        return a_init.copy(), compute_obj_true(a_init)

    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    obj = compute_obj_true(a_opt)

    if verbose:
        elapsed = time.monotonic() - start
        print(f"    L-BFGS (beta={beta:.0f}): obj={obj:.8f}, "
              f"nit={result.nit}, nfev={result.nfev}, time={elapsed:.1f}s")

    return a_opt, obj

        # Find the i that most contributes to conv[k_star]
        # conv[k_star] = sum_j a[j] * a[k_star - j]
        valid_j = [j for j in range(n) if 0 <= k_star - j < n]
        if not valid_j:
            break
        contribs = {j: a[j] * a[k_star - j] for j in valid_j}
        # Find pair to adjust
        j_sorted = sorted(contribs, key=lambda j: -contribs[j])

        improved = False
        for j in j_sorted[:5]:
            i = k_star - j
            if i == j:
                continue
            if i < 0 or i >= n or j < 0 or j >= n:
                continue
            # Try moving mass from the larger of a[i], a[j] to the smaller
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i

            for frac in [0.01, 0.05, 0.1, 0.2]:
                amount = a[src] * frac
                if amount <= 0:
                    continue
                a_try = a.copy()
                a_try[src] -= amount
                a_try[dst] += amount
                if a_try[src] < 0:
                    continue
                s = float(np.sum(a_try))
                if s < 0.01:
                    continue
                obj_try = compute_obj_true(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try random pair swap
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                amount = min(a[i], a[j]) * 0.05
                if amount < 1e-8:
                    continue
                a_try = a.copy()
                a_try[i] -= amount
                a_try[j] += amount
                s = float(np.sum(a_try))
                if s < 0.01:
                    continue
                obj_try = compute_obj_true(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: L-BFGS with softmax parameterization (continuation in beta)
    + LP polishing + coordinate descent + diverse restarts + length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj_true(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj_true(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: L-BFGS low beta (smooth, fast convergence) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: L-BFGS beta=10, t={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 100.0)
        a1, obj1 = lbfgs_optimize(best_seq, beta=10.0, time_budget=alloc, verbose=True)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  ** obj={best_score:.8f}")

    # === Phase 2: L-BFGS medium beta ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 2: L-BFGS beta=30, t={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 100.0)
        a2, obj2 = lbfgs_optimize(best_seq, beta=30.0, time_budget=alloc, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  ** obj={best_score:.8f}")

    # === Phase 3: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: LP Polish, t={remaining:.0f}s ===")
        alloc = min(remaining * 0.18, 150.0)
        a3, obj3 = lp_polish(best_seq, time_budget=alloc, max_iters=40, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  ** obj={best_score:.8f}")

    # === Phase 4: L-BFGS high beta (sharp) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: L-BFGS beta=80, t={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 100.0)
        a4, obj4 = lbfgs_optimize(best_seq, beta=80.0, time_budget=alloc, verbose=True)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  ** obj={best_score:.8f}")

    # === Phase 5: LP Polish round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 5: LP Polish r2, t={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 100.0)
        a5, obj5 = lp_polish(best_seq, time_budget=alloc, max_iters=30, verbose=True)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  ** obj={best_score:.8f}")

    # === Phase 6: Coordinate descent polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Coord Descent, t={remaining:.0f}s ===")
        alloc = min(remaining * 0.08, 30.0)
        a6, obj6 = coordinate_descent_polish(best_seq, time_budget=alloc)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  ** obj={best_score:.8f}")

    # === Phase 7: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 7: Diverse Restarts, t={remaining:.0f}s ===")
        n_cur = len(best_seq)

        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            mode = r % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best for warm-starting
            blend = 0.4 + 0.1 * r
            a_r = project_simplex((1.0 - blend) * a_r + blend * best_seq)

            # Quick L-BFGS
            alloc_sgd = min(remaining * 0.20, 60.0)
            if alloc_sgd > 15:
                a_r, obj_r = lbfgs_optimize(a_r, beta=30.0, time_budget=alloc_sgd)

            # Quick LP
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.15, 40.0)
                a_r, obj_r = lp_polish(a_r, time_budget=alloc_lp, max_iters=15, verbose=False)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): ** obj={best_score:.8f}")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 8: Length Exploration, t={remaining:.0f}s ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best + 50, n_best + 100, n_best + 200,
            n_best - 50, 600, 700, 900, 1000, 1100
        ]))
        n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_best) >= 30]
        n_options = n_options[:4]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))

            alloc_sgd = min(remaining * 0.35, 50.0)
            if alloc_sgd > 15:
                a_try, obj_try = lbfgs_optimize(a_try, beta=30.0, time_budget=alloc_sgd)

            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.25, 35.0)
                a_try, obj_try = lp_polish(a_try, time_budget=alloc_lp, max_iters=15, verbose=False)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 9: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 9: Final Polish, t={remaining:.0f}s ===")
        # L-BFGS with very high beta
        alloc1 = min(remaining * 0.35, 80.0)
        if alloc1 > 15:
            a_f, obj_f = lbfgs_optimize(best_seq, beta=150.0, time_budget=alloc1, verbose=True)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final L-BFGS: ** obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 25:
            alloc2 = min(remaining * 0.5, 60.0)
            a_f2, obj_f2 = lp_polish(best_seq, time_budget=alloc2, max_iters=25, verbose=True)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LP: ** obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc3 = min(remaining * 0.4, 20.0)
            a_f3, obj_f3 = coordinate_descent_polish(best_seq, time_budget=alloc3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final CoordDesc: ** obj={best_score:.8f}")

    # === Phase 10: Last-ditch random perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for attempt in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            noise = rng.normal(0, 0.002, len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.35, 15.0)
            if alloc > 5:
                a_p, obj_p = lp_polish(a_p, time_budget=alloc, max_iters=8, verbose=False)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Last-ditch {attempt}: ** obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj_true(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
