                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value), probs.status if hasattr(prob, 'status') else prob.status
        except Exception:
            continue
    return None, None, None


def line_search(a, h, n_coarse=50, n_fine=30, t_max=1.5):
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * h
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    if 0.0 < best_t < t_max:
        step_coarse = t_max / n_coarse
        lo = max(0.0, best_t - 2 * step_coarse)
        hi = min(t_max, best_t + 2 * step_coarse)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * h
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    
    # Also try multiplicative step
    s_a = float(np.sum(a))
    if s_a > 1e-10:
        a_norm = a / s_a
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            a_mult = a_norm * (1.0 + t * (h - a_norm))
            a_mult = np.maximum(a_mult, 0)
            s_m = float(np.sum(a_mult))
            if s_m > 0.01:
                a_mult = a_mult / s_m
                obj = compute_obj(a_mult)
                if obj < best_obj:
                    best_obj = obj
                    best_t = -t  # negative t indicates multiplicative
    
    return best_t, best_obj


def smooth_objective_and_grad(x, T=0.01):
    """Log-sum-exp smoothed objective with softmax parameterization."""
    e = np.exp(x - x.max())
    a = e / e.sum()
    n = len(a)
    conv = compute_conv(a)
    # Clip for numerical stability
    conv_clipped = np.clip(conv, -1e10, 1e10)
    max_conv = conv_clipped.max()
    if max_conv > 0:
        lse = max_conv + T * np.log(np.sum(np.exp((conv_clipped - max_conv) / T)))
        f = 2.0 * n * lse
    else:
        f = 2.0 * n * max_conv
    
    # Gradient with respect to a
    weights = np.exp((conv_clipped - max_conv) / T)
    weights = weights / weights.sum()
    # grad_f_wrt_a = 2*n * sum_k weights[k] * grad_a(b[k])
    n_conv = len(conv_clipped)
    grad_a = np.zeros(n)
    for k in range(n_conv):
        w = weights[k]
        if w < 1e-8:
            continue
        # db[k]/da[i] = 2*a[k-i] if 0<=k-i<n
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad_a[i] += 2.0 * w * a[j]
    
    grad_f_wrt_a = 2.0 * n * grad_a
    
    # Chain rule: grad_x = grad_a^T @ (da/dx)
    # da[i]/dx[j] = a[i]*(delta(i,j) - a[j])
    # So grad_x[j] = sum_i grad_a[i] * a[i]*(delta(i,j) - a[j])
    #              = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
    dot = float(np.dot(grad_f_wrt_a, a))
    grad_x = grad_f_wrt_a * a - a * dot
    
    return f, grad_x
