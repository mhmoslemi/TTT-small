Improved step function optimization using:
1. FFT-accelerated soft-max gradient descent with adaptive beta

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = (u * rho_candidates) > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_gradient_active(a, active_set):
    """
    Compute the subgradient of max(conv(a,a)) at the active set.
    grad_i = 2 * sum_{k in active_set} a_{k-i} (where valid)
    This is the exact subgradient when we know the active set.
    """
    n = len(a)
    conv = compute_conv(a)
    # Active set: indices where conv is close to max
    M = float(np.max(conv))
    # Use the provided active_set (indices into conv array)
    # grad_i = 2 * sum_{k in active_set, 0 <= k-i < n} a_{k-i}
    # This is a cross-correlation of a with the indicator of active_set
    indicator = np.zeros(2 * n - 1)
    for k in active_set:
        if 0 <= k < 2 * n - 1:
            indicator[k] = 1.0
    # cross-correlation: q_i = sum_k indicator[k] * a_{k-i}
    # Use FFT
    a_rev = a[::-1]
    size = 1
    while size < n + len(indicator) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    I = rfft(indicator, size)
    full = irfft(A * I, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def kkt_solve(a_init, n_active=None, tol=1e-8, max_iter=200):
    """
    Solve KKT conditions: find a such that for the active set S,
    2 * sum_{k in S} a_{k-i} = lambda for all i with a_i > 0.
    
    Uses iterative active-set method.
    """
    a = project_simplex(a_init.astype(np.float64))
    n = len(a)
    
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Identify active set
        if n_active is not None:
            threshold = M - n_active * 1e-6
        else:
            threshold = M - 1e-6
        
        active_mask = conv >= threshold
        active_set = np.where(active_mask)[0]
        
        if len(active_set) == 0:
            active_mask = conv >= M - 0.01 * M
            active_set = np.where(active_mask)[0]
        
        # Compute subgradient at active set
        grad = compute_gradient_active(a, active_set)
        
        # For KKT: grad_i = lambda for all i with a_i > 0
        # So lambda = average of grad over support of a
        support = a > 1e-10
        if not np.any(support):
            break
        lam = float(np.mean(grad[support]))
        
        # Check KKT residual
        residual = np.abs(grad[support] - lam)
        max_res = float(np.max(residual)) if len(residual) > 0 else 0
        
        # Check complementarity: for i with a_i = 0, grad_i <= lambda
        zeros = ~support
        if np.any(zeros):
            comp_violation = float(np.max(grad[zeros] - lam))
        else:
            comp_violation = 0
        
        if max_res < tol and comp_violation < tol:
            break
        
        # Update: move a in direction that equalizes grad on support
        # a_new_i = a_i - mu * (grad_i - lam) for i in support
        # For i not in support: a_new_i = max(0, a_i - mu * grad_i)
        mu = 0.1
        a_new = a.copy()
        a_new[support] -= mu * (grad[support] - lam)
        a_new[zeros] -= mu * np.maximum(grad[zeros], 0)
        a = project_simplex(a_new)
    
    return a


def soft_max_gd_v2(a, time_budget=200.0, beta_start=10.0, beta_end=100.0,

    Improved soft-max gradient descent with:
    - Adaptive beta
    - Periodic KKT polishing
    - Optional symmetrization

    """

        a = project_simplex(a)

    best_obj = compute_obj(a)

    stall = 0

        size = 1

        P = rfft(p, size)
        full = irfft(A * P, size)

        grad = 2.0 * q

        v = momentum * v + grad
        a_new = a - lr * v

        a_new = project_simplex(a_new)

            best_it = it
            stall = 0

            if stall > 50:
                v = np.zeros(n)
                stall = 0
                a = best_a.copy()

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def grad_active_fft(a, S):
    """Compute gradient 2*sum_{k in S} a_{k-i} for all i using FFT."""
    n = len(a)
    c = np.zeros(2 * n - 1)
    S_valid = S[(S >= 0) & (S < 2 * n - 1)]
    c[S_valid] = 1.0
    
    # Need N >= n + (2n-1) - 1 = 3n - 2
    N = 1
    while N < 3 * n:
        N <<= 1
    
    A = rfft(a, N)
    C = rfft(c, N)
    R = irfft(np.conjugate(A) * C, N)
    g = R[:n]
    return 2.0 * g

Now, for the KKT solver, the key step is:
1. Compute gradient at active set (using FFT)
2. For support T: grad[i] = lambda for all i in T
3. This means: the gradient is constant on the support

The "KKT equation" is:
2 * sum_{k in S} a[k-i] = lambda for all i in T

This is a LINEAR system in the unknowns {a[i] : i in T} and lambda!

Let me write it as a matrix equation:
For each i in T: sum_{k in S} a[k-i] = lambda/2

Let T = {t_0, t_1, ..., t_{p-1}} (sorted)
Let S = {s_0, s_1, ..., s_{m-1}}

For i = t_j: sum_{l=0}^{m-1} a[s_l - t_j] = lambda/2 (where a[x] = 0 if x not in [0,n-1])

So the (j, k) entry of the matrix is 1 if s_l - t_j = t_k for some l, summed over all l.
Actually, it's: M[j, k] = number of l such that s_l - t_j = t_k (i.e., s_l = t_j + t_k)

And the equation is: sum_k M[j, k] * a[t_k] = lambda/2 for all j.

Plus: sum_k a[t_k] = 1.

This is a (p+1) x (p+1) linear system!

The matrix M has entries:
M[j, k] = |{l : s_l = t_j + t_k}|

This can be computed efficiently: for each pair (j, k), check if t_j + t_k is in S.

If |T| = p and |S| = m, building M takes O(p^2 * log(m)) (using a set for S).
Solving the system takes O(p^3).

For p ~ n = 800, that's 800^3 = 512M operations, which might be a bit slow but doable.

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_polish(a, time_budget=100.0, max_iters=40, verbose=False):
    """LP-based polishing: find best response direction and blend."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0
        a_norm = a / float(np.sum(a))
        A = build_conv_matrix(a_norm)
        
        g = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g <= M_var, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        g_val = None
        for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        g_val = g_val / s
                    break
            except Exception:
                continue
        
        if g_val is None:
            break
        
        # Line search for optimal blend
        best_t = 0.0
        best_obj_t = best_obj
        for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t_try) * a + t_try * g_val
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t_try
        
        if best_t > 0 and best_obj_t < best_obj - 1e-12:
            a_new = (1 - best_t) * a + best_t * g_val
            a_new = project_simplex(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        if verbose:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 4:
            noise = rng.normal(0, 0.005 * (stall - 2), n)
            a = project_simplex(best_a + noise)
            stall = 0
    
    return best_a, best_obj


def arcsine_init(n, alpha=1.0):
    """Arcsine-like initialization with tunable sharpness."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / ((1 - (2 * x - 1) ** 2) ** (alpha / 2.0) + 1e-6)
    return project_simplex(a)


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=float)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n / 4) ** 2) / (2 * (n / 6) ** 2)) + \
            np.exp(-((idx - 3 * n / 4) ** 2) / (2 * (n / 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / np.cosh(2.5 * (2 * x - 1))
    elif mode == 6:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 4
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(5, n)] = 0.4
        a[max(0, n - 5):n - 1] = 0.4
        if n > 6:
            a[1:-1] += 0.03
    elif mode == 8:
        # Bezier-like: mass concentrated at edges with smooth falloff
        t = np.linspace(0, 1, n)
        a = t ** 3 + (1 - t) ** 3 + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimization combining FFT gradient descent,
    KKT linear system solver, LP polishing, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = arcsine_init(n_current, alpha=1.0)
        best_score = compute_obj(best_seq)
        print(f"Default arcsine init (n={n_current}): obj={best_score:.8f}")
    else:
        # Symmetrize parent
        symm_parent = symmetrize(best_seq)
        symm_parent = project_simplex(symm_parent)
        obj_symm = compute_obj(symm_parent)
        if obj_symm < best_score:
            best_score = obj_symm
            best_seq = symm_parent.copy()
            print(f"  Symmetrized parent: obj={best_score:.8f}")
    
    # Try arcsine as alternative
    a_arc = arcsine_init(n_current, alpha=1.0)
    obj_arc = compute_obj(a_arc)
    if obj_arc < best_score:
        best_score = obj_arc
        best_seq = a_arc.copy()
        print(f"Arcsine init: obj={best_score:.8f}")
    
    # Try different alpha
    for alpha in [0.7, 1.3, 2.0]:
        a_try = arcsine_init(n_current, alpha=alpha)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Arcsine alpha={alpha}: obj={best_score:.8f}")
    
    # === Phase 1: Soft-max GD (smooth) ===
    remaining = deadline - time.monotonic()
    if remaining > 150:
        print(f"\n=== Phase 1: SMax GD (beta 5->30), ~{remaining:.0f}s left ===")
        alloc = min(remaining * 0.2, 250.0)
        a1, obj1 = soft_max_gd(
            best_seq, time_budget=alloc,
            beta_start=5.0, beta_end=30.0,
            lr_start=0.012, momentum=0.9, symm=True,
            kkt_every=400, verbose=True, seed=seed
        )
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 done: obj={obj1:.8f} (no improvement)")
    
    # === Phase 2: KKT solver (exact for active set) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 2: KKT Solver, ~{remaining:.0f}s left ===")
        a2, obj2 = kkt_solve(best_seq, max_outer=80, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 done: obj={obj2:.8f} (no improvement)")
    
    # === Phase 3: Soft-max GD (sharp) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 3: SMax GD (beta 30->100), ~{remaining:.0f}s left ===")
        alloc = min(remaining * 0.2, 200.0)
        a3, obj3 = soft_max_gd(
            best_seq, time_budget=alloc,
            beta_start=30.0, beta_end=100.0,
            lr_start=0.005, momentum=0.92, symm=True,
            kkt_every=300, verbose=True, seed=seed + 1
        )
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 done: obj={obj3:.8f} (no improvement)")
    
    # === Phase 4: KKT + LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: KKT + LP Polish, ~{remaining:.0f}s left ===")
        # KKT
        a4, obj4 = kkt_solve(best_seq, max_outer=50, verbose=False)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
        
        # LP
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc_lp = min(remaining * 0.4, 120.0)
            a4b, obj4b = lp_polish(best_seq, time_budget=alloc_lp, max_iters=30, verbose=True)
            if obj4b < best_score:
                best_score = obj4b
                best_seq = a4b.copy()
                print(f"  Phase4 LP: obj={best_score:.8f} **")
    
    # === Phase 5: Multi-length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 5: Length Exploration, ~{remaining:.0f}s left ===")
        n_best = len(best_seq)
        
        # Generate candidate lengths
        n_options = set()
        for delta in [-100, -50, 50, 100, 200, 300]:
            n_try = n_best + delta
            if 300 <= n_try <= 2000:
                n_options.add(n_try)
        # Also try some "round" numbers
        for n_round in [500, 600, 700, 850, 900, 1000, 1200]:
            if abs(n_round - n_best) >= 50:
                n_options.add(n_round)
        
        n_options = sorted(n_options)[:6]
        print(f"  Trying lengths: {n_options}")
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            
            # Interpolate
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(a_try)
            
            # Quick optimize: SMax GD + KKT
            alloc_sgd = min(remaining * 0.3, 50.0)
            if alloc_sgd > 10:
                a_try, _ = soft_max_gd(
                    a_try, time_budget=alloc_sgd,
                    beta_start=15.0, beta_end=50.0,
                    lr_start=0.008, momentum=0.9, symm=True,
                    kkt_every=300, verbose=False, seed=seed + 500
                )
            
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try, _ = kkt_solve(a_try, max_outer=30, verbose=False)
            
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 6: Diverse Restarts, ~{remaining:.0f}s left ===")
        n_cur = len(best_seq)
        
        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            
            mode = r % 9
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best
            blend = 0.5 + 0.1 * r
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)
            a_r = symmetrize(a_r)
            a_r = project_simplex(a_r)
            
            alloc = min(remaining * 0.35, 60.0)
            if alloc > 10:
                a_r, _ = soft_max_gd(
                    a_r, time_budget=alloc,
                    beta_start=10.0, beta_end=50.0,
                    lr_start=0.008, momentum=0.9, symm=True,
                    kkt_every=300, verbose=False, seed=seed + 700 + r * 10
                )
                a_r, _ = kkt_solve(a_r, max_outer=25, verbose=False)
            
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
    
    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Final Polish, ~{remaining:.0f}s left ===")
        
        # KKT with tight tolerance
        a_f, obj_f = kkt_solve(best_seq, max_outer=40, verbose=False)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final KKT: obj={best_score:.8f} **")
        
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.5, 60.0)
            a_f2, obj_f2 = lp_polish(best_seq, time_budget=alloc, max_iters=20, verbose=False)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
        
        # One more KKT
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_f3, obj_f3 = kkt_solve(best_seq, max_outer=30, verbose=False)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Last KKT: obj={best_score:.8f} **")
    
    # === Phase 8: Tiny perturbations + KKT ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for attempt in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            noise = rng.normal(0, 0.002, len(best_seq))
            a_p = symmetrize(best_seq + noise)
            a_p = project_simplex(a_p)
            a_p, obj_p = kkt_solve(a_p, max_outer=15, verbose=False)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Perturb {attempt}: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
