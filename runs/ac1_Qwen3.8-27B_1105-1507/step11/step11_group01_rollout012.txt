"""
Optimizes step function coefficients using parametric initialization via differential
evolution, peak-targeted coordinate descent, block moves, and systematic n-sweep.
Key innovations: (1) low-dimensional cosine-series parameterization optimized via DE,
(2) peak-lag-targeted coordinate updates, (3) block coordinate descent for escaping
local minima, (4) fine-grained n exploration with shape-preserving resampling.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2n*max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def parametric_sequence(params, n):
    """
    Generate sequence from cosine-series parameters.
    params = [c1, c2, ..., cm, p1, p2, ..., pm]
    a[j] = max(0, 1 + sum_i c_i * cos(2*pi*i*j/n + p_i))
    """
    m = len(params) // 2
    j = np.arange(n)
    a = np.ones(n)
    for i in range(m):
        c = params[i]
        p = params[m + i]
        a += c * np.cos(2.0 * np.pi * (i + 1) * j / n + p)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def de_parametric(n, m, time_budget, seed=42, initial_params=None):
    """
    Differential evolution over the parametric space.
    Returns best params and best sequence.
    """
    rng = np.random.default_rng(seed)
    dim = 2 * m
    pop_size = max(20, min(60, 3 * dim))
    
    # Initialize population
    pop = np.zeros((pop_size, dim))
    pop[0] = np.zeros(dim)  # uniform (all c=0)
    for i in range(1, pop_size):
        pop[i] = rng.uniform(-0.5, 0.5, dim)
    
    if initial_params is not None:
        pop[1] = initial_params[:dim]
    
    # Evaluate initial population
    objs = np.zeros(pop_size)
    best_seqs = [None] * pop_size
    for i in range(pop_size):
        a = parametric_sequence(pop[i], n)
        objs[i] = compute_obj(a)
        best_seqs[i] = a
    
    best_idx = np.argmin(objs)
    best_obj = objs[best_idx]
    best_seq = best_seqs[best_idx].copy()
    best_params = pop[best_idx].copy()
    
    start = time.monotonic()
    F = 0.8
    CR = 0.7
    gen = 0
    
    while time.monotonic() - start < time_budget:
        gen += 1
        # Adaptive F
        F_eff = F * (1.0 - 0.3 * min(1.0, gen / 200.0))
        
        for i in range(pop_size):
            if time.monotonic() - start >= time_budget:
                break
            
            r1, r2 = rng.choice(pop_size - 1, 2, replace=False)
            if r1 >= i: r1 += 1
            if r2 >= i: r2 += 1
            if r1 == r2:
                r2 = (r2 + 1) % pop_size
                if r2 == i: r2 = (r2 + 1) % pop_size
            
            # Mutation
            v = pop[best_idx] + F_eff * (pop[r1] - pop[r2])
            
            # Crossover
            trial = pop[i].copy()
            cross_idx = rng.integers(0, dim)
            for j in range(dim):
                if rng.random() < CR or j == cross_idx:
                    trial[j] = v[j]
            
            # Evaluate
            a = parametric_sequence(trial, n)
            obj = compute_obj(a)
            
            if obj <= objs[i]:
                pop[i] = trial
                objs[i] = obj
                best_seqs[i] = a
                if obj < best_obj:
                    best_obj = obj
                    best_seq = a.copy()
                    best_params = trial.copy()
                    best_idx = i
        
        if gen % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    DE parametric gen {gen}: best_obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_params, best_seq, best_obj

                    best_x = x
            
            # Golden section
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 1.5
            if x_hi - x_lo < 1e-15:
                x_hi = x_lo + 1e-10
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            
            for _ in range(50):
                if x_hi - x_lo < 1e-14 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            
            if abs(x_opt - v) < 1e-16:
                continue
            
            delta = x_opt - v
            a[i] = x_opt
            S += delta
            
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki
            
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj_now = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
        
        # Stall detection
        if sweep % 10 == 0:
            if best_obj < prev_best - 1e-13:
                prev_best = best_obj
                stall = 0
            else:
                stall += 1
        
        if verbose and sweep % 10 == 0:
            print(f"    CD+ sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}")
    
    return best_a


def block_coordinate_descent(a, time_budget=100.0, seed=42, block_size=4):
    """
    Block coordinate descent: optimize groups of `block_size` coordinates jointly
    using a simple local search (random perturbation + acceptance).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Choose a block of coordinates
        start_idx = rng.integers(0, max(1, n - block_size))
        idx = np.arange(start_idx, start_idx + block_size)
        idx = idx[idx < n]
        
        if len(idx) < 2:
            continue
        
        # Try to optimize this block via random directions
        best_block_obj = best_obj
        best_block_a = best_a.copy()
        
        for trial in range(50):
            # Random perturbation of the block
            perturbation = rng.normal(0, S / n * 0.3, len(idx))
            a_try = a.copy()
            a_try[idx] = np.maximum(a_try[idx] + perturbation, 0)
            
            S_try = float(np.sum(a_try))
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_block_obj:
                best_block_obj = obj_try
                best_block_a = a_try.copy()
        
        if best_block_obj < best_obj - 1e-15:
            best_obj = best_block_obj
            best_a = best_block_a.copy()
            a = best_a
            conv = compute_conv(a)
        
        it += 1
        if it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    BCD it {it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a
