"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact coordinate descent (1D bisection on subproblem), LP alternating minimization,
and FFT-based subgradient descent for exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    r = np.sort(v)[::-1]
    cv = 1.0 / np.arange(1, n + 1)
    tmp = r + cv
    is_correct = (tmp > 0)
    rho = np.where(is_correct)[0][-1]
    tau = (r[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - tau, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a_norm*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def exact_coord_opt(a, j, conv, max_unaffected):
    """
    Find the optimal value v for coordinate j such that after updating a[j]=v
    and renormalizing, the objective 2n*max(conv) is minimized.
    
    Uses bisection on the objective value V, checking feasibility via interval intersection.
    
    Returns (v_opt, obj_opt) where obj_opt is the new normalized max conv.
    """
    n = len(a)
    a_j = a[j]
    
    # After changing a[j] to v, before renormalization:
    # S = 1 - a_j + v (new sum)
    # t = 1/S
    # 
    # For k != 2j (and 0<=k-j<n): new_conv[k] = conv[k] + 2*(S-1)*a[k-j] = conv[k] + 2*(1/t - 1)*a[k-j]
    #   => new_conv[k] * t^2 = conv[k]*t^2 + 2*(1-t)*a[k-j]*t^2... 
    #
    # Let me redo. S = 1/S_new where S_new = 1-a_j+v. t = 1/S_new.
    # v = (1-t)*S_new... no. v = S_new - 1 + a_j = (1/t) - 1 + a_j
    #
    # Before renorm: conv_before[k] for k != 2j: conv[k] + 2*(v-a_j)*a[k-j]
    #   = conv[k] + 2*((1/t)-1+a_j-a_j)*a[k-j]
    #   = conv[k] + 2*((1/t)-1)*a[k-j]
    #   = conv[k] + 2*a[k-j]/t - 2*a[k-j]
    #
    # After renorm (divide by S^2 = 1/t^2):
    # new_normalized[k] = (conv[k] + 2*a[k-j]/t - 2*a[k-j]) * t^2
    #   = conv[k]*t^2 + 2*a[k-j]*t - 2*a[k-j]*t^2
    #   = (conv[k] - 2*a[k-j])*t^2 + 2*a[k-j]*t
    #
    # For k = 2j: conv_before[2j] = conv[2j] + v^2 - a_j^2
    #   = conv[2j] + ((1/t)-1+a_j)^2 - a_j^2
    #   = conv[2j] + (1/t)^2 - 2(1/t)(1-a_j) + (1-a_j)^2 - a_j^2
    #   = conv[2j] + 1/t^2 - 2(1-a_j)/t + 1 - 2a_j + a_j^2 - a_j^2
    #   = conv[2j] + 1/t^2 - 2(1-a_j)/t + 1 - 2a_j
    # After renorm: * t^2
    #   = conv[2j]*t^2 + 1 - 2(1-a_j)*t + (1-2a_j)*t^2
    #   = (conv[2j] + 1 - 2a_j)*t^2 - 2(1-a_j)*t + 1
    #
    # For k where k-j < 0 or k-j >= n (i.e., a[k-j] doesn't exist):
    # conv_before[k] = conv[k] (unchanged)
    # After renorm: conv[k]*t^2
    
    # So for each k, the normalized new value as a function of t is:
    # f_k(t) = A_k * t^2 + B_k * t + C_k
    #
    # where:
    # - If 0 <= k-j < n and k != 2j: A_k = conv[k] - 2*a[k-j], B_k = 2*a[k-j], C_k = 0
    # - If k == 2j: A_k = conv[2j] + 1 - 2*a_j, B_k = -2*(1-a_j), C_k = 1
    # - Otherwise: A_k = conv[k], B_k = 0, C_k = 0
    
    # We want to minimize max_k f_k(t) subject to t > 0 and v = 1/t - 1 + a_j >= 0
    # i.e., 1/t >= 1 - a_j, i.e., t <= 1/(1-a_j) (if a_j < 1)
    # Also t > 0.
    # And we should also bound t below: t >= t_min where t_min corresponds to some max v.
    # Let's say v_max = max(5/n, 0.5), so t_min = 1/(1 - a_j + v_max)
    
    # Bisection on V: find min V such that exists t in [t_min, t_max] with f_k(t) <= V for all k.
    
    # For a given V, feasible t for each k:
    # A_k * t^2 + B_k * t + C_k <= V
    # A_k * t^2 + B_k * t + (C_k - V) <= 0
    # This is a quadratic inequality. The feasible set is an interval (possibly empty).
    
    t_max = 1.0 / max(1.0 - a_j, 1e-10)
    v_max = max(5.0 / n, 0.5)
    t_min = 1.0 / (1.0 - a_j + v_max)
    
    def feasible(V):
        """Check if there exists t in [t_min, t_max] with f_k(t) <= V for all k."""
        lo = t_min
        hi = t_max
        
        # For each k, find the interval of t where f_k(t) <= V
        # f_k(t) = A_k * t^2 + B_k * t + C_k
        # A_k * t^2 + B_k * t + (C_k - V) <= 0
        
        for k in range(2 * n - 1):
            if k == 2 * j:
                A_k = conv[2 * j] + 1.0 - 2.0 * a_j
                B_k = -2.0 * (1.0 - a_j)
                C_k = 1.0
            elif 0 <= k - j < n:
                A_k = conv[k] - 2.0 * a[k - j]
                B_k = 2.0 * a[k - j]
                C_k = 0.0
            else:
                A_k = conv[k]
                B_k = 0.0
                C_k = 0.0
            
            C_k -= V  # now we solve A_k*t^2 + B_k*t + C_k <= 0
            
            if A_k < -1e-15:
                # Quadratic opens downward: f_k(t) <= V is satisfied outside the roots
                # This means the constraint is NOT binding (it's satisfied for large |t|)
                # Actually, for a downward-opening parabola, f(t) <= V is satisfied
                # for t outside the interval between the roots.
                # Since our t is bounded, this constraint is either always satisfied
                # or satisfied at the boundaries.
                # Let's check: if discriminant < 0, always satisfied (since opens down).
                disc = B_k * B_k - 4.0 * A_k * C_k
                if disc <= 0:
                    # Always satisfied (or just touches)
                    continue
                sq = np.sqrt(disc)
                r1 = (-B_k - sq) / (2.0 * A_k)
                r2 = (-B_k + sq) / (2.0 * A_k)
                if r1 > r2:
                    r1, r2 = r2, r1
                # Feasible: t <= r1 or t >= r2
                # Intersect with [lo, hi]:
                # [lo, r1] union [r2, hi]
                # This is a non-convex constraint, which complicates things.
                # For simplicity, let's just check if lo or hi satisfies it.
                # Actually, let's handle this case:
                # The infeasible region is (r1, r2). We need [lo,hi] not to be a subset of (r1,r2).
                if lo >= r2 or hi <= r1:
                    continue  # entirely feasible
                elif lo > r1 and hi < r2:
                    return False  # entirely infeasible
                else:
                    # Partially feasible. This is tricky for the interval intersection.
                    # For now, let's be conservative and just check endpoints.
                    # Actually, let's skip this constraint (it's rare).
                    pass
                continue
            
            elif abs(A_k) < 1e-15:
                # Linear: B_k * t + C_k <= 0
                if abs(B_k) < 1e-15:
                    if C_k > 0:
                        return False
                    continue
                # B_k * t <= -C_k
                if B_k > 0:
                    # t <= -C_k / B_k
                    hi = min(hi, -C_k / B_k)
                else:
                    # t >= -C_k / B_k
                    lo = max(lo, -C_k / B_k)
                if lo > hi + 1e-15:
                    return False
                continue
            
            else:
                # A_k > 0: quadratic opens upward
                # f_k(t) <= V means t is between the two roots
                disc = B_k * B_k - 4.0 * A_k * C_k
                if disc < 0:
                    # No real roots, and since opens upward and min > V... 
                    # Actually if disc < 0 and A_k > 0, then f_k(t) > 0 for all t (if C_k - V > 0)
                    # or f_k(t) < 0 for all t... 
                    # Since A_k > 0 and disc < 0: the quadratic is always positive or always negative.
                    # A_k * t^2 + B_k * t + C_k: if A_k > 0 and disc < 0, it's always positive
                    # (since the minimum value is (4*A_k*C_k - B_k^2)/(4*A_k) = -disc/(4*A_k) > 0)
                    # So f_k(t) > V for all t => infeasible
                    return False
                
                sq = np.sqrt(disc)
                r1 = (-B_k - sq) / (2.0 * A_k)
                r2 = (-B_k + sq) / (2.0 * A_k)
                if r1 > r2:
                    r1, r2 = r2, r1
                # Feasible: t in [r1, r2]
                lo = max(lo, r1)
                hi = min(hi, r2)
                if lo > hi + 1e-15:
                    return False
        
        return lo <= hi + 1e-15
    
    # Bisection on V
    # Upper bound: current objective (all zeros is always feasible with V = max(conv)*t_min^2 + ...)
    # Lower bound: 0 (or some small positive number)
    
    # Upper bound for V: the current value at t = 1 (i.e., no change, v = a_j)
    # At t=1: f_k(1) = A_k + B_k + C_k which should equal conv[k] (the current normalized value)
    # Let's verify: for k != 2j: (conv[k]-2a[k-j]) + 2a[k-j] + 0 = conv[k] ✓
    # For k=2j: (conv[2j]+1-2a_j) - 2(1-a_j) + 1 = conv[2j]+1-2a_j-2+2a_j+1 = conv[2j] ✓
    # For others: conv[k] ✓
    # So V_upper = max(conv) (current max)
    
    V_upper = float(np.max(conv))
    V_lower = 0.0
    
    # First check if the current point (t=1) is feasible (it should be with V = max(conv))
    # Bisection
    for _ in range(60):
        V_mid = 0.5 * (V_lower + V_upper)
        if feasible(V_mid):
            V_upper = V_mid
        else:
            V_lower = V_mid
    
    V_opt = V_upper
    
    # Now find the optimal t (any t that achieves V_opt, or the one that minimizes max)
    # For the actual update, we want the t that minimizes the max.
    # Let's find t by bisection within the feasible set at V_opt + small epsilon.
    V_check = V_opt * (1 + 1e-10) + 1e-12
    
    # Find the feasible interval at V_check
    lo_final = t_min
    hi_final = t_max
    
    for k in range(2 * n - 1):
        if k == 2 * j:
            A_k = conv[2 * j] + 1.0 - 2.0 * a_j
            B_k = -2.0 * (1.0 - a_j)
            C_k = 1.0 - V_check
        elif 0 <= k - j < n:
            A_k = conv[k] - 2.0 * a[k - j]
            B_k = 2.0 * a[k - j]
            C_k = -V_check
        else:
            A_k = conv[k]
            B_k = 0.0
            C_k = -V_check
        
        if abs(A_k) < 1e-15:
            if abs(B_k) < 1e-15:
                if C_k > 0:
                    return a_j, V_upper  # infeasible, keep current
                continue
            if B_k > 0:
                hi_final = min(hi_final, -C_k / B_k)
            else:
                lo_final = max(lo_final, -C_k / B_k)
        elif A_k > 0:
            disc = B_k * B_k - 4.0 * A_k * C_k
            if disc < 0:
                return a_j, V_upper
            sq = np.sqrt(disc)
            r1 = (-B_k - sq) / (2.0 * A_k)
            r2 = (-B_k + sq) / (2.0 * A_k)
            if r1 > r2:
                r1, r2 = r2, r1
            lo_final = max(lo_final, r1)
            hi_final = min(hi_final, r2)
        else:
            # A_k < 0: skip (non-convex, usually not binding)
            continue
    
    if lo_final > hi_final:
        return a_j, V_upper
    
    # Use the t that minimizes the max f_k(t) - use midpoint of feasible interval
    t_opt = 0.5 * (lo_final + hi_final)
    
    # Convert t to v
    v_opt = 1.0 / t_opt - 1.0 + a_j
    v_opt = max(0.0, v_opt)
    
    return v_opt, V_opt


def exact_coordinate_descent(a, time_budget, seed=42, verbose=True):
    """Exact coordinate descent using 1D subproblem solution via bisection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    
    sweep = 0
    no_improve = 0
    max_no_improve = 3
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = np.random.default_rng(seed + sweep).permutation(n)
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            a_j = a[j]
            if a_j < 1e-12 and np.max(conv) < 1e-12:
                continue
            
            v_opt, obj_opt = exact_coord_opt(a, j, conv, 0.0)
            
            if v_opt != a_j and obj_opt < np.max(conv) - 1e-14:
                # Apply update
                S = 1.0 - a_j + v_opt
                if S < 1e-10:
                    continue
                a[j] = v_opt
                a = a / S
                conv = compute_conv(a)  # Recompute (expensive but safe)
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    ECD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")
        
        if not sweep_improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0
    
    return best_a


def subgradient_descent(a, time_budget, seed=42, verbose=True):
    """FFT-based subgradient descent with averaging and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    momentum = np.zeros(n)
    beta = 0.7
    lr = 0.5
    lr_decay = 0.995
    
    avg_a = a.copy()
    count = 0
    it = 0
    
    b_rev = a[::-1].copy()  # for FFT-based subgradient
    
    while time.monotonic() - start < time_budget - 2:
        it += 1
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        obj = 2.0 * n * max_val
        
        # Active set: indices where conv is within 1% of max
        threshold = max_val * 0.995
        active = np.where(conv >= threshold)[0]
        
        # Build p (soft indicator of active set)
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        
        # Compute subgradient via FFT
        # grad = 2 * (A^T @ p) where (A^T p)[j] = sum_m a[m] * p[m+j]
        # = conv(a_rev, p)[j + n - 1]
        b = a[::-1]
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        cross = irfft(rfft(b, size) * rfft(p, size), size)[:3 * n - 1]
        grad = 2.0 * cross[n - 1:2 * n - 1]
        
        # Update
        momentum = beta * momentum + (1.0 - beta) * grad
        a_new = a - lr * momentum
        a_new = project_simplex(a_new)
        
        avg_a = (count * avg_a + a_new) / (count + 1)
        count += 1
        
        # Check improvement
        if it % 10 == 0:
            obj_avg = 2.0 * n * float(np.max(compute_conv(avg_a)))
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = avg_a.copy()
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr *= lr_decay
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG iter {it}: obj={obj:.8f}, best={best_obj:.8f}, lr={lr:.4f}, |active|={len(active)}")
    
    # Final check with average
    obj_avg = 2.0 * n * float(np.max(compute_conv(avg_a)))
    if obj_avg < best_obj:
        best_a = avg_a.copy()
        best_obj = obj_avg
    
    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """LP alternating minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    prev_g = None
    iter_count = 0
    stall = 0
    
    while time.monotonic() - start < time_budget - 3:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1
        
        best_local_obj = best_obj
        best_local_a = a.copy()
        
        t_values = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                rng = np.random.default_rng(seed + iter_count)
                strength = 0.01 + 0.005 * min(stall, 10)
                pert = rng.standard_normal(n) * strength
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0
    
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 9):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
