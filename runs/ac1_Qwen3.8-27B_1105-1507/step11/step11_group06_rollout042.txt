"""
Improved optimization combining LP-based steps, smooth gradient descent with

Key improvements: SOCP-like smooth objective, better LP handling, wider n search.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build the matrix A such that (A @ g)[k] = (a * g)(k) for the full convolution."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # (a*g)(k) = sum_j a[j] * g[k-j]
        # So for fixed j, a[j] * g[k-j] contributes to row k
        # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        # But k is bounded by 0 to 2n-2
        k_start = max(0, j)
        k_end = min(m, j + n)
        # For row k, the contribution is a[j] * g[k-j]
        # k-j ranges from k_start - j to k_end - 1 - j
        g_start = k_start - j
        g_end = k_end - j
        A[k_start:k_end, g_start:g_end] = a[j]
    return A


def solve_lp(a):
    """Solve the LP to find the best direction g that minimizes the linearized convolution peak."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smooth_obj_and_grad(a, alpha=100.0):
    """Compute smooth approximation of max(conv) and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    # Log-sum-exp for numerical stability
    max_conv = float(np.max(conv))
    shifted = alpha * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    logsumexp = max_conv + np.log(np.sum(exp_shifted)) / alpha
    # Gradient: d/d a_i of (1/alpha) log sum_k exp(alpha * conv[k])
    # = (1/alpha) * sum_k alpha * exp(alpha * conv[k]) * d(conv[k])/da_i / sum_k exp(alpha * conv[k])
    # = sum_k exp(alpha * conv[k]) * d(conv[k])/da_i / sum_k exp(alpha * conv[k])
    weights = exp_shifted / np.sum(exp_shifted)  # These are the "softmax" weights
    # d(conv[k])/da_i = a[k-i] + a[i-k] (with bounds checking)
    # conv[k] = sum_j a[j] * a[k-j]
    # d(conv[k])/da[i] = a[k-i] + a[i-k] (if both indices valid)
    grad = np.zeros(n)
    for i in range(n):
        for k in range(2 * n - 1):
            j1 = k - i
            j2 = i - k
            if 0 <= j1 < n:
                grad[i] += weights[k] * a[j1]
            if 0 <= j2 < n:
                grad[i] += weights[k] * a[j2]
    return logsumexp, grad

        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, 0:k_end - k_start] = a[j]
    return A

                s = float(np.sum(g_val))
                if s > 1e-10:

"""
Hybrid LP + smooth gradient optimization for minimum peak polynomial convolution.
Uses sqrt-triangular initialization (matching sqrt of uniform polynomial),
CLARABEL LP solver, log-sum-exp smoothed subgradients, and aggressive
n-value exploration to beat the 1.5027 target.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A such that (A @ g)[k] = (a * g)[k] (full linear convolution)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. (a*g)(k) <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.clip(g_var.value, 0, None)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def sqrt_triangular_init(n, epsilon=0.5):
    """Initialize with coefficients approximating sqrt of triangular polynomial."""
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(idx + epsilon) + 1.0 / np.sqrt(n - 1.0 - idx + epsilon)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = sqrt_triangular_init(n)
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = sqrt_triangular_init(n, epsilon=1.0)
    elif mode == 7:
        a = 1.0 / (0.1 + 3.0 * np.abs(2 * x - 1) ** 3)
    else:
        a = u_shaped_init(n, sharpness=2.0)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def compute_smooth_subgrad(a, alpha=100.0):
    """Compute smoothed subgradient of max(conv) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = alpha * (conv - max_c)
    exp_s = np.exp(shifted)
    weights = exp_s / float(np.sum(exp_s))
    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * float(np.dot(a, weights[i:i + n]))
    return grad


def smooth_grad_descent(a, time_budget=10.0, alpha=100.0, lr0=0.01):
    """Run smooth gradient descent with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.7
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        grad = compute_smooth_subgrad(a, alpha=alpha)
        lr_t = lr0 / (1.0 + 0.01 * it)
        momentum = beta * momentum - lr_t * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            a = a_new
            patience += 1
            if patience > 200:
                a = best_a.copy()
                momentum = np.zeros(n)
                lr0 = max(0.001, lr0 * 0.5)
                patience = 0
        it += 1
    return best_a, best_obj


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def line_search(a, g, t_max=1.2):
    """Exhaustive line search for best t in (a, (1-t)*a + t*g)."""
    best_t = 0.0
    best_obj = compute_obj(a)
    # Coarse
    ts = np.linspace(0.0, t_max, 30)
    for t in ts:
        a_try = np.maximum((1.0 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    # Fine

"""
Hybrid LP + smooth gradient optimization for minimum peak polynomial convolution.
Uses sqrt-triangular initialization (matching sqrt of uniform polynomial),
CLARABEL LP solver, log-sum-exp smoothed subgradients, and aggressive
n-value exploration to beat the 1.5027 target.
Key ideas:
- SQRT triangular init: f[i] ~ 1/sqrt(i) + 1/sqrt(n-1-i) approximates sqrt of uniform poly
- LP linearization: at each step, fix one factor of convolution, LP on the other
- Smooth gradient escape: log-sum-exp smoothing when LP stalls
- Aggressive n exploration: try 300-2000 range
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A such that (A @ g)[k] = (a * g)[k] for full linear convolution."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. (a*g)(k) <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.clip(g_var.value, 0, None)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None

    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(idx + epsilon) + 1.0 / np.sqrt(n - 1.0 - idx + epsilon)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def compute_smooth_subgrad(a, alpha=100.0):
    """Compute smoothed subgradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = alpha * (conv - max_c)
    exp_s = np.exp(shifted)
    weights = exp_s / float(np.sum(exp_s))
    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * float(np.dot(a, weights[i:i + n]))
    return grad

                momentum = np.zeros(n)

            alloc = min(remaining * 0.25, 70.0)
            a_opt = optimize_hybrid(a_start, time_budget=alloc, max_iters=60)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Final intensive optimization at best n ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_final = len(best_seq)
        print(f"\n=== Phase 3: Final Polish at n={n_final} (budget={remaining:.0f}s) ===")
        a_sg, obj_sg = smooth_grad_descent(best_seq, time_budget=min(remaining * 0.15, 30.0), alpha=150.0, lr0=0.005)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  After SG: obj={best_score:.8f} **")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            alloc = min(remaining2 - 5, 200.0)
            a_final = optimize_hybrid(best_seq, time_budget=alloc, max_iters=100, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
            else:
                print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 4: Last-ditch squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_sq, obj_sq = smooth_grad_descent(best_seq, time_budget=min(remaining - 3, 15.0), alpha=200.0, lr0=0.003)
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_sq.copy()
            print(f"  Squeeze: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
