import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_step(a, K=5, top_fraction=0.01):
    """Compute subgradient of max conv, focusing on top K lags."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Find top lags
    if K == 'fraction':
        threshold = max_val * (1 - top_fraction)
        active_lags = np.where(conv >= threshold)[0]
    else:
        # Find top K lags
        idx = np.argsort(conv)[::-1]
        active_lags = idx[:K]
        # Also include all lags within 99% of max
        threshold = max_val * 0.99
        extra = np.where(conv >= threshold)[0]
        active_lags = np.unique(np.concatenate([active_lags, extra]))
    
    # Compute subgradient
    subgrad = np.zeros(n)
    for lag in active_lags:
        # d/d(a[i]) conv[lag] = 2*a[lag-i] if 0 <= lag-i < n
        lo = max(0, lag - (n - 1))
        hi = min(n - 1, lag)
        for i in range(lo, hi + 1):
            subgrad[i] += 2.0 * a[lag - i]
    subgrad /= len(active_lags)
    
    return subgrad, max_val, active_lags

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses hybrid fast subgradient with Nesterov momentum, adaptive active-set tracking,
periodic LP direction finding, spectral analysis, and multi-n exploration.
Key improvements over previous: much faster core iterations (no LP per step),
better momentum scheduling, adaptive active-set size, and more aggressive n-search.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    if s != 1.0:
        a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_subgradient(a, top_k=None, top_frac=None):
    """
    Compute subgradient of max_k conv(a,a)[k] w.r.t. a.
    Uses top_k lags or top_frac fraction of max value.
    Returns (subgradient, max_val, active_lags).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))

    if top_k is not None and top_k > 0:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        active_lags = idx
    elif top_frac is not None:
        threshold = max_val * (1.0 - top_frac)
        active_lags = np.where(conv >= threshold)[0]
    else:
        # Default: top 10 or 1% whichever is larger
        top_k_eff = max(10, n // 100)
        idx = np.argpartition(conv, -top_k_eff)[-top_k_eff:]
        active_lags = idx

    # Compute subgradient: g[i] = (2/|S|) * sum_{k in S} a[k-i] (if 0<=k-i<n)
    subgrad = np.zeros(n)
    for k in active_lags:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            # subgrad[i] += 2*a[k-i] for i in [i_start, i_end]
            # a[k-i] for i=i_start..i_end is a[k-i_start] down to a[k-i_end]
            subgrad[i_start:i_end + 1] += 2.0 * a[k - i_end:k - i_start + 1][::-1]

    subgrad /= len(active_lags)
    return subgrad, max_val, active_lags


def project_simplex(a):
    """Project onto probability simplex (non-negative, sum to 1)."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_solve_fast(a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def subgradient_gd(a, time_budget, verbose=False, seed=42,
                   initial_step=0.05, momentum_init=0.5,
                   top_k_init=20, lp_every=80, max_iters=100000):
    """
    Nesterov accelerated subgradient descent with adaptive active set.
    Periodically uses LP for better direction.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    v = np.zeros(n)
    step = initial_step
    momentum = momentum_init
    it = 0
    stall = 0
    consecutive_lp_fails = 0

    # Track best over a window for adaptive step
    window_obj = []

    while time.monotonic() - start < time_budget:
        if it >= max_iters:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        # Adaptive top_k: start broad, narrow as we converge
        progress = min(1.0, it / 5000.0)
        top_k = max(3, int(top_k_init * (1.0 - 0.7 * progress)))

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
            sy = 1.0
        y = y / sy

        # Compute subgradient at y
        g, max_val, active_lags = compute_subgradient(y, top_k=top_k)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break

        # Momentum update
        v = momentum * v - step * g / gn
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        window_obj.append(obj_new)
        if len(window_obj) > 50:
            window_obj.pop(0)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        # Adapt step size
        if stall > 20:
            step *= 0.7
            stall = 0
        elif it % 100 == 0 and it > 0:
            # Check if we're making progress
            if len(window_obj) >= 20:
                recent_avg = np.mean(window_obj[-20:])
                older_avg = np.mean(window_obj[:20]) if len(window_obj) >= 40 else recent_avg
                if recent_avg < older_avg - 1e-8:
                    step = min(step * 1.05, 0.5)

        # Periodic LP enhancement
        if lp_every > 0 and it % lp_every == 0 and it > 0:
            g_lp, M_lp = lp_solve_fast(a)
            if g_lp is not None:
                consecutive_lp_fails = 0
                # Line search along LP direction
                best_t_obj = best_obj
                best_t_a = best_a.copy()
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1.0 - t) * a + t * g_lp
                    a_try = project_simplex(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_t_a = a_try.copy()
                if best_t_obj < best_obj:
                    best_obj = best_t_obj
                    best_a = best_t_a.copy()
                    a = best_a.copy()
                    v = np.zeros(n)
                elif verbose and it % (lp_every * 5) == 0:
                    print(f"    LP it{it}: M_lp={M_lp:.6f}, no improvement")
            else:
                consecutive_lp_fails += 1
                if consecutive_lp_fails > 10:
                    lp_every = -1  # Disable LP

        # Occasional random restart from best
        if it % 500 == 250 and stall > 50:
            a = best_a + 0.01 * rng.standard_normal(n)
            a = project_simplex(a)
            v = np.zeros(n)
            step = initial_step * 0.5
            stall = 0

        # Update momentum (gradually increase)
        momentum = min(0.9, momentum_init + 0.4 * progress)

        it += 1

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    SubGD it{it:6d}: obj={best_obj:.10f}, step={step:.5f}, "
                  f"top_k={top_k}, mom={momentum:.3f}, t={elapsed:.1f}s")

    return best_a


def active_set_equalize(a, time_budget, verbose=False, seed=7, max_iters=2000):
    """
    Targeted active set equalization: try to make the top convolution values equal.
    Uses barycentric subgradient with adaptive weights.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = project_simplex(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    step = 0.01
    it = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        if time.monotonic() - start > time_budget - 0.5:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Find top lags and their values
        top_k = 8
        idx = np.argpartition(conv, -top_k)[-top_k:]
        idx = idx[np.argsort(conv[idx])[::-1]]  # Sort by value descending
        top_vals = conv[idx]

        # If top values are already nearly equal, use very small steps
        spread = top_vals[0] - top_vals[-1]
        if spread < max_val * 1e-5:
            step *= 0.5

        # Weighted subgradient: weight each lag by how far it is from the mean
        mean_top = np.mean(top_vals)
        weights = (top_vals - mean_top + spread * 0.5)
        weights = np.maximum(weights, 0.01)
        weights /= np.sum(weights)

        # Compute weighted subgradient
        subgrad = np.zeros(n)
        for w, k in zip(weights, idx):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end >= i_start:
                subgrad[i_start:i_end + 1] += 2.0 * w * a[k - i_end:k - i_start + 1][::-1]

        gn = float(np.linalg.norm(subgrad))
        if gn < 1e-15:
            break
        subgrad /= gn

        # Try multiple step sizes
        improved = False
        for s_try in [step, step * 0.5, step * 2.0, step * 0.1]:
            a_try = a - s_try * subgrad
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            step *= 0.8
            a = best_a.copy()
            # Small random perturbation to escape
            if rng.random() < 0.3:
                a = a + 0.005 * rng.standard_normal(n)
                a = project_simplex(a)

        it += 1
        if verbose and it % 200 == 0:
            print(f"    ASE it{it:4d}: obj={best_obj:.10f}, spread={spread:.6e}, step={step:.6f}")

    return best_a


def spectral_optimize(a, time_budget, verbose=False, seed=13):

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

def coordinate_descent(a, time_budget, verbose=False, max_sweeps=50):
    """
    Coordinate descent: optimize one coordinate at a time.
    For each a[i], find the best value given all others fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = project_simplex(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    order = np.arange(n)

    sweep = 0
    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        if time.monotonic() - start > time_budget - 1.0:
            break

        # Randomize order each sweep after first few
        if sweep > 2:
            rng.shuffle(order)

        for i in order:
            if time.monotonic() - start > time_budget - 0.5:
                break

            # Save original
            orig = a[i]
            other_sum = float(np.sum(a)) - orig

            # Try a few candidate values for a[i]
            # The optimal a[i] is where the max convolution is minimized
            # Binary search or golden section
            best_local = orig
            best_local_obj = best_obj

            # Try: 0, orig*0.5, orig, orig*1.5, orig*2, and their normalized versions
            candidates = [0.0, orig * 0.3, orig * 0.7, orig, orig * 1.3, orig * 1.7, orig * 2.0,
                          other_sum * 0.01, other_sum * 0.05]
            candidates = [c for c in candidates if c >= 0 and c < other_sum + 1.0]

            for c in candidates:
                a[i] = c
                s = float(np.sum(a))
                if s < 0.01:
                    a[i] = orig
                    continue
                obj_try = compute_obj(a)
                if obj_try < best_local_obj - 1e-14:
                    best_local_obj = obj_try
                    best_local = c

            a[i] = best_local
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            if best_local_obj < best_obj - 1e-14:
                best_obj = best_local_obj
                best_a = a.copy()

        sweep += 1
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep{swap:3d}: obj={best_obj:.10f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid subgradient + LP + spectral + coordinate descent.
    Multi-n exploration with adaptive time allocation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Try to load parent ---
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)

    # === Phase 1: Quick multi-n screening ===
    print(f"\n=== Phase 1: Multi-n Screening ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_candidates = sorted(set([
            n_current, n_current - 100, n_current - 50,
            n_current + 50, n_current + 100, n_current + 200,
            600, 700, 900, 1000, 1100, 1200, 1400
        ]))
        n_candidates = [n for n in n_candidates if 400 <= n <= 2000]

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            for mode in [0, 1, 5]:
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break
                a_init = make_init(n_try, mode, rng)
                # Quick GD
                alloc = min(remaining * 0.15, 15.0)
                if alloc < 3:
                    break
                a_screen = subgradient_gd(a_init, time_budget=alloc,
                                           seed=n_try + mode, lp_every=-1,
                                           initial_step=0.03, max_iters=5000)
                obj_screen = compute_obj(a_screen)
                if obj_screen < best_score:
                    best_score = obj_screen
                    best_seq = a_screen.copy()
                    n_current = n_try
                    print(f"  ** n={n_try} mode{mode}: obj={best_score:.10f}")
                else:
                    print(f"  n={n_try} mode{mode}: obj={obj_screen:.10f}")

    # === Phase 2: Main subgradient optimization on best ===
    print(f"\n=== Phase 2: Main Subgradient (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 200:
        alloc = min(remaining * 0.35, 500.0)
        a2 = subgradient_gd(best_seq, time_budget=alloc, verbose=True, seed=42,
                            initial_step=0.05, momentum_init=0.5,
                            top_k_init=20, lp_every=60, max_iters=200000)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 SubGD: obj={best_score:.10f} **")
        else:
            print(f"  Phase2 SubGD: obj={obj2:.10f} (no improvement)")

    # === Phase 3: Active set equalization ===
    print(f"\n=== Phase 3: Active Set Equalization ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(remaining * 0.15, 150.0)
        a3 = active_set_equalize(best_seq, time_budget=alloc, verbose=True, seed=7)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 ASE: obj={best_score:.10f} **")
        else:
            print(f"  Phase3 ASE: obj={obj3:.10f}")

    # === Phase 4: Spectral optimization ===
    print(f"\n=== Phase 4: Spectral Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(remaining * 0.10, 100.0)
        a4 = spectral_optimize(best_seq, time_budget=alloc, verbose=True, seed=13)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 Spectral: obj={best_score:.10f} **")
        else:
            print(f"  Phase4 Spectral: obj={obj4:.10f}")

    # === Phase 5: Second subgradient round (finer) ===
    print(f"\n=== Phase 5: Fine Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(remaining * 0.20, 250.0)
        a5 = subgradient_gd(best_seq, time_budget=alloc, verbose=True, seed=1234,
                            initial_step=0.02, momentum_init=0.7,
                            top_k_init=8, lp_every=40, max_iters=200000)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5 Fine SubGD: obj={best_score:.10f} **")
        else:
            print(f"  Phase5 Fine SubGD: obj={obj5:.10f}")

    # === Phase 6: Multi-n refinement ===
    print(f"\n=== Phase 6: Multi-n Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n_options = sorted(set([
            len(best_seq) - 100, len(best_seq) - 50,
            len(best_seq) + 50, len(best_seq) + 100, len(best_seq) + 200
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2500 and n != len(best_seq)]
        n_options.sort(key=lambda x: abs(x - len(best_seq)))
        n_options = n_options[:4]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            a_try = resample(best_seq, n_try)
            alloc = min(remaining * 0.25, 60.0)
            if alloc > 10:
                a_try = subgradient_gd(a_try, time_budget=alloc, seed=n_try,
                                        initial_step=0.03, lp_every=100, max_iters=50000)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.10f}")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Active set equalize (very fine)
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_f1 = active_set_equalize(best_seq, time_budget=alloc, seed=314)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final ASE: obj={best_score:.10f} **")

    # Ultra-fine subgradient
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_f2 = subgradient_gd(best_seq, time_budget=alloc, seed=271828,
                               initial_step=0.005, momentum_init=0.8,
                               top_k_init=5, lp_every=30, max_iters=50000)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final SubGD: obj={best_score:.10f} **")

    # One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 20 and len(best_seq) <= 2000:
        alloc = min(remaining * 0.5, 60.0)
        a_f3 = subgradient_gd(best_seq, time_budget=alloc, seed=55555,
                               initial_step=0.01, momentum_init=0.6,
                               top_k_init=5, lp_every=10, max_iters=20000)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
