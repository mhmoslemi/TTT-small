def smooth_max_and_grad(a, tau):
    """Compute smooth max of conv(a,a) and gradient. Returns (value, grad)."""
    n = len(a)
    S = 1
    while S < 3 * n - 2:
        S <<= 1
    
    conv = compute_conv(a)  # length 2n-1
    c_shifted = conv / tau
    c_shifted -= np.max(c_shifted)  # for numerical stability
    p = np.exp(c_shifted)
    p /= p.sum()
    
    value = tau * (np.log(p.sum()) + np.max(conv / tau))  # = tau * logsumexp(conv/tau)
    # Actually: logsumexp(conv/tau) = max(conv/tau) + log(sum(exp(conv/tau - max(conv/tau))))
    # = max(conv)/tau + log(sum(exp(conv/tau - max(conv)/tau)))
    # So: value = tau * (max(conv)/tau + log(sum(...))) = max(conv) + tau * log(sum(...))
    # But that's not right. Let me just compute it directly.
    
    # value = tau * log(sum(exp(conv/tau)))
    # For numerical stability: = tau * (max(conv)/tau + log(sum(exp((conv - max(conv))/tau))))
    # = max(conv) + tau * log(sum(exp((conv - max(conv))/tau)))
    # = max(conv) + tau * log(1/n_effective)  approximately
    
    # Actually, we already computed p = exp(conv/tau - max(conv/tau)) / sum(...)
    # So sum(exp(conv/tau)) = exp(max(conv)/tau) * sum(p * sum(exp(...))) ... 
    # Let me just compute: value = tau * (np.max(conv)/tau + np.log(np.sum(np.exp(conv/tau - np.max(conv)/tau))))
    
    max_c = np.max(conv)
    value = max_c + tau * np.log(np.sum(np.exp((conv - max_c) / tau)))
    
    # Gradient
    fa = rfft(a, S)
    fp = rfft(p, S)
    grad = 2.0 * irfft(np.conj(fa) * fp, S)[:n]

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FFT-based exponentiated gradient with annealed smooth-max for fast polishing,
(2) LP alternating with adaptive constraint management and geometric updates,
(3) vectorized coordinate descent, (4) multi-restart with diverse perturbations.
Key innovation: O(n log n) gradient computation via FFT cross-correlation enables
many more optimization steps than the LP-only approach of the parent.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def smooth_max_value_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) using log-sum-exp, and its gradient.
    Gradient computed via FFT cross-correlation: grad = 2 * IFFT(conj(FFT(a)) * FFT(p)).
    Returns (value, gradient_array).
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / tau
    exp_shifted = np.exp(shifted)
    p = exp_shifted / float(np.sum(exp_shifted))
    value = max_c + tau * float(np.log(np.sum(exp_shifted)))

    # Gradient: 2 * (p cross-correlate a)
    # (p ⊛ a)[m] = sum_j p[j+m] * a[j]
    # = IFFT(conj(FFT(a, S)) * FFT(p, S))[:n] with S >= 3n-2
    S = 1
    while S < 3 * n - 2:
        S <<= 1
    fa = rfft(a, S)
    fp = rfft(p, S)
    grad = 2.0 * irfft(np.conj(fa) * fp, S)[:n]
    return value, grad


def exponentiated_gradient(a, time_budget, seed=42, verbose=True):
    """
    Exponentiated gradient (mirror descent on simplex) with annealed temperature.
    Work on normalized sequence (sum=1). Uses multiplicative updates:
    a[j] *= exp(-eta * grad[j]), then normalize.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Temperature schedule: geometric decrease
    initial_max_conv = float(np.max(compute_conv(a)))
    n_tau_levels = 12
    tau_ratio = 0.65
    tau0 = initial_max_conv / 3.0
    steps_per_tau = 80

    for tau_idx in range(n_tau_levels):
        if time.monotonic() - start > time_budget - 2:
            break
        tau = tau0 * (tau_ratio ** tau_idx)
        if tau < initial_max_conv * 0.005:
            tau = initial_max_conv * 0.005

        for step in range(steps_per_tau):
            if time.monotonic() - start > time_budget - 1:
                break

            val, grad = smooth_max_value_and_grad(a, tau)

            # Adaptive step size
            grad_max = float(np.max(np.abs(grad)))
            if grad_max < 1e-15:
                break
            # Step size: ensure exp(-eta*grad) doesn't underflow/overflow
            eta = min(1.0 / max(grad_max, 1e-10), 5.0)
            # Scale down for large tau (smoother => smaller effective gradient)
            eta *= min(1.0, tau / (initial_max_conv * 0.1 + 1e-10))

            # Multiplicative update
            log_update = -eta * grad
            # Clip to avoid numerical issues
            log_update = np.clip(log_update, -50.0, 50.0)
            a = a * np.exp(log_update)
            a = np.maximum(a, 1e-15)
            s = float(np.sum(a))
            if s < 1e-10:
                a = best_a.copy()
                break
            a = a / s

            # Check improvement
            obj = compute_obj_norm(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()

    if verbose:
        elapsed = time.monotonic() - start
        print(f"    EG done: obj={best_obj:.10f}, time={elapsed:.1f}s")
    return best_a


def build_conv_matrix_subset(a_norm, active_indices):
    """Build matrix A (len(active_indices) x n) for subset of constraints."""
    n = len(a_norm)
    k = len(active_indices)
    A = np.zeros((k, n))
    for row, ki in enumerate(active_indices):
        lo = max(0, ki - n + 1)
        hi = min(n, ki + 1)
        for j in range(lo, hi):
            A[row, j] = a_norm[ki - j]
    return A


def solve_lp(a_norm, warm_start=None, active_indices=None):
    """
    Solve LP: min M s.t. (a_norm*g)[k] <= M for k in active_indices, sum(g)=1, g>=0.
    If active_indices is None, use all constraints.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a_norm)
    if active_indices is None:
        active_indices = np.arange(2 * n - 1)

    k_constraints = len(active_indices)
    if k_constraints == 0:
        return None, None

    A = np.zeros((k_constraints, n))
    a_n = a_norm
    for row, ki in enumerate(active_indices):
        lo = max(0, ki - n + 1)
        hi = min(n, ki + 1)
        # A[row, j] = a_norm[ki - j] for j in [lo, hi)
        for j in range(lo, hi):
            A[row, j] = a_n[ki - j]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'ECOS', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """
    Vectorized coordinate descent on normalized sequence (sum=1).
    Exploits O(n) convolution update per coordinate change.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 4
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            if a_j < 1e-12:
                candidates = np.array([0.0, mean_val * 0.1, mean_val * 0.5, mean_val, mean_val * 2, mean_val * 5])
            else:
                candidates = np.array([
                    a_j * 0.3, a_j * 0.6, a_j * 0.85,
                    a_j * 1.15, a_j * 1.4, a_j * 2.0,
                    mean_val * 0.4, mean_val * 0.8, mean_val * 1.2, mean_val * 2, mean_val * 5
                ])
                candidates = candidates[(candidates >= 0) & (candidates <= 0.5)]
                candidates = np.unique(candidates)

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_unscaled = max(max_unaffected, float(np.max(affected))) if len(affected) > 0 else max_unaffected
                obj_trial = 2.0 * n * max_conv_unscaled / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta
                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 9):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def noise_perturb(a, rng, strength=0.01):
    """Add relative noise perturbation."""
    n = len(a)
    pert = rng.standard_normal(n) * strength
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, n_new, n_old):
    """Resample a sequence from length n_old to n_new using linear interpolation."""
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines exponentiated gradient, LP alternating,
    coordinate descent, and multi-restart exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    # Initialize from parent or default
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        print(f"Parent: n={a.size}, obj={compute_obj_norm(a):.10f}")
    except Exception:
        a = np.ones(800) / 800.0
        print(f"Default: n=800, obj={compute_obj_norm(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    rng = np.random.default_rng(seed)
    n_best = len(best_a)

    # === Phase 1: Quick EG warm-up (30s) ===
    print("\n--- Phase 1: EG Warm-up ---")
    p1_budget = min(30.0, (deadline - time.monotonic()) * 0.1)
    if p1_budget > 5:
        a1 = exponentiated_gradient(best_a, time_budget=p1_budget, seed=seed, verbose=True)
        obj1 = compute_obj_norm(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            print(f"  P1: {best_obj:.10f} **")
        else:
            print(f"  P1: {obj1:.10f}")

    # === Phase 2: LP Alternating v2 (main workhorse, ~45% time) ===
    print("\n--- Phase 2: LP Alternating v2 ---")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.48, 600.0)
    a2 = lp_alternate_v2(best_a, time_budget=p2_budget, seed=seed, verbose=True)
    obj2 = compute_obj_norm(a2)
    if obj2 < best_obj:
        best_obj = obj2
        best_a = a2.copy()
        print(f"  P2: {best_obj:.10f} **")
    else:
        print(f"  P2: {obj2:.10f}")

    # === Phase 3: EG polish after LP (20s) ===
    print("\n--- Phase 3: EG Polish ---")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        p3_budget = min(20.0, remaining * 0.1)
        a3 = exponentiated_gradient(best_a, time_budget=p3_budget, seed=seed + 1000, verbose=False)
        obj3 = compute_obj_norm(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  P3: {best_obj:.10f} **")
        else:
            print(f"  P3: {obj3:.10f}")

    # === Phase 4: CD Polish (15% time) ===
    print("\n--- Phase 4: CD Polish ---")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_budget = min(remaining * 0.2, 200.0)
        a4 = coordinate_descent_polish(best_a, time_budget=p4_budget, seed=seed + 2000, verbose=True)
        obj4 = compute_obj_norm(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            print(f"  P4: {best_obj:.10f} **")
        else:
            print(f"  P4: {obj4:.10f}")

    # === Phase 5: Multi-restart exploration ===
    print("\n--- Phase 5: Multi-Restart ---")
    remaining = deadline - time.monotonic()
    n_restarts = 5
    if remaining > n_restarts * 40:
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            # Different perturbation strategies
            if r % 3 == 0:
                a_try = fourier_perturb(best_a, rng, strength=0.008 + 0.004 * r)
            elif r % 3 == 1:
                a_try = noise_perturb(best_a, rng, strength=0.005 + 0.003 * r)
            else:
                # Perturb in a structured way: amplify certain frequencies
                n = len(best_a)
                freq_pert = np.zeros(n)
                target_freq = rng.choice([2, 3, 5, 7, 11])
                amp = 0.01 + 0.005 * r
                phase = rng.uniform(0, 2 * np.pi)
                freq_pert = amp * np.cos(target_freq * 2.0 * np.pi * np.arange(n) / n + phase)
                a_try = best_a * np.exp(freq_pert)
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                else:
                    a_try = best_a.copy()

            # Short optimization: EG + LP + CD
            alloc = min(remaining * 0.18, 120.0)
            if alloc > 30:
                a_try = exponentiated_gradient(a_try, time_budget=alloc * 0.25, seed=seed + 5000 + r * 100, verbose=False)
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    a_try = lp_alternate_v2(a_try, time_budget=alloc * 0.45, seed=seed + 6000 + r * 100, verbose=False)
                remaining = deadline - time.monotonic()
                if remaining > 15:
                    a_try = coordinate_descent_polish(a_try, time_budget=min(alloc * 0.2, 30.0),
                                                     seed=seed + 7000 + r * 100, verbose=False)

            obj_try = compute_obj_norm(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  Restart {r}: {best_obj:.10f} **")
            else:
                print(f"  Restart {r}: {obj_try:.10f}")

    # === Phase 6: Length exploration ===
    print("\n--- Phase 6: Length Exploration ---")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_options = sorted(set([
            n_best - 100, n_best - 50, n_best + 50, n_best + 100,
            n_best - 200, n_best + 200,
            int(n_best * 0.8), int(n_best * 1.2)
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_best]

        for n_try in n_options[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break

            a_try = resample_sequence(best_a, n_try, n_best)
            # Quick optimization
            alloc = min(remaining * 0.2, 100.0)
            if alloc > 30:
                a_try = exponentiated_gradient(a_try, time_budget=alloc * 0.3, seed=n_try, verbose=False)
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    a_try = lp_alternate_v2(a_try, time_budget=alloc * 0.5, seed=n_try + 1000, verbose=False)
                remaining = deadline - time.monotonic()
                if remaining > 10:
                    a_try = coordinate_descent_polish(a_try, time_budget=min(alloc * 0.15, 20.0),
                                                     seed=n_try + 2000, verbose=False)

            obj_try = compute_obj_norm(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: {best_obj:.10f} **")
            else:
                print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 7: Final polish ===
    print("\n--- Phase 7: Final Polish ---")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_fin = coordinate_descent_polish(best_a, time_budget=min(remaining * 0.6, 60.0),
                                          seed=seed + 42424, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final CD: {best_obj:.10f} **")
        else:
            print(f"  Final CD: {obj_fin:.10f}")

        # One more EG at the end with very low temperature
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_fin2 = exponentiated_gradient(best_a, time_budget=min(remaining * 0.5, 15.0),
                                            seed=seed + 99999, verbose=False)
            obj_fin2 = compute_obj_norm(a_fin2)
            if obj_fin2 < best_obj:
                best_obj = obj_fin2
                best_a = a_fin2.copy()
                print(f"  Final EG: {best_obj:.10f} **")

    # Final validation and return
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
