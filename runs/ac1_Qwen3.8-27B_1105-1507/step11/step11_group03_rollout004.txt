    """
    Approximate the inner LP: min_g max_k (a*g)(k) s.t. g>=0, sum g=1.
    Uses projected subgradient with adaptive step size.
    """
    n = len(a)
    # Initialize g as uniform
    g = np.ones(n) / n
    
    # Precompute: for each lag k, the subgradient is a subshift of a
    # sg_k[j] = a[k-j] if 0 <= k-j < n, else 0
    
    best_g = g.copy()
    best_val = np.inf
    eta = eta0
    
    for it in range(n_iters):
        # Compute a*g
        ag = np.convolve(a, g)  # length 2n-1
        max_val = float(np.max(ag))

3. Find the max of `b`, say at lag k*.
4. Reduce `a` at positions that contribute most to `b[k*]` (i.e., positions i where a[i]*a[i+k*] is large).
5. Renormalize.
6. Repeat.

- At each step, identify the peak lag k*.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses hybrid subgradient + LP fixed-point iteration with adaptive active sets,
multi-scale initialization, and spectral post-processing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Efficient projection onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def convolve_with_shifts(a, g, lags):
    """
    Compute (a*g)(k) for specific lags k in `lags`.
    Returns array of values at those lags.
    """
    n = len(a)
    result = np.zeros(len(lags))
    for i, k in enumerate(lags):
        # (a*g)(k) = sum_j a[k-j]*g[j] for valid j
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_max >= j_min:
            result[i] = np.dot(a[k - np.arange(j_min, j_max + 1)], g[j_min:j_max+1])
    return result


def subgradient_inner(a, g_init=None, n_iters=500, eta0=None, verbose=False):
    """
    Approximate solution of min_g max_k (a*g)(k) s.t. g>=0, sum g=1.
    Uses projected subgradient with adaptive step size.
    Returns (g, max_val).
    """
    n = len(a)
    if g_init is None:
        g = np.ones(n) / n
    else:
        g = g_init.copy()
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 0:
            g = g / s
        else:
            g = np.ones(n) / n
    
    if eta0 is None:
        eta = 1.0 / np.sqrt(n)
    
    best_g = g.copy()
    best_val = np.inf
    
    for it in range(n_iters):
        # Compute a*g (full convolution)
        ag = np.convolve(a, g)
        max_val = float(np.max(ag))
        max_k = int(np.argmax(ag))
        
        if max_val < best_val:
            best_val = max_val
            best_g = g.copy()
        
        # Find active lags
        threshold = max_val * 0.995
        active = np.where(ag >= threshold)[0]
        
        # Compute subgradient (average of active rows)
        sg = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_max >= j_min:
                sg[j_min:j_max+1] += a[k - np.arange(j_min, j_max + 1)]
        sg /= len(active)
        
        # Step and project
        g = g - eta * sg
        g = simplex_project(g)
        
        # Decay step size
        eta *= 0.9995
    
    return best_g, best_val


def lp_inner_active(a, n_active_max=300, threshold_frac=0.8):
    """
    Solve the inner LP using only active constraints.
    min M s.t. (a*g)(k) <= M for k in active_set, g>=0, sum g=1.
    """
    n = len(a)
    # Compute a*a to find active lags
    aa = compute_conv(a)
    max_val = float(np.max(aa))
    threshold = max_val * threshold_frac
    active_lags = np.where(aa >= threshold)[0]
    
    if len(active_lags) > n_active_max:
        # Keep only the top ones
        vals = aa[active_lags]
        top_idx = np.argsort(vals)[::-1][:n_active_max]
        active_lags = active_lags[top_idx]
    
    m = len(active_lags)
    
    # Build constraint matrix
    A = np.zeros((m, n))
    for idx, k in enumerate(active_lags):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_max >= j_min:
            A[idx, j_min:j_max+1] = a[k - np.arange(j_min, j_max + 1)]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def hybrid_inner(a, g_init=None, n_subgrad=200, use_lp=True, lp_every=50):
    """
    Hybrid inner solver: subgradient for speed, LP for accuracy.
    """
    n = len(a)
    g, val = subgradient_inner(a, g_init, n_iters=n_subgrad)
    
    if use_lp:
        g_lp, M_lp = lp_inner_active(a)
        if g_lp is not None:
            # Blend: use LP solution if it's better
            ag_lp = np.convolve(a, g_lp)
            if float(np.max(ag_lp)) < val:
                g = g_lp
                val = float(np.max(ag_lp))
    
    return g, val


def water_filling_step(a, eta=0.01, n_peaks=3):
    """
    Water-filling: reduce values at positions contributing most to the peak.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Find top peaks
    top_peaks = np.argsort(conv)[::-1][:n_peaks]
    
    a_new = a.copy()
    for k in top_peaks:
        # Positions contributing to peak k: i where a[i]*a[i+k] is large
        j_min = max(0, k - n + 1) if k > n - 1 else 0
        j_max = min(k, n - 1)
        if j_max >= j_min:
            contrib = a[j_min:j_max+1] * a[k - np.arange(j_min, j_max + 1)]
            # Reduce a at positions with high contribution
            a_new[j_min:j_max+1] -= eta * contrib
    
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s


def smoothed_peak_grad(a, tau=10.0):
    """
    Compute gradient of smoothed max of conv(a,a).
    Uses log-sum-exp smoothing.
    Returns (gradient, exact_max).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    
    # Softmax weights
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    
    # Gradient: d/d a_i [sum_k p_k * (a*a)(k)]
    # = 2 * sum_k p_k * (a shifted by k)[i]
    # = 2 * (p_reversed * a) at appropriate lags
    # Using FFT:
    N = 1
    while N < 4 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # The cross-correlation
    cross = ifft(P * np.conj(A)).real
    grad = 2.0 * cross[:n]
    
    return grad, exact_max


def fixpoint_iteration(a, time_budget, verbose=False, seed=42, use_lp=True,
                       subgrad_iters=200, lp_every=20, max_outer=500):
    """
    Main fixed-point iteration with hybrid inner solver.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    t = 0.3  # blending parameter
    stall_count = 0
    g_warm = None  # warm start for inner solver
    
    for outer_it in range(max_outer):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        # Solve inner problem
        if use_lp and outer_it % lp_every == 0:
            g, M = hybrid_inner(a, g_warm, n_subgrad=subgrad_iters, use_lp=True)
        else:
            g, M = subgradient_inner(a, g_warm, n_iters=subgrad_iters)
        
        if g is None:
            continue
        
        g_warm = g  # warm start for next iteration
        
        # Try different blending parameters
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t
        
        for tc in t_candidates:
            a_try = (1.0 - tc) * a + tc * g
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-12:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = tc
        
        if best_local_obj < best_obj - 1e-12:
            improvement = best_obj - best_local_obj
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            t = best_local_t
            stall_count = 0
            if verbose and outer_it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"  FP it{outer_it:3d}: obj={best_obj:.10f}, t={t:.3f}, M={M:.6f}, elapsed={elapsed:.1f}s")
        else:
            stall_count += 1
            # Perturbation strategies
            if stall_count <= 5:
                # Small random perturbation
                a = a + 0.005 * rng.standard_normal(n)
            elif stall_count <= 15:
                # Gradient-based perturbation (toward reducing peak)
                grad, _ = smoothed_peak_grad(a, tau=50)
                a = a - 0.002 * grad
            elif stall_count <= 25:
                # Water-filling step
                a = water_filling_step(a, eta=0.005)
            elif stall_count <= 35:
                # Spectral perturbation
                A_freq = fft(a, 2*n)
                # Reduce high-frequency content
                freqs = np.fft.fftfreq(2*n)
                mask = np.abs(freqs) > n / (2*n)
                A_freq[mask] *= 0.9
                a = ifft(A_freq).real[:n]
            else:
                # Major restart
                a = best_a + 0.05 * rng.standard_normal(n)
                stall_count = 0
                t = 0.2
            
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()
                g_warm = None
        
        if stall_count > 60:
            # Full restart from best
            a = best_a.copy()
            g_warm = None
            stall_count = 0
            t = 0.15

def spectral_refine(a, time_budget, verbose=False, seed=42):
    """
    Spectral refinement: work in frequency domain to reduce convolution peak.
    The key idea: conv(a,a) in time domain = |FFT(a)|^2 in frequency domain.
    So the peak of conv is the inverse FFT of |FFT(a)|^2.
    To reduce the peak, we want |FFT(a)|^2 to be as flat as possible.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    N = 1
    while N < 4 * n:
        N <<= 1
    
    # Target: flat |A(f)|^2 spectrum
    # This means |A(f)| should be constant (like a chirp)
    # But we need a >= 0, so we can't have a perfectly flat spectrum.
    # Strategy: iteratively modify the spectrum to be flatter, then project back to a >= 0.
    
    for it in range(200):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target magnitude: geometric mean of current and uniform
        target_mag = np.sqrt(mag) * (float(np.mean(mag)) ** 0.5)
        # Actually, let's use a blend of current and flat
        alpha = 0.1 + 0.2 * min(1.0, it / 50.0)
        flat_mag = np.full(N, float(np.mean(mag)))
        new_mag = (1 - alpha) * mag + alpha * flat_mag
        
        # Reconstruct with new magnitude, same phase
        A_new = new_mag * np.exp(1j * phase)
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            continue
        a_new = a_new / sn
        
        # Blend for stability
        a_try = 0.8 * a + 0.2 * a_new
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            continue
        a_try = a_try / st
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            if verbose and it % 20 == 0:
                print(f"  Spectral it{it}: obj={best_obj:.10f}")
        else:
            # Try smaller step
            a_try2 = 0.95 * a + 0.05 * a_new
            a_try2 = np.maximum(a_try2, 0)
            a_try2 = a_try2 / a_try2.sum()
            obj_try2 = compute_obj(a_try2)
            if obj_try2 < best_obj - 1e-12:
                best_obj = obj_try2
                best_a = a_try2.copy()
                a = a_try2
            else:
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a


def gd_polish(a, time_budget, verbose=False, seed=42):
    """
    Gradient descent polish using log-sum-exp smoothed max.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    tau_levels = [5, 20, 80, 300, 1000, 3000]
    
    for tau in tau_levels:
        lr = 1.0 / np.sqrt(n)
        v = np.zeros(n)  # momentum
        beta = 0.9
        
        max_iters = 3000
        for it in range(max_iters):
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 1:
                return best_a
            
            # Nesterov step
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
            else:
                y = y / sy
            
            grad, _ = smoothed_peak_grad(y, tau)
            
            # Gradient of the full objective (with constraint sum=1 handled by projection)
            # Actually, we're optimizing over the simplex, so we project after each step
            v = beta * v - lr * grad
            a_new = y + v
            a_new
