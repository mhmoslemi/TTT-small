import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def active_set_gradient(a, top_k=20, beta_w=5.0):
    """
    Compute weighted gradient of max convolution using top-k active lags.
    grad_i = sum_m w_m * 2 * a[k_m - i] for valid k_m - i.
    """
    n = len(a)
    conv = compute_conv(a)
    # Get top-k lags
    k = min(top_k, 2 * n - 1)
    idx = np.argsort(conv)[::-1][:k]
    vals = conv[idx]
    # Softmax weights over top-k
    w = np.exp(beta_w * (vals - vals[0]))
    w = w / w.sum()
    # Compute gradient via shifts
    grad = np.zeros(n)
    for m in range(k):
        if w[m] < 1e-12:
            continue
        km = idx[m]
        # grad[i] += w[m] * 2 * a[km - i] for i in [max(0,km-n+1), min(n-1,km)]
        lo = max(0, km - n + 1)
        hi = min(n - 1, km)
        if lo > hi:
            continue
        # a[km - i] for i in [lo, hi] is a[km-hi : km-lo+1] reversed
        segment = a[km - hi : km - lo + 1][::-1]
        grad[lo:hi + 1] += w[m] * 2.0 * segment
    return grad


def backtracking_line_search(a, direction, obj_curr, c1=1e-4, rho=0.5, max_iter=30):
    """Backtracking line search with Armijo condition."""
    n = len(a)
    # Compute gradient of objective (normalized)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    obj_val = 2.0 * n * max_val  # assuming sum(a)=1
    
    # Directional derivative (approximate using active set)
    idx_peak = np.where(conv >= max_val * 0.9999)[0]
    dir_deriv = 0.0
    for km in idx_peak:
        lo = max(0, km - n + 1)
        hi = min(n - 1, km)
        segment = a[km - hi : km - lo + 1][::-1]
        dir_deriv += 2.0 * np.dot(segment, direction[lo:hi + 1])
    
    step = 0.1
    for _ in range(max_iter):
        a_try = project_simplex(a + step * direction)
        obj_try = compute_obj(a_try)
        if obj_try <= obj_val + c1 * step * dir_deriv:
            return step, a_try, obj_try
        step *= rho
    return step, a_try, obj_try


def soft_max_gradient_descent(a, time_budget=200.0, beta_start=5.0, beta_end=100.0,
                              lr_start=0.01, momentum=0.9, verbose=True, seed=42,
                              use_line_search=False):
    """Soft-max gradient descent with momentum on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    lr = lr_start
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.7)
        lr = lr_start * (1.0 - 0.4 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via cross-correlation
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A = rfft(a_rev, size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        # momentum update
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)

            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

                pert = rng.normal(0, 0.003, n)

        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    SMGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
        
        if it - best_it > 5000 and it > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            it = best_it
            lr *= 0.7
    
    return best_a, best_obj


def active_set_subgradient(a, time_budget=60.0, top_k=20, lr_start=0.05,
                           verbose=True, seed=0):
    """
    Subgradient method using top-k active lags with backtracking line search.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)
    lr = lr_start
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        obj_val = 2.0 * n * c_max
        
        # Compute active-set gradient
        grad = active_set_gradient(a, top_k=top_k, beta_w=8.0)
        
        # Ensure gradient points in descent direction
        # Project gradient: remove mean component (tangent to simplex)
        grad_tangent = grad - np.mean(grad)
        
        # Backtracking line search
        step = lr
        found = False
        a_try = a
        obj_try = obj_val
        for _ in range(20):
            a_candidate = project_simplex(a + step * grad_tangent)
            obj_candidate = compute_obj(a_candidate)
            if obj_candidate < obj_val:
                a_try = a_candidate
                obj_try = obj_candidate
                found = True
                break
            step *= 0.5
        
        if found and obj_try < best_obj:
            a = a_try
            best_obj = obj_try
            best_a = a.copy()
            stall = 0
            # Decrease lr slightly
            lr = max(0.001, lr * 0.99)
        else:
            a = a_try if found else a
            stall += 1
            if stall > 50:
                # Reset and perturb
                a = project_simplex(best_a + rng.normal(0, 0.005, n))
                lr = lr_start
                stall = 0
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    ASGD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]
    if solver is not None:
        solvers = [solver] + [s for s in solvers if s != solver]
    
    for solver_try in solvers:
        try:
            prob.solve(solver=solver_try)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    start = time.monotonic()

    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search for optimal blend
        best_t = 0.0
        best_obj_t = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:

                best_t = t
        
        if best_t > 0 and best_obj_t < best_obj - 1e-10:

def spectral_flatten(a, time_budget=30.0, verbose=False):
    """
    Spectral flattening: iterate DFT -> flatten magnitude -> IDFT -> project.
    This pushes the sequence toward having a flat power spectrum.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        A = fft(a)
        # Target: flatten the magnitude
        mag = np.abs(A)
        # Blend toward flat spectrum
        target_mag = np.full(n, np.mean(mag))
        blend = 0.1  # Gentle blending
        new_mag = (1 - blend) * mag + blend * target_mag
        # Keep phase
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        a_new = ifft(new_A).real
        a_new = np.maximum(a_new, 0)
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slowly increase blend
            blend = min(0.3, blend * 1.02)
        else:
            break
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"    Spectral it{it}: obj={best_obj:.8f}")
    
    return best_a, best_obj


def pairwise_local_search(a, time_budget=30.0, verbose=False):

def u_shaped_init(n, sharpness=3.0):

    # === Phase 1: Soft-max GD (smooth, low beta) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: SoftMax GD (beta 3->30), time={remaining:.0f}s ===")
        alloc = min(remaining * 0.18, 200.0)
        a1, obj1 = soft_max_gradient_descent(
            best_seq, time_budget=alloc,
            beta_start=3.0, beta_end=30.0,
            lr_start=0.012, momentum=0.9,
            verbose=True, seed=seed
        )
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj1:.8f}")
    
    # === Phase 2: Active-set subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 2: Active-Set Subgradient, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 150.0)
        a2, obj2 = active_set_subgradient(
            best_seq, time_budget=alloc,
            top_k=30, lr_start=0.03,
            verbose=True, seed=seed + 1
        )
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f}")
    
    # === Phase 3: Soft-max GD (high beta, sharp) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 3: SoftMax GD (beta 30->200), time={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 180.0)
        a3, obj3 = soft_max_gradient_descent(
            best_seq, time_budget=alloc,
            beta_start=30.0, beta_end=200.0,
            lr_start=0.004, momentum=0.93,
            verbose=True, seed=seed + 2
        )
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f}")
    
    # === Phase 4: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.18, 180.0)
        a4, obj4 = lp_polish(best_seq, time_budget=alloc, max_iters=35, verbose=True)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj4:.8f}")
    
    # === Phase 5: Spectral flattening + local search ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Spectral + Local, time={remaining:.0f}s ===")
        alloc_spec = min(remaining * 0.3, 40.0)
        a5, obj5 = spectral_flatten(best_seq, time_budget=alloc_spec, verbose=True)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Spectral: obj={best_score:.8f} **")
        
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc_local = min(remaining * 0.5, 50.0)
            a5b, obj5b = pairwise_local_search(best_seq, time_budget=alloc_local, verbose=False)
            if obj5b < best_score:
                best_score = obj5b
                best_seq = a5b.copy()
                print(f"  Local: obj={best_score:.8f} **")
    
    # === Phase 6: Multi-restart with diverse inits ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Diverse Restarts, time={remaining:.0f}s ===")
        n_cur = len(best_seq)
        
        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            mode = r % 8
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.7
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)
            
            alloc_sgd = min(remaining * 0.3, 60.0)
            if alloc_sgd > 10:
                a_r, obj_r = soft_max_gradient_descent(
                    a_r, time_budget=alloc_sgd,
                    beta_start=10.0, beta_end=80.0,
                    lr_start=0.006, momentum=0.92,
                    verbose=False, seed=seed + 100 + r
                )
            
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.4, 40.0)
                a_r, obj_r = lp_polish(a_r, time_budget=alloc_lp, max_iters=15, verbose=False)
            
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
    
    # === Phase 7: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 7: Length Exploration, time={remaining:.0f}s ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(400, n_best - 100), n_best + 100, n_best + 200,
            600, 700, 900, 1000, 1200
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 50]
        n_options = n_options[:3]
        print(f"  Trying n values: {n_options}")
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))
            
            alloc_sgd = min(remaining * 0.4, 50.0)
            if alloc_sgd > 10:
                a_try, obj_try = soft_max_gradient_descent(
                    a_try, time_budget=alloc_sgd,
                    beta_start=15.0, beta_end=60.0,
                    lr_start=0.007, momentum=0.91,
                    verbose=False, seed=seed + 200
                )
            
            remaining = deadline - time.monotonic()
            if remaining > 25:
                alloc_lp = min(remaining * 0.3, 35.0)
                a_try, obj_try = lp_polish(a_try, time_budget=alloc_lp, max_iters=15, verbose=False)
            
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 8: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Final Polish, time={remaining:.0f}s ===")
        
        # High-beta soft-max GD
        alloc1 = min(remaining * 0.35, 80.0)
        if alloc1 > 15:
            a_f, obj_f = soft_max_gradient_descent(
                best_seq, time_budget=alloc1,
                beta_start=50.0, beta_end=300.0,
                lr_start=0.002, momentum=0.94,
                verbose=True, seed=seed + 300
            )
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final SMGD: obj={best_score:.8f} **")
        
        # Active-set subgradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.3, 40.0)
            a_f2, obj_f2 = active_set_subgradient(
                best_seq, time_budget=alloc2,
                top_k=50, lr_start=0.01,
                verbose=False, seed=seed + 400
            )
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final ASGD: obj={best_score:.8f} **")
        
        # LP
        remaining = deadline - time.monotonic()
        if remaining > 25:
            alloc3 = min(remaining * 0.5, 60.0)
            a_f3, obj_f3 = lp_polish(best_seq, time_budget=alloc3, max_iters=25, verbose=True)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
    
    # === Phase 9: Last-ditch ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for attempt in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            # Structured perturbation
            n = len(best_seq)
            x = np.linspace(0, 1, n)
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.01 * np.sin(2 * np.pi * freq * x + phase)
            a_p = project_simplex(best_seq + pert)
            
            alloc = min(remaining * 0.4, 20.0)
            if alloc > 5:
                a_p, obj_p = lp_polish(a_p, time_budget=alloc, max_iters=8, verbose=False)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Last-ditch {attempt}: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
