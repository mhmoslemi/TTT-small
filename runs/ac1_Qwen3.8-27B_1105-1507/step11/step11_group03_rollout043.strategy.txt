# --- Supportfunctionforconvolutioncompute ---
def compute_conv(a, mode='full'):
    n = len(a)
    if n == 0:
        return np.array([])
    size_exp = 1
    while size_exp < 2*n-1:
        size_exp *= 2
    fa = rfft(a, size=size_exp)
    conv_result = irfft(fa * fa, size=size_exp)
    if mode=='full':
        return conv_result[:2*n-1]
    else:  # mode should be for full or valid likely
        return conv_result[:2*n-1]

# --- Evaluationhelperforconsistencyusingfullconvolution ---
def calculate_objective_for_sequence(seq):
    n = len(seq)
    if n == 0:
        return np.inf
    conv = compute_conv(seq)
    max_val = np.max(conv)
    total_sum = np.sum(seq)
    if total_sum < 0.01:
        total_sum = 0.01
    numerator = 2 * n * max_val
    denominator = total_sum**2
    return numerator / denominator
