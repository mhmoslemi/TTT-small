"""
Optimizes step function coefficients using a hybrid strategy:
1. Fast coordinate descent with O(n) convolution updates
2. LP-based direction finding with multiple regularization levels
3. Peak shaving: targeted modifications at peak lags
4. Subgradient descent with Polyak step size
5. Multi-length exploration

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_budget=20.0, lambda_reg=0.0):
    """
    Solve LP: min M + lambda*||g - a_norm||_1
    s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if lambda_reg > 0:
        objective = cp.Minimize(M + lambda_reg * cp.norm1(g - a_norm))
    else:
        objective = cp.Minimize(M)
    
    prob = cp.Problem(objective, constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    """
    Fast coordinate descent with O(n) convolution updates.
    Each coordinate update only requires updating O(n) convolution entries.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            # The convolution entries that depend on a[i]:
            # (a*a)[k] involves a[i] when k-i is in [0, n-1]
            # i.e., k in [i, i+n-1]
            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            # a[k-i] for these k values
            ai_contrib = a[ks - i]  # a[0], a[1], ..., a[n-1] (truncated)

            # Separate the diagonal term (k = 2i)
            k_diag = 2 * i
            if k_diag >= k_min and k_diag <= k_max:
                idx_diag = k_diag - k_min
                # At the diagonal, the contribution is a[i] itself (squared)
                # Linear part (excluding diagonal):
                mask = np.ones(len(ks), dtype=bool)
                mask[idx_diag] = False
                ks_lin = ks[mask]
                ai_lin = ai_contrib[mask]
                c_lin = conv[ks_lin].copy()
                c_diag = float(conv[k_diag])
            else:
                ks_lin = ks
                ai_lin = ai_contrib
                c_lin = conv[ks_lin].copy()
                c_diag = None

            def f_1d(x):
                if S_other + x < 1e-15:
                    return 1e30
                if len(ks_lin) > 0:
                    lin_vals = c_lin + 2.0 * (x - v) * ai_lin
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                if c_diag is not None:
                    quad_val = c_diag + x * x - v * v
                else:
                    quad_val = 0.0
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) ** 2
                return 2.0 * n * max_val / denom

            # Try some candidate values
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v, v * 0.5, v * 0.7, v * 1.3, v * 1.5,
                          1.0 / n, 2.0 / n, 3.0 / n]
            candidates = sorted(set(max(0.0, min(10.0 / n, float(c))) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section around best_x
            x_lo = max(0.0, best_x - scale * 1.5)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-10

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(gs_iters):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            # Update convolution
            delta = x_opt - v
            if c_diag is not None:
                conv[k_diag] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * ai_lin
            a[i] = x_opt
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

    return best_a


def peak_shave(a, n_iter=50, seed=42):
    """
    Targeted peak shaving: identify peak lags and reduce contributing coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    budget = 60.0

    it = 0
    while time.monotonic() - start < budget and it < n_iter * 10:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find peak lags (within 5% of max)
        threshold = max_val * 0.95
        peak_ks = np.where(conv >= threshold)[0]

        # For each peak lag k, the gradient w.r.t. a[i] is 2*a[k-i] (if 0<=k-i<n)
        # We want to reduce a[i] where it contributes most to peaks
        grad = np.zeros(n)
        for k in peak_ks:
            # a[k-i] is valid when 0 <= k-i < n, i.e., max(0,k-n+1) <= i <= min(n-1, k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                indices = np.arange(i_lo, i_hi + 1)
                grad[indices] += 2.0 * a[k - indices]

        # Normalize gradient
        grad_sum = float(np.sum(grad))
        if grad_sum < 1e-15:
            break

        # Reduce coordinates with high gradient, increase others
        # Multiplicative update: a[i] *= exp(-lr * grad[i])
        lr = 0.1 / (float(np.max(grad)) + 1e-15)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Adaptive: reduce learning rate
            lr *= 0.5
            a_new = a * np.exp(-lr * grad)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / float(np.sum(a_new))
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

        it += 1

    return best_a


def subgradient_descent(a, time_budget=60.0, tau=100.0, seed=42):
    """
    Subgradient descent with Polyak-type step size.
    Uses the lower bound 1/(2n-1) for the normalized objective / (2n).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Lower bound on max(conv) for normalized a: 1/(2n-1)
    # So lower bound on objective: 2n/(2n-1)
    lb = 2.0 * n / (2 * n - 1) + 0.01  # add margin

    # Use smoothed gradient
    it = 0
    lr = 1.0
    # Running average for Polyak
    a_avg = a.copy()
    count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj = 2.0 * n * max_conv

        # Compute smoothed gradient of max(conv(a,a))
        # Using softmax weighting
        shift = conv - max_conv
        exp_vals = np.exp(tau * shift)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient of (a*a)[k] w.r.t. a[i] is 2*a[k-i] if 0<=k-i<n
        # Smoothed gradient: sum_k p[k] * 2*a[k-i]
        # This is 2 * (p convolve a) evaluated appropriately
        # (p * a)[i] = sum_k p[k] * a[k-i] (with bounds)
        # Use FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        grad_full = ifft(P * np.conj(A)).real  # This gives (p * a) in circular sense
        # Actually we need sum_k p[k] * a[k-i] for i=0..n-1
        # This is the cross-correlation of p and a
        # Let me compute it directly
        grad = np.zeros(n)
        for k in range(len(p)):
            if p[k] < 1e-10:
                continue
            i_lo = max(0, k - n + 1)
            i_hi = n - 1  # k-i <= k, and we need k-i >= 0, so i <= k
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                indices = np.arange(i_lo, i_hi + 1)
                grad[indices] += 2.0 * p[k] * a[k - indices]

        # Polyak step: lr * (obj - lb) / ||grad||^2
        g_norm_sq = float(np.dot(grad, grad))
        if g_norm_sq > 1e-20:
            polyak_lr = (obj - lb) / g_norm_sq
            lr_eff = min(lr, polyak_lr)
        else:
            lr_eff = lr

        a_new = a - lr_eff * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Update running average
        count += 1
        a_avg = (count - 1) / count * a_avg + 1 / count * a_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        obj_avg = 2.0 * n * float(np.max(compute_conv(a_avg)))
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()

        a = a_new
        lr *= 0.999  # slow decay
        it += 1

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two gaussian bumps
        x = np.arange(n) / n
        a = np.exp(-20 * (x - 0.25)**2) + np.exp(-20 * (x - 0.75)**2)
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2 + 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.arange(n) / n
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 2
        a = a / a.sum()
    elif mode == 6:
        # sinc-like
        x = np.arange(n) - n / 2
        a = np.sinc(x / (n * 0.3)) ** 2
        a = np.maximum(a, 0)
        if a.sum() > 0:
            a = a / a.sum()
        else:
            a = np.ones(n) / n
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines coordinate descent, LP, peak shaving,
    and subgradient methods with multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Initial CD ===
    print(f"\n=== Phase 1: Initial Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        t0 = time.monotonic()
        a_cd = coordinate_descent_fast(best_seq, time_budget=p1_budget, seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Peak Shaving ===
    print(f"\n=== Phase 2: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t0 = time.monotonic()
        a_ps = peak_shave(best_seq, n_iter=100, seed=seed+1)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PeakShave: obj={best_score:.8f} **")
        else:
            print(f"  PeakShave: obj={obj_ps:.8f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP Polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.08, 40.0)
            if alloc < 8:
                break

            # Try different regularization
            lam = [0.0, 0.05, 0.1, 0.2][rnd % 4]
            t0 = time.monotonic()
            g_lp, M_lp = solve_lp_minpeak(best_seq, time_budget=alloc, lambda_reg=lam)
            if g_lp is not None:
                best_t_obj = best_score
                best_a_local = best_seq.copy()
                for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1.0 - t) * best_seq + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_a_local = a_try.copy()

                if best_t_obj < best_score:
                    best_score = best_t_obj
                    best_seq = best_a_local.copy()
                    print(f"  LP round {rnd} (lam={lam}): obj={best_score:.8f} **")
                else:
                    print(f"  LP round {rnd} (lam={lam}): obj={best_t_obj:.8f}")
            else:
                print(f"  LP round {rnd}: solver failed")
            print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 4: CD after LP ===
    print(f"\n=== Phase 4: Post-LP Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 120.0)
    if p4_budget > 20:
        t0 = time.monotonic()
        a_cd2 = coordinate_descent_fast(best_seq, time_budget=p4_budget, seed=seed+100)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Subgradient Descent ===
    print(f"\n=== Phase 5: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        t0 = time.monotonic()
        a_sg = subgradient_descent(best_seq, time_budget=alloc, tau=50.0, seed=seed+200)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  SubGrad: obj={obj_sg:.8f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 6: More CD + Peak Shaving ===
    print(f"\n=== Phase 6: Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t0 = time.monotonic()
        a_cd3 = coordinate_descent_fast(best_seq, time_budget=min(remaining*0.3, 80.0), seed=seed+300)
        obj_cd3 = compute_obj(a_cd3)
        if obj_cd3 < best_score:
            best_score = obj_cd3
            best_seq = a_cd3.copy()
        a_ps2 = peak_shave(best_seq, n_iter=50, seed=seed+400)
        obj_ps2 = compute_obj(a_ps2)
        if obj_ps2 < best_score:
            best_score = obj_ps2
            best_seq = a_ps2.copy()
        print(f"  Refinement: obj={best_score:.8f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try nearby lengths
    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) >= 30]
    n_options = n_options[:5]
    print(f"  Checking lengths: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick CD
        alloc_cd = min(remaining * 0.15, 40.0)
        if alloc_cd > 10:
            a_try = coordinate_descent_fast(a_try, time_budget=alloc_cd, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Diverse restarts ===
    print(f"\n=== Phase 8: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.2, 50.0)
        if alloc > 15:
            a_r = coordinate_descent_fast(a_r, time_budget=alloc, seed=5000 + r * 100)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 100.0)
        a_final = coordinate_descent_fast(best_seq, time_budget=alloc, seed=seed+9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
        print(f"  Final CD: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # One more LP
        g_lp, _ = solve_lp_minpeak(best_seq, time_budget=min(remaining*0.4, 30.0), lambda_reg=0.0)
        if g_lp is not None:
            for t in [0.1, 0.2, 0.3, 0.5, 1.0]:
                a_try = (1-t) * best_seq + t * g_lp
                a_try = np.maximum(a_try, 0) / float(np.sum(a_try))
                obj_try = compute_obj(a_try)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    break
            # Final CD
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_final2 = coordinate_descent_fast(best_seq, time_budget=min(remaining*0.5, 30.0), seed=seed+77777)
                obj_final2 = compute_obj(a_final2)
                if obj_final2 < best_score:
                    best_score = obj_final2
                    best_seq = a_final2.copy()
        print(f"  Final: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_budget=20.0, lambda_reg=0.0):
    """
    Solve LP: min M + lambda*||g - a_norm||_1
    s.t. (a_norm * g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    if lambda_reg > 0:
        objective = cp.Minimize(M + lambda_reg * cp.norm1(g - a_norm))
    else:
        objective = cp.Minimize(M)

    prob = cp.Problem(objective, constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def coordinate_descent(a, time_budget=100.0, seed=42, gs_iters=25):
    """
    Fast coordinate descent with O(n) per-coordinate convolution updates.
    Uses golden section search with adaptive early termination.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a).copy()
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve_sweeps = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_obj_before = best_obj
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            ai_contrib = a[ks - i]

            k_diag = 2 * i
            has_diag = (k_diag >= k_min and k_diag <= k_max)
            if has_diag:
                idx_diag = k_diag - k_min
                mask = np.ones(len(ks), dtype=bool)
                mask[idx_diag] = False
                ks_lin = ks[mask]
                ai_lin = ai_contrib[mask]
                c_lin = conv[ks_lin].copy()
                c_diag = float(conv[k_diag])
            else:
                ks_lin = ks
                ai_lin = ai_contrib
                c_lin = conv[ks_lin].copy()
                c_diag = None

            # f_1d: objective as function of x = a[i]
            # max_val = max(max_k≠diag (c_k + 2*(x-v)*a[k-i]), c_diag + x^2 - v^2)
            # obj = 2n * max_val / (S_other + x)^2

            # Quick check: evaluate at current value and a few candidates
            def eval_f(x):
                if S_other + x < 1e-15:
                    return 1e30
                if len(ks_lin) > 0:
                    lin_vals = c_lin + 2.0 * (x - v) * ai_lin
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                if c_diag is not None:
                    quad_val = c_diag + x * x - v * v
                else:
                    quad_val = 0.0
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                return 2.0 * n * max_val / denom

            # Try candidates
            scale = max(v, 1.0 / n, 1e-7)
            candidates = [0.0, v, v * 0.3, v * 0.6, v * 1.4, v * 1.8,
                          1.0 / n, 1.5 / n, 2.5 / n]
            # Clamp
            max_val_allowed = 10.0 / n
            candidates = sorted(set(max(0.0, min(max_val_allowed, float(c))) for c in candidates))

            best_x = v
            best_f = eval_f(v)
            for x in candidates:
                if x == v:
                    continue
                fx = eval_f(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # If no improvement from candidates, skip golden section
            if best_x == v:
                continue

            # Golden section around best_x
            x_lo = max(0.0, best_x - scale * 2.0)
            x_hi = best_x + scale * 3.0
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-10

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_f(c_pt)
            fd = eval_f(d_pt)

            for _ in range(gs_iters):
                if x_hi - x_lo < 1e-12 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_f(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_f(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-13:
                continue

            # Update convolution in O(n)
            delta = x_opt - v
            if has_diag:
                conv[k_diag] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * ai_lin
            a[i] = x_opt
            S += delta

        # End of sweep: check improvement
        if S > 1e-10:
            obj_now = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj_now < best_obj - 1e-12:
                best_obj = obj_now
                best_a = a.copy()
                no_improve_sweeps = 0
            else:
                no_improve_sweeps += 1

        # If stuck, perturb
        if no_improve_sweeps >= 5:
            pert = rng.normal(0, 0.005 * no_improve_sweeps, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / float(a.sum())
            conv = compute_conv(a).copy()
            S = float(a.sum())
            obj_now = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
            no_improve_sweeps = 0

    return best_a


def peak_shave(a, time_budget=40.0, seed=42):
    """
    Targeted peak shaving using FFT-based smoothed gradient.
    Identifies peak lags and applies multiplicative updates to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    N = 1
    while N < 3 * n:
        N <<= 1

    it = 0
    lr = 2.0 / np.sqrt(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj = 2.0 * n * max_conv

        # Softmax weighting on convolution values
        tau = 50.0
        shift = conv - max_conv
        exp_vals = np.exp(tau * shift)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient via FFT cross-correlation: grad[i] = 2 * sum_k p[k] * a[k-i]
        P = fft(p, N)
        A = fft(a, N)
        grad_full = ifft(P * np.conj(A)).real
        grad = 2.0 * grad_full[:n]

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Multiplicative update
        step = lr * grad / g_norm
        step = np.clip(step, -1.0, 1.0)
        a_new = a * np.exp(-step)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Try smaller step
                a_new = a * np.exp(-0.1 * step)
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / float(np.sum(a_new))
                obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                lr *= 0.7
                stall = 0

        it += 1

    return best_a


def subgradient_descent(a, time_budget=50.0, tau_init=30.0, tau_end=200.0, seed=42):
    """
    Subgradient descent with Polyak step size and running average.
    Uses FFT for gradient computation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    # Lower bound: 2n/(2n-1) for normalized a
    lb = 2.0 * n / (2.0 * n - 1.0) + 0.005

    N = 1
    while N < 3 * n:
        N <<= 1

    lr = 0.5
    a_avg = a.copy()
    count = 0
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj = 2.0 * n * max_conv

        # Smoothed gradient
        shift = conv - max_conv
        exp_vals = np.exp(tau * shift)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        P = fft(p, N)
        A = fft(a, N)
        grad_full = ifft(P * np.conj(A)).real
        grad = 2.0 * grad_full[:n]

        g_norm_sq = float(np.dot(grad, grad))
        if g_norm_sq < 1e-20:
            break

        # Polyak step
        polyak_lr = max(0.01, (obj - lb) / g_norm_sq)
        lr_eff = min(lr, polyak_lr)

        # Projected gradient step
        a_new = a - lr_eff * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        count += 1
        a_avg = ((count - 1) / count) * a_avg + (1.0 / count) * a_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        obj_avg = 2.0 * n * float(np.max(compute_conv(a_avg)))
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()

        a = a_new
        lr =(lr * 0.998 + 0.002 * 0.5 / np.sqrt(n))
        it += 1

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64) / n
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        a = np.exp(-15 * (x - 0.25) ** 2) + np.exp(-15 * (x - 0.75) ** 2)
    elif mode == 3:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 4:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    elif mode == 5:
        a = (x * (1 - x)) ** 0.5 + 0.05
    elif mode == 6:
        # Symmetric U-shape
        a = 0.5 + 0.5 * np.cos(4 * np.pi * x)
    elif mode == 7:
        # Multi-frequency
        a = np.ones(n)
        for f in [2, 5, 11]:
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2*np.pi))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    # Load parent

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 1: Coordinate Descent (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 400.0)
    if p1_budget > 30:
        t0 = time.monotonic()
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, seed=seed, gs_iters=25)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.10f} **")
        else:
            print(f"  CD: obj={obj_cd:.10f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        lams = [0.0, 0.0, 0.05, 0.1, 0.0, 0.05]
        for rnd in range(min(6, len(lams))):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.06, 35.0)
            if alloc < 8:
                break

            t0 = time.monotonic()
            g_lp, M_lp = solve_lp(best_seq, time_budget=alloc, lambda_reg=lams[rnd])
            elapsed_lp = time.monotonic() - t0

            if g_lp is not None:
                best_t_obj = best_score
                best_a_local = best_seq.copy()
                for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
                    a_try = (1.0 - t) * best_seq + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_a_local = a_try.copy()

                if best_t_obj < best_score:
                    best_score = best_t_obj
                    best_seq = best_a_local.copy()
                    print(f"  LP r{rnd} (lam={lams[rnd]}): obj={best_score:.10f} **")
                else:
                    print(f"  LP r{rnd} (lam={lams[rnd]}): obj={best_t_obj:.10f}")
            else:
                print(f"  LP r{rnd}: solver failed ({elapsed_lp:.1f}s)")
            print(f"  LP time: {elapsed_lp:.1f}s")

    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t0 = time.monotonic()
        a_ps = peak_shave(best_seq, time_budget=min(remaining * 0.05, 40.0), seed=seed + 10)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PeakShave: obj={best_score:.10f} **")
        else:
            print(f"  PeakShave: obj={obj_ps:.10f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 4: CD after improvements ===
    print(f"\n=== Phase 4: Post-Improvement CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 120.0)
    if p4_budget > 20:
        t0 = time.monotonic()
        a_cd2 = coordinate_descent(best_seq, time_budget=p4_budget, seed=seed + 500, gs_iters=30)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.10f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.10f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Subgradient Descent ===
    print(f"\n=== Phase 5: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        t0 = time.monotonic()
        alloc = min(remaining * 0.07, 50.0)
        a_sg = subgradient_descent(best_seq, time_budget=alloc, tau_init=20.0, tau_end=150.0, seed=seed + 800)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SubGrad: obj={best_score:.10f} **")
        else:
            print(f"  SubGrad: obj={obj_sg:.10f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

        # CD after subgradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cd3 = coordinate_descent(best_seq, time_budget=min(remaining * 0.08, 60.0), seed=seed + 900)
            obj_cd3 = compute_obj(a_cd3)
            if obj_cd3 < best_score:
                best_score = obj_cd3
                best_seq = a_cd3.copy()
                print(f"  CD after SG: obj={best_score:.10f} **")

    # === Phase 6: More LP rounds ===
    print(f"\n=== Phase 6: Additional LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.04, 30.0)
            if alloc < 5:
                break
            lam = [0.0, 0.1, 0.3][rnd]
            t0 = time.monotonic()
            g_lp, _ = solve_lp(best_seq, time_budget=alloc, lambda_reg=lam)
            if g_lp is not None:
                for t in [0.1, 0.2, 0.3, 0.5, 1.0]:
                    a_try = (1.0 - t) * best_seq + t * g_lp
                    a_try = np.maximum(a_try, 0) / float(np.sum(a_try))
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        print(f"  LP6 r{rnd}: obj={best_score:.10f} **")
                        break
            print(f"  LP6 time: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_candidates = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 150,
        n_current + 200,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500 and abs(n - n_current) >= 30]
    n_candidates = n_candidates[:5]
    print(f"  Trying: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc_cd = min(remaining * 0.12, 35.0)
        if alloc_cd > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd, seed=n_try, gs_iters=20)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 8: Diverse restarts ===
    print(f"\n=== Phase 8: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.15, 40.0)
        if alloc > 10:
            a_r = coordinate_descent(a_r, time_budget=alloc, seed=7000 + r * 100, gs_iters=20)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.10f}")

    # === Phase 9: Final intensive CD ===
    print(f"\n=== Phase 9: Final CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 120.0)
        t0 = time.monotonic()
        a_final = coordinate_descent(best_seq, time_budget=alloc, seed=seed + 99999, gs_iters=35)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
        print(f"  Final CD: obj={best_score:.10f}")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    # === Phase 10: One last LP + CD ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        g_lp, _ = solve_lp(best_seq, time_budget=min(remaining * 0.3, 25.0), lambda_reg=0.0)
        if g_lp is not None:
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 1.0]:
                a_try = (1.0 - t) * best_seq + t * g_lp
                a_try = np.maximum(a_try, 0) / float(np.sum(a_try))
                obj_try = compute_obj(a_try)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_final2 = coordinate_descent(best_seq, time_budget=min(remaining * 0.6, 40.0),
                                              seed=seed + 111111, gs_iters=30)
                obj_final2 = compute_obj(a_final2)
                if obj_final2 < best_score:
                    best_score = obj_final2
                    best_seq = a_final2.copy()
        print(f"  Final: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
