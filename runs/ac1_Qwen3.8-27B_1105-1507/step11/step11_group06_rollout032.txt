```python
    """Solve the FW-direction LP using only active constraints."""
    n = len(a)
    a_norm = a / np.sum(a)
    # Build only the active constraint rows
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        for j in range(n):
            if 0 <= k - j < n:
                A[idx, j] = a_norm[k - j]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.maximum(g.value, 0)
        except:
            continue
    return None
```

```python
def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size *= 2
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

    n = len(a)
```

```python
    """Compute subgradient of max_k (a*a)_k at the given a for lag k_star."""
    n = len(a)

    r = k_star - np.arange(n)  # r[j] = k_star - j
```

"""
Step function optimization using LP-based Frank-Wolfe, accelerated subgradient,
and multi-n exploration. Optimizes 2*n*max(conv(a,a))/(sum(a))^2.
Key: asymmetric sequences can outperform symmetric ones.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Compute convolution of a and b."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgrad(a, k_star):
    """Subgradient of max_k (a*a)_k at lag k_star."""
    n = len(a)
    grad = np.zeros(n)
    # grad[j] = 2*a[k_star - j] if 0 <= k_star-j < n and k_star != 2j
    # grad[j] = a[j] if k_star == 2j
    j_range = np.arange(n)
    r = k_star - j_range
    valid = (r >= 0) & (r < n)
    grad[valid] = 2.0 * a[r[valid]]
    # Fix diagonal
    if k_star % 2 == 0:
        j0 = k_star // 2
        if 0 <= j0 < n:
            grad[j0] = a[j0]
    return grad


def solve_lp_full(a, solver_list=None):
    """Solve the FW-direction LP: min M s.t. (a*g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build convolution matrix A where A[k,j] = a_norm[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_reduced(a, active_lags, solver_list=None):
    """Solve LP with only active constraints (faster)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[idx, j_start:j_end + 1] = a_norm[k - np.arange(j_start, j_end + 1)]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish_cutting(a, time_budget=100.0, max_iters=200):
    """LP-based FW with cutting planes."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    # Initial active set
    conv_a = compute_conv(a)
    active = np.argsort(-conv_a)[:80]
    
    while time.monotonic() - start < time_budget and it < max_iters:
        if time.monotonic() - start > time_budget - 10:
            break
        
        g, M_val = solve_lp_reduced(a, active)
        if g is None:
            # Fallback to full LP
            g = solve_lp_full(a)
            if g is None:
                break
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in np.linspace(0, 1.0, 20):
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 0.01:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 1e-6 and best_obj_ls < best_obj - 1e-12:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                best_obj = best_obj_ls
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            # Try t=1
            obj_direct = compute_obj(g)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g.copy()
                a = g.copy()
                stall = 0
        
        # Update active set
        conv_g = compute_conv_ab(a, g)
        violated = np.where(conv_g > M_val + 1e-6)[0]
        if len(violated) > 0:
            active = np.unique(np.concatenate([active, violated]))
            # Keep top 150 by violation
            if len(active) > 150:
                violations = conv_g[active] - M_val
                top_idx = np.argsort(-violations)[:150]
                active = active[top_idx]
        
        it += 1
        if stall >= 5:
            # Perturb to escape
            rng = np.random.default_rng(it)
            freq = rng.integers(1, max(2, n // 50))
            phase = rng.uniform(0, 2 * np.pi)
            idx = np.arange(n, dtype=np.float64)
            pert = 0.05 * np.cos(2 * np.pi * freq * idx / n + phase)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            # Reset active set
            conv_a = compute_conv(a)
            active = np.argsort(-conv_a)[:80]
            stall = 0
    
    return best_a, best_obj


def accelerated_subgrad(a, time_budget=30.0):
    """Nesterov-accelerated subgradient on the exact objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()
    y = a.copy()
    t_val = 1.0
    it = 0
    best_it = 0
    lr_scale = 0.3
    
    while time.monotonic() - start < time_budget:
        y_norm = y / float(np.sum(y))
        conv = compute_conv(y_norm)
        k_star = int(np.argmax(conv))
        grad = compute_subgrad(y_norm, k_star)
        
        lr = lr_scale / np.sqrt(t_val + 1.0)
        x_new = x - lr * grad
        x_new = np.maximum(x_new, 0)
        s_new = float(np.sum(x_new))
        if s_new < 1e-10:
            x_new = x
            s_new = 1.0
        x_new = x_new / s_new
        
        t_new = (1 + np.sqrt(1 + 4 * t_val * t_val)) / 2
        beta = (t_val - 1) / t_new if t_val > 0 else 0
        y = x_new + beta * (x_new - x)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = x_new
        else:
            y = y / s_y
        
        x = x_new
        t_val = t_new
        
        obj_x = compute_obj(x)
        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x.copy()
            best_it = it
        
        it += 1
        if it - best_it > 3000:
            x = best_a.copy()
            y = best_a.copy()
            t_val = 1.0
            best_it = it
            lr_scale = max(0.05, lr_scale * 0.5)
    
    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:

    elif mode == 2:

    elif mode == 3:

    a = np.maximum(a, 0)
    s = float(np.sum(a))

def interpolate_to_n(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s
