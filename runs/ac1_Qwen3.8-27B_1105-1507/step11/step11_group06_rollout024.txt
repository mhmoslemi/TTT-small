    # Then (A @ g)[k] = sum_i A[k,i] * g[i] = sum_i a[k-i] * g[i]

    A = build_conv_matrix(a_norm)

                s = float(np.sum(g_val))
                if s > 1e-10:

    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def mirror_descent_step(a, conv, eta):
    n = len(a)
    max_val = float(np.max(conv))
    tol = 1e-12 * max_val + 1e-30
    argmax_idx = np.where(conv >= max_val - tol)[0]
    n_argmax = len(argmax_idx)

    grad = np.zeros(n)
    for k in argmax_idx:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        if i_min <= i_max:
            j_start = k - i_max
            j_end = k - i_min
            grad[i_min:i_max + 1] += 2.0 * a[j_start:j_end + 1]

    if n_argmax > 1:
        grad /= float(n_argmax)

    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-300)
    s = float(a_new.sum())
    if s < 1e-10:
        return a.copy()
    return a_new / s


def md_polish(a, time_budget=30.0):
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 1.0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        a_new = mirror_descent_step(a, conv, eta)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = max(eta * 0.9995, 0.05)
        else:
            no_improve += 1
            eta *= 0.7
            if no_improve > 50 or eta < 1e-4:
                break
    return best_a, best_obj


def golden_line_search(a, g, f0, t_max=1.2, n_evals=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def f(t):
        a_try = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = float(a_try.sum())
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    lo = 0.0
    hi = t_max
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)

    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

    t_opt = (lo + hi) / 2.0
    f_opt = f(t_opt)
    if f_opt >= f0:
        return 0.0, f0
    return t_opt, f_opt


def accelerated_subgradient(a, time_budget=30.0):
    n = len(a)
    a = a / a.sum()
    x = a.copy()
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t = 1
    t0 = 2.0 / 3.0

    while time.monotonic() - start < time_budget:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x + (t / t_new) * (x - y)
        y_new = np.maximum(y_new, 0.0)
        s = float(y_new.sum())
        if s < 1e-10:
            y = a.copy()
            x = a.copy()
            break
        y_new = y_new / s
        t = t_new
        y = y_new

        conv = compute_conv(y)
        max_val = float(np.max(conv))
        tol = 1e-10 * max_val
        argmax_idx = np.where(conv >= max_val - tol)[0]

        grad = np.zeros(n)
        for k in argmax_idx:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            if i_min <= i_max:
                j_start = k - i_max
                j_end = k - i_min
                grad[i_min:i_max + 1] += 2.0 * y[j_start:j_end + 1]
        if len(argmax_idx) > 1:
            grad /= float(len(argmax_idx))

        t_cur = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        x_new = y - (t / (2.0 * t_cur)) * grad
        x_new = np.maximum(x_new, 0.0)
        s = float(x_new.sum())
        if s < 1e-10:
            x = best_a.copy()
            y = best_a.copy()
            break
        x_new = x_new / s
        x = x_new

        obj = compute_obj(y)
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()

        obj_x = compute_obj(x)
        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x.copy()

    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7.0
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = 1.0 / (0.1 + 3.0 * np.sin(np.pi * x) ** 2)
    else:
        a = np.cos(np.pi * x) ** 2 + 0.1
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def lp_polish(a, time_budget=100.0, max_iters=200, use_md_perturb=True, verbose=False):
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345 + it)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5.0:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        t_opt, obj_opt = golden_line_search(a, g_lp, best_obj, t_max=1.2, n_evals=30)
        improved = False

        if t_opt > 1e-6 and obj_opt < best_obj - 1e-13:
            a_new = np.maximum((1.0 - t_opt) * a + t_opt * g_lp, 0.0)
            s_new = float(a_new.sum())
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                    improved = True

        if not improved:
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj - 1e-13:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
                improved = True

        if not improved:
            stall += 1

        if stall >= 3 and use_md_perturb:
            if stall <= 5:
                conv_a = compute_conv(a)
                for _ in range(5 + stall * 2):
                    a = mirror_descent_step(a, conv_a, 0.3 / stall)
                    conv_a = compute_conv(a)
            elif stall <= 8:
                mode = it % 7
                a_div = diverse_init(n, rng, mode=mode)
                a = 0.2 * a_div + 0.8 * best_a
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            else:
                a = best_a.copy()
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.02 * stall * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            stall = 0

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.10f}, t={t_opt:.4f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0.0)
    s = float(a_dst.sum())
    if s < 0.01:
        return None
    return a_dst / s


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(parent_arr.sum())
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            print(f"Parent loaded: n={parent_arr.size}, obj={best_score:.10f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        best_seq = diverse_init(800, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, obj={best_score:.10f}")

    n_best = len(best_seq)

    # Try alternative initializations at same n
    for mode in range(7):
        a_alt = diverse_init(n_best, rng, mode=mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Alt init mode={mode}: obj={best_score:.10f} **")

    # Phase 1: Mirror descent warm-up
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: Mirror Descent Warm-up (n={n_best}) ===")
        md_budget = min(remaining * 0.05, 45.0)
        a_md, obj_md = md_polish(best_seq, time_budget=md_budget)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD done: obj={best_score:.10f} **")
        else:
            print(f"  MD done: obj={obj_md:.10f} (no improvement)")

    # Phase 2: Intensive LP polishing
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 2: LP Polish (n={n_best}, budget={min(remaining*0.45, 600):.0f}s) ===")
        alloc = min(remaining * 0.45, 600.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, max_iters=500,
                         use_md_perturb=True, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 2 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase 2 done: obj={obj_lp:.10f} (no improvement)")

    # Phase 3: Accelerated subgradient refinement
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Accelerated Subgradient ===")
        alloc = min(remaining * 0.1, 60.0)
        a_sg, obj_sg = accelerated_subgradient(best_seq, time_budget=alloc)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase 3: obj={best_score:.10f} **")
        else:
            print(f"  Phase 3: obj={obj_sg:.10f} (no improvement)")

    # Phase 4: LP polish after subgradient
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Polish Round 2 ===")
        alloc = min(remaining * 0.15, 120.0)
        a_lp2 = lp_polish(best_seq, time_budget=alloc, max_iters=100,
                          use_md_perturb=True, verbose=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 4: obj={best_score:.10f} **")
        else:
            print(f"  Phase 4: obj={obj_lp2:.10f} (no improvement)")

    # Phase 5: n Exploration
    remaining = deadline - time.monotonic()
    if remaining > 180:
        print(f"\n=== Phase 5: n Exploration ===")
        n_cur = len(best_seq)
        n_options = sorted(set([
            n_cur - 50, n_cur - 30, n_cur - 10,
            n_cur + 10, n_cur + 30, n_cur + 50, n_cur + 100,
            600, 700, 900, 1000, 1200, 1400
        ]))
        n_options = [n for n in n_options if 200 <= n <= 2500 and abs(n - n_cur) >= 10]
        n_options.sort(key=lambda x: abs(x - n_cur))
        n_options = n_options[:8]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            if a_try is None:
                continue
            alloc = min(remaining * 0.25, 80.0)
            if alloc > 20:
                a_try2 = lp_polish(a_try, time_budget=alloc, max_iters=40,
                                   use_md_perturb=True, verbose=False)
            else:
                a_try2 = a_try
            obj_try = compute_obj(a_try2)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try2.copy()
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")
                # Update best for next interpolation
                if obj_try < compute_obj(interpolate_sequence(a_try2, len(a_try2))):
                    pass

    # Phase 6: Final intensive LP on best n
    remaining = deadline - time.monotonic()
    if remaining > 90:
        print(f"\n=== Phase 6: Final Intensive LP ===")
        alloc = min(remaining * 0.4, 300.0)
        a_final = lp_polish(best_seq, time_budget=alloc, max_iters=300,
                            use_md_perturb=True, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_final:.10f}")

    # Phase 7: Mirror descent squeeze
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: MD Squeeze ===")
        alloc = min(remaining * 0.15, 40.0)
        a_md2, obj_md2 = md_polish(best_seq, time_budget=alloc)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  MD squeeze: obj={best_score:.10f} **")

    # Phase 8: Final short LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining - 10, 50.0)
        if alloc > 10:
            a_sq = lp_polish(best_seq, time_budget=alloc, max_iters=30,
                             use_md_perturb=False, verbose=False)
            obj_sq = compute_obj(a_sq)
            if obj_sq < best_score:
                best_score = obj_sq
                best_seq = a_sq.copy()
                print(f"  Final squeeze: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.array(result))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
