    """Compute softmax weights and the gradient of smooth max."""
    # conv has length 2n-1
    max_conv = np.max(conv)
    shifted = conv - max_conv  # for numerical stability
    exp_shifted = np.exp(t * shifted)
    p = exp_shifted / np.sum(exp_shifted)
    return p, max_conv + np.log(np.sum(exp_shifted)) / t

def projected_gradient_step(a, grad, alpha):
    """Take a gradient step and project onto the simplex {sum=1, a>=0}."""
    b = a - alpha * grad
    # Project onto simplex
    # Find tau such that sum(max(b - tau, 0)) = 1
    # Use bisection or sorting
    mu = np.sort(b)[::-1]
    cv = np.cumsum(mu)

    tau = (cv[rho] - 1) / (rho + 1.0)
    a_new = np.maximum(b - tau, 0)

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_norm(a):
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def simplex_project(b):
    """Project b onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(b)
    mu = np.sort(b)[::-1]
    cv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        tau = (np.sum(b) - 1.0) / n
        return np.maximum(b - tau, 0)
    rho = rho[-1]
    tau = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(b - tau, 0)

def smooth_max_gradient(a, t):
    """Compute the gradient of the smooth max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    max_conv = np.max(conv)
    shifted = t * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: grad[m] = 2 * sum_j p[m+j] * a[j]
    # = 2 * (a * reverse(p))[2n-2-m]
    L = 1
    while L < 3 * n:
        L <<= 1
    A = rfft(a, L)
    P_rev = rfft(p[::-1], L)
    conv_ap = irfft(A * P_rev, L)
    grad = 2.0 * conv_ap[n-1:2*n-1][::-1]
    
    return grad, conv, p

def smooth_max_value(conv, t):
    max_conv = np.max(conv)
    shifted = t * (conv - max_conv)
    return max_conv + np.log(np.sum(np.exp(shifted))) / t

def adam_pgd(a, t_schedule, n_iters, verbose=True):
    """
    Adam + Projected Gradient Descent with smooth max.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 1.0 / np.sqrt(n)  # initial learning rate
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t_count = 0
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    
    start = time.monotonic()
    
    for it in range(n_iters):
        # Get current t
        if it < len(t_schedule):
            t = t_schedule[it]
        else:
            t = t_schedule[-1]
        
        # Compute gradient
        grad, conv, p = smooth_max_gradient(a, t)
        
        # Adam update
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        # Step with backtracking
        alpha = lr / (1 + it * 0.01)  # diminishing
        a_try = a - alpha * update
        a_new = simplex_project(a_try)
        
        # Check improvement
        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    PGD iter {it}: obj={best_obj:.10f}, t={t:.1f}, time={elapsed:.1f}s")
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth-max projected gradient descent (Adam) as the primary optimizer,
with LP alternating and coordinate descent for final polishing.
Exploits FFT-based gradient computation for O(n log n) per iteration.

import numpy as np
import time
from scipy.fft import rfft, irfft


def next_pow2(m):
    return 1 << (m - 1).bit_length()


def compute_conv(a):
    n = len(a)
    if n <= 1:
        return np.array([float(a[0])**2]) if n == 1 else np.array([])
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return np.real(irfft(fa * fa, size)[:2 * n - 1])


def compute_obj_norm(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(b):
    """Project vector b onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(b)
    if n == 0:
        return b
    mu = np.sort(b)[::-1]
    cv = np.cumsum(mu)
    rho_candidates = np.nonzero(mu * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho_candidates) == 0:
        tau = (np.sum(b) - 1.0) / n
    else:
        rho = rho_candidates[-1]
        tau = (cv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(b - tau, 0.0)
    # Ensure exact sum = 1 (numerical fix)
    diff = 1.0 - np.sum(result)
    if abs(diff) > 1e-15:
        # Add/subtract from the largest component
        idx = np.argmax(result)
        result[idx] += diff
    return result


def smooth_max_grad(a, t):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*(a*a)[k])) with respect to a.
    Returns (gradient, conv, smoothed_max_value).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Numerically stable softmax
    max_c = float(np.max(conv))
    shifted = t * (conv - max_c)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / float(np.sum(exp_shifted))
    smooth_val = max_c + float(np.log(np.sum(exp_shifted))) / t
    
    # Gradient: grad[m] = 2 * sum_{j=0}^{n-1} p[m+j] * a[j]
    # Compute via FFT: grad[m] = 2 * (a * reverse(p))[2n-2-m]
    L = next_pow2(3 * n)
    A = rfft(a, L)
    P_rev = rfft(p[::-1], L)
    conv_ap = np.real(irfft(A * P_rev, L))
    
    # Extract: grad[m] = 2 * conv_ap[2n-2-m] for m=0,...,n-1
    # Indices: conv_ap[2n-2], conv_ap[2n-3], ..., conv_ap[n-1]
    grad = 2.0 * conv_ap[n-1:2*n-1][::-1]
    
    return grad, conv, smooth_val


def adam_pgd(a, time_budget, seed=0, verbose=True):
    """
    Adam + Projected Gradient Descent for minimizing smooth max of conv(a,a)
    subject to a being on the probability simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    rng = np.random.default_rng(seed)
    
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 2.0 / np.sqrt(n)  # base learning rate
    
    # t schedule: start small, increase over time
    t_start = 5.0
    t_end = 100.0
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    start = time.monotonic()
    it = 0
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() - start < time_budget and no_improve < max_no_improve:
        # Schedule t
        fraction = min(1.0, (time.monotonic() - start) / time_budget)
        t = t_start + (t_end - t_start) * fraction ** 0.5
        
        # Compute gradient
        grad, conv, _ = smooth_max_grad(a, t)
        
        # Adam update
        t_count += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        
        # Adaptive step size
        alpha = lr / (1.0 + 0.001 * it)
        
        # Gradient step + projection
        a_try = a - alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = simplex_project(a_try)
        
        # Evaluate
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      PGD {it}: obj={best_obj:.10f}, t={t:.1f}, alpha={alpha:.4f}, time={elapsed:.1f}s")
    
    elapsed = time.monotonic() - start
    if verbose:
        print(f"      PGD done: {it} iters, best={best_obj:.10f}, time={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a_norm*g)[k] <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        try:
            g.value = np.asarray(warm_start).copy()
        except Exception:
            pass
    
    for solver_name in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver_name))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """LP alternating minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0
    
    while time.monotonic() - start < time_budget - 5:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1
        
        best_local_obj = best_obj
        best_local_a = a.copy()
        t_values = [1.0, 0.7, 0.5, 0.3, 0.15, 0.1, 0.05]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                print(f"      LP {iter_count}: obj={best_obj:.10f}, time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall >= 6:
                strength = 0.003 + 0.002 * min(stall, 8)
                pert = rng.standard_normal(n) * strength
                a = best_a + pert * best_a
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """Coordinate descent on normalized sequence."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 8
    
    while time.monotonic() - start < time_budget:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))
            
            mean_val = 1.0 / n
            if a_j < 1e-12:
                candidates = np.array([0.0, mean_val * 0.1, mean_val * 0.5, mean_val, mean_val * 2])
            else:
                candidates = np.array([
                    a_j * 0.3, a_j * 0.5, a_j * 0.7, a_j * 1.0,
                    a_j * 1.3, a_j * 1.7, a_j * 2.5,
                    mean_val * 0.5, mean_val, mean_val * 2, mean_val * 4
                ])
                candidates = candidates[(candidates >= 0) & (candidates <= 0.5)]
                candidates = np.unique(candidates)
            
            best_trial_obj = best_obj
            best_trial_val = a_j
            
            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue
                
                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)
                
                max_conv_unscaled = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_unscaled / (S_new * S_new)
                
                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val
            
            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta
                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)
                
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    
    return best_a


def make_raised_cosine(n, beta=1.0):
    """Create a raised cosine window."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * beta * (x - 0.5)))
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def make_gaussian(n, sigma=0.3):
    """Create a Gaussian window."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    s = np.sum(a)
    return a / s


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
