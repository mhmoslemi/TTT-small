def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)

    a_norm = a / s

    """Build (2n-1) x n matrix A where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))

            continue
        stall_count = 0

        # Adaptive step size: try large steps first, bisect if needed
        # Use recent best t as starting point
        t_start = t_history[-1] if t_history else 0.3
        
        # Try a set of t values centered around t_start
        t_candidates = []
        for factor in [1.0, 0.7, 1.4, 0.5, 2.0, 0.3, 3.0]:
            t_val = min(1.0, max(0.01, t_start * factor))
            t_candidates.append(t_val)
        t_candidates.append(1.0)
        t_candidates.append(0.1)
        
        # Deduplicate and sort
        t_candidates = sorted(set(round(t, 4) for t in t_candidates), reverse=True)

        best_local_obj = best_obj
        best_local_a = a.copy()
        best_local_t = t_start

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-12:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-12:
            # Momentum update: blend with previous direction
            direction = best_local_a - a
            momentum = beta_mom * momentum + (1 - beta_mom) * direction
            a_mom = best_local_a + 0.1 * momentum
            a_mom = np.maximum(a_mom, 0)
            sm = float(np.sum(a_mom))
            if sm > 1e-10:
                a_mom = a_mom / sm
                obj_mom = compute_obj(a_mom)
                if obj_mom < best_local_obj:
                    best_local_obj = obj_mom
                    best_local_a = a_mom.copy()

            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            t_history.append(best_local_t)
            if len(t_history) > 10:
                t_history = t_history[-10:]
            if verbose and it % 50 == 0:
                print(f"    LP it{it:4d}: obj={best_obj:.10f}, t={best_local_t:.4f}")
        else:
            # Stuck: try perturbation
            stall_count += 1
            if stall_count % 3 == 1:
                # Subgradient step targeting active lags
                conv = compute_conv(a)
                max_val = float(np.max(conv))
                active = np.where(conv >= max_val * 0.999)[0]
                subgrad = np.zeros(n)
                for lag in active:
                    for i in range(n):
                        j = lag - i
                        if 0 <= j < n:
                            subgrad[i] += 2.0 * a[j]
                subgrad /= max(len(active), 1)
                gn = np.linalg.norm(subgrad)
                if gn > 1e-15:
                    for step in [0.001, 0.003, 0.01, 0.03]:
                        a_try = a - step * subgrad / gn
                        a_try = np.maximum(a_try, 0)
                        st = float(np.sum(a_try))
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            a = a_try
                            break
            elif stall_count % 3 == 2:
                # Random perturbation
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                # Frequency perturbation
                freqs = rng.integers(1, 8, size=2)
                phases = rng.uniform(0, 2 * np.pi, size=2)
                x = np.arange(n)
                pert = np.zeros(n)
                for f, p in zip(freqs, phases):
                    pert += 0.01 * np.sin(2 * np.pi * f * x / n + p)
                a = a + pert
                a = np.maximum(a, 0)
                a = a / a.sum()

            if stall_count > 30:
                a = best_a.copy()
                stall_count = 0
                t_history = [0.3]
                momentum = np.zeros(n)

    return best_a


def subgradient_refine(a, time_budget, verbose=False, seed=42, max_iters=5000):
    """Subgradient descent with adaptive step size targeting active lags."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    step_history = []
    gamma = 1.0
    c_step = 0.5  # initial step size scale

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        active = np.where(conv >= max_val * 0.998)[0]

        # Compute subgradient (average over active lags)
        subgrad = np.zeros(n)
        for lag in active:
            lo = max(0, lag - (n - 1))
            hi = min(n - 1, lag)
            for i in range(lo, hi + 1):
                subgrad[i] += 2.0 * a[lag - i]
        subgrad /= max(len(active), 1)

        gn = np.linalg.norm(subgrad)
        if gn < 1e-15:
            break
        subgrad = subgrad / gn

        # Adaptive step size
        if len(step_history) >= 2:
            if step_history[-1] > step_history[-2]:
                c_step *= 0.95
            else:
                c_step *= 1.05
        c_step = max(0.0001, min(0.5, c_step))

        # Try decreasing step sizes
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1]:
            step = c_step * mult
            a_try = a - step * subgrad
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                step_history.append(step)
                if len(step_history) > 5:
                    step_history = step_history[-5:]
                improved = True
                break

        if not improved:
            step_history.append(0)
            if len(step_history) > 5:
                step_history = step_history[-5:]
            #Perturb
            if it % 10 == 0:
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
            if it > 100 and it % 100 == 0:
                a = 0.95 * a + 0.05 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

        if verbose and it % 500 == 0:
            print(f"    SG it{it:5d}: obj={best_obj:.10f}, step={c_step:.6f}")

    return best_a


def gradient_descent_smooth(a, time_budget, verbose=False, seed=42):
    """Smoothed gradient descent with log-sum-exp approximation to max."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    tau_levels = [5, 15, 50, 150, 500, 1500, 4000]
    N = 1
    while N < 4 * n:
        N <<= 1

    for tau in tau_levels:
        lr = 1.0 / np.sqrt(n)
        v = np.zeros(n)
        beta = 0.9
        max_iters_level = 3000

        for it in range(max_iters_level):
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 0.5:
                break

            # Nesterov
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute smoothed gradient
            conv = compute_conv(y)
            exact_max = float(np.max(conv))
            shifted = (conv - exact_max) / tau
            shifted = np.clip(shifted, -500, 500)
            exp_vals = np.exp(shifted)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            # Gradient via FFT
            P = fft(p, N)
            A_fft = fft(y, N)
            cross = ifft(P * np.conj(A_fft)).real
            g = 2.0 * cross[:n]

            # Project gradient onto sum=0 subspace
            g -= np.mean(g)

            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            sn = float(np.sum(a_new))
            if sn < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / sn

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            lr = max(1e-8, lr * 0.99999)

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x / 2), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(n - 1, 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(n - 1, 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        # Exponential ramp (asymmetric)
        x = np.arange(n) / n
        a = np.exp(-3 * x)
        return a / a.sum()
    elif mode == 7:
        # Multi-scale cosine
        x = np.arange(n) / n
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 8:
        # Flat center, tapered edges
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        width = n * 0.3
        a = np.where(d < width, 1.0, np.exp(-(d - width) / (n * 0.05)))
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()


def propose_candidate(budget_s=1500):

    Main search function. Combines LP fixed-point, subgradient refinement,
    smoothed GD, and multi-n exploration with careful time management.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0  # safety buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(42)

    # --- Load parent ---
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_seq = parent.copy()
                best_score = compute_obj(parent)
                n_parent = len(parent)
                print(f"Loaded parent: n={n_parent}, obj={best_score:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n_parent = 800
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init: n={n_parent}, obj={best_score:.10f}")

    # === Phase 1: Brief GD warm-up ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.04, 30.0)
        a1 = gradient_descent_smooth(best_seq, time_budget=alloc, verbose=False)
        obj1 = compute_obj(a1)
        if obj1 < best_score:

            best_seq = a1.copy()
            print(f"Phase 1 (GD warm): obj={best_score:.10f}")

            best_score = obj2

    # === Phase 3: Subgradient Refinement ===

        alloc = min(remaining * 0.15, 150.0)

            print(f"Phase 3 (SG): obj={best_score:.10f} **")
        else:
            print(f"Phase 3 (SG): obj={obj3:.10f}")

    # === Phase 4: Second LP Round (with reduced constraints for speed) ===
    remaining = deadline - time.monotonic()
    if remaining > 150:
        lp_budget2 = min(remaining * 0.25, 300.0)
        print(f"\nPhase 4: LP Round 2 - Reduced (budget={lp_budget2:.0f}s)")

            best_score = obj4

            print(f"Phase 4 (LP2): obj={best_score:.10f} **")

    # === Phase 5: Multi-n Exploration ===
    remaining = deadline - time.monotonic()
    print(f"\nPhase 5: Multi-n Exploration (remaining={remaining:.0f}s)")
    if remaining > 100:
        n_current = len(best_seq)
        n_options = sorted(set([
            n_current - 100, n_current - 50,
            n_current + 50, n_current + 100, n_current + 200,
            600, 700, 900, 1000, 1100, 1200, 1400, 1600
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_current]
        n_options.sort(key=lambda x: abs(x - n_current))

        a7 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True,
                           seed=9999, max_iters=2000, use_reduced=False)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7

            print(f"Phase 7 (Final LP): obj={best_score:.10f} **")
        else:
            print(f"Phase 7 (Final LP): obj={obj7:.10f}")

    # === Phase 8: Final subgradient micro-refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.5, 60.0)
        print(f"\nPhase 8: Final SG Polish (budget={alloc:.0f}s)")
