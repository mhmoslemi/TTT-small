"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LSE-smoothed gradient descent with Nesterov acceleration, LP alternating
minimization, coordinate descent polishing, multi-scale initialization,
and diverse restart strategies.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_smoothed_objective(a, tau):
    """Compute LSE-smoothed objective: 2n * tau * log(sum exp(conv/tau))."""
    n = len(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    val = conv_max + tau * np.log(np.sum(np.exp((conv - conv_max) / tau)))
    return 2.0 * n * val


def lse_gradient(a, tau):
    """
    Gradient of 2n * tau * log(sum_k exp(conv[k]/tau)) w.r.t. a.
    Result: 4n * C where C[j] = sum_m p[m+j]*a[m], p = softmax(conv/tau).
    """
    n = len(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    exp_vals = np.exp((conv - conv_max) / tau)
    p = exp_vals / np.sum(exp_vals)

    # C[j] = sum_{m=0}^{n-1} p[m+j] * a[m]
    # Computed via: C[j] = (p_flipped * a)[2n-2-j]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    p_flipped = p[::-1].copy()
    fp = rfft(p_flipped, size)
    fa = rfft(a, size)
    conv_result = irfft(fp * fa, size)
    C = conv_result[2 * n - 2: n - 2: -1]

    return 4.0 * n * C


def nesterov_lse_optimize(a, time_budget, seed=42, verbose=True):
    """
    Nesterov-accelerated projected gradient on LSE-smoothed objective.
    a must be on the simplex (sum=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Start with larger tau (smoother) and anneal
    tau_schedule_start = max(0.01, compute_obj_norm(a) / (2.0 * n) * 0.1)
    tau_schedule_end = max(0.001, compute_obj_norm(a) / (2.0 * n) * 0.01)

    # Nesterov parameters
    t = 1.0
    x_prev = a.copy()
    x_curr = a.copy()
    eta = 1.0  # initial step size

    iter_count = 0
    stall = 0
    max_stall = 200

    while time.monotonic() - start < time_budget - 2:
        # Anneal tau
        elapsed_frac = min(1.0, (time.monotonic() - start) / time_budget)
        tau = tau_schedule_start * (tau_schedule_end / tau_schedule_start) ** elapsed_frac
        tau = max(tau, 1e-5)

        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_next
        y = x_curr + momentum * (x_curr - x_prev)

        # Compute gradient at y (project y first for stability)
        y_proj = simplex_project(y)
        grad = lse_gradient(y_proj, tau)

        # Backtracking line search
        a_trial = simplex_project(y_proj - eta * grad)
        obj_trial = compute_obj_norm(a_trial)

        backtrack_count = 0
        while obj_trial >= compute_obj_norm(y_proj) - 1e-15 and backtrack_count < 15:
            eta *= 0.5
            a_trial = simplex_project(y_proj - eta * grad)
            obj_trial = compute_obj_norm(a_trial)
            backtrack_count += 1

        # Update
        x_prev = x_curr
        x_curr = a_trial
        t = t_next
        iter_count += 1

        if obj_trial < best_obj - 1e-12:
            best_obj = obj_trial
            best_a = a_trial.copy()
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                # Perturb to escape
                pert = rng.standard_normal(n) * 0.01
                a_pert = best_a + pert * best_a
                a_pert = simplex_project(a_pert)
                x_prev = a_pert
                x_curr = a_pert
                t = 1.0
                eta = max(eta, 0.1)
                stall = 0

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"      Nesterov iter {iter_count}: obj={best_obj:.10f}, tau={tau:.6f}, eta={eta:.6f}, t={elapsed:.1f}s")

    return best_a

        # For very large n, use a simplified approach
        return None, None
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS', 'PDLP']:
        try:
            prob.solve(solver=getattr(cp, solver), timeout=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """LP alternating minimization with adaptive step size and warm starts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 5:
        t0 = time.monotonic()
        lp_time = min(20, time_budget - 10 - (time.monotonic() - start))
        if lp_time < 3:
            break
        g_lp, M_lp = solve_lp(a, warm_start=prev_g, time_limit=lp_time)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        # Adaptive line search
        best_local_obj = best_obj
        best_local_a = a.copy()
        best_t = 0.0

        t_values = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 6:
                strength = 0.003 + 0.002 * min(stall, 8)
                pert = rng.standard_normal(n) * strength
                a = best_a + pert * best_a
                a = simplex_project(a)
                prev_g = None
                stall = 0

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """
    Coordinate descent on normalized sequence (sum=1).
    Efficiently updates convolution when changing one coordinate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 4
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Try several candidate values
            candidates = []
            if a_j < 1e-12:
                candidates = [0.0, mean_val * 0.2, mean_val * 0.5, mean_val, mean_val * 2]
            else:
                candidates = [a_j * 0.3, a_j * 0.6, a_j * 0.85,
                              a_j * 1.15, a_j * 1.5, a_j * 2.0,
                              mean_val * 0.5, mean_val * 1.5, mean_val * 3]
                candidates = [c for c in candidates if 0 <= c <= 0.5]

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_unscaled = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_unscaled / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def multi_scale_initialize(target_n, seed=42):
    """Solve at increasing scales to get a good initial guess."""
    rng = np.random.default_rng(seed)
    scales = [max(50, target_n // 16), max(100, target_n // 8),
              max(200, target_n // 4), max(400, target_n // 2)]
    scales = [s for s in scales if s <= target_n]
    if not scales or scales[-1] < target_n:
        scales.append(target_n)
    # Remove duplicates and ensure sorted
    scales = sorted(set(scales))

    a = np.ones(scales[0]) / scales[0]
    for i, n_scale in enumerate(scales):
        if i > 0:
            # Interpolate from previous scale
            a_prev = a
            x_old = np.linspace(0, 1, len(a_prev))
            x_new = np.linspace(0, 1, n_scale)
            a = np.interp(x_new, x_old, a_prev)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

        # Quick LP at this scale
        time_per_scale = min(30, 120 / len(scales))
        a = lp_alternate(a, time_budget=time_per_scale, seed=seed + i * 100, verbose=False)
        a = coordinate_descent_polish(a, time_budget=10, seed=seed + i * 200, verbose=False)

    # Final interpolation to target_n
    if len(a) != target_n:
        x_old = np.linspace(0, 1, len(a))
        x_new = np.linspace(0, 1, target_n)
        a = np.interp(x_new, x_old, a)
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s

    return a


def diverse_initializations(n, seed=42):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian
    sigma = 0.25
    g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc[0] = 0
    rc[-1] = 0
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())

    # Triangular
    tri = 1 - np.abs(2 * x - 1)
    tri = np.maximum(tri, 1e-10)
    inits.append(tri / tri.sum())

    # Broad Gaussian
    sigma2 = 0.4
    g2 = np.exp(-0.5 * ((x - 0.5) / sigma2) ** 2)
    inits.append(g2 / g2.sum())

    # Periodic modulation
    mod = 1 + 0.3 * np.cos(2 * np.pi * 3 * x)
    mod = np.maximum(mod, 0.1)
    inits.append(mod / mod.sum())

    # Sinc-like
    z = (x - 0.5) * 4
    sinc = np.sinc(z)
    sinc = np.maximum(sinc, 0.01)
    inits.append(sinc / sinc.sum())

    return inits


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining multi-scale init, LSE-Nesterov, LP alternating,
    CD polish, diverse restarts, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    # Determine target length
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size == 0 or parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        target_n = parent.size
        parent = parent / parent.sum()
        print(f"Parent: n={target_n}, obj={compute_obj_norm(parent):.10f}")
    except Exception:
        target_n = 800
        parent = np.ones(target_n) / target_n
        print(f"Default: n={target_n}, obj={compute_obj_norm(parent):.10f}")

    best_a = parent.copy()
    best_obj = compute_obj_norm(parent)
    rng = np.random.default_rng(seed)

    # === Phase 0: Multi-scale initialization (5% time) ===
    print("\n--- Phase 0: Multi-scale Init ---")
    p0_budget = min(budget_s * 0.05, 40.0)
    remaining = deadline - time.monotonic()
    if remaining > p0_budget and target_n > 200:
        a_ms = multi_scale_initialize(target_n, seed=seed)
        obj_ms = compute_obj_norm(a_ms)
        if obj_ms < best_obj:
            best_obj = obj_ms
            best_a = a_ms.copy()
            print(f"  Multi-scale: {best_obj:.10f} **")
        else:
            print(f"  Multi-scale: {obj_ms:.10f} (parent better)")

    # === Phase 1: LSE-Nesterov gradient (25% time) ===
    print("\n--- Phase 1: LSE-Nesterov ---")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p1_budget = min(remaining * 0.3, budget_s * 0.25)
        a1 = nesterov_lse_optimize(best_a, time_budget=p1_budget, seed=seed, verbose=True)
        obj1 = compute_obj_norm(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            print(f"  Phase 1: {best_obj:.10f} **")
        else:
            print(f"  Phase 1: {obj1:.10f}")

    # === Phase 2: LP Alternating (30% time) ===
    print("\n--- Phase 2: LP Alternating ---")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.4, budget_s * 0.30)
        a2 = lp_alternate(best_a, time_budget=p2_budget, seed=seed + 1000, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  Phase 2: {best_obj:.10f} **")
        else:
            print(f"  Phase 2: {obj2:.10f}")

    # === Phase 3: CD Polish (10% time) ===
    print("\n--- Phase 3: CD Polish ---")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.25, budget_s * 0.10)
        a3 = coordinate_descent_polish(best_a, time_budget=p3_budget, seed=seed + 2000, verbose=False)
        obj3 = compute_obj_norm(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  Phase 3: {best_obj:.10f} **")
        else:
            print(f"  Phase 3: {obj3:.10f}")

    # === Phase 4: Diverse restarts (20% time) ===
    print("\n--- Phase 4: Diverse Restarts ---")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        p4_total = min(remaining * 0.4, budget_s * 0.20)
        n_restarts = max(2, int(p4_total / 60))
        per_restart = p4_total / n_restarts

        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            # Choose perturbation strategy
            strategy = r % 4
            if strategy == 0:
                a_try = fourier_perturb(best_a, rng, strength=0.015)
            elif strategy == 1:
                # Blend with a diverse init
                inits = diverse_initializations(target_n, seed=seed + r * 7)
                a_try = 0.7 * best_a + 0.3 * inits[r % len(inits)]
                a_try = simplex_project(a_try)
            elif strategy == 2:
                # Random perturbation
                a_try = best_a + rng.standard_normal(target_n) * 0.02
                a_try = simplex_project(a_try)
            else:
                # Coarse-to-fine: blur then re-sharpen
                from scipy.ndimage import gaussian_filter1d
                a_blur = gaussian_filter1d(best_a, sigma=target_n * 0.03)
                a_try = 0.5 * best_a + 0.5 * a_blur
                a_try = simplex_project(a_try)

            # Optimize the perturbed state
            alloc = min(per_restart, remaining - 20)
            if alloc > 30:
                a_try = lp_alternate(a_try, time_budget=alloc * 0.5, seed=seed + 5000 + r * 100, verbose=False)
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    a_try = nesterov_lse_optimize(a_try, time_budget=min(alloc * 0.3, remaining - 10),
                                                   seed=seed + 6000 + r * 100, verbose=False)

            obj_try = compute_obj_norm(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  Restart {r} (strategy {strategy}): {best_obj:.10f} **")
            else:
                print(f"  Restart {r} (strategy {strategy}): {obj_try:.10f}")

    # === Phase 5: Length exploration (10% time) ===
    print("\n--- Phase 5: Length Exploration ---")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = []
    for delta in [-50, -25, 25, 50, 100, -100, 150, 200]:
        n_try = n_best + delta
        if 300 <= n_try <= 2000 and n_try != n_best:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_a)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s

        alloc = min(remaining * 0.35, 100.0)
        if alloc > 40:
            a_try = lp_alternate(a_try, time_budget=alloc * 0.5, seed=n_try * 7, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try = coordinate_descent_polish(a_try, time_budget=min(alloc * 0.3, remaining - 10),
                                                  seed=n_try * 13, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 6: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n--- Phase 6: Final Polish ---")
        p6_budget = min(remaining * 0.7, 60.0)
        a_fin = coordinate_descent_polish(best_a, time_budget=p6_budget, seed=seed + 42424, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final CD: {best_obj:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_fin2 = nesterov_lse_optimize(best_a, time_budget=min(remaining * 0.6, 40.0),
                                           seed=seed + 99999, verbose=False)
            obj_fin2 = compute_obj_norm(a_fin2)
            if obj_fin2 < best_obj:
                best_obj = obj_fin2
                best_a = a_fin2.copy()
                print(f"  Final Nesterov: {best_obj:.10f} **")

    # Final validation and return
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
