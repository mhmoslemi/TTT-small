"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based soft-max gradient descent with staircase tau scheduling,
combined with reduced-constraint LP fixed-point polishing and diverse asymmetric
initializations. Key improvement over prior work: much faster gradient evaluation
via FFT cross-correlation enables 10x more iterations in the same budget.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def soft_max_grad(a, tau):
    """
    Compute gradient of F_tau(a) = tau * log(sum_k exp(c_k/tau))
    where c = full_conv(a, a).
    Returns (gradient, convolution, max_conv).
    """
    n = len(a)
    c = compute_conv(a)
    c_max = float(np.max(c))
    # Numerically stable soft-max weights
    shifted = (c - c_max) / tau
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))
    # Gradient: grad_i = 2 * sum_k p_k * a_{k-i} (cross-correlation)
    N2 = 1
    while N2 < len(p) + n - 1:
        N2 <<= 1
    P = fft(p, N2)
    A = fft(a, N2)
    grad = 2.0 * ifft(np.conj(P) * A).real[:n]
    return grad, c, c_max


def l2_energy_grad(a):
    """
    Gradient of sum_k c_k^2 (energy of convolution).
    This equals 4 * (a * a * a) convolved appropriately.
    Specifically: grad_i = 4 * sum_{j,k: j+k=2i} ... 
    Actually: d/da_i sum_k c_k^2 = 4 * sum_k c_k * (dc_k/da_i)
    dc_k/da_i = 2*a_{k-i} (if valid)
    So grad_i = 8 * sum_k c_k * a_{k-i} = 8 * cross_correlation(c, a)_i
    """
    n = len(a)
    c = compute_conv(a)
    N2 = 1
    while N2 < len(c) + n - 1:
        N2 <<= 1
    C = fft(c, N2)
    A = fft(a, N2)
    grad = 8.0 * ifft(np.conj(C) * A).real[:n]
    return grad, c


def fast_projected_gd(a, time_budget, tau_levels=None, step_scale=0.05, verbose=False):
    """
    Fast projected gradient descent with staircase tau.
    Uses FFT-based gradient for O(n log n) per iteration.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        # Staircase from smooth to sharp
        tau_levels = [5.0, 10.0, 25.0, 60.0, 150.0, 400.0, 1000.0, 2500.0, 6000.0]

    all_tau = list(reversed(tau_levels))  # start sharp, go smooth? No, start smooth.
    all_tau = tau_levels  # start smooth, end sharp

    for tau in all_tau:
        if time.monotonic() - start >= time_budget:
            break

        v = np.zeros(n)
        beta = 0.85
        iters = 0
        max_iters = 30000
        lr = step_scale / n  # initial learning rate

        while iters < max_iters and time.monotonic() - start < time_budget:
            # Nesterov lookahead
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            else:
                y = y / sy

            # Compute gradient
            g, c_vals, c_max = soft_max_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-18:
                break

            # Adaptive step: target displacement of step_scale/n
            target_step = step_scale / n
            lr = target_step / max(g_norm, 1e-15)
            lr = min(lr, 20.0 * n)  # cap

            v_new = beta * v - lr * g
            a_new = y + v_new
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = y.copy()
                v = np.zeros(n)
                s_new = 1.0
            else:
                a_new = a_new / s_new
                v = v_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                # Slightly increase step on success
                step_scale = min(step_scale * 1.002, 0.5)
            else:
                # No improvement: reduce step
                step_scale = max(step_scale * 0.8, 0.001)
                # Revert
                a_new = y.copy()
                v = beta * v

            a = a_new
            iters += 1

        if verbose and it % 10 == 0:
            pass
        if verbose:
            print(f"    tau={tau:7.1f}: obj={best_obj:.10f}, iters={iters}, step={step_scale:.4f}")

        # Reset momentum between tau levels
        step_scale = max(0.02, step_scale * 0.7)

    return best_a


def lp_solve_reduced(a, K=80, max_time=5.0):
    """
    Solve min M s.t. (a*g)_k <= M for top K lags, g >= 0, sum(g) = 1.
    Uses reduced constraint set for speed.
    """
    n = len(a)
    c = compute_conv(a)
    # Get top K lags
    top_idx = np.argsort(c)[::-1][:K]

    # Build constraint matrix for these lags only: A[k, i] = a[i] if 0 <= k-i < n
    A = np.zeros((K, n))
    for idx, k in enumerate(top_idx):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[idx, i] = a[j]

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver, verbose=False)
            if time.monotonic() - t0 > max_time:
                continue
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None

def make_asymmetric_init(n, mode=0, rng=None):
    """Create diverse asymmetric initializations."""
    if rng is None:
        rng = np.random.default_rng(42)

    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Linear ramp (strongly asymmetric)
        a = np.linspace(0.1, 1.0, n)
    elif mode == 2:
        # Exponential decay
        a = np.exp(-np.arange(n) / (n / 3.0))
    elif mode == 3:
        # Two-hump asymmetric
        x = np.arange(n) / n
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 1.5 * x) + 0.1 * np.sin(4 * np.pi * x)
    elif mode == 4:
        # Gaussian off-center
        x = np.arange(n) / n
        a = np.exp(-8.0 * (x - 0.35) ** 2) + 0.1
    elif mode == 5:
        # Superposition of shifted Gaussians
        x = np.arange(n) / n
        a = np.exp(-12.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.7) ** 2) + 0.05
    elif mode == 6:
        # Parabolic
        x = np.arange(n) / n
        a = 1.0 - 0.5 * (2 * x - 0.6) ** 2 + 0.05
    elif mode == 7:
        # Random smooth (interpolated noise)
        k = max(5, n // 20)
        noise = rng.uniform(0.5, 1.5, k)
        x_old = np.linspace(0, 1, k)
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, noise)
    elif mode == 8:
        # Sine-modulated
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x + 0.5) + 0.1 * np.sin(2 * np.pi * 7 * x)
    elif mode == 9:
        # Power law
        x = np.arange(n) / n
        a = x ** 0.7 + 0.1
    else:
        a = rng.uniform(0.2, 1.0, n)

    a = np.maximum(a, 0.001)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def active_set_descent(a, time_budget, verbose=False, seed=42):
    """
    Targeted descent: at each step, compute the subgradient of max(c)
    and take a projected step. Uses the exact max (not soft approximation).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    v = np.zeros(n)
    beta = 0.9
    step = 0.01 / n

    for it in range(50000):
        if time.monotonic() - start >= time_budget:
            break

        c = compute_conv(a)
        c_max = float(np.max(c))
        # Active set: lags at or near the max
        threshold = c_max * (1.0 - 1e-6)
        active = np.where(c >= threshold)[0]

        # Subgradient: average of gradients over active lags
        # grad_i = 2 * a_{k-i} for active lag k
        # So subgrad_i = (2/|S|) * sum_{k in S} a_{k-i}
        # This is (2/|S|) * cross_correlation(indicator_S, a)_i
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)

        N2 = 1
        while N2 < len(c) + n - 1:
            N2 <<= 1
        W = fft(w, N2)
        A = fft(a, N2)
        subgrad = 2.0 * ifft(np.conj(W) * A).real[:n]

        # Also blend with L2 gradient for smoother descent
        g_l2, _ = l2_energy_grad(a)
        g_l2n = np.linalg.norm(g_l2)
        if g_l2n > 1e-15:
            subgrad = 0.7 * subgrad + 0.3 * g_l2 / g_l2n * np.linalg.norm(subgrad) / (np.linalg.norm(subgrad) + 1e-15)

        sg_norm = np.linalg.norm(subgrad)
        if sg_norm < 1e-18:
            break

        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy

        v = beta * v - step * subgrad / sg_norm
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = a
            v = np.zeros(n)
        else:
            a_new = a_new / sn

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            step = min(step * 1.01, 0.1 / n)
        else:
            step = max(step * 0.7, 1e-6 / n)
            a_new = y
            v = beta * v

        a = a_new
        it += 1

        # Periodic perturbation
        if it % 500 == 0 and it > 0:
            a = a + 0.003 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)

    return best_a

        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 LP: obj={best_score:.10f} **")
        else:
            print(f"  Phase3 LP: obj={obj3:.10f}")

    # --- Phase 4: Active-set descent ---
    print(f"\n=== Phase 4: Active-Set Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.12, 120.0)
        a4 = active_set_descent(best_seq, time_budget=alloc, verbose=False, seed=77)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 ActiveSet: obj={best_score:.10f} **")
        else:
            print(f"  Phase4 ActiveSet: obj={obj4:.10f}")

    # --- Phase 5: Second GP round with different K ---
    print(f"\n=== Phase 5: LP with More Constraints ===")
    remaining = deadline - time.monotonic()
    if remaining > 80 and len(best_seq) <= 1200:
        alloc = min(remaining * 0.15, 150.0)
        a5 = lp_fixed_point_fast(best_seq, time_budget=alloc, K=200, verbose=True, seed=555)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5 LP-K200: obj={best_score:.10f} **")
        else:
            print(f"  Phase5 LP-K200: obj={obj5:.10f}")

    # --- Phase 6: Multi-n exploration ---
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_current = len(best_seq)
        n_options = []
        for dn in [-100, -50, 50, 100, 150, 200]:
            n_try = n_current + dn
            if 400 <= n_try <= 2000:
                n_options.append(n_try)
        # Also try some fixed values
        for nf in [700, 800, 900, 1000, 1100]:
            if 400 <= nf <= 2000 and nf != n_current:
                n_options.append(nf)
        n_options = sorted(set(n_options), key=lambda x: abs(x - n_current))[:4]
        print(f"  Testing n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_try = resample_sequence(best_seq, n_try)
            alloc1 = min(remaining * 0.15, 30.0)
            if alloc1 > 10:
                a_try = fast_projected_gd(a_try, time_budget=alloc1,
                                           tau_levels=[10, 50, 200], verbose=False)
            remaining = deadline - time.monotonic()
            if n_try <= 1200 and remaining > 40:
                alloc2 = min(remaining * 0.15, 40.0)
                a_try = lp_fixed_point_fast(a_try, time_budget=alloc2, K=60,
                                             verbose=False, seed=n_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # --- Phase 7: Final intensive polish ---
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.20, 60.0)
        a_f = fast_projected_gd(best_seq, time_budget=alloc,
                                 tau_levels=[20, 60, 200, 600, 1500, 4000],
                                 step_scale=0.03, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25 and len(best_seq) <= 1200:
        alloc = min(remaining * 0.30, 70.0)
        a_f2 = lp_fixed_point_fast(best_seq, time_budget=alloc, K=100,
                                    verbose=True, seed=9999)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 40.0)
        a_f3 = active_set_descent(best_seq, time_budget=alloc, verbose=False, seed=31415)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final ActiveSet: obj={best_score:.10f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
        print(f"  evaluate_sequence gave better: {best_score:.10f}")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
