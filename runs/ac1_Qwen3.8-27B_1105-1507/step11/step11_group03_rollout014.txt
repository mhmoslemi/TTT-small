def subgradient_polish(a, time_budget, seed=42, verbose=False):

        return a

    best_obj = compute_obj(a)
    start = time.monotonic()

    step_scale = 1.0 / np.sqrt(n)

    for it in range(20000):

        if remaining < 2:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        max_lags = np.where(conv >= max_val * (1.0 - 1e-6))[0]

        # Subgradient of max_l conv(a,a)_l
        subgrad = np.zeros(n)
        for lag in max_lags:
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    subgrad[i] += 2.0 * a[j]
        subgrad /= len(max_lags)

        # Projected subgradient step (project onto simplex a>=0, sum=1)
        gn = np.linalg.norm(subgrad)
        if gn < 1e-15:
            break
        subgrad = subgrad / gn

        # Try multiple step sizes (backtracking)
        improved = False
        for step_mult in [0.5, 1.0, 2.0, 4.0, 8.0]:
            step = step_scale * step_mult / (it + 10) ** 0.4
            a_try = a - step * subgrad
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True

def resample_seq(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)

    # === Phase 1: Quick multi-init screening ===
    print(f"\n=== Phase 1: Initialization Screening (n={n_current}) ===")
    for mode in [1, 2, 3, 4, 5, 7]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_init = make_init(n_current, mode, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            # Quick LP warm-start to see potential
            alloc = min(remaining * 0.03, 15.0)
            if alloc > 8:
                a_test = lp_fixed_point_enhanced(a_init, time_budget=alloc, seed=mode, verbose=False)
                obj_test = compute_obj(a_test)
                if obj_test < best_score:
                    best_score = obj_test
                    best_seq = a_test.copy()
                    print(f"  Init mode {mode}: obj={best_score:.10f} **")
                else:
                    print(f"  Init mode {mode}: obj={obj_test:.10f} (screened)")
            else:
                best_score = obj_init
                best_seq = a_init.copy()
                print(f"  Init mode {mode}: obj={best_score:.10f} **")
        else:
            print(f"  Init mode {mode}: obj={obj_init:.10f}")

    # === Phase 2: Main LP fixed-point (primary engine) ===
    print(f"\n=== Phase 2: Main LP Fixed-Point ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        lp_budget = min(remaining * 0.35, 500.0)
        print(f"  Budget: {lp_budget:.0f}s")
        a2 = lp_fixed_point_enhanced(best_seq, time_budget=lp_budget, seed=42, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 LP-FP: obj={best_score:.10f} **")
        else:
            print(f"  Phase2 LP-FP: obj={obj2:.10f}")

    # === Phase 3: Subgradient polish ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.12, 120.0)
        a3 = subgradient_polish(best_seq, time_budget=alloc, seed=77, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 Subgrad: obj={best_score:.10f} **")
        else:
            print(f"  Phase3 Subgrad: obj={obj3:.10f}")

    # === Phase 4: Second LP round (different seed) ===
    print(f"\n=== Phase 4: Second LP Round ===")
    remaining = deadline - time.monotonic()
    if remaining > 100 and len(best_seq) <= 1500:
        lp_budget2 = min(remaining * 0.20, 250.0)
        # Perturb before second round to explore different basin
        a_pert = best_seq + 0.01 * rng.standard_normal(len(best_seq))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        # Also try the unperturbed version
        a4a = lp_fixed_point_enhanced(a_pert, time_budget=lp_budget2 * 0.5, seed=555, verbose=False)
        obj4a = compute_obj(a4a)
        a4b = lp_fixed_point_enhanced(best_seq, time_budget=lp_budget2 * 0.5, seed=999, verbose=True)
        obj4b = compute_obj(a4b)
        best_round4 = min(obj4a, obj4b)
        best_seq4 = a4a if obj4a < obj4b else a4b
        if best_round4 < best_score:
            best_score = best_round4
            best_seq = best_seq4.copy()
            print(f"  Phase4 LP2: obj={best_score:.10f} **")
        else:
            print(f"  Phase4 LP2: best={best_round4:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_options = sorted(set([
            n_current - 100, n_current - 50, n_current + 50, n_current + 100,
            n_current + 200, n_current - 200,
            700, 850, 900, 950, 1000, 1100
        ]))
        n_options = [n for n in n_options if 500 <= n <= 1500 and n != n_current]
        n_options.sort(key=lambda x: abs(x - n_current))
        n_options = n_options[:6]
        print(f"  Candidates: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            a_try = resample_seq(best_seq, n_try)
            # Quick LP
            alloc = min(remaining * 0.12, 40.0)
            if alloc > 15 and n_try <= 1200:
                a_try = lp_fixed_point_enhanced(a_try, time_budget=alloc,
                                                 seed=n_try, verbose=False)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP micro-polish
    if remaining > 40 and len(best_seq) <= 1500:
        alloc = min(remaining * 0.30, 150.0)
        a_f1 = lp_fixed_point_enhanced(best_seq, time_budget=alloc,
                                        seed=31415, verbose=True)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # Subgradient final
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 100.0)
        a_f2 = subgradient_polish(best_seq, time_budget=alloc, seed=27182)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final Subgrad: obj={best_score:.10f} **")

    # Ultra-fine LP
    remaining = deadline - time.monotonic()
    if remaining > 25 and len(best_seq) <= 1200:
        alloc = min(remaining * 0.4, 100.0)
        a_f3 = lp_fixed_point_enhanced(best_seq, time_budget=alloc,
                                        seed=98765, verbose=False)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Ultra LP: obj={best_score:.10f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
