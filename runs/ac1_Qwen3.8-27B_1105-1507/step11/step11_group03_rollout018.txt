def coordinate_descent(a, time_budget, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    max_sweeps = 50

    for sweep in range(max_sweeps):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget - 1:
                break
            # Try changing coordinate i
            a_backup = a.copy()
            old_conv = compute_conv(a)
            old_sum = float(np.sum(a))
            
            # Try several values
            candidates = [
                a[i] * 0.5, a[i] * 0.75, a[i] * 1.25, a[i] * 1.5,
                a[i] + 0.01, a[i] - 0.01, a[i] * 0.1, a[i] * 2.0,
                max(0, a[i] - 0.005), a[i] + 0.005
            ]
            # Also try based on the gradient info
            # The subgradient w.r.t. a_i at max lags
            max_lags = np.where(old_conv >= np.max(old_conv) * 0.999)[0]
            for lag in max_lags[:2]:
                j = lag - i
                if 0 <= j < n and j != i:
                    candidates.append(max(0, a[i] - 0.01 * a[j]))
                    candidates.append(a[i] + 0.001 * a[j])
            
            best_local = best_obj
            best_val = a[i]
            for val in candidates:
                val = max(0.0, val)
                if abs(val - a[i]) < 1e-12:
                    continue
                a[i] = val
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = best_val
                    continue
                a_norm = a / s_new
                conv_new = compute_conv(a_norm)
                obj_new = 2.0 * n * float(np.max(conv_new))
                if obj_new < best_local:
                    best_local = obj_new
                    best_val = val
                    a[i] = val
                    break
            a[i] = best_val
            
            if best_local < best_obj - 1e-12:
                best_obj = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            break
        if verbose and sweep % 5 == 0:
            print(f"      CD sweep {sweep}: obj={best_obj:.10f}")

    return best_a


def subgradient_step(a, step_size, rng):
    """One subgradient step on the non-smooth max objective."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Find max lags
    threshold = max_val * (1 - 0.001)
    max_lags = np.where(conv >= threshold)[0]
    
    # Subgradient: for each max lag k, grad_k has (grad_k)_i = 2*a_{k-i} (if valid)
    # Average over max lags
    subgrad = np.zeros(n)
    for k in max_lags:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                subgrad[i] += 2.0 * a[j]
    subgrad /= len(max_lags)
    
    # Step in negative subgradient direction
    a_new = a - step_size * subgrad
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP fixed-point iteration with exact line search, coordinate descent
polishing, subgradient methods, and systematic multi-n exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build the convolution matrix A where (A@g)_k = conv(a,g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M for all k, g >= 0, sum(g) = 1."""
    n = len(a)
    A = build_conv_matrix(a)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))

            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t

    return best_t, best_obj, a + best_t * (direction - a) if best_t > 0 else None


def coordinate_descent(a, time_budget, verbose=False, max_sweeps=100):
    """Coordinate descent: optimize one coordinate at a time with smart candidates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        sweep_improved = False
        # Use random order for diversity
        order = rng.permutation(n)

        for idx in order:
            i = int(idx)
            if time.monotonic() - start > time_budget - 1:
                break

            old_val = a[i]
            if old_val < 1e-8:
                # Try small positive values
                candidates = [0.001, 0.005, 0.01 / n, 0.05 / n]
            else:
                candidates = [
                    old_val * 0.5, old_val * 0.7, old_val * 1.3,
                    old_val * 1.5, old_val * 0.8, old_val * 1.2,
                    max(0, old_val - 0.01 / n), old_val + 0.01 / n
                ]

            best_local_obj = best_obj
            best_local_val = old_val

            for val in candidates:
                val = max(0.0, float(val))
                if abs(val - old_val) < 1e-10:
                    continue
                a[i] = val
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = best_local_val
                    continue
                a_norm = a / s_new
                conv_new = compute_conv(a_norm)
                obj_new = 2.0 * n * float(np.max(conv_new))
                a[i] = best_local_val
                if obj_new < best_local_obj:
                    best_local_obj = obj_new
                    best_local_val = val

            if best_local_obj < best_obj - 1e-13:
                a[i] = best_local_val
                best_obj = best_local_obj
                best_a = a.copy()
                sweep_improved = True

        if not sweep_improved:
            break
        if verbose and sweep % 10 == 0:
            print(f"      CD sweep {sweep:3d}: obj={best_obj:.10f}")

    return best_a


def lp_fixed_point(a, time_budget, verbose=False, seed=42, max_iters=500):
    """LP fixed-point with exact line search and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall_count = 0
    lp_fail_count = 0

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            lp_fail_count += 1
            if lp_fail_count > 8:
                break
            a = best_a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue
        lp_fail_count = 0

        t_opt, obj_opt, a_new = exact_line_search(a, g_lp, best_obj)

        if a_new is not None and obj_opt < best_obj - 1e-14:
            a_new = np.maximum(a_new, 0)
            sn = float(np.sum(a_new))
            a_new = a_new / sn
            best_obj = obj_opt
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"    LP it{it:3d}: obj={best_obj:.10f}, t={t_opt:.4f}")
        else:
            stall_count += 1
            # Adaptive perturbation
            strategy = stall_count % 6
            if strategy == 1:
                a = best_a + 0.015 * rng.standard_normal(n)
            elif strategy == 2:
                x = np.arange(n)
                f = rng.integers(1, max(2, n // 4))
                p = rng.uniform(0, 2 * np.pi)
                a = best_a + 0.02 * np.sin(2 * np.pi * f * x / n + p)
            elif strategy == 3:
                # Target the max lags: reduce contribution
                conv = compute_conv(a)
                max_val = np.max(conv)
                max_lags = np.where(conv >= max_val * 0.999)[0]
                pert = np.zeros(n)
                for lag in max_lags[:5]:
                    for i in range(n):
                        j = lag - i
                        if 0 <= j < n:
                            pert[i] -= 0.005 * a[j]
                a = best_a + pert
            elif strategy == 4:
                # Dirichlet mixing
                a = 0.85 * best_a + 0.15 * rng.dirichlet(np.full(n, 2.0))
            elif strategy == 5:
                # Emphasize corners
                corners = np.zeros(n)
                corners[:n//4] = 0.01
                corners[3*n//4:] = 0.01
                a = best_a + corners
            else:
                a = best_a + 0.01 * rng.standard_normal(n)

            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()

            if stall_count > 50:
                if verbose:
                    print(f"    LP stalling at it{it}, breaking")
                break

    return best_a


def spectral_init(n, mode=0):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(1, n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / np.maximum(a, 0.01).sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.arange(n) / n
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        rng = np.random.default_rng(99)
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

    """

    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_current = len(best_seq)

    # === Phase 1: Quick structured init screening ===
    remaining = deadline - time.monotonic()

        print(f"\n=== Phase 1: Init Screening ===")
        for mode in [1, 2, 4, 5]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_init = spectral_init(n_current, mode)
            # Quick LP polish (short budget)

            if alloc > 5:
                a_scr = lp_fixed_point(a_init, time_budget=alloc, verbose=False, seed=mode)

    remaining = deadline - time.monotonic()

        a3 = coordinate_descent(best_seq, time_budget=alloc, verbose=True)

    # === Phase 4: Second LP round ===
    remaining = deadline - time.monotonic()
    if remaining > 150:
