def project_simplex(v):
    """
    Projects a vector onto the simplex {x >= 0, sum(x) = 1}
    """
    n = len(v)
    u = np.sort(v)[::-1]  # descending order
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n+1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]  # largest valid index
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def to_bit_lengths(seq):
    # Convert sequences to bit-length representation for Fourier analysis
    # But simplified to avoid unnecessary computations
    n = len(seq)
    if n == 0:
        return np.array([])
    arr = np.zeros(2 * n - 1)
    for i in range(n):
        for j in range(n):
            arr[i+j] += seq[i] * seq[j]
    # Equivalent to np.convolve(seq, seq)
    return arr
