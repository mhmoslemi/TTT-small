```python
    """
    Solve min_g max_m (a*g)_m s.t. g >= 0, sum(g) = 1
    using mirror descent (exponentiated gradient) with log-sum-exp smoothing.
    """
    n = len(a)
    g = np.ones(n) / n
    
    lr = 1.0 / np.sqrt(n)
    
    for it in range(max_iters):
        c = compute_conv_cross(a, g)  # length 2n-1
        max_c = np.max(c)

        shifted = c / tau - max_c / tau
        p = np.exp(shifted)
        p /= np.sum(p)

        # This is the cross-correlation of p with a (reversed)

            m_end = min(2*n-1, k + n - 1)

        log_weights -= np.max(log_weights)

        g_new /= np.sum(g_new)
        
        g = g_new
    
    return g
```

```python
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. Fast mirror descent (exponentiated gradient) for the LP subproblem
2. Sparse LP via cvxpy for precision
3. Multi-n exploration
4. Spectral perturbation for escaping local minima

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def mirror_descent_lp(a, tau_levels=None, max_iters_per_tau=2000, time_budget=None):
    """
    Solve min_g max_m (a*g)_m s.t. g >= 0, sum(g) = 1
    using mirror descent with log-sum-exp smoothing (staircase tau schedule).
    Returns the optimal g.
    """
    n = len(a)
    if tau_levels is None:
        tau_levels = [n*0.5, n*0.2, n*0.1, n*0.05, n*0.02, n*0.01, n*0.005, n*0.002, n*0.001]
    
    # Precompute FFT of a
    L = 1
    while L < 3 * n:
        L <<= 1
    A_fft = fft(a, L)
    
    g = np.ones(n) / n
    start = time.monotonic()
    
    for tau in tau_levels:
        lr = 1.0 / (tau * np.log(2*n))
        for it in range(max_iters_per_tau):
            if time_budget is not None and time.monotonic() - start > time_budget:
                return g
            
            # Compute c = conv(a, g)
            c = compute_conv_cross(a, g)  # length 2n-1
            max_c = np.max(c)
            
            # Compute p(m) = exp(c[m]/tau) / Z
            shifted = (c - max_c) / tau
            p = np.exp(shifted)
            p_sum = np.sum(p)
            if p_sum < 1e-300:
                p = np.zeros(2*n-1)
                p[np.argmax(c)] = 1.0
            else:
                p /= p_sum
            
            # Compute grad[k] = sum_m p[m] * a[m-k] via FFT
            # grad = cross-correlation of p and a
            P_fft = fft(p, L)
            corr = ifft(conj(P_fft) * A_fft).real
            grad = corr[n-1:2*n-1]  # lags 0 to n-1
            
            # Mirror descent update
            log_g = np.log(g + 1e-300)
            log_g_new = log_g - lr * grad
            log_g_new -= np.max(log_g_new)
            g_new = np.exp(log_g_new)
            g_sum = np.sum(g_new)
            if g_sum < 1e-300:
                break
            g_new /= g_sum
            
            # Check convergence
            if np.max(np.abs(g_new - g)) < 1e-12:
                break
            g = g_new
    
    return g


def sparse_lp_solve(a):

                method = 'MD(fallback)'
            else:
                break
        if g_sol is None:
            break

        best_local = best_obj
        best_local_a = best_a.copy()
```

The LP is:
min M
s.t. A @ g - M <= 0 (2n-1 constraints)
     sum(g) = 1
     g >= 0

In standard form for `linprog`:
min c @ x
s.t. A_ineq @ x <= b_ineq
     A_eq @ x = b_eq
     x >= 0 (bounds)

Variables: x = [g_0, ..., g_{n-1}, M] (n+1 variables)
Objective: c = [0, ..., 0, 1] (minimize M)
Inequality: [A, -1] @ x <= 0 (2n-1 constraints)
Equality: [1, ..., 1, 0] @ x = 1
Bounds: g_i >= 0 (lower bound 0), M free (lower bound -inf)

```python
from scipy.optimize import linprog

def lp_solve_scipy(a):
    n = len(a)
    
    # Build A matrix (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n
    A_toeplitz = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(j, min(j+n, 2*n-1)):
            A_toeplitz[k, j] = a[k-j]
    
    # Full constraint matrix: (2n-1) x (n+1)
    # [A_toeplitz, -1] @ [g, M] <= 0
    A_ineq = np.zeros((2*n-1, n+1))
    A_ineq[:, :n] = A_toeplitz
    A_ineq[:, n] = -1.0
    b_ineq = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Bounds: g_i >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        M = result.x[n]
        g = np.maximum(g, 0)
        sg = g.sum()
        if sg > 1e-10:
            return g / sg, M
    return None, None
```

```python
def build_toeplitz(a):
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx  # shape (2n-1, n)
    valid = (diff >= 0) & (diff < n)
    A = np.where(valid, a[diff.clip(0, n-1)], 0)
    return A
```
