                break

            a_j = a[j]
            s = float(np.sum(a))
            d = s - a_j
            if d < 1e-14:
                continue

            # Determine search range
            x_max = max(3.0 * a_j + 1.0 / (4.0 * n), 1.0 / (4.0 * n))
            x_max = min(x_max, 2.0 / n)

            # Golden section search
            x_lo = 0.0
            x_hi = x_max
            x1 = x_lo + (1.0 - g_ratio) * (x_hi - x_lo)
            x2 = x_lo + g_ratio * (x_hi - x_lo)
            f1 = h_of_x(x1, c, j, n, a_j, d, a)
            f2 = h_of_x(x2, c, j, n, a_j, d, a)

            for _ in range(35):
                if f1 < f2:
                    x_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = x_lo + (1.0 - g_ratio) * (x_hi - x_lo)
                    f1 = h_of_x(x1, c, j, n, a_j, d, a)
                else:
                    x_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = x_lo + g_ratio) * (x_hi - x_lo)
                    f2 = h_of_x(x2, c, j, n, a_j, d, a)

            x_opt = (x_lo + x_hi) / 2.0
            f_opt = h_of_x(x_opt, c, j, n, a_j, d, a)

            # Check boundary x=0
            f_0 = float(np.max(c)) / (d) ** 2
            if f_0 < f_opt:
                x_opt = 0.0
                f_opt = f_0

            f_current = float(np.max(c)) / (s) ** 2
            if f_opt < f_current - 1e-14:
                delta = x_opt - a_j
                c[j:j + n] += 2.0 * delta * a[0:n]
                if 2 * j <= 2 * n - 2:
                    c[2 * j] += delta * delta
                a[j] = x_opt
                sweep_improved = True

                new_obj = compute_obj_from_conv(a, c)
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()

        if not sweep_improved:
            stall += 1
            if stall >= 3:
                # Perturb to escape
                a_pert = a * np.exp(0.01 * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                    c = compute_conv_fft(a)
                    new_obj = compute_obj_from_conv(a, c)
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = a.copy()
                stall = 0
        else:
            stall = 0

        sweep += 1
        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={time.monotonic()-start:.1f}s")

    return best_a


def mirror_descent_optimize(a, time_budget=200.0, tau_init=50.0, tau_end=3000.0,
                            verbose=False, seed=777):
    """Mirror descent (multiplicative weights) with LSE-smoothed gradient."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 0.5:
            break

        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 0.8

        c = compute_conv_fft(a)
        max_c = float(np.max(c))
        shifted = tau * (c - max_c)
        if shifted.max() > 500:
            shifted = shifted - shifted.max()
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient: 2 * (a_rev * p)[n-1:2n-1]
        b = a[::-1]
        N_fft = 1
        while N_fft < 3 * n - 1:
            N_fft <<= 1
        conv_grad = np.fft.irfft(np.fft.rfft(b, N_fft) * np.fft.rfft(p, N_fft), N_fft)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]

        g_max = float(np.max(np.abs(grad)))
        if g_max < 1e-15:
            break
        eta = 2.0 / (g_max + 1e-10)
        eta *= (1.0 + 0.3 * (1.0 - progress))

        exponent = np.clip(-eta * grad, -50, 50)
        a_new = a * np.exp(exponent)
        s_new = float(np.sum(a_new))
        if s_new < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Occasional perturbation
        if it % 50 == 30 and progress < 0.8:
            temp = 0.005 * (1.0 - progress)
            a_new = a_new + temp * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 30:
                a_pert = a * np.exp(0.02 * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                stall = 0

        a = a_new
        it += 1
        if verbose and it % 100 == 0:
            print(f"    MD it{it}: obj={best_obj:.10f}, tau={tau:.0f}, time={time.monotonic()-start:.1f}s")

    return best_a


def spectral_init(n, iterations=50, seed=42):
    """Gerchberg-Saxton style spectral initialization for flat autocorrelation."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n)
    a = a / np.sum(a)

    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    for it in range(iterations):
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        angle = np.angle(A)
        # Flatten the magnitude
        mean_mag = float(np.mean(mag))
        target_mag = mean_mag  # flat spectrum
        # Blend: partially flatten
        alpha = 0.3 + 0.7 * (it / iterations)
        new_mag = (1 - alpha) * mag + alpha * target_mag
        A_new = new_mag * np.exp(1j * angle)
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a = a_new / s
        else:
            a = np.ones(n) / n

    return a


def lp_direction(a):
    """LP-based direction finding for the linearized problem."""
    n = len(a)
    a_norm = a / np.sum(a)
    c_aa = compute_conv_fft(a_norm)

    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1) + 1
        for j in range(lo, hi):
            A[k, j] = a_norm[k - j]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2.0 * (A @ g) - M <= c_aa, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
