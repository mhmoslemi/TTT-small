Optimizes step function coefficients using spectral shaping, peak-lag-aware
coordinate descent, and subgradient methods to minimize the normalized
maximum autocorrelation.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def spectral_shaping(a, time_budget=100.0, alpha_init=0.5, verbose=False):

    Iteratively adjusts the frequency-domain representation to
    make |FFT(a)|^2 more uniform.

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

def subgradient_method(a, time_budget=100.0, verbose=False):
    """
    Projected subgradient method on max_k conv(a,a).
    Step size follows Polyak rule.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(a @ np.cumsum(np.zeros(1))))  # placeholder
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    # Lower bound: 2n * (1/(2n-1)) = 2n/(2n-1)
    lb = 2.0 * n / (2.0 * n - 1.0)
    
    step_scale = 1.0 / np.sqrt(n)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        cur_val = float(np.max(conv))
        cur_obj = 2.0 * n * cur_val
        
        # Subgradient of max_k conv[k] at current point
        # = gradient of conv[k_peak]
        g = np.zeros(n)
        for i in range(n):
            val = 0.0
            if i >= k_peak:
                val += a[i - k_peak]
            if i + k_peak < n:
                val += a[i + k_peak]
            g[i] = val
        
        # Also include the effect on other lags that are close to the peak
        # (to make the subgradient more informative)
        threshold = 0.95 * cur_val
        near_peak_lags = np.where(conv >= threshold)[0]
        for k in near_peak_lags:
            if k == k_peak:
                continue
            for i in range(n):
                val = 0.0
                if i >= k:
                    val += a[i - k]
                if i + k < n:
                    val += a[i + k]
                g[i] += 0.1 * val  # small weight for near-peak lags
        
        # Projected subgradient step
        # We want to maintain sum(a) = 1 and a >= 0
        # Simple approach: a_new = a - step * g, then project
        # But this might make a negative. Better: multiplicative update
        # a_new[i] = a[i] * exp(-step * g[i])
        
        g_mean = float(np.mean(g))
        g_centered = g - g_mean  # center to preserve sum approximately
        
        lr = step_scale * (best_obj - cur_obj) / (max(np.dot(g, a - best_a), 1e-10) + 1e-10)
        lr = min(max(lr, 1e-4), 0.5)
        
        a_new = a * np.exp(-lr * g_centered)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        if verbose and it % 200 == 0:
            print(f"    Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}")
    
    return best_a

"""
Optimizes step function coefficients using a multi-phase strategy combining:
1. Spectral power-spectrum flattening for coarse global shaping
2. Peak-lag-aware coordinate descent with exact 1D minimization
3. Smoothed gradient (log-sum-exp) coordinate descent for refined local search
4. Multi-length exploration and diverse restarts
Key innovation: spectral shaping to reduce autocorrelation peak globally,
then fine-grained CD focused on the peak-contributing coordinates.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def spectral_shaping(a, time_budget=100.0, alpha_init=0.4, verbose=False):
    """
    Flatten the power spectrum to reduce autocorrelation peak.
    Uses iterated power compression in the frequency domain.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    it = 0
    while time.monotonic() - start < time_budget:
        A = fft(a, L)
        P = np.abs(A) ** 2
        P_mean = np.mean(P)
        if P_mean < 1e-20:
            break

        progress = min(1.0, it / 2000.0)
        eta = alpha_init * (1.0 - 0.7 * progress)

        P_clipped = np.maximum(P, P_mean * 0.001)
        A_new = A * (P_mean / P_clipped) ** eta

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            it += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # Blend to avoid divergence
        a = 0.7 * a_new + 0.3 * a
        a = np.maximum(a, 0)
        a = a / float(np.sum(a))

        it += 1

        if verbose and it % 500 == 0:
            print(f"    Spec it{it}: obj={best_obj:.8f}, eta={eta:.4f}")

    return best_a


def compute_smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    if max_conv < 1e-30:
        return np.zeros(n), conv
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42, peak_aware=True):
    """
    Exact coordinate descent with optional peak-lag awareness.
    Uses golden section search for exact 1D minimization.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1
        conv = compute_conv(a)
        S = float(np.sum(a))

        if peak_aware:
            k_peak = int(np.argmax(conv))
            # Priority: coordinates contributing most to peak
            contrib = np.zeros(n)
            for i in range(n):
                c = 0.0
                if i >= k_peak:
                    c += a[i - k_peak]
                if i + k_peak < n:

        else:
            order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:

            k_max = min(2 * n - 2, i + n - 1)

            c_k = conv[ks].copy()

            c_k_lin = c_k[mask]

            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0

                max_val = max(max_lin, quad_val)

                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Candidate evaluation
            scale = max(v, 1.0 / n, 1e-6)

                          max(0.0, v * 1.3), max(0.0, v * 1.5), 1.0 / n, 2.0 / n]

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.5
            if x_hi - x_lo < 1e-12:

                    fd = fc

                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:

            a[i] = x_opt

            S += delta

            if S > 1e-10:

                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    SmoothedCD: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")

    return best_a


def multiplicative_descent(a, time_budget=60.0, verbose=False):
    """Multiplicative weight/update method with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 30.0 + 400.0 * progress
        lr = (6.0 / np.sqrt(n)) * (1.0 - 0.6 * progress)

        try:
            grad, _ = compute_smoothed_gradient(a, tau)
        except Exception:
            break

        g_centered = grad - float(np.mean(grad))
        update = np.exp(-lr * g_centered)
        update = np.clip(update, 0.05, 20.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** rng.uniform(0.5, 3.0)
        a = a / a.sum()
    elif mode == 4:
        # Two-bump structure
        c1, c2 = n // 4, 3 * n // 4
        w = n // 6
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Log-periodic
        x = np.linspace(0.01, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * np.log10(x) * rng.uniform(2, 20))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines spectral shaping, peak-aware CD, smoothed CD,
    multiplicative descent, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Spectral Shaping ===
    print(f"\n=== Phase 1: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.12, 120.0)
        a_spec = spectral_shaping(best_seq, time_budget=alloc, alpha_init=0.35, verbose=True)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Spectral: obj={obj_spec:.8f}")

    # === Phase 2: Peak-Aware Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 2: Peak-Aware CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.30, 350.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True, seed=seed, peak_aware=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  PeakCD: obj={best_score:.8f} **")
        else:
            print(f"  PeakCD: obj={obj_cd:.8f}")

    # === Phase 3: Smoothed Gradient CD ===
    print(f"\n=== Phase 3: Smoothed CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.12, 100.0)
        a_smooth = smoothed_cd(best_seq, time_budget=alloc, tau_init=50.0, tau_end=600.0, verbose=True)
        obj_smooth = compute_obj(a_smooth)
        if obj_smooth < best_score:
            best_score = obj_smooth
            best_seq = a_smooth.copy()
            print(f"  SmoothedCD: obj={best_score:.8f} **")
        else:
            print(f"  SmoothedCD: obj={obj_smooth:.8f}")

    # === Phase 4: Follow-up Peak CD ===
    print(f"\n=== Phase 4: Follow-up CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 150.0)
        a_cd2 = coordinate_descent(best_seq, time_budget=alloc, verbose=False, seed=seed + 777, peak_aware=True)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 5: Multiplicative Descent ===
    print(f"\n=== Phase 5: Multiplicative Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.08, 60.0)
        a_md = multiplicative_descent(best_seq, time_budget=alloc, verbose=False)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), n_best - 50, n_best + 50,
        n_best + 100, n_best + 150, n_best + 200,
        700, 850, 950, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500]
    # Remove those too close to current
    n_options = [n for n in n_options if abs(n - n_best) > 40]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc_cd = min(remaining * 0.15, 45.0)
        if alloc_cd > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd, verbose=False,
                                       seed=n_try, peak_aware=True)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_r = diverse_init(n_current, rng, mode=1 + r)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                     seed=9000 + r * 333, peak_aware=True)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc, verbose=False,
                                     seed=seed + 19999, peak_aware=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # One more smoothed CD pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_sf = smoothed_cd(best_seq, time_budget=alloc, tau_init=100.0, tau_end=800.0, verbose=False)
        obj_sf = compute_obj(a_sf)
        if obj_sf < best_score:
            best_score = obj_sf
            best_seq = a_sf.copy()
            print(f"  Final Smoothed: obj={best_score:.8f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cf = coordinate_descent(best_seq, time_budget=min(remaining * 0.5, 40.0),
                                  verbose=False, seed=seed + 25000, peak_aware=True)
        obj_cf = compute_obj(a_cf)
        if obj_cf < best_score:
            best_score = obj_cf
            best_seq = a_cf.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
