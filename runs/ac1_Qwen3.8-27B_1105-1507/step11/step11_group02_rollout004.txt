def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2. Assumes sum(a)=1."""

        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def compute_smooth_grad(a, tau):

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c)**2) / (2*w**2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 4:
        # Sum of sinusoids
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in range(1, 6):
            a += 0.3 * np.sin(2*np.pi*f*x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        a = 1.0 - np.abs(2*np.linspace(0, 1, n) - 1)
        a += 0.3 * rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 7:
        # Concentrated in middle
        x = np.arange(n)
        c = n / 2
        a = np.exp(-((x - c)**2) / (2*(n/8)**2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def compute_smooth_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_circ = ifft(P * np.conj(A)).real[:n]
    return 2.0 * grad_circ, max_c, p

def golden_section_1d(f_func, x_lo, x_hi, max_iter=45):
    """Golden section search minimizing f on [x_lo, x_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if x_hi - x_lo < 1e-14:
        return (x_lo + x_hi) / 2.0

    fc = f_func(c_pt)
    fd = f_func(d_pt)
    for _ in range(max_iter):

            fc = fd

    # Build candidate points
    scale = max(v, 1.0 / n, 1e-6)
    candidates = [0.0, v * 0.3, v * 0.5, v * 0.7, v * 1.3, v * 1.5, v * 2.0, 1.0 / n, 2.0 / n]
    candidates = sorted(set(max(0.0, float(c)) for c in candidates))

    best_x = v
    best_f = f_1d_obj(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    for x in candidates:
        fx = f_1d_obj(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
        if fx < best_f:
            best_f = fx
            best_x = x

    # Golden section with wider bracket
    x_lo = max(0.0, best_x - scale * 1.5)
    x_hi = best_x + scale * 3.0
    if x_hi - x_lo < 1e-12:
        x_hi = x_lo + 1e-8

    def make_f(x):
        return f_1d_obj(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

    x_opt = golden_section_1d(make_f, x_lo, x_hi, max_iter=42)

    if abs(x_opt - v) < 1e-14:
        return a, conv, S, False

    delta = x_opt - v
    a_new = a.copy()
    a_new[i] = x_opt
    conv_new = conv.copy()
    if 0 <= k_quad <= max_idx:
        conv_new[k_quad] += x_opt * x_opt - v * v
    if len(ks_lin) > 0:
        conv_new[ks_lin] += 2.0 * delta * a_ki_lin
    S_new = S + delta
    return a_new, conv_new, S_new, True


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Active-set-aware coordinate descent with greedy ordering.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Identify active set
        max_conv = float(np.max(conv))
        if max_conv < 1e-30:
            break
        threshold = max_conv * 0.99
        active_lags = np.where(conv >= threshold)[0]

        # Compute priority
        priority = np.zeros(n)
        for k in active_lags:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            priority[i_min:i_max + 1] += 1.0

        if sweep % 4 == 0:
            order = np.argsort(-priority + rng.normal(0, 0.2, n))
        else:
            order = np.concatenate([
                np.argsort(-priority[:n // 2])[:n // 2],
                np.arange(n // 2, n)[rng.permutation(n // 2)]
            ])

        sweep_improved = False
        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            a, conv, S, improved = coord_desc_step(a, i, conv, n)
            if improved:
                sweep_improved = True
                if S > 1e-10:
                    obj = 2.0 * n * float(np.max(conv)) / (S * S)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

        if not sweep_improved:
            no_improve_count += 1
        else:
            no_improve_count = 0

        # Noise injection on stagnation
        if no_improve_count >= 3:
            scale_noise = 0.03 * min(no_improve_count, 5)
            pert = rng.normal(0, scale_noise, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                a = a / S_check
                conv = compute_conv(a)
                obj_check = 2.0 * n * float(np.max(conv))
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()
                no_improve_count = 0

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s, active={len(active_lags)}")

    return best_a
