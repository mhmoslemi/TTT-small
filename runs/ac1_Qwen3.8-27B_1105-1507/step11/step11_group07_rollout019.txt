   2. Compute gradient: g[i] = a[k*-i] + a[i-k*] (with valid indices)
   3. Project: v = g - mean(g) (to stay on the simplex tangent)
   4. Update: a_new = a - lr * v, then project to simplex
   5. Line search for optimal lr
   
   This is a pure subgradient method, which converges slowly but is very simple. We can combine it with:
   - Momentum (use average of recent gradients)
   - When the max is achieved at multiple positions, use the average gradient
   - Soft-max smoothing (as before) for better behavior
   
   Actually, I think the key improvement will come from:
   1. Better handling of the flat region (variance minimization first)
   2. More aggressive LP polishing with different LP formulations
   3. Better length choice

   QP: minimize ||A @ g - c||_2^2 subject to sum(g) = 1, g >= 0
   
   where c is the target convolution (e.g., the current conv(a,a) with its peak reduced). This would try to make conv(a, g) as close to a flat target as possible.
   
   Or: minimize (max(A@g) - min(A@g))^2 ... this is not a QP.
   
   Or: minimize ||A@g - t*1||_2^2 + penalty * max(A@g)
   
   I think the most promising new ideas are:
   1. **L-BFGS-B on the variance objective** (new)
   2. **Subgradient method on the max with momentum** (simpler than soft-max GD)
   3. **Better LP: minimize the range (max - min) of A@g instead of just max**
   4. **Adaptive length: try to find the optimal n**

"""
Hybrid spectral optimization for step function peak minimization.
Uses variance minimization via L-BFGS-B as initialization, followed by
soft-max gradient descent, improved LP polishing (range minimization),
and length exploration.
"""
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_var_conv(a):
    """Compute variance of conv(a,a) / (sum(a))^4 (normalized)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return 0.0
    a_norm = a / s
    conv = compute_conv(a_norm)
    # conv sums to 1 (since sum(a_norm)=1)
    var = float(np.sum(conv**2)) - 1.0 / (2 * n - 1)
    return var


def compute_var_grad(a):
    """
    Compute gradient of ||conv(a,a)||_2^2 / (sum(a))^2.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.zeros(n)
    
    a_norm = a / s
    conv = compute_conv(a_norm)  # length 2n-1
    
    # Need to compute T1[i] + T2[i] for each i
    # T1[i] = sum_j conv[i+j] * a_norm[j] (cross-correlation of conv and a_norm)
    # T2[i] = sum_j conv[i-j] * a_norm[j] (convolution of conv and a_norm)
    
    # Total length needed: len(conv) + len(a_norm) - 1 = 2n-1 + n - 1 = 3n-2
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    C = rfft(conv, size)
    A = rfft(a_norm, size)
    
    # T1 = cross-correlation = IFFT(conj(A) * C)
    T1_full = irfft(np.conj(A) * C, size)
    T1 = T1_full[:n]
    
    # T2 = convolution = IFFT(A * C)
    T2_full = irfft(A * C, size)
    T2 = T2_full[:n]
    
    # Gradient of ||conv(a_norm, a_norm)||_2^2 w.r.t. a_norm:
    # = 2 * (T1 + T2)
    # Gradient of f = ||conv||^2 / s^2 w.r.t. a:
    # Using chain rule: f = ||conv(a/s, a/s)||^2 / s^2... 
    # Actually let me just compute gradient of ||conv(a,a)||_2^2 / (sum a)^2 directly.
    
    # f(a) = ||conv(a,a)||_2^2 / (sum a)^2
    # df/da[i] = 2 * (T1_raw[i] + T2_raw[i]) / s^2 - 2 * ||conv(a,a)||_2^2 / s^3
    # where T1_raw and T2_raw use a (not a_norm)
    
    # Let me recompute with a instead of a_norm:
    # conv_raw = conv(a, a) = s^2 * conv(a_norm, a_norm) = s^2 * conv
    # ||conv_raw||_2^2 = s^4 * ||conv||_2^2
    
    # T1_raw[i] = sum_j conv_raw[i+j] * a[j] = s^2 * sum_j conv[i+j] * (s * a_norm[j]) = s^3 * T1[i]
    # T2_raw[i] = s^3 * T2[i]
    
    # df/da[i] = 2 * s^3 * (T1[i] + T2[i]) / s^4 - 2 * s^4 * ||conv||_2^2 / s^5
    # = 2 * (T1[i] + T2[i]) / s - 2 * s * ||conv||_2^2 / s^2
    # = 2/s * ((T1[i] + T2[i]) - ||conv||_2^2)
    
    conv_sq_sum = float(np.sum(conv ** 2))
    
    grad = (2.0 / s) * ((T1 + T2) - conv_sq_sum)
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lbfgs_var_minimize(a, time_budget=60.0, verbose=False):
    """
    Minimize ||conv(a,a)||_2^2 / (sum(a))^2 using L-BFGS-B.
    This flattens the convolution.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    s = float(np.sum(a))
    a = a / s  # normalize to sum=1
    
    start = time.monotonic()
    
    def objective(x):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            raise TimeBudgetExceeded()
        x = np.maximum(x, 0)
        if float(np.sum(x)) < 1e-10:
            return 1e10
        return compute_var_conv(x)
    
    def gradient(x):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            raise TimeBudgetExceeded()
        return compute_var_grad(x)
    
    try:
        result = minimize(
            objective, a, method='L-BFGS-B', jac=gradient,
            bounds=[(0, None)] * n,
            options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-8, 'disp': False}
        )
        if result.success or result.fun < compute_var_conv(a):
            return np.maximum(result.x, 0)
    except (TimeBudgetExceeded, Exception) as e:
        if verbose:
            print(f"  L-BFGS: {e}")
    
    return a


class TimeBudgetExceeded(Exception):
    pass


def soft_max_gd(a, time_budget=200.0, beta_start=10.0, beta_end=80.0,
                lr_start=0.005, momentum=0.9, verbose=False, seed=42):
    """Enhanced soft-max gradient descent with adaptive learning rate."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.5 * progress ** 0.5)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: 2 * (p cross-correlated with a)
        p_len = len(p)
        size = 1
        while size < n + p_len - 1:
            size <<= 1
        A = rfft(a[::-1], size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        # Momentum + adaptive step
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 50:
                v = np.zeros(n)
                stall = 0
                a = project_simplex(best_a + rng.normal(0, 0.003, n))
        
        a = a_new
        it += 1
        
        if verbose and it % 1000 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"lr={lr:.5f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_minrange(a):
    """
    Solve LP: minimize (M - m) s.t. m <= conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    This tries to make conv(a, g) as flat as possible.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    m = cp.Variable()
    constraints = [A @ g <= M, A @ g >= m, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M - m), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search_blend(a, g_dir, time_budget=3.0):
    """Golden section search for optimal blend factor."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 2.0
    
    def obj_at(t):
        if t < 0:
            return np.inf
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)
    
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    start = time.monotonic()
    for _ in range(25):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
    
    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    
    # Also try some fixed points
    for t_try in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try
    
    return t_best, obj_best


def lp_polish(a, time_budget=120.0, max_iters=50, verbose=False):
    """LP-based polishing with both min-peak and min-range LPs."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 4:
            break
        
        # Alternate between min-peak and min-range LPs
        if it % 2 == 0:
            g_lp = solve_lp_minpeak(a)[0]
        else:
            g_lp = solve_lp_minrange(a)
            if g_lp is None:
                g_lp = solve_lp_minpeak(a)[0]
        
        if g_lp is None:
            break
        
        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.05))
        
        if t_opt > 0.01 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

            else:

        else:

        if verbose and it % 5 == 0:

                  f"elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 5:
            noise_scale = 0.005 * min(stall - 3, 3)
            a = project_simplex(best_a + rng.normal(0, noise_scale, n))
            stall = 0
    
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        # Chirp-like
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) * 5.0 / n)
        a = np.abs(a) + 0.05
    elif mode == 6:
        # Two-hump
        a = np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2) + 0.02
    elif mode == 7:
        # Sharp edges
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(4, n)] = 0.3
        a[max(0, n-5):n-1] = 0.3
        if n > 4:
            a[1:-1] += 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
