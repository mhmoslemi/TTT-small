    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed < 0:
            break
        progress = min(1.0, elapsed / time_budget)
        
        # p schedule: start at p_start, ramp up to p_end
        p_power = p_start + (p_end - p_start) * progress ** 0.8
        
        conv = compute_conv(a)
        grad = lp_gradient(a, p_power, conv)
        
        g_norm = float(np.max(np.abs(grad)))
        if g_norm < 1e-15:
            break
        
        # Adaptive step size
        eta = 1.0 / (g_norm + 1e-10)
        # Dampen for large p (sharper landscape)
        eta *= 1.0 / (1.0 + 0.05 * p_power)
        # Also dampen as we progress
        eta *= (1.0 - 0.3 * progress)
        
        a_new = mirror_step(a, grad, eta)
        
        # Small random perturbation for exploration
        if it % 50 == 25 and progress < 0.7:
            temp = 0.005 * (1.0 - progress)
            a_new = a_new * np.exp(temp * rng.standard_normal(n))
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Perturb to escape
                strength = 0.01 * (1.0 + progress)
                a = best_a * np.exp(strength * rng.standard_normal(n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                stall = 0
        
        a = a_new
        it += 1
        
        if verbose and (it % 100 == 0 or it == 1):
            print(f"    LP-anneal it{it}: p={p_power:.1f}, obj={best_obj:.8f}, "
                  f"eta={eta:.5f}, stall={stall}, time={elapsed:.0f}s")
    
    return best_a

def coordinate_descent(a, time_budget, verbose=True):
    """
    Coordinate descent: for each coordinate, find the best value that minimizes the objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Cycle through coordinates
        for i in range(n):
            # Save current value
            ai = a[i]
            if ai < 1e-15:
                # Try small positive values
                for trial in [1e-3, 0.01, 0.05] / n:
                    a[i] = trial
                    s_new = sum_a - ai + trial
                    if s_new <= 0:
                        continue
                    # Compute new conv (approximate: update only affected lags)
                    # For simplicity, recompute max
                    a_try = a / s_new
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a.copy()
                        sum_a = s_new
                    a[i] = ai
                continue
            
            # Try values around current
            # The objective as a function of x = a[i] is:
            # f(x) = 2n * max_k (base_k + c_k * x) / (sum_a - ai + x)^2
            # where base_k = conv[k] - a[i]*a[k-i] - a[k-i]*a[i] + a[i]^2 (if k=2i)
            # Actually, let's just try a few values
            
            # Compute the "residual" convolution (without a[i])
            # conv[k] = sum_j a[j]*a[k-j]
            # Terms involving a[i]: 2*a[i]*a[k-i] (if k≠2i) or a[i]^2 (if k=2i)
            
            # For trial x: new_conv[k] = conv[k] - (contribution of a[i]) + (contribution of x)
            # contribution of a[i] to conv[k] = 2*a[i]*a[k-i] if 0<=k-i<n and k!=2i
            #                                        = a[i]^2 if k=2i
            # contribution of x to conv[k] = 2*x*a[k-i] if 0<=k-i<n and k!=2i
            #                                = x^2 if k=2i
            
            # Let's compute this efficiently
            # delta[k] = new_conv[k] - conv[k] = (x - ai) * (2*a[k-i] - a[i]*(1 if k==2i else 0)) ... 
            # This is getting complicated. Let me just try a few values.
            
            trials = [ai * 0.5, ai * 0.7, ai * 0.9, ai, ai * 1.1, ai * 1.3, ai * 2.0]
            trials = [max(0.0, t) for t in trials]
            
            best_x = ai
            best_f = best_obj
            
            for x in trials:
                a[i] = x
                s_try = sum_a - ai + x
                if s_try <= 0:
                    a[i] = ai
                    continue
                a_try = a / s_try
                f_try = compute_obj(a_try)
                if f_try < best_f:
                    best_f = f_try
                    best_x = x
            
            a[i] = best_x
            sum_a = sum_a - ai + best_x
            if best_f < best_obj:
                best_obj = best_f
                best_a = a.copy()
        
        it += 1
        if verbose and it % 10 == 0:
            print(f"    CD cycle {it}: obj={best_obj:.8f}, time={time.monotonic()-start:.0f}s")
    
    return best_a

def spectral_nudge(a, alpha, time_budget, verbose=False):
    """
    Blend the spectrum with a flat target spectrum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        # Compute padded FFT for linear convolution
        size = 1
        while size < 2 * n:
            size <<= 1
        fa = rfft(a, size)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # Target: flat magnitude = mean
        mean_mag = float(np.mean(mag))
        if mean_mag < 1e-15:
            break
        
        # Blend: new_mag = |mag|^(1-alpha) * mean_mag^alpha
        new_mag = mag ** (1.0 - alpha) * mean_mag ** alpha
        new_fa = new_mag * np.exp(1j * phase)
        a_new = irfft(new_fa, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a
