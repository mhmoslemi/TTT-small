import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp

def compute_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2*n-1]

def compute_obj(a):
    """2n * max(conv(a,a)) / (sum(a))^2, assuming a >= 0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def lse_grad_fft(a, T):
    """Compute gradient of 2n * LSE_T(conv(a,a)) using FFT.
    a should be normalized (sum = 1).
    Returns gradient of the same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    # For numerical stability
    exp_vals = np.exp(T * (conv - cmax))
    w = exp_vals / exp_vals.sum()
    
    # grad[i] = 4n * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is 4n * cross_correlation(w, a)[i] for i=0,...,n-1
    N = 1
    while N < 3*n:
        N *= 2
    W = fft(w, N)
    A = fft(a, N)
    R = ifft(W * np.conj(A))
    grad = 4.0 * n * np.real(R[:n])
    return grad, conv, w

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = int(rho_indices[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

def lse_gradient_descent_fast(a, time_budget, T_start=5.0, T_end=100.0):
    """Fast LSE gradient descent with FFT-based gradient."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.5  # will be adjusted by backtracking
    it = 0
    stall = 0
    max_stall = 50
    
    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac
        
        grad, conv, w = lse_grad_fft(a, T)
        
        # Skip constraint gradient (project onto simplex after step)
        a_old = a.copy()
        obj_old = 2.0 * n * (np.max(conv) + np.log(np.sum(np.exp(T*(conv - np.max(conv))))) / T)
        
        t = lr
        improved = False
        for _ in range(8):
            a_try = a_old - t * grad
            a_try = project_simplex(a_try)
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * (np.max(conv_try) + np.log(np.sum(np.exp(T*(conv_try - np.max(conv_try))))) / T)
            if obj_try < obj_old - 1e-14:
                a = a_try
                improved = True
                break
            t *= 0.5
        
        if not improved:
            stall += 1
            if stall > max_stall:
                break
        else:
            stall = 0
            lr = t * 1.05
        
        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
        
        it += 1
    
    return best_a

def multi_peak_subgrad(a, top_k, time_budget):
    """Multi-peak subgradient descent with FFT acceleration."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:

    a = a / s

        a_old = a.copy()
        obj_old = compute_obj(a)
        t = lr
        improved = False
        for _ in range(10):
            momentum = beta * momentum + (1-beta) * (-t * grad)
            # Actually let's use standard momentum
            momentum = beta * momentum - t * grad
            a_try = a_old + momentum  # Wait, this is wrong. Let me use a different formulation.
            # Standard momentum: v = beta*v - t*grad; x = x + v
            # But we need to project
            a_try = project_simplex(a_old + momentum)
            obj_try = compute_obj(a_try)
            if obj_try < obj_old - 1e-14:

                improved = True
                break
            t *= 0.5
            momentum = beta * momentum - t * grad
        
        if not improved:
            stall += 1
            momentum *= 0.5
            lr *= 0.8
        else:
            stall = 0
        
        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
        
        if stall > 100:

    return best_a

def lp_solve(a, n_constraints, time_per_solve=5.0):
    """Solve the LP: min M s.t. (a*g)[k] <= M for top-k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_a = compute_conv(a_norm)
    top_indices = np.argsort(conv_a)[::-1][:n_constraints]
    
    # Build constraint matrix
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid k-j
    # Coefficient of g[m] in (a*g)[k] is a[k-m] (if 0 <= k-m < n)
    A = np.zeros((len(top_indices), n))
    for idx, k in enumerate(top_indices):
        k = int(k)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[idx, m] = a_norm[j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def lp_blend(a, g_lp):
    """Find best blend between a and g_lp."""
    best_t = 0.0
    best_obj = compute_obj(a)
    
    for t in np.linspace(0.05, 1.0, 20):
        a_try = (1-t)*a + t*g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    
    return best_t, best_obj

def lp_polish(a, time_budget, n_constraints_start=40, n_constraints_end=15, max_iters=50):
    """Iterative LP polishing."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start >= time_budget:
            break
        if time_budget - (time.monotonic() - start) < 5:
            break
        
        frac = min(1.0, it / max(1, max_iters-1))
        nc = int(round(n_constraints_start + frac * (n_constraints_end - n_constraints_start)))
        nc = max(5, min(nc, 2*n-2))
        
        g_lp = lp_solve(best_a, nc)
        if g_lp is None:
            break
        
        t_opt, obj_opt = lp_blend(best_a, g_lp)
        
        if t_opt > 0 and obj_opt < best_obj - 1e-12:
            a_new = (1-t_opt)*best_a + t_opt*g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_new.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        if stall >= 5:
            rng = np.random.default_rng(it*7919+42)
            noise = rng.normal(0, 0.005*stall, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj(best_a)
            stall = 0
    
    return best_a

def coordinate_descent(a, time_budget, max_sweeps=5):
    """Coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        improved_sweep = False
        
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            
            ai_old = a[i]
            if ai_old < 1e-12:
                continue
            
            # Binary search for optimal value of a[i]
            # The objective as a function of a[i] (with others fixed) is:
            # 2n * max_k (conv_k) / (sum - a[i] + a[i])^2 = 2n * max_k conv_k / sum^2
            # Since sum is fixed, we just need to minimize max_k conv_k
            # conv_k depends on a[i] as: conv_k += a[i]*(a[k-i] + a[i-k]) for each k
            # (with the understanding that out-of-range terms are 0)
            
            # Try a few candidate values
            lo = 0.0
            hi = ai_old * 3 + 0.01/n
            best_val = ai_old
            best_local = best_obj
            
            for _ in range(15):
                mid = (lo + hi) / 2
                a[i] = mid
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai_old
                    continue
                a_try = a / s_new
                obj_try = compute_obj(a_try)
                a[i] = ai_old  # restore
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = mid
                if obj_try < compute_obj(a):  # Hmm this is expensive
                    hi = mid
                else:
                    lo = mid
            
            # Actually binary search on a non-convex function doesn't work well.
            # Let's just try a few values.
            candidates = [0.0, ai_old*0.5, ai_old*0.8, ai_old*1.2, ai_old*1.5, ai_old*2.0]
            best_val = ai_old
            best_local = best_obj
            for cv in candidates:
                a[i] = cv
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai_old
                    continue
                a_try = a / s_new
                obj_try = compute_obj(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv
            
            if best_val != ai_old:
                a[i] = best_val
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    improved_sweep = True
            
        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
        
        if not improved_sweep:
            break
    
    return best_a

def init_arcsine(n):
    """Arcsine distribution on {0,...,n-1}."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2*x - 1)**2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 0 else np.ones(n)/n

def init_power(n, alpha):
    """Power law concentration at edges."""
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.maximum(d, 1/(2*n))**(-alpha)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 0 else np.ones(n)/n

def init_bi_gaussian(n, sigma):

def coord_step_fast(a, i, conv):
    """Find optimal a[i] given current a and conv. Returns new a[i] and updated conv."""
    n = len(a)
    ai_old = a[i]
    s_old = float(np.sum(a))
    
    # delta_conv[k] = da * (a[k-i] + a[i-k]) where da = new_ai - ai_old
    # Build the "influence vector"
    infl = np.zeros(2*n-1)
    for k in range(2*n-1):
        val = 0.0
        if 0 <= k-i < n:
            val += a[k-i]
        if 0 <= i-k < n:
            val += a[i-k]
        infl[k] = val
    
    # Now conv_new[k] = conv[k] + da * infl[k]
    # sum_new = s_old + da
    # obj(da) = 2n * max_k (conv[k] + da*infl[k]) / (s_old + da)^2
    # We want to minimize this over da >= -ai_old (so new_ai >= 0)
    
    # This is a 1D optimization problem. Use golden section or ternary search.
    # The function is not necessarily unimodal, but in practice it should be close.
    # Let's sample at a few points and pick the best.
    
    ai_candidates = [0.0, ai_old*0.25, ai_old*0.5, ai_old*0.75, ai_old, ai_old*1.25, ai_old*1.5, ai_old*2.0, ai_old*3.0]
    
    best_da = 0.0
    best_obj = compute_obj(a)
    
    for ai_new in ai_candidates:
        da = ai_new - ai_old
        if s_old + da < 0.01:
            continue
        conv_new_max = np.max(conv + da * infl)
        obj_candidate = 2.0 * n * conv_new_max / (s_old + da)**2
        if obj_candidate < best_obj:
            best_obj = obj_candidate
            best_da = da
    
    # Refine around best
    if best_da != 0.0:
        ai_best = ai_old + best_da
        for frac in [0.5, 1.5, 0.25, 1.75, 0.75, 1.25]:
            ai_try = ai_best * frac if ai_best > 0 else ai_best + 0.01
            if ai_try < 0:
                continue
            da = ai_try - ai_old
            if s_old + da < 0.01:
                continue
            conv_new_max = np.max(conv + da * infl)
            obj_candidate = 2.0 * n * conv_new_max / (s_old + da)**2
            if obj_candidate < best_obj:
                best_obj = obj_candidate
                best_da = da
    
    new_ai = ai_old + best_da
    da = best_da
    conv_updated = conv + da * infl
    return new_ai, conv_updated, best_obj

A better approach: iteratively add constraints. Start with the top 5 peaks, solve, then check if any new peaks have appeared, add them, and re-solve. This is like a cutting-plane method.

```python
def lp_cutting_plane(a, time_budget, max_constraints=50):
    """LP with cutting plane: iteratively add violated constraints."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    start = time.monotonic()
    
    # Start with top 5 peaks
    conv_a = compute_conv(a_norm)
    active_set = set(np.argsort(conv_a)[::-1][:5].tolist())
    
    best_g = None
    best_M = np.inf
    
    while time.monotonic() - start < time_budget:
        # Build LP with current active set
        indices = sorted(active_set)
        A = np.zeros((len(indices), n))
        for idx, k in enumerate(indices):
            k = int(k)
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    A[idx, m] = a_norm[j]
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        g_val = g_val / s
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            break
        
        # Check for violated constraints
        ag = compute_conv_cross(a_norm, g_val)  # (a*g) via FFT
        violated = np.where(ag > M.value + 1e-6)[0]
        
        if len(violated) == 0:
            # LP is feasible with all constraints
            best_g = g_val
            best_M = M.value
            break
        
        # Add most violated constraints
        violated_sorted = violated[np.argsort(ag[violated])[::-1][:5]]
        for v in violated_sorted:
            active_set.add(int(v))
            if len(active_set) > max_constraints:
                # Remove least active constraint
                # Just keep the top ones by a_norm value or something
                break
    
    return best_g, best_M
```

Optimized step function coefficient search using:
1. FFT-accelerated LSE smoothed gradient descent (O(n log n) per step)
2. Cutting-plane LP for precise local optimization
3. Fast coordinate descent with O(n) per-coordinate evaluation
4. Multi-peak subgradient descent
5. Diverse initialization and n-exploration

Key innovation: FFT-based gradient computation enables ~100x more iterations
than the previous O(n^2) approach, allowing deeper local optimization.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Linear cross-correlation: (a*b)[k] = sum_j a[j]*b[k-j], length 2n-1."""
    na, nb = len(a), len(b)
    N = 1
    while N < na + nb - 1:
        N <<= 1
    fa = fft(a, N)
    fb = fft(b, N)
    return ifft(fa * fb, N).real[:na + nb - 1]
