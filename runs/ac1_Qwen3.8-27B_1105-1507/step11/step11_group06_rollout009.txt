import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def quadratic_line_search(a, g, t_max=1.5, n_coarse=50, n_fine=40):
    """Line search using the actual quadratic objective along the line a->g."""
    n = len(a)
    best_t = 0.0
    best_obj = compute_obj(a)

    # Coarse search
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Fine search around best
    if 0.0 < best_t < t_max:
        span = t_max / n_coarse
        lo = max(0.0, best_t - span)
        hi = min(t_max, best_t + span)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t

    return best_t, best_obj


def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def nesterov_subgrad(a, time_budget=60.0, multi_eps=0.02, lr0=0.05):
    """Nesterov-accelerated projected subgradient with multi-argmax."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()  # current iterate
    y = a.copy()  # extrapolated iterate
    t_k = 1.0
    momentum = np.zeros(n)
    beta = 0.75
    lr = lr0
    it = 0
    patience = 0
    best_patience = 0

    while time.monotonic() - start < time_budget:
        # Compute convolution at y
        conv_y = compute_conv(y)
        max_val = np.max(conv_y)

        # Multi-argmax: active set
        threshold = max_val * (1.0 - multi_eps)
        active = np.where(conv_y >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv_y))]

        # Compute averaged subgradient
        idx = np.arange(n)
        grad = np.zeros(n)
        for k in active:
            j1 = k - idx
            j2 = idx - k
            mask1 = (j1 >= 0) & (j1 < n)
            mask2 = (j2 >= 0) & (j2 < n)
            grad[mask1] += y[j1[mask1]]
            grad[mask2] += y[j2[mask2]]
        grad /= len(active)

        # Center the gradient (to maintain sum constraint approximately)
        grad = grad - np.mean(grad)

        # Momentum update
        lr_t = lr / (1.0 + 0.003 * it)
        momentum = beta * momentum - lr_t * grad

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new
        y_new = x + alpha * (x - y)
        y_new = np.maximum(y_new, 0)
        s_yn = y_new.sum()
        if s_yn > 1e-10:
            y_new = y_new / s_yn
        else:
            y_new = best_a.copy()

        # Step from y_new
        x_new = y_new + momentum
        x_new = project_simplex(x_new)

        # Evaluate
        obj = compute_obj(x_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = x_new.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive: adjust multi_eps as we converge
        if patience > 200:
            multi_eps = max(0.005, multi_eps * 0.7)

        x = x_new
        y = y_new
        t_k = t_new
        it += 1

        # Reset if stuck too long
        if patience > 800:
            x = best_a.copy()
            y = best_a.copy()
            momentum = np.zeros(n)
            lr *= 0.5
            patience = 0
            multi_eps = 0.02
            if lr < 0.002:
                break

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):

def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s

            print(f"  Phase 2: obj={obj_ns:.8f} (no improvement)")

    # === Phase 3: LP polish after subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: LP Polish Round 2 (budget={min(remaining*0.2, 150):.0f}s) ===")
        alloc = min(remaining * 0.2, 150.0)
        a_lp2 = lp_polish(best_seq, time_budget=alloc, max_iters=80, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 4: n Exploration ===")
        n_cur = len(best_seq)
        n_options = sorted(set([
            n_cur - 100, n_cur - 50,
            n_cur + 50, n_cur + 100, n_cur + 200,
            n_cur + 400, n_cur + 600,
            500, 600, 700, 900, 1000, 1200, 1400, 1600, 2000
        ]))
        # Filter valid and sufficiently different
        n_options = [n for n in n_options if 300 <= n <= 3000 and abs(n - n_cur) >= 30]
        # Sort by closeness to current best
        n_options.sort(key=lambda x: abs(x - n_cur))
        n_options = n_options[:8]
        print(f"  Candidates: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            if a_try is None:
                continue

            # Quick subgradient + short LP
            alloc_sg = min(remaining * 0.25, 40.0)
            a_try, obj_try = nesterov_subgrad(a_try, time_budget=alloc_sg, multi_eps=0.03, lr0=0.04)

            alloc_lp = min(remaining * 0.2, 50.0)
            if alloc_lp > 10:
                a_try = lp_polish(a_try, time_budget=alloc_lp, max_iters=30, verbose=False)
                obj_try = compute_obj(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Final Polish ===")
        n_final = len(best_seq)
        # Alternate subgradient and LP
        for round_idx in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            alloc_sg = min(remaining * 0.3, 30.0)
            if alloc_sg > 5:
                a_sg, obj_sg = nesterov_subgrad(best_seq, time_budget=alloc_sg,
                                                multi_eps=0.01, lr0=0.02)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()

            remaining = deadline - time.monotonic()
            alloc_lp = min(remaining * 0.4, 40.0)
            if alloc_lp > 8:
                a_lp_f = lp_polish(best_seq, time_budget=alloc_lp, max_iters=40, verbose=False)
                obj_lp_f = compute_obj(a_lp_f)
                if obj_lp_f < best_score:
                    best_score = obj_lp_f
                    best_seq = a_lp_f.copy()
                    print(f"  Round {round_idx}: obj={best_score:.8f} **")

    # === Phase 6: Last squeeze with tight params ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print(f"\n=== Phase 6: Last Squeeze ===")
        alloc = min(remaining - 8, 30.0)
        a_sq, obj_sq = nesterov_subgrad(best_seq, time_budget=alloc, multi_eps=0.005, lr0=0.01)
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_sq.copy()
            print(f"  Squeeze SG: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sq2 = lp_polish(best_seq, time_budget=min(remaining - 5, 25.0), max_iters=30, verbose=False)
            obj_sq2 = compute_obj(a_sq2)
            if obj_sq2 < best_score:
                best_score = obj_sq2
                best_seq = a_sq2.copy()
                print(f"  Squeeze LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
