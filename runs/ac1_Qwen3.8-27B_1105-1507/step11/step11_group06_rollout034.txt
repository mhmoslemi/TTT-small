Hybrid optimization: Nesterov subgradient + LP refinement + KKT polish.
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequence a.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_subgrad(a):
    """Compute subgradient of max(conv) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad, k_max, float(np.max(conv))

def nesterov_subgrad(a, time_budget=60.0):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    mu = 1.0  # momentum parameter
    lr0 = 0.1
    it = 0
    best_it = 0
    patience = 500
    
    # Pre-compute for speed
    while time.monotonic() - start < time_budget:
        # Adaptive learning rate
        lr_t = lr0 / (1.0 + 0.01 * it)
        
        # Subgradient
        grad, k_max, max_conv = compute_subgrad(y)
        
        # Projected gradient step in the simplex
        # Use the "heavy ball" method
        v = mu * v - lr_t * grad
        y_new = y + v
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = best_a.copy()
            v = np.zeros(n)
        else:
            y_new = y_new / s_new
        
        # Nesterov update
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        y = y_new + (mu / t_curr) * (y_new - y)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        
        # Evaluate and track best
        obj = compute_obj(y)
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()
            best_it = it
            # Reset momentum on improvement
            mu = max(0.5, mu * 0.99)
        it += 1
        
        # Patience-based restart
        if it - best_it > patience:
            y = best_a.copy()
            v = np.zeros(n)
            mu = 1.0
            lr0 = lr0 * 0.8
            patience = max(100, patience // 2)
            best_it = it
    
    return best_a

"""
Hybrid optimization: Nesterov accelerated subgradient + sparse LP refinement + KKT polish.
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequence a.
Key innovations: sparse LP formulation, Nesterov acceleration, adaptive active-set LP,
multi-scale n exploration, and KKT-based final refinement.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgrad_and_info(y):
    n = len(y)
    conv = compute_conv(y)
    max_conv = float(np.max(conv))
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += y[j1]
        if 0 <= j2 < n:
            grad[i] += y[j2]
    return grad, max_conv, k_max, conv


def nesterov_subgrad(a, time_budget=60.0, seed=42):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    y = a.copy()
    v = np.zeros(n)
    mu = 0.9
    lr0 = 0.5
    it = 0
    best_it = 0
    patience = 800
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        lr_t = lr0 / (1.0 + 0.003 * it)
        grad, max_conv, k_max, _ = compute_subgrad_and_info(y)
        v = mu * v - lr_t * grad
        y_new = y + v
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = best_a.copy()
            v = np.zeros(n)
        else:
            y_new = y_new / s_new
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        y = y_new + (mu / t_curr) * (y_new - y)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        obj = compute_obj(y)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()
            best_it = it
        it += 1
        if it - best_it > patience:
            y = best_a.copy()
            v = np.zeros(n)
            mu = 0.9
            lr0 = max(0.05, lr0 * 0.7)
            patience = max(200, patience // 2)
            best_it = it
            jitter = rng.uniform(-0.01, 0.01, n)
            y = best_a * (1.0 + jitter)
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y > 1e-10:
                y = y / s_y

    return best_a


def build_sparse_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for m in range(n):
        max_i = min(n, 2 * n - 1 - m)
        for i in range(max_i):
            rows.append(m + i)
            cols.append(m)
            vals.append(a[i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_full(a, top_k=None):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_sparse_conv_matrix(a_norm)

    if top_k is not None and top_k < 2 * n - 1:
        conv_current = compute_conv(a_norm)
        active_indices = np.argsort(conv_current)[::-1][:top_k]
        active_indices.sort()
        A_red = A[active_indices, :]
    else:
        A_red = A

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_red @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_steps=20):
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, 1.0, n_steps)
    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 0.01:
            continue
        a_try = a_try / s_try
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < 1.0:
        lo = max(0.0, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)
        ts_fine = np.linspace(lo, hi, 15)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def lp_polish(a, time_budget=100.0, max_iters=200, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(7777 + n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        top_k = min(2 * n - 1, max(50, 2 * n // 3))
        g_lp, M_lp = solve_lp_full(a, top_k=top_k)

        if g_lp is None:
            g_lp, M_lp = solve_lp_full(a, top_k=None)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search(a, g_lp, n_steps=25)
        if t_opt > 1e-8 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj - 1e-14:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0

        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={elapsed:.1f}s")

        if stall >= 4:
            strength = 0.02 * stall
            x = np.linspace(0, 1, n)
            mode = it % 5
            if mode == 0:
                pert = np.exp(-((x - 0.2) ** 2) / 0.02)
            elif mode == 1:
                pert = np.exp(-((x - 0.8) ** 2) / 0.02)
            elif mode == 2:
                pert = np.exp(-((x - 0.5) ** 2) / 0.05)
            elif mode == 3:
                pert = np.cos(4 * np.pi * x)
            else:
                pert = rng.uniform(-1, 1, n)
            a = best_a * np.exp(strength * pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0

def kkt_refine(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        active = np.where(conv >= max_conv * 0.999)[0]
        k_list = active.tolist()

        grad_total = np.zeros(n)
        for k in k_list:
            for i in range(n):
                j1 = k - i
                j2 = i - k
                if 0 <= j1 < n:
                    grad_total[i] += a[j1]
                if 0 <= j2 < n:
                    grad_total[i] += a[j2]

        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break

        mean_grad = float(np.dot(grad_total, a))
        direction = grad_total - mean_grad * a
        direction = direction[support]

        lr = 0.1 / (1.0 + 0.1 * (len(active)))
        best_t = 0.0
        for t in [lr * 0.1, lr * 0.5, lr, lr * 2, lr * 5]:
            a_try = a.copy()
            a_try[support] -= t * (grad_total[support] - mean_grad * a[support])
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj = compute_obj(a_try)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_t = t

        if best_t > 0:
            a_new = a.copy()
            a_new[support] -= best_t * (grad_total[support] - mean_grad * a[support])
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                a = a_new
                best_a = a_new.copy()
            else:
                break
        else:
            break

    return best_a


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n * 0.25) ** 2) / (2 * (n * 0.08) ** 2)) + \
            np.exp(-((idx - n * 0.75) ** 2) / (2 * (n * 0.08) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7.0
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = 1.0 / (0.05 + 2.0 * np.abs(2 * x - 1) ** 3)
    elif mode == 7:
        a = np.exp(-3.0 * (2 * x - 1) ** 2) + 0.3 * np.exp(-3.0 * (2 * x - 1) ** 4)
    else:
        a = np.ones(n) * (0.5 + 0.5 * np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    if n_src < 2:
        return np.ones(n_dst) / n_dst
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s

            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = (r + 4) % 8
            a_r = diverse_init(n_cur, rng, mode=mode)
            a_r = 0.3 * a_r + 0.7 * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            alloc = min(remaining * 0.35, 50.0)
            if alloc > 15:
                a_r = lp_polish(a_r, time_budget=alloc, max_iters=30, verbose=False)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
