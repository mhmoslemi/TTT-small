"""
Optimizes step function coefficients using an active-set coordinate descent
combined with LP-based direction finding and multi-length exploration.
Key improvements over previous approach:
1. Active-set CD: only optimizes coordinates that affect peak convolution lags
2. Faster 1D optimization with adaptive bracketing
3. Aggressive LP polishing with warm starts
4. Multi-length exploration with intelligent resampling
5. Better time management with dynamic budget allocation
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_normalized(a):
    """Objective assuming sum(a) = 1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def active_set_cd(a, time_budget=300.0, alpha=0.05, verbose=False, seed=42):
    """
    Active-set coordinate descent.
    Only optimizes coordinates that affect the peak lags.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1
        max_conv = float(np.max(conv))
        # Active lags: those within alpha of max
        threshold = max_conv * (1.0 - alpha)
        active_lags = np.where(conv >= threshold)[0]
        
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(conv))])

        # Determine which coordinates are "active" (affect active lags)
        active_coords = set()
        for k in active_lags:
            # Lag k is affected by coordinates i where max(0, k-n+1) <= i <= min(k, n-1)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            for i in range(i_lo, i_hi + 1):
                active_coords.add(i)
        
        # Convert to numpy for ordering
        active_coord_arr = np.array(list(active_coords)) if active_coords else np.arange(n)
        order = active_coord_arr[rng.permutation(len(active_coord_arr))]

        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-15:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Quick evaluation at a few points
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, max(0.0, v - scale * 0.5), v,
                          max(0.0, v + scale * 0.5), 1.0 / n, 2.0 / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section (shorter: 35 iterations)
            scale = max(best_x, 1.0 / n, 1e-6)
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(35):
                if x_hi - x_lo < 1e-12 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        # Full recompute every 10 sweeps
        if sweep % 10 == 0:
            conv = compute_conv(a)
            S = float(np.sum(a))
            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    ASCD sweep {sweep}: obj={best_obj:.8f}, active={len(active_coords)}, t={elapsed:.1f}s")

    return best_a


def lp_polish_fast(a, time_budget=60.0, seed=42, verbose=False):
    """
    LP-based polishing: solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Then blend.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        a_norm = a / float(np.sum(a))
        A = build_conv_matrix(a_norm)

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solved = False
        solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
        for solver in solvers:
            try:
                if time.monotonic() - start > time_budget - 3:
                    break
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        # Try multiple blending factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                # Perturb to escape
                pert = rng.normal(0, 0.02 * stall, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0

        if verbose:
            print(f"    LP: obj={best_obj:.8f}, stall={stall}")

    return best_a

def multiplicative_update(a, time_budget=60.0, seed=42, verbose=False):
    """Multiplicative weight update based on subgradient."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    lr = 5.0 / np.sqrt(n)

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        lr_eff = lr * (1.0 - 0.8 * progress)

        conv = compute_conv(a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]

        # Subgradient: for the lag k* that achieves max,
        # d/d a_j conv[k*] = a[k*-j] + a[j-k*] (if valid)
        k_star = int(max_idx)
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= k_star - j < n:
                grad[j] += a[k_star - j]
            if 0 <= j - k_star < n:
                grad[j] += a[j - k_star]
        grad *= 2.0  # Because conv = a*a, derivative w.r.t. a is 2*the convolution with a

        # Also consider near-max lags for a smoother gradient
        thresh = max_val * 0.95
        for k in np.where(conv >= thresh)[0]:
            k = int(k)
            if k == k_star:
                continue
            for j in range(n):
                if 0 <= k - j < n:
                    grad[j] += 0.3 * a[k - j]
                if 0 <= j - k < n:
                    grad[j] += 0.3 * a[j - k]

        # Multiplicative update
        norm_grad = np.max(np.abs(grad)) + 1e-10
        grad_norm = grad / norm_grad
        update = np.exp(-lr_eff * grad_norm)
        update = np.clip(update, 0.05, 20.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    MU it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 / (x**2 + 0.1)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * x * (1-x))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = x**3 + (1-x)**3 + 0.1
        a = a / a.sum()
    elif mode == 5:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2 + 0.05
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            s = float(np.sum(parent_arr))

                parent = parent_arr / s

                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
