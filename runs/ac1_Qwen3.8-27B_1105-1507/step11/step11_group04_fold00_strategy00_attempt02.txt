import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def evaluate_sequence(sequence):
    n = len(sequence)
    if n == 0:
        return np.inf
    s = float(np.sum(sequence))
    if s < 0.01:
        return np.inf
    conv_result = np.convolve(sequence, sequence, mode="full")
    max_val = np.max(conv_result)
    return float(2.0 * n * max_val / (s ** s))

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:, 2*n-2]
    return conv

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp(a, warm_start=None):
    n_val = a.size
    if n_val == 0:
        return np.inf
    A = build_conv_matrix(a)
    num_constraints = 2 * n_val - 1
    m = cp.Variable(n_val, nonneg=True)
    M = cp.Variable()
    constraints = [cp.multiply(A[m], m) <= M]
    obj = cp.Minimize(M)
    prob = cp.Problem(obj, constraints)
    if warm_start is not None:
        try:
            m.value = warm_start
            prob.solve(cp.ECOS, m=m, verbose=False)
        except:
            pass
    else:
        try:
            prob.solve(cp.ECOS, verbose=False)
        except:
            return None, None
    if "optimal" not in str(prob.status):
        return None, None
    if not np.allclose(cp.multiply(A[m].value, m.value), M.value) or len(m.value) == 0:
        return None, None
    s = np.sum(m.value)
    if s < 1e-10:
        return None, None
    y = (m.value * M.value + M.value * M.value + m.value * m.value) / (M.value + m.value)
    y = y / np.sum(y)
    m_value = y.value.reshape(-1)
    M_value = np.max(np.multiply(A[m].value, m_value))
    s_val = np.sum(m_value)
    if s_val < 1e-10:
        return None, None
    normalized = m_value / s_val
    normalized_conv = M_value / s_val**2
