The optimal n depends on the specific sequence shape. For the uniform sequence, the objective is always 2 regardless of n. For non-uniform sequences, there's an optimal n.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. Mirror descent (multiplicative weights) with adaptive LSE temperature
2. Exact coordinate descent for per-coordinate optimization
3. LP-based direction finding (top-k peak linearization)
4. Multi-n exploration with resampling
5. Simulated annealing restarts for escaping local minima
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_at_peak(a, peak_idx, top_k_indices=None, p=None):
    """
    Compute subgradient of max conv at given peak.
    If p is provided (probability distribution over lags), compute smoothed gradient.
    grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i.
    """
    n = len(a)
    if p is None:
        # Single peak
        p = np.zeros(2 * n - 1)
        p[peak_idx] = 1.0
    
    # grad[i] = 2 * sum_j a[j] * p[i+j]
    # = 2 * (conv(a_rev, p))[n-1 + i] where a_rev = a[::-1]
    a_rev = a[::-1]
    N_fft = 1
    while N_fft < n + len(p) - 1:
        N_fft <<= 1
    c = irfft(rfft(a_rev, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def lse_probability(conv, tau):
    """Compute softmax probability distribution over convolution values."""
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = shifted - shifted.max()  # numerical stability
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return p


def topk_probability(conv, k=5):
    """Uniform distribution over top-k peaks."""
    idx = np.argsort(conv)[-k:]
    p = np.zeros(len(conv))
    p[idx] = 1.0 / k
    return p


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights: a_i <- a_i * exp(-eta * grad_i), renormalize."""
    exponent = np.clip(-eta * grad, -50, 50)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def coordinate_descent_step(a, i, conv, max_iter=50):
    """
    Optimize coordinate i of a, keeping others fixed.
    Returns new value of a[i] and the new max conv.
    """
    n = len(a)
    a_old = a[i]
    S_sum = float(np.sum(a)) - a_old
    
    # Compute c_k = 2*a[k-i] if 0<=k-i<n else 0
    # And handle the special case k=2i where it's a[i]^2 not 2*a[i]*a[k-i]
    c = np.zeros(2 * n - 1)
    valid = (k_idx := np.arange(2 * n - 1) - i)
    # Hmm, let me just do it directly
    for k in range(2 * n - 1):
        j = k - i
        if 0 <= j < n:
            if k == 2 * i:
                # conv[k] has a[i]^2 term
                pass  # handle specially
            else:
                c[k] = 2.0 * a[j]
    
    # S[k] = conv[k] - a[i] * c[k] (for k != 2i)
    # For k = 2i: S[2i] = conv[2i] - a[i]^2
    S = conv.copy()
    for k in range(2 * n - 1):
        j = k - i
        if 0 <= j < n:
            if k == 2 * i:
                S[k] = conv[k] - a_old * a_old
            else:
                S[k] = conv[k] - a_old * c[k]
    
    # Now optimize: minimize 2*n*max(S[k] + x*c[k] or S[2i]+x^2) / (S_sum + x)^2
    # Subject to x >= 0
    
    def obj_1d(x):
        if x < 0:
            return np.inf
        new_conv = S + x * c
        if 2 * i < 2 * n - 1:
            new_conv[2 * i] = S[2 * i] + x * x
        s_new = S_sum + x
        if s_new < 0.01:
            return np.inf
        return 2.0 * n * float(np.max(new_conv)) / (s_new * s_new)
    
    # Use bounded minimization
    x_max = max(10.0 * a_old, S_sum * 2.0 / n, 0.1)
    res = minimize_scalar(obj_1d, bounds=(0, x_max), method='bounded',
                          options={'xatol': 1e-10})
    x_new = max(0.0, res.x)
    
    a[i] = x_new
    return x_new
