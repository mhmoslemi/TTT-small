import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_soft_gradient(a, p):
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q

def multiplicative_update(a, eta, beta, time_per_iter=None):
    """One step of multiplicative (mirror descent) update."""
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    if c_max == 0:
        return a, conv
    exp_vals = np.exp(beta * (conv - c_max))
    p = exp_vals / float(np.sum(exp_vals))
    grad = compute_soft_gradient(a, p)
    # Multiplicative update
    a_new = a * np.exp(-eta * grad)
    s = float(np.sum(a_new))
    if s < 1e-30:
        return a, conv
    a_new = a_new / s
    return a_new, conv

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective assuming a is already normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_soft_gradient(a, p):
    """Gradient of soft-conv objective via FFT cross-correlation."""
    n = len(a)
    p_len = len(p)
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - n + 1)
        hi = min(2 * n - 1, j + n - 1)
        C[lo:hi + 1, j] = a[hi - j:lo - j + 1][::-1]
    return C

def adam_softmax_gd(a, time_budget, beta_start=8.0, beta_end=60.0,
                    lr_start=0.01, seed=42, verbose=True):

    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    best_t = 0

    while time.monotonic() - start < time_budget:

def frank_wolfe(a, time_budget, max_iters=500000, verbose=True):
    """Frank-Wolfe (conditional gradient) with exact subgradient."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad_i = 2*a_{k_star - i} if 0 <= k_star-i <= n-1
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        grad = np.zeros(n)
        indices = np.arange(lo, hi + 1)
        grad[indices] = 2.0 * a[k_star - indices]

        # Linear minimization: argmin grad_i
        i_min = int(np.argmin(grad))

        # Step size: Gamma = 2/(it+2)
        gamma = 2.0 / (it + 2)

        e_min = np.zeros(n)
        e_min[i_min] = 1.0
        a_new = (1 - gamma) * a + gamma * e_min
        a_new = project_simplex(np.maximum(a_new, 0))

        obj_new = compute_obj(a_new)
        if np.isfinite(obj_new) and obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        a = a_new
        it += 1

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a, best_obj


def solve_lp_minpeak(a, solver_order=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))

    # Build convolution matrix
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - n + 1)
        hi = min(2 * n - 1, j + n - 1)
        C[lo:hi + 1, j] = a_norm[hi - j:lo - j + 1][::-1]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [C @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_order is None:
        solver_order = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDDLP]

    for solver in solver_order:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_polish(a, time_budget, max_iters=200, verbose=True):
    """LP-based best-response polishing with line search."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            stall += 1
            if stall >= 3:
                break
            continue

        t_opt, obj_opt = line_search_blend(a, g_lp)

        if t_opt > 0.01 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if np.isfinite(obj_new) and obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M_lp={M_lp:.6f}, t_elapsed={elapsed:.1f}s")

        if stall >= 5:
            # Perturb and restart
            noise_scale = 0.005 * (stall - 3)
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(best_a + noise)
            stall = 0

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(np.maximum(1 - (2 * x - 1) ** 2, 1e-10))
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(3, n)] = 0.3
        a[max(0, n - 4):n - 1] = 0.3
        if n > 4:
            a[1:-1] += 0.05
    elif mode == 8:
        # Double-peaked
        a = np.exp(-50 * (x - 0.2) ** 2) + np.exp(-50 * (x - 0.8) ** 2) + 0.01
    elif mode == 9:
        # Power law
        a = (x + 0.01) ** (-0.5) + (1 - x + 0.01) ** (-0.5) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(np.maximum(a_new, 0))


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase hybrid optimization.
    Phase 1: Mirror descent (fast global)
    Phase 2: Adam soft-max (smooth local)
    Phase 3: Frank-Wolfe (exact subgradient)
    Phase 4: LP polishing (best-response)
    Phase 5: Diverse restarts
    Phase 6: Length exploration
    Phase 7: Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    n_current = 800
    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-init (n={n_current}): obj={best_score:.8f}")

    # Try arcsine
    a_arc = arcsine_init(len(best_seq))
    obj_arc = compute_obj(a_arc)
    if obj_arc < best_score:
        best_score = obj_arc
        best_seq = a_arc.copy()
        print(f"Arcsine init: obj={best_score:.8f}")

    # Symmetrize
    best_seq, best_score = try_symmetrize(best_seq, best_score)

    # === Phase 1: Mirror Descent ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: Mirror Descent, budget={min(remaining*0.18, 180):.0f}s ===")
        alloc = min(remaining * 0.18, 180.0)
        a1, obj1 = mirror_descent(best_seq, time_budget=alloc,
                                  beta_start=3.0, beta_end=35.0,
                                  eta_start=0.3, eta_end=1.2,
                                  seed=seed, verbose=True)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 MD: obj={obj1:.8f} (no improvement)")
        best_seq, best_score = try_symmetrize(best_seq, best_score)

    # === Phase 2: Adam Soft-Max ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 2: Adam Soft-Max, budget={min(remaining*0.18, 180):.0f}s ===")
        alloc = min(remaining * 0.18, 180.0)
        a2, obj2 = adam_softmax_gd(best_seq, time_budget=alloc,
                                   beta_start=8.0, beta_end=55.0,
                                   lr_start=0.01, seed=seed + 1, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 Adam: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 Adam: obj={obj2:.8f} (no improvement)")
        best_seq, best_score = try_symmetrize(best_seq, best_score)

    # === Phase 3: Frank-Wolfe ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Frank-Wolfe, budget={min(remaining*0.06, 60):.0f}s ===")
        alloc = min(remaining * 0.06, 60.0)
        a3, obj3 = frank_wolfe(best_seq, time_budget=alloc, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 FW: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 FW: obj={obj3:.8f} (no improvement)")

    # === Phase 4: LP Polishing (primary refinement) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: LP Polishing, budget={min(remaining*0.28, 350):.0f}s ===")
        alloc = min(remaining * 0.28, 350.0)
        a4, obj4 = lp_polish(best_seq, time_budget=alloc, max_iters=300, verbose=True)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj4:.8f} (no improvement)")
        best_seq, best_score = try_symmetrize(best_seq, best_score)

    # === Phase 5: Diverse Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 5: Diverse Restarts, budget={remaining:.0f}s ===")
        n_cur = len(best_seq)
        modes_to_try = [7, 8, 9, 0, 2, 4, 6]

        for r in range(len(modes_to_try)):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            mode = modes_to_try[r]
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best
            blend = 0.5 + 0.1 * r
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)

            # Quick mirror descent
            alloc_md = min(remaining * 0.2, 60.0)
            if alloc_md > 15:
                a_r, obj_r = mirror_descent(a_r, time_budget=alloc_md,
                                            beta_start=5.0, beta_end=30.0,
                                            eta_start=0.3, eta_end=1.0,
                                            seed=seed + 100 + r, verbose=False)

            # Quick LP
            remaining = deadline - time.monotonic()
            if remaining > 40:
                alloc_lp = min(remaining * 0.15, 60.0)
                if alloc_lp > 10:
                    a_r, obj_r = lp_polish(a_r, time_budget=alloc_lp, max_iters=30, verbose=False)

            a_r, obj_r = try_symmetrize(a_r, obj_r)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Length Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 6: Length Exploration, budget={remaining:.0f}s ===")
        n_best = len(best_seq)

        n_candidates = set()
        for delta in [-200, -100, -50, 50, 100, 200, 300, 400, 500]:
            n_try = n_best + delta
            if 300 <= n_try <= 2500:
                n_candidates.add(n_try)
        # Add some absolute values
        for n_abs in [400, 500, 600, 700, 1000, 1200, 1500, 1800, 2000]:
            if abs(n_abs - n_best) >= 50:
                n_candidates.add(n_abs)

        # Sort by distance to current best (try nearby first)
        n_sorted = sorted(n_candidates, key=lambda x: abs(x - n_best))
        n_sorted = n_sorted[:5]  # Try at most 5 new lengths
        print(f"  Trying n values: {n_sorted}")

        for n_try in n_sorted:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_try = resample_sequence(best_seq, n_try)

            # Optimize at new length
            alloc_opt = min(remaining * 0.35, 80.0)
            if alloc_opt > 15:
                a_try, obj_try = mirror_descent(a_try, time_budget=alloc_opt * 0.4,
                                                beta_start=5.0, beta_end=30.0,
                                                eta_start=0.3, eta_end=1.0,
                                                seed=seed + 300, verbose=False)
                remaining = deadline - time.monotonic()
                alloc_lp = min(remaining * 0.3, 50.0)
                if alloc_lp > 10:
                    a_try, obj_try = lp_polish(a_try, time_budget=alloc_lp, max_iters=25, verbose=False)

            a_try, obj_try = try_symmetrize(a_try, obj_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Intensive Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Final Polish, budget={remaining:.0f}s ===")

        # Adam with high beta
        alloc1 = min(remaining * 0.3, 80.0)
        if alloc1 > 20:
            a_f, obj_f = adam_softmax_gd(best_seq, time_budget=alloc1,
                                         beta_start=30.0, beta_end=70.0,
                                         lr_start=0.004, seed=seed + 500, verbose=False)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final Adam: obj={best_score:.8f} **")

        # Mirror descent one more time
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.25, 60.0)
        if alloc2 > 15:
            a_f2, obj_f2 = mirror_descent(best_seq, time_budget=alloc2,
                                          beta_start=10.0, beta_end=50.0,
                                          eta_start=0.2, eta_end=0.8,
                                          seed=seed + 600, verbose=False)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final MD: obj={best_score:.8f} **")

        # LP polish
        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.5, 100.0)
        if alloc3 > 15:
            a_f3, obj_f3 = lp_polish(best_seq, time_budget=alloc3, max_iters=80, verbose=True)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        best_seq, best_score = try_symmetrize(best_seq, best_score)

    # === Phase 8: Last-ditch random perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Last-ditch perturbations ===")
        for attempt in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            noise = rng.normal(0, 0.002, len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.5, 15.0)
            if alloc > 5:
                a_p, obj_p = lp_polish(a_p, time_budget=alloc, max_iters=10, verbose=False)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Perturb {attempt}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    # Ensure all values are valid for evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_score = val
    else:
        result = [1.0] * 100
        final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_score:.8f}, time={elapsed:.1f}s ===")

    return result
