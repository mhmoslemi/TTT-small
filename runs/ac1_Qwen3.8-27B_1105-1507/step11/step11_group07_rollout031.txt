```python
"""
Optimized step function search combining soft-max gradient descent,
intensive LP polishing, subgradient averaging, and length exploration.
Target: minimize 2n * max(conv(a,a)) / (sum(a))^2 below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1 x n) matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def soft_max_gd(a, time_budget=200.0, beta_start=5.0, beta_end=60.0,
                lr_start=0.01, momentum=0.9, seed=42):
    """Soft-max gradient descent on simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.7)
        lr = lr_start * (1.0 - 0.5 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via cross-correlation
        n_conv = len(p)
        a_rev = a[::-1]
        size = 1
        while size < n + n_conv - 1:
            size <<= 1
        A_fft = rfft(a_rev, size)
        P_fft = rfft(p, size)
        full = irfft(A_fft * P_fft, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                v = np.zeros(n)
                a = project_simplex(best_a + rng.normal(0, 0.003, n))
                stall = 0
        
        a = a_new
        it += 1
        
        if it % 200 == 0:
            print(f"    SMGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def subgradient_averaging(a, time_budget=200.0, seed=999):
    """Polyak-Ruppert subgradient method with averaging."""

    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Running average
    avg_a = np.zeros(n)
    count = 0
    
    # Step size schedule
    eta0 = 0.5
    gamma = 0.5  # decay rate
    
    it = 0
    while time.monotonic() - start < time_budget:
        eta = eta0 / (1 + gamma * it)
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        S = np.where(conv >= max_val - 1e-12)[0]
        
        # Subgradient
        s = np.zeros(n)
        for k in S:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    s[j] += 2.0 * a[idx]
        s /= len(S)
        
        # Update
        a_new = project_simplex(a - eta * s)
        
        # Averaging
        avg_a += a_new
        count += 1
        avg_candidate = avg_a / count
        obj_avg = compute_obj(avg_candidate)
        obj_curr = compute_obj(a_new)
        
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = avg_candidate.copy()
        if obj_curr < best_obj:
            best_obj = obj_curr
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        if it % 100 == 0:
            print(f"    SUB it{it}: obj_curr={obj_curr:.8f}, obj_avg={obj_avg:.8f}, "
                  f"best={best_obj:.8f}, eta={eta:.4f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj
```

Let me also add one more idea: **active set LP**. At the optimal solution, only a few of the 2n-1 constraints are active (binding). If we can identify the active set, we can solve a much smaller LP.

The active set at a point a is: S_active = {k : conv(a,a)[k] = max(conv(a,a))}.

If we only enforce the active constraints, the LP becomes:
min M s.t. conv(a, g)[k] <= M for k in S_active, sum(g) = 1, g >= 0

This has |S_active| + 1 constraints instead of 2n-1 + 1. If |S_active| is small (say 1-5), this is much faster.

But this is a relaxation (fewer constraints), so the solution might violate the inactive constraints. We'd need to check and add violated constraints (cutting plane approach).

This is essentially the "separating hyperplane" approach:
1. Solve LP with current active set
2. Check if solution violates any inactive constraint
3. If yes, add that constraint and re-solve
4. Repeat until no violations

This could be much faster than solving the full LP! Let me implement this.

```python
def active_set_lp(a, max_inner_iters=20):
    """Active set LP: start with binding constraints, add violated ones."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Initial active set: indices where conv(a,a) is max
    conv_a = compute_conv(a_norm)
    max_val = float(np.max(conv_a))
    # Include near-max constraints too (within tolerance)
    tol = 0.01 * max_val
    active_set = list(np.where(conv_a >= max_val - tol)[0])
    
    for _ in range(max_inner_iters):
        # Solve LP with active set only
        A_rows = build_conv_matrix(a_norm)[active_set, :]
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_rows @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
```

```python
"""
Step function optimization combining:
- Soft-max gradient descent (FFT-accelerated) for smooth exploration
- Intensive LP polishing for precise refinement
- Subgradient averaging (Polyak-Ruppert) for global convergence
- Active-set LP acceleration
- Length exploration for optimal sequence size
Target: minimize 2n*max(conv(a,a))/(sum(a))^2 below 1.5027
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1 x n) matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_full(a):
    """Solve full LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_active_set(a, max_inner=15):
    """Active-set LP: solve with only binding constraints, iteratively add violated."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_full = build_conv_matrix(a_norm)

    conv_a = compute_conv(a_norm)
    max_val = float(np.max(conv_a))
    tol = 0.02 * max_val
    active_set = list(np.where(conv_a >= max_val - tol)[0])

    for _ in range(max_inner):
        if len(active_set) > 2 * n:
            active_set = list(np.argsort(A_full @ (a / float(np.sum(a)))))[-2 * n:]

        A_sub = A_full[active_set, :]
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_sub @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        g_val = g_val / s
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        conv_g = A_full @ g_val
        M_val = float(M.value)
        violations = conv_g - M_val
        max_viol = float(np.max(violations))

        if max_viol < 1e-7:
            return g_val, M_val

        k_new = int(np.argmax(violations))
        if k_new not in active_set:
            active_set.append(k_new)

    return solve_lp_full(a)


def soft_max_gd(a, time_budget=200.0, beta_start=5.0, beta_end=60.0,
                lr_start=0.01, momentum=0.9, seed=42):
    """Soft-max gradient descent with momentum on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.6)
        lr = lr_start * (1.0 - 0.4 * progress)

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))

        n_conv = len(p)
        a_rev = a[::-1]
        size = 1
        while size < n + n_conv - 1:
            size <<= 1
        A_fft = rfft(a_rev, size)
        P_fft = rfft(p, size)
        full = irfft(A_fft * P_fft, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q

        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 60:
                v = np.zeros(n)
                a = project_simplex(best_a + rng.normal(0, 0.002, n))
                stall = 0

        a = a_new
        it += 1

        if it % 300 == 0:
            print(f"    SMGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")

        if it - best_it > 4000 and it > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            it = best_it
            lr *= 0.7

    return best_a, best_obj


def lp_polish_intensive(a, time_budget=400.0, max_iters=1000):

    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    rng = np.random.default_rng(77777)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        if use_active:
            g_lp, M_lp = solve_lp_active_set(a)
        else:
            g_lp, M_lp = solve_lp_full(a)

        if g_lp is None:
            use_active = not use_active
            g_lp, M_lp = solve_lp_full(a) if not use_active else solve_lp_active_set(a)
            if g_lp is None:
                break

        # Strategy 1: Use LP solution directly
        obj_direct = compute_obj(g_lp)

        # Strategy 2: Blend with current
        best_t = 0.0
        best_blend_obj = best_obj
        a_try_direct = project_simplex(g_lp)
        objs = [obj_direct]
        t_vals = [1.0]

            a_blend = (1.0 - t) * a + t * g_lp
            a_blend = project_simplex(np.maximum(a_blend, 0))
            obj_blend = compute_obj(a_blend)
            objs.append(obj_blend)
            t_vals.append(t)
            if obj_blend < best_blend_obj:
                best_blend_obj = obj_blend
                best_t = t

        # Pick best overall
        best_idx = int(np.argmin(objs))
        best_candidate_obj = objs[best_idx]
        best_candidate_t = t_vals[best_idx]

        if best_candidate_obj < best_obj - 1e-12:
            if best_candidate_t >= 1.0:
                a_new = project_simplex(g_lp)
            else:
                a_new = (1.0 - best_candidate_t) * a + best_candidate_t * g_lp
                a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LPIP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, "
                  f"t_best={best_candidate_t:.2f}, elapsed={elapsed:.0f}s, remaining={remaining:.0f}s")

        if stall >= 5:
            if stall <= 9:
                noise_scale = 0.003 * (stall - 4)
                a = project_simplex(best_a + rng.normal(0, noise_scale, n))
            elif stall <= 14:
                x = np.linspace(0, 1, n)
                sharp = 2.0 + 2.0 * rng.random()
                a_div = 1.0 / (0.03 + sharp * (2.0 * x - 1.0) ** 2)
                a = project_simplex(0.4 * a_div + 0.6 * best_a)
            else:
                a = diverse_init(n, rng, mode=int(rng.integers(0, 8)))
                a = project_simplex(0.3 * a + 0.7 * best_a)
            stall = 0
            use_active = not use_active

    return best_a, best_obj


def compute_subgradient(a, S):
    """Compute subgradient of max_k conv(a,a)_k at point a, active set S."""
    n = len(a)
    m = 2 * n - 1
    h = np.zeros(m)
    h[S] = 1.0 / len(S)
    s = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(m, j + n)
        s[j] = 2.0 * float(np.dot(a[:k_end - k_start], h[k_start:k_end]))
    return s


def subgradient_averaging(a, time_budget=200.0, seed=999):
    """Polyak-Ruppert subgradient method with averaged iterates."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    avg_a = np.zeros(n)
    count = 0
    eta0 = 0.3
    gamma = 0.3
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        eta = eta0 / (1.0 + gamma * it)

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        tol = 1e-10 + 1e-6 * max_val
        S = list(np.where(conv >= max_val - tol)[0])

        s = compute_subgradient(a, S)

        a_new = project_simplex(a - eta * s)

        avg_a += a_new
        count += 1
        avg_candidate = avg_a / count
        obj_avg = compute_obj(avg_candidate)
        obj_curr = compute_obj(a_new)

        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = avg_candidate.copy()
            stall = 0
        elif obj_curr < best_obj:
            best_obj = obj_curr
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > 100:
            a = project_simplex(best_a + rng.normal(0, 0.005, n))
            avg_a = np.zeros(n)
            count = 0
            stall = 0

        a = a_new
        it += 1

        if it % 200 == 0:
            print(f"    SUB it{it}: obj_curr={obj_curr:.8f}, obj_avg={obj_avg:.8f}, "
                  f"best={best_obj:.8f}, eta={eta:.5f}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations of length n."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2.0 * x - 1.0) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2.0 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2.0 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4.0 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(9) / 9.0
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4.0 * (2.0 * x - 1.0) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2.0 * x - 1.0)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(4.0 * (2.0 * x - 1.0))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(5, n)] = 0.5
        a[max(0, n - 6):n - 1] = 0.5
        if n > 6:
            a[1:-1] += 0.03
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2.0 * x - 1.0) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1.0 - (2.0 * x - 1.0) ** 2))
    return project_simplex(a)


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(np.maximum(a_new, 0))

    remaining = deadline - time.monotonic()
    if remaining > 150:
        print(f"\n=== Phase 6: Length Exploration, budget={remaining:.0f}s ===")
        n_best = len(best_seq)

        n_options = sorted(set([
            max(400, n_best - 200), max(400, n_best - 100),
            n_best + 100, n_best + 200, n_best + 400,
            600, 900, 1100, 1300, 1600
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_best) >= 80]
        n_options = n_options[:4]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break

            a_try = interpolate_sequence(best_seq, n_try)

            alloc_sgd = min(remaining * 0.3, 60.0)
            if alloc_sgd > 15:
                a_try, obj_try = soft_max_gd(a_try, time_budget=alloc_sgd,
                                             beta_start=10.0, beta_end=40.0,
                                             lr_start=0.008, momentum=0.9,
                                             seed=seed + 300)

            remaining = deadline - time.monotonic()
            if remaining > 40:
                alloc_lp = min(remaining * 0.4, 80.0)
                a_try, obj_try = lp_polish_intensive(a_try, time_budget=alloc_lp, max_iters=200)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Intensive Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 7: Final Polish, budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.7, 200.0)
        a_f, obj_f = lp_polish_intensive(best_seq, time_budget=alloc, max_iters=1000)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final: obj={best_score:.8f} **")
        else:
            print(f"  Final: obj={obj_f:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # One more soft-max pass with high beta
            alloc2 = min(remaining * 0.6, 80.0)
            a_f2, obj_f2 = soft_max_gd(best_seq, time_budget=alloc2,
                                       beta_start=30.0, beta_end=100.0,
                                       lr_start=0.003, momentum=0.93,
                                       seed=seed + 500)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final SMGD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
