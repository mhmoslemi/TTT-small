"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_budget=15.0, warm_start=None):
    """Solve LP: min M s.t. conv(a_norm, g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        g.value = warm_start
        M.value = float(np.max(A @ warm_start))

    start = time.monotonic()
    for solver in ['MOSEK', 'CLARABEL', 'SCS']:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def coordinate_descent(a, time_budget=300.0, seed=42):
    """Fast coordinate descent with piecewise-quadratic 1D minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    # Precompute index info for each coordinate
    # For coordinate i, the affected convolution indices are [i, i+n-1]
    # The quadratic term is at 2i, linear terms at i+j for j != i
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            # Indices affected: k from i to i+n-1
            k_min = i
            k_max = i + n - 1
            k_quad = 2 * i  # the quadratic term

            # Linear terms: conv[k] changes by 2*(x-v)*a[k-i] for k in [i, i+n-1], k != 2i
            # Quadratic term: conv[2i] changes by (x^2 - v^2)

            # Get the linear coefficients
            if k_quad >= k_min and k_quad <= k_max:
                # Split into linear part (excluding quad) and quad part
                # Linear: for k in [k_min, k_max], k != k_quad:
                #   new_conv[k] = conv[k] + 2*(x-v)*a[k-i]
                # Quad: new_conv[2i] = conv[2i] + x^2 - v^2
                
                # The max is max(max_linear(x), quad_val(x))
                # where max_linear(x) = max over k != quad of (conv[k] + 2*a[k-i]*(x-v))
                # and quad_val(x) = conv[2i] + x^2 - v^2
                
                # To find the optimal x, we need to minimize:
                # f(x) = 2*n * max(max_linear(x), quad_val(x)) / (S_other + x)^2
                
                # This is a 1D problem. We can solve it with golden section search.
                
                # For efficiency, precompute the linear coefficients
                ks = np.arange(k_min, k_max + 1)
                a_coeffs = a[ks - i]  # a[k-i] for each k
                c_vals = conv[ks].copy()
                
                # Mask out the quadratic index
                mask = ks != k_quad
                ks_lin = ks[mask]
                a_lin = a_coeffs[mask]
                c_lin = c_vals[mask]
                c_quad = float(conv[k_quad])

                # We want to minimize f(x) = 2n * max(max(c_lin + 2*a_lin*(x-v)), c_quad + x^2 - v^2) / (S_other+x)^2
                # subject to x >= 0
                
                # Golden section search
                # Initial bracket: [0, max(2v, 1/n * 2)]
                x_lo = 0.0
                x_hi = max(2.0 * v + 0.1/n, 0.5/n)
                
                def f_1d(x):
                    if len(ks_lin) > 0:
                        lin_vals = c_lin + 2.0 * a_lin * (x - v)
                        max_lin = float(np.max(lin_vals))
                    else:
                        max_lin = 0.0
                    quad_val = c_quad + x * x - v * v
                    max_val = max(max_lin, quad_val)
                    denom = (S_other + x) * (S_other + x)
                    if denom < 1e-30:
                        return 1e30
                    return max_val / denom  # We minimize max_val/denom, equivalent to the full obj
                
                c_pt = x_hi - gr * (x_hi - x_lo)
                d_pt = x_lo + gr * (x_hi - x_lo)
                fc = f_1d(c_pt)
                fd = f_1d(d_pt)

                for _ in range(50):
                    if x_hi - x_lo < 1e-15 * max(1.0, x_hi):
                        break
                    if fc < fd:
                        x_hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = x_hi - gr * (x_hi - x_lo)
                        fc = f_1d(c_pt)
                    else:
                        x_lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = x_lo + gr * (x_hi - x_lo)
                        fd = f_1d(d_pt)

                x_opt = max(0.0, (x_lo + x_hi) / 2.0)
                if abs(x_opt - v) < 1e-15:
                    continue

                # Update conv
                delta = x_opt - v
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_lin
                conv[k_quad] += x_opt * x_opt - v * v
                a[i] = x_opt
                S += delta

                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

    return best_a


def lp_direction(a, time_budget=20.0):

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
