import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1 for speed."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_soft_gradient(a, p):
    """
    Compute gradient of soft-max objective.
    grad_i = 2 * sum_j a_j * p_{i+j}
    Uses FFT cross-correlation for O(n log n) efficiency.
    """
    n = len(a)
    p_len = len(p)  # should be 2n-1
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

            t = 0

        return compute_obj(a_try / s)
    
    c = lo + gr * (hi - lo)

    fc = obj_at(c)
    fd = obj_at(d)
    
    start = time.monotonic()
    for _ in range(20):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2)

def diverse_init(n, rng, mode=0):

        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +

    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7

    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        # Sharp edges: most mass at two endpoints
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(3, n)] = 0.3
        a[max(0, n-4):n-1] = 0.3
        if n > 4:
            a[1:-1] += 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
