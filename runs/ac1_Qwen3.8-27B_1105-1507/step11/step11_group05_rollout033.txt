    n = len(a)

    if s < 1e-10:

    conv = compute_conv(a_norm)

    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)

            # grad[lo:hi+1] += wval * a[ki-hi : ki-lo+1][::-1]
            # When j is in [lo, hi], k-j is in [ki-hi, ki-lo], and a[k-j]
            grad[lo:hi + 1] += wval * a[ki - hi:ki - lo + 1][::-1]
    return 4.0 * n * grad


def lse_gradient_descent(a, time_budget, T_start=15.0, T_end=150.0, verbose=False):
    """LSE smoothed gradient descent with backtracking and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / n
    momentum = np.zeros(n)
    beta = 0.5
    it = 0
    best_it = 0
    patience = 800

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac

        grad = lse_gradient_fft(a, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Nesterov-like momentum
        momentum = beta * momentum - lr * grad / (gnorm + 1e-15)
        a_try = project_simplex(a + momentum)

        # Check true objective with backtracking
        obj_old = compute_obj(a)
        improved = False
        for retry in range(6):
            obj_try = compute_obj(a_try)
            if obj_try < obj_old - 1e-13:
                a = a_try
                improved = True
                break
            lr *= 0.6
            momentum = beta * momentum - lr * grad / (gnorm + 1e-15)
            a_try = project_simplex(a + momentum)

        if not improved:
            # Random perturbation escape
            rng_local = np.random.default_rng(it * 104729 + 7)
            noise = rng_local.normal(0, 0.002, n)
            a_try = project_simplex(a + noise)
            obj_try = compute_obj(a_try)
            if obj_try < obj_old:
                a = a_try
                improved = True
            if not improved:
                lr = max(1e-8, lr * 0.5)

        obj_true = compute_obj(a)
        if obj_true < best_obj - 1e-14:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 1.0 / n * 0.3
            beta = 0.3
            patience = max(200, patience // 2)
            best_it = it

        if verbose and it % 500 == 0:
            print(f"      LSE-GD it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def exact_subgrad_descent(a, time_budget, top_k=5, verbose=False):
    """Exact subgradient method with adaptive step size and restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / n
    it = 0
    best_it = 0
    patience = 600
    stall_count = 0

    while time.monotonic() - start < time_budget:
        grad = multi_peak_subgrad(a, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Polyak-like step size
        obj_current = compute_obj(a)
        # Use a heuristic step size
        step = lr / (gnorm + 1e-15)
        a_try = project_simplex(a - step * grad)

        improved = False
        for retry in range(5):
            obj_try = compute_obj(a_try)
            if obj_try < obj_current - 1e-14:
                a = a_try
                improved = True
                break
            step *= 0.5
            a_try = project_simplex(a - step * grad)

        if not improved:
            stall_count += 1
            if stall_count >= 5:
                # Perturb and restart
                rng_local = np.random.default_rng(it * 31 + 17)
                noise = rng_local.normal(0, 0.003 * min(stall_count, 10), n)
                a = project_simplex(best_a + noise)
                stall_count = 0
                lr = 2.0 / n
            else:
                lr = max(1e-8, lr * 0.8)
        else:
            stall_count = 0
            lr = min(10.0 / n, step * 1.2)

        obj_true = compute_obj(a)
        if obj_true < best_obj - 1e-14:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            lr = 1.0 / n
            patience = max(200, patience // 2)
            best_it = it

        if verbose and it % 500 == 0:
            print(f"      SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def build_lp_matrix(a_norm, top_m):
    """Build constraint matrix for top_m peaks: A[g] gives conv(a_norm, g) at those peaks."""
    n = len(a_norm)
    conv_a = compute_conv(a_norm)
    top_indices = np.argsort(conv_a)[::-1][:top_m]

    A = np.zeros((len(top_indices), n))
    for idx, k in enumerate(top_indices):
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            A[idx, lo:hi + 1] = a_norm[k - hi:k - lo + 1][::-1]
    return A


def lp_polish(a, time_budget, top_m_start=25, top_m_end=10, max_iters=40, verbose=False):
    """Iterative LP polishing: solve min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0

    solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS]

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        frac = min(1.0, it / max(1, max_iters - 1))
        top_m = int(round(top_m_start + frac * (top_m_end - top_m_start)))
        top_m = max(5, min(top_m, 2 * n - 2))

        a_norm = best_a / float(np.sum(best_a))
        A = build_lp_matrix(a_norm, top_m)

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        for solver in solver_list:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    break
            except Exception:
                continue

        if g_val is None:
            break

        s_g = float(np.sum(g_val))
        if s_g < 1e-10:
            break
        g_val = g_val / s_g

        # Find best blend
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * best_a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_new = float(np.sum(a_try))
            if s_new < 0.01:
                continue
            a_try = a_try / s_new
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        # Refine blend
        if best_t > 0:
            for dt in [-0.1, -0.05, 0.05, 0.1, 0.15]:
                t_try = best_t + dt
                if 0 < t_try < 1.5:
                    a_try = (1 - t_try) * best_a + t_try * g_val
                    a_try = np.maximum(a_try, 0)
                    s_new = float(np.sum(a_try))
                    if s_new < 0.01:
                        continue
                    a_try = a_try / s_new
                    obj_try = compute_obj(a_try)
                    if obj_try < best_blend_obj:
                        best_blend_obj = obj_try
                        best_t = t_try

        if best_t > 0 and best_blend_obj < best_obj - 1e-14:
            a_new = (1 - best_t) * best_a + best_t * g_val
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, top_m={top_m}, t={best_t:.2f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 5:
            rng_local = np.random.default_rng(it * 7919 + 3)
            noise = rng_local.normal(0, 0.004 * stall, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj(best_a)
            stall = 0

    return best_a


def golden_section_minimize(f, a, b, tol=1e-10, max_iter=60):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2


def coordinate_descent_golden(a, time_budget, max_sweeps=5, verbose=False):
    """Coordinate descent with golden section search per coordinate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        sweep_improved = False

        # Randomize coordinate order
        rng_local = np.random.default_rng(sweep * 1000 + 42)
        order = rng_local.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            ai_old = a[i]
            if ai_old < 1e-12:
                # Try small values
                a[i] = 0.01 / n
                obj_try = compute_obj(a)
                if obj_try < best_obj:
                    a[i] = 0.01 / n
                    s = float(np.sum(a))
                    a = a / s
                    best_obj = obj_try
                    best_a = a.copy()
                    sweep_improved = True
                else:
                    a[i] = 0.0
                continue

            # Golden section search for a[i] in [0, 3*ai_old]
            def objective_at(val):
                a[i] = val
                s_local = float(np.sum(a))
                if s_local < 1e-10:
                    return 1e10
                a_scaled = a / s_local
                return compute_obj(a_scaled)

            a[i] = ai_old  # restore
            upper = 3.0 * ai_old
            opt_val = golden_section_minimize(objective_at, 0.0, upper, tol=1e-12 * ai_old, max_iter=40)

            if abs(opt_val - ai_old) > 1e-12 * ai_old:
                a[i] = max(0, opt_val)
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    obj_new = compute_obj(a)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        sweep_improved = True

        if verbose:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        if not sweep_improved:
            break

    return best_a


def make_palindromic(a):
    """Symmetrize a sequence to be palindromic."""
    n = len(a)
    result = np.zeros(n)
    for i in range(n):
        result[i] = 0.5 * (a[i] + a[n - 1 - i])
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def diverse_init(n, mode):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + \
            np.exp(-((idx - 3 * n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 4:
        a = 1.0 / (0.03 + 5.0 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    elif mode == 6:
        a = (2 * x - 1) ** 6 + 0.1
    elif mode == 7:
        a = np.exp(-6 * (2 * x - 1) ** 2) + 0.3 * np.exp(-((2 * x - 1) / 0.2) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_n_value(best_seq, n_try, time_budget):
    """Try a different sequence length by interpolation + quick optimization."""
    n_old = len(best_seq)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, best_seq)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick LSE gradient descent
    t1 = min(20.0, time_budget * 0.4)
    a_try = lse_gradient_descent(a_try, time_budget=t1, T_start=10.0, T_end=80.0)

    # Quick LP
    t2 = min(25.0, time_budget * 0.35)
    if t2 > 5:
        a_try = lp_polish(a_try, time_budget=t2, top_m_start=20, top_m_end=10, max_iters=10)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method optimization for step function coefficients:
    LSE gradient + exact subgradient + LP polishing + golden-section CD + n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Initialization ===
    n_current = 800

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("Could not load parent, using defaults")

    # Try diverse initializations
    inits = []
    for mode in range(8):
        a_init = diverse_init(n_current, mode)
        a_pal = make_palindromic(a_init)
        inits.append((f"mode{mode}", a_init))
        inits.append((f"mode{mode}_pal", a_pal))

    inits.append(("uniform", np.ones(n_current) / n_current))

    for name, a_init in inits:
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{name}': obj={obj_init:.8f} **")

    print(f"\nBest initial: obj={best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: LSE Gradient Descent (main refinement) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 1: LSE Gradient Descent ===")
        alloc = min(120.0, remaining * 0.15)
        a_lse = lse_gradient_descent(best_seq, time_budget=alloc, T_start=10.0, T_end=100.0, verbose=True)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_lse:.8f}")

    # === Phase 2: Exact Subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 2: Exact Subgradient ===")
        alloc = min(80.0, remaining * 0.1)
        a_sg = exact_subgrad_descent(best_seq, time_budget=alloc, top_k=5, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_sg:.8f}")

    # === Phase 3: LP Polishing (main workhorse) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 3: LP Polishing ===")
        alloc = min(300.0, remaining * 0.3)
        a_lp = lp_polish(best_seq, time_budget=alloc, top_m_start=30, top_m_end=10,
                         max_iters=50, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_lp:.8f}")

    # === Phase 4: LSE GD Round 2 (higher T) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 4: LSE GD Round 2 ===")
        alloc = min(60.0, remaining * 0.1)
        a_lse2 = lse_gradient_descent(best_seq, time_budget=alloc, T_start=30.0, T_end=300.0)
        obj_lse2 = compute_obj(a_lse2)
        if obj_lse2 < best_score:
            best_score = obj_lse2
            best_seq = a_lse2.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj_lse2:.8f}")

    # === Phase 5: Coordinate Descent (golden section) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 5: Coordinate Descent (Golden Section) ===")
        alloc = min(120.0, remaining * 0.15)
        a_cd = coordinate_descent_golden(best_seq, time_budget=alloc, max_sweeps=4, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase5: obj={best_score:.8f} **")
        else:
            print(f"  Phase5: obj={obj_cd:.8f}")

    # === Phase 6: LP Round 2 with perturbation ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 6: LP Round 2 (perturbed) ===")
        noise = rng.normal(0, 0.002, len(best_seq))
        a_pert = best_seq + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(120.0, remaining * 0.15)
        a_lp2 = lp_polish(a_pert, time_budget=alloc, top_m_start=20, top_m_end=8,
                          max_iters=25, verbose=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase6: obj={best_score:.8f} **")
        else:
            print(f"  Phase6: obj={obj_lp2:.8f}")

    # === Phase 7: Palindromic refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 7: Palindromic Refinement ===")
        a_pal = make_palindromic(best_seq)
        alloc1 = min(20.0, remaining * 0.2)
        a_pal = lse_gradient_descent(a_pal, time_budget=alloc1, T_start=20.0, T_end=150.0)
        alloc2 = min(15.0, remaining * 0.15)
        if alloc2 > 5:
            a_pal = lp_polish(a_pal, time_budget=alloc2, top_m_start=15, top_m_end=8, max_iters=8)
        obj_pal = compute_obj(a_pal)
        if obj_pal < best_score:
            best_score = obj_pal
            best_seq = a_pal.copy()
            print(f"  Phase7: obj={best_score:.8f} **")
        else:
            print(f"  Phase7: obj={obj_pal:.8f}")

    # === Phase 8: n-value exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Phase 8: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(400, n_best - 100),
            max(400, n_best - 50),
            n_best + 50,
            n_best + 100,
            n_best + 200,
        ]))
        n_options = [n for n in n_options if 500 <= n <= 2000 and abs(n - n_best) >= 40]
        n_options = n_options[:4]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            a_try, obj_try = try_n_value(best_seq, n_try, remaining * 0.7)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 9: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 9: Final Polish ===")

        # LSE with very high T (near-exact)
        alloc1 = min(remaining * 0.25, 40.0)
        if alloc1 > 10:
            a_f1 = lse_gradient_descent(best_seq, time_budget=alloc1, T_start=50.0, T_end=500.0)
            obj_f1 = compute_obj(a_f1)
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_f1.copy()
                print(f"  Final LSE: obj={best_score:.8f} **")

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.3, 50.0)
        if alloc2 > 15:
            a_f2 = lp_polish(best_seq, time_budget=alloc2, top_m_start=15, top_m_end=5,
                             max_iters=20, verbose=False)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.3, 60.0)
        if alloc3 > 15:
            a_f3 = coordinate_descent_golden(best_seq, time_budget=alloc3, max_sweeps=2)
            obj_f3 = compute_obj(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        # Exact subgradient
        remaining = deadline - time.monotonic()
        alloc4 = min(remaining * 0.2, 30.0)
        if alloc4 > 10:
            a_f4 = exact_subgrad_descent(best_seq, time_budget=alloc4, top_k=3)
            obj_f4 = compute_obj(a_f4)
            if obj_f4 < best_score:
                best_score = obj_f4
                best_seq = a_f4.copy()
                print(f"  Final SubGrad: obj={best_score:.8f} **")

    # === Phase 10: Multi-restart if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 10: Multi-Restart ===")
        n_cur = len(best_seq)
        restart_configs = [
            (0.01, 5), (0.02, 7), (0.015, 3)
        ]
        for noise_scale, pal_mode in restart_configs:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            noise = rng.normal(0, noise_scale, n_cur)
            a_r = best_seq + noise
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            if pal_mode:
                a_r = make_palindromic(a_r)

            t1 = min(20.0, remaining * 0.3)
            a_r = lse_gradient_descent(a_r, time_budget=t1, T_start=20.0, T_end=200.0)
            t2 = min(20.0, remaining * 0.3)
            if t2 > 5:
                a_r = lp_polish(a_r, time_budget=t2, top_m_start=15, top_m_end=8, max_iters=8)
            t3 = min(15.0, remaining * 0.2)
            if t3 > 5:
                a_r = coordinate_descent_golden(a_r, time_budget=t3, max_sweeps=1)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart (noise={noise_scale}, pal={pal_mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart (noise={noise_scale}, pal={pal_mode}): obj={obj_r:.8f}")

    # === Final: one more CD pass ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final = coordinate_descent_golden(best_seq, time_budget=min(remaining - 5, 25.0), max_sweeps=1)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"\n  Last CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
