import numpy as np

import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective with a normalized to sum=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve_warm(a, prev_g=None, solver_list=None):
    """Solve LP with warm start. Returns (g, M, dual_vars)."""
    n = len(a)
    # Build constraint matrix A where row k, col j = a[j] if 0 <= k-j < n
    # (a*g)[k] = sum_j a[j] * g[k-j]
    # So row k has a[j] at position (k-j) for valid j
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_val = g_val / sg
                    # Try to get dual variables
                    duals = None
                    try:
                        duals = np.array([c.dual_value for c in constraints[:2*n-1]])
                    except:
                        duals = None
                    return g_val, float(M_var.value), duals
        except Exception:
            continue
    return None, None, None


def subgradient_at_active(a, active_lags):
    """Compute subgradient of max(conv(a,a)) at the given active lags."""
    n = len(a)
    grad = np.zeros(n)
    for lag in active_lags:
        for i in range(n):
            j = lag - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(active_lags)
    return grad


def project_simplex(v, n):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    # Use the standard algorithm
    if n == 1:
        return np.array([max(0, v[0]) / max(1e-30, abs(v[0]))])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0) / np.sum(np.maximum(a_new, 0))

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective with a assumed to have sum=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve(a, solver_list=None):
    """Solve LP: min M s.t. (a*g)[k]<=M, g>=0, sum(g)=1. Returns (g, M, duals)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_val = g_val / sg
                    duals = None
                    try:
                        duals = np.array([c.dual_value for c in constraints[:2 * n - 1]]).real
                        duals = np.where(np.isnan(duals), 0, duals)
                    except Exception:
                        duals = None
                    return g_val, float(M_var.value), duals
        except Exception:

    return None, None, None

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1)
    w = np.maximum(v - theta, 0)
    sw = float(np.sum(w))
    if sw < 1e-30:
        return np.ones(n) / n
    return w / sw

def log_space_polish(a, time_budget, verbose=False, max_iters=3000):
    """
    Polish in log-space using mirror descent.
    Parameterize a[i] = exp(b[i]) / sum(exp(b)).
    This keeps a automatically in the interior of the simplex.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)

    # Initialize log parameters
    b = np.log(a + 1e-30).copy()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(555)

    lr = 1.0 / np.sqrt(n)
    momentum = 0.9
    v = np.zeros(n)
    obj_history = []

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        # Compute softmax from b
        b_shifted = b - np.max(b)
        exp_b = np.exp(b_shifted)
        a_curr = exp_b / np.sum(exp_b)

        # Compute subgradient of max(conv(a,a))
        conv = compute_conv(a_curr)
        max_val = float(np.max(conv))
        active = np.where(conv >= max_val * 0.999)[0]
        grad = subgradient_at_active(a_curr, active)

        # Project gradient to tangent space of simplex (sum = 0)
        grad = grad - np.mean(grad)

        # Nesterov momentum step in log-space
        v = momentum * v - lr * grad
        b_new = b + v

        # Convert back and check
        a_new = exp_b = np.exp(b_new - np.max(b_new))
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        obj_history.append(obj_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        b = b_new

        # Adaptive learning rate
        if len(obj_history) > 100:
            recent = np.array(obj_history[-50:])
            prev = np.array(obj_history[-100:-50])
            if np.mean(recent) >= np.mean(prev):
                lr *= 0.9
            else:
                lr = min(lr * 1.01, 2.0 / np.sqrt(n))

        # Periodic re-centering to prevent drift
        if it % 200 == 0 and it > 0:
            b -= np.mean(b)

        # Occasionally inject noise to escape flat regions
        if it % 500 == 499:
            b += 0.01 * rng.standard_normal(n)

        if verbose and it % 200 == 0:
            print(f"    LogSpace it{it:4d}: obj={best_obj:.12f}, lr={lr:.6f}")

        # Reduce momentum over time for convergence
        if it > max_iters // 2:
            momentum = max(0.5, momentum - 0.0005)

    return best_a

def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

    # Then: run from perturbed starts
    for s in range(n_starts):
        remaining = time_budget - (time.monotonic() - time.monotonic())  # placeholder
        a_pert = best_a + 0.02 * (s + 1) * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(per_start_budget, 60.0)
        a_result = lp_fixed_point_accelerated(a_pert, time_budget=alloc, verbose=False, seed=100 + s)
        obj_result = compute_obj(a_result)
        if obj_result < best_obj:
            best_obj = obj_result
            best_a = a_result.copy()
            if verbose:
                print(f"    MultiStart [pert{s}]: obj={best_obj:.12f}")

    return best_a


def propose_candidate(budget_s=1500):
    """
    Main optimization: accelerated LP fixed-point with Anderson extrapolation,
    log-space mirror descent polishing, active-set surgery, and multi-n search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(42)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.12f}")
    except Exception as e:
        print(f"Error loading parent: {e}")
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.12f}")

    n_current = len(best_seq)
    print(f"Starting optimization with n={n_current}, obj={best_score:.12f}")

    # === Phase 1: Primary LP Fixed-Point (largest allocation) ===
    print("\n=== Phase 1: Accelerated LP Fixed-Point ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        lp_budget = min(remaining * 0.40, 550.0)
        a1 = lp_fixed_point_accelerated(best_seq, time_budget=lp_budget, verbose=True, seed=42)

        else:
            print(f"  Phase1: obj={obj1:.12f}")

    # === Phase 2: Log-space polish ===
    print("\n=== Phase 2: Log-Space Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.12, 120.0)
        a2 = log_space_polish(best_seq, time_budget=alloc, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.12f} **")
        else:
            print(f"  Phase2: obj={obj2:.12f}")

    # === Phase 3: Second LP round (different seed) ===
    print("\n=== Phase 3: Second LP Round ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        lp_budget2 = min(remaining * 0.22, 200.0)
        a3 = lp_fixed_point_accelerated(best_seq, time_budget=lp_budget2, verbose=True, seed=777)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.12f} **")
        else:
            print(f"  Phase3: obj={obj3:.12f}")

    # === Phase 4: Active set surgery ===
    print("\n=== Phase 4: Active-Set Surgery ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.10, 80.0)
        a4 = active_set_surgery(best_seq, time_budget=alloc, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.12f} **")
        else:
            print(f"  Phase4: obj={obj4:.12f}")

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 70:
        # Try n values close to current
        n_options = sorted(set([
            n_current - 100, n_current - 50, n_current + 50, n_current + 100,
            n_current + 150, n_current + 200,
            750, 850, 950, 1050, 1150
        ]))
        n_options = [n for n in n_options if 500 <= n <= 1500 and n != n_current]
        n_options.sort(key=lambda x: abs(x - n_current))
        n_options = n_options[:4]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_try = resample(best_seq, n_try)
            alloc = min(remaining * 0.18, 60.0)
            a_try = lp_fixed_point_accelerated(a_try, time_budget=alloc, verbose=False, seed=n_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.12f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.12f}")

    # === Phase 6: Final intensive LP ===
    print("\n=== Phase 6: Final Intensive LP ===")

            best_seq = a6.copy()
            print(f"  Phase6: obj={best_score:.12f} **")
        else:
            print(f"  Phase6: obj={obj6:.12f}")

    # === Phase 7: Final log-space polish ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 60.0)
        a7 = log_space_polish(best_seq, time_budget=alloc, verbose=False)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Phase7: obj={best_score:.12f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
            print(f"  (evaluate_sequence gave better: {val:.12f})")
    else:
        print("  WARNING: evaluate_sequence returned inf, falling back")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.12f}, time={elapsed:.1f}s ===")
    return result
