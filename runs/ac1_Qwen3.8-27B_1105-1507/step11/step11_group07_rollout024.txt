def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_minpeak(a, use_symmetry=False):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Optionally enforce palindromic symmetry on g.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build conv matrix: (2n-1) x n
    # conv[a_norm, g][k] = sum_j a_norm[j] * g[k-j]
    # For efficiency, we build the matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a_norm[j] * g[k-j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        start = j
        end = j + n
        A[start:end, j] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    # Add symmetry constraints if requested
    if use_symmetry:
        for i in range(n // 2):
            constraints.append(g[i] == g[n-1-i])
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix (2n-1) x n such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, use_symmetry=True):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if use_symmetry:
        for i in range(n // 2):
            constraints.append(g[i] == g[n - 1 - i])
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Enforce symmetry
                    if use_symmetry:
                        for i in range(n // 2):
                            avg = (g_val[i] + g_val[n-1-i]) / 2
                            g_val[i] = avg
                            g_val[n-1-i] = avg
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, max_t=1.5, n_coarse=8, n_fine=10):
    """Find optimal blend t: a_new = (1-t)*a + t*g, project to simplex."""
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Coarse search
    for i in range(n_coarse):
        t = max_t * (i + 0.5) / n_coarse
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-6:
            continue
        obj_try = compute_obj(a_try / s)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    
    # Fine search around best
    if best_t > 0:
        lo = max(0.0, best_t - max_t / n_coarse)
        hi = min(max_t, best_t + max_t / n_coarse)
        for _ in range(n_fine):
            mid = (lo + hi) / 2
            a_try = (1 - mid) * a + mid * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-6:
                break
            obj_try = compute_obj(a_try / s)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = mid
                hi = mid
            else:
                lo = mid
    
    return best_t, best_obj


def lp_polish(a, time_budget, max_iters=100, use_symmetry=True, verbose=False, seed=0):
    """
    Aggressive LP polishing: iteratively solve LP and blend.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        g_lp, M_lp = solve_lp(a, use_symmetry=use_symmetry)
        if g_lp is None:
            break
        
        t_opt, obj_opt = line_search_blend(a, g_lp, max_t=1.0)
        
        improved = False
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s = float(np.sum(a_new))
            if s > 1e-6:
                a_new = a_new / s
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                # Perturb and restart
                noise_scale = 0.005 * min(stall, 5)
                noise = rng.normal(0, noise_scale, n)
                a = project_simplex(best_a + noise)
                stall = 0
        
        it += 1
        if verbose and (it % 3 == 0 or it < 5):
            gap = float(np.max(compute_conv(a))) - M_lp
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"gap={gap:.6f}, elapsed={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj


def soft_max_gd(a, time_budget, beta_start=10.0, beta_end=50.0,
                lr_start=0.006, momentum=0.9, verbose=False, seed=42):
    """Soft-max gradient descent with momentum."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.4 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via cross-correlation
        n_p = len(p)
        a_rev = a[::-1]
        size = 1
        while size < n + n_p - 1:
            size <<= 1
        A = rfft(a_rev, size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                v = np.zeros(n)
                stall = 0
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def make_init(n, mode=0):
    """Create various initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # U-shaped
        a = 1.0 / (0.01 + 10.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        # Arcsine
        a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2 + 1e-6)
    elif mode == 2:
        # Center-heavy (bell)
        a = np.exp(-8.0 * (2 * x - 1) ** 2)
    elif mode == 3:
        # Two peaks
        a = np.exp(-20.0 * (x - 0.2) ** 2) + np.exp(-20.0 * (x - 0.8) ** 2) + 0.01
    elif mode == 4:
        # Flat with noise
        rng = np.random.default_rng(42 + mode)
        a = np.ones(n) + rng.normal(0, 0.2, n)
        a = np.maximum(a, 0)
    elif mode == 5:
        # Cosine bump
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 0.5 + 0.01
    elif mode == 6:
        # Power law at edges
        a = x ** 0.5 + (1 - x) ** 0.5
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
