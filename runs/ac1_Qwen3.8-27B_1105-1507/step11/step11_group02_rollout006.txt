"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses fast coordinate descent with incremental convolution updates,
LP-based direction finding, and multiplicative weight updates.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective with sum(a)=1 assumption."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    return 2.0 * n * float(np.max(compute_conv(a_n)))


def eval_1d(x, v, S_other, n, c_lin, a_ki_lin, c_2i):
    """Evaluate 1D objective for coordinate descent at point x."""
    if c_lin.size > 0:
        max_lin = float(np.max(c_lin + 2.0 * a_ki_lin * (x - v)))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def fast_cd(a, time_budget, seed=42, verbose=False):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = 0.38196601125

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)

        for i in order:
            v = a[i]
            S_other = s - v
            if S_other < 1e-12:
                continue

            # Determine which convolution entries are affected
            k_min = i  # minimum k such that k-i >= 0 and a[k-i] exists
            k_max = min(2 * n - 2, i + n - 1)  # maximum k

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]  # values a[k-i] for k in ks

            # Separate the quadratic term (k = 2i) from linear terms
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            mask = (ks != k_quad)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_lin = conv[ks_lin].copy()

            # Quick candidates
            scale = max(v, 1.0 / n, 1e-6)
            candidates = np.array([0.0, v * 0.5, v * 0.7, v, v * 1.3, v * 1.5, 1.0/n, 2.0/n])
            candidates = np.maximum(candidates, 0.0)

            best_x = v
            best_f = 2.0 * n * float(np.max(conv)) / (s * s)

            for x in candidates:
                fx = eval_1d(x, v, S_other, n, c_lin, a_ki_lin, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.5
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-10

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_1d(c_pt, v, S_other, n, c_lin, a_ki_lin, c_2i)
            fd = eval_1d(d_pt, v, S_other, n, c_lin, a_ki_lin, c_2i)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_1d(c_pt, v, S_other, n, c_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_1d(d_pt, v, S_other, n, c_lin, a_ki_lin, c_2i)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            # Update convolution
            delta = x_opt - v
            a[i] = x_opt
            s += delta

            # Update affected conv entries
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if c_lin.size > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            # Check objective
            if s > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (s * s)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp(a, time_budget=20.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None

def lp_blend(a, time_budget, seed=42):
    """LP-based direction finding with blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a, time_budget=min(remaining * 0.4, 15.0))
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.02 * min(stall, 5), n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        a = np.exp(-((np.arange(n) - n//2) ** 2) / (2 * (n//4) ** 2))
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(0.01, 1, n)
        return (x ** rng.uniform(-3, 3)) / (x ** rng.uniform(-3, 3)).sum()
    else:
        a = 1.0 - np.abs(2 * np.linspace(0, 1, n) - 1)
        a += 0.1 * rng.uniform(0, 1, n)
        return a / a.sum()


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
