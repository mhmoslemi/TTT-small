"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating minimization, subgradient descent, CD with line search,
and multi-restart exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_with(a, g):
    """Linear convolution of a and g via FFT. Returns array of length len(a)+len(g)-1."""
    na, ng = len(a), len(g)
    size = 1
    while size < na + ng - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:na + ng - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None, verbose=False):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
            M.value = compute_obj_norm(a)  # rough initial bound
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS', 'PDLP']:
        try:
            prob.solve(solver=getattr(cp, solver), verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""

    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            best_trial_obj = 2.0 * n * float(np.max(conv))
            best_trial_val = a_j
            
            for val in vals:
                if val < 0:
                    continue
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new < 1e-10:
                    continue
                
                idx = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (idx == 2 * j)
                if np.any(~mask):
                    affected[~mask] += 2.0 * delta * a[idx[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)
                
                mc = max(max_unaffected, float(np.max(affected)))
                obj_t = 2.0 * n * mc / (S_new * S_new)
                
                if obj_t < best_trial_obj - 1e-14:
                    best_trial_obj = obj_t
                    best_trial_val = val
            
            if best_trial_val != a_j and best_trial_obj < 2.0 * n * float(np.max(conv)) - 1e-13:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta
                idx = np.arange(k_start, k_end + 1)
                mask = (idx == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[idx[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val ** 2 - a_j ** 2)
                
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 6:
                break
        
        if sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, t={elapsed:.1f}s")
    
    return best_a
