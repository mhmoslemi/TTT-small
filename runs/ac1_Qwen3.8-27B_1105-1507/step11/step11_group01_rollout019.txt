"""
Optimizes step function coefficients using accelerated LP best-response iterations
with exact line search, coordinate descent polishing, and systematic n exploration.
Key: dense LP iteration loop with warm-starting, FFT-based fast convolution,
and adaptive time allocation.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_from_conv(conv, n):
    """Compute objective 2*n*max(conv) assuming sum(a)=1."""
    return 2.0 * n * float(np.max(conv))


def evaluate_normalized(a):
    """Compute objective for a normalized sequence (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """
    Build the matrix A (2n-1 x n) such that A @ g = conv(a, g).
    Uses the structure that A[k, i] = a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1 (clipped to 2n-2)
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - j]
    return A


def solve_lp(a, solver_pref=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    if solver_pref:
        solvers = [solver_pref] + [s for s in solvers if s != solver_pref]
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val, M.value
        except Exception:
            continue
    return None, None


def coordinate_descent(a, time_budget=60.0, seed=42):
    """
    Fast coordinate descent with FFT-based convolution update.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            # Valid k range for coordinate i
            k_min = max(0, i)
            k_max = min(2 * n - 2, 2 * i)  # Actually k goes from i to i+n-1
            k_max = min(2 * n - 2, i + n - 1)
            
            if k_min > k_max:
                continue

            # For coordinate i, changing a[i] from v to x affects:
            # conv[k] for k=i,...,i+n-1: conv[k] changes by 2*(x-v)*a[k-i] (linear part)
            # Plus conv[2i] gets x^2 - v^2 (quadratic part, if 2i is in range)
            # More precisely:
            # conv[k] = sum_j a[j]*a[k-j]
            # If we change a[i] to x:
            #   conv[k] new = conv[k] - 2*v*a[k-i] + 2*x*a[k-i] + (x^2 - v^2) if k=2i
            #   = conv[k] + 2*(x-v)*a[k-i] + (x^2 - v^2)*[k==2i]
            # But we need to be careful: a[k-i] itself might be a[i] when k-i=i, i.e., k=2i.
            # So for k=2i: conv[2i] new = conv[2i] - v^2 + x^2 (the self-term)
            #   plus the cross terms: 2*(x-v)*a[k-i] for k != 2i already includes a[i] when k-i=i.
            # Let me redo this:
            # conv[k] = sum_j a[j] * a[k-j]
            # The term involving a[i] is: a[i]*a[k-i] (when j=i) + a[k-i]*a[i] (when j=k-i, if k-i != i)
            # If k-i != i (i.e., k != 2i): the contribution is 2*a[i]*a[k-i], so changing a[i] to x changes conv[k] by 2*(x-v)*a[k-i].
            # If k-i == i (i.e., k == 2i): the contribution is a[i]^2 = v^2, changing to x^2. So conv[2i] changes by x^2 - v^2.
            
            # Affected conv entries: k from i to i+n-1 (where a[k-i] is defined)
            # But we also need k-i to be in [0, n-1], so k in [i, i+n-1].
            # And k must be in [0, 2n-2].
            
            aff_k = np.arange(max(0, i), min(2*n-1, i+n))
            # For each affected k, the "other" index is k-i
            other = aff_k - i  # These are in [0, n-1]
            
            # Linear coefficients: 2*a[other] for k != 2i, and for k == 2i it's different
            lin_coeff = 2.0 * a[other].copy()
            quad_k = 2 * i
            if quad_k in aff_k:
                idx_quad = np.where(aff_k == quad_k)[0][0]
                lin_coeff[idx_quad] = 0.0  # The quadratic term handles this
            
            # The 1D function to minimize:
            # f(x) = max( max_{k != 2i} (conv[k] + 2*(x-v)*a[k-i]), conv[2i] + x^2 - v^2 )
            #      all divided by (S + x - v)^2 * (1/(2n)) ... actually we want to minimize 2n*max/(S+x-v)^2
            
            S_other = S - v
            
            def f_1d(x):
                if len(aff_k) > 0:
                    if quad_k in aff_k:
                        idx_quad = np.where(aff_k == quad_k)[0][0]
                        lin_vals = conv[aff_k] + lin_coeff * (x - v)
                        lin_vals[idx_quad] = conv[quad_k] + x*x - v*v
                    else:
                        lin_vals = conv[aff_k] + lin_coeff * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                if quad_k >= 0 and quad_k <= 2*n-2 and quad_k not in aff_k:
                    quad_val = conv[quad_k] + x*x - v*v
                    max_val = max(max_lin, quad_val)
                elif quad_k in aff_k:
                    max_val = max_lin  # Already included
                else:
                    max_val = max_lin
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Quick candidates
            scale = max(v, S / n, 1e-8)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0, S / (2*n), 0]
            candidates = [max(0.0, c) for c in candidates]
            
            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2
            if x_hi <= x_lo:
                x_hi = x_lo + 1e-6

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            # Update conv
            if len(aff_k) > 0:
                if quad_k in aff_k:
                    idx_quad = np.where(aff_k == quad_k)[0][0]
                    conv[aff_k] += lin_coeff * delta
                    conv[quad_k] += x_opt * x_opt - v * v
                else:
                    conv[aff_k] += lin_coeff * delta
                if quad_k >= 0 and quad_k <= 2*n-2 and quad_k not in aff_k:
                    conv[quad_k] += x_opt * x_opt - v * v

            S += delta
            improved = True

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if sweep % 3 == 0:
            S_now = float(np.sum(a))
            if S_now > 1e-10:
                conv = compute_conv(a)
                obj = 2.0 * n * float(np.max(conv)) / (S_now * S_now)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                a = a / S_now

    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
