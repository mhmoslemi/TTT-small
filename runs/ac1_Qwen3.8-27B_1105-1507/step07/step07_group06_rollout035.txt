Key improvements over previous: better tau scheduling, peak-directed perturbation,
structured initializations targeting asymmetric flat-convolution shapes, and
more efficient time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def softmax_from_theta(theta):
    """Compute softmax with numerical stability."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def smoothed_obj_and_grad(theta, tau, n):
    """Compute smoothed objective and gradient w.r.t. log-parameters theta."""
    a = softmax_from_theta(theta)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    convex_val = np.sum(np.exp((conv - conv_max) * tau))
    smooth_max = conv_max + np.log(convex_val) / tau
    obj_val = 2.0 * n * smooth_max

    w = tau * np.exp((conv - conv_max) * tau)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        return obj_val, np.zeros(n)
    w = w / w_sum

    c = np.zeros(n)
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        if end > 0:
            c[j] = np.dot(w[j:j + end], a[:end])

    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta


def smooth_lbfgs(a_init, tau_schedule, time_budget, max_iter_per_stage=200):
    """Smoothed L-BFGS-B with multiple tau stages."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-12)
    s = np.sum(a_init)
    if s < 1e-10:
        return a_init, compute_obj(a_init)
    a_init = a_init / s
    theta = np.log(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    for tau in tau_schedule:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        def obj_func(th):
            val, _ = smoothed_obj_and_grad(th, tau, n)
            return val

        def grad_func(th):
            _, g = smoothed_obj_and_grad(th, tau, n)
            return g

        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                              options={'maxiter': max_iter_per_stage, 'gtol': 1e-7,
                                       'maxfun': int(min(800, remaining * 50))})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    theta = result.x.copy()
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=30.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    start = time.monotonic()

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        if time.monotonic() - start > time_limit:
            break
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver, time_limit=remaining := max(5, time_limit - (time.monotonic() - start)))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternating(a, time_budget=60.0, max_iters=100):
    """LP alternating optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a, time_limit=min(20, time_budget - (time.monotonic() - start)))
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            if it % 5 == 0 and it > 0:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
            elif it > 10:
                break

    return best_a


def peak_directed_refinement(a, time_budget=30.0, n_steps=50):
    """
    Directly target the peak: reduce contributions to the peak lag
    by decreasing mass at "hot" positions and increasing at "cold" positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Use top-k lags for a "soft" peak direction
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[peak_indices]
        peak_weights = (peak_vals - np.min(peak_vals) + 1e-6)
        peak_weights = peak_weights / np.sum(peak_weights)

        # Compute pressure on each variable
        pressure = np.zeros(n)
        for k_idx, w_k in zip(peak_indices, peak_weights):
            k = int(k_idx)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w_k * a[j]

        # Normalize pressure
        p_max = np.max(pressure)
        if p_max < 1e-15:
            break
        pressure = pressure / p_max

        # Move mass from high-pressure to low-pressure positions
        mean_p = np.mean(pressure)
        # Decrease where pressure > mean, increase where pressure < mean
        delta = step_size * (pressure - mean_p)
        a_new = a + delta
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            step_size *= 0.7
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size * 1.05, 0.2)
        else:
            step_size *= 0.85

    return best_a


def generate_initialization(n, init_type, rng):
    """Generate diverse initial sequences."""
    if init_type == 0:
        # Linear ramp (asymmetric)
        a = np.linspace(0.1, 1.0, n)
    elif init_type == 1:
        # Exponential ramp
        rate = rng.uniform(1.0, 4.0)
        a = np.exp(np.linspace(0, rate, n))
    elif init_type == 2:
        # Off-center Gaussian (asymmetric center)
        center = rng.uniform(0.3, 0.7) * n
        width = rng.uniform(0.1, 0.3) * n
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
        # Add asymmetry
        a *= np.linspace(0.5, 1.5, n) ** rng.uniform(0, 1)
    elif init_type == 3:
        # Beta-like distribution (skewed)
        alpha = rng.uniform(1.0, 3.0)
        beta = rng.uniform(1.0, 3.0)
        x = np.linspace(0.001, 0.999, n)
        a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    elif init_type == 4:
        # Two-hump asymmetric
        c1 = rng.uniform(0.2, 0.4) * n
        c2 = rng.uniform(0.6, 0.8) * n
        w1 = rng.uniform(0.05, 0.15) * n
        w2 = rng.uniform(0.05, 0.15) * n
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - c1) / w1) ** 2) + rng.uniform(0.3, 0.7) * np.exp(-0.5 * ((x - c2) / w2) ** 2)
    elif init_type == 5:
        # Power law (heavy-tailed)
        p = rng.uniform(0.5, 2.0)
        x = np.arange(1, n + 1)
        a = x ** (-p)
        a = a + rng.uniform(0.1, 0.5)
    elif init_type == 6:
        # Sine-based asymmetric
        freq = rng.uniform(1.0, 3.0)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
    elif init_type == 7:
        # Plateau with sharp edges
        frac_lo = rng.uniform(0.1, 0.3)
        frac_hi = rng.uniform(0.7, 0.9)
        a = np.ones(n)
        a[:int(frac_lo * n)] = rng.uniform(0.1, 0.5)
        a[int(frac_hi * n):] = rng.uniform(0.1, 0.5)
        # Tilt the plateau
        a[int(frac_lo * n):int(frac_hi * n)] *= np.linspace(
            rng.uniform(0.7, 1.3), rng.uniform(0.7, 1.3),
            int(frac_hi * n) - int(frac_lo * n))
    else:
        # Random smooth (Fourier)
        n_freqs = rng.integers(3, 10)
        coeffs = rng.normal(0, 1, n_freqs + 1)
        coeffs[0] = np.max(np.abs(coeffs))  # Ensure positive mean
        x = np.arange(n)
        a = np.zeros(n)
        for k in range(n_freqs + 1):
            a += coeffs[k] * np.cos(2 * np.pi * k * x / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def proposed_get_best(a_in, tau_sched, t_budget):
    """Run smoothed L-BFGS on a given initialization."""
    return smooth_lbfgs(a_in, tau_sched, t_budget)

                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # ===== Phase 1: Multi-start smoothed optimization from parent =====
    print(f"\n=== Phase 1: Refine parent with multi-stage smoothed LBFGS ===")
    tau_sched_1 = [20.0, 50.0, 100.0, 200.0, 400.0, 800.0]
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 180.0)
    a_opt, obj_opt = proposed_get_best(best_seq.copy(), tau_sched_1, alloc)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  Parent refined: obj={best_score:.8f} **")
    else:
        print(f"  Parent refined: obj={obj_opt:.8f}")

    # ===== Phase 2: Diverse initializations =====
    print(f"\n=== Phase 2: Diverse Initializations ===")
    n_init = n_parent
    n_inits = 12
    for i in range(n_inits):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if i < 4:
            n_try = n_init
            init_type = i % 8
        elif i < 8:
            n_try = n_init
            init_type = (i + 3) % 8
        else:
            n_try = n_init

        a_init = generate_initialization(n_try, init_type, rng)

        # Blend with parent for some inits to guide the search
        if i % 3 == 0:
            blend = 0.5
            a_init = blend * best_seq + (1 - blend) * a_init
            a_init = a_init / np.sum(a_init)

        tau_sched = [30.0, 80.0, 150.0, 300.0, 600.0]
        alloc = min(remaining * 0.12, 80.0)
        a_opt, obj_opt = proposed_get_best(a_init, tau_sched, alloc)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_init = len(best_seq)
            print(f"  Init {i} (type={init_type}, n={n_try}): obj={best_score:.8f} **")
        else:
            print(f"  Init {i} (type={init_type}, n={n_try}): obj={obj_opt:.8f}")

    # ===== Phase 3: Peak-directed refinement =====
    print(f"\n=== Phase 3: Peak-Directed Refinement ===")
    remaining = deadline - time.monotonic()

        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # ===== Phase 5: Multi-n exploration =====
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = sorted(set([max(400, n_best - 100), n_best - 50, n_best,
                            n_best + 50, n_best + 100, n_best + 200]))
    n_options = [n for n in n_options if 400 <= n <= 1500]

        remaining = deadline - time.monotonic()
        if remaining < 30:

        if n_try == n_best:

        else:
            a_try = resample_sequence(best_seq, n_try)

    remaining = deadline - time.monotonic()

    # 6a: High-resolution smoothed optimization
    if remaining > 20:
        tau_sched_fine = [100.0, 200.0, 400.0, 800.0, 1500.0]
        alloc = min(remaining * 0.3, 80.0)
        a_opt, obj_opt = proposed_get_best(best_seq.copy(), tau_sched_fine, alloc)

    if remaining > 15:

    return result
