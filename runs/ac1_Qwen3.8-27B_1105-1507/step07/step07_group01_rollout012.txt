def compute_conv(a):
    n = len(a)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    # A[i,j] = a[i-j] if 0 <= i-j < n, else 0
    # conv(a,g)[i] = sum_j a[i-j] * g[j] (for valid i-j)
    # So A[i,j] = a[i-j] if 0 <= i-j < n
    A = np.zeros((2 * n - 1, n))

        i_start = j
        i_end = min(j + n, 2 * n - 1)
        A[i_start:i_end, j] = a[:i_end - i_start]
    return A

    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub @ x <= b_ub
    # A@g - M <= 0 for each row
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

        # Strategy: adjust phases of A to reduce peak of IFFT(|A|^2)
        # But we need to maintain that IFFT(A * conjugate(A)) = autocorrelation
        
        # Alternative: SLSQP-like approach
        # Modify A to reduce the peak of |A|^2's inverse transform
        # i.e., reduce the peak of the power spectrum
        
        # Simple approach: compress the dynamic range of |A|^2
        power = np.abs(A) ** 2
        max_power = np.max(power)
        min_power = np.min(power[power > 0]) if np.any(power > 0) else 0
        
        # Compress
        alpha = 0.1  # compression amount
        target = max_power ** (1 - alpha) * min_power ** (alpha)
        # Scale to keep DC fixed (sum(a) = 1 means A[0] = 1)
        A_new = A * np.ones(size // 2 + 1)
        for k in range(1, len(A)):
            if power[k] > 0:
                scale = (target / power[k]) ** alpha
                A_new[k] = A[k] * scale
        A_new[0] = A[0]  # Keep DC
        
        # Back to time domain
        a_new = irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Reduce compression
            alpha = max(0.01, alpha * 0.995)
        else:
            alpha = max(0.01, alpha * 0.9)
            if alpha < 0.01:
                break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
