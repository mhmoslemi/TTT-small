        a = np.maximum(a, 0)
        result = [float(x) for x in a]
        score = evaluate_sequence(result)
        print(f"Perturbed uniform n={n}: score={score:.10f}")
        if score < best_score:
            best_score = score
            best_seq = result.copy()

    #Triangular
    n = 800
    a = np.abs(1 - 2 * np.arange(n) / (n - 1))
    result = [float(x) for x in a]
    score = evaluate_sequence(result)
    print(f"Triangular n={n}: score={score:.10f}")
    if score < best_score:
        best_score = score
        best_seq = result.copy()

    # Bimodal
    n = 800
    a = np.zeros(n)
    a[:n//4] = 1.0
    a[3*n//4:] = 1.0
    result = [float(x) for x in a]
    score = evaluate_sequence(result)
    print(f"Bimodal n={n}: score={score:.10f}")
    if score < best_score:
        best_score = score
        best_seq = result.copy()

    # Cosine window
    n = 800
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x)) + 0.5
    result = [float(x) for x in a]
    score = evaluate_sequence(result)
    print(f"Cosine window n={n}: score={score:.10f}")
    if score < best_score:
        best_score = score
        best_seq = result.copy()

    # Try parent sequence if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        parent = np.minimum(parent, 1000.0)
        result = [float(x) for x in parent]
        score = evaluate_sequence(result)
        print(f"Parent (n={len(parent)}): score={score:.10f}")
        if score < best_score:
            best_score = score
            best_seq = result.copy()
    except Exception as e:
        print(f"Could not access parent: {e}")

    print(f"\nBest score: {best_score:.10f}, n={len(best_seq)}")
    print(f"Time elapsed: {time.monotonic() - start_time:.1f}s")

    return best_seq
