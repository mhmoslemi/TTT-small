"""
Minimize 2*n*max(conv(a,a))/(sum(a))² for non-negative sequences.
Uses LP alternating optimization, spectral peak-shaving, coordinate descent,
and multi-start exploration with diverse initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))²."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a, timeout=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    if n > 5000:
        return None, None
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={"time_limit": timeout})
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:

    except Exception:
        pass
    return None, None

def spectral_peak_shave(a, time_budget=120.0, verbose=False):
    """
    Spectral method: iteratively reduce the peak of the convolution by
    modifying the magnitude spectrum at frequencies contributing to the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Determine FFT size
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    max_iters = 200
    beta = 0.3

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute spectrum
        A = fft(a, N)
        # Compute convolution
        conv = ifft(A * A, N)[:2 * n - 1].real
        k_star = int(np.argmax(conv))
        peak_val = float(conv[k_star])

        # Compute peak contribution from each frequency
        # conv[k*] = (1/N) * sum_f A[f]² * exp(2πifk*/N)
        # Real contribution: (1/N) * |A[f]|² * cos(2*angle(A[f]) + 2πf*k*/N)
        freq = np.arange(N)
        phase_contrib = 2 * np.angle(A) + 2 * np.pi * freq * k_star / N
        real_contrib = (np.abs(A) ** 2) * np.cos(phase_contrib)

        # Only reduce frequencies that positively contribute to the peak
        max_contrib = np.max(np.maximum(real_contrib, 0))
        if max_contrib < 1e-15:
            break

        # Apply spectral reduction
        reduction = np.ones(N)
        positive_mask = real_contrib > 0
        reduction[positive_mask] = np.exp(-beta * real_contrib[positive_mask] / max_contrib)

        # Preserve DC component
        reduction[0] = 1.0

        A_new = A * reduction
        a_new = np.real(ifft(A_new, N)[:n])
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            beta *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            beta = min(beta * 1.1, 2.0)
            if verbose and it % 20 == 0:
                print(f"    Spectral it{it}: obj={best_obj:.8f}, peak={peak_val:.6f}, beta={beta:.3f}")
        else:
            beta *= 0.7
            if beta < 0.01:
                # Try a different approach: flatten the spectrum
                A = fft(a, N)
                mag = np.abs(A)
                # Compress dynamic range
                mag_new = mag ** 0.7  # power compression
                # Preserve DC
                mag_new[0] = mag[0]
                A_new = mag_new / mag * A
                a_new = np.real(ifft(A_new, N)[:n])
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        beta = 0.3
                    else:
                        a = best_a.copy()
                        # Try additive approach
                        rng = np.random.default_rng(it + 42)
                        a_pert = best_a * (1 + 0.05 * rng.standard_normal(n))
                        a_pert = np.maximum(a_pert, 0)
                        sp = float(np.sum(a_pert))
                        if sp > 1e-10:
                            a_pert = a_pert / sp
                            op = compute_obj(a_pert)
                            if op < best_obj:
                                best_obj = op
                                best_a = a_pert.copy()
                                a = a_pert
                        beta = 0.3

    return best_a

        conv = compute_conv(a)
        # Find all positions within 5% of max (active set)
        peak_val = float(np.max(conv))
        active_threshold = peak_val * 0.95
        active_set = np.where(conv >= active_threshold)[0]

        # Compute subgradient: average gradient over active set
        grad = np.zeros(n)
        for k in active_set:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(active_set)

        # Project gradient (remove mean to preserve sum)
        grad_centered = grad - np.mean(grad)

        # Momentum update
        momentum = mu * momentum + (1 - mu) * grad_centered

        # Update
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 1.0)
            if verbose and it % 30 == 0:
                print(f"    Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.4f}")
        else:
            lr *= 0.85
            if lr < 1e-5:

                    op = compute_obj(a_pert)
                    if op < best_obj:
                        best_obj = op
                        best_a = a_pert.copy()
                    a = best_a.copy()
                lr = 0.1

    return best_a


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, (a*a)[k] is linear in a[i] (except at k=2i where it's quadratic).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()

    for epoch in range(200):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for i in range(n):
            # Optimize a[i] with all others fixed
            a_others = a.copy()
            a_others[i] = 0
            s_others = float(np.sum(a_others))

            # Try different values of a[i]
            # The objective as a function of x = a[i]:
            #   f(x) = 2n * max_k (conv_k(x)) / (s_others + x)²
            # where conv_k(x) = C_k + d_k * x + (x² if k==2i else 0)
            # C_k = (a_others * a_others)[k]
            # d_k = 2 * a_others[k-i] (if valid) [accounts for both j=i and j=k-i terms]

            conv_base = compute_conv(a_others)
            # d_k: coefficient of x in conv_k when a[i] = x
            d = np.zeros(2 * n - 1)
            j = k = None
            for k_idx in range(2 * n - 1):
                j_idx = k_idx - i
                if 0 <= j_idx < n:
                    d[k_idx] = 2.0 * a_others[j_idx]  # two cross terms (i to k-i and k-i to i)
                    if j_idx == i:
                        d[k_idx] = 2.0 * a_others[j_idx]  # same, a[i]*a[i] contributes 2x in linear approx
                        # Actually for k=2i: conv_k = C_k + 2*a[i]*a_others[i] + a[i]^2
                        # But a_others[i] = 0, so conv_k = C_k + a[i]^2
                        d[k_idx] = 0.0  # no linear term, it's purely quadratic

            # For k = 2i: conv[k] = conv_base[k] + x^2
            # For other k: conv[k] = conv_base[k] + d[k] * x

            quad_at_2i = (2 * i < 2 * n - 1)

            # Try a grid of x values
            current_x = a[i]
            # Range: x from 0 to some max (limited by s_others to keep sum reasonable)
            x_max = max(3.0 * current_x if current_x > 0 else 2.0 * s_others / n, 5.0 * s_others / n)

            # Grid search with refinement
            x_candidates = np.concatenate([
                np.array([0.0]),
                np.linspace(0, x_max, 50),
                np.array([current_x * 0.5, current_x * 0.8, current_x * 1.2, current_x * 1.5])
            ])
            x_candidates = np.clip(x_candidates, 0, x_max)

            best_x = current_x
            best_f = np.inf

            for x in x_candidates:
                a_try = a_others.copy()
                a_try[i] = x
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_norm = a_try / s_try
                conv_try = compute_conv(a_norm)
                f_val = 2.0 * n * float(np.max(conv_try))
                if f_val < best_f:
                    best_f = f_val
                    best_x = x

            if best_x != current_x:
                a[i] = best_x
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    obj_now = compute_obj(a)
                    if obj_now < best_obj - 1e-14:
                        best_obj = obj_now
                        best_a = a.copy()
                        improved = True

        if not improved and epoch > 5:
            break

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smooth_sequence(a, sigma=3.0):
    """Apply Gaussian smoothing to the sequence."""
    n = len(a)
    kernel_size = int(6 * sigma) | 1
    x = np.arange(-kernel_size // 2, kernel_size // 2 + 1)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    a_smooth = np.convolve(a, kernel, mode='same')
    return np.maximum(a_smooth, 0)

Minimize 2*n*max(conv(a,a))/(sum(a))² for non-negative sequences.

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))²."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a, timeout=10.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)

        candidate_ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        if 0.01 < last_t < 0.99:
            candidate_ts = sorted(set(candidate_ts + [max(0.01, last_t * 0.5), min(1.0, last_t * 1.5)]))

        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall += 1
            if stall > 15:
                rng_p = np.random.default_rng((it * 7919 + n) % (2**32))
                scale = 0.01 + 0.003 * min(stall // 15, 10)
                a_pert = best_a * np.exp(rng_p.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                    a = best_a.copy()
                stall = 0

    return best_a
