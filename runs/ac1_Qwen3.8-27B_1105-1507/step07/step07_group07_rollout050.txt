from scipy.fft import rfft, irfft

    while p < m:
        p <<= 1
    return p


def compute_conv(a):

    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

    return 2.0 * n * float(np.max(conv))

    # Wait, conv is computed from a_norm = a/s, so:
    # conv[k] = sum_i (a/s)[i] * (a/s)[k-i]
    # d(conv[k])/dx_i = (d(a_norm[i])/dx_i) * a_norm[k-i] + a_norm[i] * d(a_norm[k-i])/dx_i
    # d(a_norm[i])/dx_i = (da_i/dx_i)/s - a_i/s * ds/dx_i = a_i/s - a_i/s * a_i/s ... 
    # This is getting complicated. Let me use the chain rule more carefully.
    
    # a = exp(x), s = sum(a) = sum(exp(x))
    # a_norm = a/s
    # conv[k] = sum_i a_norm[i] * a_norm[k-i]
    # 
    # d(conv[k])/dx_i = sum_j [d(a_norm[j])/dx_i * a_norm[k-j] + a_norm[j] * d(a_norm[k-j])/dx_i]
    # = sum_j [delta(j,i)*a_i/s - a_norm*j*a_i/s) * a_norm[k-j] + a_norm[j] * (delta(j,k-i)*a_i/s - a_norm[j]*a_i/s)]
    # = a_i/s * [a_norm[k-i] - sum_j a_norm[j]*a_norm[k-j]] + a_i/s * [a_norm[i] - sum_j a_norm[j]*a_norm[j]]
    # Wait, this isn't right either. Let me be more careful.
    #
    # a_norm[j] = a[j]/s = exp(x_j)/s
    # d(a_norm[j])/dx_i = [delta(i,j)*exp(x_i)*s - exp(x_j)*exp(x_i)]/s^2
    #                     = a_i/s * [delta(i,j) - a_norm[j]]
    #
    # So d(conv[k])/dx_i = sum_j [a_i/s*(delta(i,j) - a_norm[j]) * a_norm[k-j] + a_norm[j] * a_i/s*(delta(i,k-i) - a_norm[k-j])]
    # = a_i/s * [sum_j (delta(i,j) - a_norm[j]) * a_norm[k-j] + sum_j a_norm[j] * (delta(i,k-i) - a_norm[k-j])]
    # = a_i/s * [a_norm[k-i] - conv[k] + a_norm[i] - conv[k]]
    # = a_i/s * [a_norm[k-i] + a_norm[i] - 2*conv[k]]
    #
    # Therefore: d(smooth_max)/dx_i = sum_k w[k] * a_i/s * [a_norm[k-i] + a_norm[i] - 2*conv[k]]
    # = a_i/s * [sum_k w[k]*(a_norm[k-i] + a_norm[i]) - 2*sum_k w[k]*conv[k]]
    # = a_i/s * [corr_w[i] + a_norm[i]*sum(w) - 2*E_w[conv]]
    # where corr_w[i] = sum_k w[k]*a_norm[k-i] (cross-correlation of w and a_norm)
    # and sum(w) = 1, E_w[conv] = sum_k w[k]*conv[k]
    #
    # And ds/dx_i = a_i
    #
    # So dF/dx_i = 2n/s^3 * [s * dS/dx_i - 2*S*a_i]
    # = 2n/s^3 * [s * a_i/s * (corr_w[i] + a_norm[i] - 2*mean_w_conv) - 2*S*a_i]
    # = 2n/s^2 * a_i * [corr_w[i] + a_norm[i] - 2*mean_w_conv - 2*S/s]
    # Since a_norm[i] = a_i/s:
    # = 2n/s^2 * a_i * [corr_w[i] + a_i/s - 2*mean_w_conv - 2*S/s]
    # Hmm, S = smooth_max and s = sum(a). Let me just compute it numerically to verify.
    
    # Actually, for the optimization, since the objective is scale-invariant (F(a) = F(c*a) for c > 0),
    # we can work with a directly (not a_norm) and use the gradient on the simplex.
    # 
    # Let's work with a (not log). The objective is F(a) = 2n*max(a*a)/(sum(a))^2.
    # Using the smoothed version: F_s(a) = 2n*smax(a*a)/(sum(a))^2
    # where smax is the smooth max.
    #
    # dF_s/a_i = 2n * [s * d(smax)/da_i * s^2 - smax * 2*s * 1] / s^4  ... no
    # F = 2n * S / s^2
    # dF/da_i = 2n * (dS/da_i * s^2 - S * 2s * 1) / s^4 = 2n/s^2 * (dS/da_i - 2*S/s)
    # where dS/da_i = sum_k w[k] * d(conv[k])/da_i = sum_k w[k] * (a[k-i] + a[i] if k-i in range... )
    # Actually d(conv[k])/da_i = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i<n and k-i != i... no)
    # conv[k] = sum_j a[j]*a[k-j], so d(conv[k])/da_i = a[k-i] (if 0<=k-i<n) + (a[i] if k-2i is... no)
    # d(conv[k])/da_i = a[k-i] + a[i] if k-i == i, i.e., if k = 2i. Otherwise just a[k-i].
    # More precisely: d(conv[k])/da_i = sum_j [delta(j,i)*a[k-j] + delta(k-j,i)*a[j]]
    # = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i<n) [the second term when j = k-i, and we need k-j=i, so j=k-i]
    # Wait: d/da_i sum_j a[j]*a[k-j] = a[k-i] + a[i] if k-i is in [0,n-1] (first term: j=i gives a[k-i])
    # plus a[i] if k-i is in [0,n-1] (second term: k-j=i gives j=k-i, and a[j]=a[k-i]... no)
    # Let me just say: d(conv[k])/da_i = a[k-i] + [a[i] if 2i==k]... no that's wrong too.
    # 
    # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
    # d/da_i: we get a[k-i] from the term j=i (if k-i is valid), plus a[i] from the term j=k-i (if k-i is valid and k-i != i).
    # If k-i == i (i.e., k = 2i), then it's 2*a[i].
    # Otherwise (if both i and k-i are valid and different), it's a[k-i] + a[i].
    # If only one is valid, it's just that one.
    
    # For the gradient in log-parameterization:
    # dF/dx_i = dF/da_i * da_i/dx_i = dF/da_i * a_i
    
    # Let me just compute corr_w and use the formula:
    # dS/da_i = sum_k w[k] * (a[max(0,k-i) ... ] ) -- this is getting messy.
    
    # Let me use a cleaner formulation. Define the cross-correlation:
    # (w * a)[i] = sum_k w[k] * a[k-i] (with a[j]=0 for j<0 or j>=n)
    # This is the same as the convolution w * a_flip, evaluated at position n-1+i.
    
    # For dS/da_i:
    # dS/da_i = sum_k w[k] * d(conv[k])/da_i
    # d(conv[k])/da_i = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i<n, counting the j=k-i term)
    # But wait, when j=i in conv[k]=sum_j a[j]*a[k-j], we get a[i]*a[k-i].
    # The derivative w.r.t. a[i] is a[k-i] (from the first factor) + a[k-i] (from the second factor, when k-j=i, i.e., j=k-i, giving a[k-i]*a[i], derivative is a[k-i]).
    # So d(conv[k])/da_i = a[k-i] + a[k-i] = 2*a[k-i] if i != k-i (i.e., k != 2i)
    # and = 2*a[i] if k = 2i (which is the same as 2*a[k-i] since k-i=i).
    # So d(conv[k])/da_i = 2*a[k-i] if 0<=k-i<n, else 0.
    # Wait, that's only if both i and k-i are in [0,n-1].
    # If i is in [0,n-1] but k-i is not, then only one term contributes: a[k-i] doesn't exist, so d(conv[k])/da_i = 0 from that term? 
    # No: if k-i < 0 or k-i >= n, then a[k-i] = 0 (by convention), so the product a[i]*a[k-i] = 0 and its derivative is also 0.
    # Similarly for the a[k-j] factor: if j = k-i is out of range, that term doesn't exist.
    #
    # So: d(conv[k])/da_i = 2*a[k-i] if 0 <= k-i < n, else 0.
    # (The factor 2 comes from the two occurrences of a[i] in the sum, but they only both exist when k-i is valid.)
    
    # Therefore: dS/da_i = 2 * sum_k w[k]*a[k-i] (where a is extended by 0)
    # = 2 * (w * a_reflected)[i] where a_reflected[j] = a[n-1-j]... 
    # Actually, sum_k w[k]*a[k-i] is the cross-correlation of w and a at lag i.
    # In FFT terms: if W = FFT(w) and A = FFT(a), then the cross-correlation is IFFT(conj(W)*A) shifted appropriately.
    # Or equivalently, it's the convolution w * a_flip evaluated at appropriate positions.
    
    # Let me compute it as: corr[i] = sum_k w[k] * a[(k-i) mod nothing] = sum_k w[k] * a[k-i]
    # This is (w * a_rev)[n-1+i] where a_rev = a[::-1] and * is full convolution.
    # Actually, let me just use the definition directly with FFT.
    
    # corr[i] = sum_{k} w[k] * a[k-i] for 0<=i<n
    # Let's define b[j] = a[-j] = a[n-1+n-1-j]... this is getting confusing.
    # 
    # corr[i] = sum_k w[k] * a[k-i]
    # Let m = k-i, so k = m+i:
    # corr[i] = sum_m w[m+i] * a[m]
    # This is the cross-correlation <w_shifted, a> where w_shifted[m] = w[m+i].
    # In convolution terms: corr = ifft(fft(w, size) * conj(fft(a, size))) ... not quite.
    # 
    # Actually, the standard cross-correlation (w ⋆ a)[i] = sum_k w[k+i]*a[k] = sum_m w[m+i]*a[m].
    # This equals (w * a_flip_full)[n-1+i] where a_flip_full is a padded and flipped.
    # Or in FFT: IFFT(FFT(w, N) * conj(FFT(a, N))[::-1])... 
    # 
    # Let me just use the compensation: define the "anti-convolution"
    # corr[i] = sum_k w[k]*a[k-i] = (w * a_mirror)[n-1+i] where a_mirror[j] = a[n-1-j]
    # and (w * a_mirror)[p] = sum_j w[j]*a_mirror[p-j] = sum_j w[j]*a[n-1-p+j]
    # With p = n-1+i: sum_j w[j]*a[n-1-(n-1+i)+j] = sum_j w[j]*a[j-i]
    # That's sum_j w[j]*a[j-i], but I want sum_k w[k]*a[k-i]. These are the same (just rename j to k).
    
    # So: corr = full_convolution(w, a_mirror)[n-1 : n-1+n]
    
    # OK let me just implement this directly.
    
    # For now, let's use a simpler (slightly less efficient) approach:
    # Compute the gradient by: dS/da_i = 2 * corr[i] where corr[i] = sum_k w[k]*a[k-i]
    
    size = next_pow2(2 * n - 1)
    a_mirror = a[::-1]
    fa_mirror = rfft(a_mirror, size)
    fw = rfft(w, size)
    conv_w_am = irfft(fw * fa_mirror, size)
    corr = conv_w_am[n - 1:n - 1 + n]
    
    # Now: dS/da_i = 2 * corr[i]
    # dF/da_i = 2n/s^2 * (2*corr[i] - 2*S/s)  ... wait let me redo.
    # F = 2n*S/s^2
    # dF/da_i = 2n * (dS/da_i * s^2 - S * 2s) / s^4 = 2n/s^2 * (dS/da_i - 2*S/s)
    # dS/da_i = 2*corr[i]
    # dF/da_i = 2n/s^2 * (2*corr[i] - 2*S/s)
    
    # In log-parameterization: dF/dx_i = dF/da_i * a_i = 2n*a_i/s^2 * (2*corr[i] - 2*S/s)
    
    dS_da = 2.0 * corr
    dF_da = 2.0 * n / (s * s) * (dS_da - 2.0 * smooth_max / s)
    grad_x = dF_da * a  # chain rule: dF/dx_i = dF/da_i * da_i/dx_i = dF/da_i * a_i
    
    return F, grad_x


def gradient_optimize_v2(a_init, time_budget=60.0, verbose=False):
    """
    Optimized gradient with Nesterov acceleration and adaptive temperature.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Temperature schedule
    taus = [20.0/n, 10.0/n, 5.0/n, 2.5/n, 1.5/n, 1.0/n, 0.5/n, 0.3/n, 0.15/n, 0.08/n, 0.04/n]
    
    x = np.log(a) - np.log(a).max()
    
    for tau in taus:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        remaining = time_budget - elapsed
        max_iter = max(50, int(200 * remaining / time_budget))
        
        # Define objective and gradient
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_grad(xv, tau)
            return g
        
        # Use L-BFGS-B with more iterations for smaller tau
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 30})
        
        if res.fun < fun(x) - 1e-15:
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={best_obj:.10f}")
    
    return best_a


def nesterov_subgradient(a_init, time_budget=60.0, verbose=False):
    """
    Nesterov-accelerated subgradient method with Cutler-Polyak averaging.
    Works directly with the non-smooth max objective.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov parameters
    v = np.zeros(n)  # momentum
    y = a.copy()  # auxiliary variable (where we evaluate gradient)
    
    # Learning rate schedule
    T = max(1000, int(time_budget * 100))  # estimated total iterations
    
    # Cutler-Polyak averaging
    a_ave = a.copy()
    ave_count = 0
    
    for t in range(1, T + 1):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient at y
        s_y = float(np.sum(y))
        y_norm = y / s_y
        conv = compute_conv(y_norm)
        k_star = int(np.argmax(conv))
        
        # Subgradient of 2n*max(conv)/s^2 at y (in the normalized space)
        # d(2n*conv[k*]/s^2)/da_i = 2n/s^2 * (2*conv_contrib[i] - 2*conv[k*]/s * s) ... 
        # Let's work with F(a) = 2n*max(a*a)/(sum(a))^2
        # Subgradient: 2n/s^2 * (2*g_i - 2*M/s * 1) where g_i = d(max conv)/da_i = 2*a[k*-i], M = max conv
        # = 4n/s^2 * (a[k*-i] - M/s) for i=0,...,n-1
        # (with a[j]=0 for j out of range)
        
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2.0 * y_norm[j]
        
        M_val = float(conv[k_star])
        grad = 4.0 * n / (s_y * s_y) * (g - M_val / s_y)
        
        # Nesterov step
        alpha = 1.0 / (np.sqrt(t) + 1.0)
        v = 0.5 * (v + grad)  # momentum
        y = a - alpha * v
        
        # Project to simplex (a >= 0, sum(a) = 1)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y
        
        # Update a via averaging
        a_ave = a_ave * (ave_count / (ave_count + 1)) + y * (1.0 / (ave_count + 1))
        ave_count += 1
        
        # Every 50 iterations or so, check if averaged solution is better
        if t % 50 == 0:
            obj_ave = compute_obj(a_ave)
            if obj_ave < best_obj:
                best_obj = obj_ave
                best_a = a_ave.copy()
            obj_y = compute_obj(y)
            if obj_y < best_obj:
                best_obj = obj_y
                best_a = y.copy()
                a = y.copy()
    
    return best_a


def multi_peak_subgradient(a_init, time_budget=60.0, K=3, verbose=False):
    """
    Subgradient method using top-K peaks of the convolution (not just the max).
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 2.0 / n
    t = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        s = float(np.sum(a))
        a_norm = a / s
        conv = compute_conv(a_norm)
        
        # Get top K peaks
        idx = np.argsort(conv)[::-1][:K]
        
        # Weighted subgradient
        g = np.zeros(n)
        for k in idx:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += 2.0 * a_norm[j]
        
        M_val = float(conv[idx[0]])
        grad = 4.0 * n / (s * s) * (g / K - M_val / s)
        
        # Update
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = max(eta * 0.999, 0.01 / n)
        else:
            stall += 1
            if stall > 20:
                eta = max(eta * 0.9, 0.01 / n)
                if eta < 0.005 / n:
                    # Restart
                    rng = np.random.default_rng(t)
                    a = best_a * np.exp(rng.normal(0, 0.03, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    eta = 2.0 / n
                    stall = 0
    
    return best_a


def lp_alternate_v2(a, time_budget=60.0, verbose=False):
    """
    Improved LP alternating with better line search and warm starts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if time_budget - elapsed < 3.0:
            break
        
        # Build convolution matrix A (2n-1 x n) such that A @ g = conv(a, g)
        # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a[:r]
        
        # Solve LP: min M s.t. A @ g <= M, sum(g) = 1, g >= 0
        try:
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(0, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    g_val = g_val / s_g
                    
                    # Line search
                    for t in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8]:
                        a_try = (1.0 - t) * a + t * g_val
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            stall = 0
                            break
                    else:
                        stall += 1
        except Exception:
            stall += 1
        
        if stall > 5:
            # Perturb
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            scale = 0.01 + 0.005 * min(stall // 5, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            stall = 0
    
    return best_a


def mirror_descent(a_init, time_budget=30.0, verbose=False):
    """
    Mirror descent (Fübini) for the non-smooth objective on the simplex.
    Equivalent to exponentiated gradient but with different step sizes.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 1.0
    t = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        s = float(np.sum(a))
        a_norm = a / s
        conv = compute_conv(a_norm)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2.0 * a_norm[j]
        
        M_val = float(conv[k_star])
        grad = g - M_val  # simplified gradient on the simplex
        
        # Exponentiated gradient update
        a_new = a * np.exp(-eta * grad / np.max(np.abs(grad) + 1e-10))
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 10 == 0:
                eta *= 0.95
                if eta < 0.01:
                    # Restart with perturbation
                    rng = np.random.default_rng(t)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    eta = 1.0
                    stall = 0
    
    return best_a


def coordinate_descent(a_init, time_budget=30.0, verbose=False):
    """
    Coordinate descent: at each step, optimize one coordinate at a time.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    order = np.random.default_rng(42).permutation(n)
    pos = 0
    
    while time.monotonic() - start < time_budget:
        i = order[pos % n]
        pos += 1
        
        # Save old value
        old_val = a[i]
        
        # Try a range of values for a[i]
        # The objective as a function of a[i] (with others fixed) is:
        # F(a) = 2n * max_k conv_k / (sum)^2
        # conv_k depends linearly on a[i] (since conv_k = sum_j a[j]*a[k-j])
        # Specifically, conv_k = (conv_k without a[i] terms) + a[i]*(a[k-i] + a[i] if 2i==k)
        # Wait, conv_k = sum_j a[j]*a[k-j], and the terms involving a[i] are:
        # j=i: a[i]*a[k-i]
        # j=k-i: a[k-i]*a[i] (same as above if k-i != i, double if k-i == i)
        # So conv_k = base_k + a[i]*a[k-i] + (a[i]*a[k-i] if k-i != i else 0)
        # = base_k + a[i]*a[k-i]*(1 + [k==2i])
        # where base_k = conv_k - a[i]*a[k-i]*(1+[k==2i])
        
        # This is complex. Let's just try a few values.
        candidates = [old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 1.5,
                      old_val + 0.01/n, max(0, old_val - 0.01/n), 0.0, 2*old_val]
        
        for c in candidates:
            a[i] = c
            s = float(np.sum(a))
            if s < 1e-10:
                a[i] = old_val
                continue
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                break
        
        a[i] = old_val if obj >= best_obj else a[i]
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)
    elif mode == 3:
        a = 1.0 - np.abs(np.arange(n) - n / 2) / (n / 2)
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    elif mode == 4:
        c1, c2 = n // 3, 2 * n // 3
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n * n / 10)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n * n / 10))
        a = a / np.sum(a)
    elif mode == 5:
        a = np.ones(n) + 0.5 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 6:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    else:
        # Two-scale: mix of coarse and fine
        coarse_n = max(10, n // 10)
        coarse = rng.uniform(0, 1, coarse_n)
        coarse = coarse / np.sum(coarse)
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, coarse_n), coarse)
        a += 0.1 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    return a
