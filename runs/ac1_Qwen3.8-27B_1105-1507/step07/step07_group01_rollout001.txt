"""
Hybrid optimizer combining LP alternating, coordinate-wise optimization,
smooth-max gradient descent, and multi-n exploration to minimize
2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: adaptive LP with dual-guided directions, coordinate-wise
annealing for local exploitation, and systematic n-search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    if solver_list is None:
        solver_list = ['CLARABEL', 'MOSEK', 'SCS', 'ECOS', 'PDLP']
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in solver_list:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass
    
    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def smooth_max_weights(conv, T):
    """Compute softmax weights over conv bins."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = float(np.sum(exp_vals))
    return exp_vals / total

def smooth_gradient(weights, a):
    """Gradient of smooth max w.r.t. a using correlation."""
    return np.correlate(weights, a, mode='valid')

        if g_lp is None:

        # Adaptive line search

def smooth_max_gd(a, time_budget, T_init=0.001, T_min=0.00005):

    if s < 1e-10:

        grad = smooth_gradient(weights, a)

        a_new = simplex_project(a + v)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

                if lr < 1e-6:

                    a = best_a.copy()
                    v = np.zeros(n)
                else:
                    a = best_a.copy()
                    v = np.zeros(n)
                stall = 0

        ]

                    best_a_local = best_a.copy()
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                        a_try = (1-t) * a + t * g
                        a_try = np.maximum(a_try, 0)
                        st = float(np.sum(a_try))
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        obj_try = compute_obj(a_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_a_local = a_try.copy()
                    
                    if best_t_obj < best_obj - 1e-13:
                        best_obj = best_t_obj
                        best_a = best_a_local.copy()
                        a = best_a
        except Exception:
            break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        a = np.cumsum(rng.uniform(0, 1, n))
        return a / np.sum(a)
    elif mode == 3:
        center = n // 2
        width = max(1, n // 5)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a += 0.5 * np.exp(-((x - center * 0.7) ** 2) / (2 * (width // 2 + 1) ** 2))
        return a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        return a / np.sum(a)
    elif mode == 5:
        x = np.arange(n) / max(1, n - 1)
        a = 1 - np.abs(2 * x - 1)
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif mode == 6:
        # Two humps
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 10)
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        return a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
        return a
