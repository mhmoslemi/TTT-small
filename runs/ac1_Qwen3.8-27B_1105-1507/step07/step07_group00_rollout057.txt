from scipy.optimize import minimize

def penalty_objective(a, a_base, M_target, lambda_penalty):
    n = len(a)
    conv = compute_conv(a)
    # Penalty for exceeding M_target
    excess = np.maximum(conv - M_target, 0)
    penalty = np.sum(excess**2)
    # Regularization to stay near a_base (helps convergence)
    reg = 0.01 * np.sum((a - a_base)**2)
    return lambda_penalty * penalty + reg

def penalty_optimize(a, time_budget=100.0):
    n = len(a)
    a = a / a.sum()
    current_conv = compute_conv(a)
    M_current = float(np.max(current_conv))
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Start with M_target slightly below current max
    M_target = M_current * 0.99
    lambda_penalty = 100.0
    
    for outer in range(20):
        if time.monotonic() - start > time_budget:
            break
        
        res = minimize(
            penalty_objective,
            a,
            args=(a, M_target, lambda_penalty),
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        
        if res.success or res.fun < penalty_objective(a, a, M_target, lambda_penalty):
            a_new = np.maximum(res.x, 0)
            s = float(np.sum(a_new))
            if s > 1e-10:
                a_new = a_new / s
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    # Tighten M_target
                    conv_new = compute_conv(a)
                    M_target = max(float(np.max(conv_new)) * 0.99, best_obj / (2*n) * 0.95)
                    lambda_penalty *= 2
        
        # Tighten M_target regardless
        conv_cur = compute_conv(a)
        M_target = min(M_target, float(np.max(conv_cur)) * 0.98)
        
    return best_a

Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combined approach: improved LP alternating, gradient descent with analytic gradients,
simulated annealing with peak-targeting moves, and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Assume a is normalized (sum=1). Return 2n*max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:min(n, 2*n-1-j)]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Try cvxpy solvers
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternating_v2(a, time_budget=300.0, verbose=False):
    """
    Improved LP alternating with golden section line search.
    """
    n = len(a)
    a = normalize(a.copy())
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    stall = 0
    iter_count = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            # Fallback: use the current a as direction
            g_lp = a.copy()
        
        # Golden section line search on t in [0, 1]
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = normalize(a_try)
            return compute_obj_norm(a_try)
        
        # Golden section search
        gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
        lo, hi = 0.0, 1.0
        c = lo + (1 - gr) * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)
        
        # Also evaluate t=1
        f1 = obj_at_t(1.0)
        
        for _ in range(8):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = lo + (1 - gr) * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at_t(d)
        
        t_best = (lo + hi) / 2
        # Compare with t=1
        if f1 < obj_at_t(t_best):
            t_best = 1.0
        
        a_new = normalize((1.0 - t_best) * a + t_best * g_lp)
        obj_new = compute_obj_norm(a_new)
        
        iter_count += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and iter_count % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP2 it{iter_count}: obj={best_obj:.8f}, t={t_best:.4f}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 15:
                # Perturb
                rng = np.random.default_rng(iter_count * 7919 + n * 31)
                scale = 0.05 + 0.03 * min(5, stall // 15)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = normalize(a)
                best_a = a.copy()
                best_obj = compute_obj_norm(a)
                stall = 0
    
    return best_a


def gradient_descent_optimize(a, time_budget=200.0, verbose=False):
    """
    Gradient descent on the simplex with Nesterov acceleration.
    Uses subgradient of the max convolution bin.
    """
    n = len(a)
    a = normalize(a.copy())
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.9
    lr = 0.1 / np.sqrt(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        
        # Subgradient: g_i = 2*a_{k*-i} (for 0 <= k*-i < n)
        g = np.zeros(n)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            j_start = k_star - i_end
            j_end = k_star - i_start
            g[i_start:i_end + 1] = 2.0 * y[j_start:j_end + 1][::-1]
        
        # Also add contributions from top-K peaks for smoother gradient
        top_K = 5
        top_idx = np.argsort(conv_y)[::-1][:top_K]
        for k in top_idx[1:]:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += 0.5 * y[j_start:j_end + 1][::-1]
        
        # Nesterov step
        a_new = y - lr * g
        a_new = normalize(a_new)
        
        # Check objective
        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Update momentum
        beta = 0.5 * (1.0 + momentum)
        y = a_new + beta * (a_new - a)
        y = normalize(y)
        
        a = a_new
        
        # Adaptive lr
        if stall > 50:
            lr *= 0.8
            stall = 0
        if lr < 1e-6 / np.sqrt(n):
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + n)
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = normalize(a)
            y = a.copy()
            best_a = a.copy()
            best_obj = compute_obj_norm(a)
            lr = 0.1 / np.sqrt(n)
            momentum = 0.9
            stall = 0
        
        it += 1
    
    return best_a


def simulated_annealing(a, time_budget=200.0, verbose=False):
    """
    Simulated annealing with peak-targeting moves.
    Move: pick a position contributing to the peak, move mass to a non-contributing position.
    """
    n = len(a)
    a = normalize(a.copy())
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    
    T = 0.1  # initial temperature
    T_min = 1e-6
    cooling = 0.999
    
    conv = compute_conv(a)
    
    it = 0
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget:
            break
        
        k_star = int(np.argmax(conv))
        
        # Find contributors to the peak
        # (a*a)_{k_star} = sum_i a_i * a_{k_star - i}
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)
        
        # Find non-contributors (positions that don't appear in any product for k_star)
        contrib_positions = set()
        for _, i, j in contributors:
            contrib_positions.add(i)
            contrib_positions.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_positions]
        
        if not non_contrib:
            break
        
        # Pick a move
        if contributors:
            # Move mass from a high contributor to a non-contributor
            contrib_weight = np.array([c[0] for c in contributors[:20]])
            contrib_weight /= contrib_weight.sum()
            idx = np.random.choice(len(contributors[:20]), p=contrib_weight)
            _, i_hot, j_hot = contributors[idx]
            
            target = int(np.random.choice(non_contrib))
            
            # How much to move
            delta = np.random.exponential(0.02 / np.sqrt(n)) * max(a[i_hot], a[j_hot])
            delta = min(delta, max(a[i_hot], a[j_hot]) * 0.5)
            
            # Apply move (from the larger of i_hot, j_hot)
            src = i_hot if a[i_hot] >= a[j_hot] else j_hot
            a[src] -= delta
            a[target] += delta
            a = np.maximum(a, 0)
            a = normalize(a)
            
            new_conv = compute_conv(a)
            new_obj = 2.0 * n * float(np.max(new_conv))
            
            if new_obj < best_obj - 1e-13:
                best_obj = new_obj
                best_a = a.copy()
                conv = new_conv
            else:
                # Metropolis acceptance
                delta_obj = new_obj - best_obj
                if delta_obj > 0 and np.random.random() < np.exp(-delta_obj / max(T, T_min)):
                    conv = new_conv
                else:
                    # Revert
                    a[src] += delta
                    a[target] -= delta
                    a = np.maximum(a, 0)
                    a = normalize(a)
        
        # Cool
        T = max(T_min, T * cooling)
        
        # Periodic restart
        if it % 200 == 0 and it > 0:
            obj_cur = compute_obj_norm(a)
            if obj_cur > best_obj * 1.01:
                a = best_a.copy()
                conv = compute_conv(a)
        
        it += 1
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SA it{it}: obj={best_obj:.8f}, T={T:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Symmetric random
        half = n // 2
        a_half = rng.uniform(0.1, 1, half)
        a = np.concatenate([a_half, a_half[::-1]]) if n % 2 == 0 else np.concatenate([a_half, a_half[::-1][:-1]])
        a = a / a.sum()
    elif mode == 3:
        # Low-frequency smooth
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amps = rng.uniform(0.1, 0.5, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Tail-heavy
        x = np.linspace(0, 1, n)
        a = x**3 + (1-x)**3 + 0.05
        a = a / a.sum()
    elif mode == 5:
        # Periodic
        period = rng.integers(max(2, n//20), max(3, n//5))
        a = np.zeros(n)
        for i in range(n):
            a[i] = 1.0 + 0.5 * np.sin(2 * np.pi * i / period)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Binary-like
        a = rng.integers(0, 2, n).astype(float)
        a = a + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: for each pair (i, j), try to find the optimal split.
    """
    n = len(a)
    a = normalize(a.copy())
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    improved = True
    while improved and time.monotonic() - start < time_budget:
        improved = False
        k_star = int(np.argmax(conv))
        
        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() - start > time_budget:
                break
            
            # Try moving mass from i to i+1 (smoothness) or to a non-contributing position
            for j in [(i + 1) % n, (i - 1 + n) % n]:
                if i == j:
                    continue
                for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    sn = float(np.sum(a))
                    a = a / sn
                    
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                    else:
                        a[i] += delta
                        a[j] -= delta
                        a = np.maximum(a, 0)
                        a = a / sn
                        break
    
    return best_a
