    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Compute subgradient: grad[j] = 2*a[k_star-j] (valid)
        grad = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                grad[j] = 2.0 * a[k_j]

        # Project onto tangent space of simplex
        grad_proj = grad - np.mean(grad)

        # Line search (backtracking)
        best_alpha = alpha
        best_new_obj = best_obj
        for lr in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05]:
            a_try = a - lr * grad_proj
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_new_obj = obj_try
                best_alpha = lr
                break

        if best_new_obj < best_obj - 1e-14:
            best_obj = best_new_obj
            best_a = best_a if best_a is None else np.zeros(1)
            # Redo with best_alpha
            a_try = a - best_alpha * grad_proj
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try > 1e-10:
                a = a_try / s_try
            best_a = a.copy()
            # Adjust learning rate
            if best_alpha > 0 and best_alpha > alpha * 0.5:
                alpha = min(alpha * 1.05, 0.5)
            else:
                alpha *= 0.8
        else:
            alpha *= 0.7
            # Random perturbation to escape
            if it % 50 == 0:
                rng = np.random.default_rng(it * 104729 + n)
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def build_conv_matrix_sparsity(a):
    """Build the convolution matrix as a list of (rows, cols, values) for sparsity."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = i + j
            if k >= 2 * n - 1:
                break
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    return rows, cols, vals


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)

    # Build A matrix (2n-1 x n) such that A @ g = conv(a, g)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            A[k, j] = a[k - j]

    solvers = [cp.MOSEK, cp.CLARABEL, cp.HIGHS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to HiGHS linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization with line search and perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            if it % 5 == 0:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_initializations(n, rng):
    """Generate diverse initialization patterns."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp
    for slope in [1.5, 2.0, 3.0]:
        a = np.linspace(1.0, slope, n)
        inits.append(a / np.sum(a))

    # Exponential
    for rate in [1.0, 2.0, 4.0]:
        a = np.exp(np.linspace(0, rate, n))
        inits.append(a / np.sum(a))
        a2 = np.exp(-np.linspace(0, rate, n))
        inits.append(a2 / np.sum(a2))

    # Gaussian bump (varied positions)
    for center in [0.25, 0.5, 0.75]:
        for width in [0.1, 0.2, 0.3]:
            x = np.linspace(0, 1, n)
            a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
            inits.append(a / np.sum(a))

    # Sine-modulated
    for freq in [1, 2, 3]:
        for phase in [0, np.pi / 2, np.pi]:
            a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * np.linspace(0, 1, n) + phase)
            a = np.maximum(a, 0.01)
            inits.append(a / np.sum(a))

    # Concatenated blocks (asymmetric)
    for ratio in [0.3, 0.4, 0.6]:
        n1 = int(n * ratio)
        n2 = n - n1
        high = 1.0 + rng.uniform(0.5, 2.0)
        a = np.zeros(n)
        a[:n1] = high
        a[n1:] = 1.0
        inits.append(a / np.sum(a))

    # Power law
    for p in [0.5, 1.5, 2.5]:
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** p
        inits.append(a / np.sum(a))
        a2 = idx ** (-p)
        inits.append(a2 / np.sum(a2))

    return inits


def quick_smoothed_optimize(a_init, time_budget=30.0):
    """Quick smoothed optimization for multi-n exploration."""
    n = len(a_init)
    return smooth_lbfgs_fft(a_init, tau_start=20.0, tau_end=200.0,
                            n_stages=4, time_budget=time_budget, max_iter=100)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining FFT-accelerated smoothed L-BFGS-B,
    subgradient descent, LP alternating, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting optimization. Target: < 1.5027, Current: {best_score:.8f}")

    # ============ Phase 1: Main smoothed L-BFGS-B (FFT-accelerated) ============
    print(f"\n=== Phase 1: Smoothed L-BFGS-B (FFT) ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(remaining * 0.4, 600.0)

    trials = []
    # Trial 0: Parent sequence
    trials.append(best_seq.copy())
    # Trial 1: Small perturbation
    trials.append(best_seq * np.exp(rng.normal(0, 0.03, n_parent)))
    trials[-1] = np.maximum(trials[-1], 0)
    trials[-1] /= np.sum(trials[-1])
    # Trial 2: Moderate perturbation
    trials.append(best_seq * np.exp(rng.normal(0, 0.08, n_parent)))
    trials[-1] = np.maximum(trials[-1], 0)
    trials[-1] /= np.sum(trials[-1])

    for trial_idx, a_init in enumerate(trials):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.45, phase1_budget / (len(trials) - trial_idx))
        a_opt, obj_opt = smooth_lbfgs_fft(a_init, tau_start=10.0, tau_end=500.0,
                                          n_stages=8, time_budget=alloc, max_iter=400)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_parent = len(best_seq)
            print(f"  P1 Trial {trial_idx}: obj={best_score:.8f} **")
        else:
            print(f"  P1 Trial {trial_idx}: obj={obj_opt:.8f}")

    # Phase 1b: More tau stages from best
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_opt, obj_opt = smooth_lbfgs_fft(best_seq, tau_start=200.0, tau_end=2000.0,
                                          n_stages=6, time_budget=min(remaining * 0.3, 200.0),
                                          max_iter=500)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  P1 Fine-tune: obj={best_score:.8f} **")
        else:
            print(f"  P1 Fine-tune: obj={obj_opt:.8f}")

    # ============ Phase 2: Subgradient descent ============
    print(f"\n=== Phase 2: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_sg, obj_sg = subgradient_descent(best_seq, time_budget=min(remaining * 0.2, 200.0))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # ============ Phase 3: LP Alternating ============
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_lp = lp_alternating(best_seq, time_budget=min(remaining * 0.2, 180.0))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # ============ Phase 4: Multi-n exploration ============
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([n for n in [600, 700, 750, 800, 850, 900, 1000, 1200]
                            if 400 <= n <= 1500]))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_cur:
            continue  # Already optimized

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 50.0)
        a_opt, obj_opt = quick_smoothed_optimize(a_try, time_budget=alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # ============ Phase 5: Diverse initializations ============
    print(f"\n=== Phase 5: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    inits = generate_initializations(n_best, rng)
    rng.shuffle(inits)

    for idx, a_init in enumerate(inits[:10]):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Blend with best sequence for continuity
        blend = 0.3 + 0.4 * (idx / 10.0)
        a_blend = (1 - blend) * a_init + blend * best_seq
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / np.sum(a_blend)

        alloc = min(remaining * 0.1, 30.0)
        a_opt, obj_opt = smooth_lbfgs_fft(a_blend, tau_start=15.0, tau_end=300.0,
                                          n_stages=5, time_budget=alloc, max_iter=150)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Init {idx}: obj={best_score:.8f} **")
        else:
            print(f"  Init {idx}: obj={obj_opt:.8f}")

    # ============ Phase 6: Final intensive polish ============
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # 6a: High-tau smoothed optimization
    if remaining > 30:
        a_opt, obj_opt = smooth_lbfgs_fft(best_seq, tau_start=500.0, tau_end=5000.0,
                                          n_stages=5, time_budget=min(remaining * 0.3, 100.0),
                                          max_iter=500)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final Smooth (high tau): obj={best_score:.8f} **")
        else:
            print(f"  Final Smooth (high tau): obj={obj_opt:.8f}")

    # 6b: Subgradient refinement
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sg = subgradient_descent(best_seq, time_budget=min(remaining * 0.3, 60.0))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Final Subgrad: obj={obj_sg:.8f}")

    # 6c: One more LP pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 60.0))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    # 6d: Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_sg = subgradient_descent(best_seq, time_budget=min(remaining * 0.5, 40.0))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Very Final Subgrad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
        print(f"  evaluate_sequence gave better: {val:.8f}")
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
        print("  WARNING: evaluate_sequence returned inf, using fallback")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
