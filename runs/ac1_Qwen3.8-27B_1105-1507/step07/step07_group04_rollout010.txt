    return 2.0 * n * float(np.max(conv))

def compute_soft_gradient(weights, a):
    """
    Compute grad[m] = sum_k weights[k] * a[k-m] for m=0..n-1
    using FFT for efficiency.
    """
    n = len(a)
    # We want cross-correlation (weights ⊛ a)[m] = sum_j weights[j+m] * a[j]
    # Using FFT: ifft(FFT(weights) * conj(FFT(a)))
    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)[:2 * n - 1]
    return corr[:n]

def soft_mwu(a, time_budget=200.0, T_init=0.05, T_final=0.005, verbose=False):

        # Compute soft weights
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = (conv - max_conv) / T

        shifted = np.clip(shifted, -50, 50)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            break

        grad = compute_soft_gradient(weights, a)
        
        # Do a batch of MWU steps
        for _ in range(batch):
            if time.monotonic() - start > time_budget:
                break

            a_new = np.maximum(a_new, 0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

def solve_lp_minpeak(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    
    return None

def frequency_shaping(a, time_budget=60.0, verbose=False):
    """
    Try to flatten the spectrum |A(k)|^2 using iterative projection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Compute spectrum
        A = rfft(a, N)
        power = np.abs(A) ** 2
        
        # Target: geometric mean of power bins (make spectrum flat)
        # But we only care about the "active" bins
        # All bins are active since a is supported on [0, n-1]
        
        # Compute target level
        total_power = np.sum(power)
        n_bins = N // 2 + 1
        target_level = total_power / n_bins
        
        # Update magnitudes towards target (keeping phases)
        phases = np.angle(A)
        new_mags = np.sqrt(np.abs(A) ** 2 * 0.5 + target_level * 0.5)
        new_A = new_mags * np.exp(1j * phases)
        
        # Inverse FFT
        a_new = irfft(new_A, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        it += 1
        if verbose and it % 50 == 0:

def coordinate_descent(a, time_budget=30.0):
    """Fast coordinate descent polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        improved = False
        # Try moving mass from indices that contribute most to peak
        # to indices that contribute least
        contributors = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                contributors[m] = a[j]
        
        # Sort by contribution (descending) - these are "bad" indices
        # Try moving mass from high contributor to low contributor
        sorted_idx = np.argsort(contributors)[::-1]
        top_half = sorted_idx[:n//2]
        bottom_half = sorted_idx[n//2:]
        
        for i in top_half[:5]:
            for j in bottom_half[:5]:
                if i == j:
                    continue
                for frac in [0.02, 0.05]:
                    delta = frac * a[i]
                    if delta >= a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a /= s
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        a = np.maximum(a, 0)
                        a /= s
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break

    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Wait, I need to double-check the `compute_soft_gradient` function. Let me trace through it more carefully.

We have weights of length 2n-1 (indexed 0 to 2n-2) and a of length n (indexed 0 to n-1).

We want: grad[m] = sum_k weights[k] * a[k-m] for m = 0, ..., n-1

where a[j] = 0 if j < 0 or j >= n.

Let's substitute: for a given m, the sum over k goes from max(0, m) to min(2n-2, m+n-1), and the term is weights[k] * a[k-m].

Let j = k - m, so k = j + m. Then:
grad[m] = sum_j weights[j+m] * a[j] for j = max(0, -m) to min(n-1, 2n-2-m)
= sum_j weights[j+m] * a[j] for j = 0 to n-1 (since m >= 0, -m <= 0)

So grad[m] = sum_{j=0}^{n-1} weights[j+m] * a[j]

This is exactly the cross-correlation (weights ⊛ a)[m] where:
(weights ⊛ a)[m] = sum_j weights[j+m] * conj(a[j])

For real-valued a, this is sum_j weights[j+m] * a[j].

To compute this via FFT:
Cross-correlation can be computed as:
corr[m] = ifft(FFT(weights, N) * conj(FFT(a, N)))

where N >= (2n-1) + n - 1 = 3n - 2 to avoid circular aliasing.

Actually, for cross-correlation of sequences of lengths L1 and L2, we need FFT size >= L1 + L2 - 1.
Here L1 = 2n-1 (length of weights) and L2 = n (length of a).
So N >= 2n-1 + n - 1 = 3n - 2.
