import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = rfft(a, L)
    return irfft(fa * fa, L)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

def smooth_max_mirror_descent(a, n_iters=20000, time_budget=120.0, beta_init=10.0):
    """FFT-based mirror descent with smooth max gradient."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    beta = beta_init
    lr = 0.3
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        conv_max = np.max(conv)

        # Softmax weights (smooth max subgradient)
        diff = conv_max - conv
        weights = np.exp(-beta * diff)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            break
        weights /= w_sum

        # Gradient via FFT: grad[i] = sum_j weights[i+j] * a[j]
        S = fft(weights, L)
        A_fft = fft(a, L)
        grad = ifft(S * np.conj(A_fft)).real[:n]

        # Mirror descent
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                stall = 0
                beta = min(beta * 1.005, 500.0)
                lr = max(lr * 0.9998, 0.005)

                stall += 1
                lr *= 0.97

                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    lr = 0.2
                    beta = beta_init
                    stall = 0

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Smooth via random Fourier
        n_freqs = 6
        freqs = rng.uniform(1, 8, n_freqs)
        phases = rng.uniform(0, 2 * np.pi, n_freqs)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.cos(2 * np.pi * f * x + p)
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        w = max(5, n // 15)
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Random walk smoothed
        a = np.cumsum(rng.normal(0, 1, n))
        a = np.maximum(a - a.min() + 0.1, 0)
        a = a / a.sum()
    else:
        # Periodic with noise
        period = max(3, n // rng.integers(10, 30))
        a = 1.0 + 0.5 * np.cos(2 * np.pi * np.arange(n) / period)
        a += rng.normal(0, 0.1, n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a


def differential_evolution_local(a_best, budget=60.0, dim=None):
    """DE on a low-dimensional parameterization: amplitude of cos/sin basis + perturbation."""
    n = len(a_best)
    if dim is None:
        dim = min(30, n // 10)
    
    # Parameterization: a(x) = base(x) * (1 + sum c_i * cos(2*pi*i*x/n) + d_i * sin(2*pi*i*x/n))
    # where base(x) is the best sequence shape
    base = a_best.copy()
    base = base / base.sum()
    x = np.arange(n) / n
    
    # Precompute basis functions
    basis = [np.ones(n)]
    for i in range(1, dim // 2 + 1):
        basis.append(np.cos(2 * np.pi * i * x))
        basis.append(np.sin(2 * np.pi * i * x))
    
    n_params = len(basis)
    
    # Setup
    NP = min(30, max(10, n_params * 2))
    pop = np.zeros((NP, n_params))
    pop[0] = 0  # First individual is the current best (all params = 0)
    for i in range(1, NP):
        pop[i] = np.random.uniform(-0.3, 0.3, n_params)
    
    def decode(params):
        result = base.copy()
        for i in range(n_params):
            result += params[i] * basis[i]
        result = np.maximum(result, 0)
        s = result.sum()
        if s < 1e-10:
            return base.copy(), np.inf
        result = result / s
        return result, compute_obj(result)
    
    pop_objs = [np.inf] * NP
    _, pop_objs[0] = decode(pop[0])
    for i in range(1, NP):
        _, pop_objs[i] = decode(pop[i])
    
    start = time.monotonic()
    best_local_obj = pop_objs[0]
    best_local_params = pop[0].copy()
    
    gen = 0
    F = 0.8
    CR = 0.9
    
    while time.monotonic() - start < budget:
        gen += 1
        # Adaptive F
        F = 0.7 + 0.3 * np.exp(-gen / 50.0)
        
        improved = False
        for i in range(NP):
            if time.monotonic() - start > budget:
                break
            
            # Select three distinct random individuals
            idx = [j for j in range(NP) if j != i]
            a_idx, b_idx, c_idx = np.random.choice(idx, 3, replace=False)
            
            # Mutation
            v = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])
            
            # Crossover
            trial = np.zeros(n_params)
            j_rand = np.random.randint(0, n_params)
            for j in range(n_params):
                if np.random.random() < CR or j == j_rand:
                    trial[j] = v[j]
                else:
                    trial[j] = pop[i][j]
            
            # Clamp
            trial = np.clip(trial, -0.5, 0.5)
            
            # Evaluate
            _, trial_obj = decode(trial)
            
            if trial_obj < pop_objs[i]:
                pop[i] = trial
                pop_objs[i] = trial_obj
                improved = True
                if trial_obj < best_local_obj:
                    best_local_obj = trial_obj
                    best_local_params = trial.copy()
        
        if not improved and gen > 100:
            break
    
    # Decode best
    best_a, _ = decode(best_local_params)
    return best_a, best_local_obj


def peak_shifting(a, time_budget=30.0):
    """Try to reduce peak by shifting mass away from peak-contributing positions."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        k_star = int(np.argmax(conv))
        
        # Find positions that contribute most to the peak
        # conv[k_star] = sum_i a[i]*a[k_star-i]
        # Contribution of position i is a[i]*a[k_star-i]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Try to reduce the largest contributor
        i_max = int(np.argmax(contributions))
        if contributions[i_max] < 1e-8:
            break
        
        # Find a position to move mass to (one that doesn't contribute to the peak)
        # Try positions where a[k_star - i] is small
        targets = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                targets.append((a[j], i))  # (contribution weight, position)
            else:
                targets.append((0, i))  # No contribution to peak
        
        targets.sort(key=lambda x: x[0])
        
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.15]:
            delta = frac * a[i_max]
            for _, i_target in targets[:5]:
                if i_target == i_max:
                    continue
                a[i_max] -= delta
                a[i_target] += delta
                if a[i_max] < 1e-15:
                    a[i_max] = 1e-15
                    a[i_target] += delta - 1e-15
                a = a / np.sum(a)
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    break
                else:
                    a[i_max] += delta
                    a[i_target] -= delta
                    a = a / np.sum(a)
            if improved:
                break
        
        if not improved:
            # Try random small perturbations
            rng = np.random.default_rng(it)
            for _ in range(5):
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                if i1 == i2:
                    continue
                delta = 0.01 * a[i1]
                a[i1] -= delta
                a[i2] += delta
                if a[i1] >= 0:
                    a = a / np.sum(a)
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i1] += delta
                        a[i2] -= delta
                        a = a / np.sum(a)
            
            if not improved and it > 200:
                # Perturb and restart
                rng = np.random.default_rng(it * 31)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                conv = compute_conv(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining FFT smooth max descent, LP alternating,
    multi-n exploration, DE on Fourier coefficients, and peak shifting.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0  # Safety margin
    rng = np.random.default_rng(seed)

    # Load parent
    best_seq = None
    best_score = np.inf
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            s = float(np.sum(parent))
            if s > 0.01:
                best_seq = parent / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    print(f"\n=== Starting optimization (target < 1.5027) ===")
    print(f"Initial: n={len(best_seq)}, obj={best_score:.8f}")

    # === Phase 1: Smooth max mirror descent (fast, ~120s) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n--- Phase 1: Smooth Max Mirror Descent ---")
        alloc = min(remaining * 0.12, 150.0)
        a1 = smooth_max_mirror_descent(best_seq, n_iters=30000, time_budget=alloc, beta_init=15.0)
        obj1 = compute_obj(a1)
        print(f"  Result: obj={obj1:.8f}")
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()

    # === Phase 2: LP Alternating (precise, ~250s) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n--- Phase 2: LP Alternating ---")
        alloc = min(remaining * 0.2, 250.0)
        a2 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj2 = compute_obj(a2)
        print(f"  Result: obj={obj2:.8f}")
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()

    # === Phase 3: Smooth max again (refine after LP) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n--- Phase 3: Refine with Smooth Max ---")
        alloc = min(remaining * 0.1, 100.0)
        a3 = smooth_max_mirror_descent(best_seq, n_iters=20000, time_budget=alloc, beta_init=30.0)
        obj3 = compute_obj(a3)
        print(f"  Result: obj={obj3:.8f}")
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()

    # === Phase 4: Peak Shifting ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n--- Phase 4: Peak Shifting ---")
        alloc = min(remaining * 0.05, 60.0)
        a4 = peak_shifting(best_seq, time_budget=alloc)
        obj4 = compute_obj(a4)
        print(f"  Result: obj={obj4:.8f}")
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()

    # === Phase 5: Multi-n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n--- Phase 5: Multi-n Exploration ---")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 200), max(200, n_best - 100),
            n_best,
            n_best + 100, n_best + 200, n_best + 400,
            400, 500, 600, 700, 900, 1000, 1200, 1500, 2000
        ]))
        n_options = [n for n in n_options if 200 <= n <= 2500 and abs(n - n_best) > 30]
        
        # Sort by distance from current best (try nearby first)
        n_options.sort(key=lambda x: abs(x - n_best))
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_try = resample_sequence(best_seq, n_try)
            
            # Quick optimization
            alloc_s = min(remaining * 0.06, 30.0)
            a_opt = smooth_max_mirror_descent(a_try, n_iters=5000, time_budget=alloc_s, beta_init=20.0)
            
            # LP refinement
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.03, 20.0)
                a_opt = lp_alternating(a_opt, time_budget=alloc_lp)

            obj_try = compute_obj(a_opt)
            status = " ***" if obj_try < best_score else ""
            print(f"  n={n_try}: obj={obj_try:.8f}{status}")
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()

    # === Phase 6: DE on Fourier coefficients ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n--- Phase 6: Fourier DE ---")
        alloc = min(remaining * 0.08, 60.0)
        a_de, obj_de = differential_evolution_local(best_seq, budget=alloc)
        obj_de = compute_obj(a_de)
        print(f"  Result: obj={obj_de:.8f}")
        if obj_de < best_score:
            best_score = obj_de
            best_seq = a_de.copy()

    # === Phase 7: Random Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n--- Phase 7: Random Restarts ---")
        n_best = len(best_seq)
        n_restarts = 8
        restart_budget_total = min(remaining * 0.15, 120.0)
        
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            alloc_r = restart_budget_total / n_restarts
            
            if r % 4 == 0:
                a_r = random_init(n_best, rng)
            elif r % 4 == 1:
                a_r = best_seq * np.exp(rng.normal(0, 0.1, n_best))
                a_r = np.maximum(a_r, 1e-15)
                a_r = a_r / np.sum(a_r)
            elif r % 4 == 2:
                n_r = int(n_best + rng.integers(-200, 200))
                n_r = max(200, min(2000, n_r))
                a_r = resample_sequence(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
                a_r = np.maximum(a_r, 1e-15)
                a_r = a_r / np.sum(a_r)
            else:
                # Asymmetric perturbation
                a_r = best_seq.copy()
                mid = len(a_r) // 2
                a_r[:mid] *= 1.2
                a_r[mid:] *= 0.9
                a_r = np.maximum(a_r, 1e-15)
                a_r = a_r / np.sum(a_r)

            # Optimize
            a_opt = smooth_max_mirror_descent(a_r, n_iters=3000, time_budget=alloc_r * 0.4, beta_init=20.0)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_opt = lp_alternating(a_opt, time_budget=alloc_r * 0.4)

            obj_r = compute_obj(a_opt)
            status = " ***" if obj_r < best_score else ""
            print(f"  Restart {r} (n={len(a_opt)}): obj={obj_r:.8f}{status}")
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()

    # === Phase 8: Final Intensive Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n--- Phase 8: Final Polish (remaining={remaining:.0f}s) ---")
        a_final = best_seq.copy()

        # LP
        alloc = min(remaining * 0.35, 80.0)
        a_lp = lp_alternating(a_final, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            a_final = a_lp.copy()

        # Smooth max
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 50.0)
            a_sm = smooth_max_mirror_descent(a_final, n_iters=10000, time_budget=alloc, beta_init=50.0)
            obj = compute_obj(a_sm)
            print(f"  Smooth max: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_sm.copy()
                a_final = a_sm.copy()

        # Another LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.4, 50.0)
            a_lp2 = lp_alternating(a_final, time_budget=alloc)
            obj = compute_obj(a_lp2)
            print(f"  LP2: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_lp2.copy()

        # Peak shifting
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 30.0)
            a_ps = peak_shifting(a_final, time_budget=alloc)
            obj = compute_obj(a_ps)
            print(f"  Peak shift: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_ps.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
