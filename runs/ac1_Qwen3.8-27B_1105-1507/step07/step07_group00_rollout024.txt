```python
def solve_lp_for_a(g, time_budget=None):
    """Solve: min t s.t. sum_i a[i]*g[k-i] <= t for all k, sum a = 1, a >= 0."""
    n = len(g)
    num_rows = 2 * n - 1
    
    # Build constraint matrix: M[k, i] = g[k-i] if 0 <= k-i < n
    M = np.zeros((num_rows, n))
    for i in range(n):
        k_vals = np.arange(i, i + n)
        M[k_vals, i] = g[:n]
    
    # LP: min t s.t. M @ a <= t, sum(a) = 1, a >= 0
    # Variables: a[0..n-1], t
    # In cvxpy:
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [M @ a <= t, cp.sum(a) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if a.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s, float(t.value)
        except Exception:
            continue
    
    # Fallback to scipy
    from scipy.optimize import linprog
    # min t
    # s.t. M @ a - t <= 0 (num_rows constraints)
    #      sum(a) = 1
    #      a >= 0, t >= 0 (actually t can be anything, but it'll be positive)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = M
    A_ub[:, n] = -1.0  # M @ a - t <= 0
    b_ub = np.zeros(num_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        a_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(a_val))
        if s > 1e-10:
            return a_val / s, float(res.x[n])
    
    return None, None
```

```python
"""
Optimized alternating LP with multi-scale exploration for minimizing
2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

# --- Core computation functions ---

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_matrix(weights):
    """Build (2n-1) x n matrix: M[k,i] = weights[k-i] if 0<=k-i<n."""
    n = len(weights)
    num_rows = 2*n - 1
    M = np.zeros((num_rows, n))
    for i in range(n):
        k_start = i
        k_end = min(i+n, num_rows)
        M[k_start:k_end, i] = weights[:k_end-k_start]
    return M

def solve_lp(fixed_weights):
    """
    Solve: min t s.t. M @ x <= t for all k, sum(x) = 1, x >= 0
    where M[k,i] = fixed_weights[k-i].
    Returns (x_opt, t_opt) or (None, None).
    """
    n = len(fixed_weights)
    num_rows = 2*n - 1
    M = build_matrix(fixed_weights)
    
    # Try cvxpy solvers
    x = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [M @ x <= t, cp.sum(x) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val/s, float(t.value)
        except Exception:
            continue
    
    # Fallback to scipy HiGHS
    try:
        from scipy.optimize import linprog
        c = np.zeros(n+1)
        c[-1] = 1.0
        A_ub = np.zeros((num_rows, n+1))
        A_ub[:, :n] = M
        A_ub[:, n] = -1.0
        b_ub = np.zeros(num_rows)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)]*(n+1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            x_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(x_val))
            if s > 1e-10:
                return x_val/s, float(res.x[n])
    except Exception:
        pass
    
    return None, None
```

```python
def alternating_optimize(a_init, g_init, time_budget, verbose=False):
    a = a_init.copy()
    g = g_init.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iter_num = 0
    
    while time.monotonic() - start < time_budget:
        # Solve for g given a
        g_new, t_g = solve_lp(a)
        if g_new is None:
            break
        g = g_new
        
        # Solve for a_new given g
        a_new, t_a = solve_lp(g)
        if a_new is None:
            break
        
        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in [1.0, 0.8, 0.6, 0.5, 0.3, 0.2, 0.1]:
            a_try = (1-t)*a + t*a_new
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t > 0:
            a = (1-best_t)*a + best_t*a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        iter_num += 1
        if verbose and iter_num % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"  ALP it{iter_num}: obj={best_obj:.8f}, t_a={t_a:.8f}, t_g={t_g:.8f}, elapsed={elapsed:.1f}s")
        
        if stall > 40:
            # Perturb
            rng = np.random.default_rng(iter_num * 2654435761 + n)
            scale = 0.05 + 0.02 * (stall // 40)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.3), n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            g = a.copy()
            best_a = a.copy()
            best_obj = compute_obj(a)
            stall = 0
    
    return best_a
```

Let me also think about a completely different approach: **using the structure of the problem to construct good sequences analytically**.

For the problem min max_k conv(a,a)[k] s.t. sum a = 1, a >= 0:

If we think of a as a probability distribution on {0, ..., n-1}, then conv(a,a)[k] is the probability that the sum of two independent random variables (each distributed as a) equals k.

So we want to minimize the maximum point probability of the sum of two iid random variables.

This is related to the "sumset" problem in additive combinatorics. For a uniform distribution on {0, ..., n-1}, the sum is triangular (max probability 1/n at the center).

To make the sum distribution flatter, we need a distribution whose "additive energy" is spread out.

One idea: use a distribution that is uniform on a "fractal" subset. For example, if n = 3^m, use the Cantor-like distribution. But this might create gaps in the convolution.

Another idea: use a distribution that is the squared magnitude of a complex exponential (which is 1, i.e., uniform) - doesn't help.

Actually, I want to add one more technique that I haven't tried: **using the dual solution to identify "critical" bins and focus optimization there**.

When we solve the LP, the dual variables tell us which convolution bins are most important (have the largest dual values). We can use this to:
1. Focus subgradient steps on the critical bins
2. Detect when we're at a "bottleneck" (a few bins dominate the max)
3. Guide perturbations to target the critical bins

Here's how: after solving LP for a given g, the dual gives us weights y[k] for each bin k. The bins with large y[k] are the "bottleneck" bins. We can then:
- Compute conv(a,a) at those specific bins
- Try to reduce them specifically

This is a form of "successive over-relaxation" or "Gauss-Seidel" on the active set.

Let me implement this as a "targeted subgradient" step:

```python
def targeted_subgradient(a, critical_bins, step_size):
    """Take a subgradient step targeting specific convolution bins."""
    n = len(a)
    conv = compute_conv(a)
    g = np.zeros(n)
    for k in critical_bins:
        k = int(k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end+1] += a[j_start:j_end+1][::-1]
    
    a_new = a - step_size * g
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new
```

where the sum is over valid i (0 <= i < n, 0 <= k-i < n).

For a given k, the valid i range is:
- i >= 0
- i < n
- i <= k (from k-i >= 0)
- i >= k-n+1 (from k-i <= n-1)

So i ranges from max(0, k-n+1) to min(n-1, k).

The matrix M has:
M[k, i] = g[k-i] if max(0, k-n+1) <= i <= min(n-1, k), else 0.

For column i:
M[k, i] = g[k-i] if 0 <= k-i < n, i.e., i <= k <= i+n-1.
So for column i, the non-zero entries are at rows k = i, i+1, ..., i+n-1.
