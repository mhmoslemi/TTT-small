"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Theoretical minimum is 2 (achieved by uniform sequences).
Uses LP alternating, MWU, and direct optimization approaches.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. A@g <= M, sum(g) = 1, g >= 0.
    Returns (g, M_opt).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None

    """LP alternating optimization."""
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try several mixing parameters
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))

            a_try = a_try / s_try

            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1

def mwu_optimize(a, time_budget=100.0, top_k=10, verbose=False):
    """Multiplicative Weight Update optimization."""
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_indices = np.argsort(conv)[::-1][:top_k]

        # Compute gradient
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            # grad[j] += a[k-j] for valid j
            for j in range(j_start, j_end):
                grad[j] += a[k - j] if 0 <= k - j < n else 0
        grad /= len(k_indices)

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        else:
            a = best_a.copy()
            eta *= 0.8
            if eta < 0.001:
                break

    return best_a

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def build_conv_matrix(a):
    """A @ g = conv(a, g) where A is (2n-1) x n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a):
    """Minimize M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except:
            continue
    return None, None

def lp_alternate(a, time_budget, verbose=False):
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        g, M = solve_lp(a)
        if g is None:
            break
        
        # Line search on mixing parameter
        best_t_obj = np.inf
        best_t = 0.5
        best_local = best_a.copy()
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1-t)*a + t*g
            a_try = np.maximum(a_try, 0)
            st = np.sum(a_try)
            if st < 1e-10:
                continue
            a_try /= st
            obj = compute_obj(a_try)
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
                best_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return best_a

For uniform a: conv(a,g)[k] = sum_j (1/n)*g[k-j] = (1/n) * sum_j g[k-j] = (1/n) * (sum of g over the valid range)

So the constraint is sum_j g[k-j] <= 1 for all k. In particular, for k where the full range is valid (k = n-1), this gives sum(g) <= 1. So the maximum sum(g) is 1, achieved by any g with sum = 1.

So for uniform a, the LP gives any g with sum <= 1. That's not very informative.

For non-uniform a, the LP will give a specific g that's shaped to avoid the peaks. This is where the method is powerful.

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))^2` for non-negative sequences. 

Key insight: The convolution `conv(a,a)` is the autocorrelation. The uniform sequence gives objective = 2, but by making the autocorrelation "flat" (all lags approximately equal), we can approach the theoretical infimum of `2n/(2n-1)` ≈ 1. The LP alternating method from the SOTA description is the right approach - it finds sequences whose convolution with the current iterate stays below the current peak, effectively pushing the sequence towards flatter autocorrelation.

I'll implement the LP alternating with the correct formulation (maximize sum(g) subject to conv(a,g) ≤ M where M is current peak), combined with subgradient polish and multiple restarts.

"""
LP-Alternating optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the SOTA LP formulation (max sum(g) s.t. conv(a,g)<=M) with
subgradient polish, multiple restarts, and adaptive mixing.
Theoretical infimum is 2n/(2n-1) ~ 1 for large n.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_max_sum(a, M):
    """
    SOTA LP: Maximize sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns g (unnormalized) or None.
    """
    n = len(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                if np.sum(g_val) > 1e-10:
                    return g_val
        except Exception:
            continue

    # Fallback with scipy
    try:
        from scipy.optimize import linprog
        c = -np.ones(n)
        A_ub = A
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x, 0)
            if np.sum(g_val) > 1e-10:
                return g_val
    except Exception:
        pass

    return None


def solve_lp_min_peak(a):
    """
    Alternative LP: Minimize M s.t. conv(a,g)[k] <= M, sum(g)=1, g >= 0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate_sota(a, time_budget=200.0, verbose=False):
    """
    SOTA LP alternating: max sum(g) s.t. conv(a,g) <= M, then mix.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    t_mix = 0.3

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        g = solve_lp_max_sum(a, M)
        if g is None:
            break

        # Rescale g to match sum(a)
        gs = float(np.sum(g))
        if gs < 1e-10:
            break
        g = g / gs  # Now sum(g) = 1, same as sum(a)

        # Adaptive line search
        obj_before = 2.0 * n * M
        best_t_obj = np.inf
        best_t = 0.3
        best_local = best_a.copy()

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)

            if st < 1e-10:

            a_try = a_try / st
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try < best_t_obj:

                best_t = t
                best_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_local.copy()
            a = best_a
            stall = 0
            # Adapt mixing
            if best_t < 0.2:
                t_mix = max(0.05, t_mix * 0.8)
            elif best_t > 0.5:
                t_mix = min(1.0, t_mix * 1.3)
            else:
                t_mix = best_t
            if verbose:

        else:
            stall += 1
            if stall % 10 == 0 and verbose:
                print(f"    LP-SOTA stall={stall}, obj={best_obj:.8f}")
            if stall > 50:
                break

    return best_a

def subgradient_polish(a, time_budget=100.0, verbose=False):
    """
    Subgradient method with momentum for the max-conv objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.05
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]

        # Subgradient of max(conv) w.r.t. a at the peak
        # d/d(a[j]) conv[k] = a[k-j] + a[j-k] (with boundary handling)
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            # conv[k] = sum_i a[i]*a[k-i], d/d a[j] = a[k-j] + a[j-k] if both valid
            if 0 <= k_star - j < n:
                val += a[k_star - j]
            if 0 <= j - k_star < n and j - k_star != k_star - j:
                val += a[j - k_star]
            grad[j] = val

        # Center gradient to project onto sum=const
        grad_centered = grad - np.mean(grad)

        # Update with momentum
        momentum = beta * momentum + lr * grad_centered
        a_new = a - momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        it += 1

        # Decay learning rate
        if it % 200 == 0:
            lr *= 0.95

    return best_a


def random_init(n, rng):
    """Generate a random non-negative sequence with interesting structure."""
    method = rng.integers(0, 4)
    if method == 0:
        # Smooth random (low-passed white noise)
        a = np.abs(rng.normal(0, 1, n))
        # Apply simple moving average
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif method == 1:
        # Cosine-based
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        a = np.abs(np.cos(2 * np.pi * freq * x + phase)) + 0.1
    elif method == 2:
        # Exponential decay pairs
        a = np.zeros(n)
        n_center = n // 2
        width = rng.integers(n // 4, n // 2 + 1)
        a[max(0, n_center - width):min(n, n_center + width)] = 1.0
        a += np.abs(rng.normal(0, 0.3, n))
    else:
        # Random with low autocorrelation bias
        a = rng.random(n)
        # Subtract mean to reduce DC, then take abs
        a = np.abs(a - np.mean(a)) + 0.01

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a
