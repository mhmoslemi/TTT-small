    size = 1
    while size < 2 * n - 1:
        size <<= 1

    return irfft(fa * fa, size)[:2 * n - 1]

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    n = len(a)

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, max_iter=100):

    A is the conv matrix for sequence a.
    Returns (z, M, dual_y) where dual_y are dual vars for A@z <= M.
    """

    A = build_conv_matrix(a)

    constr = [A @ z <= M, cp.sum(z) == 1]

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@a_conv(z) <= M, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass

    return None, None, None


def solve_lp_reduced(a, active_mask):
    """
    Solve LP with only active constraints (reduced problem).
    Returns (z_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    A_red = A[active_mask, :]

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    prob = cp.Problem(cp.Minimize(M), [A_red @ z <= M, cp.sum(z) == 1])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except:
            continue
    return None, None


def lp_robust_alternating(a, time_budget=200.0, verbose=False):
    """
    LP alternating with:
    - 2-cycle detection and breaking
    - Adaptive mixing ratio
    - Momentum smoothing
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = None
    prev_prev_a = None
    g_momentum = a.copy()
    alpha_adapt = 0.5
    stall = 0

    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result

        # 2-cycle detection
        if prev_prev_a is not None:
            diff = np.max(np.abs(a - prev_prev_a))
            if diff < 1e-7:
                # Breaking 2-cycle: average the two oscillating points
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                a_break = a_break / np.sum(a_break)
                obj_break = compute_obj(a_break)
                if obj_break < best_obj - 1e-12:
                    best_obj = obj_break
                    best_a = a_break.copy()
                a = a_break
                result = solve_lp(a)
                if result[0] is not None:
                    g_lp, M_lp, dual_y = result
                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()

        # Update momentum
        g_momentum = 0.6 * g_momentum + 0.4 * g_lp
        g_momentum = np.maximum(g_momentum, 0)
        gs = float(np.sum(g_momentum))
        if gs > 1e-10:
            g_momentum = g_momentum / gs

        # Adaptive mixing: try multiple target points and ratios
        targets = [g_lp, g_momentum]
        best_cand_obj = np.inf
        best_cand = a.copy()

        for target in targets:
            for mix in [0.3, 0.5, 0.7, 1.0]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_a = a.copy()
            prev_prev_a = prev_a
            a = best_a
            stall = 0
            # Adapt mixing: if large steps work, be more aggressive
            alpha_adapt = min(alpha_adapt * 1.02, 0.9)
            if verbose:
                elapsed = time.monotonic() - start
                if int(elapsed * 2) % 10 == 0:
                    print(f"    LP-robust: obj={best_obj:.8f}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 15:
                # Dual-guided perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                if dual_y is not None:
                    # Find which coordinates contribute most to active peaks
                    A = build_conv_matrix(a)
                    active_idx = np.where(dual_y > 0.01 * np.max(dual_y))[0]
                    if len(active_idx) > 0:
                        # Column sums over active rows = importance of each coord
                        col_imp = np.sum(A[active_idx, :], axis=0)
                        # Perturb high-importance coords downward, low upward
                        pert = -np.sign(col_imp - np.mean(col_imp)) * 0.02
                        pert += rng.normal(0, 0.01, n)
                    else:
                        pert = rng.normal(0, 0.03, n)
                else:
                    pert = rng.normal(0, 0.03, n)

                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()
                stall = 0
                alpha_adapt = 0.5
                if verbose:
                    print(f"    [Perturbed] obj={best_obj:.8f}")
            else:
                # Small step
                a_try = a + 0.1 * (g_lp - a)
                a_try = np.maximum(a_try, 0)
                a_try = a_try / np.sum(a_try)
                if compute_obj(a_try) < best_obj:
                    a = a_try

    return best_a


def active_set_loop(a, time_budget=100.0):
    """
    Iterate: solve full LP -> get active set -> solve reduced LP -> verify -> update.
    The reduced LP has fewer constraints and may find a better local solution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_full, M_full, dual_y = result

        if dual_y is not None and np.max(dual_y) > 0:
            active_mask = dual_y > 1e-5
            n_active = int(np.sum(active_mask))

            if 0 < n_active < 2 * n - 1:
                z_red, M_red = solve_lp_reduced(a, active_mask)
                if z_red is not None:
                    # Verify: check ALL convolution constraints
                    conv_all = compute_conv(a * z_red)
                    max_all = float(np.max(conv_all))
                    if max_all <= M_full * 1.005:
                        obj_red = compute_obj(z_red)
                        if obj_red < best_obj - 1e-12:
                            best_obj = obj_red
                            best_a = z_red.copy()
                            a = z_red
                            continue

        # Standard update
        a_new = 0.4 * a + 0.6 * g_full
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            improved = False
            for step in [0.1, 0.2, 0.3]:
                a_try = (1 - step) * a + step * g_full
                a_try = np.maximum(a_try, 0)
                a_try = a_try / np.sum(a_try)
                if compute_obj(a_try) < best_obj - 1e-12:
                    a = a_try
                    improved = True
                    break
            if not improved:
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def pairwise_exchange(a, time_budget=40.0):
    """
    Pairwise exchange polish: try swapping mass between pairs of coordinates.
    More effective than single-coordinate moves.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            for frac in [0.05, 0.1, 0.2, 0.02]:
                delta = frac * min(a[i], a[j])
                if delta > 0.5 * min(a[i], a[j]):
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and it > 30:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining:
    1. LP robust alternating (2-cycle breaking + momentum)
    2. Active set reduction
    3. Multi-n exploration
    4. Pairwise exchange polish
    5. Dual-guided random restarts
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Try structured initializations
        for n_init in [400, 600, 800, 1000, 1200]:
            t = np.linspace(0, 1, n_init)
            # Various window shapes
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),           # raised cosine
                np.sin(np.pi * t) ** 2,                       # sine squared
                1.0 - 0.3 * np.cos(4 * np.pi * t),           # slightly modulated
            ]
            for w in windows:
                a_init = np.maximum(w, 0.01)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                if obj_init < best_score:
                    best_score = obj_init
                    best_seq = a_init.copy()

        if best_seq is None:
            best_seq = np.ones(800) / 800
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    a = best_seq.copy()
    n_parent = len(a)

    # === Phase 1: LP Robust Alternating (400s) ===
    print(f"\n=== Phase 1: LP Robust (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(400.0, remaining * 0.33)
    a_new = lp_robust_alternating(a, time_budget=alloc, verbose=True)
    obj = compute_obj(a_new)
    if obj < best_score:
        best_score = obj
        best_seq = a_new.copy()
        a = a_new.copy()
        print(f"  Phase 1 done: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1 done: obj={obj:.8f}")

    # === Phase 2: Active Set Loop (200s) ===
    print(f"\n=== Phase 2: Active Set (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(200.0, remaining * 0.2)
        a_new = active_set_loop(a, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj:.8f}")

    # === Phase 3: Multi-n Exploration (300s) ===
    print(f"\n=== Phase 3: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200, n_parent + 400,
        500, 600, 700, 900, 1000, 1100, 1200,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2000]
    # Sort by proximity to n_parent to try nearby first
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(50.0, remaining * 0.15)
        a_opt = lp_robust_alternating(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Targeted LP on best (250s) ===
    print(f"\n=== Phase 4: Targeted LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(250.0, remaining * 0.25)
        a_new = lp_robust_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 4 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 done: obj={obj:.8f}")

        # Active set round
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(100.0, remaining * 0.2)
            a_new = active_set_loop(a, time_budget=alloc)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  Active set: obj={best_score:.8f} **")

    # === Phase 5: Pairwise Exchange Polish (60s) ===
    print(f"\n=== Phase 5: Pairwise Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.2)
        a_new = pairwise_exchange(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Polish: obj={best_score:.8f} **")
        else:
            print(f"  Polish: obj={obj:.8f}")

    # === Phase 6: Random Restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(6, int(remaining // 40))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 3 == 0:
                # Gaussian
                a_r = best_seq * np.exp(rng.normal(0, 0.04 + 0.01 * r, n_best))
            elif r % 3 == 1:
                # Block
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 100)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(3, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.5, 1.5)
            else:
                # Sparse
                a_r = best_seq.copy()
                n_sparse = max(5, n_best // 30)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.3, 2.0, n_sparse)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.35, 35.0)
            a_opt = lp_robust_alternating(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final LP polish on absolute best ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 30.0)
        a_final = lp_robust_alternating(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid for the evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
