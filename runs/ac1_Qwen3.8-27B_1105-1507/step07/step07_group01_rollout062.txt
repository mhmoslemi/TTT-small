import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    return smooth_val, weights

        P = np.abs(A) ** 2

        # Target: flat spectrum (preserve DC)
        P_flat = np.ones_like(P) * P[0]

        # Blend current with target
        # P_desired = P * (P_flat / (P + eps))^alpha
        # This compresses values above average, expands below
        eps = 1e-15
        ratio = P_flat / (P + eps)
        P_desired = P * (ratio ** alpha)

        # Preserve DC exactly
        P_desired[0] = P[0]

        # Reconstruct with new amplitude, same phase
        mag = np.sqrt(P)
        phase = np.zeros_like(A)
        nonzero = mag > 1e-15
        phase[nonzero] = A[nonzero] / mag[nonzero]
        phase[~nonzero] = 1.0

        A_new = np.sqrt(P_desired) * phase
        a_new = irfft(A_new, n_fft)[:n]

        # Project to non-negative, unit sum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 50 == 0:
                print(f"    Spectral it{it}: obj={best_obj:.8f}, alpha={alpha:.4f}")
        else:
            # If no improvement, reduce alpha (less aggressive)
            alpha = max(0.05, alpha * 0.98)

        # Anneal alpha
        if it % 20 == 0:
            alpha = max(0.05, alpha * 0.99)

    return best_a


def spectral_shaping_v2(a, time_budget=60.0, verbose=False):
    """
    Improved spectral shaping: uses log-domain compression and
    adaptive target spectrum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_fft = 1
    while n_fft < 2 * n - 1:
        n_fft <<= 1

    # Work in log domain for better dynamics
    beta = 0.3  # compression exponent (0 = no compression, 1 = perfectly flat)
    
    for it in range(400):
        if time.monotonic() - start > time_budget:
            break

        A = rfft(a, n_fft)
        P = np.abs(A) ** 2

        # Log-compress: exp(beta * log(P)) = P^beta
        P_max = np.max(P[1:]) if len(P) > 1 else P[0]
        P_min = np.min(P[P > 1e-20]) if np.any(P > 1e-20) else 1e-20

        # Compress in log domain
        log_P = np.log(P + 1e-20)
        log_P_flat = np.log(P_max + 1e-20)  # target: uniform at max level
        log_P_desired = (1 - beta) * log_P + beta * log_P_flat
        P_desired = np.exp(log_P_desired)

        # Preserve DC
        P_desired[0] = P[0]

        # Reconstruct
        mag = np.sqrt(P + 1e-30)
        phase = np.ones_like(A, dtype=complex)
        nonzero = mag > 1e-15
        phase[nonzero] = A[nonzero] / mag[nonzero]

        A_new = np.sqrt(P_desired + 1e-30) * phase
        a_new = irfft(A_new, n_fft)[:n]

        # Project
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 50 == 0:
                print(f"    Spectral2 it{it}: obj={best_obj:.8f}, beta={beta:.4f}")

        # Anneal
        if it % 10 == 0:
            beta = min(0.8, beta * 1.01)  # gradually increase compression
            # Actually, we want to decrease it over time for fine-tuning
            beta = max(0.1, beta * 0.995)

    return best_a


def lp_alternating(a, time_budget=60.0, verbose=False):
    """
    LP alternating optimization with adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_t = 0.3

    max_iters = 200
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Line search over t
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        # Adaptive t candidates
        ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        if 0 < last_t < 1.0:
            ts = sorted(set(ts + [max(0.01, last_t * 0.5), min(0.99, last_t * 1.5)]))

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall_count = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall_count += 1
            if stall_count > 40:
                # Perturb
                rng = np.random.default_rng(it * 7919 + n)
                scale = 0.02
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                last_t = 0.3

    return best_a


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)

    # Try cvxpy first
    try:
        import cvxpy as cp
        A = build_conv_matrix(a)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solvers = ['CLARABEL', 'ECOS', 'SCS']
        for solver in solvers:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def smooth_max_gradient_descent(a, time_budget=40.0, T_init=0.001, T_min=0.0001):
    """
    Gradient descent using smooth max (log-sum-exp) with Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.003
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    consecutive_no_improve = 0

    max_iters = 3000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        smooth_val, weights = smooth_max_and_weights(conv, T)

        # Gradient: grad[i] = sum_k weights[k] * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = 2*a[k-i] (for i != k-i) or a[i] (for i = k-i)
        # Approximation: grad = 2 * correlate(weights, a, 'valid')
        # with correction for diagonal
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        # Diagonal correction: for index i, subtract weights[2*i]*a[i] if 2*i < len(conv)
        for i in range(n):
            idx = 2 * i
            if idx < len(conv):
                grad[i] -= weights[idx] * a[i]

        # Center for simplex constraint
        grad_centered = grad - np.mean(grad)

        # Nesterov
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + 0.5 * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            consecutive_no_improve = 0
        else:
            stall += 1
            consecutive_no_improve += 1
            if consecutive_no_improve > 50:
                lr *= 0.85
                if lr < 1e-6:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                consecutive_no_improve = 0

        # Temperature anneal
        if it > 20:
            T = max(T_min, T_init * (0.998 ** it))

    return best_a


def subgradient_chase(a, time_budget=30.0):
    """
    Chase all top-m peaks simultaneously with weighted subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.01
    v = np.zeros(n)
    momentum = 0.7
    top_m = max(3, n // 20)
    stall = 0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)

        # Select top-m peaks
        top_indices = np.argsort(conv)[-top_m:]
        peak_weights = np.zeros(len(conv))
        max_c = conv.max()
        for idx in top_indices:
            peak_weights[idx] = 0.5 ** ((conv[idx] - max_c) / (0.01 * max_c + 1e-10))

        # Normalize
        pw_sum = np.sum(peak_weights)
        if pw_sum > 0:
            peak_weights = peak_weights / pw_sum
        else:
            peak_weights = np.zeros_like(conv)
            peak_weights[np.argmax(conv)] = 1.0

        # Gradient: sum over selected peaks
        grad = np.zeros(n)
        for idx in top_indices:
            k = int(idx)
            w = peak_weights[k]
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w * a[j]
                    if i != j:  # double count correction (i != k-i)
                        grad[i] += w * a[j]
                    # If i == j (i.e., k = 2i), we've already added once, correct
                    # Actually: d conv[k]/d a[i] = a[k-i] + a[k-i] if k-i != i, else a[i]
                    # = 2*a[k-i] if k != 2i, else a[i]
                    # We added a[j] once for first term and a[j] once for second
                    # If k == 2i, j = k-i = i, so we added a[i] twice but should add once
                    # So subtract if k == 2i
                    if k == 2 * i:
                        grad[i] -= w * a[j]

        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + 0.5 * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence normalized to sum 1."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = rng.dirichlet(np.ones(n))
    elif mode == 3:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.1
        a = a / np.sum(a)
    elif mode == 4:
        # Random smooth (Gaussian kernel)
        a = rng.normal(0, 1, n)
        # Smooth
        kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2) ** 2)
        a = np.convolve(a, kernel, mode='same')
        a = np.abs(a)
        a = a / np.sum(a)
    else:
        a = rng.beta(2, 2, n)
        a = a / np.sum(a)
    return a

"""
Spectral-shaping + LP alternating + smooth-max gradient descent + multiplicative weights
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: frequency-domain spectral flattening for fast initial convergence,
fixed O(n) gradient via correlate, multiplicative (mirror) descent for non-negativity,
and adaptive cycle scheduling across multiple optimization strategies.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def smooth_max_and_weights(conv, T):
    """Compute smooth max value and softmax weights."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    return smooth_val, weights


def spectral_shaping(a, time_budget=60.0, verbose=False):
    """
    Spectral domain optimization: flatten power spectrum to reduce autocorrelation peaks.
    Uses log-domain compression with adaptive parameters.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_fft = 1
    while n_fft < 2 * n - 1:
        n_fft <<= 1

    beta = 0.3
    stall = 0

    for it in range(600):
        if time.monotonic() - start > time_budget:
            break

        A = rfft(a, n_fft)
        P = np.abs(A) ** 2

        # Log-domain compression toward flat
        P_safe = np.maximum(P, 1e-20)
        log_P = np.log(P_safe)
        # Target: flat at geometric mean level (preserves energy better)
        log_P_target = np.full_like(log_P, np.mean(log_P[1:]) if len(log_P) > 1 else log_P[0])
        log_P_desired = (1.0 - beta) * log_P + beta * log_P_target
        P_desired = np.exp(log_P_desired)

        P_desired[0] = P[0]

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)

    try:
        import cvxpy as cp
        A = build_conv_matrix(a)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solvers = ['CLARABEL', 'ECOS', 'SCS']
        for solver in solvers:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass

    try:
        from scipy.optimize import linprog
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

            s = float(np.sum(g_val))

        ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        if 0 < last_t < 1.0:
            ts = sorted(set([t for t in ts + [max(0.01, last_t * 0.6), min(0.99, last_t * 1.4)] if 0 < t <= 1.0]))

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall_count = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall_count += 1
            if stall_count > 35:
                rng = np.random.default_rng((it * 7919 + n) % (2**32))
                scale = 0.02 + 0.005 * min(stall_count // 35, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                last_t = 0.3

    return best_a


def smooth_max_gradient_descent(a, time_budget=40.0, T_init=0.001, T_min=0.0001):
    """
    Gradient descent using smooth max with Nesterov momentum.
    Gradient computed as 2*correlate(weights, a, 'valid') + centering.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.004
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        smooth_val, weights = smooth_max_and_weights(conv, T)

        # Gradient: 2 * sum_k weights[k] * a[k-i] = 2 * correlate(weights, a, 'valid')
        grad = 2.0 * np.correlate(weights, a, mode='valid')

        # Center for simplex
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + 0.5 * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                lr *= 0.8
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                no_improve = 0

        # Temperature anneal
        if it > 30:
            T = max(T_min, T_init * (0.997 ** it))

    return best_a


def multiplicative_descent(a, time_budget=40.0, T_init=0.001, T_min=0.0001):
    """
    Mirror descent (multiplicative weights) with smooth max.
    a_new[i] = a[i] * exp(-lr * grad[i]) / Z
    Naturally preserves non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.5
    no_improve = 0

    max_iters = 4000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        smooth_val, weights = smooth_max_and_weights(conv, T)

        # Gradient
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        # Multiplicative update
        log_update = -lr * grad_centered
        log_update -= np.max(log_update)  # numerical stability
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                lr *= 0.7
                if lr < 1e-5:
                    break
                a = best_a.copy()
                no_improve = 0

        # Temperature anneal
        if it > 30:
            T = max(T_min, T_init * (0.997 ** it))

    return best_a


def subgradient_chase(a, time_budget=30.0):
    """
    Chase top-m peaks simultaneously with weighted subgradient.
    Vectorized gradient computation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.008
    v = np.zeros(n)
    momentum = 0.75
    top_m = max(5, n // 15)
    no_improve = 0

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)

        # Weight top-m peaks with exponential falloff
        sorted_idx = np.argsort(conv)[::-1]
        top_indices = sorted_idx[:top_m]
        max_c = conv[top_indices[0]]

        # Exponential weights based on how close to max
        peak_vals = conv[top_indices]
        peak_w = np.exp(-(max_c - peak_vals) / (0.05 * max_c + 1e-10))
        peak_w = peak_w / np.sum(peak_w)

        # Build weighted weights vector (sparse)
        w = np.zeros(len(conv))
        for i, idx in enumerate(top_indices):
            w[idx] = peak_w[i]

        # Gradient: 2 * correlate(w, a, 'valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        # Momentum
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + 0.5 * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.8
                v = np.zeros(n)
                a = best_a.copy()
                no_improve = 0
                if lr < 1e-6:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence normalized to sum 1."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = rng.dirichlet(np.ones(n))
    elif mode == 3:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.1
        a = a / np.sum(a)
    elif mode == 4:
        a = rng.normal(0, 1, n)
        kernel_size = max(3, n // 50)
        kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size // 2 + 1)) ** 2)
        a = np.convolve(a, kernel, mode='same')
        a = np.abs(a)
        a = a / np.sum(a)
    else:
        a = rng.beta(2, 2, n)
        a = a / np.sum(a)
    return a
