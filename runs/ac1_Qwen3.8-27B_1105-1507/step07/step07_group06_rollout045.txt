def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        A[j:j + end, j] = a[:end]
    return A


def solve_lp_primal(a):
    """Solve primal LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_hankel_qp(p_weights, n):
    """Solve QP: min a^T P a s.t. sum(a)=1, a>=0 where P[i,j]=p_weights[i+j].
    Uses projected gradient if P is PSD, falls back to LP otherwise.
    Returns optimal a."""
    # Check if P is PSD by checking eigenvalues of a small sample
    P = np.zeros((n, n))
    for i in range(n):
        P[i, i:min(2*n-1, i+n)] = p_weights[i:i+min(n, 2*n-1-i)]
    # Fill the upper triangle
    for i in range(n):
        for j in range(i+1, n):
            if i + j < 2 * n - 1:
                P[i, j] = p_weights[i + j]
                P[j, i] = p_weights[i + j]

    # Quick PSD check via Cholesky
    is_psd = True
    try:
        np.linalg.cholesky(P + 1e-8 * np.eye(n))
    except np.linalg.LinAlgError:
        is_psd = False

    if is_psd:
        # Solve using projected gradient
        a = np.ones(n) / n
        step = 1.0 / (2.0 * np.max(np.abs(np.linalg.eigvalsh(P))))
        for _ in range(200):
            grad = 2.0 * P @ a
            grad_proj = grad - np.mean(grad)
            a_new = a - step * grad_proj
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s > 1e-10:
                a_new /= s
            if np.max(np.abs(a_new - a)) < 1e-10:
                break
            a = a_new
        return a
    else:
        # Fallback: solve LP min t s.t. a^T P a approximated... just use uniform
        # Actually, use the LP: minimize max_j (P@a)_j is not quite right
        # Just return uniform as fallback
        return np.ones(n) / n

def mirror_descent_minimax(a_init, time_budget=60.0, eta_start=1.0, eta_end=20.0, n_iterations=100):
    """
    Mirror descent / multiplicative weight update for the minimax formulation.
    min_a max_k (a*a)[k] = max_p min_a E_{k~p}[(a*a)[k]]
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initialize p as uniform
    p = np.ones(2 * n - 1) / (2 * n - 1)

    for it in range(n_iterations):
        if time.monotonic() - start > time_budget:
            break

        # Anneal eta
        if n_iterations > 1:
            frac = it / max(1, n_iterations - 1)
            eta = eta_start * (eta_end / eta_start) ** frac
        else:
            eta = eta_start

        # Update p: concentrate on peaks
        conv = compute_conv(a)
        p = p * np.exp(eta * conv)
        p = p / np.sum(p)

        # Solve inner QP: min a^T P a s.t. sum(a)=1, a>=0
        P = np.zeros((n, n))
        for i in range(n):
            j_max = min(n, 2 * n - 1 - i)
            P[i, i:i + j_max] = p[i:i + j_max]
        # Symmetrize
        for i in range(n):
            for j in range(i + 1, n):
                if i + j < 2 * n - 1:
                    val = p[i + j]
                    P[i, j] = val
                    P[j, i] = val

        # Check PSD and solve
        try:
            eigvals = np.linalg.eigvalsh(P)
            if eigvals[0] > -1e-8:
                # Use projected gradient
                P_pos = P + 1e-8 * np.eye(n)  # ensure positive definite
                a_qp = np.ones(n) / n
                L = 2.0 * np.max(np.abs(eigvals)) + 1e-8
                step = 1.0 / L
                for _ in range(100):
                    grad = 2.0 * P_pos @ a_qp
                    grad_proj = grad - np.mean(grad)
                    a_new = a_qp - step * grad_proj
                    a_new = np.maximum(a_new, 0)
                    s_new = np.sum(a_new)
                    if s_new > 1e-10:
                        a_new /= s_new
                    if np.max(np.abs(a_new - a_qp)) < 1e-12:
                        break
                    a_qp = a_new
            else:
                # P not PSD, use LP fallback: minimize max_j (P@a)_j
                # This is: min t s.t. P@a <= t*1, sum(a)=1, a>=0
                a_qp = a  # fallback
        except Exception:
            a_qp = a

        # Blend
        for t_blend in [1.0, 0.7, 0.5, 0.3]:
            a_try = (1.0 - t_blend) * a + t_blend * a_qp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = best_a
                break
        else:
            # No improvement, perturb
            if it % 10 == 0:
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    return best_a


def smoothed_lbfgs(a_init, tau_start, tau_end, time_budget, max_iter=200):
    """Smoothed L-BFGS-B optimization with tau annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    n_stages = 5
    taus = np.geomspace(tau_start, tau_end, n_stages)

    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        stage_budget = min(remaining * 0.8, 80.0)

        try:
            def obj_func(th):
                a = _softmax(th)
                conv = compute_conv(a)
                conv_max = np.max(conv)
                return 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)

            def grad_func(th):
                a = _softmax(th)
                conv = compute_conv(a)
                conv_max = np.max(conv)
                w = tau * np.exp((conv - conv_max) * tau)
                w = w / np.sum(w)
                c = np.zeros(n)
                for j in range(n):
                    end = min(2 * n - 1, j + n)
                    c[j] = np.dot(w[j:end], a[:end - j])
                mean_c = np.dot(c, a)
                return 4.0 * n * a * (c - mean_c)

            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-7, 'maxfun': 400})
            if result.x is not None:
                a_new = _softmax(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def _softmax(theta):
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smart_perturb(a, rng, scale=0.05):
    """Apply a smart perturbation that preserves the overall shape."""
    n = len(a)
    # Low-frequency perturbation
    freq = rng.integers(1, max(2, n // 100))
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = 1.0 + scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    a_new = a * perturbation
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    # Phase 2: Mirror Descent Minimax

    remaining = deadline - time.monotonic()
    if remaining > 40:
        for eta_end in [10.0, 20.0, 50.0]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.15, 60.0)
            a_md = mirror_descent_minimax(best_seq, time_budget=alloc,
                                          eta_start=0.5, eta_end=eta_end,
                                          n_iterations=80)
            obj_md = compute_obj(a_md)
            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                n_parent = len(best_seq)
                print(f"  MD eta_end={eta_end}: obj={best_score:.8f} **")
            else:
                print(f"  MD eta_end={eta_end}: obj={obj_md:.8f}")

    # Phase 3: Smoothed L-BFGS-B
    print(f"\n=== Phase 3: Smoothed L-BFGS-B ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    for trial in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if trial == 0:
            a_init = best_seq.copy()
        elif trial == 1:
            a_init = smart_perturb(best_seq, rng, scale=0.03)
        else:
            # Asymmetric ramp
            ramp = np.linspace(0.5, 1.5, n_best)
            a_init = best_seq * ramp
            a_init = np.maximum(a_init, 0)
            a_init = a_init / np.sum(a_init)

        alloc = min(remaining * 0.15, 80.0)
        a_opt, obj_opt = smoothed_lbfgs(a_init, tau_start=10.0, tau_end=500.0,
                                         time_budget=alloc, max_iter=150)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Trial {trial}: obj={best_score:.8f} **")
        else:
            print(f"  Trial {trial}: obj={obj_opt:.8f}")

    # Phase 4: Multi-n exploration
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 100), max(300, n_cur - 200),
        n_cur,
        n_cur + 100, n_cur + 200, n_cur + 400,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_cur:
            continue

        a_try = resample_sequence(best_seq, n_try)
        a_try = np.maximum(a_try, 1e-10)
        a_try = a_try / np.sum(a_try)

        # Quick LP optimization
        alloc = min(remaining * 0.15, 50.0)
        a_opt = lp_alternating_improved(a_try, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 5: Random restarts
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        init_type = r % 4
        if init_type == 0:
            # Linear ramp
            slope = rng.uniform(0.3, 3.0)
            a_r = np.linspace(1.0, slope, n_best)
            a_r = a_r / np.sum(a_r)
        elif init_type == 1:
            # Exponential
            rate = rng.uniform(0.5, 4.0)
            a_r = np.exp(np.linspace(0, rate, n_best))
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            # Blend with parent
            blend = rng.uniform(0.2, 0.6)
            noise = np.exp(rng.normal(0, 0.3, n_best))
            a_r = blend * best_seq + (1 - blend) * noise
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            # Two-segment (step-like asymmetry)
            split = rng.integers(n_best // 3, 2 * n_best // 3)
            a_r = np.ones(n_best)
            a_r[:split] = rng.uniform(0.3, 2.0)
            a_r[split:] = rng.uniform(0.3, 2.0)
            a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.1, 30.0)
        a_opt = lp_alternating_improved(a_r, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (type={init_type}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_opt:.8f}")

    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        # Extended LP alternating
        alloc = min(remaining * 0.35, 120.0)
        a_lp = lp_alternating_improved(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Mirror descent polish
        alloc = min(remaining * 0.25, 60.0)
        a_md = mirror_descent_minimax(best_seq, time_budget=alloc,
                                       eta_start=1.0, eta_end=30.0,
                                       n_iterations=100)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")
        else:
            print(f"  Final MD: obj={obj_md:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more smoothed L-BFGS with high tau
        alloc = min(remaining * 0.35, 40.0)
        a_opt, obj_opt = smoothed_lbfgs(best_seq, tau_start=50.0, tau_end=1000.0,
                                         time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")
        else:
            print(f"  Final Smooth: obj={obj_opt:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
