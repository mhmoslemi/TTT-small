    # Term 2 for j in [0, n-1]: conv_aw[j]
    term2 = conv_aw[:n]
    # Term 1 for j in [0, n-1]: conv_awf[2n-2-j]
    term1 = conv_awf[2*n-2 : n-1 : -1]  # indices 2n-2, 2n-3, ..., n-1
    
    grad = term1 + term2
    return grad, b

def mirror_descent_peak(a, time_budget=30.0, verbose=False):
    """
    Mirror descent (multiplicative updates) targeting the peak of auto-convolution.
    a_j <- a_j * exp(-eta * s[j]) where s[j] is subgradient at peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    eta = 0.1
    stall = 0
    
    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))
        
        # Compute subgradient
        s_j = np.zeros(n)
        for j in range(n):
            val = 0.0
            kminusj = k_star - j
            if 0 <= kminusj < n:
                val += a[kminusj]
            jminusk = j - k_star
            if 0 <= jminusk < n:
                val += a[jminusk]
            s_j[j] = val
        
        # Center the gradient (sum should be ~0 for normalization)
        s_j = s_j - np.mean(s_j)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * s_j)
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.02, 1.0)
        else:
            stall += 1
            eta *= 0.95
            if stall > 100:
                # Perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                eta = 0.1
                stall = 0
            if eta < 1e-8:
                break
        
        if verbose and it % 2000 == 0:
            print(f"          MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}")
    
    return best_a

def softmax_gd(a, time_budget=60.0, verbose=False):
    """
    Gradient descent using softmax approximation of max.
    Adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    # N2 for the weighted gradient computation
    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1
    
    tau = 50.0
    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        
        # Softmax weights
        b_max = np.max(b)
        w = np.exp(tau * (b - b_max))
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient
        fw = rfft(w, N2)
        fa2 = rfft(a, N2)
        conv_aw = irfft(fa2 * fw, N2)[:3 * n - 2]
        
        w_flip = w[::-1]
        fwf = rfft(w_flip, N2)
        conv_awf = irfft(fa2 * fwf, N2)[:3 * n - 2]
        
        term2 = conv_aw[:n]
        term1 = conv_awf[2*n-2 : n-1 : -1]
        grad = (term1 + term2) / tau  # The 1/tau from the softmax derivative
        
        # Center
        grad = grad - np.mean(grad)
        
        # Momentum
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new
        
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.003, 3.0)
            # Slowly increase tau (sharpen the peak focus)
            tau = min(tau * 1.001, 500.0)
            momentum = min(momentum + 0.0005, 0.92)
        else:
            stall += 1
            eta *= 0.98
            v *= 0.5
            tau = max(tau * 0.99, 5.0)
            if stall > 200:
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.7
                tau = 50.0
                stall = 0
            if eta < 1e-8:
                break
        
        if verbose and it % 5000 == 0:
            print(f"          SoftGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}")
    
    return best_a

def solve_lp_for_g(a, max_constraints=200):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for k in selected set, sum(g)=1, g>=0.
    Uses only a subset of constraints for speed with large n.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Select important constraints: those near the peak of a*a
    b = compute_conv(a)
    # Sort by value, take top max_constraints
    k_indices = np.argsort(b)[::-1][:max_constraints]
    k_indices = np.sort(k_indices)
    
    # Build constraint matrix
    m = len(k_indices)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for idx, k in enumerate(k_indices):
        for j in range(n):
            k_minus_j = k - j
            if 0 <= k_minus_j < n:
                A_ub[idx, j] = a[k_minus_j]
        A_ub[idx, n] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    try:
        from scipy.optimize import linprog
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g
    except Exception:
        pass
    
    # Fallback: try with fewer constraints
    if m > 50:
        return solve_lp_for_g(a, max_constraints=50)
    
    return None

def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing with adaptive constraint selection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = solve_lp_for_g(a, max_constraints=min(300, 2 * n - 1))
        if g_lp is None:
            break
        
        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 15:
                break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def periodic_init(n, period, width, rng):
    """Create a periodic-like initialization."""
    a = np.zeros(n)
    positions = list(range(0, n, period))
    for p in positions:
        for d in range(-width, width + 1):
            idx = p + d
            if 0 <= idx < n:
                a[idx] += np.exp(-d**2 / (width * 0.5))
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

"""
Hybrid optimizer combining softmax-approximated gradient descent, mirror descent,
sparse-LP polishing, coordinate-style perturbations, and multi-scale n exploration
to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def compute_obj_fast(a, N_cache=None):
    """Compute objective with cached FFT size."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    if N_cache is None:
        N_cache = 1
        while N_cache < 2 * n - 1:
            N_cache <<= 1
    fa = rfft(a_norm, N_cache)
    b = irfft(fa * fa, N_cache)[:2 * n - 1]
    return 2.0 * n * float(np.max(b))

    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj_fast(a)
    start = time.monotonic()

        N2 <<= 1

    eta = 0.3
    momentum = 0.6
    v = np.zeros(n)
    stall = 0
    max_stall = 300

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        # Compute b = a * a
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum

def mirror_descent_pulse(a, time_budget=30.0, verbose=False):

        b = irfft(fa * fa, N)[:2 * n - 1]

                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                v = np.zeros(n)
                eta_gd = 0.3
                eta_md = 0.05
                tau = 40.0
                stall = 0
            if eta_gd < 1e-9 and eta_md < 1e-8:
                break

        if verbose and it % 10000 == 0:
            print(f"          Hybrid it{it}: obj={best_obj:.8f}")

    return best_a


def solve_lp_sparse(a, max_constraints=150):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for top-k constraints, sum(g)=1, g>=0.
    Only uses the most binding constraints for speed.
    """
    n = len(a)
    if n > 1200:
        return None  # Too large for LP
    a = a / np.sum(a)

    b = compute_conv(a)
    # Select top constraints (highest b values)
    k_indices = np.argsort(b)[::-1][:min(max_constraints, 2 * n - 1)]
    k_indices = np.sort(k_indices)

    m = len(k_indices)
    A_ub = np.zeros((m, n + 1))

    for idx, k in enumerate(k_indices):
        # (a*g)[k] = sum_j a[k-j]*g[j] for valid k-j
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            A_ub[idx, j] = a[k - j]
        A_ub[idx, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    try:
        from scipy.optimize import linprog
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g
    except Exception:
        pass

    return None


def lp_polish_sparse(a, time_budget=60.0, verbose=False):
    """LP polishing using sparse constraints."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    stall = 0

    for it in range(80):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_sparse(a, max_constraints=min(150, 2 * n - 1))
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_fast(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 12:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=None):
    """Generate diverse random initializations."""
    if mode is None:
        mode = int(rng.integers(0, 7))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0 + 3.0 * (x - 0.5) ** 2
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.15
        a = a / np.sum(a)
    elif mode == 5:
        # Periodic peaks
        period = max(n // 8, 10)
        a = np.zeros(n)
        for p in range(0, n, period):
            a[p] = 1.0
        a += 0.1
        a = a / np.sum(a)
    else:
        # Smooth random
        a = np.exp(rng.normal(0, 0.5, n))
        a = a / np.sum(a)
    return a


def symmetrize(a):
    """Symmetrize: a = (a + a[::-1]) / 2."""
    return (a + a[::-1]) / 2.0


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines softmax GD, mirror descent, hybrid, LP polishing,
    and multi-n exploration with adaptive time allocation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj_fast(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj_fast(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Intensive softmax GD ===
    print(f"\n=== Phase 1: Softmax GD (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.20, 200.0)
    a_opt = softmax_gd(best_seq, time_budget=alloc, verbose=True)
    obj = compute_obj_fast(a_opt)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  ** SoftGD: {best_score:.8f} **")
    else:
        print(f"  SoftGD: {obj:.8f}")

    # === Phase 2: Hybrid (softmax + mirror) ===
    print(f"\n=== Phase 2: Hybrid (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.18, 150.0)
        a_opt = softmax_mirror_hybrid(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj_fast(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** Hybrid: {best_score:.8f} **")
        else:
            print(f"  Hybrid: {obj:.8f}")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.12, 80.0)
        a_opt = mirror_descent_pulse(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj_fast(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** MD: {best_score:.8f} **")
        else:
            print(f"  MD: {obj:.8f}")

    # === Phase 4: LP Polish ===
    print(f"\n=== Phase 4: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50 and len(best_seq) <= 1200:
        alloc = min(remaining * 0.15, 100.0)
        a_opt = lp_polish_sparse(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj_fast(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** LP: {best_score:.8f} **")
        else:
            print(f"  LP: {obj:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        n_best + 200, n_best + 400, n_best + 800,
        1000, 1200, 1500, 1800, 2000, 2500
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 3000 and n != n_best][:6]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc1 = min(remaining * 0.25, 50.0)
        a_opt = softmax_gd(a_try, time_budget=alloc1)
        obj = compute_obj_fast(a_opt)

        remaining2 = deadline - time.monotonic()
        if remaining2 > 25:
            alloc2 = min(remaining2 * 0.25, 30.0)
            a_opt2 = mirror_descent_pulse(a_opt, time_budget=alloc2)
            obj2 = compute_obj_fast(a_opt2)
            if obj2 < obj:
                obj = obj2
                a_opt = a_opt2

        remaining3 = deadline - time.monotonic()
        if remaining3 > 25 and n_try <= 1200:
            alloc3 = min(remaining3 * 0.2, 25.0)
            a_opt3 = lp_polish_sparse(a_opt, time_budget=alloc3)
            obj3 = compute_obj_fast(a_opt3)
            if obj3 < obj:
                obj = obj3
                a_opt = a_opt3

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 6: Second round on best ===
    print(f"\n=== Phase 6: Second Round ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:

        alloc = min(remaining * 0.25, 100.0)
        a_opt = softmax_gd(best_seq, time_budget=alloc)
        obj = compute_obj_fast(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** SoftGD2: {best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.2, 60.0)
            a_opt = softmax_mirror_hybrid(best_seq, time_budget=alloc)
            obj = compute_obj_fast(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** Hybrid2: {best_score:.8f} **")

        remaining = deadline - time.monotonic()

            alloc = min(remaining * 0.2, 40.0)
            a_opt = lp_polish_sparse(best_seq, time_budget=alloc)
            obj = compute_obj_fast(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** LP2: {best_score:.8f} **")

    # === Phase 7: Random restarts ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_curr = len(best_seq)
    n_restarts = max(2, min(6, int((deadline - time.monotonic()) / 50.0)))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_r = random_init(n_curr, rng, mode=r % 7)
        alloc = min(remaining * 0.2, 35.0)
        a_opt = softmax_gd(a_r, time_budget=alloc)
        obj = compute_obj_fast(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** Restart {r}: {best_score:.8f} **")
        else:
            print(f"  Restart {r}: {obj:.8f}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 80.0)
        a_final = softmax_gd(best_seq, time_budget=alloc * 0.5)
        obj_f = compute_obj_fast(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  ** Final SoftGD: {best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.4, 35.0)
            a_md = mirror_descent_pulse(best_seq, time_budget=alloc)
            obj_m = compute_obj_fast(a_md)
            if obj_m < best_score:
                best_score = obj_m
                best_seq = a_md.copy()
                print(f"  ** Final MD: {best_score:.8f} **")

        if remaining > 15 and len(best_seq) <= 1000:

                print(f"  ** Final LP: {best_score:.8f} **")

    val = evaluate_sequence(result)

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
