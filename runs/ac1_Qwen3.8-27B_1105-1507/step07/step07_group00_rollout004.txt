def project_simplex(v):

    return None, None, None

def solve_lp_max_sum(a, M_target):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target, g >= 0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M_target])
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except ImportError:
        pass

    return None

def dual_based_subgradient(a, dual_weights, n):
    """Compute subgradient using dual weights from LP."""
    g = np.zeros(n)
    for k, w in enumerate(dual_weights):
        if w > 1e-8:
            g += w * conv_subgradient(a, k, n)
    return g

def parabolic_line_search(a, g_dir, a_prev_obj, n, max_iter=10):
    """
    Parabolic line search: find best t in [0,1] for a + t*(g_dir - a).
    """
    best_t = 0.0
    best_obj = a_prev_obj
    best_a = a.copy()

    # Sample a few points
    ts = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    objs = []
    for t in ts:
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            objs.append(np.inf)
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        objs.append(obj)
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_a = a_try.copy()

    # Parabolic interpolation on best 3 points
    idx = np.argsort(objs)
    if len(idx) >= 3:
        i1, i2, i3 = idx[0], idx[1], idx[2]
        t1, t2, t3 = ts[i1], ts[i2], ts[i3]
        f1, f2, f3 = objs[i1], objs[i2], objs[i3]
        if f1 < np.inf and f2 < np.inf and f3 < np.inf:
            denom = (f1 * (t2 - t3) + f2 * (t3 - t1) + f3 * (t1 - t2))
            if abs(denom) > 1e-15:
                t_par = (f1 * t2 * t3 * (t2 - t3) + f2 * t3 * t1 * (t3 - t1) + f3 * t1 * t2 * (t1 - t2)) / denom
                if 0.01 < t_par < 0.99:
                    a_try = (1 - t_par) * a + t_par * g_dir
                    a_try = np.maximum(a_try, 0)
                    s = float(np.sum(a_try))
                    if s > 1e-10:
                        a_try = a_try / s
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_t = t_par
                            best_a = a_try.copy()

    return best_a, best_obj, best_t

def active_set_polish(a, time_budget=60.0, tol=1e-6):
    """
    Polish using active set: identify peaks near max, then do targeted reduction.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find peaks within tol of max
        active = np.where(conv >= max_val - tol * max_val)[0]

        if len(active) == 0:
            break

        # Compute weighted subgradient from active peaks
        weights = conv[active] - (max_val - tol * max_val)
        weights = np.maximum(weights, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(active))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, active):
            g += w * conv_subgradient(a, int(k), n)

        # Newton-like step: use the Hessian approximation B_k for active peaks
        # Try a damped step
        alpha = 0.1 / np.sqrt(n)
        improved = False
        for decay in [1.0, 0.5, 0.25, 0.1]:
            a_try = a - alpha * decay * g
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try moving mass from peak contributors to non-contributors
            k_star = int(active[0])
            contributors = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    c = a[i] * a[j]
                    contributors.append((c, i, j))
            contributors.sort(reverse=True)

            non_contrib = set(range(n))
            for _, i, j in contributors:
                non_contrib.discard(i)
                non_contrib.discard(j)
            non_contrib_list = sorted(non_contrib)

            if not non_contrib_list:
                break

            found = False
            for c, i, j in contributors[:10]:
                if c < 1e-10:
                    break
                for target in non_contrib_list[:5]:
                    for frac in [0.01, 0.03, 0.05, 0.1]:
                        delta = frac * a[i]
                        if delta <= 0:
                            continue
                        a[i] -= delta
                        a[target] += delta
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a.copy()
                            found = True
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[target] -= delta
                        if found:
                            break
                if found:
                    break

            if not found:
                # Reduce tolerance and continue
                tol *= 0.5
                if tol < 1e-9:
                    break

        it += 1

    return best_a

def lp_alternating_improved(a, time_budget=300.0, verbose=False):
    """
    Improved LP alternating with dual-based gradients and parabolic line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iter_count = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        # Solve LP
        g_lp, M_lp, dual = solve_lp_dual(a)
        if g_lp is None:
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = solve_lp_max_sum(a, M_cur)
            if g_lp is None:
                break

        # Use dual-based subgradient as alternative direction
        if dual is not None and np.sum(dual) > 1e-8:
            g_dual = dual_based_subgradient(a, dual, n)
            # Combine: use LP solution as primary, dual gradient for direction
            # Blend the two directions
            g_combined = 0.7 * g_lp + 0.3 * project_simplex(g_dual)
            g_combined = np.maximum(g_combined, 0)
            g_combined = g_combined / g_combined.sum()
        else:
            g_combined = g_lp

        # Parabolic line search
        a_try, obj_try, t_try = parabolic_line_search(a, g_combined, best_obj, n)

        iter_count += 1
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            last_t = t_try
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LPI it{iter_count}: obj={best_obj:.8f}, t={t_try:.3f}, elapsed={elapsed:.1f}s")
        else:
            # Try pure LP direction
            a_try2, obj_try2, t_try2 = parabolic_line_search(a, g_lp, best_obj, n)
            if obj_try2 < best_obj - 1e-12:
                best_obj = obj_try2
                best_a = a_try2.copy()
                a = a_try2.copy()
                last_t = t_try2
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    # Structured perturbation
                    conv = compute_conv(a)
                    k_star = int(np.argmax(conv))
                    # Create perturbation based on the peak structure
                    rng = np.random.default_rng(iter_count * 7919 + n)
                    scale = min(0.5, 0.05 + 0.03 * (stall // 20))
                    # Perturb more in the "contributor" region
                    contrib_mask = np.zeros(n)
                    for i in range(n):
                        j = k_star - i
                        if 0 <= j < n:
                            contrib_mask[i] = 1.0
                    contrib_mask = contrib_mask / contrib_mask.sum() if contrib_mask.sum() > 0 else 1.0/n
                    noise = rng.normal(0, scale, n) * (0.5 + contrib_mask)
                    a = best_a * np.exp(noise)
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    else:
                        a = best_a.copy()
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
                    last_t = 0.5

    return best_a

def mirror_descent_v2(a, time_budget=200.0, topK=5, verbose=False):
    """
    Improved mirror descent with adaptive top-K and annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n) * 1.5
    min_lr = 1e-5
    temperature = 3.0
    decay_temp = 0.999
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Adaptive top-K
        K = min(topK, max(2, len(conv) // max(1, n // 10)))
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Softmax weights
        weights = np.exp((peak_vals - max_val) * temperature / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            g += w * conv_subgradient(a, k, n)

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = max(min_lr, lr * 0.997)
            temperature = max(1.0, temperature * decay_temp)
        else:
            stall += 1
            if stall > 200:
                # Anneal: random perturbation
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.15, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(min_lr, lr * 0.6)
                temperature = 3.0
                stall = 0
            else:
                lr = max(min_lr, lr * 0.985)
                temperature = max(1.0, temperature * 0.995)

        it += 1

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import PchipInterpolator
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        interp = PchipInterpolator(x_old, a)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Inverted triangle
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # U-shaped (arcsine-like)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - 0.9 * x**2 + 0.1))
        a = a / a.sum()
    elif mode == 5:
        # Smooth random
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amplitudes = rng.uniform(0.1, 0.4, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amplitudes):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        a = np.arange(1, n + 1, dtype=float) ** -0.5
        a = a / a.sum()
    elif mode == 7:
        # Smoothed random (lowpass)
        a = rng.uniform(0, 1, n)
        # Simple moving average smoothing
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
    elif mode == 8:
        # Two-scale: large bumps with fine structure
        coarse = rng.uniform(0, 1, max(1, n // 20))
        coarse_full = np.interp(np.arange(n), np.arange(len(coarse)), coarse)
        fine = rng.normal(0, 0.1, n)
        a = coarse_full + fine
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def spectral_perturb(a, k_star, n, rng, scale=0.1):
    """
    Perturb a using the eigenvector structure of B_{k_star}.
    B_k has eigenvalues that are roots of unity related to the reversal structure.
    """
    # The matrix B_k for the center peak (k = n-1) is the anti-identity
    # Its eigenvectors are symmetric and antisymmetric combinations
    # We can perturb along these directions
    if k_star >= n - 1:
        # Symmetric perturbation
        sym = a + a[::-1]
        sym = sym / (2 * sym.sum())
        asym = a - a[::-1]
        # Scale asymmetry
        perturbation = scale * asym / (np.std(asym) + 1e-10)
        a_new = a + perturbation
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        return a_new
    return a

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization with:
    - LP alternating with dual-based gradients
    - Mirror descent with adaptive temperature
    - Active set polishing
    - Multi-n exploration
    - Structured restarts
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0  # Leave safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Mirror Descent (fast, good for initial improvement) ===
    print(f"\n=== Phase 1: Mirror Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    md_budget = min(remaining * 0.2, 180.0)
    if md_budget > 30:
        a_md = mirror_descent_v2(best_seq, time_budget=md_budget, topK=5, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 2: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    for rnd in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(remaining * 0.25, 300.0)
        if alloc < 40:
            break
        a_lp = lp_alternating_improved(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Active Set Polish ===
    print(f"\n=== Phase 3: Active Set Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        polish_budget = min(remaining * 0.1, 80.0)
        a_pol = active_set_polish(best_seq, time_budget=polish_budget)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Active set: obj={best_score:.8f} **")
        else:
            print(f"  Active set: obj={obj_pol:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-100, -50, 50, 100, 200, 300]:
        n_try = n_best + delta
        if 400 <= n_try <= 1500:
            n_options.append(n_try)
    # Add some fixed options
    for n_fixed in [500, 600, 700, 800, 900, 1000, 1100, 1200]:
        if n_fixed != n_best and 400 <= n_fixed <= 1500:
            n_options.append(n_fixed)

    n_options = sorted(set(n_options))

    for n_try in n_options[:10]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick MD then LP
        alloc = min(remaining * 0.15, 100.0)
        if alloc > 30:
            half = alloc / 2
            a_temp = mirror_descent_v2(a_try, time_budget=min(half, 40.0), topK=4)
            obj_temp = compute_obj(a_temp)
            if obj_temp < compute_obj(a_try):
                a_try = a_temp
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.1, 60.0)
            if lp_alloc > 20:
                a_opt = lp_alternating_improved(a_try, time_budget=lp_alloc)
                obj_opt = compute_obj(a_opt)
            else:
                a_opt = a_try
                obj_opt = compute_obj(a_opt)
        else:
            a_opt = a_try
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Structured Restarts ===
    print(f"\n=== Phase 5: Structured Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_restarts = 5

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = (r * 3) % 9  # Spread across modes
        a_r = diverse_init(n_best, rng, mode=mode)

        # Blend with best (more weight on best)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s = a_r.sum()
        if s > 1e-10:
            a_r = a_r / s

        restart_budget = min(remaining * 0.15, 100.0)
        # Quick MD then LP
        if restart_budget > 40:
            a_temp = mirror_descent_v2(a_r, time_budget=restart_budget * 0.3, topK=4)
            obj_temp = compute_obj(a_temp)
            if obj_temp < compute_obj(a_r):
                a_r = a_temp
            remaining = deadline - time.monotonic()
            a_opt = lp_alternating_improved(a_r, time_budget=min(remaining * 0.1, 60.0))
            obj_r = compute_obj(a_opt)
        else:
            a_opt = a_r
            obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        md_alloc = min(remaining * 0.2, 60.0)
        a_md = mirror_descent_v2(best_seq, time_budget=md_alloc, topK=7)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        lp_alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_alternating_improved(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        ap_alloc = min(remaining * 0.3, 50.0)
        a_ap = active_set_polish(best_seq, time_budget=ap_alloc)
        obj_ap = compute_obj(a_ap)
        if obj_ap < best_score:
            best_score = obj_ap
            best_seq = a_ap.copy()
            print(f"  Final Active Set: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more LP with longer time
        a_last = lp_alternating_improved(best_seq, time_budget=min(remaining * 0.5, 40.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
