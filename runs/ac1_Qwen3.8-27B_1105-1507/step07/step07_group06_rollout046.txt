    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)
    a = a / np.sum(a)

    M_bound = np.max(A @ current_g) * 1.5  # Initial bound

    for cp_iter in range(max_iter):

        conv_vals = A @ current_g

        active_set.update(top_k.tolist())

        active_idx = sorted(active_set)

        A_ub[:, n] = -1.0
        b_ub = np.full(len(active_idx), M_bound)
        
        A_eq = np.zeros((1, n + 1))

                      bounds=bounds, method='highs')

            break

        s = np.sum(g_new)
        if s < 1e-10:
            break

        M_new = float(res.x[n])
        
        # Check if solution violates inactive constraints
        conv_new = A @ g_new
        max_inactive = 0
        worst_inactive = -1
        for k in range(2*n-1):

            return g_new, M_new

    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def lp_alternating(a, time_budget, max_iters=5000, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stuck_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g, M = solve_lp(a)
        if g is None:
            break
        
        # Line search
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        
        if improved:
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 30:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000000) + it)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                stuck_count = 0
    
    return best_a

def smooth_obj_and_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    shifted = (conv - conv_max) * tau
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    obj = 2.0 * n * (conv_max + np.log(np.sum(exp_vals)) / tau)
    
    # Significant indices
    threshold = 1.0 / (50.0 * len(w))
    sig = np.where(w > threshold)[0]
    
    grad = np.zeros(n)
    for k in sig:
        wk = w[k]
        # Term 1: for j = 0 to n-1-k, grad[j] += wk * a[j+k]
        if n - k > 0:
            grad[:n-k] += wk * a[k:]
        # Term 2: for j = k to n-1, grad[j] += wk * a[j-k]
        if k < n:
            grad[k:n] += wk * a[:n-k]
    
    grad *= 2.0 * n
    return obj, grad

def smooth_optimize(a_init, time_budget, tau_start=20.0, tau_end=500.0):
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    taus = np.geomspace(tau_start, tau_end, 6)
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        stage_budget = min(remaining * 0.7, 60.0)
        
        def obj_func(x):
            val, _ = smooth_obj_and_grad(x, tau)
            return val
        
        def grad_func(x):
            _, g = smooth_obj_and_grad(x, tau)
            return g
        
        try:
            result = minimize(obj_func, a, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': 200, 'gtol': 1e-7, 'maxfun': 300})
            if result.x is not None:
                a_new = np.maximum(result.x, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
        except Exception:
            pass
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
