        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def cross_corr_wn_a(w, a):
    """Compute x[i] = sum_k w[k]*a[k-i] for i=0..n-1 using FFT."""
    n = len(a)
    size = next_pow2(len(w) + n - 1)
    fw = rfft(w, size)
    fa_flip = rfft(a[::-1], size)
    conv_result = irfft(fw * fa_flip, size)
    return conv_result[n - 1:2 * n - 1]


def smoothed_obj_and_grad(x, tau):
    """Smoothed logsumexp objective and gradient for a=softmax(x)."""
    n = len(x)
    x_shift = x - np.max(x)
    a = np.exp(x_shift)
    s = float(np.sum(a))
    a = a / s

    conv = compute_conv(a)
    c = conv / tau
    c_max = float(np.max(c))
    exp_c = np.exp(c - c_max)
    sum_exp = float(np.sum(exp_c))
    w = exp_c / sum_exp

    logmax_approx = c_max + np.log(sum_exp)
    F = 2.0 * n * logmax_approx

    xcor = cross_corr_wn_a(w, a)
    grad_a = (2.0 * n / tau) * xcor
    mean_grad = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - mean_grad)

    return F, grad_x


def make_fun(tau):
    def fun(xv):
        xv2 = xv - np.max(xv)
        val, _ = smoothed_obj_and_grad(xv2, tau)
        return val
    return fun


def make_fgrad(tau):
    def fgrad(xv):
        xv2 = xv - np.max(xv)
        _, g = smoothed_obj_and_grad(xv2, tau)
        return g
    return fgrad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    taus = [15.0 / n, 8.0 / n, 4.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]

    x = np.log(a)
    x = x - x.max()

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break

        res = minimize(make_fun(tau), x, method='L-BFGS-B', jac=make_fgrad(tau),
                       options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 60})

        if res.fun < make_fun(tau)(x):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={best_obj:.10f}")
            else:
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={obj_new:.10f} (no improve)")

    return best_a


def fw_subgradient(a):
    """Compute subgradient of 2n*max(conv(a,a)) at a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * n * a[j]
    return grad, k_star, conv


def fw_optimize(a_init, time_budget=20.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        grad, k_star, conv = fw_subgradient(a)

        dot_val = float(np.dot(a - best_a, grad))
        if dot_val < 1e-11:
            break

        j = int(np.argmin(grad))
        e_j = np.zeros(n)
        e_j[j] = 1.0
        t = 2.0 / (it + 2)
        a_new = (1.0 - t) * a + t * e_j
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 10000 == 0:
                print(f"    FW it={it}: obj={best_obj:.10f}")

    return best_a


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, use_cvxpy=True):
    n = len(a)
    a_norm = a / np.sum(a)

    if use_cvxpy and n <= 2000:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    try:
        A = build_conv_matrix(a_norm)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternate_one_step(a):
    n = len(a)
    a_norm = a / np.sum(a)
    g_lp, M_lp = solve_lp_minpeak(a_norm)
    if g_lp is None:
        return a, compute_obj(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8, 0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85, 0.95]:
        a_try = (1.0 - t) * a_norm + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def lp_alternate_optimize(a, time_budget=60.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        a_new, obj_new = lp_alternate_one_step(a)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.10f}")
        else:
            stall += 1
            if stall > 8:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 8, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def spectral_shaping(a, time_budget=30.0, verbose=False):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = next_pow2(2 * n)
    alphas = [0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1]

    for alpha in alphas:
        if time.monotonic() - start > time_budget:
            break

        for _ in range(200):
            if time.monotonic() - start > time_budget:
                break

            fa = rfft(a, size)
            power = np.abs(fa) ** 2
            power_smooth = np.maximum(power, 1e-30)
            power_flattened = power_smooth ** alpha
            power_flattened = power_flattened / np.mean(power_flattened) * np.mean(power_smooth)

            mag_new = np.sqrt(power_flattened)
            phase_old = np.angle(fa)
            fa_new = mag_new * np.exp(1j * phase_old)
            a_new = irfft(fa_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s = float(np.sum(a_new))
            if s < 1e-10:
                break
            a_new = a_new / s

            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

    return best_a


def multi_peak_gradient(a, time_budget=30.0, verbose=False):
    """Push down multiple near-peak positions simultaneously."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 2.0 / n

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        # Use all lags within 20% of max
        threshold = c_max * 0.8
        active = conv >= threshold

        if not np.any(active):
            k_star = int(np.argmax(conv))
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = 2.0 * a[j]
        else:
            w = np.zeros_like(conv)
            w[active] = conv[active]
            w = w / np.sum(w)
            grad = cross_corr_wn_a(w, a)
            grad = 2.0 * n * grad

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = max(eta * 0.9995, 0.05 / n)
        else:
            eta *= 0.9
            if eta < 0.01 / n:
                rng = np.random.default_rng(it % 100000)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                eta = 2.0 / n

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    mode = int(rng.integers(0, 8))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)
    elif mode == 3:
        a = 1.0 - np.abs(np.arange(n) - n / 2) / (n / 2)
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 4:
        c1, c2 = n // 3, 2 * n // 3
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n * n / 16)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n * n / 16))
        a = a / np.sum(a)
    elif mode == 5:
        a = np.ones(n) + 0.3 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 6:
        a = np.cos(np.pi * np.arange(n) / n) ** 2 + 0.1
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    else:
        a = 1.0 / (1.0 + 0.5 * (np.arange(n) / n - 0.5) ** 2)
        a = a + 0.05 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining smoothed gradient, Frank-Wolfe, LP alternating,
    spectral shaping, multi-peak gradient, and n-exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)

    # === Phase 1: Smoothed gradient (primary) ===
    print(f"\n=== Phase 1: Gradient Optimization (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_grad = gradient_optimize(best_seq, time_budget=min(remaining * 0.2, 120.0), verbose=True)
        obj = compute_obj(a_grad)
        if obj < best_score:
            best_score = obj
            best_seq = a_grad.copy()
            print(f"  After gradient: obj={best_score:.10f}")

    # === Phase 2: Multi-peak gradient ===
    print(f"\n=== Phase 2: Multi-Peak Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_mp = multi_peak_gradient(best_seq, time_budget=min(remaining * 0.08, 40.0), verbose=True)
        obj = compute_obj(a_mp)
        if obj < best_score:
            best_score = obj
            best_seq = a_mp.copy()
            print(f"  After multi-peak: obj={best_score:.10f}")
        else:
            print(f"  Multi-peak: obj={obj:.10f}")

    # === Phase 3: Frank-Wolfe ===
    print(f"\n=== Phase 3: Frank-Wolfe ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fw = fw_optimize(best_seq, time_budget=min(remaining * 0.05, 25.0), verbose=True)
        obj = compute_obj(a_fw)
        if obj < best_score:
            best_score = obj
            best_seq = a_fw.copy()
            print(f"  After FW: obj={best_score:.10f}")
        else:
            print(f"  FW: obj={obj:.10f}")

    # === Phase 4: LP Alternating ===
    print(f"\n=== Phase 4: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_lp = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.12, 90.0), verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  After LP: obj={best_score:.10f}")
        else:
            print(f"  LP: obj={obj:.10f}")

    # === Phase 5: Spectral shaping ===
    print(f"\n=== Phase 5: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_spec = spectral_shaping(best_seq, time_budget=min(remaining * 0.05, 30.0), verbose=True)
        obj = compute_obj(a_spec)
        if obj < best_score:
            best_score = obj
            best_seq = a_spec.copy()
            print(f"  After spectral: obj={best_score:.10f}")
        else:
            print(f"  Spectral: obj={obj:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_candidates = sorted(set([
        max(200, n_parent - 200), max(200, n_parent - 100), n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200, n_parent + 400,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 2000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.1, 30.0)
        a_opt = gradient_optimize(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 7: Intensify on best ===
    print(f"\n=== Phase 7: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_g = gradient_optimize(best_seq, time_budget=min(remaining * 0.15, 80.0), verbose=True)
        obj = compute_obj(a_g)
        if obj < best_score:
            best_score = obj
            best_seq = a_g.copy()
            print(f"  Gradient2: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_mp2 = multi_peak_gradient(best_seq, time_budget=min(remaining * 0.1, 40.0))
            obj = compute_obj(a_mp2)
            if obj < best_score:
                best_score = obj
                best_seq = a_mp2.copy()
                print(f"  Multi-peak2: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_lp2 = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.15, 60.0), verbose=True)
            obj = compute_obj(a_lp2)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp2.copy()
                print(f"  LP2: obj={best_score:.10f}")

    # === Phase 8: Random restarts ===
    print(f"\n=== Phase 8: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    restart_budget = min(25.0, remaining * 0.06)
    n_restarts = max(3, int(remaining / (restart_budget + 5)))
    n_restarts = min(n_restarts, 8)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < restart_budget + 5:
            break

        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.04, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            n_r = n_best + int(rng.integers(-150, 150))
            n_r = max(400, min(1800, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.03, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 2:
            a_r = best_seq + 0.03 * rng.normal(0, 1, n_best)
            a_r = (a_r + a_r[::-1]) / 2.0
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 3:
            x = np.linspace(0, np.pi, n_best)
            a_r = np.cos(x) ** 2 + 0.2 + 0.05 * rng.normal(0, 1, n_best)
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)
        elif r == 4:
            # Bimodal
            c1 = n_best // 4
            c2 = 3 * n_best // 4
            a_r = np.exp(-((np.arange(n_best) - c1) ** 2) / (2 * (n_best / 8) ** 2)) + \
                  np.exp(-((np.arange(n_best) - c2) ** 2) / (2 * (n_best / 8) ** 2))
            a_r = a_r + 0.05 * rng.normal(0, 1, n_best)
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)
        else:
            a_r = random_init(n_best, rng)

        a_opt = gradient_optimize(a_r, time_budget=restart_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_f = gradient_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final grad: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_mp3 = multi_peak_gradient(best_seq, time_budget=min(remaining * 0.2, 30.0))
            obj = compute_obj(a_mp3)
            if obj < best_score:
                best_score = obj
                best_seq = a_mp3.copy()
                print(f"  Final multi-peak: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_lp3 = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.3, 30.0))
            obj = compute_obj(a_lp3)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp3.copy()
                print(f"  Final LP: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_fw2 = fw_optimize(best_seq, time_budget=min(remaining * 0.3, 20.0))
            obj = compute_obj(a_fw2)
            if obj < best_score:
                best_score = obj
                best_seq = a_fw2.copy()
                print(f"  Final FW: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
