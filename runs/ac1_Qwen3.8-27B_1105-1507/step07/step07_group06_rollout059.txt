```python
    conv = compute_conv(a)
    conv_max = np.max(conv)

    w = w / np.sum(w)

    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    theta = cssv[rho-1] / rho
    return np.maximum(v - theta, 0)
```

where Z is the normalizing constant. This is equivalent to gradient descent in the Bregman divergence with the negative entropy.

The step size eta can be chosen via backtracking.

Let me try a different angle entirely. What if I use **Nesterov's accelerated gradient** but with the **exact (non-smooth) subgradient** and the **proximal operator** being the projection onto the simplex?

For the non-smooth function `f(a) = 2n * max_k conv(a,a)[k]`:
- Subgradient at a is: `2n * 2 * a[k* - j]` for the argmax bin k* (or a convex combination of argmax bins)
- Proximal operator: projection onto simplex

This gives us the **accelerated proximal gradient (FISTA)** for the non-smooth function. The issue is that the subgradient might not be the best direction to search (it only considers one bin at a time).

To handle this better, I can use a **randomized subgradient**: at each iteration, pick the argmax bin (or a weighted random bin among the top-k peaks) and compute the subgradient.


This is a non-convex problem (quadratic constraints). But the LP alternating gives us a sequence of feasible solutions, and the LP dual gives us a lower bound on what the next iterate can achieve.

Let me also add a new technique: **ADMM-style splitting.**

The problem: min f(a) = 2n * max_k (a^T M_k a) s.t. a >= 0, sum(a) = 1

ADMM splitting: Introduce auxiliary variable b = a, and solve:

```python
"""
Optimized step function search using:
1. Fast LP alternating (HiGHS solver) as main optimization
2. FISTA with adaptive log-sum-exp smoothing for initialization
3. Peak-targeted coordinate descent for final polishing
4. Multi-n exploration and random restarts
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

# --- Top-level helper functions ---

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 1.0/n)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(ind[cond]))
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    
    # Use HiGHS via scipy linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g, float(res.x[n])
    except Exception as e:
        pass
    
    # Fallback: try cvxpy
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        return g_val / s_g, float(M.value)
            except:
                continue
    except:
        pass
    
    return None, None

def smooth_max_and_weights(conv, tau):
    """Compute smooth max and softmax weights."""
    m = np.max(conv)
    shifted = (conv - m) * tau
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    w = exp_shifted / np.sum(exp_shifted)
    smooth_val = m + max_shifted / tau + np.log(np.sum(exp_shifted)) / tau
    return smooth_val, w

def smooth_obj_grad(a, tau):
    """Compute smooth objective and gradient."""
    n = len(a)
    conv = compute_conv(a)
    smooth_val, w = smooth_max_and_weights(conv, tau)
    obj = 2.0 * n * smooth_val
    
    # Gradient: 4n * c where c[j] = sum_k w[k] * a[k-j]
    c = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        if k_end > k_start:
            c[j] = np.dot(w[k_start:k_end], a[:k_end - k_start])
    
    grad = 4.0 * n * c
    return obj, grad

def smooth_obj_grad_fast(a, tau):
    """Faster version using FFT for the cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    smooth_val, w = smooth_max_and_weights(conv, tau)
    obj = 2.0 * n * smooth_val
    
    # c[j] = dot(w[j:j+n], a) = cross-correlation
    # This is conv(w, flip(a)) at specific indices
    # Or: c = ifft(conj(ifft(w)) * ifft(a)) ... let me just use direct for safety
    # Actually, c[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    # This is the cross-correlation of w and a, which can be computed via FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a, size)
    # cross-correlation: ifft(conj(fa) * fw) gives sum a[m]*w[m+j]
    # Which is c[j] = sum_m a[m] * w[j+m]
    c = irfft(np.conj(fa) * fw, size)[:n]
    
    grad = 4.0 * n * c
    return obj, grad

def fista_smooth(a_init, taus, time_budget, max_iter_per_tau=500):
    """FISTA with tau annealing."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        a_f = a.copy()
        y = a_f.copy()
        t = 1.0
        # Estimate L: use a rough bound
        # L ~ 4n * (2*max(a) * n + tau * max(a) * n^2) ... very loose
        # Use backtracking instead
        L = 4.0 * n * (2.0 + tau) * 2.0  # rough estimate
        step = 1.0 / L
        
        iter_budget = int(remaining * 100)  # rough
        iter_budget = max(iter_budget, 50)
        iter_budget = min(iter_budget, max_iter_per_tau)
        
        for it in range(iter_budget):
            if time.monotonic() - start > time_budget:
                break
            
            _, grad_y = smooth_obj_grad_fast(y, tau)
            
            # Gradient step
            z = y - step * grad_y
            x = project_simplex(z)
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y_new = x + ((t - 1.0) / t_new) * (x - a_f)
            a_f = x
            y = y_new
            t = t_new
            
            # Adaptive step size (very rough)
            if it % 100 == 50:
                obj_check = compute_obj(a_f)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a_f.copy()
        
        # After each tau, check
        obj_check = compute_obj(a_f)
        if obj_check < best_obj:
            best_obj = obj_check
            best_a = a_f.copy()
            a = a_f.copy()
    
    # Final check
    obj_final = compute_obj(a_f)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a_f.copy()
    
    return best_a, best_obj

def lp_alternating(a_init, time_budget, verbose=False):
    """LP alternating optimization with better line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve_count = 0
    
    for it in range(10000):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        # Estimate time per LP solve
        if it > 0:
            avg_time = elapsed / it
            if avg_time > (time_budget - elapsed) / 5:
                break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search: try multiple t values
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        best_t = 0
        
        # Golden section or simple grid
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
                best_t = t
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            no_improve_count = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.10f}, t={best_t:.2f}, M_lp={M_lp:.6f}")
        else:
            no_improve_count += 1
            if no_improve_count > 20:
                # Perturb to escape local min
                rng = np.random.default_rng(it * 7919 + n + int(time.monotonic() * 1000))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                no_improve_count = 0
            elif it % 50 == 0:
                break
    
    return best_a

def peak_targeted_descent(a_init, time_budget):
    """Coordinate descent targeting the peak bin."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # The peak bin k_star: conv[k_star] = sum_j a[j]*a[k_star-j]
        # To reduce this, decrease a[j] where a[k_star-j] is large
        # Move mass to positions that don't contribute to the peak
        
        improved = False
        # Sort indices by contribution to peak
        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]
            else:
                contributions[j] = 0
        
        # Move mass from high-contribution to low-contribution positions
        sorted_idx = np.argsort(contributions)[::-1]
        
        for move_frac in [0.005, 0.01, 0.02]:
            for i_pos in range(min(20, n)):
                src = sorted_idx[i_pos]
                # Find a destination that reduces the peak
                # Try positions far from k_star
                dst_candidates = [j for j in range(n) if abs(j - k_star) > n // 3 and j != src]
                if not dst_candidates:
                    continue
                dst = dst_candidates[np.argmax(a[dst_candidates])]  # Take from largest
                
                delta = move_frac * a[src]
                if delta > 0.5 * a[src]:
                    continue
                
                a[src] -= delta
                a[dst] += delta
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[src] += delta
                    a[dst] -= delta
                    break
            
            if improved:
                break
        
        if not improved:
            # Try random pairwise swaps
            rng = np.random.default_rng(it * 31 + n)
            for _ in range(100):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.01 * a[i] * rng.uniform(0.1, 1.0)
                if delta > a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
            
            if not improved and it > 10:
                break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
"""
Optimized step function search using:
1. Quick subgradient initialization with top-peak weighting
2. Extended LP alternating (HiGHS) as main optimizer with adaptive line search
3. Peak-targeted coordinate descent for final polishing
4. Multi-n sweep and exploration
5. Random restarts with diverse asymmetric initializations
Key insight: exploit asymmetry (symmetric sequences bounded below by 2).
Focus time on LP alternating which gives steady linear convergence to local optima.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(ind[cond]))
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g, float(res.x[n])
    except Exception:
        pass

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        return g_val / s_g, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    return None, None


def subgradient_topk(a, k_peaks=5, temperature=30.0):
    """Compute weighted subgradient using top-k peak bins."""
    n = len(a)
    conv = compute_conv(a)
    if len(conv) <= k_peaks:
        top_idx = np.argsort(conv)[::-1]
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        top_idx = np.argsort(conv)[::-1][:k_peaks]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        weights = np.exp((peak_vals - max_val) * temperature)
        weights = weights / np.sum(weights)

    grad = np.zeros(n)
    for w_k, k in zip(weights, top_idx):
        k = int(k)
        for j in range(n):
            k_j = k - j
            if 0 <= k_j < n:
                grad[j] += w_k * 2.0 * a[k_j]

    grad *= 2.0 * n
    return grad

        grad = subgradient_topk(a, k_peaks=k_peaks)

        t = step
        improved = False
        for _ in range(12):
            z = a - t * grad
            x = project_simplex(z)
            obj_x = compute_obj(x)
            if obj_x < best_obj - 1e-15:
                best_obj = obj_x
                best_a = x.copy()
                a = x.copy()
                improved = True
                break
            t *= 0.5

        if not improved:

            if step < 1e-10:

        else:

def smooth_lbfgs(a_init, taus, time_budget, max_iter=150):
    """Smoothed L-BFGS-B with tau annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    def softmax_from_theta(theta):
        theta_shift = theta - np.max(theta)
        exp_t = np.exp(theta_shift)
        return exp_t / np.sum(exp_t)

    def smooth_obj_grad(theta, tau):
        a_local = softmax_from_theta(theta)
        conv = compute_conv(a_local)
        conv_max = np.max(conv)
        obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau)))) / tau

        w = tau * np.exp((conv - conv_max) * tau)
        w = w / np.sum(w)

        c = np.zeros(n)
        for j in range(n):
            k_end = min(2 * n - 1, j + n)
            if k_end > j:
                c[j] = np.dot(w[j:k_end], a_local[:k_end - j])

        mean_c = np.dot(c, a_local)
        grad_theta = 4.0 * n * a_local * (c - mean_c)
        return obj_val, grad_theta

    from scipy.optimize import minimize

    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        theta = np.log(np.maximum(a_local, 1e-15))

        def obj_func(th):
            val, _ = smooth_obj_grad(th, tau)
            return val

        def grad_func(th):
            _, g = smooth_obj_grad(th, tau)
            return g

        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'gtol': 1e-6, 'maxfun': 300})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a_local = a_new.copy()
        except Exception:
            pass

    return best_a, best_obj

def peak_targeted_descent(a_init, time_budget):
    """Coordinate descent targeting the peak bin with smart moves."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(400):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Compute contributions to peak
        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]
            else:
                contributions[j] = 0.0

        sorted_idx = np.argsort(contributions)[::-1]
        improved = False

        for move_frac in [0.003, 0.007, 0.015]:
            for i_pos in range(min(15, n)):
                src = int(sorted_idx[i_pos])
                if a[src] < 1e-10:
                    continue

                # Find good destination: low contribution to peak, has some mass
                src_contrib = contributions[src]
                if src_contrib < 1e-12:
                    break

                # Try several destinations
                dst_candidates = []
                for j in range(n):
                    if j == src:
                        continue
                    k_j = k_star - j
                    if 0 <= k_j < n:
                        new_contrib_delta = a[j] * move_frac * a[src]
                    else:
                        new_contrib_delta = 0.0
                    dst_candidates.append((j, new_contrib_delta))

                dst_candidates.sort(key=lambda x: x[1])
                for dst, _ in dst_candidates[:5]:
                    delta = move_frac * a[src]
                    if delta > 0.8 * a[src]:
                        continue
                    a[src] -= delta
                    a[dst] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[dst] -= delta

            if improved:
                break

        if not improved:
            rng = np.random.default_rng(it * 31 + n)
            for _ in range(80):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.005 * a[i] * rng.uniform(0.5, 2.0)
                if delta > 0.5 * a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta

            if not improved and it > 20:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    Main search: combines subgradient init, LP alternating, smoothed L-BFGS-B,
    peak-targeted descent, multi-n exploration, and random restarts.
    Time allocation biased heavily toward LP alternating (the most effective local optimizer).
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, obj={best_score:.8f}, budget={budget_s}s")

    # === Phase 0: Quick n-sweep to identify best n range ===
    print(f"\n=== Phase 0: n-sweep ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_sweep = sorted(set([
            max(400, n_parent - 300), max(400, n_parent - 150),
            n_parent, n_parent + 100, n_parent + 200,
            max(400, n_parent + 400),
        ]))
        n_sweep = [n for n in n_sweep if 400 <= n <= 2000]

        for n_try in n_sweep:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            if n_try == n_parent:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)

            # Very quick LP (3s) to get a ballpark
            alloc = min(3.0, remaining * 0.15)
            a_quick = lp_alternating(a_try, time_budget=alloc)
            obj_quick = compute_obj(a_quick)
            print(f"  n={n_try}: obj={obj_quick:.6f}")

            if obj_quick < best_score:
                best_score = obj_quick
                best_seq = a_quick.copy()
                n_parent = n_try
                print(f"  -> New best: n={n_try}, obj={best_score:.8f} **")

    # === Phase 1: Subgradient + L-BFGS-B initialization (fast) ===
    print(f"\n=== Phase 1: Initialization ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Subgradient
        alloc_sg = min(remaining * 0.02, 20.0)
        a_sg, obj_sg = subgradient_init(best_seq, time_budget=alloc_sg, k_peaks=5)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

        # L-BFGS-B
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc_lb = min(remaining * 0.03, 25.0)
            a_lb, obj_lb = smooth_lbfgs(best_seq, [10.0, 50.0, 200.0],
                                         time_budget=alloc_lb, max_iter=100)
            if obj_lb < best_score:
                best_score = obj_lb
                best_seq = a_lb.copy()
                print(f"  L-BFGS-B: obj={best_score:.8f} **")
            else:
                print(f"  L-BFGS-B: obj={obj_lb:.8f}")

    # === Phase 2: Main LP alternating (45% of time) ===
    print(f"\n=== Phase 2: Main LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = remaining * 0.40
        print(f"  Allocated: {alloc:.0f}s")
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Main LP done: obj={best_score:.8f} **")
        else:
            print(f"  Main LP done: obj={obj_lp:.8f}")

    # === Phase 3: Peak-targeted refinement ===
    print(f"\n=== Phase 3: Peak-targeted descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.05, 40.0)
        a_peak = peak_targeted_descent(best_seq, time_budget=alloc)
        obj_peak = compute_obj(a_peak)
        if obj_peak < best_score:
            best_score = obj_peak
            best_seq = a_peak.copy()
            print(f"  Peak: obj={best_score:.8f} **")
        else:
            print(f"  Peak: obj={obj_peak:.8f}")

    # === Phase 4: Multi-n fine-tuning ===
    print(f"\n=== Phase 4: Multi-n exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 100), max(400, n_cur - 50),
        n_cur + 50, n_cur + 100, max(400, n_cur + 200),
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500 and n != n_cur]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.08, 50.0)
        a_opt = lp_alternating(a_try, time_budget=alloc)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = n_try
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Second intensive LP pass ===
    print(f"\n=== Phase 5: Second LP pass ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 200.0)
        print(f"  Allocated: {alloc:.0f}s")
        a_lp2 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  2nd LP: obj={best_score:.8f} **")
        else:
            print(f"  2nd LP: obj={obj_lp2:.8f}")

    # === Phase 6: Random restarts ===
    print(f"\n=== Phase 6: Random restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        init_type = r % 4
        if init_type == 0:
            slope = rng.uniform(1.5, 4.0)
            a_r = np.linspace(0.05, slope, n_best)
        elif init_type == 1:
            rate = rng.uniform(1.5, 5.0)
            a_r = np.exp(np.linspace(0, rate, n_best))
        elif init_type == 2:
            a_r = best_seq * np.exp(rng.normal(0, 0.15, n_best))
        else:
            # Cosine ramp
            phase = rng.uniform(0, np.pi)
            a_r = 1.0 + rng.uniform(0.5, 1.5) * np.cos(np.linspace(phase, phase + rng.uniform(1, 3), n_best))
            a_r = np.maximum(a_r, 0.01)

        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        alloc = min(remaining * 0.12, 35.0)
        a_opt = lp_alternating(a_r, time_budget=alloc)
        obj_final = compute_obj(a_opt)

        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (type={init_type}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_final:.8f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.4, 80.0)
        a_final = lp_alternating(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 30.0)
        a_pk = peak_targeted_descent(best_seq, time_budget=alloc)
        obj_pk = compute_obj(a_pk)
        if obj_pk < best_score:
            best_score = obj_pk
            best_seq = a_pk.copy()
            print(f"  Final peak: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
