def compute_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_optimize(a, time_budget=200.0, top_k=10, verbose=False):
    """Additive subgradient with momentum, targeting top-K convolution peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.7
    lr = 0.1
    no_improve = 0

        max_b = float(np.max(conv))
        obj = 2.0 * n * max_b
        
        k_indices = np.argsort(conv)[::-1][:top_k]

        grad = np.zeros(n)

            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    grad[j] += a[kj]

        grad -= np.mean(grad)

        a_new = a + momentum
        a_new = np.maximum(a_new, 0)

            a_new = a_new / s_new

def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M_init=None):
    """Solve: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.PDLP, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except:
            continue
    return None, None

def lp_alternate(a, time_budget=500.0, verbose=False):

Instead of linear convolution, consider circular convolution (periodic extension). The circular auto-convolution is:
c[k] = sum_j a[j] * a[(j+k) mod n]

The max of c is what we want to minimize (approximately, since for a "smooth" sequence, the linear and circular convolutions are similar away from the boundaries).

The advantage of circular convolution is that it's diagonalized by the DFT:

This makes the optimization much cleaner. We can work directly in the frequency domain.

**Spectral optimization for circular convolution:**

min max_k IDFT(|F|^2)[k]
s.t. a = IDFT(F) >= 0, sum(a) = 1

4. a_new = real(IDFT(F_new))
5. Project: a_new = max(0, a_new), renormalize
6. Repeat

This is similar to the "Gerchberg-Saxton" algorithm used in phase retrieval!

6. a_new = real(IDFT(F_new)), project to non-negative, renormalize
