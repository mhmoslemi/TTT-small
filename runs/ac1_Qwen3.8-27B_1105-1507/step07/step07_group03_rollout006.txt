import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv_fft(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, warm_start=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

def spectral_projection(a, n_iters=500, time_budget=60.0, relaxation=0.5, target_flat=True):
    """FFT-based flat spectrum iteration with projection onto non-negative simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    relax = relaxation

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-12)
        phase = A / mag_safe

        if target_flat:
            # Target: flat magnitude (all equal)
            target_mag = np.ones(L) * np.mean(mag)
        else:
            # Target: minimize peak by reducing large magnitudes
            target_mag = mag
            thresh = np.percentile(mag, 75)
            target_mag[mag > thresh] = thresh

        A_new = phase * target_mag
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            stall += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if stall < 50 and it % 100 == 0:
                relax = max(0.1, relaxation * 0.9)
        else:
            stall += 1
            if stall > 40:
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.05 + 0.01 * min(stall // 40, 5), n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                relax = relaxation

    return best_a


def multi_peak_subgradient(a, n_iters=3000, time_budget=60.0):
    """Subgradient method targeting top peaks of convolution."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    m_peaks = 8

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fft(a)
        top_idx = np.argsort(conv)[-m_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        # Exponential weighting for peaks
        temp = max(0.0001, (max_val - top_val[0]) * 0.3)
        if max_val - top_val[0] < 1e-10:
            weights = np.ones(m_peaks) / m_peaks
        else:
            weights = np.exp((top_val - max_val) / temp)
            weights /= weights.sum()

        # Compute subgradient (weighted sum of convolution contributions)
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            for i in range(lo, hi + 1):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * a[j]

        # Normalize gradient
        grad_mean = np.mean(grad)
        grad = grad - grad_mean
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        grad = grad / grad_norm

        # Adaptive step size
        alpha = 2.0 / (it + 10)
        alpha = min(alpha, 0.3)

        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new

    return best_a


def mirror_descent_exp(a, n_iters=3000, time_budget=60.0):

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def golay_like_init(n, rng):
    """Generate a Golay/Barker-like structured initialization."""
    # Start with a pseudo-random ±1 sequence, then take absolute value
    # and smooth it
    signs = rng.choice([-1, 1], n)
    a = np.cumsum(signs).astype(float)
    a = np.abs(a) + 0.1
    # Smooth
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def chirp_init(n):

        obj = compute_obj(a_md)
        print(f"  Mirror descent: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # Phase 5: LP again (refined)
    print(f"\n=== Phase 5: LP Refined (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 200.0)
        a_lp2 = lp_alternating(best_seq, time_budget=alloc, max_iters=100, verbose=True)
        obj = compute_obj(a_lp2)
        print(f"  LP refined: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()

    # Phase 6: Coordinate descent polish
    print(f"\n=== Phase 6: Coordinate Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 40.0)
        a_cd = coordinate_descent_2swap(best_seq, time_budget=alloc)
        obj = compute_obj(a_cd)
        print(f"  Coord descent: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()

    # Phase 7: Multi-n exploration
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100, n_best + 200,
        400, 500, 600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if abs(n_try - n_best) < 30:
            continue

        a_try = resample_sequence(best_seq, n_try)
        # Quick optimization
        alloc1 = min(remaining * 0.05, 15.0)
        a_opt = spectral_projection(a_try, n_iters=200, time_budget=alloc1, relaxation=0.5)

        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc2 = min(remaining * 0.08, 30.0)
        a_opt = lp_alternating(a_opt, time_budget=alloc2, max_iters=30)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # Phase 8: Random restarts with structured inits
    print(f"\n=== Phase 8: Structured Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if r == 0:
            a_r = golay_like_init(n_best, rng)
        elif r == 1:
            a_r = chirp_init(n_best)
        elif r == 2:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.1, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 3:
            # Different n with chirp
            n_r = max(300, min(1200, n_best + int(rng.integers(-200, 200))))
            a_r = chirp_init(n_r)
        elif r == 4:
            # Haar-like structure
            n_r = n_best
            a_r = np.ones(n_r)
            for level, width in enumerate([n_r // 4, n_r // 8, n_r // 16]):
                if width < 2:
                    break
                for start in range(0, n_r, width * 2):
                    if rng.random() > 0.5:
                        a_r[start:start + width] *= 0.5
                    else:
                        a_r[start:start + width] *= 1.5
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)
        elif r == 5:
            # Exponential decay + noise
            x = np.arange(n_best)
            a_r = np.exp(-x / (n_best / 3)) + rng.uniform(0, 0.1, n_best)
            a_r = a_r / np.sum(a_r)
        elif r == 6:
            # Sine wave
            x = np.linspace(0, 4 * np.pi, n_best)
            a_r = (1 + np.sin(x)) / 2 + 0.1
            a_r = a_r / np.sum(a_r)
        else:
            # Antisymmetric perturbation
            a_r = best_seq.copy()
            mid = n_best // 2
            a_r[:mid] *= 1.2
            a_r[mid:] *= 0.85
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        # Optimize
        alloc_fft = min(15.0, remaining * 0.1)
        a_opt = spectral_projection(a_r, n_iters=300, time_budget=alloc_fft, relaxation=0.5)

        remaining = deadline - time.monotonic()
        if remaining < 20:
            alloc_lp = 10.0
        else:
            alloc_lp = min(30.0, remaining * 0.2)
        a_opt = lp_alternating(a_opt, time_budget=alloc_lp, max_iters=20)

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_opt = multi_peak_subgradient(a_opt, n_iters=2000, time_budget=min(20.0, remaining * 0.3))

        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # Phase 9: Final intensive LP polish
    print(f"\n=== Phase 9: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 120.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, max_iters=50, verbose=True)
        obj = compute_obj(a_final)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()

    # Phase 10: Final subgradient + CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 40.0)
        a_sg = multi_peak_subgradient(best_seq, n_iters=3000, time_budget=alloc)
        obj = compute_obj(a_sg)
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_descent_2swap(best_seq, time_budget=min(remaining * 0.3, 20.0))
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
