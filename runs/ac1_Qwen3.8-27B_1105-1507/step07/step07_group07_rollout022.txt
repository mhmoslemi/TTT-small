"""
Advanced low-autocorrelation non-negative sequence optimizer.
Combines adaptive LSE-smoothed L-BFGS-B, exact quadratic subgradient steps,
coordinate descent, multi-resolution search, and structured perturbations
to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def _npp2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _conv(a):
    n = len(a)
    size = _npp2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def _obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    an = a / s
    c = _conv(an)
    return 2.0 * n * float(np.max(c))


def _lse_fg(x, tau):
    n = len(x)
    xs = x - np.max(x)
    u = np.exp(xs)
    s = float(np.sum(u))
    u = u / s
    conv = _conv(u)
    cs = conv / tau
    cm = float(np.max(cs))
    ec = np.exp(cs - cm)
    se = float(np.sum(ec))
    w = ec / se
    F = cm + np.log(se)
    size = _npp2(2 * n - 1)
    ur = u[::-1]
    fur = rfft(ur, size)
    fw = rfft(w, size)
    cr = irfft(fw * fur, size)
    corr = cr[n - 1:n - 1 + n]
    dc = float(np.dot(corr, u))
    grad = u * 2.0 * (corr - dc)
    return F, grad


def _pow_fg(x, p):
    """Power-mean smoothed objective: (sum(conv^p))^(1/p) / (sum(u))^2 approx."""
    n = len(x)
    xs = x - np.max(x)
    u = np.exp(xs)
    s = float(np.sum(u))
    u = u / s
    conv = _conv(u)
    # Power mean: (sum conv^p)^(1/p), gradient via chain rule
    cp = conv ** p
    sp = float(np.sum(cp))
    F = sp ** (1.0 / p)
    # dF/dconv[k] = (1/p) * sp^(1/p - 1) * p * conv[k]^(p-1) = sp^(1/p-1) * conv[k]^(p-1)
    dw = sp ** (1.0 / p - 1.0) * conv ** (p - 1)
    # dF/du[m] = sum_k dw[k] * 2*u[k-m] (valid)
    size = _npp2(2 * n - 1)
    ur = u[::-1]
    fur = rfft(ur, size)
    fdw = rfft(dw, size)
    cr = irfft(fdw * fur, size)
    corr = cr[n - 1:n - 1 + n]
    grad_u = 2.0 * corr
    dc = float(np.dot(grad_u, u))
    grad = u * (grad_u - dc)
    return F, grad


def _subgrad_exact_step(a, k_star, max_alpha):
    """
    Take an exact quadratic step on the active constraint (a*a)[k_star].
    Returns new a (normalized, non-negative).
    """
    n = len(a)
    # Gradient of (a*a)[k_star] w.r.t. a[i] = 2*a[k_star - i] for valid k_star-i
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    for i in range(lo, hi + 1):
        g[i] = 2.0 * a[k_star - i]
    # Project onto sum(a)=1 (remove mean)
    g_proj = g - np.mean(g)
    if np.max(np.abs(g_proj)) < 1e-15:
        return a.copy()
    # Optimal alpha for quadratic: minimize (a - alpha*g_proj)*(a - alpha*g_proj) at k_star
    # = (a*a)[k] - 2*alpha*(a*gp)[k] + alpha^2*(gp*gp)[k]
    # (x*y)[k] = sum_i x[i]*y[k-i] valid
    def _xyk(x, y, k):
        n2 = len(x)
        lo2 = max(0, k - n2 + 1)
        hi2 = min(k, n2 - 1)
        if lo2 > hi2:
            return 0.0
        return float(np.dot(x[lo2:hi2 + 1], y[k - hi2:k - lo2 + 1][::-1]))

    num = _xyk(a, g_proj, k_star)
    den = _xyk(g_proj, g_proj, k_star)
    if den < 1e-18:
        alpha = max_alpha
    else:
        alpha = num / den
    alpha = max(0.0, min(alpha, max_alpha))
    a_new = a - alpha * g_proj
    a_new = np.maximum(a_new, 0.0)
    sn = float(np.sum(a_new))
    if sn < 1e-10:
        return a.copy()
    return a_new / sn


def _coord_descent(a, n_iter=50):
    """Coordinate descent: for each coord, optimize 1D slice."""
    n = len(a)
    a = a.copy()
    best_obj = _obj(a)
    for it in range(n_iter):
        improved = False
        for i in range(n):
            a0 = a[i]
            a[i] = 0.0
            base_conv = _conv(a)
            base_sum = float(np.sum(a))
            # Objective as function of a[i] = t:
            # sum = base_sum + t, conv changes: conv[k] += t*(a[k-i] + a[k-i]) for valid... 
            # Actually conv[k] = (a*a)[k]. Adding t to a[i] changes:
            # conv[k] changes by t*(a[k-i]) + t*(a[k-i]) = 2*t*a[k-i] (for valid k-i)
            # Wait: conv[k] = sum_j a[j]*a[k-j]. If a[i] increases by t:
            # The terms involving a[i] are: a[i]*a[k-i] (j=i) and a[k-i]*a[i] (j=k-i if k-i != i)
            # So delta conv[k] = t*a[k-i] + t*a[k-i] = 2*t*a[k-i] (if k-i valid and k-i != i)
            # Or t*a[k-i] + t*a[i] = 2*t*a[i] (if k-i = i, i.e., k=2i)
            # In all cases: delta conv[k] = 2*t*a[k-i] for 0<=k-i<n
            # New conv[k] = base_conv[k] + 2*t*a_original[k-i] (for valid k-i)
            # But a_original = a with a[i]=0, so a_original[k-i] is the old value.
            # Actually, let's just evaluate directly for a few t values.
            
            # Sample t values
            ts = [0.0, a0, 2*a0, 0.5*a0, 0.1*a0, 5*a0]
            ts = [t for t in ts if t >= 0]
            best_t = a0
            best_v = _obj(a)
            for t in ts:
                a[i] = t
                s_try = float(np.sum(a))
                if s_try < 1e-10:
                    a[i] = best_t
                    continue
                try_obj = 2.0 * n * float(np.max(_conv(a / s_try)))
                if try_obj < best_v:
                    best_v = try_obj
                    best_t = t
            a[i] = best_t
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
                if _obj(a) < best_obj - 1e-12:
                    best_obj = _obj(a)
                    improved = True
        if not improved:
            break
    return a


def _resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

    best_a = a.copy()
    best_obj = _obj(a)

    x = np.log(a) - np.log(a).max()
    
    # Adaptive tau schedule
    taus = [20.0/n, 10.0/n, 5.0/n, 3.0/n, 2.0/n, 1.0/n, 0.5/n, 0.3/n, 0.2/n, 0.1/n, 0.05/n]

def _lp_minpeak(a, time_budget=5.0):
    """Solve LP: min M s.t. conv(a,g)[k]<=M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    try:
        # Build convolution matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a_norm[:r]
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_budget})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            sv = float(np.sum(g_val))
            if sv > 1e-10:
                return g_val / sv
    except Exception:
        pass
    return None

def _structured_perturb(a, mode, rng):
    """Apply structured perturbation."""
    n = len(a)
    a = a.copy()
    if mode == 0:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        taper = 0.5 + 0.5 * np.cos(x)
        a = a * (1.0 + 0.3 * (taper - 0.5))
    elif mode == 1:
        # Anti-symmetric adjustment
        a = a + 0.1 * (a - a[::-1])
    elif mode == 2:
        # Center boost
        center = np.exp(-((np.arange(n) - n/2.0)**2) / (2 * (n/6.0)**2))
        a = a * (1.0 + 0.2 * center)
    elif mode == 3:
        # Edge boost
        edges = np.exp(-((np.arange(n) - n/2.0)**2) / (2 * (n/6.0)**2))
        a = a * (1.0 + 0.2 * (1.0 - edges))
    elif mode == 4:
        # Two-peak structure
        p1 = np.exp(-((np.arange(n) - n*0.25)**2) / (2*(n*0.1)**2))
        p2 = np.exp(-((np.arange(n) - n*0.75)**2) / (2*(n*0.1)**2))
        a = a * (0.8 + 0.4 * (p1 + p2))
    else:
        # Random smooth perturbation
        noise = rng.normal(0, 1, n)
        kernel = np.ones(5) / 5
        noise = np.convolve(noise, kernel, mode='same')
        a = a * np.exp(0.05 * noise)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

    # === Phase 2: Power-Mean Smoothing ===
    print(f"\n=== Phase 2: Power-Mean Smoothing ===")
    if _remaining() > 40:
        alloc = min(_remaining() * 0.1, 60.0)
        a_pm = _multi_peak_smooth_optimize(best_seq, time_budget=alloc)
        obj = _obj(a_pm)
        if obj < best_score:
            best_score = obj
            best_seq = a_pm.copy()
            print(f"  Power-mean: obj={best_score:.10f}")
        else:
            print(f"  Power-mean: obj={obj:.10f} (no improve)")
    
    # === Phase 3: Subgradient with exact steps ===
    print(f"\n=== Phase 3: Subgradient Exact Steps ===")
    if _remaining() > 30:
        alloc = min(_remaining() * 0.1, 60.0)
        a_sg = _subgrad_optimize(best_seq, time_budget=alloc)
        obj = _obj(a_sg)
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.10f}")
        else:
            print(f"  Subgrad: obj={obj:.10f} (no improve)")
    
    # === Phase 4: LP Alternating ===
    print(f"\n=== Phase 4: LP Alternating ===")
    if _remaining() > 60:
        alloc = min(_remaining() * 0.15, 120.0)
        a_lp = _lp_alternate(best_seq, time_budget=alloc)
        obj = _obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  LP alt: obj={best_score:.10f}")
        else:
            print(f"  LP alt: obj={obj:.10f} (no improve)")
    
    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    if _remaining() > 30:
        a_cd = _coord_descent(best_seq, n_iter=100)
        obj = _obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  Coord desc: obj={best_score:.10f}")
        else:
            print(f"  Coord desc: obj={obj:.10f} (no improve)")
    
    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best_current = len(best_seq)
    n_candidates = sorted(set([
        n_best_current,
        max(300, n_best_current - 200),
        max(300, n_best_current - 100),
        min(2000, n_best_current + 100),
        min(2000, n_best_current + 200),
        min(2000, n_best_current + 400),
        400, 500, 600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2000]
    
    for n_try in n_candidates:
        if _remaining() < 40:
            break
        if n_try == n_best_current:
            continue
        
        a_try = _resample(best_seq, n_try)
        alloc = min(_remaining() * 0.08, 30.0)
        a_opt = _gradient_optimize(a_try, time_budget=alloc)
        obj = _obj(a_opt)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")
    
    # === Phase 7: Structured Perturbations ===
    print(f"\n=== Phase 7: Structured Perturbations ===")
    n_cur = len(best_seq)
    for mode in range(6):
        if _remaining() < 30:
            break
        a_pert = _structured_perturb(best_seq, mode, rng)
        alloc = min(_remaining() * 0.06, 30.0)
        a_opt = _gradient_optimize(a_pert, time_budget=alloc)
        obj = _obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  Perturb {mode}: obj={best_score:.10f} **")
        else:
            print(f"  Perturb {mode}: obj={obj:.10f}")
    
    # === Phase 8: Final Polish (combined) ===
    print(f"\n=== Phase 8: Final Polish ===")
    n_final = len(best_seq)
    
    # LSE with very fine tau
    if _remaining() > 30:
        alloc = min(_remaining() * 0.15, 60.0)
        a_f = _gradient_optimize(best_seq, time_budget=alloc)
        obj = _obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final LSE: obj={best_score:.10f}")
    
    # Subgradient polish
    if _remaining() > 20:
        alloc = min(_remaining() * 0.08, 30.0)
        a_s = _subgrad_optimize(best_seq, time_budget=alloc)
        obj = _obj(a_s)
        if obj < best_score:
            best_score = obj
            best_seq = a_s.copy()
            print(f"  Final subgrad: obj={best_score:.10f}")
    
    # LP polish
    if _remaining() > 30:
        alloc = min(_remaining() * 0.1, 40.0)
        a_l = _lp_alternate(best_seq, time_budget=alloc)
        obj = _obj(a_l)
        if obj < best_score:
            best_score = obj
            best_seq = a_l.copy()
            print(f"  Final LP: obj={best_score:.10f}")
    
    # Coordinate descent polish
    if _remaining() > 20:
        a_c = _coord_descent(best_seq, n_iter=50)
        obj = _obj(a_c)
        if obj < best_score:
            best_score = obj
            best_seq = a_c.copy()
            print(f"  Final coord: obj={best_score:.10f}")
    
    # === Phase 9: Random Restarts ===
    print(f"\n=== Phase 9: Random Restarts ===")
    n_r = len(best_seq)
    n_restarts = min(8, max(3, int(_remaining() / 25)))
    
    for r in range(n_restarts):
        if _remaining() < 25:
            break
        
        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_r))
        elif r == 1:
            n_alt = n_r + int(rng.integers(-150, 150))
            n_alt = max(400, min(1800, n_alt))
            a_r = _resample(best_seq, n_alt)
            a_r = a_r * np.exp(rng.normal(0, 0.02, n_alt))
        elif r == 2:
            # Symmetric version
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = a_r * np.exp(rng.normal(0, 0.02, n_r))
        elif r == 3:
            # Raised cosine
            x = np.linspace(0, 2*np.pi, n_r)
            a_r = 0.5 + 0.5*np.cos(x) + 0.3
            a_r = a_r + 0.05 * rng.normal(0, 1, n_r)
        elif r == 4:
            # Gaussian
            a_r = np.exp(-((np.arange(n_r) - n_r/2.0)**2) / (2*(n_r*0.25)**2))
            a_r = a_r + 0.05 * rng.normal(0, 1, n_r)
        else:
            a_r = rng.uniform(0.1, 1.0, n_r)
        
        a_r = np.maximum(a_r, 0)
        a_r = a_r / np.sum(a_r)
        
        alloc = min(_remaining() * 0.06, 25.0)
        a_opt = _gradient_optimize(a_r, time_budget=alloc)
        obj = _obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_r = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
