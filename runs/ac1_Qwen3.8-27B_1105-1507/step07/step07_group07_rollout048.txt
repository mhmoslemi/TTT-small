Let me think about the top-K smoothing more carefully.

Instead of log-sum-exp over all 2n-1 values of conv, we can use:
- Sort conv values
- Take top K values
- Compute weighted average (with exponential weights based on how much above the median they are)

Or even simpler: just use the max (subgradient = 0 for all non-max lags, = a[k*-i] for the max lag). This is what the exponentiated gradient does, and it's actually quite effective for this type of problem.

The issue with pure subgradient (max) is oscillation: you reduce one peak, another takes over, etc. This is mitigated by:
- Momentum (average past gradients)
- Smoothing (use log-sum-exp or top-K average instead of pure max)
- Adaptive step size

Let me implement a hybrid:
- Use a "smoothed max" that's a convex combination of the top few peaks
- Use Nesterov momentum
- Adaptive step size with backtracking

```
Initialize momentum m = 0

for iteration:

    a = a + m

```

def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""

    return 2.0 * n * float(np.max(conv))

def subgradient_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Subgradient descent with Nesterov momentum and top-K smoothing.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    lr = 0.05
    beta = 0.9  # momentum
    top_k = 10  # number of top peaks to use
    
    # Adaptive parameters
    patience = 0
    max_patience = 100
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov lookahead
        a_look = a + beta * momentum
        a_look = np.maximum(a_look, 1e-30)
        a_look = a_look / np.sum(a_look)
        
        # Compute convolution at lookahead point
        conv = compute_conv(a_look)
        
        # Top-K weights
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]
        # Softmax weights over top values
        w_max = np.max(top_vals)
        w_exp = np.exp(top_vals - w_max)
        w_top = w_exp / np.sum(w_exp)
        
        # Build full weight vector
        w = np.zeros(2 * n - 1)
        w[top_idx] = w_top
        
        # Compute gradient
        grad = subgradient_step(a_look, w, lr, beta)
        
        # Momentum update
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)
        
        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            # Slightly increase lr
            lr = min(lr * 1.001, 0.1)
        else:
            a = a  # keep old
            patience += 1
            lr *= 0.95
            if patience > max_patience:
                # Reset momentum and perturb
                momentum = np.zeros(n)
                lr = 0.05
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                patience = 0
                max_patience = int(max_patience * 1.5)
        
        if verbose and it % 1000 == 0:
            print(f"    SG it={it}: obj={best_obj:.8f}, lr={lr:.5f}")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_budget=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    try:
        A = build_conv_matrix(a_norm)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternate_optimize(a_init, time_budget=60.0, verbose=False):
    """
    LP alternating optimization.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search over blend parameter
        found_improvement = False
        for t in [0.5, 0.3, 0.7, 0.9, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                found_improvement = True
                break
        
        if found_improvement:
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 5:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 5, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
    
    return best_a


def cosine_basis_optimize(a_init, time_budget=30.0, n_bands=20):
    """
    Optimize using Fourier cosine/sine basis parameterization.
    a[i] = max(0, sum_j (c_j * cos(j*pi*x_i) + d_j * sin(j*pi*x_i)))
    where x_i = i/(n-1)
    """
    n = len(a_init)
    x = np.linspace(0, 1, n)
    
    # Build basis matrix
    # Basis: 1, cos(pi*x), sin(pi*x), cos(2*pi*x), sin(2*pi*x), ...
    basis_cols = [np.ones(n)]
    for j in range(1, n_bands + 1):
        basis_cols.append(np.cos(j * np.pi * x))
        basis_cols.append(np.sin(j * np.pi * x))
    B = np.column_stack(basis_cols)  # shape (n, 2*n_bands+1)
    
    n_params = B.shape[1]
    
    # Initial parameters: project a_init onto the basis
    params0 = np.linalg.lstsq(B, a_init, rcond=None)[0]
    
    def basis_obj(params):
        a = B @ params
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            return 1e10
        a = a / s
        return compute_obj(a)
    
    # Optimize
    best_params = params0.copy()
    best_val = basis_obj(params0)
    start = time.monotonic()
    
    # Use Powell's method (doesn't need gradients)
    res = minimize(basis_obj, params0, method='Powell',
                   options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-10})
    
    if time.monotonic() - start < time_budget:
        if res.fun < best_val:
            best_val = res.fun
            best_params = res.x
    
    # Also try Nelder-Mead
    if time.monotonic() - start < time_budget:
        res2 = minimize(basis_obj, params0, method='Nelder-Mead',
                        options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
        if res2.fun < best_val:
            best_val = res2.fun
            best_params = res2.x
    
    a_opt = B @ best_params
    a_opt = np.maximum(a_opt, 0)
    a_opt = a_opt / np.sum(a_opt)
    
    return a_opt


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Hann window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a + 0.05 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 3:
        # Cosine window (raised cosine)
        a = np.cos(np.pi * x) ** 2
        a = a + 0.05 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 4:
        # Asymmetric window (exponential decay from one side)
        a = np.exp(-3 * x) + 0.3
        a = a / np.sum(a)
    elif mode == 5:
        # Asymmetric (parabolic, shifted)
        c = 0.4 + 0.2 * rng.uniform(0, 1)
        a = 1.0 - np.abs(x - c) / max(c, 1-c)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 6:
        # Gaussian bump
        c = 0.3 + 0.4 * rng.uniform(0, 1)
        w = 0.1 + 0.2 * rng.uniform(0, 1)
        a = np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a + 0.05
        a = a / np.sum(a)
    else:
        # Sum of two Gaussians (asymmetric)
        c1 = 0.2 + 0.3 * rng.uniform(0, 1)
        c2 = 0.5 + 0.3 * rng.uniform(0, 1)
        w1 = 0.1 + 0.15 * rng.uniform(0, 1)
        w2 = 0.1 + 0.15 * rng.uniform(0, 1)
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + \
            np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def high_freq_subgradient(a_init, time_budget=40.0, verbose=False):
    """
    Fast subgradient method with very aggressive top-1 (pure max) approach.
    Uses adaptive step size and momentum.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    lr = 0.03
    beta = 0.95  # high momentum for smoothness
    patience = 0
    
    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max conv at k_star
        # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Project to zero-mean (scale invariance)
        grad = grad - np.mean(grad)
        
        # Nesterov momentum
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                lr *= 0.9
            if patience > 200:
                # Restart momentum
                momentum = np.zeros(n)
                lr = 0.03
                # Small perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                patience = 0
        
        if verbose and it % 5000 == 0:
            print(f"    HF-SG it={it}: obj={best_obj:.8f}")
    
    return best_a


def accelerate_with_smoothing(a_init, time_budget=60.0, verbose=False):
    """
    Subgradient with log-sum-exp smoothing and adaptive temperature.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.03
    
    # Temperature schedule
    taus = [5.0/n, 2.0/n, 1.0/n, 0.5/n, 0.2/n, 0.1/n, 0.05/n, 0.02/n]
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        tau_start = time.monotonic()
        tau_budget = min(time_budget - (tau_start - start), time_budget / len(taus))
        
        patience = 0
        it = 0
        while time.monotonic() - tau_start < tau_budget:
            if time.monotonic() - start > time_budget:
                break
            
            conv = compute_conv(a)
            
            # Log-sum-exp weights
            conv_scaled = conv / tau
            c_max = np.max(conv_scaled)
            exp_c = np.exp(conv_scaled - c_max)
            w = exp_c / np.sum(exp_c)
            
            # Gradient
            grad = subgradient_step(a, w, lr, beta)
            
            # Nesterov
            momentum = beta * momentum - lr * grad
            a_new = a + momentum
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / np.sum(a_new)
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                patience = 0
                lr = min(lr * 1.002, 0.06)
            else:
                patience += 1
                lr *= 0.98
                if patience > 100:
                    momentum = np.zeros(n)
                    lr = 0.03
                    patience = 0
            
            it += 1
        
        if verbose:
            print(f"    AS tau={tau:.4f}: obj={best_obj:.8f} ({it} iters)")
    
    return best_a
