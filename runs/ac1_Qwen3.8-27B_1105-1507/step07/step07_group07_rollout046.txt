= a[k-j] * I(0<=k-j<n) + a[j-k] * I(0<=j-k<n)

For k in [0, 2n-2]:
- If j in [0, k]: the first term a[k-j] is valid (k-j in [0,k] ⊆ [0,n-1] if k < n)
- If j in [k-n+2, n-1]: the second term a[j-k] is valid

This is getting complicated. Let me just use the FFT-based approach for the gradient.

For the subgradient: conv_{k*}(a) = <a, S_{k*} a> where S_{k*} is the shift matrix. The gradient is (S_{k*} + S_{k*}^T) a.

In FFT terms, if we zero-pad a to length N >= 2n-1, then:
- The autocorrelation can be computed via circular correlation of the zero-padded sequence.
- The gradient at the peak lag k* is the cross-correlation of a with a shifted by k*.

"""
Optimized low-autocorrelation non-negative sequence search.
Uses Adam with log-sum-exp smoothing, subgradient with momentum, LP refinement,
spectral initialization, and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def logsumexp_conv(a, tau):
    conv = compute_conv(a)
    c = conv / tau
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    sum_exp = np.sum(exp_c)
    logsumexp_val = c_max + np.log(sum_exp)
    w = exp_c / sum_exp
    return logsumexp_val, w


def smoothed_grad(x, tau):
    """Gradient of logsumexp(conv(exp(x),exp(x))/tau) / (sum(exp(x)))^2 w.r.t. x."""
    n = len(x)
    a = np.exp(x - np.max(x))
    s = float(np.sum(a))
    conv = compute_conv(a)
    c = conv / tau
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    
    # Cross-correlation: corr[i] = sum_k w[k] * a[k-i] (with bounds)
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def subgradient_conv(a, k_star):
    """Subgradient of max conv at a, using peak lag k_star."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_star - i
        j2 = i - k_star
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad


def adam_optimize(a_init, time_budget=60.0, verbose=False):
    """Adam optimizer with log-sum-exp smoothing and adaptive temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = np.log(a) - np.log(a).max()
    
    # Temperature schedule: many fine steps
    base_tau = 5.0 / n
    n_temps = 12
    taus = [base_tau * (0.5 ** i) for i in range(n_temps)]
    temps = [1.0 / (n * 20), 0.5 / (n * 20), 0.2 / (n * 20), 0.1 / (n * 20)]
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    
    for t_idx, tau in enumerate(taus):
        if time.monotonic() - start > time_budget:
            break
        
        # Align m, v with current x scale
        lr = temps[min(t_idx, len(temps)-1)]
        
        n_iters = min(300, max(50, int(time_budget * 15 / n_temps)))
        
        for it in range(n_iters):
            if time.monotonic() - start > time_budget:
                break
            
            F, grad = smoothed_grad(x, tau)
            t_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t_count)
            v_hat = v / (1 - beta2**t_count)
            
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            x = x - np.max(x)  # renormalize for stability
            a = np.exp(x)
            a = a / np.sum(a)
            
            if it % 50 == 0 or it == n_iters - 1:
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
        if verbose:
            print(f"    Adam tau={tau:.6f}: obj={best_obj:.8f}")
    
    return best_a


def subgradient_optimize(a_init, time_budget=60.0, verbose=False):
    """Subgradient method with momentum and Polyak averaging for the true max objective."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Polyak step size: need upper bound H
    # H = 2n * max_possible_conv / (sum^2). For sum=1, max conv <= 1 (all mass at one point)
    # So H = 2n. Lower bound F* >= 2n/(2n-1) approx 1.
    H = 2.0 * n
    F_lower = 1.0  # rough lower bound
    
    momentum = np.zeros(n)
    beta = 0.9
    x_avg = a.copy()
    
    n_iters = 5000
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        obj = 2.0 * n * conv[k_star]
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Polyak step size
        gap = obj - F_lower
        step = gap / (H + 1e-10)
        
        # Subgradient
        sg = subgradient_conv(a, k_star)
        
        # Projected gradient step with momentum
        momentum = beta * momentum - step * sg
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Polyak averaging
        t = float(it + 1)
        x_avg = (t * x_avg + a_new) / (t + 1)
        
        if obj < compute_obj(x_avg):
            pass  # keep x_avg
        else:
            x_avg = a_new.copy()
        
        a = a_new
        
        # Decay momentum
        if it % 500 == 0 and it > 0:
            beta = max(0.5, beta * 0.8)
    
    # Check averaged solution
    obj_avg = compute_obj(x_avg)
    if obj_avg < best_obj:
        best_obj = obj_avg
        best_a = x_avg.copy()
    
    if verbose:
        print(f"    Subgrad: obj={best_obj:.8f}")
    
    return best_a


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=5.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 500:
        # For large n, use scipy linprog with HiGHS
        try:
            a_norm = a / np.sum(a)
            A = build_conv_matrix(a_norm)
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * (n + 1)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'timelimit': timeout})
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(res.x[n])
        except Exception:
            pass
    else:
        try:
            a_norm = a / np.sum(a)
            A = build_conv_matrix(a_norm)
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * (n + 1)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(res.x[n])
        except Exception:
            pass
    return None, None


def lp_refine(a, time_budget=30.0, verbose=False):
    """LP-based refinement with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    num_steps = min(20, int(time_budget / 3))
    
    for step in range(num_steps):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a, timeout=min(3.0, time_budget - (time.monotonic() - start)))
        if g_lp is None:
            break
        
        # Line search
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            # Perturb and try again
            rng = np.random.default_rng(step * 12345 + 678)
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            if step > 5:
                break
    
    if verbose:
        print(f"    LP refine: obj={best_obj:.8f}")
    
    return best_a


def spectral_init(n, rng):
    """Generate initial sequence using spectral methods for low autocorrelation."""
    mode = rng.integers(0, 4)
    if mode == 0:
        #Raised cosine
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.5
        a = a + 0.1 * rng.normal(0, 1, n)
    elif mode == 1:
        #Two-lobe
        c = n * 0.3
        a = np.exp(-((np.arange(n) - c) ** 2) / (n * 0.1)) + \
            np.exp(-((np.arange(n) - (n - c)) ** 2) / (n * 0.1)) + 0.2
        a = a + 0.05 * rng.normal(0, 1, n)
    elif mode == 2:
        #Spectral flat: random phase in frequency domain
        N = next_pow2(2 * n - 1)
        mag = np.ones(N // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        A = mag * np.exp(1j * phase)
        A[0] = mag[0]
        a = irfft(A, N)[:n]
        a = a - np.min(a) + 0.1
    else:
        #Smooth random
        a = rng.uniform(0, 1, n)
        #Apply low-pass filter
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=n * 0.05)
        a = a + 0.1
    
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)


def combined_optimize(a_init, time_budget=120.0, verbose=False):
    """Run Adam + subgradient + LP in sequence."""
    n = len(a_init)
    start = time.monotonic()
    
    # Phase A: Adam with log-sum-exp
    alloc_a = min(time_budget * 0.5, 60.0)
    a1 = adam_optimize(a_init, time_budget=alloc_a, verbose=verbose)
    elapsed = time.monotonic() - start
    if verbose:
        print(f"    After Adam: {elapsed:.1f}s, obj={compute_obj(a1):.8f}")
    
    # Phase B: Subgradient with momentum
    remaining = time_budget - elapsed
    if remaining > 15:
        alloc_b = min(remaining * 0.4, 40.0)

        a3 = lp_refine(a1, time_budget=alloc_c, verbose=verbose)

            if n_try < 50:
                continue
            a_try = resample_sequence(best_seq, n_try)
            # Quick single subgradient pass
            a_try = subgradient_optimize(a_try, time_budget=min(5.0, remaining * 0.3), verbose=False)
            obj = compute_obj(a_try)
            if obj < best_score:
                best_score = obj
                best_seq = a_try.copy()
                print(f"  Length tweak n={n_try}: obj={best_score:.10f} **")
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
