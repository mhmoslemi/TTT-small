        max_iter = max(50, int(remaining * 5))
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 50})
        
        if res.x is not None:
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.5f}: obj={best_obj:.10f}")
    
    return best_a


def topk_smoothed_obj_and_grad(a, tau, K=3):
    """
    Objective based on sum of top K peaks (instead of log-sum-exp over all peaks).
    This focuses optimization on the most important peaks.
    """
    n = len(a)
    conv = compute_conv(a)
    S = float(np.sum(a))
    
    # Get top K values
    top_k_vals = np.sort(conv)[-K:][::-1]
    
    # Smoothed max of top K: logsumexp of top K / tau
    # Actually, let's use: average of top K peaks (simpler)
    # Or: logsumexp of all peaks but with emphasis on top ones
    
    # Let's use: (sum of top K)^1/K / S^2 (geometric mean of top K)
    # Or simpler: just use the mean of top K
    peak_val = float(np.mean(top_k_vals))
    obj = 2.0 * n * peak_val / (S * S)
    
    # Gradient: d(peak_val)/da_i where peak_val = mean of top K conv values
    # For a smooth approximation, use softmax over top K
    top_k_scaled = top_k_vals / tau
    top_k_max = np.max(top_k_scaled)
    exp_topk = np.exp(top_k_scaled - top_k_max)
    W_topk = exp_topk / np.sum(exp_topk)
    
    # Get the indices of top K
    top_k_idx = np.argsort(conv)[-K:][::-1]
    
    # corr[i] = sum_{k in top_K} W_topk[k] * 2 * a[k-i] * 1_{0<=k-i<n}
    size = next_pow2(2 * n - 1)
    corr = np.zeros(n)
    for idx, w in zip(top_k_idx, W_topk):
        k = int(idx)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                corr[i] += w * 2.0 * a[j]
    
    grad_a = (2.0 * n / (S * S)) * corr / K - (2.0 * n * peak_val) / (S * S * S)
    # Hmm, this isn't quite right either. Let me just use the standard logsumexp gradient
    # but computed only over top K peaks.
    
    return obj, grad_a


def lp_polish(a, time_budget=30.0):
    """
    LP-based polishing: solve LP to find g that minimizes peak of conv(a,g),
    then blend.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Build constraint matrix: A[g] = conv(a, g), so A is (2n-1) x n
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    
    from scipy.optimize import linprog
    # LP: min M s.t. A@b <= M*1, sum(b)=1, b>=0
    # Variables: [b (n), M (1)]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@b - M <= 0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    while time.monotonic() - start < time_budget:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g = np.maximum(res.x[:n], 0)
                sg = float(np.sum(g))
                if sg > 1e-10:
                    g = g / sg
                    M_opt = float(res.x[n])
                    
                    # Line search for blending
                    for t in [0.5, 0.7, 0.9, 1.0, 0.3, 0.6, 0.8]:
                        a_try = (1 - t) * best_a + t * g
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try > 1e-10:
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = a_try
                    # Update A for next iteration
                    for k in range(2 * n - 1):
                        for i in range(n):
                            j = k - i
                            if 0 <= j < n:
                                A[k, i] = a[j]
        except Exception:
            break
    
    return best_a


def diverse_init(n, rng, mode):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / np.sum(a)
    elif mode == 2:
        # Smooth (low-pass)
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif mode == 3:
        # Raised cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x - np.pi))
        a = a + 0.1 * rng.uniform(0, 1, n)
        return a / np.sum(a)
    elif mode == 4:
        # Sinc^2 inspired
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
        a = a + 0.05 * rng.uniform(0, 1, n)
        return a / np.sum(a)
    elif mode == 5:
        # Two-lobe
        x = np.arange(n)
        a = np.exp(-((x - n*0.25)**2) / (n*0.05)) + np.exp(-((x - n*0.75)**2) / (n*0.05))
        a = a + 0.05 * rng.uniform(0, 1, n)
        return a / np.sum(a)
    elif mode == 6:
        # Flat with edge tapering
        a = np.ones(n)
        taper = min(n // 5, 50)
        for i in range(taper):
            a[i] *= (i + 1) / taper
            a[n-1-i] *= (i + 1) / taper
        a = a + 0.05 * rng.uniform(0, 1, n)
        return a / np.sum(a)
    elif mode == 7:
        # Parabolic
        x = np.linspace(-1, 1, n)
        a = 1 - x * x
        a = a + 0.1 * rng.uniform(0, 1, n)
        return a / np.sum(a)
    else:
        # Frequency domain construction
        N = next_pow2(2 * n - 1)
        freqs = np.arange(N // 2 + 1)
        # Create a spectrum with controlled flatness
        spec_mag = 1.0 / (1.0 + rng.exponential(0.5, N // 2 + 1))
        spec_phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        spec = spec_mag * np.exp(1j * spec_phase)
        # Make it conjugate symmetric
        spec_full = np.zeros(N, dtype=complex)
        spec_full[:N // 2 + 1] = spec
        for i in range(1, N // 2):
            if N - i < N // 2 + 1:
                spec_full[N - i] = np.conj(spec[i])
            else:
                spec_full[N - i] = np.conj(spec_full[i - N + N])  # wrap
        # Actually, let me just use ifft of a random conjugate-symmetric spectrum
        spec_cs = np.zeros(N, dtype=complex)
        for k in range(N // 2 + 1):
            if k == 0 or k == N // 2:
                spec_cs[k] = rng.uniform(0.5, 1.0)
            else:
                spec_cs[k] = spec_mag[k] * np.exp(1j * spec_phase[k])
                spec_cs[N - k] = np.conj(spec_cs[k])
        a = np.real(ifft(spec_cs))[:n]
        a = np.abs(a) + 0.01
        return a / np.sum(a)


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
