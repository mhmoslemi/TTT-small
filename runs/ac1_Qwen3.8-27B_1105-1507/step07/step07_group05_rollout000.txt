Uses smoothed-max Adam GD, multiplicative weight updates, LP alternating,

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    """Build matrix B (2n-1 x n) such that B @ g = conv(a, g)."""

    return B

def solve_lp(a):
    """Solve LP: min M s.t. B@g <= M*1, sum(g)=1, g>=0."""
    n = len(a)
    B = build_conv_matrix(a)
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [B @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    try:
        from scipy.optimize import linprog
        # Variables: [g_0, ..., g_{n-1}, M]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([B, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None

def smoothed_max_gradient(a, tau, N):
    """
    Compute gradient of tau*log(sum(exp(b/tau))) where b = conv(a,a).
    Returns (gradient, b, softmax_weights).
    """
    n = len(a)
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2*n-1]
    
    # Softmax weights
    b_shift = b - np.max(b)
    p = np.exp(b_shift / tau)
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p = p / p_sum
    
    # Gradient: grad[i] = (1/tau) * sum_k p[k] * a[k-i]
    # = (1/tau) * conv(a, p_rev)[2n-2-i] for i=0..n-1
    p_rev = p[::-1]
    fp = rfft(p_rev, N)
    conv_ap = irfft(fa * fp, N)
    # We need indices 2n-2, 2n-3, ..., n-1 from conv_ap
    # In Python: conv_ap[2*n-2 : n-2 : -1] gives n elements
    grad = conv_ap[n-1:2*n-1][::-1] / tau
    
    return grad, b, p

def adam_smoothed_gd(a, time_budget=60.0, lr=0.05, tau_start=0.005, tau_end=0.0005, verbose=False):
    """Adam optimizer with smoothed max objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    
    start = time.monotonic()
    stall = 0
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        t += 1
        
        # Anneal temperature
        progress = min(1.0, t / 30000)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        grad, b, p = smoothed_max_gradient(a, tau, N)
        
        # Center gradient (sum constraint)
        grad_mean = np.mean(grad)
        grad = grad - grad_mean
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            if lr < 1e-8:
                break
            continue
        a_new = a_new / s_new
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.01, 0.5)
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.95
            if stall > 500:
                # Reset
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = 0.05
                stall = 0
            if lr < 1e-8:
                break
        
        if verbose and it % 5000 == 0:
            print(f"            Adam it{it}: obj={best_obj:.8f}, tau={tau:.6f}, lr={lr:.4f}")
    
    return best_a

def multiplicative_wu(a, time_budget=30.0, eta=0.1, verbose=False):
    """
    Multiplicative weight update (mirror descent on simplex).
    a_i <- a_i * exp(-eta * grad_i) / Z
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    start = time.monotonic()
    stall = 0
    
    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2*n-1]
        k_star = int(np.argmax(b))
        
        # Subgradient of max at k_star: grad[i] = a[k_star - i] if 0 <= k_star-i < n
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            grad[i] = a[k_star - i]
        
        # Center
        grad = grad - np.mean(grad)
        
        # Multiplicative update
        shift = eta * (grad - np.max(grad))
        a_new = a * np.exp(shift)
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.02, 1.0)
        else:
            stall += 1
            eta *= 0.99
            if stall > 200:
                a = best_a.copy()
                eta = 0.1
                stall = 0
            if eta < 1e-6:
                break
    
    return best_a

def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        g = solve_lp(a)
        if g is None:
            break
        
        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 15:
                break
    
    return best_a

def random_init(n, rng):
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5)**2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        c1, c2 = n//4, 3*n//4
        a = np.exp(-((np.arange(n) - c1)**2) / (n//8)**2) + np.exp(-((np.arange(n) - c2)**2) / (n//8)**2)
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    elif mode == 6:
        # S-shaped (S-curve)
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
        a = a / np.sum(a)
    else:
        # Multi-modal
        n_peaks = rng.integers(3, 7)
        centers = rng.uniform(0, n, n_peaks).astype(int)
        a = np.zeros(n)
        for c in centers:
            width = max(n // (20 * n_peaks), 2)
            a += np.exp(-((np.arange(n) - c)**2) / (2 * width**2))
        a = a + 0.01 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    return a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)
