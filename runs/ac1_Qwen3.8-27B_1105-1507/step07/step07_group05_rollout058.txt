        # So k ranges from j to j+n-1
        start = j
        end = j + n
        A[start:end, j] = a[:n]
    return A

    # Variables: g[0..n-1], M
    # c = [0, 0, ..., 0, 1] (minimize M)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ g - M <= 0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

        M_opt = float(res.x[n])
        # Dual variables for inequality constraints
        dual = res.ineqlin.marginals  # shape (2n-1,)
        dual = np.maximum(dual, 0)
        return g_opt, M_opt, dual
    return None

2. Return max(a - τ, 0)

Hybrid LP-dual + FISTA optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovations:
1. LP dual variables as optimal gradient weights
2. FISTA (accelerated proximal gradient) with simplex projection
3. Adaptive LP frequency based on convergence
4. Multi-n exploration

import numpy as np
import time
from scipy.fft import rfft, irfft

# Global cache for LP result
_lp_cache = None

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2*n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    b = compute_conv(a_n)
    return 2.0 * n * float(np.max(b))

def softmax_weights(b, T):

    w = np.exp(b_shift / T)
    s = np.sum(w)
    if s < 1e-30:
        return np.ones(len(b)) / len(b)
    return w / s

def compute_weighted_gradient(a, w):
    """
    Compute gradient of F(a) = sum_k w[k]*conv(a,a)[k] with respect to a.
    result: grad[i] = 2 * sum_m a[m] * w[m+i]
    """
    n = len(a)
    N = 1
    while N < 3*n - 2:
        N <<= 1
    
    w_rev = w[::-1]
    fa = rfft(a, N)
    fw_rev = rfft(w_rev, N)
    conv_awrev = irfft(fa * fw_rev, N)[:3*n-2]
    
    # grad[i] = 2 * conv_awrev[2n-2-i]
    # = 2 * conv_awrev[n-1:2n-1][::-1]
    grad = 2.0 * conv_awrev[n-1:2*n-1][::-1]
    return grad

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_cvxpy(a):
    """Solve LP using cvxpy with warm start."""
    import cvxpy as cp
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return None
    a = a / s
    
    # Build A matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    # Try solvers in order of preference
    for solver in [cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = np.sum(g_val)
                if gs > 1e-10:
                    g_val = g_val / gs
                    # Get dual variables
                    dual = None
                    try:
                        dual = np.array([constr.dual_value for constr in constraints[0].cons]) 
                    except:
                        pass
                    if dual is None:
                        try:
                            dual = np.maximum(constraints[0].dual_value, 0)
                        except:
                            dual = None
                    if dual is None or not np.all(np.isfinite(dual)):
                        dual = softmax_weights(compute_conv(a), 0.01)
                    return g_val, float(M.value), dual
        except Exception:
            continue
    return None

def solve_lp_scipy(a):
    """Solve LP using scipy as fallback."""
    from scipy.optimize import linprog
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return None
    a = a / s
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = np.sum(g_val)
        if gs > 1e-10:
            g_val = g_val / gs
            dual = np.maximum(res.ineqlin.marginals, 0) if hasattr(res.ineqlin, 'marginals') else None
            if dual is None:
                dual = softmax_weights(compute_conv(a), 0.01)
            return g_val, float(res.x[n]), dual
    return None

def solve_lp(a):
    """Solve LP, using cvxpy first, scipy as fallback."""
    try:
        result = solve_lp_cvxpy(a)
        if result is not None:
            return result
    except:
        pass
    return solve_lp_scipy(a)

def fista_optimize(a, weights, time_budget=60.0, momentum_start=1.0):
    """
    FISTA optimization with fixed weights.
    Minimizes F(a) = sum_k w[k]*conv(a,a)[k] subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = simplex_project(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t = 1.0
    a_prev = a.copy()
    L = 1.0 / (2 * np.sum(weights) * np.max(a + 1/n))  # rough Lipschitz estimate
    # Actually, use backtracking
    L = 4.0 * float(np.max(weights)) * float(np.sum(a)) * 2 * n  # cruder but safe
    
    Nfft = 1
    while Nfft < 3*n - 2:
        Nfft <<= 1
    
    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + (t - 1.0) / t_new * (a - a_prev)
        y = np.maximum(y, 0)
        sy = np.sum(y)
        if sy < 1e-10:
            y = a
            sy = 1.0
        else:
            y = y / sy
        
        # Backtracking line search
        step = 1.0 / L
        for _ in range(20):
            grad_y = compute_weighted_gradient(y, weights)
            center_grad = grad_y - np.sum(grad_y) / n  # center for invariance
            a_new = y - step * center_grad
            a_new = simplex_project(a_new)
            
            # Check descent condition
            F_y = float(np.dot(weights, compute_conv(y)))
            F_new = float(np.dot(weights, compute_conv(a_new)))
            # Simplified: just accept if it looks reasonable
            break  # Skip actual backtracking for speed
        
        a_new = simplex_project(y - step * (grad_y - np.mean(grad_y)))
        a_prev = a
        a = a_new
        t = t_new
        
        # Evaluate actual objective periodically
        if it % 100 == 0 or it < 10:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a

def hybrid_optimize(a, time_budget=120.0, verbose=False):
    """
    Hybrid LP + FISTA optimization.
    Alternates between LP (to get optimal weights) and FISTA (fast inner iterations).
    """
    n = len(a)
    a = simplex_project(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if verbose:
        print(f"  Hybrid start: obj={best_obj:.8f}")
    
    lp_solve = solve_lp
    lp_freq = 500  # solve LP every 500 FISTA iterations
    
    # Initial: solve LP to get good weights
    lp_result = lp_solve(a)
    if lp_result is not None:
        g_lp, M_lp, dual = lp_result
        w = dual / np.sum(dual) if np.sum(dual) > 0 else softmax_weights(compute_conv(a), 0.01)
        
        # Check if g_lp is better
        obj_g = compute_obj(g_lp)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g_lp.copy()
            a = g_lp.copy()
            if verbose:
                print(f"  LP: obj={best_obj:.8f}")
    else:
        b = compute_conv(a)
        w = softmax_weights(b, 0.01)
    
    fista_count = 0
    stall = 0
    prev_best = best_obj
    
    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # FISTA phase
        fista_time = min(remaining * 0.3, 30.0)
        a_new = fista_optimize(a, w, time_budget=fista_time)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if verbose:
                print(f"  FISTA: obj={best_obj:.8f}")
        
        # Check for stall
        if best_obj > prev_best - 1e-10:
            stall += 1
        else:
            stall = 0
        prev_best = best_obj
        
        # LP phase (more frequent if stalled)
        lp_time = min(remaining * 0.2, 20.0)
        if stall > 2 or fista_count % lp_freq == 0:
            lp_result = lp_solve(a)
            if lp_result is not None:
                g_lp, M_lp, dual = lp_result
                obj_g = compute_obj(g_lp)
                if obj_g < best_obj:
                    best_obj = obj_g
                    best_a = g_lp.copy()
                    a = g_lp.copy()
                    if verbose:
                        print(f"  LP: obj={best_obj:.8f}")
                
                # Update weights
                if np.sum(dual) > 0:
                    w = dual / np.sum(dual)
                else:
                    b = compute_conv(a)
                    T = max(0.001, 0.01 * np.exp(-fista_count / 5000))
                    w = softmax_weights(b, T)
            else:
                b = compute_conv(a)
                T = max(0.001, 0.01 * np.exp(-fista_count / 5000))
                w = softmax_weights(b, T)
        
        fista_count += 1
    
    return best_a
