"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using LP alternating,
multiplicative weight descent, FFT flat spectrum, and peak-chasing coordinate descent.
Improvements over previous: more aggressive LP usage, better step sizes, peak chasing,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_mix(a, b):
    """Linear convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, use_cvxpy=True):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1
    
    if use_cvxpy:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.HIGHS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    
    # Fallback to linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def mpd_step(a, peaks, eta, n_peaks=5):
    """Multiplicative weight descent step targeting multiple peaks."""
    n = len(a)
    conv = compute_conv(a)
    
    # Get top peaks
    top_idx = np.argsort(conv)[-n_peaks:]
    top_val = conv[top_idx]
    max_val = top_val[-1]
    
    # Softmax weights based on peak values
    temp = max(0.001, (max_val - top_val[0]) * 0.5)
    weights = np.exp((top_val - max_val) / temp)
    weights /= weights.sum()
    
    # Compute weighted subgradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        ki = int(k)
        for i in range(n):
            j = ki - i
            if 0 <= j < n:
                grad[i] += w * a[j]
    
    # Multiplicative update
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - eta * grad
    # Normalize
    log_a_new -= np.max(log_a_new)  # numerical stability
    a_new = np.exp(log_a_new)
    s = float(np.sum(a_new))
    if s > 1e-30:
        a_new = a_new / s
    return a_new, grad

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1

    # Use scipy linprog with HiGHS (fastest for this problem size)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def mpd_compress(a, n_iters=3000, time_budget=60.0, n_peaks=5):
    """Multiplicative weight descent with multi-peak subgradient and cooling."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    log_a = np.log(a)
    eta = 0.3
    cool = 0.998
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[-n_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        temp = max(0.001, (max_val - top_val[0]) * 0.5)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            for i in range(lo, hi + 1):
                j = ki - i
                grad[i] += w * a[j]

        log_a = log_a - eta * grad
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        s = float(np.sum(a_new))
        if s > 1e-15:
            a_new = a_new / s
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    eta = max(eta * 0.7, 0.01)
                if stall > 200:
                    rng = np.random.default_rng(it * 7 + n)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    log_a = np.log(a)
                    best_obj = compute_obj(a)
                    best_a = a.copy()
                    stall = 0
                    eta = 0.2
        else:
            eta *= 0.5

    return best_a


def fft_flat(a, n_iters=300, time_budget=60.0, relaxation=0.5):
    """FFT flat spectrum iteration (Gerchberg-Saxton style)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    cur_relax = relaxation

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe
        A_new = (1 - cur_relax) * A + cur_relax * phase

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                cur_relax = max(0.1, cur_relax * 0.997)
            else:
                stall += 1
                if stall > 25:
                    rng = np.random.default_rng(it * 3 + n)
                    a = best_a * np.exp(rng.normal(0, 0.06, n))

                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

def peak_chase(a, time_budget=30.0):
    """Targeted peak reduction via mass redistribution."""
    n = len(a)
    a = np.maximum(a, 0).copy()

    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Find contributing pairs for the peak
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)

        # Get pair contributions
        pairs = []
        for i in range(lo, hi + 1):
            j = k_star - i
            pairs.append((i, j, a[i] * a[j]))
        pairs.sort(key=lambda x: -x[2])

        # Try reducing top contributors
        for i, j, val in pairs[:5]:
            if val < 1e-12:
                break
            for frac in [0.03, 0.07, 0.12]:
                delta = frac * a[i]
                if delta < 1e-15 or a[i] - delta < -1e-15:
                    continue
                # Try moving to several cold positions
                candidates = [(i + n // 3) % n, (i + 2 * n // 3) % n,
                              max(0, i - k_star - 1) % n, min(n - 1, i + k_star + 1)]
                for target in set(candidates):
                    if target == i or target == j:
                        continue
                    a_try = a.copy()
                    a_try[i] = a[i] - delta
                    a_try[target] = a[target] + delta
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    conv_try = compute_conv(a_try)
                    obj_try = 2.0 * n * float(np.max(conv_try))
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Small random moves
            rng = np.random.default_rng(it * 31 + n)
            for _ in range(15):
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

        widths = rng.uniform(n // 15, n // 4, n_modes)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 4:
        # Alternating pattern
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 1.5, n // 2 + 1)
        a[1::2] = rng.uniform(0.1, 0.7, n // 2)
        a = a / a.sum()
    elif mode == 5:
        # Exponential decay
        rate = rng.uniform(0.01, 0.1)
        a = np.exp(-rate * np.arange(n))
        a = a / a.sum()
    elif mode == 6:
        # Smoothed random
        a = rng.uniform(0, 1, n)
        k = max(3, n // 40)
        if k % 2 == 0:
            k += 1
        a = np.convolve(a, np.ones(k) / k, mode='same')
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        # Binary-like
        a = (rng.random(n) > 0.3).astype(float)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with momentum, MPD, FFT flat spectrum,
    peak chasing, multi-n exploration, and diverse random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_p = len(best_seq)
    print(f"Starting: n={n_p}, obj={best_score:.8f}")

    # === Phase 1: Quick MPD warmup ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.05, 40.0)
        print(f"\n[Phase 1] MPD warmup (budget={alloc:.0f}s)")
        a1 = mpd_compress(best_seq, n_iters=3000, time_budget=alloc, n_peaks=5)
        obj1 = compute_obj(a1)
        print(f"  obj={obj1:.8f} {'**' if obj1 < best_score else ''}")
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()

    # === Phase 2: FFT flat spectrum ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n[Phase 2] FFT flat spectrum")
        for relax in [0.5, 0.3]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.05, 25.0)
            a2 = fft_flat(best_seq, n_iters=400, time_budget=alloc, relaxation=relax)
            obj2 = compute_obj(a2)
            print(f"  relax={relax}: obj={obj2:.8f} {'**' if obj2 < best_score else ''}")
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()

    # === Phase 3: LP Alternating (main workhorse) ===
    n_lp = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n[Phase 3] LP Alternating (n={n_lp})")
        # Round 1: momentum version
        alloc = min(remaining * 0.15, 90.0)
        a3a = lp_alternate(best_seq, time_budget=alloc, momentum_beta=0.3)
        obj3a = compute_obj(a3a)
        print(f"  LP momentum: obj={obj3a:.8f} {'**' if obj3a < best_score else ''}")
        if obj3a < best_score:
            best_score = obj3a
            best_seq = a3a.copy()

        # Round 2: simple version
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(remaining * 0.15, 90.0)
            a3b = lp_simple_alternate(best_seq, time_budget=alloc)
            obj3b = compute_obj(a3b)
            print(f"  LP simple: obj={obj3b:.8f} {'**' if obj3b < best_score else ''}")
            if obj3b < best_score:
                best_score = obj3b
                best_seq = a3b.copy()

        # Round 3: more LP
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(remaining * 0.1, 60.0)
            a3c = lp_simple_alternate(best_seq, time_budget=alloc)
            obj3c = compute_obj(a3c)
            print(f"  LP round 3: obj={obj3c:.8f} {'**' if obj3c < best_score else ''}")
            if obj3c < best_score:
                best_score = obj3c
                best_seq = a3c.copy()

    # === Phase 4: MPD with different peak settings ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n[Phase 4] MPD refinement")
        for npk in [3, 8]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.06, 35.0)
            a4 = mpd_compress(best_seq, n_iters=4000, time_budget=alloc, n_peaks=npk)
            obj4 = compute_obj(a4)
            print(f"  MPD peaks={npk}: obj={obj4:.8f} {'**' if obj4 < best_score else ''}")
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()

    # === Phase 5: Peak chasing ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n[Phase 5] Peak chasing")
        alloc = min(remaining * 0.05, 30.0)
        a5 = peak_chase(best_seq, time_budget=alloc)
        obj5 = compute_obj(a5)
        print(f"  obj={obj5:.8f} {'**' if obj5 < best_score else ''}")
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (current best n={len(best_seq)})")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        300, 400, 500, 600, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and abs(n - n_best) > 50]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample(best_seq, n_try)
        # Quick pipeline: MPD + LP
        alloc_total = min(remaining * 0.1, 45.0)
        a_opt = mpd_compress(a_try, n_iters=2000, time_budget=alloc_total * 0.4, n_peaks=5)
        rem_inner = deadline - time.monotonic()
        if rem_inner > 15:
            a_opt = lp_simple_alternate(a_opt, time_budget=min(alloc_total * 0.6, rem_inner * 0.4))

        obj = compute_obj(a_opt)
        st = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{st}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 7: Diverse random restarts ===
    print(f"\n[Phase 7] Random restarts")
    n_best = len(best_seq)
    for r in range(20):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if r < 5:
            a_r = diverse_init(n_best, rng)
        elif r < 10:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
            a_r = np.maximum(a_r, 1e-6)
            a_r = a_r / a_r.sum()
        elif r < 15:
            # Different n
            n_r = int(n_best + rng.integers(-200, 200))
            n_r = max(200, min(1500, n_r))
            a_r = resample(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 1e-6)
            a_r = a_r / a_r.sum()
        else:
            a_r = diverse_init(int(n_best * rng.uniform(0.7, 1.3)), rng)

        n_r = len(a_r)
        alloc_total = min(remaining * 0.06, 35.0)
        a_opt = mpd_compress(a_r, n_iters=1500, time_budget=alloc_total * 0.5, n_peaks=5)
        rem_inner = deadline - time.monotonic()
        if rem_inner > 10:
            a_opt = lp_simple_alternate(a_opt, time_budget=min(alloc_total * 0.5, rem_inner * 0.4))

        obj_r = compute_obj(a_opt)
        st = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (n={n_r}): obj={obj_r:.8f}{st}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # === Phase 8: Final intensive polish ===
    print(f"\n[Phase 8] Final polish (n={len(best_seq)})")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Intensive LP
        alloc = min(remaining * 0.3, 90.0)
        a_f1 = lp_simple_alternate(best_seq, time_budget=alloc)
        obj_f1 = compute_obj(a_f1)
        print(f"  LP final: obj={obj_f1:.8f} {'**' if obj_f1 < best_score else ''}")
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()

        # MPD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.2, 40.0)
            a_f2 = mpd_compress(best_seq, n_iters=5000, time_budget=alloc, n_peaks=5)
            obj_f2 = compute_obj(a_f2)
            print(f"  MPD final: obj={obj_f2:.8f} {'**' if obj_f2 < best_score else ''}")
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()

        # Peak chase
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.15, 25.0)
            a_f3 = peak_chase(best_seq, time_budget=alloc)
            obj_f3 = compute_obj(a_f3)
            print(f"  Peak final: obj={obj_f3:.8f} {'**' if obj_f3 < best_score else ''}")
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.35, 50.0)
            a_f4 = lp_simple_alternate(best_seq, time_budget=alloc)
            obj_f4 = compute_obj(a_f4)
            print(f"  LP final2: obj={obj_f4:.8f} {'**' if obj_f4 < best_score else ''}")
            if obj_f4 < best_score:
                best_score = obj_f4
                best_seq = a_f4.copy()

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = min(best_score, val)
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
