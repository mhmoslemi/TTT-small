def compute_conv(a):
    """Fast convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Fast linear convolution a*a via FFT. Returns length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_cg(a, max_rounds=30):

    A_ub = np.zeros((2 * n - 1, n + 1))

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

def spectral_init(n, n_iters=2000, temperature=0.4, init_type='flat'):
    """
    Iterative flat spectrum with non-negativity projection.
    Makes |DFT(a)| approximately constant -> flat autocorrelation.
    """
    if init_type == 'raised_cosine':
        x = np.arange(n) / n
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        a = np.maximum(a, 1e-6)
    elif init_type == 'sinusoidal':
        x = np.arange(n) / n
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 1e-6)
    else:
        a = np.ones(n)

    a = a / a.sum()
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    for it in range(n_iters):
        A = fft(a, L)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-15)

        # Target: flat magnitude with slight taper at edges
        target_mag = np.ones(L)
        # Slight taper to help with non-negativity
        t = np.arange(L)
        target_mag *= 1.0 - 0.1 * np.cos(2 * np.pi * t / L) ** 2

        A_new = phase * target_mag
        # Relaxed update
        A_mix = (1 - temperature) * A + temperature * A_new
        a_new = ifft(A_mix).real[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new /= s

        if it > 100 and np.sum(np.abs(a_new - a)) < 1e-12:
            break
        a = a_new

    return a

def mirror_descent_multi(a, n_iters=3000, time_budget=40.0):
    """Mirror descent with multi-peak softmax subgradient."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.3
    m_peaks = 8

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[-m_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        # Softmax weights (sharper = more focus on top peak)
        temp = max(0.001, (max_val - top_val[0]) * 0.8)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        # Weighted subgradient (vectorized)
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * a[j]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad /= grad_norm

        # Mirror descent step
        a_new = a * np.exp(-alpha * grad * 10)
        a_new = np.maximum(a_new, 1e-15)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new /= s
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.97
        else:
            alpha *= 0.5

        # Periodically boost alpha
        if it % 500 == 499 and alpha < 0.3:
            alpha = min(alpha * 1.5, 0.5)

        if alpha < 0.005:
            break

    return best_a

                continue

            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = frac * a[i]
                a[i] -= delta
                a[k_new] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a[i] += delta
                    a[k_new] -= delta
                    break
                a /= s
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[k_new] -= delta
                    break

        if not improved and it > 100:
            # Try global perturbation
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a /= a.sum()
            obj_new = compute_obj(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()

        if not improved and it > 200:
            break

    return best_a


def simulated_annealing_search(a_init, n, time_budget=60.0):
    """SA with peak-targeting moves."""
    rng = np.random.default_rng(999)
    a = np.maximum(a_init, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 0.1
    T_min = 0.001
    cooling = 0.995

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        if T < T_min:
            break

        # Propose move
        move_type = rng.integers(0, 3)
        a_try = a.copy()

        if move_type == 0:
            # Perturb random elements
            idx = rng.integers(0, n, size=5)
            for i in idx:
                a_try[i] *= np.exp(rng.normal(0, 0.3))
        elif move_type == 1:
            # Peak-targeting: reduce contributing, increase non-contributing
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
            i = rng.integers(0, n)
            j = k_star - i
            if 0 <= j < n:
                k_new = rng.integers(0, n)
                delta = 0.05 * a_try[i]
                a_try[i] -= delta
                a_try[k_new] += delta
        else:
            # Swap mass between two positions
            i1, i2 = rng.integers(0, n, 2)
            if i1 != i2:
                delta = 0.03 * max(a_try[i1], a_try[i2])
                a_try[i1] -= delta
                a_try[i2] += delta

        a_try = np.maximum(a_try, 1e-10)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try /= s

        obj_try = compute_obj(a_try)
        delta_obj = obj_try - best_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()

        T *= cooling

    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines spectral init, LP alternating, multi-n exploration,
    peak-targeted coordinate descent, mirror descent, and SA diversification.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Spectral initialization and quick LP (0-90s) ===
    print(f"\n=== Phase 1: Spectral Init + Quick LP ===")
    n_cur = len(best_seq)
    remaining = deadline - time.monotonic()

    # Try different spectral initializations
    for init_type in ['flat', 'raised_cosine', 'sinusoidal']:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_test = spectral_init(n_cur, n_iters=1500, temperature=0.4, init_type=init_type)
        obj = compute_obj(a_test)
        print(f"  Spectral ({init_type}): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_test.copy()

    # Quick LP on best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_lp = lp_alternate(best_seq, max_iters=50, time_budget=min(remaining * 0.3, 40.0), verbose=True)
        obj = compute_obj(a_lp)
        print(f"  Quick LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # === Phase 2: Intensive LP alternating (90-360s) ===
    print(f"\n=== Phase 2: Intensive LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 180.0)
        a_lp = lp_alternate(best_seq, max_iters=300, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  LP phase2: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # === Phase 3: Multi-n exploration (360-720s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = []
    for n_try in range(400, 3001, 100):
        if abs(n_try - n_best) > 50:
            n_options.append(n_try)
    # Add specific values
    for extra in [n_best - 50, n_best + 50, n_best + 100, n_best + 200, 500, 1000, 1500, 2000]:
        if 300 <= extra <= 3000 and extra not in n_options:
            n_options.append(extra)
    n_options = sorted(set(n_options))

    # Sort by proximity to best first, then explore
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if abs(n_try - n_best) < 30:
            continue

        # Resample and optimize
        a_try = resample(best_seq, n_try)

        # Spectral init at new n
        a_spec = spectral_init(n_try, n_iters=800, temperature=0.4)
        # Blend
        a_blend = 0.4 * a_spec + 0.6 * a_try
        a_blend = np.maximum(a_blend, 1e-10)
        a_blend /= a_blend.sum()

        alloc = min(remaining * 0.08, 25.0)
        a_opt = lp_alternate(a_blend, max_iters=40, time_budget=alloc)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try

    # === Phase 4: Mirror descent + SA on best (720-1080s) ===
    print(f"\n=== Phase 4: MD + SA (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.2, 80.0)
        a_md = mirror_descent_multi(best_seq, n_iters=5000, time_budget=alloc)
        obj = compute_obj(a_md)
        print(f"  MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 50.0)
        a_sa = simulated_annealing_search(best_seq, len(best_seq), time_budget=alloc)
        obj = compute_obj(a_sa)
        print(f"  SA: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_sa.copy()

    # === Phase 5: Peak-targeted CD (1080-1260s) ===
    print(f"\n=== Phase 5: Peak Targeting CD (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.1, 40.0)
        a_cd = peak_targeting_cd(best_seq, time_budget=alloc)
        obj = compute_obj(a_cd)
        print(f"  CD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()

        # LP after CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.15, 30.0)
            a_lp2 = lp_alternate(best_seq, max_iters=50, time_budget=alloc)
            obj = compute_obj(a_lp2)
            print(f"  LP after CD: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_lp2.copy()

    # === Phase 6: Final intensive polish (1260-1480s) ===
    print(f"\n=== Phase 6: Final Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Multiple LP rounds
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.25, 50.0)
            a_final = lp_alternate(best_seq, max_iters=100, time_budget=alloc, verbose=True)
            obj = compute_obj(a_final)
            print(f"  Final LP rnd{rnd}: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.2, 20.0)
            a_cd2 = peak_targeting_cd(best_seq, time_budget=alloc)
            obj = compute_obj(a_cd2)
            print(f"  Final CD: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_cd2.copy()

        # One more spectral
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_f = len(best_seq)
            a_spec_f = spectral_init(n_f, n_iters=1000, temperature=0.2)
            # Blend with current best
            a_blend_f = 0.3 * a_spec_f + 0.7 * best_seq
            a_blend_f = np.maximum(a_blend_f, 1e-10)
            a_blend_f /= a_blend_f.sum()
            obj_f = compute_obj(a_blend_f)
            print(f"  Final spectral blend: obj={obj_f:.8f}")
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_blend_f.copy()

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
