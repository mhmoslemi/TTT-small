        A = rfft(a, N)

        target_mag = np.sqrt(np.mean(np.abs(A)**2) * len(A))  # or some target
        # Keep phase, set magnitude
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:

        a_new = a_new / s
        a = a_new
    return a

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral shaping (Gerchberg-Saxton), adaptive softmax gradient descent,
LP polishing, and multi-scale exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def softmax_gradient(a, tau, N=None):
    """
    Compute gradient of softmax objective: tau * log(sum exp(b[k]/tau))
    where b = conv(a, a).
    Returns gradient array of length n.
    """
    n = len(a)
    if N is None:
        N = 1
        while N < 3 * n - 1:
            N <<= 1
    
    # Compute b = conv(a, a)
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]
    
    # Compute weights w[k] = exp(b[k]/tau) / sum(exp(b[k']/tau))
    b_shifted = b / tau - np.max(b) / tau  # numerical stability
    w = np.exp(b_shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    
    # Gradient: grad[i] = 2 * sum_m a[m] * w[m+i]
    # This is 2 * cross_correlation(a, w) at lag i
    # = 2 * conv(a_flipped, w)[n-1+i] for i=0,...,n-1
    a_flip = a[::-1]
    faf = rfft(a_flip, N)
    fw = rfft(w, N)
    conv_aw = irfft(faf * fw, N)
    grad = 2.0 * conv_aw[n - 1:2 * n - 1]
    
    return grad, b, w


def spectral_shaping(a, n_iter=200, gamma=0.5, time_budget=30.0):
    """
    Gerchberg-Saxton inspired spectral shaping.
    Iteratively modifies the spectrum to be flatter.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        
        A = rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: make magnitude flatter
        # Use gamma-blending: mag_new = mag^(1-gamma) * target_mag^gamma
        target_mag = np.full_like(mag, np.mean(mag[mag > 0])) if np.any(mag > 0) else np.ones_like(mag)
        
        # Actually, a better target: we want |A|^2 to be flatter
        # So we want |A| to be closer to constant
        # Use: mag_new = mag * (mean_mag / mag)^gamma = mag^(1-gamma) * mean_mag^gamma
        mean_mag = np.mean(mag)
        if mean_mag < 1e-10:
            break
        mag_new = mag * (mean_mag / np.maximum(mag, 1e-10)) ** gamma
        # Or equivalently: mag_new = mag ** (1-gamma) * meanmag ** gamma
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def adaptive_gd(a, time_budget=60.0, verbose=False):
    """
    Adaptive softmax gradient descent with momentum.
    Temperature decreases over time.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / s
    
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Initial temperature: spread over the range of b
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]
    b_range = np.max(b) - np.min(b)
    tau = max(b_range * 0.3, 1e-6)
    
    eta = 0.5  # learning rate
    momentum = 0.8
    v = np.zeros(n)
    
    stall = 0
    max_stall = 50
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient
        grad, b, w = softmax_gradient(a, tau, N)
        
        # Center gradient (remove mean to maintain sum)
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v *= 0.5
            continue
        a_new = a_new / s_new
        
        # Evaluate
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase learning rate and decrease temperature
            eta = min(eta * 1.01, 3.0)
            tau *= 0.995
        else:
            stall += 1
            eta *= 0.95
            v *= 0.7
            if stall > max_stall:
                # Reset with perturbation
                rng_local = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng_local.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.8
                stall = 0
                max_stall = min(max_stall * 1.5, 200)
                # Reset temperature
                fa = rfft(a, N)
                b = irfft(fa * fa, N)[:2 * n - 1]
                b_range = np.max(b) - np.min(b)
                tau = max(b_range * 0.3, 1e-6)
        
        # Decay temperature slowly
        tau = max(tau * 0.9999, 1e-8)
        
        if verbose and it % 500 == 0:
            print(f"            GD it{it}: obj={best_obj:.8f}, tau={tau:.2e}, eta={eta:.4f}")
    
    return best_a


def subgradient_peak_chase(a, time_budget=30.0, K=3):
    """
    Subgradient method chasing the top-K peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / s
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    eta = 0.02
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        
        # Find top-K peaks
        top_k = np.argsort(b)[-K:]
        
        # Subgradient: average of gradients at top peaks
        # grad[i] = 2 * sum_{k in top_k} a[k-i] (with bounds checking)
        grad = np.zeros(n)
        for k in top_k:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        
        # Actually this is O(n*K) which is fine for moderate n
        # But let's use the FFT approach for efficiency
        # grad[i] = 2 * sum_{k in top_k} a[k-i] * 1[0<=k-i<n]
        # = 2 * (delta_topk * a)[i] where delta_topk is 1 at top_k positions
        # This is a cross-correlation
        
        # Actually for small K, direct computation is fine
        # Let me use the efficient version:
        # Create w that is 1/K at top_k positions
        w = np.zeros(2 * n - 1)
        for k in top_k:
            w[k] = 1.0 / K
        
        a_flip = a[::-1]
        N2 = 1
        while N2 < 3 * n - 1:
            N2 <<= 1
        faf = rfft(a_flip, N2)
        fw = rfft(w, N2)
        conv_aw = irfft(faf * fw, N2)
        grad = 2.0 * conv_aw[n - 1:2 * n - 1]
        
        # Center
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.2)
        else:
            eta *= 0.95
            v *= 0.5
            if eta < 1e-8:
                break
    
    return best_a


def solve_lp_for_direction(a):
    """
    Given fixed a (normalized), solve LP:
    min M s.t. conv(a, g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns g or None.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix: A @ g gives conv(a, g)
    # conv(a, g)[k] = sum_i a[i] * g[k-i]
    # So row k of A has A[k, j] = a[j-k] if 0<=j-k<n, else 0... 
    # Actually: conv(a,g)[k] = sum_i a[i]*g[k-i]
    # For fixed k, the coefficient of g[j] is a[k-j] if 0<=k-j<n, i.e., k-n+1<=j<=k
    # So A[k,j] = a[k-j] if max(0,k-n+1) <= j <= min(k, n-1)
    
    from scipy.optimize import linprog
    
    # C: minimize M (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub @ [g; M] <= b_ub
    # conv(a,g)[k] - M <= 0 for each k
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq @ [g; M] = b_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception as e:
        print(f"            LP error: {e}")
    
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / s
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = solve_lp_for_direction(a)
        if g_lp is None:
            break
        
        # Try different step sizes
        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            a = (1 - best_t) * best_a + best_t * g_lp
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 15:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diversified_init(n, rng):
    """Generate a diverse random initialization."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # U-shaped
        x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = 4 * (x - 0.5) ** 2 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n // 8) ** 2) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n // 8) ** 2)
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n)) + 0.3
        a = a / np.sum(a)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.7
        a = a / np.sum(a)
    elif mode == 6:
        # Random Fourier features (non-negative projection)
        freqs = rng.uniform(-1, 1, min(n // 4, 50))
        phases = rng.uniform(0, 2 * np.pi, min(n // 4, 50))
        a = np.ones(n) + 0.5 * np.cos(2 * np.pi * np.outer(np.arange(n), freqs) + phases).sum(axis=1)
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s if s > 0 else np.ones(n) / n
    else:
        # Smooth random (low-pass filtered noise)
        a = rng.normal(1.0, 0.5, n)
        # Simple moving average smoothing
        kernel_size = max(3, n // 50)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a
