def plateau_gd(a, time_budget=100.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Parameters
    eta = 0.5  # initial step size
    momentum = 0.85
    v = np.zeros(n)
    threshold = 0.9  # start with 90% of max
    stall = 0
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    N2 = 1
    while N2 < 3 * n:
        N2 <<= 1
    
    a_flip = a[::-1]  # will update each iteration
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        b_max = float(np.max(b))
        
        # Compute plateau gradient
        mask = (b >= threshold * b_max).astype(np.float64)
        n_plateau = max(int(np.sum(mask)), 1)
        
        a_flip = a[::-1]
        fm = rfft(mask, N2)
        faf = rfft(a_flip, N2)
        conv_ma = irfft(fm * faf, N2)
        grad = 2.0 * conv_ma[n-1:2*n-1] / n_plateau
        
        # Center gradient
        grad = grad - np.mean(grad)
        
        # Momentum
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 3.0)
            # Gradually lower threshold to include more peaks
            if it > 100:
                threshold = max(threshold * 0.999, 0.5)
        else:
            stall += 1
            eta *= 0.95
            v *= 0.5
            if stall > 50:
                # Perturb and reset
                rng = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                threshold = 0.9
                stall = 0
            if eta < 1e-10:
                break
    
    return best_a

def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    b = compute_conv(a)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for i in range(n):
            k = i + j  # conv(a,g)[i+j] includes a[i]*g[j]
            if k < 2 * n - 1:
                A[k, j] += a[i]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))

        return a

            break

        best_t = 0.5
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

                best_t = t

            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()

            stall += 1

"""
Adaptive soft-L∞ gradient descent with LP polishing, plateau tracking,
and multi-scale n exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: adaptive beta schedule for smooth-to-peak transition,
dual-plateau gradient (combining soft and hard thresholding),
frequency-domain perturbation for escape, and aggressive multi-n search.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    b = compute_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for i in range(n):
            k = i + j
            if k < 2 * n - 1:
                A[k, j] += a[i]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def soft_linf_gd(a, time_budget=100.0, beta_start=3.0, beta_end=30.0, verbose=False):
    """
    Gradient descent on soft L∞ norm (log-sum-exp approximation) of auto-convolution.
    Beta increases over time to transition from 'flatten all' to 'shave the peak'.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.3
    momentum = 0.88
    v = np.zeros(n)
    beta = beta_start
    stall = 0

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    N2 = 1
    while N2 < 4 * n:
        N2 <<= 1

    total_iters = 50000
    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break

        # Schedule beta: linearly increase from beta_start to beta_end
        progress = min(it / (total_iters * 0.7), 1.0)
        beta_cur = beta_start + (beta_end - beta_start) * progress

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        b_max = float(np.max(b))

        if b_max < 1e-15:
            break

        # Softmax weights with temperature 1/beta
        b_shifted = b - b_max
        w = np.exp(beta_cur * b_shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum

        # Gradient: 2 * sum_k w[k] * a[k-i]
        a_flip = a[::-1]
        fw = rfft(w, N2)
        faf = rfft(a_flip, N2)
        conv_wa = irfft(fw * faf, N2)
        grad = 2.0 * conv_wa[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.008, 1.5)
            momentum = min(momentum + 0.0003, 0.96)
        else:
            stall += 1
            eta *= 0.975
            v *= 0.5
            if stall > 60:
                rng = np.random.default_rng(it + 4242)
                # Frequency-domain perturbation: add random phase noise
                fa_a = rfft(a, N)
                magnitude = np.abs(fa_a)
                phase = np.angle(fa_a) + rng.normal(0, 0.1, len(fa_a))
                fa_new2 = magnitude * np.exp(1j * phase)
                a_pert = irfft(fa_new2, N)[:n]
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a = a_pert / s_pert
                else:
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                v = np.zeros(n)
                eta = 0.3
                momentum = 0.88
                stall = 0
            if eta < 1e-12:
                break

    return best_a


def plateau_gd(a, time_budget=100.0, verbose=False):
    """
    Plateau-based gradient descent with adaptive threshold.
    Tracks all peaks above threshold*max and averages their gradients.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.4
    momentum = 0.85
    v = np.zeros(n)
    threshold = 0.95
    stall = 0

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    N2 = 1
    while N2 < 4 * n:
        N2 <<= 1

    for it in range(80000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        b_max = float(np.max(b))

        if b_max < 1e-15:
            break

        # Hard threshold plateau
        mask = (b >= threshold * b_max).astype(np.float64)
        n_plateau = max(int(np.sum(mask)), 1)

        a_flip = a[::-1]
        fm = rfft(mask, N2)
        faf = rfft(a_flip, N2)
        conv_ma = irfft(fm * faf, N2)
        grad = 2.0 * conv_ma[n - 1:2 * n - 1] / n_plateau
        grad = grad - np.mean(grad)

        # Momentum
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.005, 2.0)
            # Gradually lower threshold
            if it > 50:
                threshold = max(threshold - 0.0003, 0.5)
        else:
            stall += 1
            eta *= 0.97
            v *= 0.5
            if stall > 80:
                rng = np.random.default_rng(it + 8888)
                a = best_a * np.exp(rng.normal(0, 0.06, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.4
                threshold = 0.92
                stall = 0
            if eta < 1e-12:
                break

    return best_a


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization for polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 12:
                break

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.3
        a = a / np.sum(a)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.7
        a = a / np.sum(a)
    elif mode == 6:
        # Sine-squared with randomness
        x = np.arange(n) / n
        a = np.sin(np.pi * x) ** 2 + rng.uniform(0, 0.3, n)
        a = a / np.sum(a)
    else:
        # Multi-modal
        n_modes = rng.integers(2, 5)
        centers = rng.uniform(0, n, n_modes)
        widths = rng.uniform(n // 20, n // 5, n_modes)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    return np.maximum(a, 0)

    n_options = sorted(set([
        n_current + 100, n_current + 200, n_current + 400,
        n_current + 800,
        max(400, n_current - 200),
        1000, 1200, 1600, 2000, 2500
    ]))
    n_options = [n for n in n_options if 500 <= n <= 3000 and n != n_current]
    n_options = n_options[:8]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / np.sum(a_try)

        # Quick optimization on new n
        alloc = min(remaining * 0.12, 80.0)
        a_opt = soft_linf_gd(a_try, time_budget=alloc * 0.5, beta_start=5.0, beta_end=20.0)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score - 1e-6:
            # Further polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20 and n_try <= 1200:
                a_polish = lp_alternating(a_opt, time_budget=min(remaining2 * 0.3, 40.0))
                obj_polish = compute_obj(a_polish)
                if obj_polish < obj_opt:
                    a_opt = a_polish
                    obj_opt = obj_polish

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 7: Random restarts
    print(f"\n=== Phase 7: Random Restarts ===")
    n_use = len(best_seq)
    n_restarts = max(2, int((deadline - time.monotonic()) / 45.0))
    n_restarts = min(n_restarts, 10)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_r = random_init(n_use, rng)
        alloc = min(remaining * 0.15, 40.0)
        a_opt = soft_linf_gd(a_r, time_budget=alloc, beta_start=5.0, beta_end=25.0)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # Phase 8: Final intensive polish
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # LP polish
        if len(best_seq) <= 1200:
            alloc = min(remaining * 0.35, 100.0)
            a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  ** Final LP: obj={best_score:.8f} **")

        # Soft GD polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.4, 60.0)
            a_sg = soft_linf_gd(best_seq, time_budget=alloc, beta_start=15.0, beta_end=40.0)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  ** Final SoftGD: obj={best_score:.8f} **")

        # Dual plateau polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.4, 40.0)
            a_dp = dual_plateau_gd(best_seq, time_budget=alloc)
            obj_dp = compute_obj(a_dp)
            if obj_dp < best_score:
                best_score = obj_dp
                best_seq = a_dp.copy()
                print(f"  ** Final DualPlateau: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
