"""
Log-sum-exp smoothed optimization with exact coordinate descent for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses annealed log-sum-exp for smooth gradients,
exact 1D coordinate descent for max-convolution, and LP polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def lse_objective_and_grad(a, t):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    f_t(a) = t * log(sum_k exp(b_k / t)) where b = conv(a, a)
    Returns (f_t, grad) where grad = df_t/da.
    Assumes sum(a) = 1 and a >= 0.
    """
    n = len(a)
    b = compute_conv(a)
    
    # Numerical stability: subtract max
    b_max = np.max(b)
    exp_b = np.exp((b - b_max) / t)
    log_sum = b_max / t + np.log(np.sum(exp_b))
    f_t = t * log_sum
    
    # Softmax weights
    w = exp_b / np.sum(exp_b)
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (with a[m]=0 outside [0,n-1])
    # This is 2 * (conv(a_rev, w))[i + n - 1]
    a_rev = a[::-1]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fa_rev = rfft(a_rev, N)
    fw = rfft(w, N)
    conv_aw = irfft(fa_rev * fw, N)[:3 * n - 2]
    grad = 2.0 * conv_aw[n - 1:2 * n - 1]
    
    return f_t, grad


def lse_adam_optimize(a, time_budget=60.0, t_schedule=None, verbose=False):
    """
    Adam optimization on the log-sum-exp smoothed objective with annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    if t_schedule is None:
        t_schedule = np.geomspace(1.0, 0.02, 10)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0
    
    for t in t_schedule:
        # Run Adam for this temperature
        iters = max(10, int(time_budget / (len(t_schedule) * 2)))
        for it in range(iters):
            if time.monotonic() - start > time_budget:
                break
            
            f_t, grad = lse_objective_and_grad(a, t)
            
            # Adam update
            step_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**step_count)
            v_hat = v / (1 - beta2**step_count)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project: non-negative and sum = 1
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                continue
            a_new = a_new / s_new
            
            # Check improvement
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                a = a_new  # still use the new point for continuing
            
            if verbose and it % 50 == 0:
                print(f"    LSE-Adam t={t:.4f} it{it}: obj={best_obj:.8f}, cur={obj_new:.8f}")
        
        if time.monotonic() - start > time_budget:
            break
    
    return best_a

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build the matrix: (A @ g)[k] = conv(a, g)[k] = sum_j a[j]*g[k-j]
    # For fixed a, this is linear in g.
    # (A @ g)[k] = sum_j a[|k-j|] * g[j] ... no, let me think.
    # conv(a,g)[k] = sum_j a[j]*g[k-j] (with valid indices)
    # = sum_j g[j]*a[k-j] (by symmetry of the sum)
    # So A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, build A as a Hankel-like matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for valid k-j
        k_lo = j  # k-j >= 0 => k >= j
        k_hi = min(j + n, 2 * n - 1)  # k-j < n => k < j+n
        A[k_lo:k_hi, j] = a[:k_hi - k_lo]
    
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception as e:
        print(f"    LP error: {e}")
    
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a


def chase_peak(a, time_budget=30.0):
    """Subgradient chase on the peak of the convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.05

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        grad = grad - np.mean(grad)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 0.3)
        else:
            eta *= 0.95
            if eta < 1e-7:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 10) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.2
        a = a / np.sum(a)
    else:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.5
        a = a / np.sum(a)
    return a
