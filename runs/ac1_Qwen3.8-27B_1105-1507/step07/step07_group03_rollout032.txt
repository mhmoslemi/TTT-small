import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def peak_focused_mirror_descent(a, n_iters=5000, time_budget=60.0, beta_schedule='adaptive'):
    """Mirror descent with peak-focused subgradient and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = np.maximum(a, 1e-15) / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.15  # step size
    patience = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Adaptive beta for soft-max weighting
        if beta_schedule == 'adaptive':
            beta = max(5.0, 50.0 / (it + 10))
        else:
            beta = 50.0
        
        # Soft-max weights on peaks
        shifted = conv - max_conv
        weights = np.exp(beta * shifted)
        total_w = float(np.sum(weights))
        if total_w < 1e-30:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= total_w
        
        # Compute subgradient: grad_j = sum_k weights[k] * a[k-j]
        grad = np.zeros(n)
        for k in range(len(conv)):
            wk = weights[k]
            if wk < 1e-10:
                continue
            k = int(k)
            # conv[k] = sum_i a[i] * a[k-i]
            # d(conv[k])/d(a[j]) = a[k-j] (if 0 <= k-j < n)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                j = k - i
                if 0 <= j < n:
                    grad[j] += wk * a[i]
        
        # Mirror descent: a_j *= exp(-eta * grad_j)
        a_new = a * np.exp(-eta * grad)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            # Increase eta slightly if improving

                eta = min(eta * 1.05, 0.5)
        else:
            patience += 1
            a = best_a  # revert
            if patience % 20 == 0:
                eta *= 0.9
            if patience > 200:

                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                patience = 0
                eta = 0.15
    
    return best_a

def fast_lp_alternating(a, time_budget=60.0):
    """LP alternating with active set maintenance for speed."""
    from scipy.optimize import linprog
    
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    patience = 0
    last_t = 0.3
    
    # Build constraint matrix once (it depends on a, so rebuild each iteration)
    max_iters = 200
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        a_norm = a / np.sum(a)
        # Build A matrix: A[k, j] = a_norm[k-j] if 0 <= k-j < n else 0
        # conv(a_norm, g)[k] = sum_j a_norm[k-j] * g[j]
        m = 2 * n - 1
        A = np.zeros((m, n))
        for j in range(n):
            # A[k, j] = a_norm[k-j] for valid k-j
            valid_start = max(0, j)  # k-j >= 0 => k >= j
            valid_end = min(m, j + n)  # k-j < n => k < j+n
            for k in range(valid_start, valid_end):
                A[k, j] = a_norm[k - j]
        
        # LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        # Variables: [g_0, ..., g_{n-1}, M]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0 or res.x is None:
            break
        
        g_lp = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g_lp))
        if s_g < 1e-15:
            break
        g_lp = g_lp / s_g
        
        # Line search
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()
        
        # Try a range of t values
        t_candidates = np.concatenate([
            np.linspace(0.05, 0.5, 10),
            np.linspace(0.5, 1.0, 10)
        ])
        if 0 < last_t < 1:
            t_candidates = np.concatenate([t_candidates, 
                                           [last_t * 0.5, last_t * 0.75, last_t * 1.0, last_t * 1.25, last_t * 1.5]]
                                          )
            t_candidates = t_candidates[(t_candidates > 0) & (t_candidates <= 1.0)]
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            patience = 0
        else:
            patience += 1
            if patience > 10:
                rng = np.random.default_rng(it * 17 + n)
                a = best_a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-15:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                patience = 0
    
    return best_a

def freq_domain_shape(a, n_iters=300, time_budget=30.0):
    """Frequency-domain optimization: shape |FFT(a)|^2 to minimize peak of IFFT."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    # Current spectrum
    A_fft = fft(a, L)
    log_mag = np.log(np.maximum(np.abs(A_fft), 1e-15))
    phase = A_fft / np.maximum(np.abs(A_fft), 1e-15)
    
    eta = 0.1
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Current autocorrelation
        spectrum_power = np.abs(A_fft) ** 2
        autocorr = ifft(spectrum_power).real[:2 * n - 1]
        max_ac = float(np.max(autocorr))
        
        # We want to reduce the peak of autocorr
        # In frequency domain, the autocorrelation peak at lag k gets contribution
        # from all frequency bins. The "blame" for peak at lag k is related to
        # the frequency bins where the phase creates constructive interference at lag k.
        
        # Alternative: directly optimize the power spectrum
        # We want IFFT(|A|^2) to have minimum peak
        # Gradient of max(autocorr) w.r.t. |A|^2 is: delta at the peak lag -> FFT gives exp(2*pi*i*k*f)
        
        # Peak lag
        k_peak = int(np.argmax(autocorr))
        
        # The gradient of autocorr[k_peak] w.r.t. |A_f|^2 is:
        # d/d|A_f|^2 IFFT(|A|^2)[k] = real(e^{-2*pi*i*f*k/L}) / L * ... 
        # Actually, autocorr[k] = (1/L) sum_f |A_f|^2 * e^{2*pi*i*f*k/L}
        # So d(autocorr[k_peak])/d(|A_f|^2) = (1/L) * cos(2*pi*f*k_peak/L)
        
        freqs = np.arange(L)
        grad_power = (1.0 / L) * np.cos(2 * np.pi * freqs * k_peak / L)
        
        # Update power spectrum: reduce where gradient is positive
        new_log_mag = log_mag - eta * grad_power
        new_log_mag = np.maximum(new_log_mag, -20)  # prevent too small values
        
        # Reconstruct
        new_mag = np.exp(new_log_mag)
        new_A_fft = new_mag * phase
        a_new = ifft(new_A_fft).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            A_fft = fft(a_new, L)
            log_mag = np.log(np.maximum(np.abs(A_fft), 1e-15))
            phase = A_fft / np.maximum(np.abs(A_fft), 1e-15)
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.95
            # Revert
            A_fft = fft(best_a, L)
            log_mag = np.log(np.maximum(np.abs(A_fft), 1e-15))
            phase = A_fft / np.maximum(np.abs(A_fft), 1e-15)
    
    return best_a

def multi_peak_subgradient(a, n_iters=3000, time_budget=60.0):
    """Subgradient with multiple peaks and adaptive thresholding."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    threshold_factor = 0.9  # only consider peaks within this fraction of max
    patience = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        mean_conv = float(np.mean(conv))
        
        # Adaptive threshold
        threshold = max_conv - (max_conv - mean_conv) * (1 - threshold_factor)
        
        # Find peaks above threshold
        peak_indices = np.where(conv >= threshold)[0]
        
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        
        # Weights: proportional to how much above threshold
        above = conv[peak_indices] - threshold
        above = np.maximum(above, 0)
        total = float(np.sum(above))
        if total < 1e-15:
            weights = np.zeros(len(peak_indices))
            weights[0] = 1.0
        else:
            weights = above / total
        
        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                j = k - i
                if 0 <= j < n:
                    grad[j] += w * a[i]
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        alpha = 0.5 / (grad_norm * np.sqrt(it + 10))
        alpha = min(alpha, 0.3)
        
        # Multiplicative update (mirror descent)
        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            alpha *= 0.5
            continue
        a_new = a_new / s_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 50:
                # Try a different threshold
                threshold_factor = min(threshold_factor + 0.1, 0.99)
                if threshold_factor >= 0.99:
                    threshold_factor = 0.5 + np.random.random() * 0.4
                patience = 0
                a = best_a.copy()
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n * n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def randomized_rounding(a, n_strips=10, rng=None):
    """Randomized rounding: split into strips, randomly permute/modify.”""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = a.copy()
    
    # Randomly permute chunks
    chunk_size = max(1, n // n_strips)
    for _ in range(3):
        # Pick a random chunk and shift it
        start = rng.integers(0, n - chunk_size)
        end = start + chunk_size
        shift = rng.integers(1, chunk_size)
        a[start:end] = np.roll(a[start:end], shift)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    
    return a

"""
Optimized search for step function coefficients minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses cycling between LP and mirror descent, FFT-based subgradients, peak equalization,
and multi-n exploration with structured restarts.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_with_b(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def peak_focused_md(a, n_iters=5000, time_budget=60.0):
    """Mirror descent with peak-focused subgradient using FFT for speed."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = np.maximum(a, 1e-15) / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.12
    patience = 0
    L = 1
    while L < 3 * n:
        L <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Adaptive beta for soft-max weighting
        beta = max(10.0, 80.0 / (it + 10))

        # Soft-max weights on peaks
        shifted = conv - max_conv
        weights = np.exp(beta * shifted)
        total_w = float(np.sum(weights))
        if total_w < 1e-30:
            weights = np.zeros_like(conv)
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= total_w

        # Compute subgradient via FFT cross-correlation
        # grad[j] = sum_k weights[k] * a[k-j] (for valid k-j)
        w_padded = np.zeros(L)
        w_padded[:len(weights)] = weights
        a_padded = np.zeros(L)
        a_padded[:n] = a

        W = rfft(w_padded, L)
        A = rfft(a_padded, L)

        # Cross-correlation: grad[j] = sum_k w[k] * a[k-j]
        # = ifft(W * conj(A)) but we need the right form
        # Actually: ifft(fft(w) * conj(fft(a))) gives circular cross-correlation
        # cross[j] = sum_k w[k] * a[(k-j) mod L]
        # For zero-padded vectors with L >= len(w) + len(a) - 1, first n elements are linear cross-correlation
        cross = ifft(W * np.conj(A)).real[:n]
        grad = 2.0 * cross

        # Mirror descent step
        a_new = a * np.exp(-eta * grad)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:

            best_a = a_new.copy()
            a = a_new

                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                patience = 0
                eta = 0.12

    return best_a


def fast_lp_step(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    from scipy.optimize import linprog

    n = len(a)
    a_norm = a / np.sum(a)

    # Build constraint matrix A where A[k,j] = a_norm[k-j] if 0<=k-j<n
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a_norm[:k_end - k_start]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status != 0 or res.x is None:
        return None, None

    g = np.maximum(res.x[:n], 0)
    s_g = float(np.sum(g))
    if s_g < 1e-15:
        return None, None
    g = g / s_g
    return g, float(res.x[n])

def freq_domain_shape(a, n_iters=500, time_budget=30.0):
    """Frequency-domain optimization to flatten autocorrelation peak."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    A_fft = fft(a, L)
    log_mag = np.copy(np.log(np.maximum(np.abs(A_fft), 1e-15)))
    phase = A_fft / np.maximum(np.abs(A_fft), 1e-15)

    eta = 0.08
    patience = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Current autocorrelation
        power = np.exp(2.0 * log_mag)
        autocorr = ifft(power).real[:2 * n - 1]
        k_peak = int(np.argmax(autocorr))

        # Multi-peak: use top few peaks
        top_peaks = np.argsort(autocorr)[-3:]
        top_peaks = top_peaks[np.argsort(autocorr[top_peaks])[::-1]]

        # Gradient in frequency domain
        freqs = np.arange(L)
        grad_power = np.zeros(L)
        for kp in top_peaks:
            grad_power += np.cos(2.0 * np.pi * freqs * kp / L) / L

        new_log_mag = log_mag - eta * grad_power
        new_log_mag = np.maximum(new_log_mag, -15.0)

        new_mag = np.exp(new_log_mag)
        new_A_fft = (new_mag / np.maximum(new_mag, 1e-15)) * (np.exp(new_log_mag) / np.maximum(np.exp(new_log_mag), 1e-15))
        new_A_fft = np.exp(new_log_mag + 0j) * phase
        a_new = ifft(new_A_fft).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            A_fft = fft(a_new, L)
            log_mag = np.log(np.maximum(np.abs(A_fft), 1e-15))
            phase = A_fft / np.maximum(np.abs(A_fft), 1e-15)
            patience = 0
        else:
            patience += 1
            log_mag = np.log(np.maximum(np.abs(fft(best_a, L)), 1e-15))
            phase = fft(best_a, L) / np.maximum(np.abs(fft(best_a, L)), 1e-15)
            if patience % 20 == 0:
                eta *= 0.9

    return best_a


def peak_equalization(a, n_iters=200, time_budget=30.0):
    """Directly reduce the top peaks by moving mass from contributing positions."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]
        mean_val = float(np.mean(conv))

        # Find positions that contribute most to peak at k_star
        # Contribution of position i to peak k_star: a[i] * a[k_star - i]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]

        # Find positions with low contribution to ALL top peaks
        top_peaks = np.argsort(conv)[-5:]
        low_contrib = np.ones(n)
        for kp in top_peaks:
            contrib_kp = np.zeros(n)
            for i in range(n):
                j = kp - i
                if 0 <= j < n:
                    contrib_kp[i] = a[i] * a[j]
            low_contrib *= 1.0 + contrib_kp / max(mean_val, 1e-15)

        # Move mass from high-contribution to low-contribution positions
        top_contrib_idx = np.argsort(contributions)[-10:]
        low_contrib_idx = np.argsort(low_contrib)[:10]

        improved = False
        for si in top_contrib_idx:
            for di in low_contrib_idx:
                if si == di or a[si] < 1e-6:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[si]
                    a[si] -= delta
                    a[di] += delta
                    if a[si] < 0:
                        a[si] = 0
                        a[di] -= delta
                        continue
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[si] += delta
                        a[di] -= delta
                        break

        if not improved and it > 50:
            break

        a = best_a.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = float(np.sum(a_new))
    if s > 1e-15:
        a_new = a_new / s
    return np.maximum(a_new, 0)

            if obj < best_score - 1e-13:
                best_score = obj
                best_seq = a_pe.copy()
                n_current = len(best_seq)
                print(f"  Cycle {cycle_count} PE: obj={obj:.8f}")

        if cycle_count % 5 == 0:
            print(f"  ... cycle {cycle_count}, best={best_score:.8f}, remaining={phase2_end - time.monotonic():.0f}s")

    # ============ Phase 3: Multi-n exploration (20% time) ============
    t3 = (deadline - start_time) * 0.20
    phase3_end = start_time + t3

    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(100, n_best - 200), max(100, n_best - 100),
        max(100, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        400, 500, 600, 700, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2500 and abs(n - n_best) > 30]

    for n_try in n_options:
        remaining = phase3_end - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick MD optimization
        alloc = min(remaining * 0.3, 25.0)
        a_opt = peak_focused_md(a_try, n_iters=2000, time_budget=alloc)
        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(a_opt)

    # ============ Phase 4: Random restarts (5% time) ============
    phase4_end = start_time + (deadline - start_time) * 0.05
    print(f"\n=== Phase 4: Random Restarts ===")

    for r in range(5):
        remaining = phase4_end - time.monotonic()
        if remaining < 20:
            break

        n_best = len(best_seq)
        init_type = r % 4
        if init_type == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.04, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 1:
            n_r = max(200, min(1500, n_best + int(rng.integers(-100, 100))))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.06, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            x = np.linspace(0, 2 * np.pi, n_best)
            a_r = 1.0 + 0.4 * np.cos(2 * x + rng.uniform(0, 2 * np.pi)) + \
                       0.2 * np.cos(7 * x + rng.uniform(0, 2 * np.pi))
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)
        else:
            # Structured perturbation
            a_r = best_seq.copy()
            mid = n_best // 2
            a_r[:mid] *= rng.uniform(0.8, 1.3)
            a_r[mid:] *= rng.uniform(0.7, 1.2)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        alloc = min(15.0, remaining * 0.4)
        a_opt = peak_focused_md(a_r, n_iters=1500, time_budget=alloc)
        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # ============ Phase 5: Final polish (5% time) ============
    print(f"\n=== Phase 5: Final Polish ===")
    n_final = len(best_seq)
    print(f"  Polishing n={n_final}, best={best_score:.8f}")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.35, 50.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, max_iters=30)
        obj = compute_obj(a_lp)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            n_final = len(best_seq)

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 40.0)
        a_md = peak_focused_md(best_seq, n_iters=5000, time_budget=alloc)
        obj = compute_obj(a_md)
        print(f"  Final MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()
            n_final = len(best_seq)

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 20.0)
        a_pe = peak_equalization(best_seq, n_iters=200, time_budget=alloc)
        obj = compute_obj(a_pe)
        print(f"  Final PE: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_pe.copy()

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 20.0)
        a_fd = freq_domain_shape(best_seq, n_iters=300, time_budget=alloc)
        obj = compute_obj(a_fd)
        print(f"  Final Freq: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fd.copy()

    # One more LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 30.0)
        a_last = lp_alternating(best_seq, time_budget=alloc, max_iters=20)
        obj = compute_obj(a_last)
        print(f"  Last LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_last.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
