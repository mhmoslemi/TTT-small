"""
Optimized LP-alternating method with peak-chasing supplement for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(A_row, a):
    """
    Build the convolution matrix for LP.
    A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    Returns the matrix as a dense array for small n, or use structured.
    """
    n = len(a)
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # A[k, i] = a[k-i] for k in [i, i+n-1]
        A[i:i+n, i] = a
    return A

def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a, g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(None, a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M → c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality: A @ g - M <= 0 for each row → [A, -1] @ [g, M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

from scipy.optimize import linprog

    if n == 0:
        return np.array([0.0])

    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):

    for i in range(n):

    """
    LP alternating optimization with adaptive step size and perturbation restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    last_t = 0.3

        best_t = None
        best_a_local = None

    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
        return max_c, weights
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    return smooth_val, weights


def compute_gradient_smooth(weights, a):
    """Compute gradient of smooth max w.r.t. a using correlate."""
    grad = np.correlate(weights, a, mode='valid')
    return grad


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_gradient_descent(a, time_budget=40.0, T_init=0.0005, T_min=0.00002):
    """
    Gradient descent using smooth max (log-sum-exp) with Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.003
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        smooth_val, weights = smooth_max_and_weights(conv, T)
        grad = compute_gradient_smooth(weights, a)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0 and lr < 0.015:
                lr = min(lr * 1.04, 0.015)
        else:
            stall += 1
            if stall > 50:
                lr *= 0.6
                if lr < 1e-6:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 100:
            T = max(T_min, T_init * (0.995 ** it))

    return best_a


def peak_chase(a, time_budget=30.0):
    """
    Chase-the-peak with weighted subgradient and momentum.
    Targets top-K convolution bins.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.01
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    K = max(3, n // 20)  # Number of top bins to consider

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)

        # Find top K bins
        top_idx = np.argpartition(conv, -K)[-K:]
        # Weight by proximity to max
        weights_k = conv[top_idx] / max_conv
        weights_k = weights_k ** 10  # Emphasize the top ones
        weights_k = weights_k / np.sum(weights_k)

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights_k, top_idx):
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += w * 2.0 * a[idx]

        grad_centered = grad - np.mean(grad)
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped (bimodal at edges)
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal in center
        center = n // 2
        width = max(1, n // 4)
        x = np.arange(n)
        a = np.exp(-((x - center * 0.7) ** 2) / (2 * width ** 2))
        a += np.exp(-((x - center * 1.3) ** 2) / (2 * width ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        a = a / np.sum(a)
    elif mode == 5:
        # Triangle
        x = np.arange(n) / max(1, n - 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    elif mode == 6:
        # Random sparse-ish
        a = rng.uniform(0, 1, n)
        a[a < 0.3] *= 0.1
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: LP alternating + smooth max + peak chase + multi-n + restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Intensive LP Alternating (60% of budget) ===
    print(f"\n=== Phase 1: LP Alternating ===")
    remaining = deadline - time.monotonic()
    phase1_budget = remaining * 0.55

    a = best_seq.copy()
    n_rounds = 4
    for round_idx in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = phase1_budget / (n_rounds - round_idx)
        alloc = min(alloc, remaining * 0.7)
        print(f"  LP Round {round_idx+1}/{n_rounds} (budget={alloc:.0f}s)")
        a_new = lp_alternating(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"    -> obj={best_score:.8f}")
        else:
            print(f"    -> obj={obj:.8f} (no improvement, keeping best)")

    # === Phase 2: Smooth Max Gradient Descent ===
    print(f"\n=== Phase 2: Smooth Max GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.12, 80.0)
        a_sm = smooth_max_gradient_descent(best_seq.copy(), time_budget=alloc, T_init=0.0004, T_min=0.00001)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            a = a_sm.copy()
            print(f"  Smooth Max: obj={best_score:.8f} **")
        else:
            print(f"  Smooth Max: obj={obj_sm:.8f}")

    # === Phase 3: Peak Chase ===
    print(f"\n=== Phase 3: Peak Chase ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 50.0)
        a_pc = peak_chase(best_seq.copy(), time_budget=alloc)
        obj_pc = compute_obj(a_pc)
        if obj_pc < best_score:
            best_score = obj_pc
            best_seq = a_pc.copy()
            a = a_pc.copy()
            print(f"  Peak Chase: obj={best_score:.8f} **")
        else:
            print(f"  Peak Chase: obj={obj_pc:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50),
        n_best,
        min(2000, n_best + 50), min(2000, n_best + 100), min(2000, n_best + 200),
        600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc_lp = min(remaining * 0.1, 25.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.005:
            alloc_sm = min(remaining * 0.05, 15.0)
            a_opt2 = smooth_max_gradient_descent(a_opt, time_budget=alloc_sm, T_init=0.0003, T_min=0.00001)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Random Restarts ===
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        init_mode = r % 7
        n_r = n_best + int(rng.integers(-50, 51))
        n_r = max(400, min(1500, n_r))
        a_r = random_init(n_r, rng, mode=init_mode)

        alloc_r = min(remaining * 0.1, 30.0)
        a_opt = lp_alternating(a_r, time_budget=alloc_r * 0.6)

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = smooth_max_gradient_descent(a_opt, time_budget=min(alloc_r * 0.3, remaining * 0.2),
                                                T_init=0.0003, T_min=0.00001)

        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = n_r
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        for polish in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.25, 30.0)
            a_lp = lp_alternating(a_final, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_final = a_lp.copy()
                print(f"  Polish LP {polish}: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sm = smooth_max_gradient_descent(a_final, time_budget=min(remaining * 0.2, 15.0),
                                               T_init=0.0001, T_min=0.000005)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Final Smooth: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 8:
            a_pc = peak_chase(a_final, time_budget=min(remaining * 0.15, 10.0))
            obj_pc = compute_obj(a_pc)
            if obj_pc < best_score:
                best_score = obj_pc
                best_seq = a_pc.copy()
                print(f"  Final Peak: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
