"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using spectral initialization,
alternating LP optimization, subgradient polishing, and multi-n exploration.
Key innovation: convolution-square-root spectral init + diversified LP iterations.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def spectral_init(n):
    """Initialize via convolution square-root of a flat target."""
    N = 1
    while N < 2 * n:
        N <<= 1
    # Target: flat convolution
    c = np.ones(2 * n - 1) / (2 * n - 1)
    c_padded = np.zeros(N)
    c_padded[:2 * n - 1] = c
    C = np.fft.fft(c_padded)
    # Take square root in frequency domain
    B = np.sqrt(C + 0j)
    # Ensure Hermitian symmetry for real output
    for k in range(1, N // 2):
        B[k] = (B[k] + np.conj(B[N - k])) / 2
    if N % 2 == 0:
        B[N // 2] = np.real(B[N // 2])
    b_padded = np.real(np.fft.ifft(B))
    b = np.maximum(b_padded[:n], 0)
    s = float(np.sum(b))
    if s > 1e-10:
        b = b / s
    else:
        b = np.ones(n) / n
    return b


def cosine_init(n, eta=1.0):
    """Edge-enhanced cosine initialization: high at edges, low at center."""
    if n <= 2:
        return np.ones(n) / n
    x = np.arange(n)
    # eta > 0: edge-enhanced; eta < 0: center-enhanced
    b = 1.0 + eta * np.cos(2.0 * np.pi * x / (n - 1))
    b = np.maximum(b, 0.01)
    b = b / np.sum(b)
    return b


def uniform_init(n):
    return np.ones(n) / n


def random_init(n, rng):
    mode = int(rng.integers(0, 4))
    if mode == 0:
        b = rng.uniform(0.1, 1.0, n)
    elif mode == 1:
        b = 1.0 + rng.uniform(-0.5, 0.5, n)
    elif mode == 2:
        # Smooth random
        b = rng.uniform(0, 1, n)
        k = max(3, n // 40)
        kernel = np.exp(-0.5 * (np.arange(-k, k + 1)) ** 2 / (k / 2.0) ** 2)
        b = np.convolve(b, kernel / kernel.sum(), mode='same')
    else:
        # Two-peak
        c1 = n * rng.uniform(0.1, 0.35)
        c2 = n * rng.uniform(0.65, 0.9)
        w = n * rng.uniform(0.1, 0.25)
        x = np.arange(n)
        b = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    b = np.maximum(b, 0)
    s = float(np.sum(b))
    if s < 1e-10:
        return np.ones(n) / n
    return b / s


def build_conv_matrix(b):
    """Build dense matrix A (2n-1 x n) such that A @ g = conv(b, g)."""
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = b[:end - j]
    return A


def solve_lp(b):
    """Solve LP: min t s.t. conv(b,g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(b)
    if n <= 1:
        return np.array([1.0])
    A = build_conv_matrix(b)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def alternating_optimize(b, time_budget, verbose=False, max_iters=500):
    """Alternating LP with adaptive step and restarts."""
    n = len(b)
    b = np.maximum(b, 0).astype(np.float64)
    s = float(np.sum(b))
    if s < 1e-10:
        return b
    b = b / s
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345 + n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        g = solve_lp(b)
        if g is None:
            break
        # Try multiple step sizes
        best_t = 1.0
        best_t_obj = np.inf
        for t in [0.4, 0.6, 0.8, 0.95, 1.0]:

            b_try = np.maximum(b_try, 0)

            obj_try = compute_obj(b_try)
            if obj_try < best_t_obj:

                best_try = b_try.copy()
        if best_t_obj < best_obj - 1e-14:

            best_b = best_try.copy()
            b = best_try
            stall = 0

                if s > 1e-10:

                stall = 0
    return best_b


def subgradient_polish(b, time_budget, verbose=False):
    """Subgradient method on the max-conv objective."""
    n = len(b)
    b = np.maximum(b, 0).astype(np.float64)
    s = float(np.sum(b))
    if s < 1e-10:
        return b
    b = b / s
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    lr = 0.02
    rng = np.random.default_rng(999 + n)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(b)
        k_star = int(np.argmax(conv))
        # Subgradient: d/d b_i of (b*b)[k_star] = b[k_star - i] (valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = b[j]
        # Center to preserve sum constraint (projected gradient)
        grad_centered = grad - np.mean(grad)
        b_new = b - lr * grad_centered
        b_new = np.maximum(b_new, 0)
        s_new = float(np.sum(b_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        b_new = b_new / s_new
        obj_new = compute_obj(b_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_b = b_new.copy()
            b = b_new
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.85
            if lr < 1e-5:
                b = best_b * np.exp(rng.normal(0, 0.03, n))
                b = np.maximum(b, 0)
                s = float(np.sum(b))
                if s > 1e-10:
                    b = b / s
                    best_obj = compute_obj(b)
                    best_b = b.copy()
                lr = 0.02
    return best_b


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                best_seq = b_opt.copy()
                print(f"  ** n={n_nb}: score={best_score:.8f} **")
            else:
                print(f"  n={n_nb}: score={ev:.8f}")

    # Phase 4: Random restarts near best
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: Random restarts ===")
        n_best = len(best_seq)
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            # Perturb best
            scale = 0.05 + 0.03 * r
            b_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            b_r = np.maximum(b_r, 0)
            sr = float(np.sum(b_r))
            if sr < 1e-10:
                continue
            b_r = b_r / sr
            alloc = min(20.0, remaining * 0.1)
            b_opt = alternating_optimize(b_r, time_budget=alloc, max_iters=100)
            ev = evaluate_sequence([float(x) for x in b_opt])
            if ev < best_score:
                best_score = ev
                best_seq = b_opt.copy()
                print(f"  Restart {r}: score={best_score:.8f} **")
            else:
                print(f"  Restart {r}: score={ev:.8f}")

    # Phase 5: Final polish with all methods
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Final polish (remaining={remaining:.0f}s) ===")
        b_final = best_seq / np.sum(best_seq)

        # LP
        alloc = min(remaining * 0.4, 120.0)
        b_f1 = alternating_optimize(b_final, time_budget=alloc, max_iters=300)
        ev = evaluate_sequence([float(x) for x in b_f1])
        if ev < best_score:
            best_score = ev
            best_seq = b_f1.copy()
            b_final = best_seq / np.sum(best_seq)
            print(f"  Final LP: score={best_score:.8f} **")
        else:
            print(f"  Final LP: score={ev:.8f}")

        # Subgradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 60.0)
            b_f2 = subgradient_polish(b_final, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in b_f2])
            if ev < best_score:
                best_score = ev
                best_seq = b_f2.copy()
                print(f"  Final subgrad: score={best_score:.8f} **")
            else:
                print(f"  Final subgrad: score={ev:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
