**LP Alternating Algorithm:**
1. Normalize `a` so `sum(a) = 1`
2. Solve LP to get `g*` minimizing `max_k conv(a, g*)[k]`
3. Update `a = (1-t)*a + t*g*` for some step size `t`
4. Normalize `a`
5. Repeat

The key question is: what step size `t` to use? The previous code tries multiple values. A better approach might be to use a backtracking line search: start with `t=1` and reduce if the objective increases.

Actually, since we're doing `a = (1-t)*a + t*g*`, the objective as a function of `t` is:
`F((1-t)*a + t*g*) = 2n * max_k conv((1-t)*a + t*g*, (1-t)*a + t*g*) / ((1-t)*sum(a) + t*sum(g*))^2`

Since `sum(a) = sum(g*) = 1`, the denominator is 1. The numerator involves `conv` of a convex combination, which is a quadratic in `t`. The max of quadratics is convex in `t`, so we can do a 1D line search.

But it's simpler to just try a few values of `t` and pick the best.

LP-Alternating + Smoothed Gradient Optimization for Minimum Peak Autocorrelation.
Uses HiGHS LP solver for fast inner iterations, with diverse initializations,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for i in range(n):
        k_start = i
        k_end = min(2*n-1, i+n)
        A[k_start:k_end, i] = a[:k_end-k_start]
    return A

def solve_lp(a):
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def lp_alternate(a, time_budget, max_iter=500, verbose=False):
    n = len(a)

    best_obj = compute_obj(a)

            best_obj = best_t_obj
            best_a = best_a_local.copy()

def smooth_gradient_descend(a, tau, time_budget, max_iter=200):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        conv_max = np.max(conv)
        
        # Softmax weights over convolution
        w = tau * np.exp((conv - conv_max) * tau)
        w = w / np.sum(w)
        
        # Compute c[j] = sum_k w[k] * a[k-j] (convolution of w with a, reversed)
        # c[j] = sum_k w[k] * a[k-j] where valid
        # This is like cross-correlation
        c = np.zeros(n)
        for j in range(n):
            # c[j] = sum_k w[k] * a[k-j]
            # k-j must be in [0, n-1], k in [0, 2n-2]
            # So k in [j, j+n-1]
            k_min = j
            k_max = min(2*n-1, j+n)
            if k_min < k_max:
                c[j] = np.dot(w[k_min:k_max], a[:k_max-k_min])
        
        mean_c = np.dot(c, a)
        grad = 2.0 * n * (c - mean_c)  # subgradient of 2n*max(conv(a,a))
        
        # Project gradient
        grad_proj = grad - np.mean(grad)
        
        a_new = a - step_size * grad_proj
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                step_size = min(step_size * 1.1, 0.5)
            else:
                step_size *= 0.85
        else:
            step_size *= 0.7
    
    return best_a, best_obj

def subgradient_polish(a, time_budget, max_iter=500):
    """Subgradient method with step size decay."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.1
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        conv_max = np.max(conv)
        k_star = int(np.argmax(conv))
        
        # Subgradient at k_star: grad[j] = a[k_star - j] + a[k_star - j] (if k_star != 2j)
        # or 2*a[j] if 2j = k_star
        grad = np.zeros(n)
        for j in range(n):
            k_minus_j = k_star - j
            if 0 <= k_minus_j < n:
                if k_minus_j == j:
                    grad[j] = 2.0 * a[j]
                else:
                    grad[j] = a[k_minus_j]
        # Wait, this is the gradient of conv(a,a)[k_star] w.r.t. a
        # conv(a,a)[k] = sum_j a[j]*a[k-j]
        # d/da[m] conv(a,a)[k] = a[k-m] + a[m] if k-m != m, or 2*a[m] if k-m == m
        # But only if k-m is in [0, n-1]
        grad = np.zeros(n)
        for m in range(n):
            km = k_star - m
            if 0 <= km < n:
                if km == m:
                    grad[m] += 2.0 * a[m]
                else:
                    grad[m] += a[km]
        
        # Scale: objective is 2n * conv(a,a)[k_star] (at the max)
        # So subgradient of objective is 2n * grad
        # But we need to project onto the simplex
        
        grad_proj = grad - np.mean(grad)
        step_decay = 1.0 / (1.0 + it * 0.01)
        a_new = a - step_size * step_decay * grad_proj
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj
