descent polishing. Key difference from previous approach: uses proper second-order
information via L-BFGS and a reparameterization that handles non-negativity automatically.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_convolve_xy(x, y, N=None):
    """Compute full convolution of x and y via FFT."""
    if N is None:
        N = 1
        while N < len(x) + len(y) - 1:
            N <<= 1
    return irfft(rfft(x, N) * rfft(y, N), N)[:len(x) + len(y) - 1]


def safe_softmax(x, temp=1.0):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted / temp)
    return e / np.sum(e)


def objective_and_grad(x, beta, n):
    """
    Compute objective and gradient for L-BFGS in x-space.
    x is the log-parameter, a = softmax(x).
    Objective: (2n/beta) * log(sum(exp(beta * conv(a,a))))
    """
    a = safe_softmax(x)
    b = compute_conv_fft(a)
    
    max_b = np.max(b)
    shifted = beta * (b - max_b)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(np.sum(exp_vals)) / beta
    obj = 2.0 * n * log_sum
    
    s = exp_vals / np.sum(exp_vals)  # softmax(beta * b)
    
    # conv_s_a[j] = sum_l s[l] * a[l-j] = conv(s, a_rev, 'full')[n-1+j]
    a_rev = a[::-1]
    conv_s_a = compute_convolve_xy(s, a_rev, N=None)
    # conv_s_a has length (2n-1) + n - 1 = 3n-2
    # We need indices n-1 to 2n-2 (inclusive), which is n elements
    conv_s_a = conv_s_a[n-1:2*n-1]
    
    dot_s_b = np.dot(s, b)
    
    grad = 4.0 * n * a * (conv_s_a - dot_s_b)
    
    return obj, grad


def lbfgs_optimize(a_init, beta, n, maxiter=5000, tol=1e-12, time_budget=60.0):

        obj, grad = objective_and_grad(x, beta, n)

                          method='L-BFGS-B',
                          options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12,
                                   'maxfun': 200000, 'maxls': 50})
        if res.fun < best_f:
            best_f = res.fun
            best_x = res.x
    except RuntimeError:
        pass
    except Exception:
        pass
    
    a_out = softmax_fn(best_x)
    b_out = conv_fft(a_out)
    return a_out, 2.0 * n * float(np.max(b_out))


def beta_anneal(a_init, n, budget_sec):
    """Anneal beta from low to high for progressively tighter approximation."""
    start = time.monotonic()
    cur = a_init / np.sum(a_init)
    betas = [20, 100, 500, 2000, 10000, 50000, 200000]
    
    for i, beta in enumerate(betas):
        rem = budget_sec - (time.monotonic() - start)
        if rem < 3:
            break
        # Allocate more time to later (higher beta) stages
        alloc = max(2.0, rem * 0.7)
        cur, obj = run_single_lbfgs(cur, beta, n, alloc)
        print(f"      beta={beta:>7d}: obj={obj:.8f} (evals done)")
    
    b = conv_fft(cur)
    return cur, 2.0 * n * float(np.max(b))


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 1e-10)


def make_random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 7)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 3:
        a = np.exp(-((np.arange(n) - n // 2) ** 2) / (2 * max(n // 6, 1) ** 2))
    elif mode == 4:
        a = 1.0 / (np.arange(1, n + 1) ** 0.3)
    elif mode == 5:
        # Binary-like (0.1 or 1.0)
        a = np.where(rng.random(n) > 0.3, 1.0, 0.1)
    else:
        # Super-Gaussian
        a = np.abs(np.sin(np.pi * np.arange(n) / n)) ** 3 + 0.1
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def fine_tune_high_beta(a_init, n, deadline):
    """Run L-BFGS with very high beta for final polishing."""
    ac = np.clip(a_init, 1e-15, None)
    ac = ac / np.sum(ac)
    x0 = np.log(ac)
    best_a = ac.copy()
    best_obj = true_obj(ac, n)
    
    for beta in [50000, 200000, 500000, 1000000]:
        rem = deadline - time.monotonic()
        if rem < 5:
            break
        ctx = LBFGSContext(time.monotonic() + rem * 0.8)
        try:
            res = sp_minimize(obj_grad_with_ctx, x0, args=(beta, n, ctx),
                              method='L-BFGS-B',
                              options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-13,
                                       'maxfun': 100000, 'maxls': 50})
            if res.fun < np.inf:
                a_new = softmax_fn(res.x)
                obj_new = true_obj(a_new, n)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new
                    x0 = res.x
                    print(f"      Fine-tune beta={beta}: obj={obj_new:.8f} **")
                else:
                    print(f"      Fine-tune beta={beta}: obj={obj_new:.8f}")
        except (RuntimeError, Exception):
            pass
    
    return best_a, best_obj


def local_refine(a_init, n, budget_sec):
    """
    Quick local refinement: perturb coordinates and re-run short L-BFGS.
    Helps escape very shallow local minima.
    """
    start = time.monotonic()
    best_a = a_init.copy()
    best_obj = true_obj(best_a, n)
    rng_local = np.random.default_rng(int(time.monotonic() * 1000))
    
    attempts = 0
    while (time.monotonic() - start) < budget_sec and attempts < 20:
        attempts += 1
        # Small Gaussian perturbation in log-space
        x = np.log(np.clip(best_a, 1e-15, None))
        x += rng_local.normal(0, 0.02, n)
        a_try = softmax_fn(x)
        
        # Quick L-BFGS at moderate beta
        rem = budget_sec - (time.monotonic() - start)
        if rem < 3:
            break
        a_opt, obj_opt = run_single_lbfgs(a_try, 10000, n, max(2.0, rem * 0.4))
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: L-BFGS with log-sum-exp smoothing, beta-annealing,
    multi-n exploration, random restarts, and high-beta fine-tuning.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)
    
    best_a = None
    best_obj = np.inf
    best_n = 0
    
    # Load parent
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        pa = np.clip(pa, 0, None)
        if pa.size > 10 and np.isfinite(pa).all() and np.sum(pa) > 0.01:
            pa = pa / np.sum(pa)
            n_p = len(pa)
            obj_p = true_obj(pa, n_p)
            best_a = pa.copy()
            best_obj = obj_p
            best_n = n_p
            print(f"Parent loaded: n={n_p}, obj={obj_p:.8f}")
    except Exception:
        pass
    
    if best_a is None:
        best_a = np.ones(800) / 800.0
        best_n = 800
        best_obj = 2.0
        print("No parent, using uniform n=800: obj=2.0")
    
    # === Phase 1: Beta-anneal on parent n ===
    print(f"\n=== Phase 1: Beta-anneal (n={best_n}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        alloc = min(rem * 0.35, 400.0)
        a1, obj1 = beta_anneal(best_a, best_n, alloc)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1
            print(f"  ** Phase 1 improved: obj={best_obj:.8f}")
        else:
            print(f"  Phase 1: obj={obj1:.8f} (no improvement)")
    
    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    n_base = best_n
    n_candidates = set()
    for delta in [-200, -100, -50, 0, 50, 100, 200, 400, 700, 1000]:
        n_cand = n_base + delta
        if 300 <= n_cand <= 3000:
            n_candidates.add(n_cand)
    # Also try some fixed values
    for nv in [500, 700, 900, 1100, 1400, 1800, 2200]:
        if 300 <= nv <= 3000:
            n_candidates.add(nv)
    
    n_list = sorted(n_candidates)
    # Prioritize n values close to best_n
    n_list.sort(key=lambda x: abs(x - best_n))
    n_list = n_list[:10]  # Limit to 10 different n values
    
    for n_try in n_list:
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        if n_try == best_n:
            # Already handled in Phase 1, do a quick refinement
            alloc = min(rem * 0.08, 40.0)
            a_opt, obj_opt = local_refine(best_a, best_n, alloc)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt
                print(f"  ** n={n_try} refine: obj={best_obj:.8f}")
            continue
        
        a_try = resample_seq(best_a, n_try)
        alloc = min(rem * 0.12, 100.0)
        a_opt, obj_opt = beta_anneal(a_try, n_try, alloc)
        label = "  **" if obj_opt < best_obj else "     "
        print(f"{label} n={n_try:>5d}: obj={obj_opt:.8f}")
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt
            best_n = n_try
    
    # === Phase 3: Random restarts ===
    print(f"\n=== Phase 3: Random Restarts (n={best_n}) ===")
    n_restarts = min(6, max(2, int((deadline - time.monotonic()) / 80)))
    for r in range(n_restarts):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        a_r = make_random_init(best_n, rng)
        alloc = min(rem * 0.2, 120.0)
        a_opt, obj_opt = beta_anneal(a_r, best_n, alloc)
        label = "  **" if obj_opt < best_obj else "     "
        print(f"{label} Restart {r} (n={best_n}): obj={obj_opt:.8f}")
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt
    
    # === Phase 4: Also try restarts with different n ===
    print(f"\n=== Phase 4: Cross-n Random Restarts ===")
    n_alt_list = [n for n in [best_n + 200, best_n - 200, best_n + 500]
                  if 300 <= n <= 2500 and n != best_n]
    for n_alt in n_alt_list[:3]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        a_r = make_random_init(n_alt, rng)
        alloc = min(rem * 0.2, 100.0)
        a_opt, obj_opt = beta_anneal(a_r, n_alt, alloc)
        label = "  **" if obj_opt < best_obj else "     "
        print(f"{label} Cross-n restart (n={n_alt}): obj={obj_opt:.8f}")
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt
            best_n = n_alt
    
    # === Phase 5: Local refinement around best ===
    print(f"\n=== Phase 5: Local Refinement ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        alloc = min(rem * 0.3, 150.0)
        a_r, obj_r = local_refine(best_a, best_n, alloc)
        if obj_r < best_obj:
            best_obj = obj_r
            best_a = a_r
            print(f"  ** Local refine: obj={best_obj:.8f}")
    
    # === Phase 6: High-beta fine-tuning ===
    print(f"\n=== Phase 6: High-Beta Fine-Tuning ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        a_f, obj_f = fine_tune_high_beta(best_a, best_n, deadline)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f
            print(f"  ** Fine-tune: obj={best_obj:.8f}")
    
    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_obj:
            best_obj = val
    else:
        # Fallback
        result = [1.0] * 100
        best_obj = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
