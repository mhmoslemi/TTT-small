"""
Hybrid optimization: IFT (Iterative Fourier Transform) for flat-spectrum initialization,
LP polishing via HiGHS, FFT-based peak-targeted gradient descent, peak-peeling,
and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Novel contributions: spectral flattening via IFT, adaptive peak-peeling with
mass transfer, and structured coarse-to-fine multi-scale search.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def compute_obj_and_peak(a):
    """Return (objective, peak_value, peak_lag)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf, 0, 0
    a_norm = a / s
    b = compute_conv(a_norm)
    max_b = float(np.max(b))
    k_star = int(np.argmax(b))
    return 2.0 * n * max_b, max_b, k_star


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def ift_flat_spectrum(a, time_budget=60.0, verbose=False, clip_mode='hard'):
    """
    Iterative Fourier Transform to flatten |DFT(a)|, driving auto-correlation flat.
    clip_mode: 'hard' (max(a,0)), 'square' (a^2), 'exp' (softmax-like)
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    target_mag = 1.0 / np.sqrt(N)
    stall = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        A = rfft(a, N)
        phase = np.angle(A)

        A_new = target_mag * np.exp(1j * phase)
        a_raw = irfft(A_new, N)[:n]

        if clip_mode == 'hard':
            a_new = np.maximum(a_raw, 0.0)
        elif clip_mode == 'square':
            a_new = a_raw ** 2
        else:  # 'exp'
            a_shifted = a_raw - np.max(a_raw)
            a_new = np.exp(a_shifted)

        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            stall += 1
            if stall > 50:
                break
            continue

        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Blend with best to escape
                a = 0.5 * (a + best_a)
                a = a / np.sum(a)
                stall = 0

        if verbose and it % 2000 == 0:
            print(f"          IFT it{it}: obj={best_obj:.8f}, stall={stall}")

    return best_a


def ift_weighted(a, time_budget=60.0, verbose=False):
    """
    IFT with adaptive target magnitude: boost frequency bins where |A|^2 is
    below target, reduce where above. Uses a power law for smooth transition.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    # Power for aggressive flattening: higher = more aggressive
    power = 0.5
    stall = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        A = rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Adaptive target: geometric mean of current magnitudes
        log_mag = np.log(mag + 1e-30)
        target = np.exp(np.mean(log_mag))

        # Blend: new_mag = target^alpha * mag^(1-alpha)
        alpha = min(0.3 + 0.5 * it / 5000.0, 0.8)
        new_mag = target ** alpha * (mag + 1e-30) ** (1.0 - alpha)

        A_new = new_mag * np.exp(1j * phase)
        a_raw = irfft(A_new, N)[:n]

        a_new = np.maximum(a_raw, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            stall += 1
            if stall > 100:
                break
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                alpha = max(0.1, alpha - 0.1)
                a = 0.7 * best_a + 0.3 * a
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

        if verbose and it % 2000 == 0:
            print(f"          IFTW it{it}: obj={best_obj:.8f}, alpha={alpha:.3f}")

    return best_a


def fft_gradient_descent(a, time_budget=120.0, mode='peak', verbose=False):
    """
    FFT-based gradient descent with peak-targeted or energy-mode gradient.
    Peak mode: subgradient of max(conv) focused on the peak lag.
    Energy mode: gradient of ||conv||^2.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    stall = 0
    max_stall = 300

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        if mode == 'peak':
            # Subgradient: focus on the peak lag using softmax weighting
            b_max = np.max(b)
            # Use multiple peaks (top-k weighted subgradient)
            temperature = 0.01 * b_max + 1e-6
            b_shifted = b - b_max
            w = np.exp(b_shifted / temperature)
            w_sum = float(np.sum(w))
            if w_sum < 1e-30:
                w = np.zeros_like(b)
                w[int(np.argmax(b))] = 1.0
            else:
                w = w / w_sum
            # Gradient: c[i] = sum_m w[m] * a[m-i] (cross correlation of w with a)
            a_flip = a[::-1]
            faf = rfft(a_flip, N)
            fw = rfft(w, N)
            c_full = irfft(faf * fw, N)[:3 * n - 2]
            c = c_full[n - 1: 2 * n - 1]
        else:
            # Energy gradient: c[i] = sum_m a[m] * b[m+i]
            a_flip = a[::-1]
            faf = rfft(a_flip, N)
            fb = rfft(b, N)
            c_full = irfft(faf * fb, N)[:3 * n - 2]
            c = c_full[n - 1: 2 * n - 1]

        # Center to maintain sum constraint
        c = c - np.mean(c)

        v = momentum * v - eta * c
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 3.0)
            momentum = min(momentum + 0.0005, 0.95)
        else:
            stall += 1
            eta *= 0.97
            v *= 0.5
            if stall > max_stall:
                rng_local = np.random.default_rng(hash((it, n)) % (2**32))
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.7
                stall = 0
                max_stall = min(max_stall + 200, 800)
            if eta < 1e-8:
                break

        if verbose and it % 2000 == 0:
            print(f"          GD({mode}) it{it}: obj={best_obj:.8f}, eta={eta:.4f}")

    return best_a


def peak_peel(a, time_budget=30.0, verbose=False):
    """
    Peak-peeling: iteratively identify the peak lag and transfer mass
    from coordinates that contribute most to the peak to those that don't.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv(a)
        k_star = int(np.argmax(b))
        peak_val = b[k_star]

        # Contribution of each coordinate i to the peak:
        # a[i] appears in (a*a)[k_star] as a[i]*a[k_star-i] if 0 <= k_star-i < n
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Total peak = sum(contrib)
        # We want to reduce coordinates with high contrib
        # Transfer mass to low-contrib coordinates

        # Sort by contribution
        order = np.argsort(contrib)
        # Top contributors (to decrease)
        n_transfer = max(n // 10, 1)
        donors = order[-n_transfer:]  # highest contribution
        recipients = order[:n_transfer]  # lowest contribution

        # Transfer amount: proportional to the excess contribution
        mass_to_transfer = peak_val * 0.05  # 5% of peak mass
        donor_total = np.sum(a[donors])
        if donor_total < 1e-10:
            continue

        # How much to take from each donor (proportional to their value)
        proportions = a[donors] / donor_total
        amounts = mass_to_transfer * proportions

        a_trial = a.copy()
        a_trial[donors] -= amounts
        a_trial = np.maximum(a_trial, 0)

        # Distribute to recipients proportionally
        recip_total = np.sum(a[recipients])
        if recip_total < 1e-10:
            # Distribute uniformly among recipients
            a_trial[recipients] += mass_to_transfer / len(recipients)
        else:
            recip_props = a[recipients] / recip_total
            a_trial[recipients] += mass_to_transfer * recip_props

        s_trial = float(np.sum(a_trial))
        if s_trial < 1e-10:
            continue
        a_trial = a_trial / s_trial

        obj_trial = compute_obj(a_trial)
        if obj_trial < best_obj - 1e-13:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([a.sum()])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Power law tail
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x ** 0.3
        a = a / np.sum(a)
    elif mode == 6:
        # Periodic structure
        period = rng.integers(3, 20)
        pattern = rng.uniform(0.1, 1.0, period)
        a = np.tile(pattern, n // period + 1)[:n]
        a = a / np.sum(a)
    else:
        # Sine-modulated
        freq = rng.uniform(1, 5)
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
        a = a / np.sum(a)
    return a

def compute_conv(a):

            best_a = (1.0 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 12:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([float(a.sum())])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = int(rng.integers(0, 8))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = 0.5
