import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. 2011
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_step(a, conv, tau, K):

"""
Optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Subgradient with soft-argmax multi-peak tracking + LP alternating
+ parametric exploration + multi-n search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = int(rho_candidates[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum = 1 (fix floating point)
    result_sum = result.sum()
    if result_sum > 0:
        result = result / result_sum
    return result


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Using scipy linprog with HiGHS
    # Variables: g[0..n-1], M
    # Minimize M -> c = [0,...,0,1]
    # Constraints: A@ <= M -> A@ - M <= 0
    #              sum(g) = 1
    #              g >= 0, M >= 0 (actually M can be anything, but it'll be positive)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0 for each row
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    # Fallback: try cvxpy
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(M.value)
    except Exception:
        pass
    
    return None, None

            else:
                w = w / w_sum
            
            g = np.zeros(n)
            for idx in range(K):
                if w[idx] < 1e-12:
                    continue
                k = int(top_k[idx])
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                segment = a[k - hi : k - lo + 1][::-1]
                g[lo:hi + 1] += 2.0 * w[idx] * segment
            
            # Project gradient onto tangent space
            g -= np.mean(g)
            
            # Scale gradient
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            # Backtracking line search
            improved = False
            for _ in range(15):
                a_new = a - alpha * g / g_norm
                a_new = simplex_project(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    alpha = min(alpha * 1.1, 10.0)
                    no_improve = 0
                    improved = True
                    break
                else:
                    alpha *= 0.6
            
            if not improved:
                no_improve += 1
                if no_improve > 30:
                    # Perturb and reset step size
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = simplex_project(a)
                    alpha = 1.0 / np.sqrt(n)
                    no_improve = 0
    
    return best_a, best_obj


def parametric_explore(a_center, time_budget=200.0, n_components=15, pop_size=30, seed=42):
    """
    Explore in a low-dimensional parametric space around a_center.
    Parametrization: a_i = a_center_i * exp(sum_m w_m * phi_m(i)) / Z
    where phi_m are basis functions (cosines).
    """
    n = len(a_center)
    a_center = np.maximum(a_center, 1e-10)
    
    rng = np.random.default_rng(seed)
    
    # Define basis functions
    basis = np.zeros((n_components, n))
    for m in range(n_components):
        # Different frequencies
        freq = (m + 1) * np.pi / n
        basis[m] = np.cos(freq * np.arange(n))
    # Add a linear ramp
    basis[0] = np.linspace(-1, 1, n)
    
    best_a = a_center.copy()
    best_obj = compute_obj(a_center)
    start = time.monotonic()
    
    # Differential Evolution
    # Population: each individual is a vector of n_components weights
    pop = np.zeros((pop_size, n_components))
    pop[0] = np.zeros(n_components)  # first individual is the center
    for i in range(1, pop_size):
        pop[i] = rng.normal(0, 0.3, n_components)
    
    def evaluate_individual(weights):
        a = a_center * np.exp(np.dot(weights, basis.T))
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            return np.inf
        a = a / s
        return compute_obj(a), a
    
    # Evaluate initial population
    pop_objs = np.full(pop_size, np.inf)
    pop_a = [None] * pop_size
    for i in range(pop_size):
        obj_i, a_i = evaluate_individual(pop[i])
        pop_objs[i] = obj_i
        pop_a[i] = a_i
        if obj_i < best_obj:
            best_obj = obj_i
            best_a = a_i.copy()
    
    gen = 0
    while time.monotonic() - start < time_budget:
        gen += 1
        for i in range(pop_size):
            if time.monotonic() - start > time_budget:
                break
            
            # DE/rand/1
            a_idx, b_idx, c_idx = rng.choice(pop_size, 3, replace=False)
            while a_idx == b_idx or a_idx == c_idx or a_idx == i:
                a_idx, b_idx, c_idx = rng.choice(pop_size, 3, replace=False)
            
            F = rng.uniform(0.5, 1.0)
            CR = rng.uniform(0.3, 0.8)
            
            trial = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])
            # Shuffle
            mask = rng.random(n_components) < CR
            if not mask.any():
                j0 = rng.integers(0, n_components)
                mask[j0] = True
            trial = np.where(mask, trial, pop[i])
            
            # Bound the weights
            trial = np.clip(trial, -3, 3)
            
            obj_trial, a_trial = evaluate_individual(trial)
            
            if obj_trial <= pop_objs[i]:
                pop[i] = trial
                pop_objs[i] = obj_trial
                pop_a[i] = a_trial
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
