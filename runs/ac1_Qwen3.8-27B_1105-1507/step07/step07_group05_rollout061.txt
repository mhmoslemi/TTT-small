def adam_lse_descent(a, time_budget=60.0, verbose=False):
    """Log-sum-exp annealing gradient descent with Adam optimizer."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    
    # Annealing schedule
    tau_start = 5.0
    tau_end = 5000.0
    n_steps = int(time_budget * 20)  # estimate total steps
    
    t = 0
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        t = step / max(n_steps - 1, 1)
        # Exponential annealing of tau
        tau = tau_start * (tau_end / tau_start) ** t
        
        grad, b = lse_gradient_step(a, tau, N)
        
        # Center gradient to maintain sum constraint (approximately)
        grad = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(step+1))
        v_hat = v / (1 - beta2**(step+1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2*n-1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce learning rate if stuck
            lr *= 0.95
            if lr < 1e-6:
                break
        
        if verbose and step % 2000 == 0:
            print(f"            LSE-Adam step {step}: tau={tau:.1f}, obj={best_obj:.8f}, lr={lr:.6f}")
    
    return best_a

def frank_wolfe(a, time_budget=30.0):
    """Frank-Wolfe (conditional gradient) method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2*n-1]
        k_star = int(np.argmax(b))
        
        # Subgradient at peak: g[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]
        
        # Center
        g = g - np.mean(g)
        
        # Frank-Wolfe: find vertex minimizing <g, e_j> = g[j]
        j_min = int(np.argmin(g))
        
        # Step size: gamma = 2/(it+2) (standard FW rate) or line search
        gamma = 2.0 / (it + 2)
        a_new = (1 - gamma) * a + gamma * np.eye(n)[j_min]
        
        # Line search
        best_t = gamma
        best_obj_t = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, gamma, 1.0]:
            a_try = (1 - t) * a + t * np.eye(n)[j_min]
            fa_try = rfft(a_try, N)
            b_try = irfft(fa_try * fa_try, N)[:2*n-1]
            obj_try = 2.0 * n * float(np.max(b_try))
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
        
        a_new = (1 - best_t) * a + best_t * np.eye(n)[j_min]
        
        if best_obj_t < best_obj - 1e-12:
            best_obj = best_obj_t
            best_a = a_new.copy()
            a = a_new
    
    return best_a

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix A (2n-1 x n) such that A @ g = conv(a, g)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, min(2*n-1, j+n)):
            A[k, j] = a[k - j]
    
    try:
        from scipy.optimize import linprog
        # Minimize M
        # Variables: g[0..n-1], M
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        # A_ub @ [g, M] <= b_ub
        # conv(a,g)[k] - M <= 0 for all k
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        
        # sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None

def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            a = (1 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, rng):
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Circulant-inspired: use roots of unity magnitude
        x = rng.uniform(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * rng.uniform(0, 1) * np.arange(n) / n)
        a = np.abs(a) + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Tent function variant
        x = np.arange(n) / n
        a = 1.0 - np.abs(2 * x - 1)
        a = a + 0.1
        a = a / np.sum(a)
    else:
        # Random periodic
        period = int(rng.integers(max(2, n//10), max(3, n//3)))
        base = rng.uniform(0.5, 1.5, min(period, n))
        a = np.tile(base, n // len(base) + 1)[:n]
        a = a / np.sum(a)
    return a

def peak_chase(a, time_budget=20.0):
    """Subgradient chase on the peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.03
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2*n-1]
        k_star = int(np.argmax(b))
        
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
        
        grad = grad - np.mean(grad)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2*n-1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.2)
        else:
            eta *= 0.97
            if eta < 1e-7:
                break
    
    return best_a

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LSE-Adam annealing, multi-peak subgradient method, spectral shaping,
LP polishing, and coordinate descent with multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    b = compute_conv(a_n)
    return 2.0 * n * float(np.max(b))


def lse_objective_and_grad(a, tau, N):
    """
    Compute LSE(tau * conv(a,a)) and its gradient with respect to a.
    Assumes a is normalized (sum=1).
    Returns (lse_value, gradient, b).
    """
    n = len(a)
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]

    # Softmax weights
    tau_b = tau * b
    tau_b_shifted = tau_b - np.max(tau_b)
    exp_vals = np.exp(tau_b_shifted)
    w = exp_vals / np.sum(exp_vals)

    # Gradient: df/da[i] = 2*n*tau * 2 * sum_m a[m] * w[i+m]
    # = 4*n*tau * cross_correlation(w, a)[i]
    # where cross[i] = sum_{m=0}^{n-1} a[m] * w[i+m]

    # w has length 2n-1
    w_rev = w[::-1]  # length 2n-1
    # Full conv of a (len n) and w_rev (len 2n-1) has length 3n-2
    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1
    fa2 = rfft(a, N2)
    fw2 = rfft(w_rev, N2)
    conv_full = irfft(fa2 * fw2, N2)[:3 * n - 2]
    cross = conv_full[n - 1::-1]  # cross[i] = conv_full[n-1-i]

    obj = 2.0 * n * (np.log(np.sum(exp_vals) / exp_vals.size) + np.max(tau_b) / tau)
    # Actually LSE(tau*b) = log(sum(exp(tau*b))) = log(sum(exp(tau_b_shifted))) + tau*b_max
    lse_val = np.log(np.sum(exp_vals)) + np.max(tau_b)

    grad = 4.0 * n * tau * cross
    return lse_val, grad, b


def adam_lse_descent(a, time_budget=60.0, verbose=False, tau_start=5.0, tau_end=10000.0):
    """Log-sum-exp annealing with Adam optimizer."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5

    # Estimate total steps
    est_steps = int(time_budget * 30)

    t = 0
    stall_count = 0
    max_stall = 50

    for step in range(est_steps):
        if time.monotonic() - start > time_budget:
            break
        t = step / max(est_steps - 1, 1)

        # Annealing: exponential then linear ramp
        if t < 0.7:
            tau = tau_start * (tau_end / tau_start) ** (t / 0.7)
        else:
            frac = (t - 0.7) / 0.3
            tau = tau_end * (0.5 + 0.5 * frac)

        lse_val, grad, b = lse_objective_and_grad(a, tau, N)

        # Center gradient to approximately preserve sum
        grad = grad - np.mean(grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            lr = min(lr * 1.01, 3.0)
        else:
            stall_count += 1
            lr *= 0.96
            if stall_count > max_stall:
                # Perturb and reset

                a = best_a * np.exp(rng_local.normal(0, 0.03, n))

                if s > 1e-10:
                    a = a / s
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.5
                stall_count = 0
                max_stall = min(max_stall + 50, 300)

        if lr < 1e-7:
            break

        if verbose and step % 3000 == 0:

def multi_peak_subgrad(a, time_budget=30.0, K=5, verbose=False):
    """Subgradient method focused on top K peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.02

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Find top K peaks
        top_k = np.argsort(b)[-K:]

        # Average subgradient from top K peaks
        grad = np.zeros(n)
        for k_star in top_k:
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= K
        grad = grad - np.mean(grad)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.15)
        else:
            eta *= 0.97
            if eta < 1e-7:
                break

    return best_a


def spectral_shaping(a, time_budget=30.0, verbose=False):
    """
    Spectral relaxation: shape the power spectrum to be flat, then project back.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    alpha = 0.5
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        # Current spectrum
        fa = rfft(a, N)
        power = np.abs(fa) ** 2

        # Wiener filter: reduce peaks in power spectrum
        # A_new = A * exp(-alpha * power)
        fa_new = fa * np.exp(-alpha * power / (np.max(power) + 1e-10))

        # Inverse FFT
        a_new = irfft(fa_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / s_new

        # Evaluate
        fa_check = rfft(a_new, N)
        b = irfft(fa_check * fa_check, N)[:2 * n - 1]
        obj = 2.0 * n * float(np.max(b))

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.01, 5.0)
        else:
            alpha *= 0.95
            if alpha < 0.01:
                break

    return best_a


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)

    # Build constraint matrix A (2n-1 x n): (A @ g)[k] = conv(a, g)[k]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, min(2 * n - 1, j + n)):
            A[k, j] = a[k - j]

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0

        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization for precise polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            a = (1 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            stall = 0
            if verbose and it % 10 == 0:
                print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall > 20:
                break

    return best_a


def peak_chase(a, time_budget=20.0):
    """Subgradient chase on the single peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.02

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(8000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        grad = grad - np.mean(grad)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.003, 0.1)
        else:
            eta *= 0.98
            if eta < 1e-8:
                break

    return best_a


def coordinate_descent_fast(a, time_budget=30.0, verbose=False):
    """Fast coordinate descent with FFT-based updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    n_sweeps = 20
    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        cur_max = np.max(b)

        improved = False
        # Random order
        order = np.random.permutation(n)

        for i in order:
            # Compute base (conv without coordinate i's contribution)
            # b[k] = base[k] + 2*a[i]*a[k-i] (for 0<=k-i<n, k!=2i)
            #       base[k] + a[i]^2 (for k=2i)
            # base[k] = b[k] - contribution

            # Try a few candidate values for a[i]
            # The optimal x minimizes max_k (base[k] + 2*x*a[k-i])
            # Simple approach: try x=0, x=current/2, x=current
            candidates = [0.0, a[i] * 0.3, a[i] * 0.7, a[i], a[i] * 1.4]

            best_x = a[i]
            best_peak = np.inf

            for x in candidates:
                # Compute new b efficiently: b_new = b - old_contrib + new_contrib
                b_try = b.copy()
                # Remove old contribution
                for k in range(max(0, i), min(2 * n - 1, i + n)):
                    km = k - i
                    if 0 <= km < n:
                        if k == 2 * i:
                            b_try[k] -= a[i] * a[i]
                        else:
                            b_try[k] -= 2 * a[i] * a[km]
                # Add new contribution
                for k in range(max(0, i), min(2 * n - 1, i + n)):
                    km = k - i
                    if 0 <= km < n:
                        if k == 2 * i:
                            b_try[k] += x * x
                        else:
                            b_try[k] += 2 * x * a[km]

                peak = np.max(b_try)
                if peak < best_peak:
                    best_peak = peak
                    best_x = x

            if best_x < a[i] - 1e-10 or best_x > a[i] + 1e-10:
                a[i] = best_x
                improved = True

        # Renormalize
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        obj = 2.0 * n * float(np.max(b))
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        elif not improved:
            break

        if verbose and sweep % 5 == 0:
            print(f"            CD sweep {sweep}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    else:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x ** 0.3
        a = a / np.sum(a)
    return a


def combined_optimize(a, time_budget=60.0, verbose=False):
    """Run multiple optimization methods in sequence."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    methods = [
        ("LSE", adam_lse_descent, 0.35),
        ("Spectral", spectral_shaping, 0.15),
        ("MultiPeak", multi_peak_subgrad, 0.15),
        ("LP", lp_polish, 0.25),
        ("PeakChase", peak_chase, 0.10),
    ]

    for name, func, frac in methods:
        remaining = max(0, start + time_budget - time.monotonic())
        if remaining < 10:
            break
        alloc = min(remaining * frac, 40.0)
        if alloc < 5:
            continue
        result = func(best_a, time_budget=alloc, verbose=verbose) if 'time_budget' in func.__code__.co_varnames else func(best_a, time_budget=alloc)
        obj = compute_obj(result)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = result.copy()
            if verbose:
                print(f"            Combined {name}: {best_obj:.8f}")

    return best_a
