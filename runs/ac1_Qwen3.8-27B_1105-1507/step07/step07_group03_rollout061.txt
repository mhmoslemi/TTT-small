```python
def lbfgs_optimize(a_init, t_values, max_iter=200):
    """Optimize using L-BFGS on smoothed objective with softmax parameterization."""
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    # Initialize z such that softmax(z) = a
    z = np.log(a)
    z -= z.mean()  # for numerical stability
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in t_values:
        # Define objective and gradient for this t
        def obj_z(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            conv = compute_conv(a)
            # Smoothed max
            tc = t * conv
            tc_max = tc.max()
            return (tc_max + np.log(np.sum(np.exp(tc - tc_max)))) / t
        
        def grad_z(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            conv = compute_conv(a)
            tc = t * conv
            tc_max = tc.max()
            w = np.exp(tc - tc_max)
            w = w / w.sum()
            
            # grad_a[i] = 2 * sum_j w[i+j] * a[j]
            idx = precomputed_idx  # shape (n, n)
            M = w[idx]
            grad_a = 2.0 * (M @ a)
            
            # Convert to grad_z via softmax chain rule
            grad_z = a * (grad_a - np.sum(a * grad_a))
            return grad_z
        
        # Run L-BFGS
        from scipy.optimize import minimize
        res = minimize(obj_z, z, jac=grad_z, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'ftol': 1e-10})
        
        a_new = np.exp(res.x - res.x.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a
```

```python
Optimization using L-BFGS on log-sum-exp smoothed max-convolution objective,
combined with mirror descent, multi-n exploration, and random restarts.
Key innovation: proper gradient computation for the smooth approximation.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def precompute_idx(n):
    """Precompute index matrix for gradient computation."""
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    return i + j  # shape (n, n), values 0 to 2n-2

def lbfgs_smoothed(a_init, idx, t_schedule, max_iter_per_t=100, time_budget=60):
    """L-BFGS optimization with log-sum-exp smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    z = np.log(a) - np.log(a).mean()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for t in t_schedule:
        if time.monotonic() - start > time_budget:
            break
            
        def obj_fn(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            conv = compute_conv(a)
            tc = t * conv
            tc_max = tc.max()
            val = (tc_max + np.log(np.sum(np.exp(tc - tc_max)))) / t
            return val
        
        def grad_fn(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            conv = compute_conv(a)
            tc = t * conv
            tc_max = tc.max()
            w = np.exp(tc - tc_max)
            w = w / w.sum()
            M = w[idx]
            grad_a = 2.0 * (M @ a)
            return a * (grad_a - np.dot(a, grad_a))
        
        try:
            res = minimize(obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                          options={'maxiter': max_iter_per_t, 'ftol': 1e-12, 'gtol': 1e-8})
            a_new = np.exp(res.x - res.x.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0)
            if a_new.sum() < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
        except Exception as e:
            pass
    
    return best_a
```

```python
"""
Optimization using L-BFGS on log-sum-exp smoothed max-convolution objective.
Key innovation: proper analytic gradient computation for the smooth approximation
of the max, enabling much faster convergence than subgradient methods.
Combined with mirror descent, multi-n exploration, and random restarts.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def precompute_idx(n):
    """Precompute index matrix for gradient computation. M[i,j] = i+j."""
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    return i + j


def lbfgs_smoothed(a_init, idx, t_schedule=None, max_iter_per_t=200, time_budget=60.0):
    """L-BFGS optimization with log-sum-exp smoothing of the max convolution."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    z0 = np.log(a) - np.log(a).mean()

    if t_schedule is None:
        t_schedule = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    z = z0.copy()
    for ti, t in enumerate(t_schedule):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        iters = max(50, int(max_iter_per_t * min(1.0, remaining / 30.0)))

        def obj_fn(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            conv = compute_conv(a)
            tc = t * conv
            tc_max = tc.max()
            return (tc_max + np.log(np.sum(np.exp(tc - tc_max)))) / t

        def grad_fn(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            conv = compute_conv(a)
            tc = t * conv
            tc_max = tc.max()
            w = np.exp(tc - tc_max)
            w = w / w.sum()
            M = w[idx]
            grad_a = 2.0 * (M @ a)
            return a * (grad_a - np.dot(a, grad_a))

        try:
            res = minimize(obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-14, 'gtol': 1e-10,
                                    'maxls': 50})
            z_new = res.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    z = z_new
        except Exception:
            pass

    return best_a


def lbfgs_active_set(a_init, idx, top_k=5, n_iters=300, time_budget=60.0):
    """L-BFGS on sum of top-k peaks (active set approach)."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    z0 = np.log(a) - np.log(a).mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    z = z0.copy()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        a_curr = np.exp(z - z.max())
        a_curr = a_curr / a_curr.sum()
        conv = compute_conv(a_curr)
        top_idx = np.argsort(conv)[-top_k:]
        top_val = conv[top_idx]

        # Softmax weights over top peaks
        max_v = top_val[-1]
        temp = max(0.01, (max_v - top_val[0]) * 0.3)
        w = np.exp((top_val - max_v) / temp)
        w = w / w.sum()

        # Build full weight vector (sparse)
        w_full = np.zeros(2 * n - 1)
        for wi, ki in zip(w, top_idx):
            w_full[int(ki)] = wi

        def obj_fn(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            conv = compute_conv(a)
            return float(np.sum(w_full * conv))

        def grad_fn(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            M = w_full[idx]
            grad_a = 2.0 * (M @ a)
            return a * (grad_a - np.dot(a, grad_a))

        try:
            res = minimize(obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                           options={'maxiter': 30, 'ftol': 1e-14, 'gtol': 1e-10})
            z_new = res.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    z = z_new
        except Exception:
            break

    return best_a


def mirror_descent(a, n_iters=3000, time_budget=60.0):
    """Mirror descent (exponential weights) with multi-peak subgradient."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.05
    m_peaks = 8

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[-m_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        temp = max(0.005, (max_val - top_val[0]) * 0.3)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            valid_i = np.arange(max(0, ki - n + 1), min(n, ki + 1))
            valid_j = ki - valid_i
            grad[valid_i] += w * a[valid_j]

        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.99
        else:
            alpha *= 0.5

        if it % 500 == 499:
            alpha = min(alpha * 1.15, 0.3)

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """Coordinate-wise refinement targeting the peak positions."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    stall = 0
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        # Move mass from high to low within peak-contributing positions
        step = max(1, n // 300)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 5 * a[j] + 1e-6:
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            # Random pair swap
            rng = np.random.default_rng(it * 13 + n)
            for _ in range(20):
                i1, i2 = rng.integers(0, n, 2)
                if i1 == i2:
                    continue
                delta = 0.005 * abs(a[i1] - a[i2]) + 0.001 * max(a[i1], a[i2])
                if a[i1] < delta:
                    delta = a[i1] * 0.5
                a[i1] -= delta
                a[i2] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                else:
                    a[i1] += delta
                    a[i2] -= delta

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        freqs = rng.uniform(0, 5, 10)
        phases = rng.uniform(0, 2 * np.pi, 10)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # Symmetric random
        half = n // 2
        a_half = rng.uniform(0.1, 1, half)
        a = np.concatenate([a_half, a_half[::-1]])[:n]
        a = a / np.sum(a)
    elif mode == 4:
        # Bump function
        centers = rng.integers(0, n, 3)
        widths = rng.integers(n // 10, n // 4, 3)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    else:
        # Smooth random
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a

        elif init_type == 2:
            n_r = int(n_best + rng.integers(-200, 200))
            n_r = max(400, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.04, n_r))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:
            # Symmetric perturbation
            a_r = best_seq.copy()
            mid = len(a_r) // 2
            a_r[:mid] *= (1.0 + rng.normal(0, 0.05, mid))
            a_r[mid:] = a_r[:mid][::-1] if len(a_r) % 2 == 0 else np.concatenate([a_r[:mid][::-1], [a_r[mid]]])
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        idx_r = precompute_idx(len(a_r))
        alloc_fft = min(remaining * 0.2, 20.0)
        a_opt = lbfgs_smoothed(a_r, idx_r,
                               t_schedule=[10.0, 50.0, 200.0],
                               max_iter_per_t=80, time_budget=alloc_fft)

        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc_md = min(remaining * 0.15, 15.0)
            a_opt = mirror_descent(a_opt, n_iters=1000, time_budget=alloc_md)

        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # Phase 9: Final intensive polish on best
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    if remaining > 60:
        idx_final = precompute_idx(n_final)
        alloc = min(remaining * 0.3, 120.0)
        a_opt = lbfgs_smoothed(best_seq, idx_final,
                               t_schedule=[200.0, 500.0, 1500.0, 5000.0, 15000.0],
                               max_iter_per_t=300, time_budget=alloc)
        obj = compute_obj(a_opt)
        print(f"  Final L-BFGS (high t): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    if remaining > 20:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 30.0)
        a_opt = coordinate_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        print(f"  Final coord: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    if remaining > 20:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 30.0)
        a_opt = mirror_descent(best_seq, n_iters=3000, time_budget=alloc)
        obj = compute_obj(a_opt)
        print(f"  Final MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
