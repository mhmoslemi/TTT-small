import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def qp_peak_shave(a, n_peaks=10, time_budget=30.0, reduction=0.01, mu=1.0, verbose=False):
    """
    Iteratively solve QP to reduce top peaks of autocorrelation.
    QP: min 0.5*mu*||delta||^2 s.t. B@delta <= rhs, sum(delta)=0, a+delta>=0
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    max_iters = 200
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        top_idx = np.argsort(conv)[-n_peaks:]
        top_idx = np.sort(top_idx)

        # Build constraint matrix B (n_peaks x n)
        B = np.zeros((n_peaks, n))
        rhs = np.zeros(n_peaks)
        for m, k in enumerate(top_idx):
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    B[m, i] = a[j]
            rhs[m] = -reduction * conv[ki]

        # Solve QP
        try:
            delta = cp.Variable(n)
            objective = cp.Minimize(0.5 * mu * cp.sum_squares(delta))
            constraints = [
                B @ delta <= rhs,
                cp.sum(delta) == 0,
                a + delta >= 0
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)

            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d = np.maximum(delta.value, -a)  # ensure a + d >= 0
                # Line search
                best_t_obj = best_obj
                best_t = 0.0
                for t in [1.0, 0.5, 0.25, 0.1]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_t = t

                if best_t > 0 and best_t_obj < best_obj - 1e-14:
                    best_obj = best_t_obj
                    a = (a + best_t * d)
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    stall = 0
                    if verbose and it % 20 == 0:
                        print(f"    QP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
                else:
                    stall += 1
                    # Try smaller reduction
                    if stall > 10:
                        reduction = max(0.001, reduction * 0.5)
                        stall = 0
            else:

        except Exception:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using QP-based peak shaving (new),
FISTA with soft-max, FFT flat spectrum, LP alternating, multi-n exploration,
and random restarts. The QP peak shaving directly targets top autocorrelation
peaks via minimum-norm linearly-constrained QP subproblems.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix_vec(a, k):
    """Return the row of the cross-correlation matrix for lag k.
    conv(delta, a)[k] = sum_i delta[i] * a[k-i] (with a[j]=0 outside [0,n-1])."""
    n = len(a)
    row = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            row[i] = a[j]
    return row


def qp_peak_shave(a, n_peaks=10, time_budget=60.0, reduction=0.01, mu=1.0, verbose=False):
    """
    Iteratively solve QP to reduce top peaks of autocorrelation.
    QP: min 0.5*mu*||delta||^2 s.t. B@delta <= rhs, sum(delta)=0, a+delta>=0
    where B[m,i] = a[k_m - i] (cross-correlation row for peak m).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    max_iters = 500
    eff_reduction = reduction

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[-n_peaks:]
        top_idx = np.sort(top_idx)

        # Build constraint matrix
        B = np.zeros((n_peaks, n))
        rhs = np.zeros(n_peaks)
        for m, k in enumerate(top_idx):
            ki = int(k)
            row = build_conv_matrix_vec(a, ki)
            B[m, :] = row
            rhs[m] = -eff_reduction * conv[ki]

        # Check feasibility: if all rhs are 0 or positive, no reduction needed
        if np.all(rhs >= 0):
            break

        # Solve QP
        try:
            delta = cp.Variable(n)
            objective = cp.Minimize(0.5 * mu * cp.sum_squares(delta))
            constraints = [
                B @ delta <= rhs,
                cp.sum(delta) == 0,
                a + delta >= 0
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)

            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d = delta.value.copy()
                d = np.clip(d, -a + 1e-15, None)
                # Line search
                best_t_obj = best_obj
                best_t = 0.0
                for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_t = t

                if best_t > 0 and best_t_obj < best_obj - 1e-14:
                    best_obj = best_t_obj
                    a = a + best_t * d
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    stall = 0
                    if verbose and it % 50 == 0:
                        print(f"      QP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, n_peaks={n_peaks}")
                else:
                    stall += 1
            else:
                stall += 1
        except Exception:
            stall += 1

        if stall > 0 and stall % 20 == 0:
            eff_reduction = max(0.001, eff_reduction * 0.5)
            if eff_reduction < 0.0015:
                # Try increasing n_peaks
                break
        if stall > 60:
            break

    return best_a


def fista_softmax(a, n_iters=3000, time_budget=60.0, t_init=0.005, verbose=False):
    """
    Nesterov-accelerated mirror descent with soft-max subgradient.
    Smooths the max of the autocorrelation using a log-sum-exp approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t = t_init
    alpha = 0.15
    tk = 1.0
    y = a.copy()
    x_prev = a.copy()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_y = compute_conv(y)
        max_val = float(np.max(conv_y))
        shifted = (conv_y - max_val) / t
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights /= wsum

        # Gradient: d/dx_i [sum_k w_k * conv(y,y)[k]]
        grad = np.zeros(n)
        for k in range(len(conv_y)):
            if weights[k] < 1e-10:
                continue
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += weights[k] * y[j]
        grad *= 2.0 * n

        # Mirror descent step in log domain
        log_new = np.log(np.maximum(y, 1e-30)) - alpha * grad
        log_max = np.max(log_new)
        x_new = np.exp(log_new - log_max)
        x_new = np.maximum(x_new, 1e-15)
        x_new = x_new / np.sum(x_new)

        # FISTA update
        tk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        y_new = x_new + ((tk - 1.0) / tk_new) * (x_new - x_prev)
        y_new = np.maximum(y_new, 1e-15)
        y_new = y_new / np.sum(y_new)

        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x_new.copy()

        x_prev = x_new
        y = y_new
        tk = tk_new

        # Decrease temperature
        t = max(t_init * 0.998, t_init * 0.1)

    return best_a


def fft_flat_spectrum(a, n_iters=300, time_budget=40.0, relaxation=0.5, verbose=False):
    """FFT-based flat spectrum iteration (Gerchberg-Saxton variant)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe

        A_flat = phase
        A_new = (1.0 - relaxation) * A + relaxation * A_flat

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 25:
                    rng = np.random.default_rng(it * 7 + n)
                    scale = 0.03 + 0.01 * min(stall // 25, 5)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s_val = float(np.sum(a))
                    if s_val > 1e-10:
                        a = a / s_val
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

    return best_a


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    # Build convolution matrix A (2n-1 x n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]

    solvers = [cp.ECOS, cp.SCS, cp.MOSEK, cp.PDLP]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP alternating optimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

        best_t_val = 0.3

        for t in candidate_ts:

            a_try = a_try / s_try

def coordinate_descent(a, time_budget=30.0):
    """Coordinate-wise refinement targeting the peak."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(80):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        step = max(1, n // 150)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 10 * a[j] and a[i] > 1e-5:
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv)) / (float(np.sum(a)) ** 2)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            rng = np.random.default_rng(it * 31)
            for _ in range(5):
                i1, i2 = rng.integers(0, n, 2)
                if i1 == i2:
                    continue
                delta = 0.02 * max(a[i1], a[i2])
                a[i1] -= delta
                a[i2] += delta
                if a[i1] < 0:
                    a[i1] = 0
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv)) / (float(np.sum(a)) ** 2)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True

        if not improved and it > 40:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        freqs = rng.uniform(0, 4, 8)
        phases = rng.uniform(0, 2 * np.pi, 8)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        c1, c2 = n // 3, 2 * n // 3
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2.0 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2.0 * w ** 2))
        a = a / np.sum(a)
    else:
        a = rng.uniform(0, 1, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a

        print(f"  FISTA: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fista.copy()

    # Phase 4: LP Alternating
    print(f"\n=== Phase 4: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.15, 100.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # Phase 5: QP + FFT alternation (escape local minima)
    print(f"\n=== Phase 5: QP/FFT Alternation ===")
    remaining = deadline - time.monotonic()
    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # FFT for global redeployment
        alloc_fft = min(remaining * 0.1, 20.0)
        a_tmp = fft_flat_spectrum(best_seq, n_iters=200, time_budget=alloc_fft, relaxation=0.4)

        # QP for local refinement
        alloc_qp = min(remaining * 0.15, 30.0)
        a_tmp = qp_peak_shave(a_tmp, n_peaks=8, time_budget=alloc_qp,
                              reduction=0.015, mu=1.0)
        obj = compute_obj(a_tmp)
        status = " **" if obj < best_score else ""
        print(f"  Round {rnd}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_tmp.copy()

    # Phase 6: Intensive QP polish
    print(f"\n=== Phase 6: Intensive QP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Use more peaks and smaller reduction for fine tuning
        alloc = min(remaining * 0.15, 80.0)
        a_qp_final = qp_peak_shave(best_seq, n_peaks=20, time_budget=alloc,
                                   reduction=0.005, mu=0.3, verbose=True)
        obj = compute_obj(a_qp_final)
        print(f"  QP (peaks=20, red=0.005, mu=0.3): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_qp_final.copy()

    # Phase 7: Coordinate descent
    print(f"\n=== Phase 7: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.06, 30.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj = compute_obj(a_cd)
        print(f"  CD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()

    # Phase 8: Multi-n exploration
    print(f"\n=== Phase 8: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        500, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500 and abs(n - n_best) >= 30]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: QP + FFT
        alloc1 = min(remaining * 0.08, 15.0)
        a_opt = qp_peak_shave(a_try, n_peaks=8, time_budget=alloc1, reduction=0.02, mu=1.0)
        alloc2 = min(remaining * 0.05, 10.0)
        a_opt = fft_flat_spectrum(a_opt, n_iters=100, time_budget=alloc2, relaxation=0.4)
        alloc3 = min(remaining * 0.05, 10.0)
        a_opt = qp_peak_shave(a_opt, n_peaks=10, time_budget=alloc3, reduction=0.01, mu=0.5)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # Phase 9: Random restarts
    print(f"\n=== Phase 9: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    n_restarts = min(8, int(remaining // 30))
    for r in range(max(1, n_restarts)):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        init_type = r % 4
        if init_type == 0:
            a_r = random_init(n_best, rng)
        elif init_type == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.06, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            n_r = int(n_best + rng.integers(-100, 100))
            n_r = max(300, min(1200, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.04, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            x = np.linspace(0, np.pi, n_best)
            a_r = np.cos(x) ** 2 + 0.5
            a_r = a_r * np.exp(rng.normal(0, 0.05, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        # Optimize: QP + FFT
        alloc_qp = min(remaining * 0.15, 15.0)
        a_opt = qp_peak_shave(a_r, n_peaks=8, time_budget=alloc_qp, reduction=0.02, mu=1.0)
        alloc_fft = min(remaining * 0.1, 10.0)
        a_opt = fft_flat_spectrum(a_opt, n_iters=150, time_budget=alloc_fft, relaxation=0.4)
        alloc_qp2 = min(remaining * 0.1, 10.0)
        a_opt = qp_peak_shave(a_opt, n_peaks=10, time_budget=alloc_qp2, reduction=0.01, mu=0.5)

        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # Phase 10: Final intensive polish
    print(f"\n=== Phase 10: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_final = best_seq.copy()

        # QP with many peaks
        alloc = min(remaining * 0.3, 60.0)
        a_qp = qp_peak_shave(a_final, n_peaks=25, time_budget=alloc,
                             reduction=0.003, mu=0.2, verbose=True)
        obj = compute_obj(a_qp)
        print(f"  Final QP (peaks=25): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_qp.copy()
            a_final = a_qp.copy()

        # FISTA
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.25, 30.0)
            a_f = fista_softmax(a_final, n_iters=2000, time_budget=alloc, t_init=0.002)
            obj = compute_obj(a_f)
            print(f"  Final FISTA: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_f.copy()
                a_final = a_f.copy()

        # More QP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 30.0)
            a_qp2 = qp_peak_shave(a_final, n_peaks=15, time_budget=alloc,
                                  reduction=0.005, mu=0.5)
            obj = compute_obj(a_qp2)
            print(f"  Final QP2: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_qp2.copy()

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 15.0)
            a_cd = coordinate_descent(best_seq, time_budget=alloc)
            obj = compute_obj(a_cd)
            print(f"  Final CD: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
