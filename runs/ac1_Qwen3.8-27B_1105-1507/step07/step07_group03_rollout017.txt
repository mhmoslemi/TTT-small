def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_fast(a_norm):
    n = len(a_norm)
    A = build_conv_matrix(a_norm)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n / new_n)

    x_new = np.linspace(0, 1, new_n)

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 0:
        a_new = a_new / s
    return a_new

        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # Cosine window with random modulation
        x = np.linspace(0, np.pi, n)
        a = (1 + np.cos(x)) / 2 + 0.1 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 4:
        # Bimodal
        c1, c2 = n // 3, 2 * n // 3
        w = max(1, n // 12)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    else:
        # Smoothed random
        a = rng.uniform(0, 1, n)
        kernel_len = max(3, n // 50)
        kernel = np.ones(kernel_len) / kernel_len
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 45.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: Fast LP alternating (main workhorse)
    print(f"\n=== Phase 1: LP Alternating (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 100.0)
        a_lp = lp_alternating_fast(best_seq, n_iters=2000, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # Phase 2: Mirror descent refinement
    print(f"\n=== Phase 2: Mirror Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.12, 80.0)
        a_md = mirror_descent(best_seq, n_iters=5000, time_budget=alloc)
        obj = compute_obj(a_md)
        print(f"  MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # Phase 3: Peak shaving
    print(f"\n=== Phase 3: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 40.0)
        a_ps = peak_shaving(best_seq, n_iters=500, time_budget=alloc)
        obj = compute_obj(a_ps)
        print(f"  Peak shave: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_ps.copy()

    # Phase 4: More LP
    print(f"\n=== Phase 4: More LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 100.0)
        a_lp2 = lp_alternating_fast(best_seq, n_iters=2000, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp2)
        print(f"  LP2: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()

    # Phase 5: FFT flat spectrum
    print(f"\n=== Phase 5: FFT Flat Spectrum ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.06, 30.0)
        a_fft = fft_flat_spectrum(best_seq, n_iters=400, time_budget=alloc, relaxation=0.4)
        obj = compute_obj(a_fft)
        print(f"  FFT: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fft.copy()

    # Phase 6: Multi-n exploration
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        500, 700, 900, 1000, 1100, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and abs(n - n_best) > 50]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc1 = min(remaining * 0.12, 25.0)
        a_opt = lp_alternating_fast(a_try, n_iters=300, time_budget=alloc1)

        alloc2 = min(remaining * 0.08, 15.0)
        a_opt = mirror_descent(a_opt, n_iters=2000, time_budget=alloc2)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try

    # Phase 7: Random restarts
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    n_restarts = min(6, max(2, int(remaining / 50)))
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        init_type = r % 5
        if init_type == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 1:
            n_r = int(n_best + rng.integers(-150, 250))
            n_r = max(200, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.06, n_r))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            a_r = random_init(n_best, rng)
        elif init_type == 3:
            # Anti-symmetric perturbation
            a_r = best_seq.copy()
            mid = n_best // 2
            a_r[:mid] *= 1.2
            a_r[mid:] *= 0.85
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            # Random smooth with structure
            n_r = int(n_best * rng.uniform(0.8, 1.3))
            n_r = max(200, min(1500, n_r))
            a_r = random_init(n_r, rng)

        # Optimize with LP + MD
        alloc = min(remaining * 0.2, 30.0)
        a_opt = lp_alternating_fast(a_r, n_iters=400, time_budget=alloc)
        alloc2 = min(remaining * 0.12, 15.0)
        a_opt = mirror_descent(a_opt, n_iters=2000, time_budget=alloc2)

        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(a_opt)

    # Phase 8: Final intensive polish
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 60.0)
        a_final = lp_alternating_fast(best_seq, n_iters=1000, time_budget=alloc, verbose=True)
        obj = compute_obj(a_final)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.25, 40.0)
            a_final = mirror_descent(best_seq, n_iters=3000, time_budget=alloc)
            obj = compute_obj(a_final)
            print(f"  Final MD: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.2, 25.0)
            a_final = peak_shaving(best_seq, n_iters=500, time_budget=alloc)
            obj = compute_obj(a_final)
            print(f"  Final PS: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 30.0)
            a_final = lp_alternating_fast(best_seq, n_iters=500, time_budget=alloc)
            obj = compute_obj(a_final)
            print(f"  Final LP2: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
