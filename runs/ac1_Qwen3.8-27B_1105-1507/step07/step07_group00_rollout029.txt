def logsumexp_gradient(f, time_budget, verbose=True):
    """Smooth logsumexp-based gradient descent with momentum."""
    n = len(f)
    f = np.maximum(f, 1e-15).copy()
    f = f / f.sum()

    best_f = f.copy()
    best_obj = compute_obj(f)
    start = time.monotonic()

    t_init = 2.0 / n
    t_min = 0.02 / n
    lr = 1.0 / n
    momentum = 0.0
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(f)

        temp = max(t_min, t_init * (1.0 - it / max(1, it + 2000)) + t_min)
        scaled = conv / temp
        scaled -= np.max(scaled)
        weights = np.exp(scaled)
        weights /= weights.sum()

        g = np.zeros(n)
        top_idx = np.argsort(conv)[::-1][:min(30, 2 * n - 1)]
        for k in top_idx:
            k = int(k)
            w = weights[k]
            if w < 1e-8:
                continue
            for j in range(n):
                jm = k - j
                jp = j - k
                grad = 0.0
                if 0 <= jm < n:
                    grad += f[jm]
                if 0 <= jp < n:
                    grad += f[jp]
                g[j] += w * grad

        v = momentum * v + lr * g
        f_new = f - v
        f_new = np.maximum(f_new, 1e-15)
        s_new = float(np.sum(f_new))
        if s_new < 1e-10:
            f_new = f.copy()
            stall += 1
            it += 1
            continue
        f_new = f_new / s_new

        obj_new = compute_obj(f_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stall = 0
            momentum = min(0.95, momentum + 0.005)
        else:
            stall += 1
            if stall > 100:
                lr = max(t_min * 0.1, lr * 0.5)

        it += 1

def coordinate_descent(f, time_budget, verbose=True):
    """Targeted coordinate descent: move mass from peak contributors to reduce peak."""
    n = len(f)
    f = np.maximum(f, 1e-15).copy()

    best_obj = compute_obj(f)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(f)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        improved = False

        # Find contributing pairs (j, k_star - j)
        contribs = []
        for j in range(n):
            jm = k_star - j
            if 0 <= jm < n:
                contribs.append((f[j] * f[jm], j, jm))
        contribs.sort(reverse=True)

        # Non-contributing positions
        contrib_set = set()
        for _, j, jm in contribs:
            contrib_set.add(j)
            contrib_set.add(jm)
        non_contrib = [i for i in range(n) if i not in contrib_set]

        for contrib_val, j, jm in contribs[:30]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < 1e-12:
                break

            target_list = non_contrib[:8] if non_contrib else [i for i in range(n) if i != j][:8]

            for target in target_list:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                    delta = frac * f[j]
                    if delta >= f[j]:
                        continue
                    # Move mass from j to target
                    f[j] -= delta
                    f[target] += delta
                    new_conv = compute_conv(f)
                    new_peak = float(np.max(new_conv))
                    new_obj = 2.0 * n * new_peak
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_f = f.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        f[j] += delta
                        f[target] -= delta
                        break

                break

        if not improved:
            stall += 1
            if stall > 50:
                # Try random perturbations

    return best_f, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def u_shaped_init(n, alpha=2.0):
    """Generate a U-shaped (edges-heavy) initialization."""
    x = np.linspace(0, 1, n)
    a = x**alpha + (1 - x)**alpha + 0.01
    return a / a.sum()


def bimodal_init(n, sep=None, width=None):
    """Generate a bimodal initialization."""
    if sep is None:
        sep = n // 3
    if width is None:
        width = n // 12
    a = np.zeros(n)
    c1 = n // 6
    c2 = n - c1
    x = np.arange(n)
    a += np.exp(-(x - c1)**2 / (2 * width**2))
    a += np.exp(-(x - c2)**2 / (2 * width**2))
    a += 1e-3
    return a / a.sum()

    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_f = parent.copy()
                best_n = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_f is None:
        best_n = 800
        best_f = u_shaped_init(best_n, alpha=2.5)
        best_score = compute_obj(best_f)
        print(f"U-shaped init (n={best_n}): obj={best_score:.8f}")

    # Phase 1: Active-set LP alternating on parent (main workhorse)
    print(f"\n=== Phase 1: Active-Set LP (n={best_n}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        f_lp, obj_lp = active_set_lp_alternating(best_f, time_budget=p1_budget, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_f = f_lp.copy()
            best_n = len(f_lp)
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # Phase 2: Logsumexp gradient polish
    print(f"\n=== Phase 2: LogSumExp Gradient (n={best_n}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 30:
        f_ls, obj_ls = logsumexp_gradient(best_f, time_budget=p2_budget, verbose=True)
        if obj_ls < best_score:
            best_score = obj_ls
            best_f = f_ls.copy()
            best_n = len(f_ls)
            print(f"  ** LS: obj={best_score:.8f} **")
        else:
            print(f"  LS: obj={obj_ls:.8f} (no improvement)")

    # Phase 3: Coordinate descent
    print(f"\n=== Phase 3: Coordinate Descent (n={best_n}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 15:
        f_cd, obj_cd = coordinate_descent(best_f, time_budget=p3_budget, verbose=True)
        if obj_cd < best_score:
            best_score = obj_cd
            best_f = f_cd.copy()
            best_n = len(f_cd)
            print(f"  ** CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # Phase 4: Second LP round (on improved solution)
    print(f"\n=== Phase 4: LP Round 2 (n={best_n}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 250.0)
    if p4_budget > 20:
        f_lp2, obj_lp2 = active_set_lp_alternating(best_f, time_budget=p4_budget, verbose=True)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_f = f_lp2.copy()
            best_n = len(f_lp2)
            print(f"  ** LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.8f} (no improvement)")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = best_n
    n_options = sorted(set([
        n_current,
        n_current + 100, n_current + 200, n_current + 300,
        max(400, n_current - 100),
        600, 1000, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and n != n_current]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        # Resample or generate fresh
        if abs(n_try - n_current) / max(n_current, 1) < 0.3:
            a_try = resample_sequence(best_f, n_try)
        else:
            # Blend resampled with a structured init
            a_resampled = resample_sequence(best_f, n_try)
            a_struct = u_shaped_init(n_try, alpha=2.0)
            a_try = 0.7 * a_resampled + 0.3 * a_struct
            a_try = np.maximum(a_try, 1e-15)

        a_try = a_try / a_try.sum()
        obj_try = compute_obj(a_try)
        print(f"  n={n_try}: init obj={obj_try:.8f}")

        # LP alternating
        alloc_lp = min(remaining * 0.3, 150.0)
        if alloc_lp > 20:
            f_opt, obj_opt = active_set_lp_alternating(a_try, time_budget=alloc_lp, verbose=False)
            if obj_opt < obj_try:
                a_try = f_opt
                obj_try = obj_opt

        remaining = deadline - time.monotonic()
        if remaining > 30 and obj_try < best_score:
            # Quick CD polish
            alloc_cd = min(remaining * 0.2, 40.0)
            f_cd2, obj_cd2 = coordinate_descent(a_try, time_budget=alloc_cd, verbose=False)
            if obj_cd2 < obj_try:
                a_try = f_cd2
                obj_try = obj_cd2

        if obj_try < best_score:
            best_score = obj_try
            best_f = a_try.copy()
            best_n = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish (n={best_n}) ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        f_final, obj_final = active_set_lp_alternating(best_f, time_budget=min(remaining * 0.4, 100.0), verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_f = f_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        f_cd3, obj_cd3 = coordinate_descent(best_f, time_budget=min(remaining * 0.4, 60.0), verbose=True)
        if obj_cd3 < best_score:
            best_score = obj_cd3
            best_f = f_cd3.copy()
            print(f"  ** Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        f_ls2, obj_ls2 = logsumexp_gradient(best_f, time_budget=min(remaining * 0.5, 50.0), verbose=False)
        if obj_ls2 < best_score:
            best_score = obj_ls2
            best_f = f_ls2.copy()
            print(f"  ** Final LS: obj={best_score:.8f} **")

    # Final validation
    if best_f is None:
        best_f = np.ones(200)
        best_score = compute_obj(best_f)

    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
