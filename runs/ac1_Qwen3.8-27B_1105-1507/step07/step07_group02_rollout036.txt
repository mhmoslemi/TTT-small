Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences of length n.
Strategy: Subgradient descent with adaptive step size, LP active-set refinement,
multi-start with diverse initializations (structured, random, Fourier-inspired),
and multi-n exploration. Key insight: non-negative sequences can have convolution
peaks away from center, enabling scores well below 2.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_score_internal(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(a, target_sum=1.0):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-30:
        return np.full(n, target_sum / n)
    return a * (target_sum / s)


def subgradient_descent(a, time_budget=60.0, verbose=False):
    """
    Subgradient descent on max(conv(a,a)) with a>=0, sum(a)=1.
    Uses argmax-based subgradient with adaptive step size.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = compute_score_internal(a)
    start = time.monotonic()
    lr = 4.0 / n
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Find all positions within 1% of max (for more robust gradient)
        threshold = max_val * 0.999
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        # Compute subgradient (average over active positions)
        grad = np.zeros(n)
        for d in active:
            idx = d - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] += 2.0 * a[idx[valid]]
        grad /= float(len(active))

        # Center to preserve sum
        grad -= np.mean(grad)

        # Adaptive step: try several step sizes
        best_lr = lr * 0.3
        best_obj = np.inf
        for trial_lr in [lr * 0.25, lr * 0.5, lr * 0.75, lr, lr * 1.5]:
            a_try = project_simplex(a - trial_lr * grad)
            obj_try = compute_score_internal(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_lr = trial_lr

        if best_obj < best_score - 1e-13:
            best_score = best_obj
            best_a = project_simplex(a - best_lr * grad)
            a = best_a
            lr = min(best_lr * 1.05, 50.0 / n)
            stall = 0
        else:
            a = project_simplex(a - best_lr * grad)
            stall += 1
            lr *= 0.9
            if stall > 30:
                rng = np.random.default_rng(it * 7919 + n)
                scale = 0.01 + 0.01 * min(stall // 30, 5)

                a = project_simplex(a)

                best_a = a.copy()
                stall = 0
                lr = 4.0 / n

        it += 1
        if verbose and it % 200 == 0:
            print(f"        subgrad it{it}: score={best_score:.8f}, lr={lr:.4f}")

def smooth_max_descent(a, time_budget=60.0, alpha_init=3.0, verbose=False):

        # = 2 * (rev(a) * weights)[n-1+i]

            size <<= 1

        fw = rfft(weights, size)
        corr = irfft(fa * fw, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        grad -= np.mean(grad)

        # Line search
        best_lr_use = lr
        best_obj = compute_score_internal(project_simplex(a - lr * grad))
        for trial in [lr * 0.5, lr * 0.25, lr * 1.5]:
            obj_t = compute_score_internal(project_simplex(a - trial * grad))
            if obj_t < best_obj:
                best_obj = obj_t
                best_lr_use = trial

        if best_obj < best_score - 1e-13:
            best_score = best_obj
            best_a = project_simplex(a - best_lr_use * grad)
            a = best_a
            lr = min(best_lr_use * 1.02, 20.0 / n)
            stall = 0
        else:
            stall += 1
            lr *= 0.95
            if stall > 40:

                best_score = compute_score_internal(a)
                best_a = a.copy()
                stall = 0
                lr = 2.0 / n
                alpha = max(1.0, alpha * 0.8)

        # Slowly increase alpha for sharper max approximation
        if it % 100 == 0:
            alpha = min(alpha * 1.2, 30.0)

        it += 1
        if verbose and it % 200 == 0:
            print(f"        smooth it{it}: score={best_score:.8f}, alpha={alpha:.1f}, lr={lr:.4f}")

    return best_a

def fourier_shaping(n, rng):
    """
    Generate a sequence designed in the frequency domain to have
    relatively flat autocorrelation.
    """
    # Create a random complex spectrum with flat magnitude
    size = 1
    while size < 2 * n:
        size <<= 1
    magnitude = np.ones(size // 2 + 1)
    # Add some random variation to avoid exact symmetry
    magnitude *= (1.0 + 0.3 * rng.standard_normal(size // 2 + 1))
    magnitude = np.abs(magnitude) + 0.1
    phase = rng.uniform(-np.pi, np.pi, size // 2 + 1)
    spec = magnitude * np.exp(1j * phase)
    # Make it Hermitian for real output
    full_spec = np.zeros(size, dtype=np.complex128)
    full_spec[:size // 2 + 1] = spec
    for k in range(1, size // 2):
        full_spec[size - k] = np.conj(spec[k])
    a = np.real(irfft(full_spec, size)[:n])
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def structured_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        # Near-constant with small noise
        a = np.ones(n) + rng.normal(0, 0.1, n)
    elif mode == 1:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.05
    elif mode == 3:
        # Random
        a = rng.uniform(0, 1, n)
    elif mode == 4:
        # Triangular (zero at ends)
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x) + 0.01
    elif mode == 5:
        # Inverse U (high at ends, low in middle)
        x = np.linspace(-1, 1, n)
        a = x ** 2 + 0.1
    elif mode == 6:
        # Sine modulation
        x = np.arange(n)
        freq = rng.uniform(1, 5)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
    else:
        # Logistic S-curve
        x = np.linspace(-5, 5, n)
        a = 1.0 / (1.0 + np.exp(-x))
    a = project_simplex(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
