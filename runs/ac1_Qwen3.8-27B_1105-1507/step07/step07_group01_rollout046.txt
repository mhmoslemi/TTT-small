from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def simplex_project(v):
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def make_symmetric(a):
    a = a.copy()
    a = (a + a[::-1]) / 2.0
    return np.maximum(a, 0)

        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 1e-15:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            print(f"  Symmetric LP: obj={obj_sym:.8f}")

    # === Phase 3: Smooth Max Gradient ===
    print("\n=== Phase 3: Smooth Max Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.15, 60.0)
        a_sm = smooth_max_gradient(a, time_budget=alloc, T_init=0.001, T_min=0.00005)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            a = a_sm
            print(f"  Smooth Max: obj={best_score:.8f}")
        else:
            print(f"  Smooth Max: obj={obj_sm:.8f}")

        # Low temperature
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.1, 30.0)
            a_sm2 = smooth_max_gradient(a, time_budget=alloc, T_init=0.0003, T_min=0.00002)
            obj_sm2 = compute_obj(a_sm2)
            if obj_sm2 < best_score:
                best_score = obj_sm2
                best_seq = a_sm2.copy()
                a = a_sm2
                print(f"  Smooth Max (low T): obj={best_score:.8f}")

    # === Phase 4: Chase Peak ===
    print("\n=== Phase 4: Chase Peak ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.08, 25.0)
        a_cp = chase_peak(a, time_budget=alloc)
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            a = a_cp
            print(f"  Chase Peak: obj={best_score:.8f}")
        else:
            print(f"  Chase Peak: obj={obj_cp:.8f}")

    # === Phase 5: Random Restarts ===
    print("\n=== Phase 5: Random Restarts ===")
    n_best = len(best_seq)
    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Different initialization strategies
        n_r = max(400, min(1500, n_best + int(rng.integers(-100, 101))))
        
        if r == 0:
            a_r = np.ones(n_r) / n_r
        elif r == 1:
            a_r = rng.uniform(0.1, 1.0, n_r)
            a_r = a_r / float(np.sum(a_r))
        elif r == 2:
            # Cosine window
            x = np.linspace(0, np.pi, n_r)
            a_r = 0.5 * (1 + np.cos(x))
            a_r = a_r / float(np.sum(a_r))
        elif r == 3:
            # Triangular
            x = np.arange(n_r) / (n_r - 1)
            a_r = 1 - np.abs(2 * x - 1)
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / float(np.sum(a_r))
        elif r == 4:
            # Symmetric random
            half = n_r // 2
            a_half = rng.uniform(0.1, 1.0, half + 1)
            a_r = np.concatenate([a_half, a_half[-2::-1]])
            a_r = a_r / float(np.sum(a_r))
        else:
            # Geometric taper
            a_r = np.exp(-0.5 * (np.arange(n_r) / n_r - 0.5) ** 2 * 4)
            a_r = a_r / float(np.sum(a_r))

        alloc_r = min(remaining * 0.12, 40.0)
        a_opt = lp_alternating(a_r, time_budget=alloc_r, symmetric=False)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            a = a_opt
            print(f"  Restart {r} (n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}): obj={obj_r:.8f}")

    # === Phase 6: Final Combined Polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        # LP
        alloc = min(remaining * 0.25, 60.0)
        a_lp = lp_alternating(a_final, time_budget=alloc, symmetric=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp
            print(f"  Final LP: obj={best_score:.8f}")

        # Symmetric LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sym = make_symmetric(a_final)
            a_sym = a_sym / float(np.sum(a_sym))
            alloc = min(remaining * 0.15, 30.0)
            a_sym = lp_alternating(a_sym, time_budget=alloc, symmetric=True)
            obj_sym = compute_obj(a_sym)
            if obj_sym < best_score:
                best_score = obj_sym
                best_seq = a_sym.copy()
                a_final = a_sym
                print(f"  Final Sym LP: obj={best_score:.8f}")

        # Smooth max
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.2, 20.0)
            a_sm = smooth_max_gradient(a_final, time_budget=alloc, T_init=0.0005, T_min=0.00003)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                a_final = a_sm
                print(f"  Final Smooth: obj={best_score:.8f}")

        # Chase peak
        remaining = deadline - time.monotonic()
        if remaining > 8:
            alloc = min(remaining * 0.15, 15.0)
            a_cp = chase_peak(a_final, time_budget=alloc)
            obj_cp = compute_obj(a_cp)
            if obj_cp < best_score:
                best_score = obj_cp
                best_seq = a_cp.copy()
                print(f"  Final Chase: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
