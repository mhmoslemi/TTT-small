def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lbfgs_smooth_optimize(a_init, T=0.002, max_iter=500, time_budget=60.0):
    """
    Optimize using L-BFGS-B on the smooth max objective.
    Works in log-space to handle non-negativity: a[i] = exp(b[i]) / sum(exp(b)).
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    # Work in log-parameterization
    b = np.log(a)

    def objective_and_grad(b_vec):
        # a = softmax(b)
        b_shifted = b_vec - np.max(b_vec)
        exp_b = np.exp(b_shifted)
        a_vec = exp_b / np.sum(exp_b)
        
        conv = compute_conv_fft(a_vec)
        smax, weights = smooth_max_logsumexp(conv, T)
        
        # Objective: 2*n*smax
        obj = 2.0 * n * smax
        
        # Gradient w.r.t. b:
        # da_vec[i]/db[j] = a_vec[i]*(delta_ij - a_vec[j])
        # d(obj)/da[i] = 2*n * (2 * correlate(weights, a_vec)[i])  [from above]
        # d(obj)/db[j] = sum_i d(obj)/da[i] * da[i]/db[j]
        #               = sum_i 2*n*2*corr[i] * a[i]*(delta_ij - a[j])
        #               = 2*n*2*corr[j]*a[j] - 2*n*2*a[j]*sum_i corr[i]*a[i]
        
        corr = np.correlate(weights, a_vec, mode='valid')
        d_obj_da = 2.0 * n * 2.0 * corr
        s_corr = np.sum(corr * a_vec)
        d_obj_db = d_obj_da * a_vec - 2.0 * n * 2.0 * s_corr * a_vec
        # Simplify: d_obj_db = 2*n*(2*corr*a - 2*s_corr*a) = 4*n*a*(corr - s_corr)
        
        return obj, d_obj_db

    start = time.monotonic()
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()

    for attempt in range(3):
        if time.monotonic() - start > time_budget:
            break
        try:
            # Use a slightly larger T for the first attempt for smoother landscape
            T_use = T * (1 + attempt * 0.5)
            res = minimize(
                lambda bv: objective_and_grad(bv)[0],
                b.copy(),
                jac=lambda bv: objective_and_grad(bv)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8,
                         'maxfun': max_iter * 3}
            )
            b_opt = res.x
            b_shifted = b_opt - np.max(b_opt)
            exp_b = np.exp(b_shifted)
            a_new = exp_b / np.sum(exp_b)
            obj_new = compute_obj_normalized(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        except Exception as e:
            pass

    return best_a


def coordinate_descent(a_init, time_budget=30.0, max_passes=50):
    """
    Coordinate descent: for each coordinate i, find optimal value via 1D search.
    Since objective is scale-invariant, we work without sum constraint.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for passe in range(max_passes):
        if time.monotonic() - start > time_budget:
            break

        # Random order for each pass
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Compute convex values w.r.t. a[i] = x
            # conv[k] = base_k + c_k * x for k != 2i
            # conv[2i] = base_2i + x^2
            # where base_k = conv[k] - (current contribution of a[i])
            # and c_k = 2*a[k-i] if 0<=k-i<n (for auto-conv, the coefficient of a[i])
            
            ai = a[i]
            if ai < 1e-15:
                # Try small positive values
                for x_try in [1e-4, 1e-3, 1e-2, 0.01, 0.02]:
                    a_try = a.copy()
                    a_try[i] = x_try
                    obj_try = compute_obj_normalized(a_try)
                    if obj_try < best_obj:
                        a = a_try.copy()
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                continue

            # Coefficient c_k for each k: 2*a[k-i] if 0<=k-i<n, else 0
            # For k in [0, 2n-2]: k-i in [0, n-1] means k in [i, i+n-1]
            c = np.zeros(2 * n - 1)
            lo = i
            hi = min(i + n - 1, 2 * n - 2)
            if lo <= hi:
                c[lo:hi+1] = 2.0 * a[lo-i:hi-i+1]
            # Special case k=2i: the coefficient is 2*a[i] (from a[i]^2 derivative)
            # But in our formulation, conv[2i] has x^2, not c* x
            # So we need: conv[2i](x) = base_2i + x^2
            # base_2i = conv[2i] - a[i]^2  (remove the quadratic part)
            # For k != 2i: conv[k](x) = base_k + c_k * x where base_k = conv[k] - c_k * a[i]
            
            # Compute base values (current conv minus contribution)
            conv_current = compute_conv_fft(a)
            base = conv_current.copy()
            # Remove linear contribution for k != 2i
            mask = np.ones(2*n-1, dtype=bool)
            if 2*i < 2*n-1:
                mask[2*i] = False
            base[mask] -= c[mask] * ai
            # For k=2i: conv[2i] = base_2i + x^2, so base_2i = conv[2i] - a[i]^2
            if 2*i < 2*n-1:
                base[2*i] = conv_current[2*i] - ai * ai
                c[2*i] = 0.0

            # Now minimize max(base[k] + c[k]*x) for k!=2i, and base[2i] + x^2
            # subject to x >= 0
            # This is a convex 1D problem. Use ternary/golden section search.
            
            # Bracket the minimum
            # Try a range of x values
            x_lo = 0.0
            x_hi = max(ai * 5.0, 0.01)
            
            # Evaluate at several points
            candidate_xs = np.concatenate([
                np.linspace(0, x_hi, 20),
                np.array([ai * 0.1, ai * 0.5, ai, ai * 1.5, ai * 2, ai * 3])
            ])
            candidate_xs = candidate_xs[candidate_xs >= 0]
            candidate_xs = np.unique(candidate_xs)
            
            best_x = ai
            best_x_obj = np.inf
            for x in candidate_xs:
                # Compute max conv value
                conv_vals = base + c * x
                if 2*i < 2*n-1:
                    conv_vals[2*i] = base[2*i] + x * x
                max_val = np.max(conv_vals)
                obj_val = 2.0 * n * max_val  # (scale invariant, sum doesn't matter for comparison)
                if obj_val < best_x_obj:
                    best_x_obj = obj_val
                    best_x = x

            if best_x != ai and best_x_obj < best_obj * (1 - 1e-10):
                a[i] = best_x
                # Renormalize
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                obj_check = compute_obj_normalized(a)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()
                    improved = True

        if not improved and passe > 10:
            break

    return best_a


def lp_alternating_fast(a_init, time_budget=45.0):
    """
    LP alternating: solve min M s.t. conv(a,g)<=M, sum(g)=1, g>=0.
    Use scipy linprog for speed.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)

    # Build constraint matrix: A (2n-1 x n) where A[k,i] = a[k-i] if 0<=k-i<n
    A_mat = np.zeros((2 * n - 1, n))
    for i in range(n):
        lo = i
        hi = min(i + n - 1, 2 * n - 2)
        if lo <= hi:
            A_mat[lo:hi+1, i] = a[lo-i:hi-i+1]

    max_iters = 200
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP: min M s.t. A@z <= M, sum(z)=1, z>=0
        # Variables: z[0..n-1], M (index n)
        c_vec = np.zeros(n + 1)
        c_vec[n] = 1.0  # minimize M
        
        # A_ub: A@z - M <= 0 for each row
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A_mat
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        
        # Equality: sum(z) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * (n + 1)
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5.0})
            if res.status != 0 or res.x is None:
                break
            z = np.maximum(res.x[:n], 0)
            s_z = float(np.sum(z))
            if s_z < 1e-10:
                break
            z = z / s_z
        except Exception:
            break

        # Line search for blending
        improved = False
        for t in [0.3, 0.5, 0.7, 0.1, 0.2, 0.4, 0.6, 0.8, 0.9, 1.0, 0.05]:
            a_try = (1 - t) * a + t * z
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            # Perturb
            if it > 20:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
            if it > 50:
                break

    return best_a


def diverse_init(n, rng, mode):
    """Generate diverse initial sequences."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        return a / np.sum(a)
    elif mode == 2:
        # Gaussian centered
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
        return a / np.sum(a)
    elif mode == 3:
        # Two peaks
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n*0.3) / (n/8))**2) + np.exp(-0.5 * ((x - n*0.7) / (n/8))**2)
        return a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x/2)**2 + 0.1
        return a / np.sum(a)
    elif mode == 5:
        # Flat middle, tapered edges
        x = np.arange(n) / (n - 1)
        a = np.ones(n)
        taper = max(1, n // 8)
        for i in range(taper):
            a[i] = (i + 1) / (taper + 1)
            a[n-1-i] = a[i]
        return a / np.sum(a)
    elif mode == 6:
        # Low autocorrelation inspired: alternating pattern with noise
        a = np.ones(n) * (1 + 0.3 * np.sin(2*np.pi*3*x/(n-1)))
        a += rng.normal(0, 0.1, n)
        a = np.maximum(a, 0)
        return a / np.sum(a)
    elif mode == 7:
        # Wide Gaussian
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n/2) / (n/2.5))**2)
        return a / np.sum(a)
    elif mode == 8:
        # Inverse U (valley in middle)
        x = np.arange(n) / (n - 1)
        a = 0.3 + 0.7 * np.abs(2*x - 1)
        return a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
        return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
