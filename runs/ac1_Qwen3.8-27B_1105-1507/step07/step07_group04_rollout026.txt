"""
Optimized step function search using:
1. Soft-max gradient descent with Adam for smooth optimization
2. LP alternating (HiGHS) for direct max-peak minimization  
3. Multi-length exploration
4. Coordinate descent polish
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:2 * n - 1]
    return conv


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_weights(conv, T):
    """Compute normalized soft-max weights."""
    shifted = T * (conv - np.max(conv))
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / s


def compute_softmax_grad(a, T):
    """
    Gradient of (1/T)*log(sum_k exp(T*b_k)) w.r.t. a.
    b = conv(a,a), so b_k = sum_i a_i * a_{k-i}.
    d(soft_max)/da_j = sum_k w_k * db_k/da_j = sum_k w_k * 2*a_{k-j}
    = 2 * sum_k w_k * a_{k-j} (where 0 <= k-j < n)
    """
    n = len(a)
    conv = compute_conv(a)
    w = softmax_weights(conv, T)
    
    # grad_j = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j]
    #        = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(a, w[j:j+n])
    return grad


def softmax_adam(a, time_budget=60.0, T_start=3.0, T_end=300.0,
                 lr=0.005, verbose=False):
    """Soft-max gradient descent with Adam."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_iter = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    T = T_start
    T_mult = 1.03
    
    while time.monotonic() - start < time_budget:
        t_iter += 1
        grad = compute_softmax_grad(a, T)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        if t_iter % 20 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        if t_iter % 30 == 0:
            T = min(T * T_mult, T_end)
        
        if verbose and t_iter % 500 == 0:
            print(f"    SoftMax+Adam: obj={best_obj:.8f}, T={T:.1f}, iter={t_iter}, "
                  f"elapsed={time.monotonic()-start:.0f}s")
    
    return best_a


def solve_lp_for_g(a):
    """
    Solve: min M s.t. (conv(a,g))_k <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS.
    Returns (g_normalized, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix A_ub: (2n-1) x (n+1)
    # Row k: (a*g)_k - M <= 0
    # (a*g)_k = sum_j a_j * g_{k-j} (with valid indices)
    # For variable g_j, coefficient in row k is a_{k-j} if 0<=k-j<n, else 0
    # For variable M, coefficient is -1
    
    # This matrix is (2n-1) x (n+1)
    # Instead of building the full matrix (which is (2n-1)*(n+1) = O(n^2)),
    # we can use the sparse structure. But for n=800, the dense matrix is 1600x801 ≈ 1.3M entries, which is fine.
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g_j appears in (a*g)_k for k = j, j+1, ..., j+n-1
        # coefficient is a_{k-j} = a_0, a_1, ..., a_{n-1}
        A[j:j+n, j] = a
    
    # Full LP matrix
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    
    return None, None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    consecutive_no_improve = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_for_g(a)
        if g_lp is None:
            # Try perturbing
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            a_pert = a * np.exp(rng.normal(0, 0.02, n))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            g_lp, M_lp = solve_lp_for_g(a_pert)
            if g_lp is None:
                break
        
        # Try different step sizes
        best_t_obj = np.inf
        best_t_val = 1.0
        best_a_local = best_a.copy()
        
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > 5:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                scale = 0.02 + 0.01 * min(consecutive_no_improve // 5, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                consecutive_no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - start
            if int(elapsed * 2) % 5 == 0:  # print ~every 2.5s
                print(f"    LP: obj={best_obj:.8f}, t={best_t_val:.2f}, elapsed={elapsed:.0f}s")
    
    return best_a


def coordinate_descent_polish(a, time_budget=30.0):
    """Coordinate descent polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        k_star = int(np.argmax(conv))
        
        # Try moving mass from peak contributors to others
        # Find which a_i contribute most to b_{k_star}
        # b_{k_star} = sum_i a_i * a_{k_star-i}
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Move from high contribution to low contribution
        sorted_idx = np.argsort(contributions)[::-1]
        for i in sorted_idx[:10]:
            for j in range(n):
                if j == i or contributions[j] > contributions[i] * 0.5:
                    continue
                for frac in [0.05, 0.02, 0.01]:
                    delta = frac * a[i]
                    if delta >= a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    a = a / s
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        a = a / s
                        break
        
        if not improved:
            # Random pair swap
            rng = np.random.default_rng(int(time.monotonic() * 100000) % (2**31))
            for _ in range(5):
                idx = rng.integers(0, n)
                jdx = rng.integers(0, n)
                if idx == jdx:
                    continue
                delta = 0.01 * a[idx]
                a[idx] -= delta
                a[jdx] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a[idx] += delta
                    a[jdx] -= delta
                    continue
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and it > 100:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized step function search combining:
1. Soft-max gradient descent with Adam (smooth landscape exploration)
2. LP alternating via HiGHS (direct max-peak minimization)
3. Targeted perturbations to escape local minima
4. Multi-length exploration
5. Coordinate descent polish
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:2 * n - 1]
    return conv


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_weights(conv, T):
    """Compute normalized soft-max weights for convolution values."""
    shifted = T * (conv - np.max(conv))
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / s


def compute_softmax_grad(a, T):
    """
    Gradient of (1/T)*log(sum_k exp(T*b_k)) w.r.t. a.
    b = conv(a,a), db_k/da_j = 2*a_{k-j} (when 0<=k-j<n).
    grad_j = 2 * sum_k w_k * a_{k-j} = 2 * sum_m w[j+m] * a[m]
    """
    n = len(a)
    conv = compute_conv(a)
    w = softmax_weights(conv, T)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(a, w[j:j + n])
    return grad


def softmax_adam(a, time_budget=60.0, T_start=3.0, T_end=300.0,
                 lr=0.005, verbose=False):
    """Soft-max gradient descent with Adam optimizer."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    t_iter = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    T = T_start
    T_mult = 1.03

    while time.monotonic() - start < time_budget:
        t_iter += 1
        grad = compute_softmax_grad(a, T)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        if t_iter % 15 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if t_iter % 25 == 0:
            T = min(T * T_mult, T_end)

        if verbose and t_iter % 400 == 0:
            print(f"    SoftMax+Adam: obj={best_obj:.8f}, T={T:.1f}, iter={t_iter}, "
                  f"elapsed={time.monotonic() - start:.0f}s")

    return best_a


def build_lp_matrices(a):
    """Build matrices for the LP: min M s.t. A_ub @ [g;M] <= 0, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm

    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * n + [(None, None)]
    return A_ub, b_ub, A_eq, b_eq, c, bounds


def solve_lp_for_g(a):
    """
    Solve: min M s.t. (conv(a,g))_k <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS. Returns (g_normalized, M) or (None, None).
    """
    n = len(a)
    if n < 2:
        return np.ones(1) / 1.0, 1.0

    A_ub, b_ub, A_eq, b_eq, c, bounds = build_lp_matrices(a)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def targeted_perturbation(a, scale=0.03):
    """
    Targeted perturbation: move mass from high-autocorrelation contributors
    to low-contribution positions.
    """
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    # Find contributors to the peak
    contributions = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            contributions[i] = a[i] * a[j]
        j2 = 2 * k_star - i
        if 0 <= j2 < n:
            contributions[i] += a[i] * a[j2]

    # Identify hot and cold spots
    hot = np.argsort(contributions)[::-1][:max(3, n // 20)]
    cold = np.argsort(contributions)[:max(3, n // 20)]

    a_new = a.copy()
    total_shift = 0.0
    for i in hot:
        for j in cold:
            if i == j:
                continue
            delta = scale * 0.1 * a[i]
            if delta < a[i] * 0.5:
                a_new[i] -= delta
                a_new[j] += delta
                total_shift += delta
            if total_shift > scale * np.sum(a):
                break
        if total_shift > scale * np.sum(a):
            break

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating with adaptive step size and targeted perturbations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    perturbation_scale = 0.02
    iter_count = 0

    while time.monotonic() - start < time_budget:
        iter_count += 1
        g_lp, M_lp = solve_lp_for_g(a)
        if g_lp is None:
            # Try with perturbed a
            a_pert = a * np.exp(np.random.default_rng(iter_count).normal(0, 0.01, n))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            g_lp, M_lp = solve_lp_for_g(a_pert)
            if g_lp is None:
                break

        # Line search: try decreasing step sizes
        best_t_obj = np.inf
        best_t_val = 1.0
        best_a_local = best_a.copy()

        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            no_improve = 0
            perturbation_scale = max(0.01, perturbation_scale * 0.95)
        else:
            no_improve += 1
            if no_improve >= 8:
                # Apply targeted perturbation
                a = targeted_perturbation(best_a, perturbation_scale)
                perturbation_scale = min(0.1, perturbation_scale * 1.3)
                no_improve = 0

                # Also try random perturbation if still stuck
                if compute_obj(a) >= best_obj:
                    rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                    a = best_a * np.exp(rng.normal(0, perturbation_scale, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP iter={iter_count}, obj={best_obj:.8f}, "
                  f"t={best_t_val:.2f}, elapsed={elapsed:.0f}s")

def coordinate_descent_polish(a, time_budget=30.0):
    """Coordinate descent: adjust pairs to reduce the peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    stall = 0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        k_star = int(np.argmax(conv))

        # Compute contributions to peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Try moving from high to low contributors
        top_idx = np.argsort(contrib)[::-1][:8]
        bot_idx = np.argsort(contrib)[:8]

        for i in top_idx:
            if improved:
                break
            for j in bot_idx:
                if i == j:
                    continue
                for frac in [0.03, 0.01, 0.005]:
                    delta = frac * a[i]
                    if delta >= a[i] * 0.8:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    a = a / s
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a = best_a * 0 + a  # restore
                        # Actually just undo
                        a[i] += delta
                        a[j] -= delta
                        a = a / s
                        break

        if not improved:
            # Random pair adjustment
            rng = np.random.default_rng(int(time.monotonic() * 100000 + it) % (2**31))
            for _ in range(3):
                idx = int(rng.integers(0, n))
                jdx = int(rng.integers(0, n))
                if idx == jdx:
                    continue
                delta = 0.005 * a[idx]
                a[idx] -= delta
                a[jdx] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a[idx] += delta
                    a[jdx] -= delta
                    continue
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[idx] += delta
                    a[jdx] -= delta
                    a = a / s

        if not improved:
            stall += 1
        else:
            stall = 0

        if stall > 50:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        alloc1 = min(lp_total * 0.5, remaining * 0.5)
        a_lp = lp_alternating(a, time_budget=alloc1, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  LP Round 1: obj={best_score:.8f} **")
        else:
            print(f"  LP Round 1: obj={obj_lp:.8f}")

        # Round 2: continue from best
        remaining = deadline - time.monotonic()
        if remaining > 60:
            alloc2 = min(lp_total * 0.3, remaining * 0.5)
            a_lp2 = lp_alternating(a, time_budget=alloc2, verbose=True)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                a = a_lp2.copy()
                print(f"  LP Round 2: obj={best_score:.8f} **")
            else:
                print(f"  LP Round 2: obj={obj_lp2:.8f}")

        # Round 3: short refinement
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc3 = min(lp_total * 0.2, remaining * 0.5)
            a_lp3 = lp_alternating(a, time_budget=alloc3)
            obj_lp3 = compute_obj(a_lp3)
            if obj_lp3 < best_score:
                best_score = obj_lp3
                best_seq = a_lp3.copy()
                a = a_lp3.copy()
                print(f"  LP Round 3: obj={best_score:.8f} **")
            else:
                print(f"  LP Round 3: obj={obj_lp3:.8f}")

    # === Phase 3: Soft-max with high T (80s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sm_budget = min(80.0, remaining * 0.08)
        print(f"\n=== Phase 3: Soft-max high-T (budget={sm_budget:.0f}s) ===")
        a_sm2 = softmax_adam(a, time_budget=sm_budget, T_start=20.0, T_end=500.0,
                            lr=0.002, verbose=True)
        obj_sm2 = compute_obj(a_sm2)
        if obj_sm2 < best_score:
            best_score = obj_sm2
            best_seq = a_sm2.copy()
            a = a_sm2.copy()
            print(f"  SoftMax2: obj={best_score:.8f} **")
        else:
            print(f"  SoftMax2: obj={obj_sm2:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Multi-n Exploration ===")
        n_parent = len(best_seq)
        n_options = sorted(set([
            max(200, n_parent - 200),
            max(200, n_parent - 100),
            n_parent,
            n_parent + 100,
            n_parent + 200,
            n_parent + 400,
        ]))
        n_options = [n for n in n_options if 200 <= n <= 2500]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)

            alloc_lp = min(25.0, remaining * 0.3)
            a_opt = lp_alternating(a_try, time_budget=alloc_lp)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 100.0)
        print(f"\n=== Phase 5: Final LP (budget={alloc:.0f}s) ===")
        a_final = lp_alternating(a, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 6: Coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 50.0)
        print(f"\n=== Phase 6: Coordinate Descent (budget={alloc:.0f}s) ===")
        a_cd = coordinate_descent_polish(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 7: Final LP burst ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_last = lp_alternating(best_seq, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"\n  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
