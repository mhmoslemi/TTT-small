    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def get_peak_lags(a, n_peaks=5):
    """Return indices of the top n_peaks lags in the convolution."""
    conv = compute_conv(a)
    idx = np.argsort(conv)[::-1][:n_peaks]

def subgradient_step(a, k_star, lr):
    """One subgradient step for minimizing max_k (a*a)[k]."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    # Scale: objective is 2n * max, so grad of objective is 2n * 2 * a_rev_shifted
    # But we're on the simplex, so project
    grad_proj = grad - np.mean(grad)
    return a - lr * grad_proj

def momentum_subgradient(a, time_budget=60.0):

        return a

            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        obj = 2.0 * n * conv[k_star]

                grad[j] = 2.0 * a[idx]
        grad_proj = grad - np.mean(grad)
        
        # Nesterov: v = momentum * v + grad_proj; a = a - lr * v
        v = momentum * v + grad_proj

        if s_new > 1e-10:

        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Try cvxpy solvers

            M = cp.Variable()

            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))

def solve_lp_smoothing(a, eps=0.1):
    """Solve LP with smoothing: min M + eps * sum((A@g)_+) to encourage flatness."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    for solver in [cp.CLARABEL, cp.ECOS]:

            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            conv_vals = A @ g

            prob = cp.Problem(cp.Minimize(M + eps * cp.sum(excess)), constraints)

        conv = compute_conv(a)

            return obj_grad(t, tau)[0]

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    """Solve LP for best direction d: min M' s.t. (a*d)[k] <= M', sum(d)=0, d>=-a"""
    n = len(a)
    a = a / np.sum(a)
    
    # Build matrix for (a*d)[k] = sum_j A[k,j] * d[j]
    # A[k,j] = a[k-j] if 0<=k-j<n, else 0 (this is the same as the convolution matrix)
    A = build_conv_matrix(a)  # A @ d = conv(a, d)

    # Constraints: A @ d - M' <= 0 (for all k), sum(d) = 0, d >= -a

This is a key improvement! The direction LP directly optimizes the linearized change in the convolution peak, which should give better directions than the previous approach.

Let me also think about a second-order correction. After finding the direction d, the actual new convolution is:
(a+d)*(a+d) = a*a + a*d + d*a + d*d = a*a + 2*(a*d) + d*d (by symmetry of convolution)

Wait, convolution is commutative, so a*d = d*a. So (a+d)*(a+d) = a*a + 2*(a*d) + d*d.

The peak of the new convolution is max_k [(a*a)[k] + 2*(a*d)[k] + (d*d)[k]].

For small d, the d*d term is negligible. But for larger steps, it matters. We can account for it by doing a line search on the step size.

Let me now write the complete algorithm:

1. **Direction LP**: Find d that minimizes the linearized peak increase
2. **Line search**: Find the best step size t in (0,1] such that a_new = a + t*d minimizes the actual objective
3. **Iterate**: Repeat until convergence or time runs out
4. **Escape**: When stuck, add perturbation or use subgradient steps

This should be more effective than the previous approach because:
- The direction is optimized for the actual objective (not a proxy)
- The line search ensures we make the best possible step in that direction
- We're working directly on the simplex (no rescaling issues)

Let me also add a "flatness" regularization: instead of just minimizing the peak, minimize the peak + small * (variance of convolution / peak). This encourages flatness beyond just the maximum.

Actually, for the LP, we can't easily add variance (it's quadratic). But we can add a penalty on the sum of the top few peaks:
- min M' + eps * sum of top-k values of (a*d)

This can be formulated as an LP with auxiliary variables.

        if s_new > 1e-10:
            a = a_new / s_new
        lr = max(lr * 0.9995, 1e-5)
        momentum = min(momentum + 0.0005, 0.98)

    return best_a


def smoothed_lbfgs(a_init, time_budget=60.0):
    """Smoothed L-BFGS with log-softmax parameterization."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    for tau in [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0]:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        def obj_func(t):
            t_shift = t - np.max(t)
            a = np.exp(t_shift)
            a = a / np.sum(a)
            conv = compute_conv(a)
            cm = np.max(conv)
            return 2.0 * n * (cm + np.log(np.sum(np.exp((conv - cm) * tau))) / tau)

        def grad_func(t):
            t_shift = t - np.max(t)
            a = np.exp(t_shift)
            a = a / np.sum(a)
            conv = compute_conv(a)
            cm = np.max(conv)
            w = tau * np.exp((conv - cm) * tau)
            w = w / np.sum(w)
            c = np.zeros(n)
            for j in range(n):
                end = min(2 * n - 1, j + n)
                if j < 2 * n - 1:
                    c[j] = np.dot(w[j:end], a[:end - j])
            mean_c = np.dot(c, a)
            return 4.0 * n * a * (c - mean_c)

        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': 100, 'gtol': 1e-5, 'maxfun': 200})
            if result.x is not None:
                t_shift = result.x - np.max(result.x)
                a_new = np.exp(t_shift)
                a_new = a_new / np.sum(a_new)
                a_new = np.maximum(a_new, 0)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Direction LP optimization + subgradient + smoothed L-BFGS + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, obj={best_score:.8f}")

    # Phase 1: Direction LP (primary new method)
    print(f"\n=== Phase 1: Direction LP Optimization ===")
    alloc = min((deadline - time.monotonic()) * 0.35, 400.0)
    a_lp = direction_lp_optimize(best_seq, time_budget=alloc)
    obj_lp = compute_obj(a_lp)
    if obj_lp < best_score:
        best_score = obj_lp
        best_seq = a_lp.copy()
        n_parent = len(best_seq)
        print(f"  Direction LP: obj={best_score:.8f} **")
    else:
        print(f"  Direction LP: obj={obj_lp:.8f}")

    # Phase 2: Smoothed L-BFGS refinement
    print(f"\n=== Phase 2: Smoothed L-BFGS ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for trial in range(2):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            if trial == 0:
                a_init = best_seq.copy()
            else:
                a_init = best_seq * np.exp(rng.normal(0, 0.03, n_parent))
                a_init = np.maximum(a_init, 0)
                a_init = a_init / np.sum(a_init)
            alloc = min(remaining * 0.2, 80.0)
            a_opt, obj_opt = smoothed_lbfgs(a_init, time_budget=alloc)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_parent = len(best_seq)
                print(f"  Smoothed trial {trial}: obj={best_score:.8f} **")
            else:
                print(f"  Smoothed trial {trial}: obj={obj_opt:.8f}")

    # Phase 3: Momentum subgradient (escape mechanism)
    print(f"\n=== Phase 3: Momentum Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for itype in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            if itype == 0:
                a_init = best_seq.copy()
            elif itype == 1:
                a_init = best_seq * np.exp(rng.normal(0, 0.05, n_parent))
                a_init = np.maximum(a_init, 0) / np.sum(np.maximum(a_init, 0))
            else:
                ramp = np.linspace(0.7, 1.3, n_parent)
                a_init = best_seq * ramp
                a_init = np.maximum(a_init, 0) / np.sum(np.maximum(a_init, 0))
            alloc = min(remaining * 0.2, 40.0)
            a_sub = momentum_subgradient(a_init, time_budget=alloc)
            obj_sub = compute_obj(a_sub)
            if obj_sub < best_score:
                best_score = obj_sub
                best_seq = a_sub.copy()
                print(f"  Subgrad {itype}: obj={best_score:.8f} **")
            else:
                print(f"  Subgrad {itype}: obj={obj_sub:.8f}")

    # Phase 4: Multi-n exploration
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = sorted(set([
        max(400, n_best - 100), n_best - 50, n_best + 50,
        n_best + 100, n_best + 200, n_best + 400
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_best]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 50.0)
        a_opt = direction_lp_optimize(a_try, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 5: Final intensive direction LP
    print(f"\n=== Phase 5: Final Direction LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a_lp = direction_lp_optimize(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final Dir LP: obj={best_score:.8f} **")
        else:
            print(f"  Final Dir LP: obj={obj_lp:.8f}")

    # Phase 6: Final subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_sub = momentum_subgradient(best_seq, time_budget=alloc)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Final Subgrad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
