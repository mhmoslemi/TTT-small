```python
"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using
LP alternating optimization with adaptive step sizes, FFT-based
initialization, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def compute_conv_fft(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_conv(a, b=None):
    """Compute convolution a*b (or a*a if b is None) via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n+m-1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # conv(a,g)[k] = sum_i a[i]*g[k-i]
        # So A[k,j] contributes a[i] where i = k - j, i.e., k = i + j
        # For fixed j, A[k,j] = a[k-j] when 0 <= k-j < n
        # So k ranges from j to j+n-1, clamped to [0, 2n-2]
        k_start = j  # k-j = 0 => k = j
        k_end = min(j + n - 1, 2*n - 2)
        for k in range(k_start, k_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix_sparse(a)
    
    # x = [g (n vars), M (1 var)]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0
    A_ub = np.hstack([A.toarray(), -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
        return g_val, float(res.x[n])
    return None, None

def fft_flat_spectrum(a, n_iters=200, time_budget=30.0, relaxation=0.5):
    """FFT-based flat spectrum iteration."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe

        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    rng = np.random.default_rng(hash((it, n)) % (2**32))
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

def lp_alternating(a, time_budget=60.0, max_iters=200):
```

```python
"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using
LP alternating optimization (block coordinate descent on min-max problem),
FFT flat-spectrum initialization, adaptive line search, multi-n exploration,
and random restarts. The LP at each step solves: min M s.t. conv(a,g)[k]<=M,
sum(g)=1, g>=0, exploiting the sparse banded structure of the convolution matrix.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def compute_conv_fft(a):
    """Compute full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv(a, b):
    """Compute convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) on failure."""
    n = len(a)
    A = build_conv_matrix_sparse(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub_sparse = A
    # We need [A | -1] @ [g; M] <= 0
    # Use dense for the -1 column (it's small)
    A_ub = np.hstack([A.toarray(), -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                return g_val, float(res.x[n])
    except Exception:
        pass
    return None, None


def fft_flat_spectrum(a, n_iters=200, time_budget=30.0, relaxation=0.5, seed=0):
    """FFT-based flat magnitude iteration (Gerchberg-Saxton variant)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe

        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:

            obj_new = compute_obj(a_new)

                a = a_new
                stall = 0

                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

            break

        # Adaptive line search: try several t values
        candidate_ts = [0.3, 0.5, 0.7, 0.8, 0.9, 1.0]
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose and (it % 20 == 0 or it == 0):
                elapsed = time.monotonic() - start
                print(f"    LP it{it:3d}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, "
                      f"t={best_t_val:.2f}, ||g-a||={diff:.2e}, time={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 8:
                scale = 0.02 + 0.01 * min(stall // 8, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init_flat_spectrum(n, rng):
    """Generate a non-negative sequence with approximately flat spectrum."""
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    phases = rng.uniform(0, 2 * np.pi, L)
    A = np.exp(1j * phases)
    a = ifft(A).real[:n]
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with FFT initialization,
    multi-n exploration, random restarts, and final polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = random_init_flat_spectrum(n_init, rng)
        best_score = compute_obj(best_seq)
        print(f"Flat-spectrum init (n={n_init}): obj={best_score:.8f}")

    # === Phase 1: FFT initialization ===
    print(f"\n=== Phase 1: FFT Flat Spectrum (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.04, 25.0)
        a_fft = fft_flat_spectrum(best_seq, n_iters=300, time_budget=alloc,
                                  relaxation=0.5, seed=seed)
        obj = compute_obj(a_fft)
        print(f"  FFT relax=0.5: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fft.copy()

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_fft2 = fft_flat_spectrum(best_seq, n_iters=200,
                                       time_budget=min(remaining * 0.03, 15.0),
                                       relaxation=0.3, seed=seed + 1)
            obj2 = compute_obj(a_fft2)
            print(f"  FFT relax=0.3: obj={obj2:.8f}")
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_fft2.copy()

    # === Phase 2: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.35, 450.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, max_iters=500,
                              verbose=True, seed=seed)
        obj = compute_obj(a_lp)
        print(f"  LP main: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        400, 500, 600, 700, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and abs(n - n_best) >= 30]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick FFT
        a_opt = fft_flat_spectrum(a_try, n_iters=100,
                                  time_budget=min(8.0, remaining * 0.05),
                                  relaxation=0.5, seed=seed + n_try)

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_opt = lp_alternating(a_opt, time_budget=min(40.0, remaining * 0.25),
                                   max_iters=80, verbose=False, seed=seed + n_try)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 4: Random restarts ===
    print(f"\n=== Phase 4: Random Restarts ===")
    n_best = len(best_seq)
    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if r % 3 == 0:
            a_r = random_init_flat_spectrum(n_best, rng)
        elif r % 3 == 1:
            # Perturbed best with larger scale
            a_r = best_seq * np.exp(rng.normal(0, 0.15, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:
            # Different n with flat spectrum init
            n_r = int(n_best + rng.integers(-150, 150))
            n_r = max(300, min(1500, n_r))
            a_r = random_init_flat_spectrum(n_r, rng)

        # Optimize
        a_opt = fft_flat_spectrum(a_r, n_iters=100,
                                  time_budget=min(10.0, remaining * 0.1),
                                  relaxation=0.5, seed=seed + 100 + r)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_opt = lp_alternating(a_opt, time_budget=min(35.0, remaining * 0.25),
                                   max_iters=60, verbose=False, seed=seed + 100 + r)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  Restart {r} (n={len(a_opt)}): obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 5: Final intensive LP ==
    print(f"\n=== Phase 5: Final LP Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 300.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, max_iters=500,
                                 verbose=True, seed=seed + 200)
        obj = compute_obj(a_final)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()

    # === Phase 6: Final FFT + short LP ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fft = fft_flat_spectrum(best_seq, n_iters=200,
                                  time_budget=min(15.0, remaining * 0.3),
                                  relaxation=0.4, seed=seed + 300)
        obj = compute_obj(a_fft)
        print(f"  Final FFT: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fft.copy()

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_lp = lp_alternating(best_seq, time_budget=min(30.0, remaining * 0.5),
                                  max_iters=100, verbose=False, seed=seed + 400)
            obj = compute_obj(a_lp)
            print(f"  Final LP2: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
