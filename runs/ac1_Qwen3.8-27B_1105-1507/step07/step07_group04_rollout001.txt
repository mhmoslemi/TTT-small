"""
Advanced LP-alternating optimizer with constraint generation, adaptive active sets,
symmetric refinement, and multi-scale exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a, k_indices):
    """Build a matrix with only rows for k_indices. Returns (A_sub, k_indices)."""
    n = len(a)
    K = len(k_indices)
    A_sub = np.zeros((K, n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        d_start = k + 1 - j_end
        d_end = k - j_start + 1
        A_sub[idx, j_start:j_end] = a[d_start:d_end][::-1]
    return A_sub


def solve_lp_constrained(a, k_indices, solver=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for k in k_indices, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A_sub = build_conv_matrix_sparse(a, k_indices)
    K = len(k_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    if solver:
        solvers = [solver] + solvers

    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((K, n + 1))
        A_ub[:, :n] = A_sub
        A_ub[:, n] = -1.0
        b_ub = np.zeros(K)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:

    except Exception:

def soft_max_lbgfs(a, time_budget=100.0, verbose=False):
    """
    Optimize using soft-max approximation of the peak with L-BFGS.
    Minimizes (sum_k exp(beta * b_k) / (2n-1))^(1/beta) for increasing beta.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Work with log-barrier to enforce positivity and sum=1
    # Use variables x where a = exp(x) / sum(exp(x)) (softmax parameterization)
    # Or simpler: use a directly with penalty for sum != 1 and a < 0

    def objective_and_grad(x):
        a_x = np.maximum(np.exp(x), 1e-10)
        a_x = a_x / np.sum(a_x)
        conv = compute_conv(a_x)
        # Soft max
        beta = 50.0
        # Clip for numerical stability
        conv_clipped = np.clip(conv, 1e-10, None)
        softmax_val = np.exp(beta * conv_clipped)
        log_sum = np.log(np.sum(softmax_val) / len(conv))
        obj = (1.0 / beta) * log_sum

        # Gradient via finite difference (expensive but reliable)
        # Actually let's compute analytically
        # d/d(a_j) of b[k] = a[k-j] + a[j-k] (with bounds)
        # d b[k] / d x_j = (1 - a_j) * (a[k-j] + a[j-k]) - a_j * sum_m (a[k-m] + a[m-k])
        # This is complex; use finite differences with FFT acceleration
        # For speed, use a crude gradient
        grad = np.zeros(n)
        for j in range(n):
            x_old = x.copy()
            x_old[j] += 1e-5
            a_new = np.maximum(np.exp(x_old), 1e-10)
            a_new = a_new / np.sum(a_new)
            conv_new = compute_conv(a_new)
            soft_new = np.exp(beta * np.clip(conv_new, 1e-10, None))
            obj_new = (1.0 / beta) * np.log(np.sum(soft_new) / len(conv_new))
            grad[j] = (obj_new - obj) / 1e-5

        return obj, grad

    # Use a simpler approach: direct gradient descent on a with projection
    beta_values = [20, 50, 100, 200, 500]
    a_current = a.copy()

    for beta in beta_values:
        if time.monotonic() - start > time_budget:
            break

        for it in range(200):
            if time.monotonic() - start > time_budget:
                break

            conv = compute_conv(a_current)
            max_b = float(np.max(conv))

            # Compute soft gradient
            w = np.exp(beta * (conv - max_b))  # Weighted peaks (normalized so max gets weight 1)
            w = w / np.sum(w)

            # Gradient of soft max w.r.t. a
            # d/d(a_j) [sum_k w_k * b[k]] = sum_k w_k * (a[k-j] + a[j-k])
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                wk = w[k]
                if wk < 1e-10:
                    continue
                j_start = max(0, k - n + 1)
                j_end = min(n, k + 1)
                # Contribution to grad[j] from b[k]: a[k-j]
                for j in range(j_start, j_end):
                    d1 = k - j  # index for a[k-j], valid if k-j < n
                    d2 = j - k  # would be a[j-k]... wait
                    # b[k] = sum_i a[i]*a[k-i]
                    # d b[k] / d a[j] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n and j != k-j)
                    # Actually: b[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
                    # d/d a[j]: a[j] appears when i=j (contributing a[k-j]) and when k-i=j i.e. i=k-j (contributing a[j])
                    # So d b[k] / d a[j] = a[k-j] * [0<=k-j<n] + a[k-j] * [0<=k-j<n and j != k-j]
                    # = a[k-j] * [0<=k-j<n] * (1 + [j != k-j])
                    # Hmm this is getting confusing. Let me just use the formula:
                    # b[k] = sum_i a[i] a[k-i]
                    # d b[k]/d a[j] = a[k-j] (from i=j term) + a[k-j] (from k-i=j term, i.e., i=k-j)
                    # = 2*a[k-j] if j != k-j, or a[j]^2... no
                    # Actually: d/d a[j] [sum_i a[i] a[k-i]]
                    # = a[k-j] (differentiating a[i] at i=j) + a[k-j] (differentiating a[k-i] at k-i=j, i.e., i=k-j)
                    # But if j = k-j (i.e., k=2j), then it's just a[j] (one term, derivative is a[j])
                    # So: d b[k]/d a[j] = a[k-j] if 0<=k-j<n, plus a[k-j] if 0<=k-j<n and k-j != j
                    # = a[k-j] * (1 + [k-2j != 0]) if 0<=k-j<n

                    pass  # Skip individual computation, use vectorized below

            # Vectorized gradient computation
            grad = np.zeros(n)
            for j in range(n):
                # d b[k] / d a[j] for all k, summed with weights w
                # k ranges from j to j+n-1 (where a[k-j] is valid)
                k_min = j
                k_max = j + n - 1
                k_min = max(0, k_min)
                k_max = min(2 * n - 2, k_max)
                if k_min > k_max:
                    continue
                ks = np.arange(k_min, k_max + 1)
                vals = a[ks - j]  # a[k-j] for k in range
                # Check if k-j == j for any k (i.e., k = 2j)
                double_j = 2 * j
                if k_min <= double_j <= k_max:
                    vals[double_j - k_min] *= 0.5  # Only counted once
                grad[j] = np.dot(w[k_min:k_max + 1], vals)

            # Project gradient: remove component along all-ones (to maintain sum)
            grad_centered = grad - np.mean(grad)

            # Adaptive step size
            step = 0.01 / (1.0 + it * 0.01)
            a_new = a_current + step * grad_centered * (-1)  # Descend
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
            a_current = a_new

            if verbose and it % 50 == 0:
                print(f"    SoftMax beta={beta}, it={it}, obj={best_obj:.8f}")

    return best_a


def pairwise_exchange_improve(a, time_budget=60.0, verbose=False):
    """
    Local search: try moving small amounts of mass between pairs of positions
    to reduce the objective. Focus on positions that contribute to the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    # Identify "important" positions: those that contribute to top peaks
    top_k = np.argsort(conv)[::-1][:10]
    # For each peak k, the contributing positions are i where k-n+1 <= i <= k (and i in [0,n))
    important = set()
    for k in top_k:
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            important.add(j)
    important_list = sorted(important)

    # Try exchanges between important and non-important positions
    non_important = [j for j in range(n) if j not in important]

    rng = np.random.default_rng(12345)

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Try a few random exchanges
        for _ in range(5):
            if time.monotonic() - start > time_budget:
                break
            if len(important_list) < 2:
                break
            i = int(rng.integers(0, len(important_list)))
            j = int(rng.integers(0, len(non_important))) if non_important else 0
            i = important_list[i]
            if i == j:
                continue
            for frac in [0.01, 0.03, 0.05, 0.02]:
                delta = frac * a[i]
                if delta <= 0 or delta > a[i] * 0.9:
                    continue
                a[i] -= delta
                a[j] += delta
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    # Update conv
                    conv = compute_conv(a)
                    k_star = int(np.argmax(conv))
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    break

        if not improved and it > 100:
            # Try larger-scale perturbation
            scale = 0.01 + 0.005 * (it // 100)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.1), n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                conv = compute_conv(a)

    return best_a


def symmetric_refinement(a, time_budget=30.0):
    """
    Enforce approximate symmetry: average with reversed sequence,
    then optimize the symmetric part.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Make symmetric
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s_sym = float(np.sum(a_sym))
    if s_sym < 1e-10:
        return a
    a_sym = a_sym / s_sym

    obj_sym = compute_obj(a_sym)
    obj_orig = compute_obj(a)

    if obj_sym < obj_orig:
        return a_sym
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multi_start_lp(a, time_budget=100.0, n_starts=3, verbose=False):
    """
    Try multiple perturbations of a, each followed by LP optimization,
    and return the best result.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(int(time.monotonic() * 100) % (2**31))

    for s_idx in range(n_starts):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:
            break
        alloc = max(10.0, remaining / (n_starts - s_idx))

        # Different perturbation strategies
        if s_idx == 0:
            # Small random perturbation
            a_pert = best_a * np.exp(rng.normal(0, 0.03, n))
        elif s_idx == 1:
            # Targeted: reduce mass at peak-contributing positions
            conv = compute_conv(best_a)
            k_peak = int(np.argmax(conv))
            a_pert = best_a.copy()
            j_start = max(0, k_peak - n + 1)
            j_end = min(n, k_peak + 1)
            # Reduce at peak contributors
            factor = 0.8
            a_pert[j_start:j_end] *= factor
            # Redistribute proportionally
            deficit = float(np.sum(best_a[j_start:j_end]) - np.sum(a_pert[j_start:j_end]))
            other = np.arange(n)
            other_mask = ~np.isin(other, range(j_start, j_end))
            if np.sum(other_mask) > 0:
                a_pert[other_mask] += deficit * a_pert[other_mask] / np.sum(a_pert[other_mask])
        else:
            # Symmetric perturbation
            a_pert = best_a.copy()
            noise = rng.normal(0, 0.02, n)
            noise = 0.5 * (noise + noise[::-1])  # Make symmetric
            a_pert = a_pert * np.exp(noise)

        a_pert = np.maximum(a_pert, 0)
        s_pert = float(np.sum(a_pert))
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Optimize with LP constraint generation
        a_opt = lp_constraint_generation(a_pert, time_budget=alloc, max_constraints=200)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj - 1e-14:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    MultiStart {s_idx}: obj={best_obj:.8f} **")
        elif verbose:
            print(f"    MultiStart {s_idx}: obj={obj_opt:.8f}")

    return best_a
