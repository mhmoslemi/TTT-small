"""
Optimized sequence search using L-BFGS with log-sum-exp smoothing on softmax parameterization,
multi-temperature schedule, subgradient polish, and multi-n exploration.
Key improvements: sharper smooth max via adaptive temperature, proper softmax gradient,
and iterative refinement from parent construction.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fft_conv(a):
    """Linear auto-convolution of a using FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def obj_from_conv(conv, n):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    return 2.0 * n * float(np.max(conv))


def true_obj(a):
    """Compute the true objective for a (assumed normalized to sum=1)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fft_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_max_val_and_weights(conv, t):
    """Compute smooth max value and exponential weights."""
    c_max = np.max(conv)
    shifted = t * (conv - c_max)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    smooth_val = c_max + np.log(total) / t
    w = exp_shifted / total
    return smooth_val, w


def grad_conv_wrt_a(a, w, n):
    """
    Compute gradient of sum_k w_k * conv_k w.r.t. a.
    Uses FFT cross-correlation: grad_a[i] = 4n * sum_j a_j * w_{i+j}
    (Actually it's 2 * sum_k w_k * 2*a_{k-i}, the factor 2 comes from the quadratic form.)
    """
    flip_a = a[::-1]
    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1
    faf = rfft(flip_a, N2)
    fw = rfft(w, N2)
    c_full = irfft(faf * fw, N2)[:3 * n - 2]
    # grad_a[i] = 2 * sum_j a_j * w_{i+j} = 2 * cross_corr(a, w)[i]
    # cross_corr at lag i = conv(flip_a, w)[n-1+i]
    grad_a_raw = c_full[n - 1:2 * n - 1]
    return 2.0 * grad_a_raw


def smooth_obj_grad(b, t, n):
    """
    Compute smooth objective and its gradient w.r.t. b (log-parameterization).
    a = softmax(b), F = 2n * (1/t) * log(sum_k exp(t * conv_k))
    """
    b_max = np.max(b)
    exp_b = np.exp(b - b_max)
    S = np.sum(exp_b)
    a = exp_b / S

    conv = fft_conv(a)
    smooth_val, w = smooth_max_val_and_weights(conv, t)
    obj = 2.0 * n * smooth_val

    # dF/da_i = 2n * sum_k w_k * 2*a_{k-i} = 4n * (cross_correlation of a and w)[i]
    grad_a = 2.0 * grad_conv_wrt_a(a, w, n)
    # Wait: grad_conv_wrt_a already includes the factor 2 from the quadratic form.
    # So dF/da_i = 2n * grad_conv_wrt_a(a, w, n)[i]
    grad_a = 2.0 * n * grad_conv_wrt_a(a, w, n)

    # dF/db_j = a_j * (dF/da_j - sum_i dF/da_i * a_i)
    dot = np.dot(grad_a, a)
    grad_b = a * (grad_a - dot)

    return obj, grad_b


def run_lbfgs(b_init, n, t, maxiter=300):
    """Run L-BFGS-B optimization for a given temperature."""
    try:
        res = minimize(
            lambda b: smooth_obj_grad(b, t, n)[0],
            b_init,
            jac=lambda b: smooth_obj_grad(b, t, n)[1],
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15}
        )
        return res.x, res.fun
    except Exception as e:
        return b_init, np.inf

            a = a_new
            eta = min(eta * 1.01, 0.1)
        else:
            eta *= decay
            if eta < 1e-10:
                break

    return best_a


def prox_onto_simplex(v, tau):
    """Project v onto {x >= 0, sum(x) = tau}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - tau))[0][-1]
    theta = (cssv[rho] - tau) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    a = a / s

    step = 1.0

            break

        k_star = int(np.argmax(conv))

            a = a_new
            step = min(step * 1.005, 5.0)
        else:

            T *= 1.001
            if step < 1e-8:
                break

    return best_a

            print(f"  Final subgrad: obj={best_obj:.10f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_obj:
        best_obj = val
        print(f"  evaluate_sequence refined: {best_obj:.10f}")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, obj={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
