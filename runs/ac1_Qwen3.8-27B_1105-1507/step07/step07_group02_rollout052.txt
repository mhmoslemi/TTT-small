What if we parameterize a as a sum of a few Gaussians? Then we optimize the parameters (centers, widths, amplitudes) of these Gaussians. This reduces the dimensionality and might be easier to optimize.

For example: a(x) = sum_{j=1}^{m} c_j * exp(-(x - mu_j)² / (2*sigma_j²))

We optimize the 3m parameters (c_j, mu_j, sigma_j for j=1..m) to minimize the objective.

With m=3-5 Gaussians, we have 9-15 parameters, which is very manageable.

The advantage: the auto-convolution of Gaussians is also a Gaussian (for same-width Gaussians), so we can compute it analytically or with fewer points.

Actually, the auto-convolution of a sum of Gaussians is a sum of convolutions of pairs of Gaussians, each of which is a Gaussian. So we can compute the convolution analytically at any point.

Let me formalize: if a = sum_j c_j * g_j where g_j is a Gaussian, then (a*a)(k) = sum_{j,l} c_j * c_l * (g_j * g_l)(k).

(g_j * g_l)(k) is a Gaussian with:
- center: mu_j + mu_l
- variance: sigma_j² + sigma_l²
- amplitude: c_j * c_l * sigma_j * sigma_l / sqrt(2*pi*(sigma_j²+sigma_l²)) * 2*sqrt(pi) (for continuous)

For discrete, it's similar but with the discrete Gaussian.

This gives us an analytical formula for the convolution at any point! We can then use a global optimizer to find the parameters that minimize the maximum of the convolution.

This is a much lower-dimensional problem (3m parameters vs n) and we can use derivative-based methods.

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses multiplicative weights (mirror descent), Gaussian mixture parameterization,
and LP-based refinement. Key insight: for non-negative sequences, the peak of
auto-convolution is NOT always at the center - we exploit this freedom.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def multiplicative_update(a, time_budget=60.0, verbose=False):
    """
    Mirror descent with exponential weights on the probability simplex.
    a_i <- a_i * exp(-lr * grad_i) / Z
    where grad_i = weighted sum of a_{k-i} over peak lags k.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    top_k = 5  # use top-K peak lags
    consecutive_fails = 0
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Find top-K peaks
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[sorted_idx]
        peak_weights = peak_vals / peak_vals.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for w, k in zip(peak_weights, sorted_idx):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w * a[j]
        
        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10 or not np.isfinite(s_new):
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 1.0)
            consecutive_fails = 0
        else:
            consecutive_fails += 1
            if consecutive_fails > 50:
                lr *= 0.7
                consecutive_fails = 0
            if lr < 1e-8:
                # Try perturbation
                rng = np.random.default_rng(it)
                a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                    else:
                        a = best_a.copy()
                lr = 0.1
                consecutive_fails = 0
        
        if verbose and it % 1000 == 0:
            print(f"  MU it{it}: obj={best_obj:.8f}, lr={lr:.6f}")
    
    return best_a

def gradient_ascent_descent(a, time_budget=60.0):
    """
    Subgradient descent on the simplex for minimizing max convolution peak.
    Uses Barzilai-Borwein step size and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    top_k = 3
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[sorted_idx]
        peak_weights = peak_vals / peak_vals.sum()
        
        grad = np.zeros(n)
        for w, k in zip(peak_weights, sorted_idx):
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += w * a[j]
        
        # Center the gradient to preserve sum
        grad_mean = float(np.mean(grad))
        grad_centered = grad - grad_mean
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad_centered
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.9
            if lr < 1e-10:
                rng = np.random.default_rng(it * 7 + 3)
                a_pert = best_a * np.exp(rng.normal(0, 0.08, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert
                        a = a_pert
                    else:
                        a = best_a.copy()
                lr = 0.01
    
    return best_a

def batch_lda_optimize(a, time_budget=60.0):
    """
    Batch optimization: find directions that reduce the peak by adjusting
    multiple coordinates at once. Uses the insight that we want to make
    a "anti-correlated" with its convolution partners at peak lags.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # For the peak lag k_star, grad[i] = a[k_star - i]
        # We want to reduce a[i] where a[k_star - i] is large,
        # and increase a[i] where a[k_star - i] is small.
        
        partner = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                partner[i] = a[j]
        
        # Direction: move mass from high-partner to low-partner coordinates
        # d[i] = partner[i] - mean(partner)
        d = partner - np.mean(partner)
        
        # Try different step sizes
        best_obj_local = np.inf
        best_a_local = best_a.copy()
        for t in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_try = a - t * d
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_local:
                best_obj_local = obj_try
                best_a_local = a_try.copy()
        
        if best_obj_local < best_obj - 1e-12:
            best_obj = best_obj_local
            best_a = best_a_local.copy()
            a = best_a_local
        else:
            # Stuck - try perturbation
            rng = np.random.default_rng(it * 13 + 7)
            a_pert = best_a * np.exp(rng.normal(0, 0.03 + 0.01 * (it % 10), n))
            a_pert = np.maximum(a_pert, 0)
            sp = float(np.sum(a_pert))
            if sp > 1e-10:
                a_pert = a_pert / sp
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert
                    a = a_pert
                else:
                    a = best_a.copy()
    
    return best_a

def gaussian_mixture_init(n, m, rng):
    """Generate a Gaussian mixture initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.zeros(n)
    for _ in range(m):
        c = rng.uniform(0.5, 2.0)
        mu = rng.uniform(0.1*n, 0.9*n)
        sig = rng.uniform(0.05*n, 0.2*n)
        a += c * np.exp(-0.5 * ((x - mu) / sig)**2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s

def optimize_gaussian_params(n, m, rng, time_budget=30.0):

def peak_replacement(a, time_budget=30.0):
    """
    Innovative approach: identify the peak lag, then find the optimal
    redistribution of mass that minimizes the peak while maintaining
    the sum. Uses a local quadratic approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # The peak is sum_i a[i]*a[k_star-i]
        # To reduce it, we can try: transfer mass from i to i' where
        # a[k_star-i] > a[k_star-i']
        
        # Compute partner values
        partner = np.zeros(n)
        valid = np.zeros(n, dtype=bool)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                partner[i] = a[j]
                valid[i] = True
        
        if not valid.any():
            break
        
        # Sort by partner value
        valid_idx = np.where(valid)[0]
        sorted_valid = valid_idx[np.argsort(partner[valid_idx])]
        
        # Transfer from high-partner to low-partner
        # Try transferring a small amount epsilon
        i_src = sorted_valid[-1]  # highest partner
        i_dst = sorted_valid[0]   # lowest partner
        
        for eps in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            a_try = a.copy()
            transfer = min(a[i_src], eps)
            a_try[i_src] -= transfer
            a_try[i_dst] += transfer
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
        else:
            # No improvement, try random moves
            rng = np.random.default_rng(it * 31 + 17)
            i_src = int(rng.integers(0, n))
            i_dst = int(rng.integers(0, n))
            if i_src == i_dst:
                continue
            eps = 0.01
            a_try = a.copy()
            transfer = min(a[i_src], eps)
            if transfer > 0:
                a_try[i_src] -= transfer
                a_try[i_dst] += transfer
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    a = best_a.copy()
    
    return best_a

def lp_alternating_fast(a, time_budget=60.0, verbose=False):
    """
    Fast LP alternating: instead of solving the full LP, use BFGS to
    minimize max_k conv(a,g)(k) subject to sum(g)=1, g>=0.
    """
    n = len(a)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    def given_g_objective(g):
        """Compute max_k conv(a, g)(k) given g on simplex."""
        # conv(a, g) via FFT
        size = 1
        while size < 2*n - 1:
            size <<= 1
        fa = rfft(a, size)
        fg = rfft(g, size)
        conv = irfft(fa * fg, size)[:2*n-1]
        return float(np.max(conv))
    
    for outer in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        # Find g that minimizes max_k conv(a,g)(k)
        # Use subgradient on the simplex
        g = np.ones(n) / n
        lr_g = 0.01
        
        for inner in range(200):
            if time.monotonic() - start > time_budget:
                break
            
            fa = rfft(a, size)
            # conv(a, g)
            fg = rfft(g, size)
            conv_ag = irfft(fa * fg, size)[:2*n-1]
            
            k_star = int(np.argmax(conv_ag))
            # grad_g[j] = a[k_star - j] (if valid)
            grad_g = np.zeros(n)
            for j in range(n):
                i = k_star - j
                if 0 <= i < n:
                    grad_g[j] = a[i]
            
            # Mirror update
            g_new = g * np.exp(-lr_g * grad_g)
            g_new = np.maximum(g_new, 0)
            s_g = float(np.sum(g_new))
            if s_g > 1e-10:
                g_new = g_new / s_g
                g = g_new
        
        # Blend
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1-t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
        
        if a is best_a and (best_obj > compute_obj(a) + 1e-6):
            a = best_a.copy()
    
    return best_a
