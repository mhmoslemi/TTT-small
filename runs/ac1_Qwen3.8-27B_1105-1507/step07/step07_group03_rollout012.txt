If i = k/2, then `dc[k]/da[i] = a[k-i] = a[i]`.

So the gradient is:
`dF/da[i] = sum_k w[k] * (2*a[k-i] - delta(i, k/2) * a[i])` for valid indices.

This is approximately `2 * correlate(w, a)[i]` for large n (the diagonal correction is negligible).

So the gradient is approximately `2 * correlate(w, a)[:n]` (with the understanding that we need to subtract the correction for the diagonal).

For the optimization, I'll use:
- Gradient: `g_i ≈ 2 * correlate(w, a)[:n]_i` (approximately, ignoring the diagonal)
- This is the direction of increasing F, so we want to move in the negative direction
- But we also need to maintain sum(a) = 1 and a >= 0

Since the objective is scale-invariant, I can work on the simplex. The projection onto the simplex (a >= 0, sum(a) = 1) is well-known.

Let me now think about a practical algorithm:

**Algorithm: Smooth Max Gradient Descent with Projection**

1. Initialize a from parent (normalized to sum=1)
2. For each iteration:
   a. Compute c = conv(a, a) via FFT
   b. Compute softmax weights: w = exp(beta * c) / sum(exp(beta * c))
   c. Compute gradient: grad = 2 * correlate(w, a)[:n] (via FFT)
   d. Center the gradient: grad -= mean(grad) (to maintain sum constraint in linear space)
   e. Take step: a_new = a - alpha * grad
   f. Project to simplex: a_new = project_simplex(a_new)
   g. Update beta (anneal: increase beta over time for sharper focus)
   h. Update alpha (adaptively)

3. Periodically check improvement and save best.

The simplex projection: given x, find the closest point on {y >= 0, sum(y) = 1}.
Algorithm: sort x in decreasing order, find the threshold, subtract it.

This is O(n log n) per iteration, which is fast.

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth-max gradient descent with adaptive beta annealing
2. Differential evolution in log-space for global search
3. LP-based alternating optimization for local refinement
4. Multi-n exploration and final polishing

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    theta = 0.0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def smooth_max_grad_descent(a, n_iters=5000, time_budget=60.0, beta_start=50, beta_end=2000):
    """
    Gradient descent on the smooth approximation of max(conv(a,a)).
    Uses adaptive step size and beta annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    alpha = 0.5
    size = 1
    while size < 4 * n:
        size <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal beta
        t = it / max(n_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        
        # Compute convolution
        fa = rfft(a, size)
        c = irfft(fa * fa, size)[:2 * n - 1]
        
        # Softmax weights
        c_shifted = c - np.max(c)
        w = np.exp(beta * c_shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum
        
        # Gradient: grad[i] = sum_k w[k] * a[k-i] (correlation)
        # Use FFT: grad = ifft(fft(w, size) * conj(fft(a, size)))[:n]
        fw = fft(w, size)
        fa_full = fft(a, size)
        grad = ifft(fw * np.conj(fa_full)).real[:n]
        
        # Note: for conv(a,a), the true gradient has a factor of 2 and a diagonal correction
        # but for large n, 2*grad is a good approximation
        grad *= 2.0
        
        # Center gradient (to approximately preserve sum)
        grad -= np.mean(grad)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            step = alpha / grad_norm
        else:
            step = alpha
        step = min(step, 1.0)
        
        a_new = a - step * grad
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.01, 2.0)  # Slightly increase step on success
        else:
            alpha *= 0.95
            if alpha < 0.01:
                alpha = 0.01
    
    return best_a

def differential_evolution_log(a_init, n_population=60, n_generations=2000, 
                                F=0.7, CR=0.8, time_budget=120.0, seed=42):
    """
    Differential evolution in log-space.
    Variables: x_i = log(a_i + eps), a_i = exp(x_i) - eps
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    
    eps = 1e-10
    x_init = np.log(a_init + eps)
    
    # Initialize population
    pop = np.zeros((n_population, n))
    pop[0] = x_init
    for i in range(1, n_population):
        if i < n_population // 2:
            # Perturbations of the best
            pop[i] = x_init + rng.normal(0, 0.05 * (1 + i * 0.05), n)
        else:
            # Random initialization (log-uniform)
            pop[i] = rng.uniform(np.log(1e-4), np.log(1.0), n)
    
    # Evaluate initial population
    objs = np.zeros(n_population)
    for i in range(n_population):
        a_i = np.exp(pop[i] - eps)
        a_i = np.maximum(a_i, 0)
        s = np.sum(a_i)
        if s < 1e-10:
            objs[i] = np.inf
        else:
            objs[i] = compute_obj(a_i / s)
    
    best_idx = np.argmin(objs)
    best_obj = objs[best_idx]
    best_a = np.exp(pop[best_idx] - eps)
    best_a = np.maximum(best_a, 0)
    best_a = best_a / np.sum(best_a)
    
    start = time.monotonic()
    no_improve = 0
    
    for gen in range(n_generations):
        if time.monotonic() - start > time_budget:
            break
        
        # Adapt F
        F_gen = F * (0.8 + 0.2 * np.random.random())
        
        for i in range(n_population):
            # Select three distinct individuals
            r1, r2, r3 = rng.choice([j for j in range(n_population) if j != i], 3, replace=False)
            
            # Mutation
            v = pop[r1] + F_gen * (pop[r2] - pop[r3])
            
            # Crossover
            trial = pop[i].copy()
            j_rand = rng.integers(0, n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
            
            # Evaluate
            a_trial = np.exp(trial - eps)
            a_trial = np.maximum(a_trial, 0)
            s = np.sum(a_trial)
            if s < 1e-10:
                obj_trial = np.inf
            else:
                obj_trial = compute_obj(a_trial / s)
            
            # Selection
            if obj_trial < objs[i]:
                objs[i] = obj_trial
                pop[i] = trial
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_a = a_trial / s
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        # Periodically inject the best into the population
        if gen % 50 == 49:
            pop[0] = np.log(np.maximum(best_a, 1e-10) + eps)
            objs[0] = best_obj
        
        # Optionally add random individuals to maintain diversity
        if gen % 100 == 99:
            idx = rng.integers(1, n_population)
            pop[idx] = rng.uniform(np.log(1e-3), np.log(1.0), n)
            a_rand = np.exp(pop[idx] - eps)
            objs[idx] = compute_obj(a_rand / np.sum(a_rand))
    
    return best_a, best_obj

import time

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)

    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    a = np.maximum(a, 0).astype(np.float64).copy()

        return a

    best_a = a.copy()

    alpha = 0.3
    size = 1

                mask[rng.integers(n)] = True
            trial = np.where(mask, v, pop[i])
            
            at = np.exp(trial - eps)
            at = np.maximum(at, 0)
            st = np.sum(at)
            obj_t = compute_obj(at / st) if st > 1e-10 else np.inf
            
            if obj_t < objs[i]:
                objs[i] = obj_t
                pop[i] = trial
                if obj_t < best_obj_de:
                    best_obj_de = obj_t
                    best_a_de = at / st
        
        if gen % 100 == 99:
            k = int(rng.integers(1, pop_size))
            pop[k] = x_init + rng.normal(0, 0.1, n)
            ak = np.exp(pop[k] - eps)
            ak = np.maximum(ak, 0)
            objs[k] = compute_obj(ak / np.sum(ak))
    
    return best_a_de, best_obj_de

def lp_alternating(a, time_budget=60.0):
    from scipy.optimize import linprog
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        # Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        # Build A matrix
        A_mat = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A_mat[j:j + r, j] = a[:r]
        
        # LP formulation: variables [g_0, ..., g_{n-1}, M]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A_mat
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 5})
        
        if res.status != 0 or res.x is None:
            break
        
        g = np.maximum(res.x[:n], 0)
        sg = np.sum(g)
        if sg < 1e-10:
            break
        g = g / sg
        
        # Line search
        best_t = 0.5
        best_t_obj = np.inf
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = np.sum(a_try)
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
        else:
            # Perturb and continue
            rng = np.random.default_rng(it)
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            best_obj = compute_obj(a)
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
