"""
Adaptive Log-Sum-Exp Gradient Descent with Frank-Wolfe and LP polishing
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: smooth LSE surrogate with adaptive temperature, 
FFT-based O(n log n) gradients, Nesterov momentum, Frank-Wolfe polish,
and multi-scale n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def lse_gradient(a, T):
    """
    Compute gradient of T*log(sum(exp(b_k/T))) where b = conv(a,a).
    Returns (gradient, p, b) where p is the softmax weight.
    """
    n = len(a)
    b = compute_conv(a)  # length 2n-1
    
    # Softmax with temperature T (numerically stable)
    b_shifted = b.astype(np.float64) - np.max(b)
    exp_b = np.exp(b_shifted / T)
    p = exp_b / np.sum(exp_b)  # length 2n-1
    
    # Compute c[m] = sum_i a[i] * p[m+i] via FFT convolution
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    a_rev = a[::-1]
    conv_result = irfft(rfft(a_rev, N) * rfft(p, N), N)
    c = conv_result[n - 1:2 * n - 1]  # length n
    
    # Gradient: grad[m] = 2*c[m] - a[m] * p[2m] (with p[2m]=0 if 2m >= 2n-1)
    grad = 2.0 * c
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < (2 * n - 1)
    grad[valid] -= a[valid] * p[diag_idx[valid]]
    
    return grad, p, b


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_gradient_descent(a, time_budget=60.0, T_start=1.0, T_end=0.01, 
                          verbose=False, n_max_iter=50000):
    """
    LSE gradient descent with Nesterov momentum and adaptive temperature.
    a is assumed to be on the simplex (sum=1, a>=0).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    b = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(b))
    start = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()  # lookahead
    v = np.zeros(n)
    beta = 0.9
    
    # Adaptive learning rate
    eta = 0.1
    patience = 0
    max_patience = 100
    
    # Temperature schedule
    n_total_steps = int(time_budget / 0.001)  # rough estimate
    T = T_start
    
    for it in range(n_max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Temperature annealing (geometric)
        progress = min(it / max(n_total_steps, 1), 1.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov lookahead
        grad_y, p_y, b_y = lse_gradient(y, T)
        
        # Momentum update
        beta_new = (1 + np.sqrt(1 + 4 * beta**2)) / 2
        v = beta * v + grad_y
        y = a - eta * (grad_y + beta_new * (v - grad_y))
        y = project_simplex(y)
        
        # Accept/reject based on actual objective
        b_new = compute_conv(y)
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = y.copy()
            a = y.copy()
            patience = 0
            eta = min(eta * 1.01, 1.0)
        else:
            patience += 1
            eta *= 0.95
            if patience > max_patience:
                # Reset
                a = best_a.copy()
                y = a.copy()
                v = np.zeros(n)
                eta = 0.1
                patience = 0
                max_patience = min(max_patience * 2, 1000)
        
        if verbose and it % 5000 == 0:
            print(f"            LSE-GD it{it}: obj={best_obj:.8f}, T={T:.4f}, eta={eta:.4f}")
        
        if eta < 1e-8:
            break
    
    return best_a


def frank_wolfe(a, time_budget=30.0, verbose=False, n_max_iter=50000):
    """
    Frank-Wolfe (conditional gradient) for min max_k (a*a)_k over simplex.
    a is on the simplex.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    b = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(b))
    start = time.monotonic()
    
    for it in range(n_max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute b and find active constraint
        b = compute_conv(a)
        k_star = int(np.argmax(b))
        
        # Subgradient: grad[m] = 2*a[(k_star-m)] - a[m]*[2m==k_star]
        # where a[j] = 0 if j not in [0, n-1]
        grad = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]
                if 2 * m == k_star:
                    grad[m] = a[j]  # diagonal correction
        
        # Linear oracle: minimize grad^T s over simplex = pick e_{argmin grad}
        j_min = int(np.argmin(grad))
        
        # Step size: use 2/(t+2) (standard FW) or line search
        t = 2.0 / (it + 2.0)
        a_new = (1 - t) * a + t * np.zeros(n)
        a_new[j_min] += t
        
        # Ensure feasibility
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        b_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        
        if it % 10000 == 0 and verbose:
            print(f"            FW it{it}: obj={best_obj:.8f}, k*={k_star}, j={j_min}")
    
    return best_a


def multiplicative_weights(a, time_budget=30.0, verbose=False):
    """
    Mirror descent with negative entropy (multiplicative weights) on the simplex.
    """
    n = len(a)
    a = a.astype(np.float64).copy() + 1e-15  # avoid zeros
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    b = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(b))
    start = time.monotonic()
    
    eta = 0.5
    
    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break
        
        b = compute_conv(a)
        k_star = int(np.argmax(b))
        
        # Subgradient
        grad = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]
                if 2 * m == k_star:
                    grad[m] = a[j]
        
        # Center gradient (for simplex)
        grad = grad - np.mean(grad)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        
        b_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 5.0)
        else:
            eta *= 0.9
            if eta < 0.01:
                eta = 0.5  # reset
    
    return best_a


def coordinate_descent(a, time_budget=30.0, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate m, the objective as a function of a[m]=x (with others fixed) is:
    F(x) = 2n * max_k q_k(x) where q_k involves x linearly or quadratically.
    We do a 1D search over x.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    b = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(b))
    start = time.monotonic()
    
    # Precompute the "background" convolution (without one element)
    # For coordinate m: conv(a,a) = conv(a_without_m, a_without_m) + contributions from a[m]
    # Contribution of a[m] to conv: a[m] * (a shifted by m) + (a shifted by m) * a[m]
    # = a[m] * (a shifted by m + a shifted by m reversed)
    
    it = 0
    for sweep in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for m in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            # Current value
            old_val = a[m]
            
            # Try a few values around old_val
            # The optimal x minimizes max_k (base_k + x * (a[(k-m) if valid] + a[(k-m) if valid and k-m≠m]))
            # = max_k (base_k + x * c_k) where c_k = contribution coefficient
            
            # Compute base convolution (a with a[m]=0) and the linear coefficient
            a_zero = a.copy()
            a_zero[m] = 0.0
            b_base = compute_conv(a_zero)
            
            # Linear coefficient: c_k = a[m+m if valid] + a[k-m if valid] (with correction for diagonal)
            # Actually, the contribution of a[m]*x to (a*a)[k] is:
            # x * a[k-m] (from the "a[m] * a[k-m]" term) + a[k-m] * x (from "a[k-m] * a[m]" term)
            # = 2x * a[k-m] if k != 2m
            # = x * a[m] + x * a[m] = x^2 if k = 2m (wait no)
            # 
            # Let me be more careful. (a*a)[k] = sum_i a[i]*a[k-i]
            # Terms involving x = a[m]:
            # i = m: x * a[k-m]
            # k-i = m, i = k-m: a[k-m] * x (if k-m ≠ m)
            # So total linear in x: x * a[k-m] + x * a[k-m] = 2x*a[k-m] (if k-m ≠ m and valid)
            # If k-m = m (i.e., k = 2m): x * a[m] = x^2 (quadratic!)
            # 
            # Wait, that's not right either. If k = 2m:
            # i = m: a[m] * a[m] = x^2
            # There's no other term with a[m] since k-m = m = i, so it's the same term.
            # So for k = 2m: contribution is x^2
            # For k ≠ 2m with 0 <= k-m < n: contribution is 2*x*a[k-m]
            # Otherwise: 0
            
            # So (a*a)[k] = base_k + 2*x*a[k-m] (k≠2m) or base_k + x^2 (k=2m)
            # where base_k = (a_zero * a_zero)[k]
            
            # We want to minimize max_k (base_k + linear/quadratic term)
            # subject to a[m] = x >= 0 and sum = 1 (so x <= 1)
            
            # This is a 1D problem. Let's do a ternary search or grid search.
            # Actually, let's try a few candidate values.
            
            best_x = old_val
            best_local_obj = np.inf
            
            # The "important" k values are those where base_k is large
            top_k = np.argsort(b_base)[-min(10, 2*n-1):][::-1]
            
            # Try values: 0, current, and values that would make the top peaks equal
            candidates = [0.0, old_val, min(old_val * 2, 1.0), old_val * 0.5]
            
            # For each top k, find x that makes base_k + 2*x*a[k-m] = best possible
            for k in top_k[:5]:
                j = k - m
                if 0 <= j < n and j != m:
                    # base_k + 2*x*a[j] should be minimized
                    # The max is determined by the largest such term
                    # Try x that would reduce this: x = 0 might help if the term is positive
                    pass
            
            # Simple approach: try a grid of x values
            x_grid = np.linspace(0, min(1.0, old_val + 0.01), 20)
            x_grid = np.append(x_grid, [old_val])
            x_grid = np.unique(x_grid)
            
            for x in x_grid:
                a_try = a_zero.copy()
                a_try[m] = x
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                b_try = compute_conv(a_try)
                obj_try = 2.0 * n * float(np.max(b_try))
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_x = x
            
            if best_x != old_val:
                a[m] = best_x
                a = a / np.sum(a)
                b = compute_conv(a)
                obj = 2.0 * n * float(np.max(b))
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
            
            it += 1
        
        if not improved and it > 0:
            # Check if we should continue
            b = compute_conv(a)
            obj = 2.0 * n * float(np.max(b))
            if obj >= best_obj:
                break
    
    return best_a


def lp_minpeak(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix: (a*g)[k] = sum_i a[max(0,k-n+1):min(n,k+1)] * g[...]
    # Actually (a*g)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * g[k-i]
    # For fixed k: sum_i a[i] * g[k-i] where 0 <= k-i < n, i.e., k-n+1 <= i <= k, 0 <= i < n
    
    from scipy.optimize import linprog
    
    # Variables: g[0], ..., g[n-1], M
    # Minimize: M
    # Constraints: (a*g)[k] - M <= 0 for k=0,...,2n-2
    #              sum(g) = 1
    #              g[i] >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Upper bound constraints: A_ub @ vars <= b_ub
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i  # index in g
            A_ub[k, j] = a[i]
        A_ub[k, n] = -1.0  # -M
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s_val = float(np.sum(g_val))
        if s_val > 1e-10:
            return g_val / s_val
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = lp_minpeak(a)
        if g_lp is None:
            break
        
        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 10 == 0:
                print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 15:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 1:
        # Monotonically increasing (key for low peak!)
        a = np.linspace(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Monotonically decreasing
        a = np.linspace(1.0, 0.1, n)
        a = a / np.sum(a)
    elif mode == 3:
        # Exponential (heavy tail)
        a = np.exp(np.linspace(0, 2, n))
        a = a / np.sum(a)
    elif mode == 4:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** 0.5
        a = a / np.sum(a)
    else:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 10) ** 2, 1))
        a = a / np.sum(a)
    return a

"""
Adaptive Log-Sum-Exp Gradient Descent with Frank-Wolfe, Multiplicative Weights,
and LP polishing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. LSE smooth surrogate with geometric temperature annealing for stable gradients
2. FFT-based O(n log n) gradient via cross-correlation (verified formula)
3. Nesterov-accelerated projected gradient on the simplex
4. Frank-Wolfe (conditional gradient) for fast linear-oracle convergence
5. Multiplicative weights (mirror descent) exploiting simplex geometry
6. Monotonically-increasing initializations to exploit asymmetry
7. Multi-scale n exploration with resampling
8. LP polishing for final precision
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def lse_gradient(a, T):
    """
    Compute gradient of T*log(sum(exp(b_k/T))) where b = conv(a,a).
    Returns (gradient, p, b).
    Gradient formula: grad[m] = 2*c[m] - a[m]*p[2m] where c[m]=sum_i a[i]*p[m+i].
    """
    n = len(a)
    b = compute_conv(a)

    # Numerically stable softmax with temperature T
    b_shifted = b.astype(np.float64) - np.max(b)
    exp_b = np.exp(b_shifted / T)
    p = exp_b / np.sum(exp_b)

    # c[m] = sum_{i=0}^{n-1} a[i] * p[m+i]
    # Use np.correlate: np.correlate(p, a, 'valid')[m] = sum_k p[k+m]*a[k] = c[m]
    c = np.correlate(p, a, 'valid')

    # Gradient: grad[m] = 2*c[m] - a[m]*p[2m] (p[2m]=0 if 2m >= 2n-1)
    grad = 2.0 * c
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < (2 * n - 1)
    grad[valid] -= a[valid] * p[diag_idx[valid]]

    return grad, p, b


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_gd_nesterov(a, time_budget=60.0, T_start=0.5, T_end=0.005, verbose=False):
    """
    Nesterov-accelerated LSE gradient descent on the simplex.
    Temperature anneals geometrically from T_start to T_end.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    b = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(b))
    start = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    beta = 0.85
    eta = 0.05
    patience = 0
    max_patience = 80
    stall_rounds = 0

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        # Geometric temperature annealing
        frac = min(it / 10000.0, 1.0)
        T = T_start * (T_end / T_start) ** frac

        # Nesterov lookahead
        grad_y, p_y, b_y = lse_gradient(y, T)

        # Momentum update (heavy ball style for stability)
        v = beta * v + grad_y
        y = a - eta * v
        y = project_simplex(y)

        # Evaluate
        b_new = compute_conv(y)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = y.copy()
            a = y.copy()
            patience = 0
            eta = min(eta * 1.008, 0.5)
            beta = min(beta + 0.0005, 0.95)
        else:
            patience += 1
            eta *= 0.97
            if patience > max_patience:
                stall_rounds += 1
                if stall_rounds > 5:
                    break
                a = best_a.copy()
                y = a.copy()
                v = np.zeros(n)
                eta = 0.05
                beta = 0.85
                patience = 0
                max_patience = min(max_patience + 40, 300)

        if verbose and it % 5000 == 0:
            print(f"              LSE-Nest it{it}: obj={best_obj:.8f}, T={T:.5f}, eta={eta:.5f}")

        if eta < 1e-6:
            break

    return best_a


def frank_wolfe(a, time_budget=30.0, verbose=False):
    """
    Frank-Wolfe (conditional gradient) for min max_k (a*a)_k over the simplex.
    Uses subgradient of the active constraint and vertex linear oracle.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    b = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(b))
    start = time.monotonic()

    # Running average for Polyak-style
    avg_a = a.copy()
    avg_obj = best_obj

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv(a)
        k_star = int(np.argmax(b))

        # Subgradient at a: grad[m] = 2*a[k_star-m] (if valid), with diagonal correction
        grad = np.zeros(n)
        m_arr = np.arange(n)
        j_arr = k_star - m_arr
        valid = (j_arr >= 0) & (j_arr < n)
        grad[valid] = 2.0 * a[j_arr[valid]]
        diag = (2 * m_arr == k_star)
        if np.any(diag & valid):
            grad[diag & valid] = a[j_arr[diag & valid]]

        # Center gradient (tangent to simplex)
        grad_c = grad - np.mean(grad)

        # Linear oracle: vertex e_j where j = argmin grad_c
        j_min = int(np.argmin(grad_c))

        # Step size: standard FW 2/(t+2)
        t = 2.0 / (it + 2.0)
        a_new = (1.0 - t) * a
        a_new[j_min] += t
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Update running average
        avg_a = (it * avg_a + a_new) / (it + 1)

        b_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(b_new))
        b_avg = compute_conv(avg_a)
        obj_avg = 2.0 * n * float(np.max(b_avg))

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
        if obj_avg < best_obj - 1e-11:
            best_obj = obj_avg
            best_a = avg_a.copy()
        a = a_new

        if verbose and it % 5000 == 0:
            print(f"              FW it{it}: obj={best_obj:.8f}, k*={k_star}")

    return best_a


def multiplicative_weights(a, time_budget=30.0, verbose=False):
    """
    Mirror descent withnegative entropy (multiplicative weights) on the simplex.
    """
    n = len(a)
    a = a.astype(np.float64).copy() + 1e-12
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    b = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(b))
    start = time.monotonic()

    eta = 0.3
    patience = 0

    for it in range(30000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv(a)
        k_star = int(np.argmax(b))

        # Subgradient
        grad = np.zeros(n)
        m_arr = np.arange(n)
        j_arr = k_star - m_arr
        valid = (j_arr >= 0) & (j_arr < n)
        grad[valid] = 2.0 * a[j_arr[valid]]
        diag = (2 * m_arr == k_star)
        if np.any(diag & valid):
            grad[diag & valid] = a[j_arr[diag & valid]]

        # Center
        grad = grad - np.mean(grad)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        b_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            eta = min(eta * 1.01, 3.0)
        else:
            patience += 1
            eta *= 0.95
            if patience > 150:
                break

        if verbose and it % 5000 == 0:
            print(f"              MW it{it}: obj={best_obj:.8f}, eta={eta:.4f}")

    return best_a


def lp_minpeak(a):

        return None  # Too slow for LP
    a = a / np.sum(a)

    from scipy.optimize import linprog

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        for i in range(i_lo, i_hi):
            j = k - i
            A_ub[k, j] = a[i]
        A_ub[k, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_val = float(np.sum(g_val))
            if s_val > 1e-10:
                return g_val / s_val
    except Exception:
        pass
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        g_lp = lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 10 == 0:
                print(f"              LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 20:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations with emphasis on asymmetric shapes."""
    mode = rng.integers(0, 8)
    if mode == 0:
        # Monotonically increasing (key for exploiting asymmetry)
        a = np.linspace(0.05, 1.0, n) ** rng.uniform(0.5, 2.0)
        a = a / np.sum(a)
    elif mode == 1:
        # Monotonically decreasing
        a = np.linspace(1.0, 0.05, n) ** rng.uniform(0.5, 2.0)
        a = a / np.sum(a)
    elif mode == 2:
        # Exponential ramp
        a = np.exp(np.linspace(0, rng.uniform(1.0, 3.0), n))
        a = a / np.sum(a)
    elif mode == 3:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** rng.uniform(0.3, 1.5)
        a = a / np.sum(a)
    elif mode == 4:
        # Uniform
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 5:
        # Smooth asymmetric (sigmoid ramp)
        x = np.linspace(-3, 3, n)
        a = 1.0 / (1.0 + np.exp(-x * rng.uniform(0.5, 2.0)))
        a = a / np.sum(a)
    elif mode == 6:
        # Truncated normal, asymmetric
        a = rng.normal(0, 1, n)
        a = np.maximum(a, 0) + 0.01
        a = a / np.sum(a)
    else:
        # Cosine window (smooth)
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) ** rng.uniform(1, 4) + 0.05
        a = a / np.sum(a)
    return a


def combined_optimize(a, time_budget=120.0, verbose=False):
    """
    Combined strategy: LSE-GD -> FW -> MW -> LSE-GD (low T).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LSE-GD (broad temperature)
    remaining = max(0.0, time_budget - (time.monotonic() - start))
    if remaining > 15:
        alloc = remaining * 0.35
        a1 = lse_gd_nesterov(a, time_budget=alloc, T_start=0.5, T_end=0.05, verbose=verbose)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            a = a1.copy()
            if verbose:
                print(f"              Combo LSE1: {best_obj:.8f}")

    # Phase 2: Frank-Wolfe
    remaining = max(0.0, time_budget - (time.monotonic() - start))
    if remaining > 15:
        alloc = remaining * 0.30
        a2 = frank_wolfe(a, time_budget=alloc, verbose=verbose)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            a = a2.copy()
            if verbose:
                print(f"              Combo FW: {best_obj:.8f}")

    # Phase 3: Multiplicative weights
    remaining = max(0.0, time_budget - (time.monotonic() - start))
    if remaining > 10:
        alloc = remaining * 0.20
        a3 = multiplicative_weights(a, time_budget=alloc, verbose=verbose)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            a = a3.copy()
            if verbose:
                print(f"              Combo MW: {best_obj:.8f}")

    # Phase 4: LSE-GD (narrow temperature, fine-tuning)
    remaining = max(0.0, time_budget - (time.monotonic() - start))
    if remaining > 10:
        alloc = remaining * 0.30
        a4 = lse_gd_nesterov(a, time_budget=alloc, T_start=0.1, T_end=0.005, verbose=verbose)
        obj4 = compute_obj(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            if verbose:
                print(f"              Combo LSE2: {best_obj:.8f}")

    return best_a
