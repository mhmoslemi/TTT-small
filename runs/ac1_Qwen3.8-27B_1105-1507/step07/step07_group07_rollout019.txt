Optimized search for low-peak non-negative autocorrelation sequences.
Combines smoothed gradient, coordinate descent, LP alternating, and spectral methods.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
