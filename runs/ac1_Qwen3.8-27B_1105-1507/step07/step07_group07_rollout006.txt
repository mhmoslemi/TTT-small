def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def solve_sqp_lp(a):
    """SQP-style LP: linearize conv(a,a) at current a, solve min M LP."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv_cur = compute_conv(a_norm)
    
    # Gradient of conv(a,a)[k] w.r.t. a[i] = 2*a[k-i] (reversed)
    # Linearized: C_k(a) ≈ C_k(a*) + 2 * sum_i a*[k-i] * (a[i] - a*[i])
    # = C_k(a*) - 2*conv(a*,a*)[k] + 2*conv(a*,a)[k]
    # So constraint: C_k(a*) - 2*C_k(a*) + 2*conv(a*,a)[k] <= M
    # i.e., 2*conv(a*,a)[k] - C_k(a*) <= M
    
    A = build_conv_matrix(a_norm)
    
    try:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        # 2*(A@g)[k] - conv_cur[k] <= M
        constraints = [2 * (A @ g) - conv_cur <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob = cp.Problem(cp.Minimize(M), constraints)
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    
    # Fallback to standard LP
    g_std, _ = solve_lp_minpeak(a)
    return g_std


def lp_alternate_step(a, use_sqp=False):
    """One LP/SQP alternating step with adaptive line search."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    if use_sqp:
        g_new = solve_sqp_lp(a_norm)
    else:
        g_new, _ = solve_lp_minpeak(a_norm)
    
    if g_new is None:
        return a_norm, compute_obj(a_norm)
    
    best_a = a_norm.copy()
    best_obj = compute_obj(a_norm)
    
    # Adaptive line search: try t=1 first, then binary search
    for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.5, 0.7, 0.3, 0.9]:
        a_try = (1.0 - t) * a_norm + t * g_new
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a, best_obj


def lp_alternate_optimize(a, time_budget=60.0, verbose=False):
    """LP/SQP alternating with adaptive method switching and perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    use_sqp = False
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        # Alternate between standard LP and SQP LP
        use_sqp = (stall > 3) or (int((time.monotonic() - start) * 2) % 3 == 0)
        
        a_new, obj_new = lp_alternate_step(a, use_sqp=use_sqp)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"      LP{'-SQP' if use_sqp else ''}: obj={best_obj:.10f} (t={elapsed:.1f}s)")
        else:
            stall += 1
            if stall > 5:
                # Perturb to escape local minimum
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 5, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
    
    return best_a


def smoothed_obj_grad(x, tau):
    """Smoothed objective and gradient using log-sum-exp approximation."""
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = float(np.sum(exp_c))
    w = exp_c / sum_exp
    
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    
    # Gradient via FFT
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """L-BFGS-B on smoothed objective with multi-temperature annealing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    taus = [20.0 / n, 10.0 / n, 5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
    
    x = np.log(a) - np.log(a).max()
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_grad(xv, tau)
            return g
        
        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50})
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={best_obj:.10f}")
        except Exception:
            continue
    
    return best_a


def expgrad_step(a, eta, k_star):
    """One exponentiated gradient step targeting peak at k_star."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
    
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / np.sum(a_new)
    return a_new


def expgrad_optimize(a_init, time_budget=30.0, verbose=False):
    """Exponentiated gradient with multi-peak targeting."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 2.0 / n
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        a_new = expgrad_step(a, eta, k_star)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = max(eta * 0.9995, 0.05 / n)
        else:
            # Try targeting second-highest peak
            conv_sorted = np.sort(conv)[::-1]
            if len(conv_sorted) > 1 and conv_sorted[1] > conv_sorted[0] * 0.9:
                # Find second peak location
                top2 = np.argsort(conv)[::-1][:2]
                k2 = int(top2[1])
                a_new = expgrad_step(a, eta * 0.5, k2)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    continue
            
            eta *= 0.85
            if eta < 0.02 / n:
                rng = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                eta = 2.0 / n
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def freq_init(n, rng, mode=0):
    """Frequency-domain based initialization."""
    if mode == 0:
        # Flat spectrum with random phases (gives white noise time domain)
        spec_mag = np.ones(n // 2 + 1)
        spec_phase = 2 * np.pi * rng.random(n // 2 + 1)
        spec = spec_mag * np.exp(1j * spec_phase)
        spec[0] = spec_mag[0]  # DC component real
        if n % 2 == 0 and n // 2 < len(spec):
            spec[n // 2] = spec_mag[n // 2]
        a = irfft(spec, n * 4)[:n]
    elif mode == 1:
        # Low-pass filtered random spectrum
        freqs = np.arange(n // 2 + 1) / (n // 2)
        spec_mag = np.exp(-2 * freqs ** 2)
        spec_phase = 2 * np.pi * rng.random(n // 2 + 1)
        spec = spec_mag * np.exp(1j * spec_phase)
        spec[0] = spec_mag[0]
        a = irfft(spec, n * 4)[:n]
    elif mode == 2:
        # Bandpass spectrum
        freqs = np.arange(n // 2 + 1) / (n // 2)
        center = 0.3 + 0.4 * rng.random()
        width = 0.1 + 0.2 * rng.random()
        spec_mag = np.exp(-((freqs - center) ** 2) / (2 * width ** 2))
        spec_phase = 2 * np.pi * rng.random(n // 2 + 1)
        spec = spec_mag * np.exp(1j * spec_phase)
        spec[0] = spec_mag[0]
        a = irfft(spec, n * 4)[:n]
    else:
        # Specific harmonic structure
        a = np.ones(n) / n
        for k in range(3):
            freq = rng.uniform(0.1, 0.5)
            amp = rng.uniform(0.1, 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def structured_init(n, rng):
    """Structured initializations based on known good patterns."""
    mode = rng.integers(0, 8)
    x = np.arange(n) / n
    
    if mode == 0:
        # Sigmoid-like (asymmetric)
        a = 1.0 / (1.0 + np.exp(-5 * rng.uniform(-2, 2) * (x - 0.5)))
    elif mode == 1:
        # Asymmetric Gaussian
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.3)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 2:
        # Power law (asymmetric)
        alpha = rng.uniform(0.5, 3.0)
        a = x ** alpha if rng.random() > 0.5 else (1 - x) ** alpha
    elif mode == 3:
        # Two Gaussian bumps
        c1, c2 = rng.uniform(0, 0.4), rng.uniform(0.6, 1.0)
        a = np.exp(-((x - c1) ** 2) / 0.02) + np.exp(-((x - c2) ** 2) / 0.02)
    elif mode == 4:
        # Cosine window (asymmetric)
        a = np.cos(np.pi * x * rng.uniform(0.5, 1.5)) ** 2
    elif mode == 5:
        # Step + smooth
        thresh = rng.uniform(0.3, 0.7)
        a = np.where(x < thresh, 1.0, 0.3)
        a = np.convolve(a, np.ones(20) / 20, mode='same')
    elif mode == 6:
        # Sine-squared with offset
        freq = rng.uniform(1, 5)
        a = np.sin(np.pi * freq * x + rng.uniform(0, np.pi)) ** 2 + 0.1
    else:
        # Random smooth (low-pass)
        a = rng.uniform(0, 1, n)
        kernel_size = max(5, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0.001)
    a = a / np.sum(a)
    return a


def diverse_init(n, rng):
    """Try multiple initialization strategies and return the best."""
    candidates = []
    candidates.append(structured_init(n, rng))
    candidates.append(freq_init(n, rng, mode=rng.integers(0, 4)))
    candidates.append(resample_sequence(np.ones(10), n))  # uniform
    
    # Add perturbed versions
    base = candidates[0]
    for _ in range(3):
        pert = base * np.exp(rng.normal(0, 0.1, n))
        pert = np.maximum(pert, 0.001)
        pert = pert / np.sum(pert)
        candidates.append(pert)
    
    best_a = candidates[0]
    best_obj = compute_obj(best_a)
    for c in candidates[1:]:
        obj = compute_obj(c)
        if obj < best_obj:
            best_obj = obj
            best_a = c
    
    return best_a


def paired_lp_optimize(a, time_budget=60.0, verbose=False):
    """Enhanced LP alternating with paired steps (LP then SQP) and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_a = a.copy()
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        iteration += 1
        use_sqp = (iteration % 3 == 0)
        
        a_new, obj_new = lp_alternate_step(a, use_sqp=use_sqp)
        
        if obj_new < best_obj - 1e-15:
            # Check for oscillation (if we're going back and forth)
            momentum = 0.0
            if iteration > 2:
                # Simple momentum: if direction is consistent, amplify
                pass
            
            best_obj = obj_new
            best_a = a_new.copy()
            prev_a = a.copy()
            a = a_new
            
            if verbose and iteration % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"      PAIR-LP iter={iteration}: obj={best_obj:.10f} (t={elapsed:.1f}s)")
        else:
            # Try small perturbation + LP
            rng = np.random.default_rng(iteration * 7919 + int(time.monotonic() * 100) % 10000)
            scale = 0.01 + 0.005 * (iteration % 5)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            
            a_new2, obj_new2 = lp_alternate_step(a_pert, use_sqp=False)
            if obj_new2 < best_obj:
                best_obj = obj_new2
                best_a = a_new2.copy()
                a = a_new2
            else:
                a = best_a
    
    return best_a

            a_r = a_r / np.sum(a_r)
        else:
            a_r = diverse_init(n_final, rng)
        
        # Optimize: LP + gradient
        alloc = restart_budget_each
        a_opt = paired_lp_optimize(a_r, time_budget=alloc * 0.6)
        remaining_r = deadline - time.monotonic()
        if remaining_r > alloc * 0.3:
            a_opt = gradient_optimize(a_opt, time_budget=min(alloc * 0.4, remaining_r * 0.2))
        
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_final = len(best_seq)
            print(f"  Restart {r} (n={len(a_opt)}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (n={len(a_opt)}): obj={obj:.10f}")
    
    # === Phase 7: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 20:
        # LP
        lp_alloc = min(remaining * 0.4, 200.0)
        a_f = paired_lp_optimize(best_seq, time_budget=lp_alloc, verbose=True)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.10f}")
        
        remaining = deadline - time.monotonic()
        if remaining > 15:
            # Gradient
            grad_alloc = min(remaining * 0.3, 100.0)
            a_g = gradient_optimize(best_seq, time_budget=grad_alloc)
            obj = compute_obj(a_g)
            if obj < best_score:
                best_score = obj
                best_seq = a_g.copy()
                print(f"  Final gradient: obj={best_score:.10f}")
        
        remaining = deadline - time.monotonic()
        if remaining > 10:
            # One more LP round
            lp_alloc = min(remaining * 0.5, 100.0)
            a_f2 = paired_lp_optimize(best_seq, time_budget=lp_alloc)
            obj = compute_obj(a_f2)
            if obj < best_score:
                best_score = obj
                best_seq = a_f2.copy()
                print(f"  Final LP2: obj={best_score:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = evaluate_sequence(result)
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
