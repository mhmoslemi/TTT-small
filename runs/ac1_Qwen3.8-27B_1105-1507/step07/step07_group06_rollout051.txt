"""
Multi-strategy optimization for flat-autocorrelation non-negative sequences.
Uses subgradient method with momentum, active-set LP, simulated annealing,
Fourier-based diverse initializations, and multi-length exploration.
Key innovations over previous approach:
1. Direct subgradient on max (with momentum) instead of smoothed L-BFGS-B
2. Active-set LP (only binding constraints) for speed
3. Simulated annealing acceptance to escape local minima
4. Structured Fourier perturbations for diversity
5. Multi-length exploration (n=400 to n=1500)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def gradient_max_conv(a, k):
    """Compute gradient of c_k = sum_i a[i]*a[k-i] w.r.t. a.
    Result: (grad)_p = 2*a[k-p] if 0<=k-p<n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    # valid p: 0 <= p < n and 0 <= k-p < n
    p_lo = max(0, k - n + 1)
    p_hi = min(n - 1, k)
    if p_lo <= p_hi:
        grad[p_lo:p_hi + 1] = 2.0 * a[k - p_hi:k - p_lo - 1:-1] if k - p_hi != k - p_lo - 1 else 2.0 * a[k - p_lo:k - p_hi + 1][::-1]
    # Simpler: just do it directly
    grad = np.zeros(n)
    for p in range(p_lo, p_hi + 1):
        grad[p] = 2.0 * a[k - p]
    return grad


def gradient_max_conv_fast(a, k):
    """Vectorized gradient of c_k w.r.t. a."""
    n = len(a)
    p_lo = max(0, k - n + 1)
    p_hi = min(n - 1, k)
    if p_lo > p_hi:
        return np.zeros(n)
    grad = np.zeros(n)
    # grad[p] = 2 * a[k-p] for p in [p_lo, p_hi]
    # k-p for p in [p_lo, p_hi] is [k-p_lo, k-p_hi] in reverse
    indices = k - np.arange(p_lo, p_hi + 1)
    grad[p_lo:p_hi + 1] = 2.0 * a[indices]
    return grad


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    s = np.sum(v)
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return v * (target_sum / s)


def build_conv_matrix_active(a, active_k):
    """Build sparse-ish matrix A active_k x n such that A @ g = conv(a, g) at active_k positions."""
    n = len(a)
    K = len(active_k)
    A = np.zeros((K, n))
    for idx, k in enumerate(active_k):
        p_lo = max(0, k - n + 1)
        p_hi = min(n - 1, k)
        if p_lo <= p_hi:
            A[idx, p_lo:p_hi + 1] = a[k - p_hi:k - p_lo - 1:-1] if p_hi - p_lo > 0 else a[k - p_lo]
    return A


def solve_lp_active(a, active_k, M_bound=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_active(a, active_k)
    K = len(active_k)
    
    # Variables: g_0, ..., g_{n-1}, M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def subgradient_optimize(a, time_budget=60.0, momentum=0.9, verbose=False):
    """Subgradient method with momentum on the simplex, targeting top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)  # momentum buffer
    K = 5  # number of top peaks to target
    base_eta = 2.0 / np.sqrt(n)
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        obj = 2.0 * n * np.max(conv)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Get top-K peak indices
        top_k = np.argsort(conv)[::-1][:K]
        
        # Combined gradient (average of top-K gradients)
        g = np.zeros(n)
        for k in top_k:
            g += gradient_max_conv_fast(a, k)
        g /= K
        
        # Project gradient to tangent space of simplex
        g_proj = g - np.mean(g)
        
        # Momentum
        v = momentum * v - base_eta * g_proj
        
        # Step
        a_new = a + v  # Note: we're minimizing, so subtract gradient -> add negative
        # Wait, v already has the negative sign from -base_eta * g_proj
        # Actually: v = momentum * v - eta * g_proj, so a_new = a + v is correct
        # (since we want to go in direction -gradient)
        
        a_new = project_simplex(a_new, 1.0)
        
        # Check improvement
        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * np.max(conv_new)
        
        # Simulated annealing: accept with probability if worse
        if obj_new <= obj or (it > 100 and np.random.random() < np.exp(-(obj_new - obj) * 50.0)):
            a = a_new
        else:
            # Revert: undo the step
            a_old = a
            # Just keep a as is, reset momentum partially
            v *= 0.5
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG it={it}, obj={obj:.8f}, best={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def lp_optimize_active(a, time_budget=60.0, verbose=False):
    """LP-based optimization with active set selection."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = np.max(conv)
        
        # Select active constraints: bins where c_k > 0.7 * max_c
        threshold = 0.7 * max_c
        active_k = np.where(conv > threshold)[0]
        
        if len(active_k) < 3:
            active_k = np.argsort(conv)[::-1][:10]
        
        g_lp, M_lp = solve_lp_active(a, active_k, M_bound=None)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_a_local = a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = project_simplex(a_try, 1.0)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose and it % 5 == 0:
                print(f"    LP it={it}, obj={best_obj:.8f}, active={len(active_k)}")
        else:
            # Perturb to escape
            if it % 5 == 0:
                rng = np.random.default_rng(it * 104729 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = project_simplex(a, 1.0)
    
    return best_a, best_obj


def fourier_perturb(a, rng, strength=0.3):
    """Perturb a by adding random Fourier modes and re-projecting."""
    n = len(a)
    a = a.copy()
    # Add random low-frequency Fourier modes
    fa = rfft(a)
    n_freqs = max(3, n // 20)
    for _ in range(n_freqs):
        f = rng.integers(1, n // 2)
        amp = strength * rng.uniform(-1, 1) / (f + 1)
        phase = rng.uniform(0, 2 * np.pi)
        fa[f] += amp * np.exp(1j * phase)
        if f < n // 2:
            fa[-f] = np.conj(fa[f])
    a_new = irfft(fa, n)
    return project_simplex(a_new, float(np.sum(a)))


def make_ramp_init(n, rng):
    """Create a random ramp-like initialization."""
    type_ = rng.integers(0, 5)
    if type_ == 0:
        # Linear ramp
        slope = rng.uniform(1.0, 3.0)
        a = np.linspace(1.0, slope, n)
    elif type_ == 1:
        # Exponential ramp
        rate = rng.uniform(1.0, 4.0)
        a = np.exp(np.linspace(0, rate, n))
    elif type_ == 2:
        # Two bumps (asymmetric)
        p1 = rng.uniform(0.1, 0.4)
        p2 = rng.uniform(0.6, 0.9)
        s1 = rng.uniform(0.05, 0.2)
        s2 = rng.uniform(0.05, 0.2)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - p1) / s1) ** 2) + 0.5 * np.exp(-((x - p2) / s2) ** 2)
    elif type_ == 3:
        # Cosine shape (anti-symmetric)
        f = rng.uniform(1, 5)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    else:
        # Power law
        p = rng.uniform(0.5, 3.0)
        x = np.linspace(0.01, 1, n)
        a = x ** p
    return project_simplex(a, 1.0)


def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / 1.0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(a_new, float(np.sum(a)))


def hybrid_optimize(a, time_budget=60.0, verbose=False):
    """Combined subgradient + LP optimization."""
    n = len(a)
    a = project_simplex(a, 1.0)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Alternate between subgradient and LP
    phase = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        if phase == 0:
            # Subgradient phase
            alloc = min(remaining * 0.4, 20.0)
            a_new, obj_new = subgradient_optimize(best_a, time_budget=alloc)
            phase = 1
        else:
            # LP phase
            alloc = min(remaining * 0.4, 25.0)
            a_new, obj_new = lp_optimize_active(best_a, time_budget=alloc)
            phase = 0
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"    Hybrid: obj={best_obj:.8f}")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_init = 800
        best_seq = make_ramp_init(n_init, rng)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    # Phase 1: Hybrid optimization from parent
    print(f"\n=== Phase 1: Hybrid (SG+LP) from parent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 120.0)
        a_opt, obj_opt = hybrid_optimize(best_seq, time_budget=alloc, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_parent = len(best_seq)
            print(f"  Hybrid parent: obj={best_score:.8f} **")
        else:
            print(f"  Hybrid parent: obj={obj_opt:.8f}")
    
    # Phase 2: Diverse initializations with hybrid optimization
    print(f"\n=== Phase 2: Diverse Initializations ===")
    n_trials = 8
    for trial in range(n_trials):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        # Choose n for this trial
        if trial < 3:
            n_try = n_parent
        elif trial < 5:
            n_try = max(400, n_parent + rng.integers(-100, 300))
        else:
            n_try = rng.integers(600, 1500)
        
        n_try = int(min(n_try, 1500))
        
        # Create diverse initialization
        if trial == 0:
            # Perturbed parent
            a_init = best_seq[:n_try] if n_try <= n_parent else resample_sequence(best_seq, n_try)
            a_init = a_init * np.exp(rng.normal(0, 0.1, n_try))
            a_init = project_simplex(a_init, 1.0)
        elif trial == 1:
            # Fourier-perturbed parent
            a_init = resample_sequence(best_seq, n_try)
            a_init = fourier_perturb(a_init, rng, strength=0.5)
        elif trial == 2:
            # Strong Fourier perturbation
            a_init = resample_sequence(best_seq, n_try)
            a_init = fourier_perturb(a_init, rng, strength=1.0)
            a_init = fourier_perturb(a_init, rng, strength=0.5)
        else:
            # Random structured init
            a_init = make_ramp_init(n_try, rng)
        
        alloc = min(remaining * 0.15, 60.0)
        a_opt, obj_opt = hybrid_optimize(a_init, time_budget=alloc)
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_parent = len(best_seq)
            print(f"  Trial {trial} (n={n_try}): obj={best_score:.8f} **")
        else:
            print(f"  Trial {trial} (n={n_try}): obj={obj_opt:.8f}")
    
    # Phase 3: Intensive subgradient with annealing on best
    print(f"\n=== Phase 3: Intensive Subgradient + Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 80.0)
        a_opt, obj_opt = subgradient_optimize(best_seq, time_budget=alloc, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  SG anneal: obj={best_score:.8f} **")
        else:
            print(f"  SG anneal: obj={obj_opt:.8f}")
    
    # Phase 4: LP polish
    print(f"\n=== Phase 4: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 60.0)
        a_opt, obj_opt = lp_optimize_active(best_seq, time_budget=alloc, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  LP polish: obj={best_score:.8f} **")
        else:
            print(f"  LP polish: obj={obj_opt:.8f}")
    
    # Phase 5: Multi-length exploration
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 200),
        max(400, n_cur - 100),
        n_cur,
        n_cur + 100,
        n_cur + 200,
        min(1500, n_cur + 400),
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        
        if n_try == n_cur:
            continue
        
        a_try = resample_sequence(best_seq, n_try)
        
        # Quick hybrid
        alloc = min(remaining * 0.15, 30.0)
        a_opt, obj_opt = hybrid_optimize(a_try, time_budget=alloc)
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # Phase 6: Final intensive polish (alternate SG and LP)
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 80.0)
        a_opt, obj_opt = hybrid_optimize(best_seq, time_budget=alloc, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final hybrid: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_opt, obj_opt = lp_optimize_active(best_seq, time_budget=alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final LP: obj={best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
