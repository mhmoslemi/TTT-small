"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses structured parametric initializations, subgradient descent targeting the convolution peak,
LP alternating optimization, frequency-domain refinement, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def compute_obj_normalized(a):
    """Compute objective assuming sum(a) = 1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=60.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_work = a.copy()
    s = float(np.sum(a_work))
    if s < 1e-10:
        return None, None
    a_work = a_work / s
    A = build_conv_matrix(a_work)

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def subgradient_descent(a, time_budget=60.0, verbose=False):
    """
    Subgradient descent on the peak of auto-convolution.
    The subgradient of max_k conv(a,a)[k] w.r.t. a[j] is:
      a[k*-j] + a[j-k*] (for valid indices), where k* = argmax conv.
    We center the gradient to preserve the sum, then project onto a>=0.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    lr = 1.0 / n
    patience = 0
    max_patience = 200

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: d/d a[j] of conv(k*) = a[k*-j] + a[j-k*] (valid indices)
        grad = np.zeros(n)
        for j in range(n):
            idx1 = k_star - j
            idx2 = j - k_star
            if 0 <= idx1 < n:
                grad[j] += a[idx1]
            if 0 <= idx2 < n:
                grad[j] += a[idx2]

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Step
        a_new = a + lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr = min(lr * 1.02, 5.0 / n)
            if verbose and it % 100 == 0:
                print(f"  SG it{it}: obj={best_obj:.10f}, lr={lr:.6f}, k*={k_star}")
        else:
            patience += 1
            lr *= 0.95
            if patience > max_patience:
                # Restart with perturbation
                rng = np.random.default_rng(it * 13 + 7)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = normalize(a)
                best_obj = compute_obj_normalized(a)
                lr = 1.0 / n
                patience = 0
                if verbose:
                    print(f"  SG restart at it{it}: obj={best_obj:.10f}")

    return best_a

            a = best_a

            break
        size = 1
        while size < 2 * n - 1:
            size <<= 1

        A = rfft(a, size)
        # Apply Gaussian taper in frequency (reduce high freq)
        freqs = np.fft.rfftfreq(size, d=1.0)
        max_freq = max(abs(freqs))
        taper = np.exp(-beta * (freqs / max_freq) ** 2)
        # Keep DC component unchanged
        taper[0] = 1.0

        A_tapered = A * taper
        a_tapered = irfft(A_tapered, size)[:n]
        a_tapered = np.maximum(a_tapered, 0)
        s_t = float(np.sum(a_tapered))
        if s_t < 1e-10:
            continue
        a_tapered = a_tapered / s_t

        obj_t = compute_obj_normalized(a_tapered)
        # Blend
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = normalize((1 - t) * a + t * a_tapered)
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try

    return best_a


def structured_init(n, mode, rng=None):
    """Generate structured initial sequences."""
    if rng is None:
        rng = np.random.default_rng(42)

    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Linear ramp (ascending)
        a = np.arange(1, n + 1, dtype=float)
    elif mode == 2:
        # Linear ramp (descending)
        a = np.arange(n, 0, dtype=float)
    elif mode == 3:
        # Quadratic (parabola)
        x = np.linspace(0, 1, n)
        a = 1.0 - (2 * x - 1) ** 2
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(0.5 * x) ** 2
    elif mode == 5:
        # Geometric (ascending)
        r = float(rng.uniform(1.001, 1.02))
        a = r ** np.arange(n)
    elif mode == 6:
        # Geometric (descending)
        r = float(rng.uniform(0.98, 0.999))
        a = r ** np.arange(n)
    elif mode == 7:
        # Asymmetric: linear * geometric
        x = np.arange(n, dtype=float)
        a = (x + 1) * (0.99 ** x)
    elif mode == 8:
        # Asymmetric: quadratic * geometric
        x = np.arange(n, dtype=float)
        norm_x = x / (n - 1)
        a = (1.0 + norm_x) * (0.95 ** x)
    elif mode == 9:
        # Sine weighted
        x = np.arange(n, dtype=float)
        a = np.sin(np.pi * (x + 1) / (n + 1)) ** 2 * (0.995 ** x)
    elif mode == 10:
        # Bimodal asymmetric
        x = np.arange(n, dtype=float)
        a = np.exp(-((x - 0.3*n)**2) / (2*(0.1*n)**2)) + 0.5*np.exp(-((x - 0.7*n)**2) / (2*(0.08*n)**2))
    elif mode == 11:
        # Power law
        x = np.arange(1, n + 1, dtype=float)
        p = float(rng.uniform(0.3, 2.0))
        a = x ** p
    elif mode == 12:
        # Exponential decay from center
        x = np.arange(n, dtype=float)
        c = n * float(rng.uniform(0.3, 0.7))
        a = np.exp(-0.5 * (x - c) ** 2 / (n * 0.2) ** 2)
    elif mode == 13:
        # Triangle asymmetric
        x = np.arange(n, dtype=float)
        peak = n * float(rng.uniform(0.2, 0.8))
        a = np.exp(-np.abs(x - peak) / (n * 0.3))
    else:
        # Random smooth
        a = rng.uniform(0.5, 1.5, n)
        kernel_size = max(3, n // 20)
        kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1)) ** 2 / (kernel_size / 3) ** 2)
        a = np.convolve(a, kernel / kernel.sum(), mode='same')

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def optimize_params_parametric(n, time_budget=30.0):
    """
    Optimize parameters of a[i] = (A + B*i) * exp(C*i) * (1 - D*|2i/n - 1|^E)
    to minimize the objective directly.
    """
    best_a = None
    best_obj = np.inf
    start = time.monotonic()

    # Parameter ranges
    A_range = (0.5, 5.0)
    B_range = (-0.01, 0.01)
    C_range = (-0.02, 0.02)
    D_range = (0.0, 2.0)
    E_range = (0.5, 5.0)

    best_params = None
    best_val = np.inf
    rng = np.random.default_rng(123)

    # Random search over parameters
    max_evals = int(2000)
    for ev in range(max_evals):
        if time.monotonic() - start > time_budget:
            break
        A = rng.uniform(*A_range)
        B = rng.uniform(*B_range)
        C = rng.uniform(*C_range)
        D = rng.uniform(*D_range)
        E = rng.uniform(*E_range)

        x = np.arange(n, dtype=float)
        a = (A + B * x) * np.exp(C * x)
        center_term = 1.0 - D * np.abs(2 * x / n - 1.0) ** E
        a = a * center_term
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            continue
        a = a / s
        obj = compute_obj_normalized(a)
        if obj < best_val:
            best_val = obj
            best_params = (A, B, C, D, E)
            best_a = a.copy()

    # Local refinement with scipy
    if best_params is not None and time.monotonic() - start < time_budget - 5:
        def param_obj(params):
            A, B, C, D, E = params
            if A < 0 or D < 0 or E <= 0:
                return 10.0
            x = np.arange(n, dtype=float)
            a = (A + B * x) * np.exp(C * x)
            center_term = 1.0 - D * np.abs(2 * x / n - 1.0) ** E
            a = a * center_term
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s < 1e-10:
                return 10.0
            a = a / s
            return compute_obj_normalized(a)

        try:
            res = minimize(param_obj, best_params, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-8,
                                    'adaptive': True})
            if res.fun < best_val:
                best_val = res.fun
                A, B, C, D, E = res.x
                x = np.arange(n, dtype=float)
                a = (A + B * x) * np.exp(C * x)
                center_term = 1.0 - D * np.abs(2 * x / n - 1.0) ** E
                a = np.maximum(a * center_term, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    best_a = a / s
                    best_obj = best_val
        except Exception:
            pass

    if best_a is not None:
        return best_a
    return None


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple strategies:
    1. Structured parametric initializations
    2. Subgradient descent on convolution peak
    3. LP alternating optimization
    4. Frequency-domain refinement
    5. Multi-length exploration
    6. Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    def try_update(a_arr, label=""):
        nonlocal best_seq, best_score
        if a_arr is None:
            return
        a_list = [float(max(0.0, min(1000.0, x))) for x in a_arr]
        s_check = sum(a_list)
        if s_check < 0.01:
            return
        score = evaluate_sequence(a_list)
        if score < np.inf and score < best_score:
            best_score = score
            best_seq = a_arr.copy()
            print(f"  ** {label}: n={len(a_arr)}, score={score:.8f} **")
        elif label:
            print(f"  {label}: n={len(a_arr)}, score={score:.8f}")

    # Phase 0: Try parent and structured initializations
    print("=== Phase 0: Initializations ===")

    # Try parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr
                try_update(parent, "Parent")
    except Exception as e:
        print(f"  Parent not available: {e}")

    # Try constant baseline
    for n_const in [400, 800]:
        if time.monotonic() > deadline:
            break
        a_const = np.ones(n_const)
        try_update(a_const, f"Const n={n_const}")

    # Try structured initializations
    n_values = [400, 600, 800, 1000, 1200, 1500]
    modes = list(range(14))

    for n_try in n_values:
        if time.monotonic() > deadline:
            break
        for mode in modes:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            a_init = structured_init(n_try, mode, rng)
            # Quick subgradient for just a few seconds
            a_quick = subgradient_descent(a_init, time_budget=min(3.0, remaining * 0.1))
            try_update(a_quick, f"n={n_try},mode={mode}")

    print(f"  After Phase 0: best_score={best_score:.8f}, n={len(best_seq) if best_seq is not None else 0}")

    if best_seq is None:
        best_seq = np.ones(800)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"  Default: score={best_score:.8f}")

    # Phase 1: Intensive subgradient descent from best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 1: Subgradient descent (remaining={remaining:.0f}s) ===")
        a_work = normalize(best_seq.copy())
        alloc = min(120.0, remaining * 0.25)
        a_sg = subgradient_descent(a_work, time_budget=alloc, verbose=True)
        try_update(a_sg, "SG Phase1")

    # Phase 2: LP alternating from best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: LP alternating (remaining={remaining:.0f}s) ===")
        a_work = normalize(best_seq.copy())
        alloc = min(180.0, remaining * 0.3)
        a_lp = lp_alternating(a_work, time_budget=alloc, verbose=True)
        try_update(a_lp, "LP Phase2")

    # Phase 3: Frequency domain refinement
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 3: Freq domain (remaining={remaining:.0f}s) ===")
        a_work = normalize(best_seq.copy())
        alloc = min(60.0, remaining * 0.15)
        a_freq = freq_domain_refine(a_work, time_budget=alloc)
        try_update(a_freq, "Freq Phase3")

    # Phase 4: Parametric optimization at current best length
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 4: Parametric optimization ===")
        n_curr = len(best_seq)
        alloc = min(60.0, remaining * 0.2)
        a_param = optimize_params_parametric(n_curr, time_budget=alloc)
        if a_param is not None:
            try_update(a_param, "Param Phase4")
            # Follow up with SG
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_sg_p = subgradient_descent(a_param, time_budget=min(30.0, remaining * 0.3))
                try_update(a_sg_p, "SG after Param")

    # Phase 5: Multi-length exploration
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Multi-length exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 200), max(200, n_best - 100),
            n_best + 50, n_best + 100, n_best + 200,
            n_best + 500,
            500, 700, 900, 1100, 1400, 1600, 2000
        ]))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_try = resample_sequence(best_seq, n_try)
            a_try = normalize(a_try)

            # Quick optimization: SG + LP
            alloc_sg = min(20.0, remaining * 0.1)
            a_opt = subgradient_descent(a_try, time_budget=alloc_sg)

            remaining = deadline - time.monotonic()
            if remaining > 10 and len(a_opt) <= 1200:
                alloc_lp = min(30.0, remaining * 0.15)
                a_opt = lp_alternating(a_opt, time_budget=alloc_lp)

            try_update(a_opt, f"n={n_try}")

    # Phase 6: Multi-start from best with perturbations
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Multi-start perturbations ===")
        n_best = len(best_seq)
        restart_budget = min(30.0, remaining * 0.06)

        for r in range(15):
            remaining = deadline - time.monotonic()
            if remaining < restart_budget + 5:
                break

            pert_scale = 0.05 + 0.02 * r
            a_pert = best_seq * np.exp(rng.normal(0, pert_scale, n_best))
            a_pert = normalize(a_pert)

            # Alternate between SG and LP
            if r % 2 == 0:
                a_opt = subgradient_descent(a_pert, time_budget=restart_budget)
            else:
                if n_best <= 1000:
                    a_opt = lp_alternating(a_pert, time_budget=restart_budget)
                else:
                    a_opt = subgradient_descent(a_pert, time_budget=restart_budget)

            try_update(a_opt, f"Perturb {r} (scale={pert_scale:.3f})")

    # Phase 7: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 7: Final polish (remaining={remaining:.0f}s) ===")
        a_final = normalize(best_seq.copy())

        # Multiple rounds of SG + LP
        for round_idx in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.2, 40.0)
            a_sg = subgradient_descent(a_final, time_budget=alloc)
            try_update(a_sg, f"Polish SG {round_idx}")

            remaining = deadline - time.monotonic()
            if remaining > 10 and len(a_final) <= 1000:
                alloc = min(remaining * 0.3, 30.0)
                a_lp = lp_alternating(normalize(best_seq.copy()), time_budget=alloc)
                try_update(a_lp, f"Polish LP {round_idx}")

            # Freq
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_freq = freq_domain_refine(normalize(best_seq.copy()), time_budget=min(15.0, remaining * 0.2))
                try_update(a_freq, f"Polish Freq {round_idx}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
