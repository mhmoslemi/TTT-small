Sequence optimizer combining LP alternating (HiGHS), QP projected gradient,

import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Algorithm from Wright (1984)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)

"""
Optimized sequence search combining accelerated LP alternating (HiGHS),
FFT-based QP projected gradient, smooth-max subgradient, and systematic
multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: FFT-accelerated QP gradient, adaptive LP step selection,
peak-aware perturbation, and coarse-to-fine n search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        grad = np.correlate(weights, a, mode='valid')

        # Center for simplex constraint
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0 and lr < 0.05:
                lr = min(lr * 1.02, 0.05)
        else:
            stall += 1
            if stall >= 40:
                lr *= 0.7
                v = np.zeros(n)
                if lr < 1e-5:
                    break
                a = best_a.copy()
                stall = 0
            elif stall >= 100:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st > 1e-10:
                    a = a / st
                v = np.zeros(n)
                stall = 0

        if it > 100:
            T = max(T_min, T_init * (0.995 ** it))

    return best_a


def peak_split_method(a, time_budget=30.0, max_iters=3000):
    """
    Direct peak-chasing: for the current peak bin k*, reduce the
    contributing elements via multiplicative update.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    rng = np.random.default_rng(55 + n)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Use top few peaks weighted
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # Subgradient at the peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = a[j]

        # Multiplicative update (mirror descent)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        st = float(np.sum(a_new))
        if st > 1e-10:
            a_new = a_new / st
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall >= 30:
                    # Try smaller eta
                    eta *= 0.5
                    if eta < 1e-4:
                        # Perturb
                        a = best_a * np.exp(rng.normal(0, 0.05, n))
                        a = np.maximum(a, 0)
                        st = float(np.sum(a))
                        if st > 1e-10:
                            a = a / st
                        eta = 0.1
                    stall = 0
        else:
            eta *= 0.5
            stall += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence (normalized)."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Smooth via sinc-like
        x = np.arange(n)
        center = n // 2
        a = np.exp(-0.5 * ((x - center) / (n / 4)) ** 2)
        a += 0.3 * rng.uniform(0, 1, n)
        a = np.maximum(a, 0) / np.sum(a)
    elif mode == 3:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / np.sum(a)
    elif mode == 4:
        # Triangle
        x = np.arange(n) / max(n - 1, 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    elif mode == 5:
        # Harmonic
        a = 1.0 / (np.arange(1, n + 1))
        a = a / np.sum(a)
    elif mode == 6:
        # Two-peak
        c1, c2 = n // 3, 2 * n // 3
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (n / 3)) + np.exp(-((x - c2) ** 2) / (n / 3))
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def full_optimize(a, time_budget=60.0):
    """Run a full optimization pipeline on sequence a."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Phase 1: LP alternating
    remaining = time_budget - (time.monotonic() - time.monotonic())  # initial
    lp_budget = min(remaining * 0.4, 25.0)
    a_lp = lp_alternate(a, time_budget=lp_budget)
    obj_lp = compute_obj(a_lp)
    if obj_lp < best_obj:
        best_obj = obj_lp
        best_a = a_lp.copy()

    # Phase 2: QP projected gradient
    elapsed = time.monotonic() - (time_monotonic_start())
    qp_budget = min((time_budget - elapsed) * 0.3, 20.0)
    if qp_budget > 2:
        a_qp = qp_projected_grad(best_a, time_budget=qp_budget)
        obj_qp = compute_obj(a_qp)
        if obj_qp < best_obj:
            best_obj = obj_qp
            best_a = a_qp.copy()

    # Phase 3: Smooth max gradient
    elapsed = time.monotonic() - time_monotonic_start()
    sm_budget = min((time_budget - elapsed) * 0.2, 15.0)
    if sm_budget > 2:
        a_sm = smooth_max_grad(best_a, time_budget=sm_budget)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_obj:
            best_obj = obj_sm
            best_a = a_sm.copy()

    # Phase 4: Peak split
    elapsed = time.monotonic() - time_monotonic_start()
    ps_budget = min((time_budget - elapsed) * 0.1, 10.0)
    if ps_budget > 2:
        a_ps = peak_split_method(best_a, time_budget=ps_budget)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_obj:
            best_obj = obj_ps
            best_a = a_ps.copy()

    return best_a


def time_monotonic_start():
    return _GLOBAL_START[0]


_GLOBAL_START = [0.0]


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple strategies:
    LP alternating, QP projected gradient, smooth-max, peak-split,
    multi-n exploration, and random restarts.
    """
    _GLOBAL_START[0] = time.monotonic()
    start_time = _GLOBAL_START[0]
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={obj:.8f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.8f}")

    # === Phase 1: Intensive LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(400.0, remaining * 0.3)
    a = best_seq.copy()
    a_lp = lp_alternate(a, time_budget=phase1_budget, verbose=True)
    obj_lp = compute_obj(a_lp)
    if obj_lp < best_score:
        best_score = obj_lp
        best_seq = a_lp.copy()
        print(f"  Phase 1 done: obj={best_score:.8f}")
    else:
        print(f"  Phase 1 done: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: QP Projected Gradient ===
    print(f"\n=== Phase 2: QP Projected Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        qp_budget = min(150.0, remaining * 0.2)
        a_qp = qp_projected_grad(best_seq, time_budget=qp_budget, verbose=True)
        obj_qp = compute_obj(a_qp)
        if obj_qp < best_score:
            best_score = obj_qp
            best_seq = a_qp.copy()
            print(f"  Phase 2 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 2 done: obj={obj_qp:.8f} (no improvement)")

    # === Phase 3: Smooth Max Gradient ===
    print(f"\n=== Phase 3: Smooth Max Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        sm_budget = min(80.0, remaining * 0.15)
        a_sm = smooth_max_grad(best_seq, time_budget=sm_budget, T_init=0.001, T_min=0.00003)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Phase 3 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 3 done: obj={obj_sm:.8f} (no improvement)")

    # === Phase 4: Peak Split ===
    print(f"\n=== Phase 4: Peak Split ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        ps_budget = min(50.0, remaining * 0.1)
        a_ps = peak_split_method(best_seq, time_budget=ps_budget)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Phase 4 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 4 done: obj={obj_ps:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 200),
        max(300, n_best - 100),
        max(300, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        500, 600, 700, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP + QP
        lp_b = min(remaining * 0.12, 20.0)
        qp_b = min(remaining * 0.08, 10.0)

        a_opt = lp_alternate(a_try, time_budget=lp_b)
        obj_try = compute_obj(a_opt)

        if remaining > 30 and obj_try < best_score + 0.02:
            a_opt2 = qp_projected_grad(a_opt, time_budget=qp_b)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Random Restarts ===
    print(f"\n=== Phase 6: Random Restarts ===")
    n_best = len(best_seq)

    for r in range(20):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        init_mode = r % 7
        n_r = int(n_best + rng.integers(-100, 101))
        n_r = max(400, min(1500, n_r))
        a_r = random_init(n_r, rng, mode=init_mode)

        # Quick optimization
        lp_b = min(remaining * 0.1, 20.0)
        a_opt = lp_alternate(a_r, time_budget=lp_b)
        obj_r = compute_obj(a_opt)

        if remaining > 20 and obj_r < best_score + 0.01:
            qp_b = min(remaining * 0.05, 10.0)
            a_opt2 = qp_projected_grad(a_opt, time_budget=qp_b)
            obj_r2 = compute_obj(a_opt2)
            if obj_r2 < obj_r:
                a_opt = a_opt2
                obj_r = obj_r2

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # Multiple rounds of LP + QP on the best
    for polish_round in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        lp_b = min(remaining * 0.2, 30.0)
        a_lp = lp_alternate(best_seq, time_budget=lp_b)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Polish LP {polish_round}: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        qp_b = min(remaining * 0.2, 20.0)
        a_qp = qp_projected_grad(best_seq, time_budget=qp_b)
        obj_qp = compute_obj(a_qp)
        if obj_qp < best_score:
            best_score = obj_qp
            best_seq = a_qp.copy()
            print(f"  Final QP: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        sm_b = min(remaining * 0.15, 15.0)
        a_sm = smooth_max_grad(best_seq, time_budget=sm_b, T_init=0.0005, T_min=0.00001)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Final SmoothMax: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 8:
        ps_b = min(remaining * 0.1, 10.0)
        a_ps = peak_split_method(best_seq, time_budget=ps_b)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Final PeakSplit: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
