    while size < 2 * n - 1:

    fa = rfft(a, size)

    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Take a subgradient step for the peak bin k_star."""
    n = len(a)
    # Gradient of conv(a,a)[k_star] w.r.t. a
    # d/d a_j = 2 * a[k_star - j] if k_star - j in [0, n-1]
    g = np.zeros(n)

    for j in range(j_min, j_max + 1):
        g[j] = 2.0 * a[k_star - j]
    
    # Project gradient onto tangent space (sum(g_projected) = 0)
    g_proj = g - np.mean(g)
    
    # Take step and project onto simplex
    a_new = a - alpha * g_proj
    a_new = project_simplex(a_new, target_sum=1.0)
    return a_new, g

"""
Hybrid optimization for low-peak-factor non-negative sequences.
Combines: (1) fast vectorized subgradient descent, (2) sparse LP alternating,
(3) FFT-based smoothed L-BFGS, (4) structured random restarts, (5) multi-n exploration.
Key improvements: vectorized gradient computation, mirror descent variant,
adaptive method switching, and more diverse exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
import scipy.sparse as sp
from scipy.optimize import minimize, linprog
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_gradient_at_bin(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a. Vectorized."""
    n = len(a)
    g = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min <= j_max:
        g[j_min:j_max + 1] = 2.0 * a[k - j_max:k - j_min + 1][::-1]
    return g


def subgradient_descent(a_init, time_budget=60.0, top_k=1, verbose=False):
    """Subgradient descent with adaptive step size on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    alpha = 0.1
    n_iter = 0
    stalled = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)

        if top_k <= 1:
            k_star = int(np.argmax(conv))
            g = get_gradient_at_bin(a, k_star)
        else:
            top_idx = np.argsort(conv)[::-1][:top_k]
            weights = np.array([0.6 ** i for i in range(top_k)])
            weights /= weights.sum()
            g = np.zeros(n)
            for w, k in zip(weights, top_idx):
                g += w * get_gradient_at_bin(a, int(k))

        g_proj = g - np.mean(g)
        g_norm = np.linalg.norm(g_proj)
        if g_norm < 1e-15:
            break

        found = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = alpha * mult * g_proj / g_norm
            a_trial = project_simplex(a - step, 1.0)
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial
                found = True
                break

        n_iter += 1
        if found:
            stalled = 0
            alpha = min(alpha * 1.02, 2.0)
            if verbose and n_iter % 500 == 0:
                print(f"    SG it{n_iter}: obj={best_obj:.8f}, alpha={alpha:.4f}")
        else:
            alpha *= 0.5
            stalled += 1
            if stalled > 200:
                rng = np.random.default_rng(n_iter * 13 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                alpha = 0.05
                stalled = 0

    return best_a, best_obj


def mirror_descent(a_init, time_budget=60.0, verbose=False):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    n_iter = 0
    stalled = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        g = get_gradient_at_bin(a, k_star)

        # Mirror descent: a_new[j] ∝ a[j] * exp(-eta * g[j])
        log_a = np.log(a) - eta * g
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        n_iter += 1

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stalled = 0
            eta = min(eta * 1.01, 5.0)
            if verbose and n_iter % 500 == 0:
                print(f"    MD it{n_iter}: obj={best_obj:.8f}, eta={eta:.4f}")
        else:

            stalled += 1

                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                eta = 0.3
                stalled = 0

    return best_a, best_obj


def build_sparse_conv_matrix(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        for i in range(r):
            rows.append(i + j)
            cols.append(j)
            vals.append(a[i])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_sparse_conv_matrix(a)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog with sparse matrix
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = sp.hstack([A, sp.csr_matrix(-np.ones((2 * n - 1, 1)))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = sp.csr_matrix((np.ones(1), (0, 0)), shape=(1, n + 1))
        # Fix: A_eq should be [1,1,...,1, 0]
        A_eq = sp.csr_matrix(np.zeros((1, n + 1)))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, time_budget=60.0, verbose=False):

def fft_smoothed_obj_grad(theta, tau, n):
    """Compute smoothed objective and gradient w.r.t. theta using FFT."""
    a = np.exp(theta - np.max(theta))
    a = a / np.sum(a)

    conv = compute_conv(a)
    conv_max = np.max(conv)
    obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)

    # Soft-argmax weights
    w = tau * np.exp((conv - conv_max) * tau)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / w_sum

    # Cross-correlation c[j] = sum_k w[k] * a[k-j]
    # Using np.correlate: correlate(w, a, 'full')[j] = sum_m w[m+j] * a[m]
    c_full = np.correlate(w, a, mode='full')
    c = c_full[:n]

    mean_c = float(np.dot(c, a))
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta


def smooth_lbfgs(a_init, tau_start, tau_end, time_budget, max_iter=200):
    """Smoothed L-BFGS-B with tau annealing and FFT gradients."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    n_stages = 6
    taus = np.geomspace(tau_start, tau_end, n_stages)

    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        stage_budget = min(remaining * 0.85, 60.0)

        def obj_func(th):
            val, _ = fft_smoothed_obj_grad(th, tau, n)
            return val

        def grad_func(th):
            _, g = fft_smoothed_obj_grad(th, tau, n)
            return g

        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-7, 'maxfun': 300})
            if result.x is not None:
                a_new = np.exp(result.x - np.max(result.x))
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a_opt, obj_opt = smooth_lbfgs(best_seq, tau_start=30.0, tau_end=800.0,
                                      time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final Smooth: obj={best_score:.8f}")

    # Phase 8: Very fine-tuned subgradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_sg, obj_sg = subgradient_descent(best_seq, time_budget=min(remaining * 0.5, 20.0),
                                           top_k=1)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final FineSG: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
