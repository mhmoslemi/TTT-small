"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive log-sum-exp with L-BFGS, Nesterov-accelerated subgradient,
LP polishing, coordinate descent on peaks, and structured perturbations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def lse_obj_and_grad(x, T, n, N_fft):
    """
    Compute smooth objective and gradient using log-sum-exp approximation.
    x is on the simplex (sum=1, x>=0). Returns (value, grad_projected).
    """
    a = np.maximum(x, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    a = a / s

    fa = rfft(a, N_fft)
    b = irfft(fa * fa, N_fft)[:2 * n - 1]

    # Log-sum-exp
    b_max = float(np.max(b))
    shifted = (b - b_max) / T
    exp_shifted = np.exp(shifted)
    w_sum = float(np.sum(exp_shifted))
    lse_val = b_max + T * np.log(w_sum)
    obj_val = 2.0 * n * lse_val

    # Weights
    w = exp_shifted / w_sum

    # Gradient: grad[i] = 4n * sum_m a[m] * w[m+i]
    a_flip = a[::-1]
    caf = rfft(a_flip, N_fft)
    c_full = irfft(caf * rfft(w, N_fft), N_fft)[:3 * n - 2]
    c = c_full[n - 1:2 * n - 1]

    grad = 4.0 * n * c
    grad = grad - np.mean(grad)  # Project onto simplex tangent

    return obj_val, grad


def nesterov_subgradient(a, time_budget=60.0, T_init=0.5, T_final=0.01):
    """
    Nesterov-accelerated subgradient method on the smooth approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    beta = 0.9

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        # Anneal temperature
        frac = min(1.0, it / 5000.0)
        T = T_init * (T_final / T_init) ** frac

        # Nesterov momentum
        t_k = 2.0 / (it + 2.0)
        y = a + (beta - 1.0) * (a - y) if it > 0 else a.copy()

        val, grad = lse_obj_and_grad(y, T, n, N_fft)
        if not np.isfinite(val):
            break

        # Step size: Polyak-style
        # Use the gap between current and best
        gap = val - best_obj
        if gap < 1e-10:
            gap = 1e-10
        gamma = min(2.0 * gap / (np.sum(grad ** 2) + 1e-20), 0.5)
        gamma = max(gamma, 1e-10)

        a_new = a - gamma * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            continue
        a_new = a_new / s_new

        # Evaluate true objective
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            beta = min(beta * 1.002, 0.99)

    return best_a


def coordinate_descent_peak(a, time_budget=30.0):
    """
    Coordinate descent targeting the top peaks of the convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N_fft)
        b = irfft(fa * fa, N_fft)[:2 * n - 1]
        k_star = int(np.argmax(b))
        peak_val = b[k_star]

        # Compute gradient of peak w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        # Center for simplex
        grad = grad - np.mean(grad)

        # Try a few step sizes
        improved = False
        for eta in [0.001, 0.002, 0.005, 0.01, 0.02]:
            a_try = a - eta * grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try coordinate moves: reduce a[i] where it contributes most to peak
            max_contrib_idx = int(np.argmax(np.abs(grad)))
            for delta in [0.001, 0.005, 0.01, 0.02, 0.05]:
                a_try = a.copy()
                # Reduce a[max_contrib_idx] and redistribute
                reduce_amt = a[max_contrib_idx] * delta
                if reduce_amt < 1e-12:
                    continue
                a_try[max_contrib_idx] -= reduce_amt
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break

            if not improved:
                # Add small noise and retry
                rng_local = np.random.default_rng(it * 7 + 3)
                a_perturbed = a * np.exp(rng_local.normal(0, 0.01, n))
                a_perturbed = np.maximum(a_perturbed, 0)
                s_p = float(np.sum(a_perturbed))
                if s_p > 1e-10:
                    a_perturbed = a_perturbed / s_p
                obj_p = compute_obj(a_perturbed)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_perturbed.copy()
                    a = a_perturbed.copy()
                elif it > 500:
                    break

    return best_a


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a = a / np.sum(a)

    # Build constraint matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]
    # conv(a,g)[k] = sum_j a[j] * g[k-j]
    # For fixed k: sum_j g[k-j] * a[j]
    # So row k of A has: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                A[k, j] = a[k - j]

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=60.0):
    """
    LP-based refinement: alternate between LP solve on current a and blending.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP with current a as "convolution partner"
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try blending
        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a


def lse_lbfgs(a, time_budget=120.0):
    """
    L-BFGS optimization using log-sum-exp smooth approximation with annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1

    # Annealing schedule: multiple rounds with decreasing T
    T_schedule = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    for T in T_schedule:
        if time.monotonic() - start > time_budget:
            break

        T_remaining = time_budget - (time.monotonic() - start)
        if T_remaining < 10:
            break
        T_alloc = min(T_remaining * 0.7, 30.0)

        def fun_and_grad(x):
            val, grad = lse_obj_and_grad(x, T, n, N_fft)
            if not np.isfinite(val):
                return 1e10, np.zeros(n)
            return val, grad

        # L-BFGS on the simplex (using log transform for non-negativity)
        # Work in log-space: x_i = exp(y_i) / sum(exp(y_j))
        a_curr = np.maximum(a, 1e-15)
        y_init = np.log(a_curr)

        def softmax_fun(y):
            # Compute softmax
            y_shifted = y - np.max(y)
            x = np.exp(y_shifted)
            x = x / np.sum(x)
            val, grad_x = lse_obj_and_grad(x, T, n, N_fft)
            if not np.isfinite(val):
                return 1e10, np.zeros(n)
            # Chain rule: dy -> dx
            # x_i = exp(y_i)/S where S = sum(exp(y_j))
            # dx_i/dy_j = x_i * (delta_ij - x_j)
            # grad_y_j = sum_i grad_x_i * dx_i/dy_j = sum_i grad_x_i * x_i * (delta_ij - x_j)
            # = x_j * grad_x_j - x_j * sum_i grad_x_i * x_i
            grad_x_centered = grad_x - np.sum(grad_x * x)
            grad_y = x * grad_x_centered
            return val, grad_y

        try:
            res = minimize(softmax_fun, y_init, method='L-BFGS-B',
                           jac=True, options={'maxiter': 200, 'maxfun': 500,
                                              'ftol': 1e-15, 'gtol': 1e-10})
            if res.success or res.fun < 1e10:
                y_opt = res.x
                y_shifted = y_opt - np.max(y_opt)
                a_new = np.exp(y_shifted)
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
        except Exception:
            pass

    # Final subgradient polish
    a_polished = nesterov_subgradient(a, time_budget=min(20.0, max(5.0, time_budget - (time.monotonic() - start))))
    obj_polished = compute_obj(a_polished)
    if obj_polished < best_obj:
        best_obj = obj_polished
        best_a = a_polished.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_perturbation(a, rng, mode=0):
    """Apply structured perturbation to create diverse starting points."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        # Sinusoidal perturbation
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        pert = 1.0 + 0.1 * np.sin(2 * np.pi * freq * x / n + phase)
    elif mode == 1:
        # Polynomial perturbation
        coeff = rng.normal(0, 0.05, 3)
        t = x / n - 0.5
        pert = 1.0 + coeff[0] * t + coeff[1] * t**2 + coeff[2] * t**3
    elif mode == 2:
        # Bump perturbation
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(n * 0.05, n * 0.2)
        amp = rng.uniform(0.05, 0.2) * rng.choice([-1, 1])
        pert = 1.0 + amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 3:
        # Multi-frequency
        pert = 1.0
        for _ in range(3):
            freq = rng.uniform(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.08)
            pert += amp * np.sin(2 * np.pi * freq * x / n + phase)
    else:
        # Gaussian noise
        pert = 1.0 + rng.normal(0, 0.05, n)

    a_new = a * np.maximum(pert, 0.1)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 10) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.2
        a = a / np.sum(a)
    elif mode == 5:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.5
        a = a / np.sum(a)
    elif mode == 6:
        # Smooth random (low-pass noise)
        a = rng.normal(0, 1, n)
        # Simple moving average smoothing
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a - np.min(a) + 0.1, 0)
        a = a / np.sum(a)
    else:
        # Von Mises-like (peaked)
        kappa = rng.uniform(5, 20)
        theta = 2 * np.pi * np.arange(n) / n
        a = np.exp(kappa * (np.cos(theta) - 1)) + 0.1
        a = a / np.sum(a)
    return a

            print(f"  CoordDesc: obj={obj_cd:.8f}")

    # === Phase 5: Second L-BFGS round (from improved point) ===
    print(f"\n=== Phase 5: L-BFGS Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 100.0)
        a_res2 = lse_lbfgs(best_seq, time_budget=alloc)
        obj_res2 = compute_obj(a_res2)
        if obj_res2 < best_score:
            best_score = obj_res2
            best_seq = a_res2.copy()
            print(f"  ** L-BFGS2: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS2: obj={obj_res2:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        n_best - 100, n_best - 200,
        1000, 1200, 1400, 1600, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and n != n_best]
    n_options = n_options[:8]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Quick L-BFGS on new n
        alloc = min(remaining * 0.15, 60.0)
        a_opt = lse_lbfgs(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Structured perturbation restarts ===
    print(f"\n=== Phase 7: Structured Perturbations ===")
    n_curr = len(best_seq)
    remaining = deadline - time.monotonic()
    n_restarts = max(3, int(remaining / 50.0))
    n_restarts = min(n_restarts, 10)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r % 5
        a_pert = structured_perturbation(best_seq, rng, mode=mode)
        alloc = min(remaining * 0.2, 40.0)
        a_opt = lse_lbfgs(a_pert, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** Pert {r} (mode {mode}): obj={best_score:.8f} **")
        else:
            print(f"  Pert {r} (mode {mode}): obj={obj_opt:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # L-BFGS
        alloc = min(remaining * 0.3, 80.0)
        a_final = lse_lbfgs(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final L-BFGS: obj={best_score:.8f} **")

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_lp2 = lp_refine(best_seq, time_budget=min(remaining * 0.3, 50.0))
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  ** Final LP: obj={best_score:.8f} **")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_cd2 = coordinate_descent_peak(best_seq, time_budget=min(remaining * 0.3, 20.0))
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  ** Final CoordDesc: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
