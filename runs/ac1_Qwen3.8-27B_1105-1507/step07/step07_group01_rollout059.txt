    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""

        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in ['CLARABEL', 'MOSEK', 'SCS', 'ECOS', 'HIGHS']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_topk_penalty(a, k=10, penalty=0.1):
    """Solve LP with a penalty on the sum of top-k convolution values.
    min M + penalty * sum(conv)  s.t. conv(a,g) <= M, sum(g)=1, g>=0
    This encourages flatter convolution profiles.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        conv_vals = A @ g
        constraints = [conv_vals <= M, cp.sum(g) == 1]
        obj = M + penalty * cp.sum(conv_vals)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        for solver in ['CLARABEL', 'MOSEK', 'SCS', 'ECOS', 'HIGHS']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except ImportError:
        pass
    return None


def line_search_blend(a, g, M_a, M_ag, M_g, max_evals=15):
    """
    Find best t for (1-t)*a + t*g using analytical initialization + refinement.
    Uses the bound f(t) = (1-t)^2*M_a + 2t(1-t)*M_ag + t^2*M_g to init.
    """
    best_obj = compute_obj(a)
    best_a = a.copy()

    # Analytical t from the quadratic bound
    denom = M_a - 2.0 * M_ag + M_g
    if abs(denom) > 1e-15:
        t_analytical = (M_a - M_ag) / denom
        t_analytical = max(0.01, min(1.0, t_analytical))
    else:
        t_analytical = 0.3

    # Candidate t values: analytical + neighbors + standard grid
    candidates = []
    for frac in [0.3, 0.5, 0.7, 1.0, 1.4]:
        t = t_analytical * frac
        if 0.01 <= t <= 1.0:
            candidates.append(t)
    candidates.extend([0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0])
    # Deduplicate and sort
    candidates = sorted(set(round(t, 4) for t in candidates))
    candidates = [t for t in candidates if 0.01 <= t <= 1.0]

    for t in candidates:
        if len(candidates) > max_evals and t > candidates[max_evals]:
            break
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def lp_alternating(a, time_budget=60.0, verbose=False):
    """
    LP alternating with analytical line search and top-k penalty LP variants.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    use_topk = False

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Alternate between standard LP and top-k penalty LP
        if use_topk:
            g = solve_lp_topk_penalty(a, k=10, penalty=0.05)
            if g is not None:
                M_ag = float(np.max(build_conv_matrix(a / np.sum(a)) @ g))
                M_g = float(np.max(compute_conv(g / np.sum(g))))
                M_a = float(np.max(compute_conv(a)))
                a_new, obj_new = line_search_blend(a, g, M_a, M_ag, M_g)
            else:
                g_info = solve_lp_maxpeak(a)
                if g_info[0] is None:
                    break
                g, M_ag = g_info
                M_a = float(np.max(compute_conv(a)))
                M_g = float(np.max(compute_conv(g)))
                a_new, obj_new = line_search_blend(a, g, M_a, M_ag, M_g)
        else:
            g, M_ag = solve_lp_maxpeak(a)
            if g is None:
                break
            M_a = float(np.max(compute_conv(a)))
            M_g = float(np.max(compute_conv(g)))
            a_new, obj_new = line_search_blend(a, g, M_a, M_ag, M_g)

        use_topk = not use_topk

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 50 == 0:
                print(f"      LP it{it}: obj={best_obj:.10f}")
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(it * 104729 + n * 31)
                scale = 0.015 + 0.005 * min(stall // 15, 4)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def smooth_max_weights(conv, T):
    """Compute smooth max value and softmax weights over convolution values."""
    max_c = np.max(conv)
    shifted = np.clip((conv - max_c) / T, -500, 500)
    exp_vals = np.exp(shifted)
    total = float(np.sum(exp_vals))
    if total < 1e-300:
        total = 1e-300
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    return smooth_val, weights


def smooth_max_gradient_descent(a, time_budget=30.0, T_init=0.001, T_min=0.00005):
    """Gradient descent using log-sum-exp smooth max approximation with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    lr = 0.004
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        _, weights = smooth_max_weights(conv, T)
        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - float(np.mean(grad))
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                lr *= 0.65
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-6:
                    break
        if it > 20:
            T = max(T_min, T_init * (0.997 ** it))
    return best_a


def subgradient_descent(a, time_budget=20.0):
    """Direct subgradient descent targeting the max convolution bin."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.01
    stall = 0

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        # Subgradient of conv[k_star] w.r.t. a[i] = a[k_star - i] if valid
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
        grad_centered = grad - float(np.mean(grad))
        a_try = simplex_project(a - lr * grad_centered)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            stall += 1
            if stall > 20:
                lr *= 0.7
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence normalized to sum 1."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a / np.sum(a)
    elif mode == 3:
        center = n // 2
        width = max(1, n // 4)
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        a = a / np.sum(a)
    elif mode == 5:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif mode == 6:
        # U-shaped (high at edges)
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with analytical line search, top-k penalty LP,
    smooth max gradient descent, subgradient descent, multi-n exploration,
    and random restarts. Targets objective < 1.5027.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent n={len(best_seq)}: obj={best_score:.10f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default n=800: obj={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: Intensive LP Alternating ===
    print("\n=== Phase 1: LP Alternating (main) ===")
    a = best_seq.copy()
    for round_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.15, 100.0)
        a = lp_alternating(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  LP round {round_idx}: obj={best_score:.10f} **")
        else:
            print(f"  LP round {round_idx}: obj={obj:.10f}")

    # === Phase 2: Smooth Max Polish ===
    print("\n=== Phase 2: Smooth Max Gradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        for T_init in [0.001, 0.0003, 0.0001]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_sm = smooth_max_gradient_descent(a, time_budget=min(30.0, remaining * 0.2),
                                                T_init=T_init, T_min=T_init * 0.05)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                a = a_sm.copy()
                print(f"  Smooth (T={T_init}): obj={best_score:.10f} **")
            else:
                print(f"  Smooth (T={T_init}): obj={obj_sm:.10f}")

    # === Phase 3: Subgradient Descent ===
    print("\n=== Phase 3: Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sg = subgradient_descent(a, time_budget=min(20.0, remaining * 0.1))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  Subgradient: obj={best_score:.10f} **")
        else:
            print(f"  Subgradient: obj={obj_sg:.10f}")

    # === Phase 4: Multi-n Exploration ===
    print("\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 150, n_best - 100, n_best - 50, n_best - 25,
        n_best, n_best + 25, n_best + 50, n_best + 100, n_best + 150, n_best + 250,
        350, 400, 450, 500, 550, 600, 650, 700, 750, 800, 850, 900, 950, 1000,
        1050, 1100, 1150, 1200, 1300, 1400, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.08, 35.0)
        a_opt = lp_alternating(a_try, time_budget=alloc * 0.7)
        # Quick smooth polish if close
        remaining = deadline - time.monotonic()
        obj_try = compute_obj(a_opt)
        if obj_try < best_score + 0.005 and remaining > 10:
            a_opt = smooth_max_gradient_descent(a_opt, time_budget=min(alloc * 0.3, remaining * 0.3),
                                                 T_init=0.0005, T_min=0.00005)
            obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Random Restarts ===
    print("\n=== Phase 5: Random Restarts ===")
    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        mode = r % 7
        n_r = max(400, min(1400, n_best + int(rng.integers(-100, 101))))
        a_r = random_init(n_r, rng, mode=mode)
        alloc = min(remaining * 0.08, 28.0)
        a_opt = lp_alternating(a_r, time_budget=alloc * 0.6)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = smooth_max_gradient_descent(a_opt, time_budget=min(alloc * 0.4, remaining * 0.3),
                                                 T_init=0.001, T_min=0.0001)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (mode={mode}, n={n_r}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}, n={n_r}): obj={obj_r:.10f}")

    # === Phase 6: Final Intensive Polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()
    for polish_round in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.25, 50.0)
        a_lp = lp_alternating(a_final, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP {polish_round}: obj={best_score:.10f} **")
        else:
            print(f"  Final LP {polish_round}: obj={obj_lp:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_f2 = smooth_max_gradient_descent(a_final, time_budget=min(remaining * 0.3, 25.0),
                                            T_init=0.0002, T_min=0.00002)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final Smooth: obj={best_score:.10f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_f3 = subgradient_descent(a_final, time_budget=min(remaining * 0.2, 15.0))
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final Subgrad: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
