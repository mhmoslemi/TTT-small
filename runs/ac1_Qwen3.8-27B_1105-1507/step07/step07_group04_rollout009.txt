    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):

        # So B[k, j] = a[j] if 0 <= k-j < n, i.e., j <= k < j+n
        # So B[j:j+n, j] = a (but clipped to 2n-1 rows)
        row_start = j
        row_end = min(j + n, 2 * n - 1)
        col_end = row_end - row_start
        B[row_start:row_end, j] = a[:col_end]
    return B

    b_ub = np.zeros(2 * n - 1)

    return None

            # d/d a[i] = a[k-i] (for i where k-i is valid) + a[i-k] (for i where i-k is valid)
            # Actually: d(a*a)[k] / da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)

            # Or just a[k-i] when only one term is valid
            for i in range(j_start, j_end):
                other = k - i
                if 0 <= other < n:
                    grad[i] += a[other]
                other2 = i - k
                if 0 <= other2 < n:
                    grad[i] += a[other2]
        
        grad /= top_k
        
        # Project gradient to sum=0 (stay on simplex)
        grad = grad - np.mean(grad)
        
        # Momentum update
        momentum = beta * momentum - step * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0

            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new

        if it % 100 == 99:
            step = max(step * 0.93, 0.001)
            beta = max(beta * 0.999, 0.5)

"""
Optimized alternating LP + log-sum-exp gradient for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses direct HiGHS LP, smooth gradient polish, and multi-scale exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute autocorrelation via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective value (scale invariant)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix B (2n-1 x n) such that B @ g = conv(a, g)."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_end = min(j + n, 2 * n - 1)
        B[j:row_end, j] = a[:row_end - j]
    return B


def solve_lp_peak(a):
    """Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s
    B = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = B
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def lse_gradient_polish(a, time_budget=120.0, verbose=False):
    """
    Log-sum-exp smoothed gradient descent with momentum.
    Uses smooth approximation of max to get a well-defined gradient.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.8
    step = 0.03
    lam = 50.0  # temperature for log-sum-exp

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        # Softmax weights over lags
        shifted = conv - np.max(conv)
        weights = np.exp(lam * shifted)
        total_w = np.sum(weights)
        weights /= total_w
        # Gradient: weighted average of per-lag gradients
        grad = np.zeros(n)
        # Only process lags with significant weight for speed
        active_mask = weights > 1.0 / (2 * n)
        if np.sum(active_mask) < 5:
            active_mask = np.argsort(weights)[::-1][:20] > 0
        active_indices = np.where(active_mask)[0]
        for k in active_indices:
            k = int(k)
            w = weights[k]
            if w < 1e-6:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for i in range(j_start, j_end):
                other = k - i
                if 0 <= other < n:
                    grad[i] += w * a[other]
                other2 = i - k
                if 0 <= other2 < n:
                    grad[i] += w * a[other2]
        # Project to sum=0
        grad = grad - np.mean(grad)
        # Momentum update
        momentum = beta * momentum - step * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        if it % 500 == 499:
            step = max(step * 0.95, 0.002)
            beta = min(beta + 0.001, 0.95)
            lam = min(lam * 1.02, 200.0)
        if verbose and it % 5000 == 4999:
            print(f"    LSE iter {it}: obj={best_obj:.8f}")

def subgradient_topk(a, time_budget=60.0, top_k=10, verbose=False):
    """Subgradient method targeting top-K peaks with adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.7
    step = 0.04
    for it in range(30000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_indices = np.argsort(conv)[::-1][:top_k]
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for i in range(j_start, j_end):
                other = k - i
                if 0 <= other < n:
                    grad[i] += a[other]
                other2 = i - k
                if 0 <= other2 < n:
                    grad[i] += a[other2]
        grad /= top_k
        grad = grad - np.mean(grad)
        momentum = beta * momentum - step * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        if it % 200 == 199:
            step = max(step * 0.96, 0.002)
    return best_a


def resample_to_n(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new /= s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Fast alternating LP (HiGHS) + LSE gradient polish + Multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent or initialize
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            sp = float(np.sum(parent))
            if sp > 0.01:
                parent = parent / sp
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Try structured initializations
        for n_init in [400, 600, 800, 1000]:
            x = np.linspace(0, 1, n_init)
            for shape in ['uniform', 'cosine', 'triangle']:
                if shape == 'uniform':
                    a_init = np.ones(n_init)
                elif shape == 'cosine':
                    a_init = 0.5 + 0.5 * np.cos(2 * np.pi * x)
                else:
                    a_init = 1.0 - 2.0 * np.abs(x - 0.5)
                a_init = np.maximum(a_init, 0.01)
                a_init = a_init / np.sum(a_init)
                obj = compute_obj(a_init)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_init.copy()
                    print(f"Init n={n_init} {shape}: obj={obj:.8f}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_cur = len(best_seq)
    a = best_seq.copy()

    # === Phase 1: Alternating LP (main workhorse, ~500s) ===
    print(f"\n=== Phase 1: Alternating LP (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(500.0, remaining * 0.35)
    a1 = alternating_lp(a, time_budget=lp_budget, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        a = a1.copy()
        print(f"  Phase 1: obj={best_score:.8f}")

    # === Phase 2: LSE gradient polish ===
    print(f"\n=== Phase 2: LSE Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        lse_budget = min(100.0, remaining * 0.1)
        a2 = lse_gradient_polish(a, time_budget=lse_budget)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            a = a2.copy()
            print(f"  Phase 2: obj={best_score:.8f}")
        else:
            print(f"  Phase 2: obj={obj2:.8f} (no improvement)")

    # === Phase 3: More LP after polish ===
    print(f"\n=== Phase 3: LP After Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        lp_budget3 = min(200.0, remaining * 0.15)
        a3 = alternating_lp(a, time_budget=lp_budget3, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            a = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_base = len(best_seq)
    n_options = sorted(set([
        max(200, n_base - 200), max(200, n_base - 100),
        max(200, n_base - 50), n_base,
        n_base + 50, n_base + 100, n_base + 200, n_base + 400,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try == n_base:
            continue
        a_try = resample_to_n(best_seq, n_try)
        alloc = min(60.0, remaining * 0.15)
        a_opt = alternating_lp(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Subgradient + LP on best ===
    print(f"\n=== Phase 5: Subgradient + Final LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_budget = min(60.0, remaining * 0.1)
        a_sg = subgradient_topk(best_seq, time_budget=sg_budget, top_k=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        remaining = deadline - time.monotonic()
        if remaining > 20:
            lp_budget5 = min(remaining * 0.3, 100.0)
            a_final = alternating_lp(best_seq, time_budget=lp_budget5)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # === Phase 6: Random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(20):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            if r % 4 == 0:
                scale = 0.03 + 0.01 * r
                a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif r % 4 == 1:
                x = np.linspace(0, 1, n_best)
                freq = (r + 2) % 7 + 1
                w = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
                a_r = best_seq * np.maximum(w, 0.1)
            elif r % 4 == 2:
                a_r = best_seq * rng.uniform(0.6, 1.4, n_best)
            else:
                # Try different n
                n_alt = n_best + rng.integers(-100, 100)
                n_alt = max(200, min(2000, n_alt))
                a_r = resample_to_n(best_seq, n_alt)
                n_best = len(a_r)

            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
            alloc = min(remaining * 0.15, 40.0)
            a_opt = alternating_lp(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
