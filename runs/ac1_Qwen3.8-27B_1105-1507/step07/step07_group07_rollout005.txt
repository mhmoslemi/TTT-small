```python
"""
Adam-based optimizer with adaptive log-sum-exp smoothing and spectral projection
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_smoothed_grad(a, tau):
    """
    Compute the gradient of 2n*logsumexp(conv(a,a)/tau) w.r.t. a.
    Returns (smoothed_obj, grad_a).
    """
    n = len(a)
    conv = compute_conv(a)
    c = conv / tau
    c_max = c.max()
    exp_c = np.exp(c - c_max)
    Z = exp_c.sum()
    w = exp_c / Z
    
    # Objective
    F = 2.0 * n * (c_max + np.log(Z))
    
    # Gradient: da_i = 2n/tau * sum_j a[j] * w[i+j]
    # Compute via FFT: g = conv(a[::-1], w)[n-1:2n-1]
    fft_size = next_pow2(3 * n - 1)
    a_rev = a[::-1]
    d = irfft(rfft(a_rev, fft_size) * rfft(w, fft_size), fft_size)
    g = d[n-1:2*n-1]  # length n
    
    da = 2.0 * n * g / tau
    return F, da
