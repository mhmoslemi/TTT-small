    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute objective with sum(a)=1 normalization assumed."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

    n = len(a)
    conv = compute_conv(a)
    # Shift for numerical stability
    max_conv = np.max(conv)
    log_weights = beta * (conv - max_conv)
    weights = np.exp(log_weights)
    weights /= np.sum(weights)
    obj = 2.0 * n * (max_conv + np.log(np.sum(weights)) / beta)
    return obj, weights, conv

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective with softmax parameterization
2. LP alternating with adaptive mixing
3. Multi-n exploration and final polish
The key innovation is efficient FFT-based gradient computation for the smoothed objective,
enabling many optimization iterations within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_smooth_obj_grad(a, beta):
    """
    Compute smooth objective and its gradient w.r.t. a (a is normalized, sum=1).
    Uses log-sum-exp smoothing of the max.
    Returns (obj, grad_a) where grad_a is the gradient of the smoothed objective.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Softmax weights for the log-sum-exp
    log_w = beta * (conv - max_conv)
    # Numerical stability
    max_log = float(np.max(log_w))
    exp_w = np.exp(log_w - max_log)
    sum_exp_w = float(np.sum(exp_w))
    weights = exp_w / sum_exp_w
    
    # Smooth objective
    obj = 2.0 * n * (max_conv + (max_log + np.log(sum_exp_w)) / beta)
    
    # Gradient: grad_a[j] = 4n * sum_m a[m] * weights[j+m]
    # = 4n * convolve(a[::-1], weights)[n-1+j]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    a_rev = a[::-1]
    fa = fft(a_rev, size)
    fw = fft(weights, size)
    conv_full = ifft(fa * fw).real[:3 * n - 2]
    grad_a = 4.0 * n * conv_full[n - 1:n - 1 + n]
    
    return obj, grad_a


def softmax_z_to_a(z):
    """Convert z to a = softmax(z) = exp(z)/sum(exp(z))."""
    z_shift = z - np.max(z)
    exp_z = np.exp(z_shift)
    a = exp_z / np.sum(exp_z)
    return a


def a_to_z(a):
    """Convert a to z = log(a) (for softmax parameterization)."""
    return np.log(np.maximum(a, 1e-30))


def lbfgs_smooth(a_init, beta_list, time_budget=600.0):
    """
    Run L-BFGS with increasing beta values.
    Optimizes in z-space where a = softmax(z).
    """
    start = time.monotonic()
    n = len(a_init)
    a_init = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a_init))
    if s < 1e-10:
        a_init = np.ones(n) / n
        s = 1.0
    a_init = a_init / s
    
    z = a_to_z(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    
    for beta in beta_list:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break
        
        print(f"    L-BFGS beta={beta:.0f}, elapsed={elapsed:.1f}s, current_obj={best_obj:.8f}")
        
        # Cache to avoid recomputation
        cache = {}
        
        def fun(z_val):
            a_val = softmax_z_to_a(z_val)
            key = (z_val[0], z_val[-1], len(z_val))
            obj, _ = compute_smooth_obj_grad(a_val, beta)
            return obj
        
        def jac(z_val):
            a_val = softmax_z_to_a(z_val)
            _, grad_a = compute_smooth_obj_grad(a_val, beta)
            # Project to z-space: grad_z[j] = a[j] * (grad_a[j] - sum(a*grad_a))
            mean_g = float(np.dot(a_val, grad_a))
            grad_z = a_val * (grad_a - mean_g)
            return grad_z
        
        try:
            res = minimize(fun, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': 2000, 'gtol': 1e-12,
                                   'ftol': 1e-15, 'maxfun': 5000,
                                   'iprint': -1})
            
            z_opt = res.x
            a_opt = softmax_z_to_a(z_opt)
            a_opt = np.maximum(a_opt, 0)
            s_opt = float(np.sum(a_opt))
            a_opt = a_opt / s_opt
            
            obj = compute_obj(a_opt)
            if obj < best_obj:
                best_obj = obj
                best_a = a_opt.copy()
                z = z_opt.copy()
            else:
                # Keep old z but try from slightly perturbed
                z = z_opt.copy()
            
            print(f"      -> obj={obj:.8f} (best={best_obj:.8f}), iters={res.nfev}")
        except Exception as e:
            print(f"      L-BFGS error: {e}")
            break
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    
    A_mat = build_conv_matrix(a)
    
    # LP: min M s.t. A_mat@g - M <= 0, sum(g)=1, g>=0, M>=0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A_mat, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternating(a, time_budget=120.0, max_iters=100):
    """LP alternating optimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Adaptive line search
        candidate_ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0.01 < last_t < 0.99:
            extra = [last_t * 0.5, last_t * 0.75, last_t * 1.25, last_t * 1.5]
            candidate_ts = sorted(set([t for t in candidate_ts + extra if 0.01 <= t <= 1.0]))
        
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()
        
        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            last_t = best_t_val
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb to escape
                rng = np.random.default_rng(it * 1000 + n)
                scale = 0.02 + 0.01 * min(stall // 10, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
    
    return best_a


def direct_subgradient(a, n_iters=2000, time_budget=120.0):
    """Direct subgradient method with adaptive step size on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.1
    momentum = 0.7
    v = np.zeros(n)
    
    for it in range(n_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: 2 * a[k_star - j] for valid j
        grad = np.zeros(n)
        for j in range(n):
            j2 = k_star - j
            if 0 <= j2 < n:
                grad[j] = 2.0 * a[j2]
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        
        # Step
        step = alpha / (1.0 + 0.01 * it)
        a_new = a - step * v
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.02, 0.5)
        else:
            alpha *= 0.97
            if alpha < 0.001:
                break
    
    return best_a


def fft_refine(a, n_iters=500, time_budget=60.0, relaxation=0.5):
    """FFT-based flat spectrum refinement (Gerchberg-Saxton style)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe
        
        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat
        
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 40:
                    rng = np.random.default_rng(it * 7)
                    a = best_a * np.exp(rng.normal(0, 0.04, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_obj = compute_obj(a)
                    stall = 0
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
