"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP alternating with golden-section line search, QP-based directions,
mirror descent (multiplicative updates), structured perturbations, and multi-n exploration.
Key innovations: golden-section step size, QP-then-LP hybrid directions, multiplicative
weight updates as escape mechanism, and adaptive n-search with resampling.
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_weights(conv, T):
    """Compute softmax weights for smooth max approximation."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    return weights


def gradient_from_weights(weights, a):
    """Compute gradient of smooth max w.r.t. a given softmax weights on conv bins."""
    return np.correlate(weights, a, mode='valid')


def golden_section_search(f, lo, hi, tol=1e-6, max_iter=50):
    """Golden section search for minimum of f on [lo, hi]. Returns (x, f(x))."""
    gr = (math.sqrt(5) - 1) / 2  # golden ratio conjugate
    a_pt = lo + (1 - gr) * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = f(a_pt)
    fb = f(b_pt)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = lo + (1 - gr) * (hi - lo)
            fa = f(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = f(b_pt)
    x_opt = (lo + hi) / 2
    return x_opt, f(x_opt)


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Try cvxpy first
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in ['CLARABEL', 'MOSEK', 'SCS', 'ECOS', 'PDLP']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def solve_qp_minpeak(a):
    """
    Solve QP: min sum_k (a*g)[k]^2 s.t. sum(g)=1, g>=0.
    Returns g_normalized or None.
    This gives a "smoother" direction than LP.
    """
    n = len(a)
    a = a / np.sum(a)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        obj = cp.sum_squares(a @ g)  # This is sum of (conv(a,g))[k]^2... no
        # Actually we need conv(a, g), not a @ g
        # conv(a, g)[k] = sum_i a[i]*g[k-i]
        # This is A @ g where A is the conv matrix
        A = build_conv_matrix(a)
        obj = cp.sum_squares(A @ g)
        constraints = [cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(obj), constraints)
        for solver in ['CLARABEL', 'MOSEK', 'SCS']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except ImportError:
        pass

    # Fallback: use the analytical solution for the case where we ignore the active constraints
    # The unconstrained min of sum_k (A@g)^2 s.t. sum(g)=1 is g ~ (A^T A)^{-1} 1
    # This is expensive for large n, so use a simple approximation
    return None


def lp_alternating_golden(a, time_budget=60.0, verbose=False):
    """
    LP alternating with golden-section line search for optimal step size.
    Also tries "over-shooting" (t > 1) via extrapolation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Golden section search for optimal t in [0.01, 1.5]
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        # Quick scan to find good bracket
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.3]
        vals = [(t, obj_at_t(t)) for t in t_candidates]
        best_t_scan = min(vals, key=lambda x: x[1])

        # Golden section in a narrower range around the best scan point
        t_center = best_t_scan[0]
        t_lo = max(0.01, t_center - 0.3)
        t_hi = min(1.5, t_center + 0.3)
        t_opt, obj_opt = golden_section_search(obj_at_t, t_lo, t_hi, tol=0.01, max_iter=15)

        # Also check the scan best
        if best_t_scan[1] < obj_opt:
            t_opt = best_t_scan[0]
            obj_opt = best_t_scan[1]

        if obj_opt < best_obj - 1e-10:
            best_obj = obj_opt
            a_try = (1.0 - t_opt) * a + t_opt * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            best_a = a_try.copy()
            a = a_try
            stall_count = 0
            if verbose and it % 20 == 0:
                print(f"        LP-GS it{it}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:
            stall_count += 1
            if stall_count > 20:
                # Structured perturbation
                rng = np.random.default_rng(hash((it, n, stall_count)) % (2**32))
                # Add a small sinusoidal perturbation
                perturbation = 0.02 * np.sin(np.linspace(0, 2 * np.pi * rng.integers(1, 5), n))
                a = (a + perturbation)
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall_count = 0

    return best_a


def mirror_descent(a, time_budget=60.0, verbose=False):
    """
    Multiplicative weight (mirror descent) updates targeting the peak.
    a[i] *= exp(-eta * a[k* - i]) then normalize.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.5
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient: d conv[k_star] / d a[i] = a[k_star - i] (if valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        # Multiplicative update
        log_scale = -eta * grad
        log_scale -= np.mean(log_scale)  # centering
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if eta < 2.0:
                eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            if stall > 30:
                eta *= 0.8
                a = best_a.copy()
                stall = 0
                if eta < 0.01:
                    break
            elif stall > 10:
                # Try smaller step
                eta *= 0.9
                a = best_a.copy()

    return best_a


def smooth_max_gd(a, time_budget=40.0, T_init=0.001, T_min=0.0001):
    """Smooth max gradient descent with adaptive temperature and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    T = T_init
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        weights = smooth_max_weights(conv, T)
        grad = gradient_from_weights(weights, a)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break

        if it > 20:
            T = max(T_min, T * 0.998)

    return best_a


def top_k_direction(a, K=5):
    """
    Compute direction that reduces the top K convolution peaks simultaneously.
    Solves: min max_{k in topK} (a*g)[k] s.t. sum(g)=1, g>=0
    Approximated by: min sum_{k in topK} (a*g)[k]^2 s.t. sum(g)=1, g>=0
    """
    n = len(a)
    conv = compute_conv(a)
    top_indices = np.argsort(conv)[-K:]

    # Build reduced constraint matrix
    A_red = np.zeros((K, n))
    for idx, k in enumerate(top_indices):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_red[idx, i] = a[j]

    # Solve: min sum_k (A_red @ g)^2 s.t. sum(g)=1, g>=0
    # Gradient of objective w.r.t. g: 2 * A_red^T @ (A_red @ g)
    # At optimum: A_red^T @ (A_red @ g) = lambda * 1 (for free variables)
    # Use projected gradient method
    g = np.ones(n) / n
    lr_pk = 0.1 / (np.max(np.sum(A_red ** 2, axis=1)) + 1e-10)

    for _ in range(200):
        Ag = A_red @ g
        grad_g = 2.0 * A_red.T @ Ag
        grad_g -= np.mean(grad_g)  # center for sum constraint
        g -= lr_pk * grad_g
        g = np.maximum(g, 0)
        s = np.sum(g)
        if s < 1e-10:
            break
        g = g / s

    return g


def combined_direction_search(a, time_budget=60.0, verbose=False):
    """
    Combine LP direction, QP direction, top-K direction, and mirror descent.
    Use golden section to find best blend.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Compute available directions
    g_lp, _ = solve_lp_minpeak(a)
    g_topk = top_k_direction(a, K=5)
    g_qp = solve_qp_minpeak(a)

    directions = []
    if g_lp is not None:
        directions.append(('LP', g_lp))
    directions.append(('TopK', g_topk))
    if g_qp is not None:
        directions.append(('QP', g_qp))

    # Also compute mirror descent direction (one step)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
    g_md = a * np.exp(-0.5 * grad)
    g_md = np.maximum(g_md, 0)
    g_md = g_md / np.sum(g_md)
    directions.append(('MD', g_md))

    # For each direction, find optimal t via golden section
    for name, g_dir in directions:
        if time.monotonic() - start > time_budget:
            break

        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g_dir
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            return compute_obj(a_try / s_try)

        # Quick scan
        t_scan = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        vals = [obj_at_t(t) for t in t_scan]
        best_i = int(np.argmin(vals))
        best_t = t_scan[best_i]
        best_val = vals[best_i]

        # Fine-tune with golden section
        if 0 < best_i < len(t_scan) - 1:
            lo = t_scan[max(0, best_i - 1)]
            hi = t_scan[min(len(t_scan) - 1, best_i + 1)]
            t_opt, val_opt = golden_section_search(obj_at_t, lo, hi, tol=0.02, max_iter=12)
            if val_opt < best_val:
                best_t, best_val = t_opt, val_opt

        if best_val < best_obj - 1e-10:
            best_obj = best_val
            a_try = (1.0 - best_t) * a + best_t * g_dir
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            best_a = a_try.copy()
            a = a_try
            if verbose:
                print(f"        Combined {name}: obj={best_obj:.8f}, t={best_t:.4f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, rng, mode=0):
    """Generate structured initial sequences."""
    x = np.arange(n)
    if mode == 0:
        # Tilted uniform
        a = np.linspace(0.5, 1.5, n)
    elif mode == 1:
        # Cosine ramp
        a = np.cos(np.linspace(0, np.pi * 0.7, n)) ** 2 + 0.1
    elif mode == 2:
        # Exponential ramp
        a = np.exp(np.linspace(-0.5, 0.5, n))
    elif mode == 3:
        # Double peak (asymmetric)
        c1, c2 = n * 0.3, n * 0.7
        w1, w2 = n * 0.15, n * 0.2
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + 0.7 * np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
    elif mode == 4:
        # S-curve (sigmoid)
        a = 1.0 / (1.0 + np.exp(-5 * (x / n - 0.4)))
    elif mode == 5:
        # Linear tilt with noise
        a = np.linspace(0.3, 1.7, n) + 0.1 * rng.normal(0, 1, n)
    elif mode == 6:
        # Inverse U (high at edges)
        a = 1.0 + 0.5 * np.abs(2 * x / n - 1)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating with golden section,
    mirror descent, combined direction search, multi-n exploration,
    and structured restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_n = 800
        best_seq = structured_init(best_n, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={best_n}): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Intensive LP Alternating with Golden Section ===
    print(f"\n=== Phase 1: LP Alternating + Golden Section (n={n_cur}) ===")
    phase1_start = time.monotonic()
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining * 0.2, 200.0)
        a_new = lp_alternating_golden(best_seq.copy(), time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  LP-GS Round {round_idx}: obj={best_score:.8f}")
        else:
            print(f"  LP-GS Round {round_idx}: obj={obj:.8f} (no improvement)")

    print(f"  Phase 1 time: {time.monotonic() - phase1_start:.1f}s")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.15)
        a_md = mirror_descent(best_seq.copy(), time_budget=alloc, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            n_cur = len(best_seq)
            print(f"  Mirror Descent: obj={best_score:.8f} **")
        else:
            print(f"  Mirror Descent: obj={obj_md:.8f}")

    # === Phase 3: Combined Direction Search ===
    print(f"\n=== Phase 3: Combined Direction Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for cd_round in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(60.0, remaining * 0.12)
            a_cd = combined_direction_search(best_seq.copy(), time_budget=alloc, verbose=True)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                n_cur = len(best_seq)
                print(f"  Combined Dir Round {cd_round}: obj={best_score:.8f} **")
            else:
                print(f"  Combined Dir Round {cd_round}: obj={obj_cd:.8f}")

    # === Phase 4: Smooth Max Gradient Descent ===
    print(f"\n=== Phase 4: Smooth Max GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for T_init in [0.002, 0.0005, 0.0001]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(40.0, remaining * 0.1)
            a_sm = smooth_max_gd(best_seq.copy(), time_budget=alloc, T_init=T_init)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                n_cur = len(best_seq)
                print(f"  Smooth Max (T={T_init}): obj={best_score:.8f} **")
            else:
                print(f"  Smooth Max (T={T_init}): obj={obj_sm:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 50, n_best - 25, n_best,
        n_best + 25, n_best + 50, n_best + 100,
        600, 700, 800, 900, 1000, 1100, 1200, 1300, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP alternating
        alloc_lp = min(remaining * 0.1, 30.0)
        a_opt = lp_alternating_golden(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        # Mirror descent polish
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.01:
            a_opt = mirror_descent(a_opt, time_budget=min(20.0, remaining * 0.08))
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Structured Restarts ===
    print(f"\n=== Phase 6: Structured Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    for r in range(15):

        # Blend with current best for warm start
        if n_r == n_final:
            a_r = 0.5 * a_r + 0.5 * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        # Optimize
        alloc_r = min(remaining * 0.1, 25.0)
        a_opt = lp_alternating_golden(a_r, time_budget=alloc_r * 0.5)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = mirror_descent(a_opt, time_budget=min(alloc_r * 0.4, remaining * 0.3))

        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_final = len(best_seq)
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for polish_round in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.2, 50.0)
            a_lp = lp_alternating_golden(best_seq.copy(), time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP {polish_round}: obj={best_score:.8f} **")
            else:
                print(f"  Final LP {polish_round}: obj={obj_lp:.8f}")

        # Combined direction final push
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_cd = combined_direction_search(best_seq.copy(), time_budget=min(30.0, remaining * 0.2), verbose=True)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final Combined: obj={best_score:.8f} **")

        # Mirror descent final
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_md = mirror_descent(best_seq.copy(), time_budget=min(20.0, remaining * 0.2))
            obj_md = compute_obj(a_md)
            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                print(f"  Final MD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
