"""
Hybrid optimizer using coordinate descent, smoothed gradient, and LP alternating
to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar, linprog
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def coordinate_descent_step(a, i, tol=1e-10):
    """Optimize coordinate i of a (normalized sum=1) via 1D search."""
    n = len(a)
    a = a.copy()
    s = 1.0  # a is normalized
    
    # Save old value
    ai_old = a[i]
    
    # For the 1D problem, we need to find a[i] that minimizes:
    # 2n * max_k conv(a,a)[k] / (sum(a))^2
    # where only a[i] varies, and we'll re-normalize.
    
    # Actually, let's parameterize: set a[i] = t, keep others fixed, then normalize.
    # a_norm[j] = a[j] / (sum(a) - ai_old + t) for j != i
    # a_norm[i] = t / (sum(a) - ai_old + t)
    
    # This is getting complex. Let me just use a simpler approach:
    # Fix all other coordinates, vary a[i], and normalize.
    
    others_sum = 1.0 - ai_old  # since sum = 1
    
    def obj_1d(t):
        t = max(t, 0.0)
        new_sum = others_sum + t
        if new_sum < 1e-10:
            return np.inf
        a_new = a.copy()
        a_new[i] = t
        a_new = a_new / new_sum
        conv = compute_conv(a_new)
        return 2.0 * n * np.max(conv)
    
    # Search range for t: [0, 2*ai_old + 0.01/n] or so
    # The optimal t should be near ai_old
    result = minimize_scalar(obj_1d, bounds=(0, max(2*ai_old, 1/n) + 0.1/n), 
                            method='bounded', options={'xatol': tol})
    
    if result.success:
        a[i] = result.x
        a = a / np.sum(a)
    
    return a

def coordinate_descent(a, max_sweeps=10, time_budget=60.0):
    """Cyclic coordinate descent."""
    n = len(a)
    a = a.copy()
    a = a / np.sum(a)
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            a_new = coordinate_descent_step(a, i)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                a = a_new
                best_obj = obj_new
                improved = True
        if not improved:
            break
    
    return a

Optimized search for low-autocorrelation non-negative sequences using
fast coordinate descent with incremental convolution, smoothed gradient
(L-BFGS-B with log-sum-exp), LP alternating, and multi-n exploration.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def coord_descent_1d(a, i, conv_full):
    """
    Optimize coordinate i given the current full convolution.
    Uses fast incremental update.
    
    a: current normalized sequence (sum=1)
    i: coordinate to optimize
    conv_full: current convolution of a with itself (length 2n-1)
    
    Returns: (new_a, new_conv, new_obj)
    """
    n = len(a)
    ai = a[i]
    others_sum = 1.0 - ai
    
    # Precompute the "base" convolution (without a[i]'s contribution)
    # and the linear coefficients
    # conv[k] = base[k] + 2*ai*a[k-i] (if 0<=k-i<n and k!=2i)
    #                  + ai^2 (if k==2i)
    
    # For a candidate t (new value of a[i]):
    # conv_new[k] = conv[k] - old_contrib[k] + new_contrib[k]
    # where old_contrib[k] = 2*ai*a[k-i] or ai^2
    #       new_contrib[k] = 2*t*a[k-i] or t^2
    
    # After normalization by (others_sum + t):
    # conv_norm_new[k] = conv_new[k] / (others_sum + t)^2
    # objective = 2n * max(conv_norm_new)
    
    # Precompute: for each k, what's the coefficient of t in conv_new[k]?
    # conv_new[k] = conv[k] + (t - ai) * a[k-i] * (1 if k-i==i else 2) - (t^2 - ai^2 if k==2i else 0)
    
    # Simpler: just compute the change.
    # Let delta = t - ai
    # conv_new[2i] = conv[2i] + 2*ai*delta + delta^2 = conv[2i] + 2*(t*ai) - ai^2 + t^2 - 2*ai*t + ai^2
    # Actually: conv[2i] includes ai^2. New value has t^2.
    # So conv_new[2i] = conv[2i] - ai^2 + t^2
    # For k != 2i, 0 <= k-i < n: conv_new[k] = conv[k] - 2*ai*a[k-i] + 2*t*a[k-i]
    # For other k: conv_new[k] = conv[k]
    
    # Define the 1D objective
    def obj_1d(t):
        if t < 0:
            t = 0.0
        s_new = others_sum + t
        if s_new < 1e-10:
            return 1e10
        # Compute max of conv_new / s_new^2
        # conv_new[k] for all k
        # Instead of full recomputation, use the structure:
        # max_k conv_new[k] / s_new^2 = max_k conv_new[k] / s_new^2
        
        # We need max of:
        # - For k = 2i: (conv[2i] - ai^2 + t^2) / s_new^2
        # - For 0<=k-i<n, k!=2i: (conv[k] + 2*(t-ai)*a[k-i]) / s_new^2
        # - For other k: conv[k] / s_new^2
        
        # The "other k" part: max of conv[k]/s_new^2 for k not affected
        # = max_oth
        # The affected parts depend on t.
        
        # For a fast evaluation, let's just compute the max directly:
        # Conv_new can be computed in O(n) using the precomputed structure.
        
        # Affected indices: k where 0 <= k-i < n → k in [i, i+n-1] (intersect [0, 2n-2])
        k_min = max(0, i)
        k_max = min(2*n-2, i+n-1)
        
        # For these k: conv_new[k] = conv[k] + 2*(t-ai)*a[k-i] (if k != 2i)
        #                                        or conv[k] - ai^2 + t^2 (if k == 2i)
        
        # Precompute for speed:
        # affected_conv[k] = conv[k] + 2*(t-ai)*a[k-i] for k in [k_min, k_max], k != 2i
        # affected_conv[2i] = conv[2i] - ai^2 + t^2

        #     max over k in [k_min,k_max], k!=2i of (conv[k] + 2*(t-ai)*a[k-i]),
        #     conv[2i] - ai^2 + t^2

        # = 2*(t-ai) * a[k*-i] + conv[k*] where k* = argmax
        # But we need the max over ALL k, so we need to compute it.
        
        # For speed, let's vectorize:
        ks = np.arange(k_min, k_max + 1)
        affected = conv_full[ks].copy()
        # Add linear term
        idx = ks - i  # these are the a indices

        # Fix the k=2i term

            affected[2*i - k_min] = conv_full[2*i] - ai*ai + t*t

            max_unaff_left = np.max(conv_full[:k_min])
        else:
            max_unaff_left = -np.inf
        if k_max < 2*n-2:
            max_unaff_right = np.max(conv_full[k_max+1:])
        else:
            max_unaff_right = -np.inf
        max_unaffected = max(max_unaff_left, max_unaff_right) if (k_min > 0 or k_max < 2*n-2) else -np.inf
        
        max_conv = max(max_affected, max_unaffected)
        return 2.0 * n * max_conv / (s_new * s_new)
    
    # Use bounded minimization
    # t should be in [0, some upper bound]
    # Upper bound: if t is very large, obj approaches 2n * (something) / t^2 * t^2 = 2n * max(a[idx])/... 
    # Actually for large t: conv_new[2i] ~ t^2, s_new ~ t, so obj ~ 2n*t^2/t^2 = 2n * 1 = 2n
    # That's worse than our current ~1.5, so the optimum is at moderate t.
    # Good upper bound: 2*ai + 0.1/n
    
    ub = max(2.0 * ai + 0.1 / n, 1.0 / n)
    result = minimize_scalar(obj_1d, bounds=(0.0, ub), method='bounded',
                             options={'xatol': 1e-12})
    
    t_opt = max(0.0, result.x)
    a_new = a.copy()
    a_new[i] = t_opt
    a_new = a_new / np.sum(a_new)
    
    # Compute new convolution
    conv_new = compute_conv(a_new)
    obj_new = 2.0 * n * np.max(conv_new)
    
    return a_new, conv_new, obj_new


def coordinate_descent(a, time_budget=120.0, verbose=False):
    """Cyclic coordinate descent with fast 1D subproblem solving."""

    a = a.copy()
    a = a / np.sum(a)
    conv = compute_conv(a)
    best_obj = 2.0 * n * np.max(conv)
    start = time.monotonic()
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        obj_start = best_obj

            if obj_new < best_obj - 1e-14:
                a = a_new
                conv = conv_new
                best_obj = obj_new

            elapsed = time.monotonic() - start
            print(f"  CD sweep {sweep}: obj={best_obj:.10f} (delta={obj_start-best_obj:.2e}) [{elapsed:.1f}s]")
        if obj_start - best_obj < 1e-12:
            break
    
    return a

"""
Optimized search for low-autocorrelation non-negative sequences.
Uses fast coordinate descent with incremental convolution updates,
LP alternating method, smoothed gradient (log-sum-exp + L-BFGS-B),
and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def obj_1d_fast(t, ai, others_sum, conv_full, a, i, n):
    """Evaluate 1D objective for coordinate i set to value t (unnormalized)."""
    if t < 0:
        t = 0.0
    s_new = others_sum + t
    if s_new < 1e-10:
        return 1e10
    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)
    ks = np.arange(k_min, k_max + 1)
    idx = ks - i
    affected = conv_full[ks] + 2.0 * (t - ai) * a[idx]
    if 2 * i >= k_min and 2 * i <= k_max:
        pos = 2 * i - k_min
        affected[pos] = conv_full[2 * i] - ai * ai + t * t
    max_affected = np.max(affected)
    if k_min > 0:
        max_unaff = np.max(np.concatenate([conv_full[:k_min], conv_full[k_max + 1:2*n-1]])) if k_max < 2*n-2 else np.max(conv_full[:k_min])
    elif k_max < 2 * n - 2:
        max_unaff = np.max(conv_full[k_max + 1:])
    else:
        max_unaff = -np.inf
    max_conv = max(max_affected, max_unaff)
    return 2.0 * n * max_conv / (s_new * s_new)


def coord_optimize_1d(a, i, conv_full, max_evals=25):
    """Golden section search for optimal value of coordinate i."""
    n = len(a)
    ai = a[i]
    others_sum = 1.0 - ai
    ub = max(3.0 * ai + 0.2 / n, 2.0 / n)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = ub
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = obj_1d_fast(x1, ai, others_sum, conv_full, a, i, n)
    f2 = obj_1d_fast(x2, ai, others_sum, conv_full, a, i, n)
    evals = 2
    for _ in range(max_evals - 2):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = obj_1d_fast(x1, ai, others_sum, conv_full, a, i, n)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = obj_1d_fast(x2, ai, others_sum, conv_full, a, i, n)
        evals += 1
    t_opt = (lo + hi) / 2.0
    a_new = a.copy()
    a_new[i] = t_opt
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)

    a_new = a_new / s
    conv_new = compute_conv(a_new)
    obj_new = 2.0 * n * np.max(conv_new)

    """Cyclic coordinate descent with golden section 1D subproblems."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()

    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_sweeps = 0

def smoothed_obj_and_grad(x, tau):
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):

    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)

    x = np.log(a) - np.log(a).max()
    # Finer temperature schedule

    for tau in taus:

                best_a = a_new.copy()

        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_alternate_optimize(a, time_budget=120.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    step_scales = [0.5, 0.7, 0.9, 1.0, 0.3, 0.6, 0.8, 1.0, 1.1, 1.2]
    scale_idx = 0
    while time.monotonic() - start < time_budget:
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        t = step_scales[scale_idx % len(step_scales)]
        scale_idx += 1
        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP alt: obj={best_obj:.10f} (t={t:.2f}) [{elapsed:.1f}s]")
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)
    elif mode == 3:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        a = a / np.sum(a)
    else:
        a = np.exp(-((np.arange(n) - n/2)**2) / (n*n/10))
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: coordinate descent + LP alternating + smoothed gradient,
    with multi-n exploration and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"\n=== Starting optimization (n={n_parent}, obj={best_score:.10f}) ===")

    # === Phase 1: Coordinate Descent (fast local refinement) ===
    print(f"\n=== Phase 1: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.2, 150.0), verbose=True)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  After CD: obj={best_score:.10f}")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_lp = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.2, 120.0), verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  After LP: obj={best_score:.10f}")

    # === Phase 3: Coordinate Descent again ===
    print(f"\n=== Phase 3: Coordinate Descent (round 2) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_cd2 = coordinate_descent(best_seq, time_budget=min(remaining * 0.15, 100.0), verbose=True)
        obj = compute_obj(a_cd2)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd2.copy()
            print(f"  After CD2: obj={best_score:.10f}")

    # === Phase 4: Smoothed Gradient ===
    print(f"\n=== Phase 4: Smoothed Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_grad = gradient_optimize(best_seq, time_budget=min(remaining * 0.15, 90.0), verbose=True)
        obj = compute_obj(a_grad)
        if obj < best_score:
            best_score = obj
            best_seq = a_grad.copy()
            print(f"  After gradient: obj={best_score:.10f}")

    # === Phase 5: More Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (round 3) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_cd3 = coordinate_descent(best_seq, time_budget=min(remaining * 0.15, 100.0), verbose=True)
        obj = compute_obj(a_cd3)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd3.copy()
            print(f"  After CD3: obj={best_score:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        max(400, n_best - 100), max(500, n_best - 50),
        n_best,
        n_best + 50, n_best + 100,
        700, 850, 900, 950, 1000, 1100
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1500]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if n_try == n_best:
            continue
        a_try = resample_sequence(best_seq, n_try)
        # Quick CD
        alloc = min(remaining * 0.1, 40.0)
        a_opt = coordinate_descent(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 7: Random restarts with CD ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    n_restarts = max(2, int(min(remaining / 40.0, 6)))
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break
        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
        elif r == 1:
            n_r = n_best + int(rng.integers(-50, 50))
            n_r = max(500, min(1200, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
        elif r == 2:
            a_r = best_seq + 0.01 * rng.normal(0, 1, n_best)
            a_r = (a_r + a_r[::-1]) / 2.0
        else:
            a_r = random_init(n_best, rng)
        a_r = np.maximum(a_r, 0)
        a_r = a_r / np.sum(a_r)
        alloc = min(remaining * 0.15, 35.0)
        a_opt = coordinate_descent(a_r, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        # CD
        alloc = min(remaining * 0.3, 60.0)
        a_f = coordinate_descent(best_seq, time_budget=alloc)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.10f}")
        # LP
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_lp = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.3, 40.0))
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.10f}")
        # Gradient
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_g = gradient_optimize(best_seq, time_budget=min(remaining * 0.3, 40.0))
            obj = compute_obj(a_g)
            if obj < best_score:
                best_score = obj
                best_seq = a_g.copy()
                print(f"  Final grad: obj={best_score:.10f}")
        # One more CD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_c = coordinate_descent(best_seq, time_budget=min(remaining * 0.4, 30.0))
            obj = compute_obj(a_c)
            if obj < best_score:
                best_score = obj
                best_seq = a_c.copy()
                print(f"  Final CD2: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_score = val
    else:
        result = [1.0] * 800
        final_score = compute_obj(np.array(result))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_score:.10f}, time={elapsed:.1f}s ===")
    return result
