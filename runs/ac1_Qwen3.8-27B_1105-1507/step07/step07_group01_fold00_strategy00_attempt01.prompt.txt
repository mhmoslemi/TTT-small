def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

    s = float(np.sum(a))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_and_weights(conv, T):
    """Compute smooth max value and softmax weights."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    return smooth_val, weights


def compute_gradient_smooth(weights, a):
    """
    Compute gradient of smooth max w.r.t. a.
    grad[i] = sum_j weights[i+j] * a[j] for valid indices.
    Uses np.correlate for efficiency.

    # grad[i] = sum_j a[j] * weights[i+j] for j=0..n-1 where i+j < len(weights)
    # This is np.correlate(weights, a, mode='valid')
    grad = np.correlate(weights, a, mode='valid')
    return grad

    M = cp.Variable()

                v = np.zeros(n)
                stall = 0

        # Temperature anneal
        if it > 50:
            T = max(T_min, T_init * (0.997 ** it))

def chase_peak_momentum(a, time_budget=30.0):
    """
    Chase-the-peak with Nesterov momentum and adaptive step size.
    Targets the maximum convolution bin.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.02
    v = np.zeros(n)
    momentum = 0.8
    stall = 0

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: d(conv[k_star])/d(a[i]) = a[k_star - i] if 0 <= k_star-i < n
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
            else:
                # For i where k_star - i < 0, check the other side
                # conv[k_star] = sum_i a[i]*a[k_star-i], so only one term per i
                pass

        # Also need to handle the case where k_star >= n
        # In that case, conv[k_star] = sum_{i=k_star-n+1}^{n-1} a[i]*a[k_star-i]
        # d/d(a[i]) = a[k_star-i] for i in [k_star-n+1, n-1]

        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v - step_size * grad_centered
        a_new = simplex_project(a + 0.5 * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0 and step_size < 0.05:
                step_size = min(step_size * 1.03, 0.05)
        else:
            stall += 1
            if stall > 40:
                step_size *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if step_size < 1e-4:
                    break

    return best_a


def qp_direction_blend(a, time_budget=20.0):

        # Also compute top-peaks weighted direction
        conv = compute_conv(a)
        top_k = max(3, n // 10)
        top_indices = np.argsort(conv)[-top_k:]

        else:
            g_peak = g_qp

        # Blend of QP and peak directions
        g_blend = 0.5 * g_qp + 0.5 * g_peak
        g_blend = np.maximum(g_blend, 0)
        g_blend = g_blend / np.sum(g_blend)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Smooth random via cumulative
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        center = n // 2
        width = max(1, n // 5)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a += 0.5 * np.exp(-((x - center * 0.6) ** 2) / (2 * (width // 2 + 1) ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        a = a / np.sum(a)
    elif mode == 5:
        # Triangle
        x = np.arange(n) / (n - 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n))
    return a
