"""
Improved optimizer for low-peak non-negative autocorrelation sequences.
Combines coordinate descent (exact 1D subproblem), smoothed gradient (L-BFGS-B),
penalty-based peak reduction, and systematic multi-n exploration.
Key novel elements: exact coordinate descent for the max-conv objective,
active-set LP with tighter focus, and structured initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(x, tau):
    """
    Compute smoothed objective and gradient for a = exp(x).
    Objective: logsumexp(conv(a,a)/tau) / (sum(a))^2
    """
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))

    conv = compute_conv(a)

    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)

    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]

    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """Optimize using L-BFGS-B on smoothed objective with adaptive temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    taus = [20.0 / n, 10.0 / n, 5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n,
            0.3 / n, 0.15 / n, 0.08 / n, 0.04 / n, 0.02 / n]

    x = np.log(a) - np.log(a).max()

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break

        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val

        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g

        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-11,
                                'maxls': 80})

        if res.success or res.fun < fun(x):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)

                best_obj = obj_new

        else:

    Coordinate descent: for each coordinate i, solve the exact 1D subproblem
    min_{x>=0} max_k (base[k] + delta_k(x)).
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    max_sweeps = 200
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for i in range(n):
            old_ai = a[i]
            if old_ai < 1e-15:
                continue

            # Compute base convolution (with a[i] = 0)
            a[i] = 0.0
            base_conv = compute_conv(a)
            # The contribution of a[i] = x to each lag:
            # For k where 0 <= k-i < n and k != 2i: delta = 2*x*a[k-i]
            # For k = 2i (if 0 <= 2i < 2n-1): delta = x^2

            # Find the relevant lags
            # k ranges from i to i+n-1 (where k-i is in [0, n-1])
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Build the coefficient vector: delta_k(x) = coef[k] * x or x^2
            # For k = 2i: coefficient is x^2 (quadratic)
            # For k != 2i: coefficient is 2*a[k-i] (linear)

            # We want to minimize max_k (base_conv[k] + delta_k(x)) for x >= 0
            # This is a convex 1D problem. Use bisection on subgradient.

            # Determine bounds for x
            x_lo = 0.0
            x_hi = max(old_ai * 3.0, 1.0 / n * 5.0)

            # Subgradient at x:
            # For each k achieving the max, the subgradient contribution is:
            #   if k == 2i: 2*x
            #   else: 2*a[k-i]
            # The subgradient is the max of (base[k] + delta_k(x)) weighted by...
            # Actually for max of functions, the subgradient is the convex hull of
            # individual gradients at the active set.

            # Simpler approach: use ternary/golden section search since it's 1D convex.
            # But the function is piecewise (max of linear and one quadratic).
            # Let's use a more direct approach: binary search on the derivative.

            def f_at_x(x):
                # Compute max conv value with a[i] = x
                vals = base_conv[k_start:k_end + 1].copy()
                for k_offset in range(len(vals)):
                    k = k_start + k_offset
                    if k == 2 * i:
                        vals[k_offset] += x * x
                    else:
                        j = k - i
                        if 0 <= j < n:
                            vals[k_offset] += 2.0 * x * a[j]
                # Also need to check lags outside [k_start, k_end]
                # But those don't involve a[i], so they're already in base_conv
                return max(np.max(vals), np.max(np.delete(base_conv, range(k_start, k_end + 1))))

            # Golden section search on [x_lo, x_hi]
            invphi = (np.sqrt(5.0) - 1.0) / 2.0
            invphi2 = invphi * invphi

            c = x_hi - invphi * (x_hi - x_lo)
            d = x_lo + invphi * (x_hi - x_lo)
            fc = f_at_x(c)
            fd = f_at_x(d)

            for _ in range(50):
                if fc < fd:
                    x_hi = d
                    d = c
                    fd = fc
                    c = x_hi - invphi * (x_hi - x_lo)
                    fc = f_at_x(c)
                else:
                    x_lo = c
                    c = d
                    fc = fd
                    d = x_lo + invphi * (x_hi - x_lo)
                    fd = f_at_x(d)

            x_opt = (x_lo + x_hi) / 2.0
            new_ai = max(0.0, x_opt)

            if abs(new_ai - old_ai) > 1e-10:
                a[i] = new_ai
                improved = True

        # Renormalize
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose and (sweep % 10 == 0):
                print(f"    coord sweep {sweep}: obj={best_obj:.10f}")

        if not improved:
            break

    return best_a


def penalty_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Penalty-based peak reduction: minimize sum(a_i^2) + mu * max(0, conv - M)^2
    with decreasing M and increasing mu.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Current peak
    conv = compute_conv(a)
    current_max = float(np.max(conv))

    # Target schedule: decrease from current_max
    targets = [current_max * (1 - 0.02 * t) for t in range(1, 21)]
    taus = [5.0 / n, 3.0 / n, 2.0 / n, 1.0 / n]

    for target in targets:
        if time.monotonic() - start > time_budget:
            break
        target_time = time_budget / 20.0

        for tau in taus:
            if time.monotonic() - start > time_budget:
                break
            t_start = time.monotonic()
            t_alloc = min(target_time, time_budget - (time.monotonic() - start))
            if t_alloc < 2.0:
                continue

            x = np.log(a) - np.log(a).max()

            def fun_pen(xv):
                xv = xv - np.max(xv)
                av = np.exp(xv)
                sv = float(np.sum(av))
                cv = compute_conv(av / sv)
                # Smooth penalty
                excess = (cv - target) / tau
                excess = np.maximum(excess, 0)
                penalty = np.sum(excess * excess)
                # Regularization (encourage spread)
                reg = np.sum(av) * 0  # no reg
                return penalty + reg

            def grad_pen(xv):
                xv = xv - np.max(xv)
                av = np.exp(xv)
                sv = float(np.sum(av))
                avn = av / sv
                cv = compute_conv(avn)
                size = next_pow2(2 * n - 1)

                # Gradient of penalty w.r.t. a
                excess = (cv - target) / tau
                w = np.where(excess > 0, 2.0 * excess, 0.0)

                # d(conv)/da[i] = 2*a_rev at appropriate offset
                a_flip = avn[::-1]
                fa_flip = rfft(a_flip, size)
                fw = rfft(w, size)
                conv_w = irfft(fw * fa_flip, size)
                corr = conv_w[n - 1:n - 1 + n]

                # dF/da = 2*corr/s - 2*(corr sum /s^2) * a/s (from chain rule through normalization)
                # For simplicity, work with unnormalized and normalize at the end
                grad_a = 2.0 * corr
                # Adjust for normalization (a = exp(x)/sum(exp(x)))
                grad_x = grad_a * av / sv - np.mean(grad_a * av) / sv * av
                grad_x = grad_x / sv
                return grad_x

            try:
                res = minimize(fun_pen, x, method='L-BFGS-B', jac=grad_pen,
                               options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-10})
                if res.fun < fun_pen(x):
                    x_opt = res.x - res.x.max()
                    a_new = np.exp(x_opt)
                    a_new = np.maximum(a_new, 0)
                    s_new = float(np.sum(a_new))
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                        obj_new = compute_obj(a_new)
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a_new.copy()
                            a = a_new
            except Exception:
                pass

    return best_a


def build_conv_matrix_rows(a, lags):
    """Build matrix rows for specified lags: A[k] @ g = conv(a, g)[k]."""
    n = len(a)
    rows = []
    for k in lags:
        row = np.zeros(n)
        for j in range(n):
            if 0 <= k - j < n:
                row[j] += a[k - j]
        rows.append(row)
    return np.array(rows)


def active_set_lp(a, num_active=50, use_highs=True):
    """
    LP: min M s.t. conv(a,g)[k] <= M for k in active lags, sum(g)=1, g>=0.
    Active lags = those nearest to the current maximum.
    """
    from scipy.optimize import linprog
    n = len(a)
    a_norm = a / np.sum(a)

    conv = compute_conv(a_norm)
    max_val = float(np.max(conv))

    # Get lags sorted by conv value (descending)
    sorted_lags = np.argsort(conv)[::-1]
    active_lags = sorted_lags[:num_active]

    # Build constraint matrix
    A = build_conv_matrix_rows(a_norm, active_lags)

    # LP: min M s.t. A@g <= M, sum(g) = 1, g >= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((len(active_lags), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active_lags))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n]), active_lags
    except Exception:
        pass

    return None, None, None


def lp_step(a, time_budget=10.0):
    """One LP alternating step with active set."""
    n = len(a)
    a_norm = a / np.sum(a)

    g_lp, M_lp, _ = active_set_lp(a_norm, num_active=min(100, 2 * n - 1))
    if g_lp is None:
        return a, compute_obj(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.4, 0.6, 0.8, 0.9]:
        a_try = (1.0 - t) * a_norm + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def lp_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating optimization with active sets."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        a_new, obj_new = lp_step(a)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and int((time.monotonic() - start) * 10) % 20 == 0:
                print(f"    LP: obj={best_obj:.10f}")
        else:
            stall += 1
            if stall > 5:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                scale = 0.02 * min(stall // 5, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def exp_gradient_optimize(a_init, time_budget=30.0, verbose=False):
    """Exponentiated gradient (multiplicative weights) for peak objective."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 2.0 / n

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient at peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        if 0 <= k_star - k_star < n:
            pass  # a[i]^2 term handled separately if k_star = 2*i

        # Actually the subgradient of max(conv) w.r.t. a[i]:
        # d/d(a[i]) conv[k_star] = sum over terms involving a[i]
        # If k_star = 2i: contribution is 2*a[i]
        # If k_star != 2i and 0 <= k_star-i < n: contribution is 2*a[k_star-i]
        # (same as above since we're computing 2*a[j] where j = k_star - i)

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s = float(np.sum(a_new))
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = max(eta * 0.998, 0.05 / n)
        else:
            eta *= 0.85
            if eta < 0.01 / n:
                rng = np.random.default_rng(it + int(time.monotonic()))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                eta = 2.0 / n

    return best_a


def geometric_init(n, r=None, rng=None):
    """Geometric sequence initialization."""
    if r is None:
        r = rng.uniform(0.7, 1.3) if rng is not None else 1.0
    a = r ** np.arange(n, dtype=float)
    a = a + 1e-10
    a = a / np.sum(a)
    return a


def cosine_init(n, rng=None):
    """Cosine-squared window initialization."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2.0) ** 2
    if rng is not None:
        a = a * np.exp(rng.normal(0, 0.05, n))
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    return a


def bimodal_init(n, rng=None):
    """Two Gaussian bumps."""
    c1 = n * 0.25
    c2 = n * 0.75
    width = n * 0.15
    a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * width ** 2)) + \
        np.exp(-((np.arange(n) - c2) ** 2) / (2 * width ** 2))
    if rng is not None:
        a = a * np.exp(rng.normal(0, 0.03, n))
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    return a


def decreasing_init(n, rng=None):
    """Monotonically decreasing (optimal for reducing center lag)."""
    a = np.exp(-np.arange(n, dtype=float) / (n * 0.3))
    if rng is not None:
        a = a * np.exp(rng.normal(0, 0.02, n))
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines coordinate descent, smoothed gradient, penalty method,
    active-set LP, and multi-n exploration with diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)

    # === Phase 1: Quick gradient + coordinate descent on parent ===
    print(f"\n=== Phase 1: Gradient + CoordDescent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_grad = gradient_optimize(best_seq, time_budget=min(remaining * 0.2, 60.0), verbose=True)
        obj = compute_obj(a_grad)
        if obj < best_score:
            best_score = obj
            best_seq = a_grad.copy()
            print(f"  After gradient: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.2, 50.0), verbose=True)
            obj = compute_obj(a_cd)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()
                print(f"  After coord descent: obj={best_score:.10f}")

    # === Phase 2: Active-set LP ===
    print(f"\n=== Phase 2: Active-Set LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_lp = lp_optimize(best_seq, time_budget=min(remaining * 0.15, 60.0), verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  After LP: obj={best_score:.10f}")

    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: Coordinate Descent Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.15, 50.0), verbose=True)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  After CD polish: obj={best_score:.10f}")

    # === Phase 4: Multi-n exploration with diverse inits ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_candidates = sorted(set([
        n_parent - 50, n_parent, n_parent + 50,
        600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 2000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Try different initializations for this n
        inits = [a_try]
        if n_try != len(best_seq):
            inits.append(decreasing_init(n_try, rng))
            inits.append(cosine_init(n_try, rng))
            inits.append(bimodal_init(n_try, rng))
            inits.append(geometric_init(n_try, rng=rng))

        best_for_n = None
        best_obj_for_n = np.inf

        for a_init in inits:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.12, 30.0)

            # Quick gradient
            a_opt = gradient_optimize(a_init, time_budget=alloc)
            obj = compute_obj(a_opt)

            if obj < best_obj_for_n:
                best_obj_for_n = obj
                best_for_n = a_opt.copy()

            # Quick coord descent
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_cd = coordinate_descent(a_opt, time_budget=min(remaining * 0.1, 20.0))
                obj_cd = compute_obj(a_cd)
                if obj_cd < best_obj_for_n:
                    best_obj_for_n = obj_cd
                    best_for_n = a_cd.copy()

        if best_for_n is not None and best_obj_for_n < best_score:
            best_score = best_obj_for_n
            best_seq = best_for_n.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        elif best_for_n is not None:
            print(f"  n={n_try}: best={best_obj_for_n:.10f} (no improve)")
        else:
            print(f"  n={n_try}: SKIPPED")

    # === Phase 5: Intensify on best ===
    print(f"\n=== Phase 5: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Gradient with fine temperature
        alloc = min(remaining * 0.2, 60.0)
        a_g = gradient_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_g)
        if obj < best_score:
            best_score = obj
            best_seq = a_g.copy()
            print(f"  Gradient2: obj={best_score:.10f}")

        # Coord descent
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.15, 40.0), verbose=True)
            obj = compute_obj(a_cd)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()
                print(f"  CD2: obj={best_score:.10f}")

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_lp = lp_optimize(best_seq, time_budget=min(remaining * 0.15, 40.0))
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP2: obj={best_score:.10f}")

        # Coord descent again
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.15, 30.0))
            obj = compute_obj(a_cd)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()
                print(f"  CD3: obj={best_score:.10f}")

    # === Phase 6: Penalty method ===
    print(f"\n=== Phase 6: Penalty Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_pen = penalty_optimize(best_seq, time_budget=min(remaining * 0.2, 60.0), verbose=True)
        obj = compute_obj(a_pen)
        if obj < best_score:
            best_score = obj
            best_seq = a_pen.copy()
            print(f"  Penalty: obj={best_score:.10f}")

        # Coord descent after penalty
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.15, 30.0))
            obj = compute_obj(a_cd)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()
                print(f"  CD4: obj={best_score:.10f}")

    # === Phase 7: Random restarts ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    restart_budget = min(25.0, remaining * 0.06)
    n_restarts = max(3, int(remaining / (restart_budget + 5)))
    n_restarts = min(n_restarts, 8)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < restart_budget + 5:
            break

        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            n_r = max(400, min(1600, n_best + int(rng.integers(-100, 100))))
            a_r = resample_sequence(best_seq, n_r) * np.exp(rng.normal(0, 0.02, n_r))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 2:
            a_r = decreasing_init(n_best, rng)
        elif r == 3:
            a_r = cosine_init(n_best, rng)
        elif r == 4:
            a_r = bimodal_init(n_best, rng)
        else:
            a_r = best_seq + 0.01 * rng.normal(0, 1, n_best)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        # Quick optimization
        a_opt = gradient_optimize(a_r, time_budget=restart_budget * 0.6)
        obj = compute_obj(a_opt)
        if remaining > restart_budget * 0.4:
            a_cd = coordinate_descent(a_opt, time_budget=restart_budget * 0.4)
            obj_cd = compute_obj(a_cd)
            if obj_cd < obj:
                obj = obj_cd
                a_opt = a_cd

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Coord descent (fast and effective for final polish)
        alloc = min(remaining * 0.3, 30.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.10f}")

        # Gradient
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_g = gradient_optimize(best_seq, time_budget=min(remaining * 0.3, 20.0))
            obj = compute_obj(a_g)
            if obj < best_score:
                best_score = obj
                best_seq = a_g.copy()
                print(f"  Final grad: obj={best_score:.10f}")

        # One more coord descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.3, 15.0))
            obj = compute_obj(a_cd)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()
                print(f"  Final CD2: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
