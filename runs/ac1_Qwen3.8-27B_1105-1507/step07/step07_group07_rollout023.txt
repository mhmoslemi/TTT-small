    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def smoothed_max_and_grad(conv, tau):
    """Compute smoothed max of conv and its gradient w.r.t. conv values."""
    c = conv / tau
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    w = exp_c / np.sum(exp_c)
    smooth_max = tau * (c_max + np.log(np.sum(exp_c)))
    return smooth_max, w * tau  # weights for gradient

Let me think about the PROTO algorithm more carefully. The original PROTO (Ozarow & Fielder, 1989) works as follows:

1. Compute `x = IDFT(X)` where `X` is the DFT of the signal.
2. Find `i*` where `|x[i*]|` is maximum.
3. For each frequency `k` where `|X[k]| > 0`, adjust `X[k]` to reduce the contribution to `x[i*]`.
4. Go to 1.

In our case, we're not minimizing the peak of `a` itself, but the peak of `conv(a, a)`. So we need to adapt.

`conv(a, a) = IDFT(|A|^2)` where `A = DFT(a, zero-padded to N ≥ 2n-1)`.

So the "time domain signal" we care about is `r = IDFT(|A|^2)`, and we want to minimize `max(r)`.

The PROTO-like step would be:
1. Compute `r = conv(a, a)`, find `k* = argmax(r)`.
2. In the frequency domain, `r[k*] = (1/N) * sum_m |A[m]|^2 * exp(2*pi*i*m*k*/N)`.

This is the tricky part. In standard PROTO, we just scale the frequency bins and go back. Here, we need to enforce the non-negativity constraint.

def projected_subgradient(a, step_size, n_smooth=1, rng=None):

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def proto_step(a, eta=0.1, N=None):

    PROTO-like spectral step to reduce autocorrelation peak.

def coordinate_swap(a, j1, j2, n_trials=10):
    """
    Try swapping mass between coordinates j1 and j2.
    Returns improved a or original.
    """
    n = len(a)
    best_a = a.copy()
    best_peak = autocorr(a).max()
    
    for delta in np.linspace(-a[j1]*0.5, a[j1]*0.5, n_trials):
        a_try = a.copy()
        a_try[j1] = max(0, a[j1] - delta)
        a_try[j2] = max(0, a[j2] + delta)

        if s < 1e-10:

        a_try /= s
        peak = autocorr(a_try).max()
        if peak < best_peak:
            best_peak = peak
            best_a = a_try.copy()

def smooth_obj_and_grad(a, tau):
    """
    Compute smoothed objective and gradient.
    f(a) = tau * log(sum_k exp(r[k]/tau)) where r = autocorr(a)
    grad_f[j] = sum_k w[k] * dr[k]/da[j]
    """
    n = len(a)
    r = autocorr(a)
    
    # Softmax weights
    r_scaled = r / tau
    r_max = r_scaled.max()
    exp_r = np.exp(r_scaled - r_max)
    w = exp_r / exp_r.sum()
    
    f = tau * (r_max + np.log(exp_r.sum()))
    
    # Gradient
    grad = np.zeros(n)
    for k in range(len(r)):

        j_end = min(n - 1, k)

            if j_start <= j_mid <= j_end:

    return a_pal

def structured_init(n, mode, rng):
    """Generate structured initial sequence of length n."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2*np.pi*x))
        a += 0.01 * rng.random(n)
        return a / a.sum()
    elif mode == 2:
        # Gaussian
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2)
        a += 0.01 * rng.random(n)
        return a / a.sum()
    elif mode == 3:
        # Periodic (period 3)
        pattern = np.array([0.5, 1.0, 0.3])
        a = np.tile(pattern, n // 3 + 1)[:n]
        a *= 1 + 0.1 * rng.random(n)
        return a / a.sum()
    elif mode == 4:
        # Two Gaussians
        c1, c2 = n//3, 2*n//3
        a = np.exp(-((np.arange(n)-c1)**2)/(n*n/30)) + np.exp(-((np.arange(n)-c2)**2)/(n*n/30))
        a += 0.01 * rng.random(n)
        return a / a.sum()
    elif mode == 5:
        # Sinusoidal modulation
        a = 1 + 0.3 * np.cos(2*np.pi*np.arange(n)/n * 3)
        a += 0.05 * rng.random(n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # Bimodal (high at edges, low in middle)
        x = np.linspace(0, 1, n)
        a = (x - 0.5)**2 + 0.05
        a += 0.01 * rng.random(n)
        return a / a.sum()
    elif mode == 7:
        # Sinc-like
        x = np.arange(n) - n/2
        a = np.sinc(x / (n/10))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 8:
        # Periodic (period p for various p)
        p = int(rng.integers(2, 10))
        pattern = rng.uniform(0.5, 1.5, p)
        a = np.tile(pattern, n // p + 1)[:n]
        return a / a.sum()
    else:
        a = rng.random(n)
        a += 0.1
        return a / a.sum()

def full_optimize(a, time_budget=60.0, verbose=False):
    """
    Full local optimization pipeline.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    
    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()
    
    # Round 1: Smoothed gradient with decreasing tau
    taus = [5.0/n, 2.0/n, 1.0/n, 0.5/n, 0.2/n, 0.1/n, 0.05/n]
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        steps = max(20, int(1000 / (2*n-1)))
        a_opt = optimize_smoothed(a, tau, n_steps=steps)
        obj = objective(a_opt)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
        a = a_opt
    
    # Round 2: PROTO-like spectral steps
    for i in range(50):
        if time.monotonic() - start > time_budget:
            break
        a_proto = proto_step(best_a, eta=0.05 + 0.05*(i//10))
        # Follow with gradient steps
        a_follow = optimize_smoothed(a_proto, 1.0/n, n_steps=30)
        obj = objective(a_follow)
        if obj < best_obj:
            best_obj = obj
            best_a = a_follow.copy()
    
    # Round 3: Coordinate swaps
    rng = np.random.default_rng(42)
    for i in range(200):
        if time.monotonic() - start > time_budget:
            break
        j1 = rng.integers(n)
        j2 = rng.integers(n)
        if j1 == j2:
            continue
        a_new = coordinate_swap(best_a, j1, j2, n_trials=8)
        obj = objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a
