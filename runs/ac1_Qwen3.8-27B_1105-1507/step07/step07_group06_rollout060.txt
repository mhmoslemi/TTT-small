def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a, solver_pref=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    if solver_pref is None:
        solver_pref = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    
    for solver in solver_pref:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))

def make_ramp(n, power=1.0, offset=1.0):
    """Create a ramp-like sequence: a_j = (j+1)^power + offset, normalized."""
    j = np.arange(n, dtype=np.float64) + 1
    a = j ** power + offset
    a = a / np.sum(a)
    return a


def make_shape(n, shape_type, rng):
    """Create various shaped initial sequences."""
    if shape_type == 0:
        # Pure linear ramp
        return make_ramp(n, power=1.0, offset=0.0)
    elif shape_type == 1:
        # Power law ramp
        p = rng.uniform(0.5, 2.0)
        return make_ramp(n, power=p, offset=0.0)
    elif shape_type == 2:
        # Exponential ramp
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(rate * np.linspace(0, 1, n))
        return a / np.sum(a)
    elif shape_type == 3:
        # Ramp with flat top
        frac = rng.uniform(0.3, 0.7)
        a = np.zeros(n)
        n_flat = int(n * frac)
        a[:n//2] = np.linspace(0.1, 1.0, n//2)
        a[n//2:n//2+n_flat] = 1.0
        a[n//2+n_flat:] = np.linspace(1.0, 0.1, n - n//2 - n_flat)
        return a / np.sum(a)
    elif shape_type == 4:
        # S-curve (sigmoid)
        center = rng.uniform(0.3, 0.7) * n
        width = rng.uniform(0.05, 0.2) * n
        x = np.arange(n)
        a = 1.0 / (1.0 + np.exp(-(x - center) / width))
        a = a + 0.01  # avoid exact zeros
        return a / np.sum(a)
    elif shape_type == 5:
        # Quadratic ramp
        j = np.arange(n, dtype=np.float64)
        a = j * j + 1
        return a / np.sum(a)
    elif shape_type == 6:
        # Square root ramp
        j = np.arange(n, dtype=np.float64) + 1
        a = np.sqrt(j)
        return a / np.sum(a)
    elif shape_type == 7:
        # Log ramp
        j = np.arange(n, dtype=np.float64) + 1
        a = np.log(j)
        return a / np.sum(a)
    elif shape_type == 8:
        # Triangular (peak in middle) - for comparison
        a = 1.0 - np.abs(np.arange(n) - n/2) / (n/2)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif shape_type == 9:
        # Inverse ramp (decreasing)
        j = np.arange(n, dtype=np.float64)
        a = (n - j) ** 1.0
        return a / np.sum(a)
    else:
        # Random smooth
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2*np.pi)
        a = 1.0 + 0.5 * np.sin(freq * np.linspace(0, 2*np.pi, n) + phase)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
