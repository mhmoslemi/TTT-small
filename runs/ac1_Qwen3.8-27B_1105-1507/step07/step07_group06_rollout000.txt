        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val/s, float(res.x[n])
    except:
        pass
    return None, None

def lp_alternating(a, time_budget=60.0, max_iter=100):
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        best_t = 0
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0-t)*a + t*g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = ((1.0-best_t)*a + best_t*g_lp)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
        else:
            # Stuck - perturb
            rng = np.random.default_rng(hash((it, n)) % 2**32)
            noise = rng.normal(0, 0.1, n)
            a = best_a * np.exp(noise)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
    
    return best_a

def peak_shaving(a, time_budget=30.0):
    """Targeted coordinate descent that specifically reduces the peak bin."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        
        # Products contributing to peak
        products = a[j_start:j_end+1] * a[k_star - j_end:k_star - j_start + 1][::-1]
        
        # Find the index with the largest product
        max_idx_local = int(np.argmax(products))
        i_big = j_start + max_idx_local
        i_small = k_star - i_big
        
        # Try reducing i_big and adding to a position that doesn't contribute to peak
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.2]:
            delta = frac * a[i_big]
            # Find a position to add to that minimizes impact on peak
            # Prefer positions far from the peak contributors
            candidates = []
            for m in range(n):
                if m == i_big or m == i_small:
                    continue
                # Check if m contributes to the peak
                if j_start <= m <= j_end:
                    # m is a contributor, check its partner
                    partner = k_star - m
                    if j_start <= partner <= j_end:
                        continue  # Both m and its partner are contributors, skip
                candidates.append(m)
            
            if not candidates:
                continue
            
            # Pick the candidate that minimizes the new peak
            best_m = candidates[0]
            best_new_obj = np.inf
            for m in candidates[:20]:  # Limit to 20 for speed
                a_test = a.copy()
                a_test[i_big] -= delta
                a_test[m] += delta
                a_test = np.maximum(a_test, 0)
                s_test = np.sum(a_test)
                if s_test < 1e-10:
                    continue
                a_test = a_test / s_test
                obj_test = compute_obj(a_test)
                if obj_test < best_new_obj:
                    best_new_obj = obj_test
                    best_m = m
            
            if best_new_obj < best_obj - 1e-13:
                a[i_big] -= delta
                a[best_m] += delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_obj = best_new_obj
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try random moves
            rng = np.random.default_rng(it * 7 + n)
            for _ in range(30):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.01 * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    continue
                a = a / s
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a = best_a.copy()
            if not improved:
                break
    
    return best_a

def smooth_obj_and_grad(theta, tau, n):
    a = np.exp(theta - np.max(theta))
    a = a / np.sum(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)
    
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)
    
    c = np.zeros(n)
    for j in range(n):
        end = min(2*n-1, j+n)
        if j < 2*n-1:
            c[j] = np.dot(w[j:end], a[:end-j])
    
    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta

def smooth_optimize(a_init, time_budget=60.0):
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    taus = [10, 30, 100, 300]
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        def obj_func(th):
            val, _ = smooth_obj_and_grad(th, tau, n)
            return val
        
        def grad_func(th):
            _, g = smooth_obj_and_grad(th, tau, n)
            return g
        
        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                            options={'maxiter': 100, 'gtol': 1e-5, 'maxfun': 300})
            if result.x is not None:
                a_new = np.exp(result.x - np.max(result.x))
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    theta = result.x.copy()
        except:
            pass
    
    return best_a, best_obj

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
