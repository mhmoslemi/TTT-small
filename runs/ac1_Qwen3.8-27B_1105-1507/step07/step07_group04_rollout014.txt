def peak_subgradient(a, top_k=1):
    """Compute subgradient of max(conv(a,a)) using top_k peaks."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    k_indices = np.argsort(conv)[::-1][:top_k]
    weights = np.ones(len(k_indices)) / len(k_indices)
    
    grad = np.zeros(n)
    for wi, k in zip(weights, k_indices):
        k = int(k)
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                grad[i] += wi * 2.0 * a_norm[ki]
    return grad, conv, k_indices

def lp_solve_for_g(a, n_constraints=None):
    """
    Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0
    where A is the convolution matrix for fixed a.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix efficiently
    # A @ g = conv(a, g), where A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = a[k-j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    # For the LP, we need A@g <= M for all k
    # This is equivalent to: for each k, sum_j a[k-j]*g[j] <= M
    
    # Use scipy linprog
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, n] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    return None, None

def resample_sequence(a, new_n):

    x_old = np.linspace(0, 1, n)

    a_new = np.interp(x_new, x_old, a)
