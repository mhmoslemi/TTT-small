import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_conv_general(a, b_arr):
    """Linear convolution of a and b_arr via FFT."""
    n1, n2 = len(a), len(b_arr)
    N = 1
    while N < n1 + n2 - 1:
        N <<= 1
    fa = rfft(a, N)
    fb = rfft(b_arr, N)
    return irfft(fa * fb, N)[:n1 + n2 - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = compute_conv_fft(a)
    return float(2.0 * n * np.max(b) / (s * s))


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    b = compute_conv_fft(a_norm)
    return float(2.0 * n * np.max(b))


def softmax_weights(b, temperature):
    """Compute softmax weights from convolution values with given temperature."""
    b_max = np.max(b)
    shifted = (b - b_max) / max(temperature, 1e-10)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.ones(len(b)) / len(b)
    return w / w_sum

    w_sum = np.sum(w)
    if w_sum < 1e-30:

    """
    Mirror descent (exponential gradient) with adaptive temperature.
    Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    eta = eta_start
    T = T_start
    momentum_v = np.zeros(n)
    momentum_coeff = 0.5

    stall = 0
    max_stall = 100
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        # Adaptive temperature: start focused (low T), spread out (high T)
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start + (T_end - T_start) * progress
        K = max(1, int(len(a) * topk_transition * progress))

        # Compute convolution and weights
        b = compute_conv_fft(a)
        if progress < 0.3:
            w = softmax_weights(b, T)
        else:
            w = topk_weights(b, K)

        # Compute gradient
        grad = compute_gradient(a, w)
        # Center gradient to preserve sum
        grad = grad - np.mean(grad)

        # Momentum
        momentum_v = momentum_coeff * momentum_v + (1 - momentum_coeff) * grad
        g_eff = momentum_v

        # Backtracking line search
        best_eta = eta
        best_new_obj = best_obj
        for trial in range(8):
            a_try = a * np.exp(-best_eta * g_eff)
            a_try = np.maximum(a_try, 1e-30)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                best_eta *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj - 1e-10:
                best_new_obj = obj_try
                break
            best_eta *= 0.5

        if best_new_obj < best_obj - 1e-12:
            a = a * np.exp(-best_eta * g_eff)
            a = np.maximum(a, 1e-30)
            a = a / np.sum(a)
            best_obj = best_new_obj
            best_a = a.copy()
            stall = 0
            eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            eta *= 0.9
            if stall > max_stall:
                # Perturbation restart
                rng_local = np.random.default_rng(it * 7 + 13)
                noise = np.exp(rng_local.normal(0, 0.1, n))
                a = best_a * noise
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                best_obj = compute_obj_normalized(a)
                momemtum_v = np.zeros(n)
                momentum_v = np.zeros(n)
                eta = eta_start
                stall = 0
                max_stall = min(max_stall * 1.5, 400)
            if eta < 1e-6:
                break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}, T={T:.3f}, K={K}, time={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, max_iters=50):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS solver.
    """
    n = len(a)
    if n > 2000:
        return None  # Too large for LP
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        # Variables: [g_0, ..., g_{n-1}, M]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 10:
                break

    return best_a


def spectral_flatten(a, n_iters=30, time_budget=30.0):
    """
    Gerchberg-Saxton-inspired spectral flattening.
    Iteratively: FFT -> flatten magnitude -> IFFT -> clip non-negative -> normalize.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # FFT
        A = fft(a)
        mag = np.abs(A)

        # Flattening: push magnitude toward geometric mean
        log_mag = np.log(mag + 1e-30)
        target = np.mean(log_mag)
        # Blend toward flat
        alpha = 0.3
        new_log_mag = (1 - alpha) * log_mag + alpha * target
        new_mag = np.exp(new_log_mag)

        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)

        # Inverse FFT
        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def mirror_descent_v2(a, time_budget=60.0, verbose=False):
    """
    Improved mirror descent with:
    - Adaptive top-K peak weighting with smooth transition
    - Backtracking with more aggressive acceptance
    - Periodic "annealing" (increase temperature to escape local minima)
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    eta = 0.05
    T = 0.1
    total_budget = time_budget
    it = 0
    anneal_counter = 0
    best_obj_ever = best_obj

    N_conv = 1
    while N_conv < 3 * n - 1:
        N_conv <<= 1

    while time.monotonic() - start < time_budget:
        it += 1
        elapsed_frac = (time.monotonic() - start) / total_budget

        # Adaptive parameters
        T = 0.05 + 0.5 * elapsed_frac
        K = max(1, int(np.ceil(1 + 20 * elapsed_frac)))

        # Compute convolution
        b = compute_conv_fft(a)
        peak_val = np.max(b)

        # Weight computation: blend softmax and top-K
        w_soft = softmax_weights(b, T)
        w_topk = topk_weights(b, K)
        blend = min(1.0, elapsed_frac * 2)
        w = (1 - blend) * w_soft + blend * w_topk
        w = w / np.sum(w)

        # Gradient
        grad = compute_gradient(a, w)
        grad = grad - np.mean(grad)

        # Mirror descent update with backtracking
        improved = False
        for trial_eta in [eta, eta * 0.5, eta * 0.25, eta * 0.125]:
            log_a = np.log(a + 1e-30) - trial_eta * grad
            # Clip to avoid underflow
            log_a = np.clip(log_a, -50, 50)
            a_try = np.exp(log_a)
            a_try = np.maximum(a_try, 1e-30)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj - 1e-12:
                a = a_try
                best_obj = obj_try
                best_a = a.copy()
                eta = min(eta * 1.03, 1.0)
                improved = True
                break

        if not improved:
            eta *= 0.95
            # Periodic annealing: perturb to escape local minima
            anneal_counter += 1
            if anneal_counter > 50:
                anneal_counter = 0
                rng_an = np.random.default_rng(it)
                noise = np.exp(rng_an.normal(0, 0.05 + 0.1 * elapsed_frac, n))
                a = best_a * noise
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                eta = 0.05

        if eta < 1e-8 and improved:
            # Try with larger perturbations
            rng_an = np.random.default_rng(it + 100)
            noise = np.exp(rng_an.normal(0, 0.2, n))
            a = best_a * noise
            a = np.maximum(a, 1e-30)
            a = a / np.sum(a)
            eta = 0.05

        if verbose and it % 30 == 0:
            print(f"    MD2 it{it}: obj={best_obj:.8f}, eta={eta:.4f}, T={T:.3f}, K={K}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_new = np.linspace(0, 1, new_n)
        a_new = a[0] * (1 - x_new) + a[1] * x_new
        return np.maximum(a_new, 0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped (edge-concentrated)
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine edges (high at edges)
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Power law tail
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    elif mode == 6:
        # Sine-modulated
        x = np.arange(n) / n
        a = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x) + 0.1
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    else:
        # Gaussian mixture
        centers = rng.uniform(0, n, 3).astype(int)
        widths = rng.uniform(n // 10, n // 4, 3)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a


def objective_direct(a):
    """Direct objective computation matching evaluate_sequence exactly."""
    n = len(a)
    if n == 0:
        return np.inf
    a = np.maximum(a, 0)
    a = np.minimum(a, 1000.0)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))
