def spectral_clipping(a, n_iters=100):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    a = a.copy()
    for it in range(n_iters):
        F = fft(a, size)
        # Make |F| flat
        mag = np.abs(F)
        phase = F / np.where(mag > 1e-10, mag, 1)
        # Target: flat magnitude, but DC bin should be sum(a)^2
        target_mag = np.ones(size)
        target_mag[0] = np.sum(a)**2 / size  # DC component
        # Actually, the DC bin of |F|^2 is (sum a)^2, so |F[0]| = sum(a)
        target_mag[0] = np.sum(a)
        F_new = target_mag * phase
        a_new = np.real(ifft(F_new, size)[:n])
        # Project to non-negative
        a = np.maximum(a_new, 0)
        if np.sum(a) > 0:
            a = a / np.sum(a)
    return a

import numpy as np

from scipy.fft import fft, ifft, rfft, irfft


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_from_conv(a, conv):
    """Compute objective given precomputed convolution."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (s * s)

        indices = list(range(n // 2 + 1))
    else:
        indices = list(range(n))

    for i in indices:

        S = total_sum - ai  # sum of all other elements

def spectral_projection(a, n_iters=50, t=0.3, time_budget=10.0):
    """
    Gerchberg-Saxton style spectral projection.
    Alternates between enforcing flat spectrum and non-negativity.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        F = fft(a, N)
        mag = np.abs(F)
        phase = np.zeros_like(F)
        nonzero = mag > 1e-10
        phase[nonzero] = F[nonzero] / mag[nonzero]

        # Target: flat magnitude
        curr_sum = float(np.sum(a))
        target_mag = np.full(N, curr_sum / np.sqrt(N))
        target_mag[0] = curr_sum / np.sqrt(N)  # Actually |F[0]| = sum(a), and we want |F[k]| = sum(a)/sqrt(N) for all k
        # Wait: if |F[k]|^2 is constant = C, then conv[k] = C/N for all k.
        # sum(conv) = sum(a)^2, so N*C/N = C = sum(a)^2, hence |F[k]| = sum(a) for all k.
        # But that gives conv[k] = sum(a)^2/N for all k, and objective = 2n*sum(a)^2/N/sum(a)^2 = 2n/N.
        # For N >= 2n-1, this is <= 2n/(2n-1). Good.
        target_mag = np.full(N, curr_sum)

        # Blend
        mag_new = (1 - t) * mag + t * target_mag
        F_new = mag_new * phase
        a_new = np.real(ifft(F_new, N))[:n]
        a = np.maximum(a_new, 0)

        if np.sum(a) < 1e-10:
            a = best_a.copy()
            break

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def spectral_projection_soft(a, n_iters=30, t_schedule=None, time_budget=10.0):

    n = len(a)

        return a

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        if t_schedule is not None and it < len(t_schedule):
            t = t_schedule[it]
        else:
            t = 0.5 * (0.9 ** it) + 0.1

        F = fft(a, N)
        mag = np.abs(F)
        phase = np.zeros_like(F)
        nonzero = mag > 1e-10
        phase[nonzero] = F[nonzero] / mag[nonzero]

        curr_sum = float(np.sum(a))
        target_mag = np.full(N, curr_sum)

        mag_new = (1 - t) * mag + t * target_mag
        F_new = mag_new * phase
        a_new = np.real(ifft(F_new, N))[:n]
        a = np.maximum(a_new, 0)

        if np.sum(a) < 1e-10:
            a = best_a.copy()
            break

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def pair_swap_improve(a, n_trials=200, time_budget=10.0):
    """Try random pair swaps and keep improvements."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - start > time_budget:
            break

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # Try averaging i and j
        avg = (a[i] + a[j]) / 2
        a_trial = a.copy()
        a_trial[i] = avg
        a_trial[j] = avg

        obj_trial = compute_obj(a_trial)
        if obj_trial < best_obj - 1e-10:
            best_obj = obj_trial
            a = a_trial
        a = a  # revert if no improvement (a wasn't modified since we used a_trial)

    return a


def random_perturb_optimize(a, rng, n_perturbs=10, time_budget=30.0):
    """Apply random perturbations and re-optimize with CD."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - start > time_budget:
            break

        scale = 0.05 + 0.05 * p / max(n_perturbs, 1)
        a_pert = a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 0)
        s = np.sum(a_pert)
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        # Quick CD
        a_opt = coord_descent_multi(a_pert, n_sweeps=20, time_budget=2.0)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj - 1e-10:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def init_uniform(n):
    return np.ones(n) / n


def init_triangular(n):
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 - np.abs(2 * x - 1)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def init_cosine(n):
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** 2
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def init_spectral(n, rng):
    """Initialize via spectral method: random phase, flat magnitude."""
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    phase = np.exp(2j * np.pi * rng.uniform(0, 1, N))
    # Conjugate symmetry for real signal
    phase[N // 2] = 1.0
    if N % 2 == 0:
        phase[:N // 2 + 1] = phase[:N // 2 + 1]
        phase[N // 2 + 1:] = np.conj(phase[1:N // 2][::-1])
    F = phase
    a = np.real(ifft(F, N))[:n]
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-strategy optimization combining
    vectorized coordinate descent, spectral projection, pair swaps,
    random perturbations, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = init_uniform(parent_n)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.8f}")

    # === Phase 1: Diverse Initialization ===
    print(f"\n=== Phase 1: Diverse Initialization ===")
    remaining = deadline - time.monotonic()
    candidates = [best_seq.copy()]

    init_types = ['uniform', 'triangular', 'cosine', 'spectral', 'random']
    n_options_init = [parent_n, max(400, parent_n - 100), min(1500, parent_n + 100),
                      min(2000, parent_n + 200), 500, 1000]

    for n_try in n_options_init:
        if time.monotonic() > deadline - 100:
            break
        for init_type in init_types[:3]:  # Use first 3 for init phase
            if time.monotonic() > deadline - 100:
                break
            if init_type == 'uniform':
                a_init = init_uniform(n_try)
            elif init_type == 'triangular':
                a_init = init_triangular(n_try)
            elif init_type == 'cosine':
                a_init = init_cosine(n_try)
            elif init_type == 'spectral':
                a_init = init_spectral(n_try, rng)
            else:
                a_init = rng.uniform(0, 1, n_try)
                a_init = a_init / np.sum(a_init)

            # Quick CD to evaluate
            a_opt = coord_descent_multi(a_init, n_sweeps=30, time_budget=5.0)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Init {init_type} n={n_try}: obj={best_score:.8f} **")

    print(f"  After Phase 1: best_score={best_score:.8f}, n={len(best_seq)}")

    # === Phase 2: Intensive Coordinate Descent ===
    print(f"\n=== Phase 2: Intensive Coordinate Descent ===")
    n_best = len(best_seq)
    a = best_seq.copy()

    # Long CD phase
    remaining = deadline - time.monotonic()
    cd_budget = min(remaining * 0.4, 500.0)
    a = coord_descent_multi(a, n_sweeps=5000, time_budget=cd_budget, verbose=True)
    obj = compute_obj(a)
    if obj < best_score:
        best_score = obj
        best_seq = a.copy()
        print(f"  After main CD: obj={best_score:.8f}")
    else:
        print(f"  Main CD: obj={obj:.8f} (no improvement)")
        a = best_seq.copy()

    # Symmetric CD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_sym = len(a)
        # Make symmetric
        a_sym = (a + a[::-1]) / 2
        a_sym = np.maximum(a_sym, 0)
        s = np.sum(a_sym)
        if s > 1e-10:
            a_sym = a_sym / s
        a_sym = coord_descent_multi(a_sym, n_sweeps=2000, symmetric=True,
                                    time_budget=min(remaining * 0.2, 100.0))
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  Symmetric CD: obj={best_score:.8f}")

    # === Phase 3: Spectral Projection + CD ===
    print(f"\n=== Phase 3: Spectral Projection ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a = best_seq.copy()
        for round_sp in range(3):
            if time.monotonic() > deadline - 30:
                break
            a_sp = spectral_projection_soft(a, n_iters=40, time_budget=min(remaining * 0.1, 20.0))
            # Polish with CD
            a_polish = coord_descent_multi(a_sp, n_sweeps=200, time_budget=min(remaining * 0.1, 30.0))
            obj_sp = compute_obj(a_polish)
            if obj_sp < best_score:
                best_score = obj_sp
                best_seq = a_polish.copy()
                a = a_polish.copy()
                print(f"  Spectral round {round_sp}: obj={best_score:.8f}")
            else:
                print(f"  Spectral round {round_sp}: obj={obj_sp:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_explore = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100), max(300, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200,
        400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 1800
    ]))
    n_explore = [n for n in n_explore if 300 <= n <= 2500]

    for n_try in n_explore:
        if time.monotonic() > deadline - 50:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # CD optimization
        alloc_cd = min(remaining * 0.1, 30.0)
        a_opt = coord_descent_multi(a_try, n_sweeps=2000, time_budget=alloc_cd)
        obj_try = compute_obj(a_opt)

        # Try spectral + CD
        if time.monotonic() > deadline - 30:
            pass
        elif obj_try < best_score + 0.02:
            a_sp = spectral_projection_soft(a_opt, n_iters=20, time_budget=5.0)
            a_opt2 = coord_descent_multi(a_sp, n_sweeps=200, time_budget=10.0)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        elif obj_try < best_score + 0.01:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Random Perturbations ===
    print(f"\n=== Phase 5: Random Perturbations ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a = best_seq.copy()
        for pert_round in range(5):
            if time.monotonic() > deadline - 30:
                break
            a_pert = random_perturb_optimize(a, rng, n_perturbs=5,
                                             time_budget=min(remaining * 0.15, 30.0))
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a = a_pert.copy()
                print(f"  Perturb round {pert_round}: obj={best_score:.8f}")
            else:
                print(f"  Perturb round {pert_round}: obj={obj_pert:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a = best_seq.copy()

        # Extended CD
        alloc = min(remaining * 0.5, 200.0)
        a_cd = coord_descent_multi(a, n_sweeps=10000, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f}")

        # Another spectral pass
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sp = spectral_projection_soft(a, n_iters=30, time_budget=min(remaining * 0.2, 15.0))
            a_final = coord_descent_multi(a_sp, n_sweeps=1000, time_budget=min(remaining * 0.3, 30.0))
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final Spectral+CD: obj={best_score:.8f}")

        # One more CD on absolute best
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_last = coord_descent_multi(best_seq, n_sweeps=5000, time_budget=min(remaining * 0.8, 60.0))
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Final Final CD: obj={best_score:.8f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
