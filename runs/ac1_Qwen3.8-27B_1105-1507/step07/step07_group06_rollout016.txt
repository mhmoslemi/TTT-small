def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=float)
    x_new = np.arange(new_n, dtype=float) * (n - 1) / (new_n - 1)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_ramp_init(n, direction='left'):
    """Create a ramp initialization."""
    if direction == 'left':
        a = np.linspace(n, 1, n)
    else:
        a = np.linspace(1, n, n)
    return a / np.sum(a)


def make_exp_init(n, rate=1.0):
    """Create an exponential initialization."""
    a = np.exp(-rate * np.arange(n) / n)
    return a / np.sum(a)


def make_gaussian_init(n, center_frac=0.3, width_frac=0.2):
    """Create a Gaussian bump initialization (off-center for asymmetry)."""
    center = center_frac * n
    width = width_frac * n
    a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    return a / np.sum(a)

"""
LP-alternating optimization with fast HiGHS solves, adaptive line search,
subgradient-based escape mechanisms, and multi-length exploration.
Focuses on maximizing the number of effective LP iterations within the time budget.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_fast(a):
    """Compute objective: 2*n*max(conv(a,a)). Assumes sum(a)=1."""
    n = len(a)
    conv = compute_conv_fft(a)
    return 2.0 * n * float(np.max(conv))


def compute_conv_cross(a, b):
    """Compute cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(fixed_vec):
    """
    Solve LP: min M s.t. conv(fixed_vec, x)[k] <= M for all k, sum(x)=1, x>=0.
    Returns (x_opt, M_opt) or (None, None).
    """
    n = len(fixed_vec)
    f = fixed_vec / np.sum(fixed_vec)

    A = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.concatenate([A, -np.ones((2 * n - 1, 1))], axis=1)
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        x_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(x_val))
        if s > 1e-10:
            return x_val / s, float(res.x[n])
    return None, None


def line_search_convex(a, g, n_evals=20):
    """
    Find best t in [0,1] minimizing max(conv((1-t)*a+t*g, (1-t)*a+t*g)).
    Uses golden-section search since the objective is convex in t.
    """
    def f_of_t(t):
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0)
        s = np.sum(x)
        if s < 1e-10:
            return np.inf
        x = x / s
        return compute_obj_fast(x)

    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)

    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)

    t_best = (lo + hi) / 2.0
    f_best = f_of_t(t_best)

    # Also check t=0 and t=1
    f0 = f_of_t(0.0)
    f1 = f_of_t(1.0)
    if f0 < f_best:
        t_best = 0.0
        f_best = f0
    if f1 < f_best:
        t_best = 1.0
        f_best = f1

    if t_best <= 1e-10:
        return a, f_best
    x = (1.0 - t_best) * a + t_best * g
    x = np.maximum(x, 0)
    s = np.sum(x)
    if s > 1e-10:
        x = x / s
    return x, f_best


def lp_alternating(a_init, time_budget, verbose=False, max_stall=40):
    """
    LP alternating optimization with golden-section line search.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()

    stall_count = 0
    it = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget:
        it += 1

        # Solve LP: fix a, find best g
        g, M_g = solve_lp_minpeak(a)
        if g is None:
            break

        # Line search between a and g
        a_new, obj_new = line_search_convex(a, g, n_evals=15)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall_count = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_g:.6f}, t={elapsed:.1f}s")
        else:
            stall_count += 1

        # If stuck, perturb with increasing scale
        if stall_count == max_stall:
            scale = 0.02
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            stall_count = 0
        elif stall_count > max_stall and stall_count % (max_stall // 2) == 0:
            scale = 0.02 + 0.01 * (stall_count // max_stall)
            scale = min(scale, 0.15)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            stall_count = 0

    if verbose:
        print(f"    LP done: {it} iters, best_obj={best_obj:.8f}")
    return best_a


def subgradient_descent(a_init, time_budget, n_steps=500):
    """Subgradient descent on the simplex for minimizing 2n*max(conv)."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()

    step_size = 2.0 / np.sqrt(n)
    decay = 0.9995

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fft(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0
        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        if j_min <= j_max:
            idx = k_star - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] = 2.0 * a[idx]

        # Project gradient onto tangent space of simplex
        grad_proj = grad - np.mean(grad)

        a_new = a - step_size * grad_proj
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj_fast(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()

        step_size *= decay

    return best_a


def smooth_lbfgs_step(a_init, time_budget, tau_start=50.0, tau_end=500.0):
    """
    Smoothed L-BFGS-B with tau annealing. Uses log-softmax parameterization.
    """
    from scipy.optimize import minimize

    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)
    theta = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()

    def smooth_obj(th):
        th_shift = th - np.max(th)
        a_local = np.exp(th_shift)
        a_local = a_local / np.sum(a_local)
        conv = compute_conv_fft(a_local)
        cm = np.max(conv)
        smooth = cm + np.log(np.sum(np.exp((conv - cm) * tau))) / tau
        return 2.0 * n * smooth

    def smooth_grad(th):
        th_shift = th - np.max(th)
        a_local = np.exp(th_shift)
        a_local = a_local / np.sum(a_local)
        conv = compute_conv_fft(a_local)
        cm = np.max(conv)
        w = tau * np.exp((conv - cm) * tau)
        w = w / np.sum(w)

        # c[j] = sum_k w[k] * a[|k-j|] ... no, c[j] = sum_k w[k] * (conv matrix row k dot a)
        # Actually: d(conv[k])/d(a[j]) = 2*a[k-j] (if valid)
        # So d(smooth_max)/d(a[j]) = sum_k w[k] * 2*a[k-j] (if valid)
        # = 2 * (w * a_rotated)[j]
        c = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min <= j_max:
                idx = k - np.arange(j_min, j_max + 1)
                c[j_min:j_max + 1] += wk * a[idx]
        c *= 2.0

        mean_c = np.dot(c, a_local)
        grad_theta = 2.0 * n * a_local * (c - mean_c)
        return grad_theta

    # Anneal tau
    n_stages = 4
    taus = np.geomspace(tau_start, tau_end, n_stages)

    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        stage_budget = min(remaining * 0.85, 60.0)

        try:
            result = minimize(smooth_obj, theta, jac=smooth_grad, method='L-BFGS-B',
                             options={'maxiter': 100, 'gtol': 1e-7, 'maxfun': 300})
            if result.x is not None:
                th_shift = result.x - np.max(result.x)
                a_new = np.exp(th_shift)
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj_fast(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=float)
    x_new = np.arange(new_n, dtype=float) * (n - 1) / max(new_n - 1, 1)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_ramp_init(n, direction='left'):
    """Create a ramp initialization."""
    if direction == 'left':
        a = np.linspace(n, 1, n)
    else:
        a = np.linspace(1, n, n)
    return a / np.sum(a)


def make_exp_init(n, rate=1.0):
    """Create an exponential initialization."""
    a = np.exp(-rate * np.arange(n) / n)
    return a / np.sum(a)


def make_gaussian_init(n, center_frac=0.3, width_frac=0.2):
    """Create a Gaussian bump initialization (off-center for asymmetry)."""
    center = center_frac * n
    width = width_frac * n
    a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    return a / np.sum(a)

            a_opt, obj_opt = smooth_lbfgs_step(a_pert, time_budget=alloc,
                                               tau_start=50.0, tau_end=800.0)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Perturbed: ** obj={best_score:.8f}")
            else:
                print(f"  Perturbed: obj={obj_opt:.8f}")

    # Phase 5: Subgradient + LP combination
    print(f"\n=== Phase 5: Subgradient + LP ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Subgradient to create a perturbation
        a_sub = subgradient_descent(best_seq, time_budget=min(remaining * 0.08, 15.0))
        obj_sub = compute_obj_fast(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Subgrad: ** obj={best_score:.8f}")
        else:
            print(f"  Subgrad: obj={obj_sub:.8f}")

        # LP from subgradient result
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.2, 50.0)
            a_opt = lp_alternating(best_seq, time_budget=alloc)
            obj_opt = compute_obj_fast(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  LP after subgrad: ** obj={best_score:.8f}")
            else:
                print(f"  LP after subgrad: obj={obj_opt:.8f}")

    # Phase 6: Intensive LP with multiple perturbations
    print(f"\n=== Phase 6: Intensive LP (perturbations) ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    for attempt in range(4):
        if remaining < 15:
            break
        scale = 0.02 + 0.02 * attempt
        a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
        a_pert = np.maximum(a_pert, 0)
        s = np.sum(a_pert)
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        alloc = min(remaining * 0.15, 50.0)
        a_opt = lp_alternating(a_pert, time_budget=alloc)
        obj_opt = compute_obj_fast(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Perturb {attempt} (scale={scale:.2f}): ** obj={best_score:.8f}")
        else:
            print(f"  Perturb {attempt} (scale={scale:.2f}): obj={obj_opt:.8f}")
        remaining = deadline - time.monotonic()

    # Phase 7: Final long LP run on best
    print(f"\n=== Phase 7: Final LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = remaining * 0.85
        print(f"  Budget: {alloc:.0f}s")
        a_opt = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_opt = compute_obj_fast(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
        print(f"  Final obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_fast(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
