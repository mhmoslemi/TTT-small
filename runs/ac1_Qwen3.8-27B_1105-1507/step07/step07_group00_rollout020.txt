import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):

    a = a / np.sum(a)

def solve_lp_relaxed(a, K=50):

def subgradient_excess(a, target_M, K=5):
    """
    Compute subgradient weighted by excess over target_M.
    More aggressive than plain top-K softmax.
    """
    n = len(a)
    conv = compute_conv(a)
    excess = np.maximum(conv - target_M, 0)
    total_excess = float(np.sum(excess))

    if total_excess < 1e-15:
        # Fall back to top-K uniform
        top_idx = np.argsort(conv)[::-1][:K]
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        # Get indices where excess is significant
        sig_idx = np.where(excess > 0.01 * target_M)[0]
        if len(sig_idx) == 0:
            sig_idx = np.argsort(conv)[::-1][:K]
        sig_idx = sig_idx[np.argsort(excess[sig_idx])[::-1]][:K]
        weights = excess[sig_idx] / total_excess

    g = np.zeros(n)
    for w, k in zip(weights, sig_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
    return g, conv


def spectral_seed(n, mode=0):
    """Generate structured initial sequences with good spectral properties."""
    x = np.arange(n)
    if mode == 0:
        # Lopsided chirp (asymmetric frequency sweep)
        freq = 0.5 + 0.3 * (x / n)
        a = (np.cos(2 * np.pi * np.cumsum(freq)) + 1) / 2 + 0.01
    elif mode == 1:
        # Asymmetric Gaussian (peak off-center)
        center = n * 0.3
        width = n * 0.2
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2)) + 0.01
    elif mode == 2:
        # Double chirp
        f1 = 0.7 * x / n
        f2 = 0.3 * (1 - x / n)
        a = (np.cos(2 * np.pi * np.cumsum(f1)) + np.cos(2 * np.pi * np.cumsum(f2)) + 2) / 4 + 0.01
    elif mode == 3:
        # Power law (heavy tail)
        a = (x + 1) ** (-0.5)
    elif mode == 4:
        # Anti-symmetric bias: high at one end, low at other
        a = np.linspace(1.0, 0.1, n) + 0.01
    elif mode == 5:
        # Binary-like (thresholded chirp)
        freq = 0.5
        a = (np.sign(np.cos(2 * np.pi * freq * x / n)) + 1) / 2
        a = a * 0.5 + 0.1
    elif mode == 6:
        # Random with spectral shaping
        rng = np.random.default_rng(mode * 1000 + n)
        a = rng.uniform(0, 1, n)
        # Low-pass filter to reduce high-frequency content
        fa = rfft(a)
        cutoff = int(n * 0.3)
        fa[cutoff:] *= 0.3
        a = np.maximum(irfft(fa, n), 0)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.001)
    return a / a.sum()

def subgradient_optimize(a, time_budget=200.0, verbose=False):
    """
    Subgradient method with excess-weighted peaks and adaptive step size.
    Uses Barzilai-Borwein step size rules.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    target_M = float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr = 0.1 / np.sqrt(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Update target: slightly below current best max
        target_M = best_obj / (2 * n) * 0.95

        g, conv = subgradient_excess(a, target_M, K=5)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # BB step size
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-15:
                lr_bb = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(lr_bb, 1.0 / g_norm)

        prev_a = a.copy()
        prev_g = g.copy()

        # Take step with projection to simplex
        a_new = a - lr * g
        # Project to simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try half step
            a_half = a - 0.5 * lr * g
            a_half = np.maximum(a_half, 0)
            s_half = float(np.sum(a_half))
            if s_half > 1e-10:

                obj_half = compute_obj(a_half)
                if obj_half < best_obj - 1e-12:
                    best_obj = obj_half
                    best_a = a_half.copy()
                    a = a_half
                    stall = 0
                    lr *= 0.7
                    continue
            lr *= 0.8
            stall += 1
            if stall > 100:
                # Reset
                a = best_a * np.exp(np.random.default_rng(it).normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                lr = 0.1 / np.sqrt(n)
                stall = 0

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

def coordinate_refine(a, time_budget=30.0):
    """
    Paired coordinate descent: move mass between positions to reduce peak.
    More aggressive than before - tries more target positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False

        # Find contributors to the peak
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)

        # For each contributing pair, try moving mass
        for i in range(i_start, i_end + 1):
            j = k_star - i
            if j < 0 or j >= n:
                continue
            contrib = a[i] * a[j]
            if contrib < 1e-10:
                continue

            # Try moving mass from i (or j) to random non-contributing positions
            for source in [i, j]:
                if a[source] < 1e-10:
                    continue
                # Target: positions far from both i and j
                distances = np.abs(np.arange(n) - source).astype(float)
                distances[source] = np.inf
                top_targets = np.argsort(distances)[::-1][:10]

                for target in top_targets[:5]:
                    for frac in [0.05, 0.1, 0.2]:
                        delta = frac * a[source]
                        if delta > a[source]:
                            continue
                        a[source] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                            improved = True
                            break
                        else:
                            a[source] += delta
                            a[target] -= delta
                            break
            if time.monotonic() - start > time_budget:
                break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def dual_feasibility_search(a_init, time_budget=100.0, verbose=False):
    """
    Binary search on target M using LP feasibility.
    Find minimum M such that there exists g >= 0 with sum(g)=1 and conv(a,g) <= M.
    Then use the found g as a direction to update a.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initial bounds on M
    conv = compute_conv(a)
    M_hi = float(np.max(conv)) * 1.5
    M_lo = float(np.min(conv)) * 0.5
    if M_lo <= 0:
        M_lo = M_hi * 0.1

    for iteration in range(20):
        if time.monotonic() - start > time_budget:
            break

        M_mid = (M_lo + M_hi) / 2

        # Check feasibility: does there exist g with conv(a,g) <= M_mid?
        g_feas = solve_lp_max_sum(a, M_mid)

        if g_feas is not None:
            # Feasible - try to use g_feas as update direction
            a_try = 0.5 * a + 0.5 * g_feas
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
            M_hi = M_mid
        else:
            M_lo = M_mid

        if verbose and iteration % 5 == 0:
            print(f"    Dual it{iteration}: M_lo={M_lo:.8f}, M_hi={M_hi:.8f}, obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, spectral seeding,
    dual feasibility search, subgradient, and coordinate refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_default = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n = len(best_seq)

    # === PHASE 1: Spectral Seeding (find better starting point) ===
    print(f"\n=== Phase 1: Spectral Seeding ===")
    remaining = deadline - time.monotonic()
    seed_budget = min(remaining * 0.08, 80.0)

    if seed_budget > 20:
        per_seed = seed_budget / 7
        for mode in range(7):
            if time.monotonic() > deadline - 30:
                break
            a_seed = spectral_seed(n, mode=mode)
            # Quick optimization: short LP alternating
            sub_budget = min(per_seed, 15.0)
            a_opt = lp_alternating_enhanced(a_seed, time_budget=sub_budget)
            obj_seed = compute_obj(a_opt)
            if obj_seed < best_score:
                best_score = obj_seed
                best_seq = a_opt.copy()
                print(f"  Spectral seed mode={mode}: obj={best_score:.8f} **")
            else:
                print(f"  Spectral seed mode={mode}: obj={obj_seed:.8f}")

    # === PHASE 2: Dual Feasibility Search ===
    print(f"\n=== Phase 2: Dual Feasibility ===")
    remaining = deadline - time.monotonic()
    dual_budget = min(remaining * 0.1, 100.0)

    if dual_budget > 20:
        a_dual = dual_feasibility_search(best_seq, time_budget=dual_budget, verbose=True)
        obj_dual = compute_obj(a_dual)
        if obj_dual < best_score:
            best_score = obj_dual
            best_seq = a_dual.copy()
            print(f"  Dual search: obj={best_score:.8f} **")
        else:
            print(f"  Dual search: obj={obj_dual:.8f}")

    # === PHASE 3: Main LP Alternating (workhorse) ===
    print(f"\n=== Phase 3: Enhanced LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(remaining * 0.4, 500.0)

    if lp_budget > 30:
        # Run multiple rounds with escape
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            alloc = min(lp_budget / (4 - rnd), remaining * 0.4)
            if alloc < 20:
                break
            a_lp = lp_alternating_enhanced(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === PHASE 4: Subgradient Polishing ===
    print(f"\n=== Phase 4: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    sg_budget = min(remaining * 0.15, 150.0)

    if sg_budget > 20:
        a_sg = subgradient_optimize(best_seq, time_budget=sg_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgradient: obj={best_score:.8f} **")
        else:
            print(f"  Subgradient: obj={obj_sg:.8f}")

    # === PHASE 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100, n_current - 50,
        n_current + 50, n_current + 100, n_current + 200,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [x for x in n_options if 400 <= x <= 1400]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization
        alloc = min(remaining * 0.15, 60.0)
        a_opt = lp_alternating_enhanced(a_try, time_budget=alloc)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === PHASE 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r % 7
        a_r = spectral_seed(n_best, mode=mode)

        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.15, 60.0)
        a_opt = lp_alternating_enhanced(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === PHASE 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # LP polish
        lp_alloc = min(remaining * 0.3, 100.0)
        a_lp = lp_alternating_enhanced(best_seq, time_budget=lp_alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Subgradient
        sg_alloc = min(remaining * 0.3, 60.0)
        a_sg = subgradient_optimize(best_seq, time_budget=sg_alloc)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Coordinate descent
        cd_alloc = min(remaining * 0.4, 30.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # One more LP
        a_last = lp_alternating_enhanced(best_seq, time_budget=min(remaining * 0.5, 40.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
