```python
        # d/da[m] = a[k-m] (if valid) + a[m-k] (if valid and m-k != k-m)

            grad[j] += a[other]
            if other != j:
```

This is O(n + K*n) but with efficient numpy operations.

Actually, I realize the formula `d/da[m] of conv(a,a)[k] = 2*a[k-m]` is beautiful. It means the gradient at the peak lag k* is just a "folded" version of a itself!

This gives us a very efficient subgradient step:
1. Find k* = argmax(conv(a,a))
2. grad = 2 * fold(a, k*) (where fold means a[k*-m] with zero padding)
3. Take a step: a_new = a - step * grad
4. Project onto simplex

And with Nesterov acceleration, this should converge quickly.

Let me now write the complete, clean implementation:

```python
"""
Advanced optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated subgradient, LP alternating, and diverse initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_gradient_at_k(a, k):
    """
    Compute gradient of conv(a,a)[k] w.r.t. a.
    Result: grad[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def conv_gradient_topk(a, top_k=5):
    """Compute average gradient from top-k peak lags."""
    n = len(a)
    conv = compute_conv(a)
    idx = np.argsort(conv)[::-1][:top_k]
    grad = np.zeros(n)
    for k in idx:
        grad += conv_gradient_at_k(a, int(k))
    grad /= len(idx)
    return grad, conv


def project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def nesterov_subgradient(a0, time_budget=60, top_k=5, verbose=False):
    """
    Nesterov-accelerated subgradient method with adaptive step size.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a0
    a = a / s

    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step_size = 0.1
    momentum = 0.0
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient at lookahead point
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            grad += conv_gradient_at_k(y, int(k))
        grad /= len(top_idx)
        
        # Step in negative gradient direction
        y_new = y - step_size * grad
        
        # Project onto simplex
        a_new = project_simplex(y_new)
        
        # Nesterov update
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        
        a = a_new
        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive step size
        if it % 200 == 199:
            step_size = max(step_size * 0.97, 0.001)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    Nesterov it={it}: obj={best_obj:.8f}, step={step_size:.4f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        # k-j in [0, n-1] means k in [j, j+n-1]
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, max_time=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    return None, None


def lp_alternating(a0, time_budget=100, verbose=False):
    """
    LP alternating optimization with adaptive line search.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Line search over t
        best_t_obj = best_obj
        best_t = None
        best_a_try = None
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_try = a_try.copy()

        if best_t is not None and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.2f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
```

```python
"""
Advanced optimizer: Nesterov subgradient + LP alternating + peak equalization.
Exploits the exact gradient formula d/da[m] conv(a,a)[k] = 2*a[k-m] for efficient
subgradient steps. Uses diverse initializations, n-sweeping, and aggressive
LP alternating with 1D line search.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """
    Gradient of conv(a,a)[k] w.r.t. a.
    Formula: grad[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.08
    prev_obj = best_obj
    patience = 0

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            grad += conv_grad_at_k(y, int(k))
        grad /= len(top_idx)
        # Center gradient (remove mean for better behavior on simplex)
        grad = grad - np.mean(grad)
        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        if it % 10 == 0 or it < 20:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 200:
                step_size = max(step_size * 0.5, 0.002)
                patience = 0
        if it % 500 == 499:
            step_size = max(step_size * 0.95, 0.003)
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      Nesterov it={it}: best={best_obj:.8f}, step={step_size:.4f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=100.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        # 1D line search: minimize obj((1-t)*a + t*g_lp) over t in [0, 1]
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 15:
                break

    return best_a


def peak_equalize(a0, time_budget=60.0, n_peaks=5, verbose=False):
    """
    Iterative peak equalization: for each top peak, try to reduce it by
    moving mass from its contributors to lower-impact positions.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(50):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:n_peaks]
        improved = False
        for k in top_k:
            k = int(k)
            # Find which a[m] contribute most to conv[k]
            contrib = np.zeros(n)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                m = k - i
                contrib[m] += a[i]
                if m != i:
                    contrib[i] += a[m]
            # Move mass from high contributors to low contributors
            top_contrib = np.argsort(contrib)[::-1][:3]
            low_contrib = np.argsort(contrib)[:3]
            for src in top_contrib:
                for dst in low_contrib:
                    if src == dst or a[src] < 1e-10:
                        continue
                    for frac in [0.05, 0.1, 0.2]:
                        delta = frac * a[src]
                        a[src] -= delta
                        a[dst] += delta
                        a = np.maximum(a, 0)
                        st = float(np.sum(a))
                        if st < 1e-10:
                            a[src] += delta
                            a[dst] -= delta
                            continue
                        a = a / st
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[dst] -= delta
                            a = a / st
                            break
        if not improved and outer > 10:
            break
        a = best_a.copy()
        if verbose:
            print(f"      PEQ iter={outer}: obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Nesterov subgradient + LP alternating + peak equalization.
    Uses exact gradient formula for efficient subgradient steps.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    # === Phase 1: Diverse inits + quick Nesterov (120s) ===
    print(f"\n=== Phase 1: Diverse Initializations ===")
    n_base = len(best_seq)
    inits = [best_seq.copy()]
    inits.append(hann_window(n_base))
    gauss = np.exp(-0.5 * ((np.arange(n_base) - n_base / 2.0) / (n_base / 5.0)) ** 2)
    gauss = np.maximum(gauss, 0)
    inits.append(gauss / np.sum(gauss))
    # Asymmetric
    asym = np.exp(-((np.arange(n_base)) / (n_base * 0.3)) ** 2)
    asym = np.maximum(asym, 0)
    inits.append(asym / np.sum(asym))
    # Random
    inits.append(rng.random(n_base))
    inits[-1] = inits[-1] / np.sum(inits[-1])
    # Triangular
    tri = np.minimum(np.arange(n_base) + 1.0, n_base - np.arange(n_base))
    inits.append(tri / np.sum(tri))

    for i, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  Init {i}: obj={best_score:.8f} **")
        else:
            print(f"  Init {i}: obj={obj:.8f}")
        if remaining > 25:
            alloc = min(15.0, remaining * 0.25)
            a_opt = nesterov_subgrad(init, time_budget=alloc, top_k=5)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Init {i}+Nest: obj={best_score:.8f} **")

    # === Phase 2: n-sweep (180s) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    n_options = [200, 300, 400, 500, 600, 700, 800, 1000, 1200, 1500, 2000]
    n_options = sorted(set(n_options + [n_base, max(100, n_base - 100), n_base + 100]))
    n_options = [n for n in n_options if 100 <= n <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(12.0, remaining * 0.2)
        a_opt = nesterov_subgrad(a_try, time_budget=alloc, top_k=5)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Main LP Alternating (550s) ===
    print(f"\n=== Phase 3: LP Alternating (main) ===")
    a = best_seq.copy()
    n_cur = len(a)
    for round_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(70.0, remaining * 0.3)
        a_new = lp_alternate(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            # Perturb to escape
            a = best_seq * np.exp(rng.normal(0, 0.015, n_cur))
            a = np.maximum(a, 0)
            a /= np.sum(a)

    # === Phase 4: Nesterov polish (200s) ===
    print(f"\n=== Phase 4: Nesterov Polish ===")
    for k_val in [3, 5, 10, 20]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(50.0, remaining * 0.3)
        a_ns = nesterov_subgrad(best_seq, time_budget=alloc, top_k=k_val, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score - 1e-12:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nest k={k_val}: obj={best_score:.8f} **")
        else:
            print(f"  Nest k={k_val}: obj={obj_ns:.8f}")

    # === Phase 5: Peak Equalization (120s) ===
    print(f"\n=== Phase 5: Peak Equalization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.5)
        a_peq = peak_equalize(best_seq, time_budget=alloc, n_peaks=5, verbose=True)
        obj_peq = compute_obj(a_peq)
        if obj_peq < best_score - 1e-12:
            best_score = obj_peq
            best_seq = a_peq.copy()
            print(f"  PEQ: obj={best_score:.8f} **")
        else:
            print(f"  PEQ: obj={obj_peq:.8f}")
        # Follow up with LP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc2 = min(40.0, remaining * 0.5)
            a_lp2 = lp_alternate(best_seq, time_budget=alloc2, verbose=True)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score - 1e-12:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP after PEQ: obj={best_score:.8f} **")

    # === Phase 6: Final LP + Nesterov (100s) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(50.0, remaining * 0.5)
        a_f = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score - 1e-12:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_f:.8f}")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(40.0, remaining * 0.6)
        a_ns2 = nesterov_subgrad(best_seq, time_budget=alloc, top_k=3)
        obj_ns2 = compute_obj(a_ns2)
        if obj_ns2 < best_score - 1e-12:
            best_score = obj_ns2
            best_seq = a_ns2.copy()
            print(f"  Final Nest: obj={best_score:.8f} **")

    # === Phase 7: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(20):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.02 + 0.005 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.3, 15.0)
            a_opt = nesterov_subgrad(a_r, time_budget=alloc * 0.5, top_k=5)
            a_opt = lp_alternate(a_opt, time_budget=alloc * 0.5)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-12:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation and return
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
