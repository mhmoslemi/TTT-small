def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

    """
    Solve LP: min sum_k w_k * conv(a,g)[k] s.t. sum(g)=1, g>=0.
    This gives a direction that balances multiple peaks.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Objective: min sum_k w_k * (A @ g)[k] = min (sum_k w_k * A[k,:]) @ g
    obj_vec = A.T @ peak_weights  # This is the gradient direction
    
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(obj_vec, g))), [cp.sum(g) == 1])

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue

    return None

def newton_smooth_optimize(a, time_budget=100.0, verbose=False):
    """
    Newton's method on the log-sum-exp smoothed objective.
    Uses the Hessian of the smooth approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 5.0 / np.sqrt(n)  # Temperature for smoothing
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        if it % 50 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    Newton it{it}: obj={best_obj:.8f}, T={T:.4f}, elapsed={elapsed:.1f}s")

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Compute gradient and Hessian of smooth objective
        exp_T = np.exp(T * (conv - max_c))
        w = exp_T / exp_T.sum()  # Softmax weights on peaks
        
        # Gradient
        grad = np.zeros(n)
        H = np.zeros((n, n))
        
        for k in range(len(conv)):
            if w[k] < 1e-10:
                continue
            wi = w[k]
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            j_start = k - i_end
            j_end = k - i_start
            
            # Gradient contribution: 2 * wi * a[j] for i in [i_start, i_end]
            grad[i_start:i_end+1] += 2.0 * wi * a[j_start:j_end+1][::-1]
            
            # Hessian: for the smooth part, H[i,j] gets contributions from all k
            # d2(conv_k)/da[i]da[j] = 2*delta(k-i-j) (with appropriate bounds)
            # Plus the softmax Hessian terms
            # This is expensive, so use a diagonal approximation
            pass
        
        # Diagonal Hessian approximation
        # d2 f / da[i]2 ≈ sum_k w[k] * 4*a[k-i]^2 (roughly)
        diag_H = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-10:
                continue
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                diag_H[i_start:i_end+1] += 4.0 * w[k] * a[j_start:j_end+1][::-1]**2
        
        # Add regularization for conditioning
        diag_H += 1e-6 * np.max(diag_H) + 1e-10
        
        # Newton step with diagonal Hessian
        step = grad / diag_H
        
        # Line search
        alpha = 1.0
        for _ in range(15):
            a_try = project_simplex(a - alpha * step)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj + 1e-12:
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                a = a_try.copy()
                stall = 0
                break
            alpha *= 0.5
        
        if stall > 0:
            stall += 1
            if stall > 50:
                # Perturb and restart
                rng = np.random.default_rng(it * 104729 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = project_simplex(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
        
        # Slightly increase T
        T = min(T * 1.005, 20.0 / np.sqrt(n))
        it += 1

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2 + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Anti-correlated: try to make a[i] * a[n-1-i] small
        x = np.linspace(0, 1, n)
        a = np.exp(-5.0 * x) + 0.1
        a = a / a.sum()
    elif mode == 8:
        # Periodic with prime period
        p = 7
        base = rng.uniform(0.5, 1.5, p)
        a = np.tile(base, n // p + 1)[:n]
        a = a / a.sum()
    elif mode == 9:
        # sine-based with random phase
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * x + phase)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_topK(a, K=5, temperature=3.0):
    """Compute subgradient using top-K peaks with softmax weighting."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n)
    weights = np.exp((peak_vals - max_val) * temperature / max_val)
    weights /= weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))

            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_weighted(a, peak_weights):

def solve_lp_second_order(a):
    """
    Solve LP that penalizes both the max peak and the sum of squared peaks.
    min M + lambda * sum_k max(0, conv_k - M/2)^2
    Approximated by: min M + lambda * sum_k conv_k^2 (quadratic)
    Falls back to min-peak LP.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    conv_vals = A @ g
    # Minimize M + small penalty on spread
    penalty = cp.sum_squares(conv_vals - cp.mean(conv_vals))
    lambda_pen = 0.01
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M + lambda_pen * penalty), constraints)
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))

            continue
    return None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 11:
        # Exponential decay from center
        center = n / 2.0
        a = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (n * 0.1)) + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating (multiple strategies) + smooth gradient descent +
    systematic n exploration + diverse restarts + final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # Phase 1: LP Alternating (main workhorse)
    print(f"\n=== Phase 1: LP Alternating (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        lp_budget = min(remaining * 0.35, 500.0)
        a_lp = lp_alternating_advanced(best_seq, time_budget=lp_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP advanced: obj={best_score:.8f} **")
        else:
            print(f"  LP advanced: obj={obj_lp:.8f} (no improvement)")

    # Phase 2: Smooth gradient descent
    print(f"\n=== Phase 2: Smooth Gradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        sgd_budget = min(remaining * 0.15, 150.0)
        a_sgd = smooth_descent_optimize(best_seq, time_budget=sgd_budget, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Smooth GD: obj={best_score:.8f} **")
        else:
            print(f"  Smooth GD: obj={obj_sgd:.8f} (no improvement)")

    # Phase 3: Second LP round (finer)
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        lp2_budget = min(remaining * 0.2, 200.0)
        a_lp2 = lp_alternating_advanced(best_seq, time_budget=lp2_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP round2: obj={best_score:.8f} **")
        else:
            print(f"  LP round2: obj={obj_lp2:.8f} (no improvement)")

    # Phase 4: Multi-n exploration
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 150, n_best + 200,
        600, 700, 900, 1000, 1100
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1400 and n != n_best]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 80.0)
        a_opt = lp_alternating_advanced(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 5: Diverse restarts
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        mode = (r * 3 + 2) % 12
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.1, 60.0)
        a_opt = lp_alternating_advanced(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # Phase 6: Final polish
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 25:
        alloc = min(remaining * 0.3, 120.0)
        a_final = lp_alternating_advanced(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_sgd2 = smooth_descent_optimize(best_seq, time_budget=alloc, verbose=False)
        obj_sgd2 = compute_obj(a_sgd2)
        if obj_sgd2 < best_score:
            best_score = obj_sgd2
            best_seq = a_sgd2.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating_advanced(best_seq, time_budget=min(remaining * 0.4, 30.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Validate
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
