"""
Optimized step function search using constraint-generation LP with mirror descent
hybrid, adaptive line search, and multi-n exploration.
Key innovations: constraint generation (Benders-style) LP for fast direction finding,
exponential tilting (mirror descent) for cheap intermediate steps, golden-section
line search, and diverse n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a, row_indices):
    """Build rows of the convolution matrix A for given row indices.
    A[k, i] = a[k-i] if 0 <= k-i < n, else 0.
    Returns matrix of shape (len(row_indices), n).
    """
    n = len(a)
    rows = []
    for k in row_indices:
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            row[i] = a[k - i]
        rows.append(row)
    return np.array(rows)


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_screened_lp(a, active_indices, max_constraints=60):
    """
    Solve the screened LP: min M s.t. conv(a,g)_k <= M for k in active_indices,
    sum(g) = 1, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Limit number of constraints
    if len(active_indices) > max_constraints:
        # Keep the most important ones (highest conv values)
        conv_a = compute_conv(a)
        sorted_idx = np.argsort(conv_a[active_indices])[::-1][:max_constraints]
        active_indices = active_indices[sorted_idx]
    
    A_screened = build_conv_matrix_rows(a, active_indices)
    m = len(active_indices)
    
    # LP: min M
    # s.t. A_screened @ g - M <= 0  (m constraints)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A_screened
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def line_search(a, g, obj_current, n_evals=20):
    """Golden-section + grid line search for best t in [0, 1]."""
    best_t = 0.0
    best_obj = obj_current
    
    # Coarse grid
    ts = np.linspace(0.0, 1.0, n_evals)
    objs = []
    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            objs.append(np.inf)
            continue
        a_try = a_try / s
        objs.append(compute_obj(a_try))
    
    objs = np.array(objs)
    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    
    # Fine grid around best
    if 0 < best_idx < n_evals - 1:
        lo = ts[max(0, best_idx - 1)]
        hi = ts[min(n_evals - 1, best_idx + 1)]
        ts_fine = np.linspace(lo, hi, 15)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t
    
    return best_t, best_obj


def mirror_descent_step(a, k_star, lr):
    """Exponential tilting step targeting peak at k_star."""
    n = len(a)
    # Gradient of conv(a,a)_{k_star} wrt a is 2*a reversed/shifted
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]
    
    # Exponential tilting: a_new[i] = a[i] * exp(-lr * g[i])
    logits = np.log(np.maximum(a, 1e-15)) - lr * g
    logits -= np.max(logits)  # for numerical stability
    a_new = np.exp(logits)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def multi_peak_mirror_descent(a, lr, T_weight=0.01):
    """Mirror descent with softmax-weighted multi-peak gradient."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    
    # Softmax weights over convolution bins
    shifted = (conv - max_c) / T_weight
    shifted = np.clip(shifted, -100, 100)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Weighted gradient
    g = np.zeros(n)
    for i in range(n):
        g[i] = 0.0
        for k in range(2 * n - 1):
            j = k - i
            if 0 <= j < n:
                g[i] += weights[k] * a[j]
        g[i] *= 2.0
    
    # Exponential tilting
    logits = np.log(np.maximum(a, 1e-15)) - lr * g
    logits -= np.max(logits)
    a_new = np.exp(logits)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = (1 + np.cos(x)) / 2 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Gaussian
        center = n // 2
        sigma = n / 4
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Two peaks
        c1, c2 = n // 4, 3 * n // 4
        sigma = n / 8
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2))
        a = a / np.sum(a)
    elif mode == 5:
        # Power law decay from center
        center = n // 2
        x = np.arange(n)
        a = 1.0 / (1.0 + 0.1 * np.abs(x - center))
        a = a / np.sum(a)
    elif mode == 6:
        # Smoothed random
        a = np.convolve(rng.uniform(0, 1, n), np.ones(10) / 10, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a
