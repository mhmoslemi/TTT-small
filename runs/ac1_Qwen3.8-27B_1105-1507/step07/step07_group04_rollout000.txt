"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating minimization with fast HiGHS solver, mirror descent
with closed-form gradients, frequency-domain shaping, and multi-length search.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_best_response(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s

    # Build constraint matrix
    A = build_conv_matrix(a)

    # LP: min M
    # vars: g_0,...,g_{n-1}, M
    # constraints: A@g - M <= 0  (2n-1 constraints)
    #              sum(g) = 1
    #              g >= 0, M >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@g - M <= 0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception as e:
        pass

    # Fallback to cvxpy
    try:
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=cp.CLARABEL)
        if g_var.value is not None:
            g_val = np.maximum(g_var.value, 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(M_var.value)
    except Exception:
        pass

    return None, None


def lp_alternating_fast(a, time_budget=100.0, max_iters=100, verbose=False):
    """
    Fast LP alternating: find best response g to a, then interpolate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_best_response(a)
        if g_lp is None:
            stall += 5
            if stall > 10:
                break
            # Perturb and retry
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            continue

        # Find optimal interpolation parameter via bisection
        best_t = 0.0
        best_t_obj = best_obj
        lo, hi = 0.0, 1.0
        for _ in range(8):
            mid = (lo + hi) / 2.0
            a_try = (1.0 - mid) * a + mid * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                hi = mid
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = mid
            if obj_try <= best_obj:
                lo = mid
            else:
                hi = mid

        if best_t_obj < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP iter {it}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 8:
                rng = np.random.default_rng(hash((it, n, stall)) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

    return best_a


def mirror_descent_step(a, k_indices, lr):
    """
    Mirror descent step on simplex.
    Gradient of conv(a,a)_k at a is: 2*a_{k-j} for valid j.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in k_indices:
        k = int(k)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                grad[j] += 2.0 * a[kj]
    grad /= len(k_indices)

    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new /= s
    return a_new


def mirror_descent_optimize(a, time_budget=100.0, verbose=False):
    """Mirror descent with adaptive step size, targeting top peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.1
    no_improve = 0
    top_k = min(10, n)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Get top peaks (only look at first n positions due to symmetry)
        k_indices = np.argsort(conv)[::-1][:top_k]

        obj_before = compute_obj(a)
        a_new = mirror_descent_step(a, k_indices, lr)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            if obj_new < obj_before:
                lr = min(lr * 1.05, 1.0)
            else:
                lr = max(lr * 0.95, 0.01)
        else:
            a = best_a.copy()
            no_improve += 1
            lr = max(lr * 0.8, 0.005)
            if no_improve > 50:
                rng = np.random.default_rng(hash((no_improve, n)) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                no_improve = 0
                lr = 0.1

        if verbose and (time.monotonic() - start) % 20 < 1:
            print(f"    MD: obj={best_obj:.10f}, lr={lr:.4f}, t={time.monotonic()-start:.0f}s")

    return best_a


def freq_domain_shaping(a, time_budget=50.0, verbose=False):
    """
    Try to flatten the spectrum in frequency domain while maintaining non-negativity.
    The idea: work on log|FFT(a)| and try to make it flat, then inverse FFT.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1

    while time.monotonic() - start < time_budget:
        # Compute FFT
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a, size)
        magnitude = np.abs(fa)

        # Target: flat magnitude
        target_mag = np.mean(magnitude[magnitude > 1e-10])
        if target_mag < 1e-10:
            break

        # Gradient in freq domain: want to reduce peaks in |FFT|
        # Perturb phase to reduce magnitude peaks
        phase = np.angle(fa)
        log_mag = np.log(np.maximum(magnitude, 1e-15))
        log_mag_target = np.mean(log_mag)

        # Adjust: reduce high-magnitude bins
        adjustment = lr * (log_mag - log_mag_target)
        new_mag = magnitude * np.exp(-adjustment)
        new_fa = new_mag * np.exp(1j * phase)

        # Inverse FFT
        a_new = irfft(new_fa, size)[:n]
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            break
        a_new /= sn

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            lr *= 0.9
            if lr < 0.01:
                break

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent: swap small amounts between pairs to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from high-gradient positions to low-gradient positions
        grads = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                grads[j] = 2.0 * a[kj]

        # Sort by gradient: want to decrease high-grad, increase low-grad
        high_idx = np.argsort(grads)[::-1][:n // 4]
        low_idx = np.argsort(grads)[:n // 4]

        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break
            i = high_idx[rng.integers(0, len(high_idx))]
            j = low_idx[rng.integers(0, len(low_idx))]
            if i == j:
                continue
            delta = 0.01 * a[i]
            if delta <= 0:
                continue
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0)
            sn = float(np.sum(a))
            if sn > 1e-10:
                a /= sn
            new_obj = compute_obj(a)
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                improved = True
            else:
                a[i] += delta
                a[j] -= delta

        if not improved:
            # Random pair swap
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.005 * max(a[i], 1e-6)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                sn = float(np.sum(a))
                if sn > 1e-10:
                    a /= sn
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta

        if not improved and time.monotonic() - start > time_budget * 0.5:
            break

    return best_a


def structured_init(n, style='cosine'):
    """Generate structured initial sequences."""
    x = np.arange(n)
    if style == 'cosine':
        a = np.cos(np.pi * x / (2 * n)) ** 2
    elif style == 'gaussian':
        center = n / 2.0
        sigma = n / 4.0
        a = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
    elif style == 'uniform':
        a = np.ones(n)
    elif style == 'triangle':
        center = n / 2.0
        a = 1.0 - np.abs(x - center) / (n / 2.0)
    elif style == 'quadratic':
        center = n / 2.0
        a = 1.0 - (x - center) ** 2 / (n / 2.0) ** 2
    elif style == 'wigner':
        # Wigner semicircle
        center = (n - 1) / 2.0
        a = np.sqrt(np.maximum(1.0 - ((x - center) / (n / 2.0)) ** 2, 0))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a /= s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimization combining LP alternating,
    mirror descent, frequency shaping, coordinate polish, and multi-length search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    n_parent = len(best_seq) if best_seq is not None else 800

    # === Phase 1: Structured initialization exploration (60s) ===
    print("\n=== Phase 1: Structured Initialization ===")
    remaining = deadline - time.monotonic()
    styles = ['cosine', 'gaussian', 'triangle', 'quadratic', 'wigner', 'uniform']

    for style in styles:
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        a_struct = structured_init(n_parent, style)
        obj_struct = compute_obj(a_struct)
        print(f"  {style} (n={n_parent}): obj={obj_struct:.10f}")
        if obj_struct < best_score:
            best_score = obj_struct
            best_seq = a_struct.copy()
            print(f"    ** NEW BEST: {best_score:.10f}")

    # Quick mirror descent on best structured init
    if remaining > 30:
        a_md = mirror_descent_optimize(best_seq, time_budget=min(30, remaining * 0.3))
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD on best init: obj={best_score:.10f} **")

    # === Phase 2: LP Alternating on best sequence (400s) ===
    print(f"\n=== Phase 2: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(400.0, remaining * 0.35)

    # Run LP alternating in chunks with perturbations between
    a_current = best_seq.copy()
    for chunk in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(lp_budget / (3 - chunk), remaining * 0.45)
        a_lp = lp_alternating_fast(a_current, time_budget=alloc, max_iters=200, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP chunk {chunk}: obj={best_score:.10f} **")
        else:
            print(f"  LP chunk {chunk}: obj={obj_lp:.10f}")

        # Perturb for next chunk
        a_current = best_seq * np.exp(rng.normal(0, 0.01, len(best_seq)))
        a_current = np.maximum(a_current, 0)
        a_current = a_current / np.sum(a_current)

    # === Phase 3: Multi-length exploration (300s) ===
    print(f"\n=== Phase 3: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()

    n_options = []
    # Try lengths around the parent and some others
    for delta in [-200, -100, -50, 0, 50, 100, 200, 300, 400]:
        n_try = n_parent + delta
        if 100 <= n_try <= 3000:
            n_options.append(n_try)
    # Add some absolute values
    for n_abs in [200, 400, 600, 800, 1000, 1200, 1500, 2000]:
        if 100 <= n_abs <= 3000 and n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: mirror descent + short LP
        alloc_md = min(15.0, remaining * 0.15)
        alloc_lp = min(30.0, remaining * 0.25)

        a_opt = mirror_descent_optimize(a_try, time_budget=alloc_md)
        a_opt = lp_alternating_fast(a_opt, time_budget=alloc_lp, max_iters=50)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 4: Refined LP on best (200s) ===
    print(f"\n=== Phase 4: Refined LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(200.0, remaining * 0.4)
        a_lp2 = lp_alternating_fast(best_seq, time_budget=alloc, max_iters=300, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Refined LP: obj={best_score:.10f} **")
        else:
            print(f"  Refined LP: obj={obj_lp2:.10f}")

    # === Phase 5: Frequency domain shaping (60s) ===
    print(f"\n=== Phase 5: Frequency Shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.15)
        a_freq = freq_domain_shaping(best_seq, time_budget=alloc)
        obj_freq = compute_obj(a_freq)
        if obj_freq < best_score:
            best_score = obj_freq
            best_seq = a_freq.copy()
            print(f"  Freq: obj={best_score:.10f} **")
        else:
            print(f"  Freq: obj={obj_freq:.10f}")

    # === Phase 6: Coordinate polish + final LP (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        alloc_cd = min(40.0, remaining * 0.25)
        a_cd = coordinate_polish(best_seq, time_budget=alloc_cd)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.10f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_final = min(80.0, remaining * 0.5)
        a_final = lp_alternating_fast(best_seq, time_budget=alloc_final, max_iters=200, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # === Phase 7: Random restarts with best structure ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            scale = 0.02 + 0.015 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            s = float(np.sum(a_r))
            if s < 1e-10:
                continue
            a_r = a_r / s

            alloc = min(remaining * 0.3, 25.0)
            a_opt = mirror_descent_optimize(a_r, time_budget=alloc * 0.4)
            a_opt = lp_alternating_fast(a_opt, time_budget=alloc * 0.6, max_iters=30)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.10f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
