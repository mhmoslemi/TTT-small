"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated subgradient with multi-peak weighting,
FFT flat-spectrum initialization, spectral perturbations, and adaptive step size.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def nesterov_subgradient(a, time_budget=300.0, n_top=5, restart_every=200):
    """
    Nesterov-accelerated subgradient method with multi-peak weighting.
    Uses adaptive step size and periodic restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Polyak step size parameters
    # Lower bound on optimum: 2n/(2n-1) ≈ 1 + 1/(2n-1)
    obj_lb = 2.0 * n / (2.0 * n - 1.0)
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum coefficient
    alpha = 0.1  # step size
    best_alpha = 0.1
    patience = 0

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        conv_y = compute_conv(y)
        
        # Multi-peak subgradient
        top_idx = np.argpartition(conv_y, -n_top)[-n_top:]
        top_val = conv_y[top_idx]
        max_val = top_val.max()
        
        # Adaptive temperature: start wide, narrow over time
        progress = min(1.0, it / 5000.0)
        temp = max(0.01, 0.5 * (1.0 - progress))
        weights = np.exp((top_val - max_val) / max(temp, 1e-6))
        weights /= weights.sum()

        # Compute subgradient: grad[i] = 2 * sum_k w_k * a[k-i] for valid k
        # This is a weighted cross-correlation
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_idx):
            ki = int(k)
            # grad[i] += 2*w_k*y[ki-i] for 0 <= ki-i < n
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            i_start = ki - j_end
            i_end = ki - j_start
            grad[i_start:i_end+1] += 2.0 * w_k * y[j_start:j_end+1][::-1]

        # Nesterov update
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-10:
            break
        
        # Polyak-like step size
        # alpha = (f(y) - f_lb) / ||grad||^2
        f_y = 2.0 * n * max_val  # (since sum(y)=1)
        alpha = min(0.5, max(0.001, (f_y - obj_lb) / (norm_grad ** 2 + 1e-10)))
        
        v = beta * v + (1.0 - beta) * (grad / norm_grad)
        y_new = y - alpha * v
        
        # Project to simplex {a >= 0, sum(a) = 1}
        y_new = project_simplex(y_new)
        
        if compute_obj(y_new) < best_obj - 1e-12:
            best_obj = compute_obj(y_new)  # redundant but safe
            best_a = y_new.copy()
            patience = 0
        else:
            patience += 1
        
        y = y_new
        
        # Periodic restart
        if patience > restart_every:
            # Add noise and restart
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            noise = rng.normal(0, 0.05, n)
            y = best_a * np.exp(noise)
            y = project_simplex(y)
            v = np.zeros(n)
            patience = 0
    
    return best_a


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the algorithm from Duchi et al. (2008)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def fft_spectral_init(n, mode='flat'):
    """Generate initial sequence using spectral methods."""
    if mode == 'flat':
        # Flat amplitude spectrum with random phases
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        rng = np.random.default_rng(42)
        phases = np.exp(1j * rng.uniform(0, 2*np.pi, size))
        # Make it Hermitian for real output
        phases[size//2+1:] = np.conj(phases[1:size//2][::-1]) if size > 2 else phases
        a = ifft(phases).real[:n]
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    elif mode == 'smooth':
        # Smoothly varying amplitude
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(2*np.pi*2*x) + 0.2 * np.cos(2*np.pi*5*x + 0.5)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 'bimodal':
        c1, c2 = n//4, 3*n//4
        w = n//8
        a = np.exp(-((np.arange(n)-c1)**2)/(2*w**2)) + np.exp(-((np.arange(n)-c2)**2)/(2*w**2))
        a = a / np.sum(a)
    else:  # 'raised_cosine'
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = a / np.sum(a)
    
    return a


def spectral_perturb(a, n_freqs=10, amplitude=0.1, time_budget=30.0):
    """Perturb the sequence in the frequency domain."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        # Randomly perturb some frequency bins
        idx = rng.integers(0, size//2 + 1, min(n_freqs, size//2 + 1))
        for i in idx:
            mag = np.abs(A[i])
            phase = np.angle(A[i])
            new_mag = mag * np.exp(rng.normal(0, amplitude))
            new_phase = phase + rng.normal(0, 0.1)
            A[i] = new_mag * np.exp(1j * new_phase)
            # Maintain Hermitian symmetry
            if i != 0 and i != size//2:
                A[size-i] = np.conj(A[i])
        
        a_new = ifft(A).real[:n]
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
    return best_a


def coordinate_refine(a, time_budget=30.0):
    """Refine by trying small perturbations of pairs of elements."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        # Try moving mass from one element to another
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        for frac in [0.01, 0.03, 0.05, 0.1]:
            delta = frac * a[i]
            a[i] -= delta
            a[j] += delta
            if a[i] < 0:
                a[i] = 0
            new_obj = compute_obj(a)
            if new_obj < best_obj - 1e-13:
                best_obj = new_obj
                best_a = a.copy()
            else:
                a[i] += delta
                a[j] -= delta
                break
    
    return best_a

"""
Optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated
subgradient with multi-peak weighting, adaptive step size, log-sum-exp smoothing,
FFT-based gradient computation, and multi-n exploration.

import numpy as np
from scipy.fft import fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def multi_peak_grad(y, conv_y, n_top, temperature):
    """Compute weighted multi-peak subgradient using FFT cross-correlation."""
    n = len(y)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    top_idx = np.argpartition(conv_y, -n_top)[-n_top:]
    top_val = conv_y[top_idx]
    max_val = top_val.max()

    weights = np.exp((top_val - max_val) / max(temperature, 1e-8))
    weights /= weights.sum()

    # Build weight vector in time domain (length 2n-1)
    w_vec = np.zeros(2 * n - 1)
    for w_k, k in zip(weights, top_idx):
        w_vec[int(k)] = w_k

    # grad[i] = 2 * sum_k w[k] * y[k-i] for valid k-i
    # = 2 * cross_correlation(w_vec, y)[i]
    # Using FFT: cross_corr = ifft(fft(w_vec, N) * conj(fft(y, N)))
    W = fft(w_vec, size)
    Y = fft(y, size)
    cross = ifft(W * np.conj(Y)).real[:n]

    return 2.0 * cross, top_idx, weights


def nesterov_optimize(a, time_budget=300.0, n_top=5, beta_mom=0.85,
                      smooth_beta_start=5.0, smooth_beta_end=50.0,
                      restart_patience=400, verbose=True):
    """Nesterov-accelerated subgradient with adaptive smoothing and step size."""
    n = len(a)
    a = np.maximum(a, 0.0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    alpha = 0.15
    best_alpha = 0.15
    no_improve = 0
    total_iters = 50000

    for it in range(total_iters):
        t_rem = time_budget - (time.monotonic() - start)
        if t_rem < 0.5:
            break

        conv_y = compute_conv(y)
        max_val = float(np.max(conv_y))
        f_curr = 2.0 * n * max_val

        # Adaptive smoothing: use log-sum-exp for smoother gradient
        progress = min(1.0, it / (total_iters * 0.7))
        smooth_beta = smooth_beta_start + (smooth_beta_end - smooth_beta_start) * progress

        # Temperature for multi-peak weighting
        temp = max(0.05, 0.3 * (1.0 - 0.5 * progress))

        # Compute gradient
        grad, top_idx, weights = multi_peak_grad(y, conv_y, n_top, temp)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Adaptive step size
        # Scale by 1/sqrt(it) for convergence, with cap
        alpha = min(1.0, best_alpha / max(1.0, np.sqrt(1.0 + it / 500.0)))
        alpha = max(0.001, alpha)

        # Nesterov momentum update
        v = beta_mom * v + (1.0 - beta_mom) * (grad / grad_norm)
        y_new = y - alpha * v

        # Project to simplex
        y_new = project_simplex(y_new)

        # Evaluate
        obj_new = 2.0 * n * float(np.max(compute_conv(y_new)))

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = y_new.copy()
            no_improve = 0
            # Increase alpha slightly on improvement
            best_alpha = min(0.5, best_alpha * 1.02)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                best_alpha *= 0.9

        y = y_new

        # Periodic restart with noise
        if no_improve >= restart_patience:

            noise_scale = 0.03 + 0.02 * min(no_improve // restart_patience, 5)
            y = best_a * np.exp(rng.normal(0, noise_scale, n))

            v = np.zeros(n)
            no_improve = 0
            best_alpha = 0.15

            print(f"    it={it}, obj={best_obj:.8f}, alpha={alpha:.4f}, "

    return best_a


def logsumexp_smooth_objective(a, beta, time_budget=200.0):
    """Optimize using log-sum-exp smoothed objective with Adam."""
    n = len(a)
    a = np.maximum(a, 0.0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam state
    lr = 0.005
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    total_iters = 30000
    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break

        # Anneal beta for log-sum-exp
        progress = it / total_iters
        cur_beta = beta + (100.0 - beta) * progress

        conv_a = compute_conv(a)
        # Smooth max via log-sum-exp
        cmax = np.max(conv_a)
        e = np.exp(cur_beta * (conv_a - cmax))
        w = e / np.sum(e)

        # Gradient via cross-correlation (same as multi-peak but with all weights)
        size2 = 1
        while size2 < 3 * n - 2:
            size2 <<= 1
        W = fft(w, size2)
        Y = fft(a, size2)
        cross = ifft(W * np.conj(Y)).real[:n]
        grad = 2.0 * cross

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad ** 2
        m_hat = m / (1.0 - b1 ** t_adam)
        v_hat = v / (1.0 - b2 ** t_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new

        # Decay learning rate
        if it % 5000 == 4999:
            lr *= 0.7

    return best_a


def spectral_refine(a, time_budget=60.0):
    """Refine in frequency domain by adjusting amplitude spectrum."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)

        # Perturb a random subset of frequency bins
        n_perturb = max(3, n // 20)
        idx = rng.choice(size // 2 + 1, size=min(n_perturb, size // 2 + 1), replace=False)
        for i in idx:
            mag = np.abs(A[i])
            phase = np.angle(A[i])
            new_mag = max(0, mag * np.exp(rng.normal(0, 0.08)))
            new_phase = phase + rng.normal(0, 0.05)
            A[i] = new_mag * np.exp(1j * new_phase)
            if 0 < i < size // 2:
                A[size - i] = np.conj(A[i])
            elif i == 0 or i == size // 2:
                A[i] = np.abs(A[i])  # Keep real for DC and Nyquist

        a_new = ifft(A).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-11:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new

    return best_a

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving small fraction of mass from i to j
        fracs = [0.02, 0.05, 0.1]
        improved = False
        for frac in fracs:
            delta = frac * a[i]
            if delta < 1e-12:
                continue
            a[i] -= delta
            a[j] += delta
            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
                improved = True
            else:
                a[i] += delta
                a[j] -= delta
                break

        if not improved and it % 2000 == 1999:
            # Try reverse direction
            delta = 0.05 * a[j]
            if delta > 1e-12:
                a[i] += delta
                a[j] -= delta
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 0:
                    a = a / s2
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a.copy()

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

    + pairwise exchange + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0.0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Nesterov (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov Subgradient (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.22, 250.0)
        a_opt = nesterov_optimize(best_seq, time_budget=alloc, n_top=5,
                                  beta_mom=0.85, smooth_beta_start=5.0,
                                  smooth_beta_end=50.0, restart_patience=300,
                                  verbose=True)
        obj = compute_obj(a_opt)
        print(f"  Phase1 Nesterov: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    print(f"\n=== Phase 2: LogSumExp Adam (n={len(best_seq)}) ===")

        alloc = min(remaining * 0.15, 150.0)
        a_opt = logsumexp_smooth_objective(best_seq, beta=10.0, time_budget=alloc)
        obj = compute_obj(a_opt)
        print(f"  Phase2 LSE-Adam: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 3: Nesterov with different params ===
    print(f"\n=== Phase 3: Nesterov (n_top=3, higher momentum) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 200.0)
        a_opt = nesterov_optimize(best_seq, time_budget=alloc, n_top=3,
                                  beta_mom=0.9, smooth_beta_start=10.0,
                                  smooth_beta_end=80.0, restart_patience=200,
                                  verbose=True)
        obj = compute_obj(a_opt)
        print(f"  Phase3 Nesterov: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 4: Spectral refinement ===
    print(f"\n=== Phase 4: Spectral Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_opt = spectral_refine(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        print(f"  Phase4 Spectral: obj={obj:.8f}")
        if obj < best_score:

            best_seq = a_opt.copy()

    # === Phase 5: Pairwise exchange ===
    print(f"\n=== Phase 5: Pairwise Exchange ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 40.0)
        a_opt = pairwise_exchange(best_seq, time_budget=alloc)

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, n_best + 300,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [x for x in n_options if 300 <= x <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if abs(n_try - n_best) < 30:
            continue

        a_try = resample_seq(best_seq, n_try)

    rng = np.random.default_rng(seed)

            n_r = max(400, min(1500, n_r))

    # === Phase 8: Final intensive polish ===

    remaining = deadline - time.monotonic()

    if remaining > 60:

    if remaining > 30:

        obj = compute_obj(a_opt)
        print(f"  Final LSE-Adam: obj={obj:.8f}")
        if obj < best_score:

            best_seq = a_opt.copy()

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Pairwise
        alloc = min(remaining * 0.3, 40.0)

    if remaining > 20:

    return result
