Hybrid LP-alternating optimization with constraint generation, 
peak-shaving coordinate descent, and multi-n exploration.
Focuses on fast LP solves using HiGHS and adaptive step sizes.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def convolve_a_with_b(a, b):
    """Compute convolution of a and b (not necessarily the same)."""
    n_a, n_b = len(a), len(b)
    size = 1
    while size < n_a + n_b - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n_a + n_b - 1]


def solve_lp_fast(a, active_indices=None, max_constraints=50):
    """
    Solve LP: min t s.t. (a*g)_k <= t for active k, sum(g)=1, g>=0.
    Uses constraint generation if active_indices is None.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Start with all constraints if n is small, else use top-K
    if active_indices is None:
        # Compute convolution with current best g (or use all)
        # For the first solve, use all constraints
        conv = compute_conv(a)
        k = np.argsort(conv)[::-1][:max_constraints]
        active_indices = sorted(k.tolist())
    
    num_constraints = len(active_indices)
    
    # Build constraint matrix
    # (a*g)_k = sum_j a_{k-j} * g_j for valid j
    # A_ub[k_idx, j] = a_{k_idx - j} if 0 <= k_idx - j < n
    A_ub = np.zeros((num_constraints, n + 1))
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A_ub[idx, j_min:j_max+1] = a[j_min:j_max+1][::-1]
        A_ub[idx, n] = -1.0  # coefficient of t
    
    b_ub = np.zeros(num_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x[:n]
            g = np.maximum(g, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                # Check if solution satisfies all constraints
                conv_check = convolve_a_with_b(a, g)
                violated = np.where(conv_check > res.x[n] + 1e-8)[0]
                if len(violated) > 0:
                    # Add violated constraints and re-solve
                    new_active = list(set(active_indices + violated.tolist()))
                    new_active = sorted(new_active)
                    return solve_lp_fast(a, active_indices=new_active, 
                                         max_constraints=max_constraints)
                return g, float(res.x[n])
    except Exception:
        pass
    
    # Fallback: solve with all constraints
    try:
        all_indices = list(range(2 * n - 1))
        A_ub_full = np.zeros((2 * n - 1, n + 1))
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                A_ub_full[k, j_min:j_max+1] = a[j_min:j_max+1][::-1]
            A_ub_full[k, n] = -1.0
        
        res = linprog(c, A_ub=A_ub_full, b_ub=np.zeros(2*n-1),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternating(a, time_budget=60.0, max_iter=200, verbose=False):
    """LP alternating optimization with backtracking line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve_count = 0
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_fast(a, max_constraints=30)
        if g_lp is None:
            break
        
        # Backtracking line search
        t = 1.0
        improved = False
        for _ in range(10):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                t *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                no_improve_count = 0
                break
            t *= 0.5
        
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 5:
                # Perturb to escape
                rng = np.random.default_rng(it * 12345 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                no_improve_count = 0
            if no_improve_count >= 20:
                break
        
        if verbose and it % 20 == 0:
            print(f"    LP alt it={it}: obj={best_obj:.8f}")
    
    return best_a

        contributors = []

def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp (asymmetric)
    ramp = np.linspace(1, n, n)
    inits.append(ramp / ramp.sum())
    
    # Reverse ramp
    inits.append(ramp[::-1] / ramp.sum())
    
    # Exponential
    exp_seq = np.exp(np.linspace(0, 2, n))
    inits.append(exp_seq / exp_seq.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    inits.append(rc / rc.sum())
    
    # Gaussian
    x = np.linspace(-2, 2, n)
    gauss = np.exp(-x**2)
    inits.append(gauss / gauss.sum())
    
    # Bimodal (two peaks)
    bimodal = np.zeros(n)
    bimodal[:n//3] = 1
    bimodal[2*n//3:] = 1
    inits.append(bimodal / bimodal.sum())
    
    # Random
    rand = rng.uniform(0.1, 1.0, n)
    inits.append(rand / rand.sum())
    
    # Sinusoidal modulation
    sin_mod = 1 + 0.5 * np.sin(2 * np.pi * np.arange(n) / (n * rng.uniform(0.1, 0.5)))
    inits.append(sin_mod / sin_mod.sum())
    
    return inits
