def resample_seq(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        progress = step / max(n_steps - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        eta = 3.0 / (np.sqrt(step + 1.0) * np.sqrt(n) * 0.5)
        
        obj, grad = objective_and_grad_direct(y, tau)
        
        a_new = mirror_descent_step(y, grad, eta)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y_new = ((t_new - 1.0) / t_new) * a_new + (t_mom / t_new) * (y - a)
        y_new = np.maximum(y_new, 0)
        s = np.sum(y_new)
        if s > 1e-10:
            y_new = y_new / s
        y = y_new
        t_mom = t_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            if it % 10 == 0 and it > 0:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
    
    return best_a


def peak_chasing_descent(a, time_budget, tau=20.0, n_steps=400):
    """
    Gradient descent focusing specifically on reducing the peak.
    Uses subgradient of hard max (plus small smoothing) with adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.3
    patience = 0
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv_fft(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max at k_star: d(conv_{k_star})/da_i
        # conv_{k_star} = sum_j a[j] * a[k_star - j] (for valid j)
        # d/da_i = a[k_star - i] + a[i - k_star] (if valid)
        # Actually for a*a: d(conv_k)/da_i = a[k-i] + a[k-i] if both in range... no
        # conv_k = sum_j a[j]*a[k-j], d(conv_k)/da_i = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n
        # Wait, that's only if i appears in exactly one term. Let me be precise:
        # conv_k = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d(conv_k)/da_i: a_i appears as a[j] when j=i (contributes a[k-i] if valid)
        #                  and as a[k-j] when k-j=i, i.e., j=k-i (contributes a[k-i] if valid)
        # So d(conv_k)/da_i = a[k-i] (if 0<=k-i<n and 0<=i<n) + a[k-i] (if 0<=k-i<n and 0<=k-i<n)
        # = 2*a[k-i] if 0 <= k-i < n, else 0 (assuming 0 <= i < n always)
        # Wait, if i = k-i (i.e., k=2i), then it's just 2*a[i] (same term counted twice in derivative)
        # Actually no: if j=i gives a[i]*a[k-i] and j=k-i gives a[k-i]*a[i], these are the same term
        # when i = k-i. So the derivative is 2*a[k-i] if i != k-i, and 2*a[i] if i = k-i.
        # Either way: d(conv_k)/da_i = 2*a[k-i] if 0 <= k-i < n.
        
        sg = np.zeros(n)
        k_star = k_star
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = 2.0 * a[j]
        
        # Add small smoothing to avoid getting stuck
        if tau > 0:
            smooth_val, w = smooth_max_and_weights(conv, tau * 0.1)
            c = np.zeros(n)
            # Fast computation using FFT
            a_rev = a[::-1]
            N = 1
            while N < 3 * n - 2:
                N <<= 1
            fa_rev = rfft(a_rev, N)
            fw = rfft(w, N)
            conv_full = irfft(fa_rev * fw, N)
            c = conv_full[n - 1:2 * n - 1]
            sg = 0.9 * sg + 0.1 * 2.0 * n * c / (n * 2.0)
        
        mean_sg = np.dot(sg, a)
        grad = sg - mean_sg  # Projected onto simplex tangent
        
        a_new = mirror_descent_step(a, grad, eta)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            eta = min(eta * 1.05, 2.0)
        else:
            patience += 1
            eta *= 0.9
            if patience > 50:
                # Perturb to escape
                rng = np.random.default_rng(step)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                patience = 0
                eta = 0.3
    
    return best_a, best_obj


def diverse_init(n, rng, method='ramp'):
    """Create diverse initializations."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'ramp_up':
        slope = rng.uniform(2.0, 8.0)
        a = np.linspace(1.0, slope, n)
        return a / np.sum(a)
    elif method == 'ramp_down':
        slope = rng.uniform(2.0, 8.0)
        a = np.linspace(slope, 1.0, n)
        return a / np.sum(a)
    elif method == 'exp_decay':
        rate = rng.uniform(2.0, 8.0)
        a = np.exp(-rate * np.linspace(0, 1, n))
        return a / np.sum(a)
    elif method == 'exp_growth':
        rate = rng.uniform(2.0, 8.0)
        a = np.exp(rate * np.linspace(0, 1, n))
        return a / np.sum(a)
    elif method == 'cosine_skew':
        phase = rng.uniform(0.5, 2.0)
        x = np.linspace(0, 1, n)
        a = 1.0 + phase * np.cos(np.pi * x)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif method == 'linear_quadratic':
        x = np.linspace(0, 1, n)
        alpha = rng.uniform(0.5, 3.0)
        beta = rng.uniform(0.5, 3.0)
        a = alpha * x + beta * x * x
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif method == 'two_ramp':
        split = rng.uniform(0.3, 0.7)
        x = np.linspace(0, 1, n)
        a = np.where(x < split, 
                     1.0 + 3.0 * x / split,
                     4.0 - 3.0 * (x - split) / (1.0 - split))
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif method == 'smooth_random':
        # Random smooth positive function
        n_knots = rng.integers(5, 20)
        knots = np.sort(rng.uniform(0, 1, n_knots))
        values = np.maximum(rng.normal(1.0, 0.5, n_knots), 0.05)
        x = np.linspace(0, 1, n)
        a = np.interp(x, knots, values)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_seq(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
