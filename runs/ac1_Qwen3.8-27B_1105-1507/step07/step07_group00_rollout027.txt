from scipy.fft import rfft, irfft
import time
import math
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])

    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix B (2n-1 x n) such that B @ g = conv(a, g)."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for i in range(n):
        length = min(n, 2 * n - 1 - i)
        B[i:i + length, i] = a[:length]
    return B


def solve_lp_for_g(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns (g_normalized, M_opt, dual_u) or (None, None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    B = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [B @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start
        except Exception:
            pass

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Compute dual variables
                    dual_u = np.zeros(2 * n - 1)
                    try:
                        for i in range(2 * n - 1):
                            if constraints[i].dual_value is not None:
                                dual_u[i] = constraints[i].dual_value
                    except Exception:
                        pass
                    return g_val, float(M.value), dual_u
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = B
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                for i in range(n):
                    j = top_k - i
                    if 0 <= j < n:
                        g_contrib[i] = best_a[j]
                g_norm = g_contrib / (np.sum(g_contrib) + 1e-15)
                noise = rng.normal(0, scale, n)
                a = best_a * (1 + noise) + 0.1 * scale * g_norm
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                last_t = 0.5
                warm_g = None

    return best_a


def spectral_init(n, rng, mode=0):
    """Generate spectral-based initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Hann window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 1:
        # Raised cosine (smoother)
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 2
    elif mode == 2:
        # Asymmetric: more mass on left
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) * (0.5 + 0.5 * x)
    elif mode == 3:
        # Asymmetric: more mass on right
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) * (0.5 + 0.5 * (1 - x))
    elif mode == 4:
        # Super-Gaussian
        sigma = 0.15
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
    elif mode == 5:
        # Flat top with smooth edges
        w = 0.3
        a = np.ones(n)
        edge = int(w * n)
        for i in range(edge):
            a[i] = 0.5 * (1 - np.cos(np.pi * i / edge))
            a[n - 1 - i] = 0.5 * (1 - np.cos(np.pi * i / edge))
    elif mode == 6:
        # Random smooth
        freqs = rng.uniform(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        coefs = rng.uniform(0.1, 0.3, size=3)
        a = np.ones(n)
        for f, p, c in zip(freqs, phases, coefs):
            a += c * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode == 7:
        # Bump in middle, asymmetric
        c = 0.4 + 0.2 * rng.random()
        sigma = 0.1 + 0.1 * rng.random()
        a = np.exp(-((x - c) ** 2) / (2 * sigma ** 2))
        a += 0.05
    else:
        a = 0.5 * (1 + np.cos(2 * np.pi * x))

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def mirror_descent_fast(a, time_budget=100.0, topK=5, verbose=False):
    """
    Mirror descent with top-K peak subgradient and adaptive step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    stall = 0
    it = 0
    momentum = 0.0
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top K peaks
        top_idx = np.argsort(conv)[::-1][:topK]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Softmax weights
        temp = 3.0
        weights = np.exp((peak_vals - max_val) * temp / max_val)
        weights /= weights.sum()

        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += w * 2.0 * a[j]

        # Momentum
        momentum = min(0.8, momentum + 0.0005)
        v = momentum * v + g

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = max(1e-6, lr * 0.999)
        else:
            stall += 1
            if stall > 200:
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.15, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(1e-6, lr * 0.5)
                v = np.zeros(n)
                momentum = 0.0
                stall = 0
            else:
                lr = max(1e-6, lr * 0.97)

        it += 1

    return best_a


def peak_equalize(a, time_budget=60.0):
    """
    Direct peak equalization: iteratively reduce the max peak by
    redistributing mass from peak-contributing positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    it = 0

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # Find the top contributors to this peak
        # C[k_star] = sum_i a[i] * a[k_star - i]
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        if not contributors or contributors[0][0] < 1e-15:
            break

        # Find positions that DON'T contribute to this peak
        contributing_set = set()
        for _, i, j in contributors:
            contributing_set.add(i)
            if j != i:
                contributing_set.add(j)
        non_contrib = [i for i in range(n) if i not in contributing_set]

        if not non_contrib:
            # All positions contribute; try to reduce the peak contributor
            non_contrib = [i for i in range(n) if i != contributors[0][1]]

        if not non_contrib:
            break

        # Try moving a small fraction from the top contributor
        improved = False
        top_contrib_val, i_from, j_from = contributors[0]

        if i_from == j_from:
            # Diagonal term: a[i]^2. Reduce a[i] and spread to non-contrib positions
            from_idx = i_from
            delta_frac = 0.01
        else:
            # Off-diagonal: reduce the larger of a[i_from], a[j_from]
            if a[i_from] >= a[j_from]:
                from_idx = i_from
            else:
                from_idx = j_from
            delta_frac = 0.01

        for target in non_contrib[:10]:
            for frac in [0.005, 0.01, 0.02, 0.03]:
                delta = min(frac, 0.5) * a[from_idx]
                if delta < 1e-15:
                    break
                a[from_idx] -= delta
                a[target] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    break
                else:
                    a[from_idx] += delta
                    a[target] -= delta

            if improved:
                break

        if not improved:
            # Try reducing all top contributors slightly
            success = False
            for _, i, j in contributors[:5]:
                if a[i] > 1e-10:
                    delta = 0.005 * a[i]
                    a[i] -= delta
                    # Distribute to random non-contrib positions
                    targets = non_contrib[:5]
                    for t in targets:
                        a[t] += delta / len(targets)
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        success = True
                        break
            if not success:
                # Revert
                a = best_a.copy()
                # Small perturbation
                rng = np.random.default_rng(it * 104729)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                conv = compute_conv(a)

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + mirror descent + peak equalization + multi-n.
    """
    global height_sequence_1

    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: Quick mirror descent to get a slightly better starting point
    print("\n=== Phase 1: Mirror Descent Warm-up ===")
    remaining = deadline - time.monotonic()
    md_budget = min(remaining * 0.08, 80.0)
    if md_budget > 10:
        a_md = mirror_descent_fast(best_seq, time_budget=md_budget, topK=5)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD warm-up: obj={best_score:.8f} **")
        else:
            print(f"  MD warm-up: obj={obj_md:.8f}")

    # Phase 2: Main LP alternating (longer run)
    print("\n=== Phase 2: LP Alternating (Main) ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(remaining * 0.40, 480.0)
    if lp_budget > 30:
        a_lp = lp_alternating_fast(best_seq, time_budget=lp_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP main: obj={best_score:.8f} **")
        else:
            print(f"  LP main: obj={obj_lp:.8f}")

    # Phase 3: Peak equalization
    print("\n=== Phase 3: Peak Equalization ===")
    remaining = deadline - time.monotonic()
    pe_budget = min(remaining * 0.10, 90.0)
    if pe_budget > 10:
        a_pe = peak_equalize(best_seq, time_budget=pe_budget)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_score:
            best_score = obj_pe
            best_seq = a_pe.copy()
            print(f"  Peak eq: obj={best_score:.8f} **")
        else:
            print(f"  Peak eq: obj={obj_pe:.8f}")

    # Phase 4: Second LP round with the improved sequence
    print("\n=== Phase 4: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    lp2_budget = min(remaining * 0.25, 250.0)
    if lp2_budget > 20:
        a_lp2 = lp_alternating_fast(best_seq, time_budget=lp2_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP round2: obj={best_score:.8f} **")
        else:
            print(f"  LP round2: obj={obj_lp2:.8f}")

    # Phase 5: Multi-n exploration
    print("\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        n_parent - 100, n_parent - 50, n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        500, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]
    # Remove duplicates and sort by distance from parent
    n_options = sorted(set(n_options), key=lambda x: abs(x - n_parent))

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: MD then LP
        alloc = min(remaining * 0.15, 80.0)
        half = alloc / 2
        if half > 10:
            a_md = mirror_descent_fast(a_try, time_budget=half, topK=3)
            if compute_obj(a_md) < compute_obj(a_try):
                a_try = a_md

        lp_alloc = min(remaining * 0.10, 60.0)
        if lp_alloc > 15:
            a_opt = lp_alternating_fast(a_try, time_budget=lp_alloc)
        else:
            a_opt = a_try

        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt

        if remaining < 70:
            break

        mode = r % 8
        a_r = spectral_init(n_best, rng, mode=mode)

        # Blend with best (asymmetric blend)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:

        # Use LP for restart
        a_opt = lp_alternating_fast(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # Phase 7: Final intensive polish
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        # LP polish
        lp_alloc = min(remaining * 0.3, 100.0)
        a_lp = lp_alternating_fast(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:

            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()

        pe_alloc = min(remaining * 0.3, 50.0)
        a_pe = peak_equalize(best_seq, time_budget=pe_alloc)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_score:
            best_score = obj_pe
            best_seq = a_pe.copy()
            print(f"  Final PE: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # MD polish with high topK
        md_alloc = min(remaining * 0.3, 40.0)
        a_md = mirror_descent_fast(best_seq, time_budget=md_alloc, topK=8)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        # One more short LP
        a_last = lp_alternating_fast(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:

        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
