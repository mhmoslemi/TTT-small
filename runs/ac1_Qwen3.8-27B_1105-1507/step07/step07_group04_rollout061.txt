    stall_count = 0

        if g_lp is None:

        phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
        
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)

            return compute_obj_normalized(a_try)
        
        # Golden section search
        lo, hi = 0.0, 1.0

        f1 = obj_at_t(c1)
        f2 = obj_at_t(c2)

                f2 = f1

"""
LP Alternating Optimizer with diverse initializations, golden-section line
search, n-adaptation, and spectral nudging for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Fast linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Objective for sequence normalized to sum=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        idx = slice(k - j_max, k - j_min + 1)
        A[k, j_min:j_max + 1] = a[idx][::-1]
    return A


def solve_lp_conv(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def solve_lp_reverse(a):
    """Solve LP: min M s.t. conv(g,a)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A_rev = build_conv_matrix(a[::-1])
    g, M = solve_lp_conv(a[::-1])
    if g is not None:
        return g[::-1], M
    return None, None


def lp_single_iteration(a, a_dir):
    """One LP iteration: find g that minimizes peak conv(a,g), then line search."""
    n = len(a)
    if a_dir == 'fwd':
        g_lp, M_lp = solve_lp_conv(a)
    else:
        g_lp, M_lp = solve_lp_reverse(a)
    if g_lp is None:
        return a, None
    return a, g_lp


def golden_section_line_search(a, g, obj_fn, max_iter=20):
    """Golden section search for optimal blend parameter t in [0, 1]."""
    phi = (np.sqrt(5) - 1) / 2.0

    def f(t):
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-12:
            return np.inf
        return obj_fn(x / s)

    lo, hi = 0.0, 1.0
    c1 = lo + phi * (hi - lo)
    c2 = hi - phi * (hi - lo)
    f1 = f(c1)
    f2 = f(c2)

    for _ in range(max_iter):
        if f1 <= f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = lo + phi * (hi - lo)
            f1 = f(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = hi - phi * (hi - lo)
            f2 = f(c2)

    t_opt = (lo + hi) / 2.0
    x = (1.0 - t_opt) * a + t_opt * g
    x = np.maximum(x, 0)
    s = float(np.sum(x))
    if s < 1e-12:
        return a, np.inf
    x = x / s
    return x, f(t_opt)


def lp_alternate(a, time_budget=100.0, verbose=False, rng=None):
    """
    LP alternating optimization with golden section line search.
    Alternates between forward and reverse LP directions.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    stall = 0
    direction = 'fwd'

    while time.monotonic() - start < time_budget:
        # Solve LP
        if direction == 'fwd':
            g_lp, M_lp = solve_lp_conv(a)
        else:
            g_lp, M_lp = solve_lp_reverse(a)

        if g_lp is None:
            stall += 5
            if stall > 30:
                break
            rng2 = np.random.default_rng(int(time.monotonic() * 7919) % (2**31))
            a = best_a * np.exp(rng2.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / float(np.sum(a))
            direction = 'fwd' if direction == 'rev' else 'rev'
            continue

        # Line search
        a_new, obj_new = golden_section_line_search(a, g_lp, compute_obj_norm)

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP [{direction}]: obj={best_obj:.10f}, M_lp={M_lp:.6f}, t_elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 10:
                # Escape: try a larger perturbation
                scale = min(0.05 + 0.01 * (stall // 10), 0.2)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                a = a_pert.copy()
                obj_pert = compute_obj_norm(a)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                stall = 0
                direction = 'fwd' if direction == 'rev' else 'rev'
            elif stall >= 5:
                direction = 'fwd' if direction == 'rev' else 'rev'

    return best_a


def mwu_fast(a, time_budget=50.0, top_k=10, eta_init=0.1):
    """Fast multiplicative weight updates targeting top peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    eta = eta_init

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_idx = np.argsort(conv)[::-1][:top_k]

        # Compute subgradient (average over active lags)
        grad = np.zeros(n)
        for k in k_idx:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n, k + 1)
            for j in range(j_min, j_max):
                grad[j] += a[k - j]
        grad /= len(k_idx)

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-15)
        a = a / float(np.sum(a))

        obj = compute_obj_norm(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        else:
            eta *= 0.8
            if eta < 0.001:
                break
            a = best_a.copy()

    return best_a


def initialize_diverse(n, rng):
    """Generate diverse initial sequences of length n."""
    candidates = []

    # 1. Uniform
    candidates.append(np.ones(n) / n)

    # 2. Cosine window
    j = np.arange(n)
    a = np.cos(np.pi * j / (2.0 * n)) ** 2
    a = np.maximum(a, 0)
    candidates.append(a / a.sum())

    # 3. Triangle
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(j - center) / (center + 1)
    a = np.maximum(a, 0)
    candidates.append(a / a.sum())

    # 4. Inverse triangle (U-shape)
    a = np.abs(j - center) / (center + 1)
    a = np.maximum(a, 0)
    candidates.append(a / a.sum())

    # 5. Ramp
    a = (j + 1) / (n * (n + 1) / 2.0)
    candidates.append(a / a.sum())

    # 6. Inverse ramp
    a = (n - j) / (n * (n + 1) / 2.0)
    candidates.append(a / a.sum())

    # 7. Two peaks
    c1 = n * 0.25
    c2 = n * 0.75
    width = n * 0.15
    a = np.exp(-0.5 * ((j - c1) / width) ** 2) + np.exp(-0.5 * ((j - c2) / width) ** 2)
    candidates.append(a / a.sum())

    # 8. Edge concentrated
    a = np.ones(n)
    a[0] = 2.0
    a[-1] = 2.0
    candidates.append(a / a.sum())

    # 9-12. Random
    for _ in range(4):
        a = rng.random(n)
        candidates.append(a / a.sum())

    # 13. Quantized (step function with 10 levels)
    a = np.round(rng.random(n) * 10) / 10.0
    a = np.maximum(a, 0.01)
    candidates.append(a / a.sum())

    # 14. Logistic curve
    a = 1.0 / (1.0 + np.exp(-5.0 * (j / n - 0.5)))
    candidates.append(a / a.sum())

    # 15. Negative logistic
    a = 1.0 - 1.0 / (1.0 + np.exp(-5.0 * (j / n - 0.5)))
    a = np.maximum(a, 0)
    candidates.append(a / a.sum())

    return candidates


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_nudge(a, strength=0.1):
    """
    Nudge the sequence to flatten its convolution spectrum.
    Works by modifying the DFT magnitude to be more uniform,
    then projecting back to non-negative real sequence.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    # The convolution in frequency domain is |A|^2
    # We want to reduce peaks in |A|^2, which means reducing |A| at high-magnitude bins
    mag = np.abs(fa)
    phase = np.angle(fa)
    # Compress the magnitude
    mag_new = mag ** (1.0 - strength) / (mag ** (1.0 - strength) + 1e-10) ** 0
    # Actually, let's use a softer approach: reduce dynamic range
    mag_min = mag[mag > 1e-10].min() if np.any(mag > 1e-10) else 1e-10
    mag_max = mag.max()
    if mag_max > mag_min * 1e-3:
        # Compress toward geometric mean
        log_mag = np.log(np.maximum(mag, 1e-15))
        log_mag_compressed = log_mag - strength * (log_mag - log_mag.mean())
        mag_new = np.exp(log_mag_compressed)
    else:
        mag_new = mag.copy()

    fa_new = mag_new * np.exp(1j * phase)
    a_new = irfft(fa_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s

            best_seq = a_opt_n.copy()
            a_main = a_opt_n.copy()
            n_best = n_try
            print(f"-> obj={best_obj:.10f} **")
        else:
            print(f"-> obj={obj_n:.10f}")

    # === Phase 4: Combined LWU + LP restarts (15% of time) ===
    remaining = deadline - time.monotonic()
    phase4_budget = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 4: MWU + LP Combined (budget={phase4_budget:.0f}s) ===")

    n_restarts = 5
    time_per_restart = phase4_budget / n_restarts

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(time_per_restart, remaining * 0.9, 30.0)

        # Perturb best solution
        scale = 0.03 + 0.02 * r
        a_r = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(np.sum(a_r))

        # Quick MWU then LP
        a_mwu = mwu_fast(a_r, time_budget=alloc * 0.3, top_k=15)
        obj_mwu = compute_obj_norm(a_mwu)

        a_final = lp_alternate(a_mwu, time_budget=alloc * 0.7, rng=rng)
        obj_final = compute_obj_norm(a_final)

        if obj_final < best_obj:
            best_obj = obj_final
            best_seq = a_final.copy()
            a_main = a_final.copy()
            print(f"  Restart {r}: MWU={obj_mwu:.8f} -> LP={best_obj:.10f} **")
        else:
            print(f"  Restart {r}: MWU={obj_mwu:.8f} -> LP={obj_final:.8f}")

    # === Phase 5: Final polish (remaining time) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Final Polish (budget={remaining:.0f}s) ===")

    if remaining > 20:
        # LP on best
        alloc = min(remaining * 0.4, 100.0)
        a_polish = lp_alternate(best_seq.copy(), time_budget=alloc, verbose=True, rng=rng)
        obj_polish = compute_obj_norm(a_polish)
        if obj_polish < best_obj:
            best_obj = obj_polish
            best_seq = a_polish.copy()
            print(f"  Final LP: obj={best_obj:.10f} **")
        else:
            print(f"  Final LP: obj={obj_polish:.10f}")

    if remaining > 15:
        # Spectral nudge + LP
        a_spec = spectral_nudge(best_seq, strength=0.05)
        alloc = min(remaining * 0.3, 60.0)
        a_spec_opt = lp_alternate(a_spec, time_budget=alloc, rng=rng)
        obj_spec = compute_obj_norm(a_spec_opt)
        if obj_spec < best_obj:
            best_obj = obj_spec
            best_seq = a_spec_opt.copy()
            print(f"  Spectral nudge: obj={best_obj:.10f} **")
        else:
            print(f"  Spectral nudge: obj={obj_spec:.10f}")

    if remaining > 10:
        # Another LP pass with different direction emphasis
        alloc = min(remaining * 0.5, 50.0)
        a_last = lp_alternate(best_seq.copy(), time_budget=alloc, rng=rng)
        obj_last = compute_obj_norm(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_obj:.10f} **")

    # === Validate with evaluate_sequence ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < np.inf and val < best_obj:
            best_obj = val
    except Exception:
        pass

    # Fallback safety
    if best_seq is None or len(best_seq) == 0:
        result = [1.0] * 100
    else:
        result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, obj={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
