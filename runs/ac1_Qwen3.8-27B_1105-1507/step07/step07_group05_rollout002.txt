"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    return None


def lse_gradient(a, b, tau, K=1):
    """
    Compute gradient of log-sum-exp smoothing of top-K peaks.
    grad[i] = sum_k w_k * a[k-i] where w_k = exp(b[k]/tau) / sum exp(b[k']/tau)
    """
    n = len(a)
    # Focus on top K peaks
    if K >= len(b):
        top_indices = np.arange(len(b))
    else:
        top_indices = np.argsort(b)[-K:]

    # Temperature-scaled weights
    b_top = b[top_indices]
    b_max = np.max(b_top)
    exp_vals = np.exp((b_top - b_max) / tau)
    w = exp_vals / np.sum(exp_vals)

    # Gradient: grad[i] = sum_{k in top} w_k * a[k-i] (valid indices)
    grad = np.zeros(n)
    for idx, k in enumerate(top_indices):
        wi = w[idx]
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += wi * a[j]
    return grad


def fft_lse_gradient(a, b, tau, K=5, N=None):
    """
    FFT-accelerated log-sum-exp gradient.
    """
    n = len(a)
    if N is None:
        N = 1
        while N < 3 * n - 1:
            N <<= 1

    if K >= len(b):
        w = np.ones(len(b)) / len(b)
    else:
        top_indices = np.argsort(b)[-K:]
        b_top = b[top_indices]
        b_max = np.max(b_top)
        exp_vals = np.exp((b_top - b_max) / tau)
        w = np.zeros(len(b))
        w[top_indices] = exp_vals / np.sum(exp_vals)

    # grad[i] = sum_k w[k] * a[k-i] = (w * a_flip) correlation
    # This is the "full" convolution of w with reversed a, shifted
    a_flip = a[::-1]
    fw = rfft(w, N)
    faf = rfft(a_flip, N)
    c_full = irfft(fw * faf, N)[:2 * n - 1]
    # c_full[i+j] = sum_k w[k] * a_flip[k-i-j+...] 
    # Actually: conv(w, a_flip)[m] = sum_k w[k] * a_flip[m-k]
    # We want grad[i] = sum_k w[k] * a[k-i] = sum_k w[k] * a_flip[n-1-(k-i)]
    # = sum_k w[k] * a_flip[n-1-k+i]
    # Let m = n-1+i-k, so k = n-1+i-m
    # grad[i] = sum_m w[n-1+i-m] * a_flip[m] = conv(a_flip, w)[n-1+i]
    # So grad[i] = conv(a_flip, w)[n-1+i] for i = 0,...,n-1
    # conv(a_flip, w) has length 2n-1
    # We need indices n-1, n, ..., 2n-2
    # But our c_full is conv(w, a_flip) which has the same values (convolution is commutative)
    # conv(w, a_flip)[m] = sum_k w[k]*a_flip[m-k]
    # conv(a_flip, w)[m] = sum_k a_flip[k]*w[m-k]
    # These are the same! So c_full[m] = sum_k w[k]*a_flip[m-k]
    # We want grad[i] = sum_k w[k]*a[k-i] = sum_k w[k]*a_flip[n-1-k+i]
    # Let j = n-1-k+i, k = n-1+i-j
    # grad[i] = sum_j w[n-1+i-j]*a_flip[j] = conv(a_flip, w)[n-1+i]
    # = conv(w, a_flip)[n-1+i] by commutativity
    # So grad[i] = c_full[n-1+i]

    # But c_full has length 2n-1, and n-1+i goes from n-1 to 2n-2, which is valid.
    grad = c_full[n - 1:2 * n - 1]
    return grad


def lse_objective(b, tau, K=5):
    """Compute log-sum-exp of top K peaks."""
    if K >= len(b):
        top = b
    else:
        top = np.sort(b)[-K:]
    b_max = np.max(top)
    return tau * (b_max + np.log(np.sum(np.exp((top - b_max) / tau))))


def nesterov_lse_gd(a, time_budget=120.0, tau_start=0.1, tau_end=0.001,
                     tau_decay=0.999, K=5, momentum_start=0.5, momentum_end=0.9,
                     verbose=False):
    """
    Nesterov momentum gradient descent on LSE-smoothed peak objective.
    Adaptive temperature cooling.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    eta = 0.5
    momentum = momentum_start
    v = np.zeros(n)
    tau = tau_start
    stall = 0
    max_stall = 100

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Compute b = conv(a, a)
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Compute LSE gradient
        grad = fft_lse_gradient(a, b, tau, K=K, N=N)

        # Center gradient (project onto sum=0 subspace)
        grad = grad - np.mean(grad)

        # Nesterov: look-ahead
        a_look = a + momentum * v
        fa_look = rfft(a_look, N)
        b_look = irfft(fa_look * fa_look, N)[:2 * n - 1]
        grad_look = fft_lse_gradient(a_look, b_look, tau, K=K, N=N)
        grad_look = grad_look - np.mean(grad_look)

        # Update
        v = momentum * v - eta * grad_look
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        # Evaluate
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 3.0)
            momentum = min(momentum + 0.0005, momentum_end)
            tau = max(tau * tau_decay, tau_end)
        else:
            stall += 1
            eta *= 0.95
            v *= 0.5
            if stall > max_stall:
                # Reset with perturbation
                rng_local = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = momentum_start
                tau = tau_start * 0.5
                stall = 0
                max_stall = min(max_stall + 50, 300)
            if eta < 1e-8:
                break

        if verbose and it % 2000 == 0:
            print(f"            LSE-GD it{it}: obj={best_obj:.8f}, tau={tau:.5f}, eta={eta:.4f}, mom={momentum:.3f}")

    return best_a

            eta = min(eta * 1.01, 2.0)
            momentum = min(momentum + 0.0005, 0.9)
            tau = max(tau * tau_decay, tau_end)
        else:
            stall += 1
            eta *= 0.95
            v *= 0.5
            if stall > max_stall:
                rng_local = np.random.default_rng(it + 99999)
                a = best_a * np.exp(rng_local.normal(0, 0.02, n))
                a = 0.5 * (a + a[::-1])
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(half_n)
                eta = 0.3
                momentum = 0.5
                tau = tau_start * 0.5
                stall = 0
                max_stall = min(max_stall + 50, 400)
            if eta < 1e-8:
                break

        if verbose and it % 2000 == 0:
            print(f"            SymLSE it{it}: obj={best_obj:.8f}, tau={tau:.5f}, eta={eta:.4f}")

    return best_a


def peak_deflation(a, time_budget=60.0, verbose=False):
    """
    Iterative peak deflation: identify top peaks, push mass away from
    positions that contribute to those peaks, redistribute to low-peak positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    alpha = 0.1
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        b_max = np.max(b)
        obj = 2.0 * n * b_max

        # Find top K peak positions
        K = min(3, 2 * n - 1)
        top_k = np.argsort(b)[-K:][::-1]

        # For each peak position k, the contribution to b[k] from a[i] is a[k-i]
        # We want to reduce a[i] where it contributes most to peaks
        pressure = np.zeros(n)
        for k in top_k:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j] * (b[k] / b_max)

        # Normalize pressure
        p_max = np.max(pressure)
        if p_max > 1e-15:
            pressure = pressure / p_max

        # Deflate: reduce a where pressure is high, increase where it's low
        avg_pressure = np.mean(pressure)
        delta = pressure - avg_pressure
        a_new = a - alpha * delta
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.02, 0.3)
        else:
            alpha *= 0.9
            if alpha < 1e-6:
                break

        if verbose and it % 500 == 0:
            print(f"            Deflation it{it}: obj={best_obj:.8f}, alpha={alpha:.4f}")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing with fine step size search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.5
        # Fine grid for t
        for t in np.concatenate([np.linspace(0.01, 0.3, 10), np.linspace(0.3, 1.0, 8)]):
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall > 10:
                break

    return best_a


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    elif mode == 6:
        # Symmetric random
        half = (n + 1) // 2
        h = rng.uniform(0.1, 1.0, half)
        if n % 2 == 0:
            a = np.concatenate([h, h[::-1]])
        else:
            a = np.concatenate([h, h[-2::-1]])
        a = a / np.sum(a)
    else:
        # Inverse U (dip in middle)
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0 - 0.8 * np.exp(-((x - 0.5) ** 2) / 0.05)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
