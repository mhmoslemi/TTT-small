Hybrid optimization: Differential Evolution on log-polynomial parametric family

+ multi-n exploration.

Key innovations over previous approach:
1. Uses LP dual variables to compute exact gradient of the LP value w.r.t. a

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def log_poly_sequence(params, n):
    """Generate normalized non-negative sequence from log-polynomial parameters."""
    t = np.linspace(0.0, 1.0, n)
    log_a = np.zeros(n)
    for d in range(len(params)):
        log_a += params[d] * t**d
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    s = np.sum(a)
    if s < 1e-300:
        return np.ones(n) / n
    return a / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        i_max = min(n - 1, 2 * n - 2 - j)
        for i in range(i_max + 1):
            A[i + j, j] = a[i]
    return A


def solve_lp_with_dual(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, mu) where mu are dual variables (normalized to sum=1).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    s = np.sum(g)
    if s < 1e-10:
        return None, None, None
    g = g / s
    M = float(res.x[n])

    mu = None

        mu_sum = np.sum(mu)

        else:
            mu = np.ones(2 * n - 1) / (2 * n - 1)

def lp_dual_gradient(a, mu, g):

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max(conv, tau):
    """Compute smoothed maximum via log-sum-exp."""
    m = np.max(conv)
    return m + np.log(np.sum(np.exp((conv - m) * tau))) / tau

    """Compute soft-argmax weights."""
    m = np.max(conv)
    w = np.exp((conv - m) * tau)
    w = w / np.sum(w)
    return w

def differential_evolution(n, n_params, budget_s, rng, n_pop=40, F=0.8, CR=0.7, max_gen=None):

    Differential evolution on log-polynomial parameters.
    Returns best params and best objective.
    """

    # Initial population
    pop = np.zeros((n_pop, n_params))
    for i in range(n_pop):
        pop[i] = rng.uniform(-3, 3, n_params)

    best_idx = np.argmin(objs)
    best_obj = objs[best_idx]
    best_params = pop[best_idx].copy()
    
    if max_gen is None:
        max_gen = 200

    while gen < max_gen:
        if time.monotonic() - start > budget_s:
            break

            if time.monotonic() - start > budget_s:

                objs[i] = obj_child
                if obj_child < best_obj:
                    best_obj = obj_child
                    best_params = child.copy()
        
        if gen % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    DE gen {gen}/{max_gen}: best_obj={best_obj:.8f} (t={elapsed:.1f}s)")
    
    return best_params, best_obj


def lp_alternating_dual_gradient(a, time_budget, verbose=False):
    """
    LP alternating with dual gradient descent on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        g, M, mu = solve_lp_with_dual(a)
        if g is None:
            break
        
        # Compute dual gradient
        grad = lp_dual_gradient(a, mu, g)
        
        # Line search: try several step sizes
        best_step_obj = best_obj
        best_step_a = best_a.copy()
        
        for eta in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_try = a - eta * grad
            a_try = simplex_project(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()
        
        # Also try blending with LP solution
        for t in [0.1, 0.3, 0.5]:
            a_blend = (1 - t) * a + t * g
            a_blend = np.maximum(a_blend, 0)
            s_blend = np.sum(a_blend)
            if s_blend > 1e-10:
                a_blend = a_blend / s_blend
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_step_obj:
                    best_step_obj = obj_blend
                    best_step_a = a_blend.copy()
        
        if best_step_obj < best_obj - 1e-14:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_a
            if verbose and it % 20 == 0:
                print(f"      LP-DG it{it}: obj={best_obj:.8f}")
        elif it % 15 == 0:
            # Perturb to escape local minimum
            rng = np.random.default_rng(it * 104729 + n)
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
    
    return best_a


def block_simulated_annealing(a, time_budget, n_blocks=12, T_start=0.01, T_end=0.0001):
    """
    Block-level simulated annealing.
    Transfers mass between blocks of the sequence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    
    # Define block boundaries
    block_size = max(1, n // n_blocks)
    block_starts = np.arange(0, n, block_size)
    n_actual_blocks = len(block_starts)
    
    rng = np.random.default_rng(12345)
    n_iters = 0
    
    while time.monotonic() - start < time_budget:
        n_iters += 1
        # Temperature schedule
        frac = min(1.0, n_iters / (time_budget * 200))  # rough estimate
        T = T_start * (T_end / T_start) ** frac
        
        # Select two distinct blocks
        b1, b2 = rng.choice(n_actual_blocks, size=2, replace=False)
        
        # Block ranges
        s1 = block_starts[b1]
        e1 = min(s1 + block_size, n)
        s2 = block_starts[b2]
        e2 = min(s2 + block_size, n)
        
        mass1 = np.sum(a[s1:e1])
        mass2 = np.sum(a[s2:e2])
        
        if mass1 < 1e-10 and mass2 < 1e-10:
            continue
        
        # Transfer fraction f of mass between blocks
        f = rng.uniform(0.05, 0.3)
        direction = rng.choice([1, -1])  # +1: from block1 to block2
        
        if direction == 1:
            transfer = f * mass1
            a[s1:e1] *= (1 - f)
            a[s2:e2] += transfer * (a[s2:e2] / (mass2 + 1e-15))
            if mass2 < 1e-15:
                a[s2:e2] = transfer / (e2 - s2)
        else:
            transfer = f * mass2
            a[s2:e2] *= (1 - f)
            a[s1:e1] += transfer * (a[s1:e1] / (mass1 + 1e-15))
            if mass1 < 1e-15:
                a[s1:e1] = transfer / (e1 - s1)
        
        a = np.maximum(a, 0)
        s_cur = np.sum(a)
        if s_cur < 1e-10:
            a = best_a.copy()
            continue
        a = a / s_cur
        
        obj_cur = compute_obj(a)
        delta = obj_cur - current_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            current_obj = obj_cur
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_a = a.copy()
        else:
            # Reject: revert
            a = best_a.copy()
            current_obj = best_obj
    
    return best_a


def multi_scale_optimization(a, time_budget):
    """
    Optimize at multiple scales: coarsen, optimize, refine.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Try a few coarser resolutions
    for coarsen in [4, 2]:
        if time.monotonic() - start > time_budget * 0.8:
            break
        
        n_coarse = max(50, n // coarsen)
        # Downsample
        a_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            lo = int(i * n / n_coarse)
            hi = int((i + 1) * n / n_coarse)
            hi = max(hi, lo + 1)
            a_coarse[i] = np.sum(a[lo:hi])
        
        a_coarse = np.maximum(a_coarse, 0)
        s = np.sum(a_coarse)
        if s > 1e-10:
            a_coarse = a_coarse / s
        
        # Optimize at coarse level
        alloc = min(time_budget * 0.2, 20.0)
        a_coarse_opt = lp_alternating_dual_gradient(a_coarse, alloc)
        
        # Upsample back
        a_up = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_coarse), a_coarse_opt)
        a_up = np.maximum(a_up, 0)
        s = np.sum(a_up)
        if s > 1e-10:
            a_up = a_up / s
            obj_up = compute_obj(a_up)
            if obj_up < best_obj:
                best_obj = obj_up
                best_a = a_up.copy()
                a = a_up
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
