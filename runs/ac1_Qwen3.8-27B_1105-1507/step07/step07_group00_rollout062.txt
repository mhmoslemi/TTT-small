Optimized minimization of 2*n*max(conv(a,a))/(sum(a))^2 using
Nesterov-accelerated smoothed gradient descent combined with LP-based
direction finding, adaptive temperature, and multi-scale exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_gradient(conv, a, T):
    """
    Compute gradient of log-sum-exp_T(conv) with respect to a.
    Uses softmax weights over all convolution bins.
    """
    n = len(a)
    max_conv = np.max(conv)
    # For numerical stability and efficiency, only compute weights for bins
    # that are "close" to the max
    # Actually, let's just do it for all bins since 2n-1 is at most ~2000
    shifted = (conv - max_conv) * T
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        # Fallback: uniform over top bins
        top_k = 10
        idx = np.argsort(conv)[::-1][:top_k]
        w = np.zeros(len(conv))
        w[idx] = 1.0 / top_k
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-12:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end+1] += w[k] * a[j]
    
    return grad

def nesterov_smoothed_descent(a, time_budget=100.0, T_init=10.0, T_final=100.0, verbose=False):
    """
    Nesterov-accelerated gradient descent on the smoothed objective
    log-sum-exp_T(conv(a,a)) with simplex projection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # extrapolated point
    beta = 0.9  # momentum parameter (will be updated)
    step = 1.0 / T_init  # initial step size
    
    it = 0
    stall = 0
    T_current = T_init
    
    while time.monotonic() - start < time_budget:
        # Adaptive temperature: increase over time (sharper peaks)
        progress = (time.monotonic() - start) / time_budget
        T_current = T_init * (T_final / T_init) ** progress
        
        # Compute gradient at extrapolated point y
        conv_y = compute_conv(y)
        grad_y = smoothed_gradient(conv_y, y, T_current)
        
        # Nesterov update
        a_old = a.copy()
        a = y - step * grad_y
        a = np.maximum(a, 0)
        s_new = float(np.sum(a))
        if s_new > 1e-10:
            a = a / s_new
        else:
            a = best_a.copy()
            break

        # Let me use a simpler momentum scheme

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of LP-based
direction finding, Nesterov-accelerated smoothed gradient descent,
adaptive line search, and multi-scale exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

# Try to import cvxpy for LP solving
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def convolve_with_filter(a, g):
    """Compute conv(a, g) efficiently using FFT."""
    n = len(a)
    m = len(g)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:n + m - 1]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    if not HAS_CVXPY:
        # Fallback: use the uniform direction
        return np.ones(n) / n, np.max(A @ (np.ones(n) / n))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Compute actual M
                    conv_g = A @ g_val
                    M_val = float(np.max(conv_g))
                    return g_val, M_val
        except Exception:
            continue
    
    return None, None


def golden_section_line_search(a, direction, obj_func, max_iter=50):
    """
    Golden section line search: find t in [0,1] minimizing obj_func((1-t)*a + t*direction).
    """
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    
    def f(t):
        x = (1.0 - t) * a + t * direction
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            return np.inf
        return obj_func(x / s)
    
    lo, hi = 0.0, 1.0
    c = lo + (1 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1 - gr) * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    
    t_best = (lo + hi) / 2
    x_best = (1.0 - t_best) * a + t_best * direction
    x_best = np.maximum(x_best, 0)
    s = float(np.sum(x_best))
    if s > 1e-10:
        x_best = x_best / s
    else:
        x_best = a.copy()
    
    return x_best, f(t_best)


def smoothed_max_gradient(conv, a, T):
    """
    Compute gradient of (1/T)*log(sum(exp(T*conv[k]))) with respect to a.
    This approximates the subgradient of max(conv) as T -> inf.
    """
    n = len(a)
    max_val = float(np.max(conv))
    shifted = (conv - max_val) * T
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        # Fallback to top-1
        k = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[k] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    # Only process significant weights
    threshold = 1e-10
    active = w > threshold
    for k in np.where(active)[0]:
        wk = w[k]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end+1] += wk * a[j]
    
    return grad


def nesterov_smoothed_descent(a, time_budget=50.0, T_init=20.0, T_end=200.0, verbose=False):
    """
    Nesterov-accelerated gradient descent on smoothed max of convolution.
    Works on the probability simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Parameters
    step = 2.0 / (T_init * np.sqrt(n))
    beta = 0.9
    t_nest = 1.0  # Nesterov counter
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T_cur = T_init * (T_end / T_init) ** progress
        
        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 1.0 - t_nest / t_new
        t_nest = t_new
        
        y = a + alpha * (a - (a - step * 0))  # simplified
        # Actually, let me use a simpler momentum scheme
        # y = a + momentum * (a - a_prev)
        # For simplicity, let's use heavy ball momentum
        
        # Compute gradient at current a
        conv_a = compute_conv(a)
        grad = smoothed_max_gradient(conv_a, a, T_cur)
        
        # Update
        a_old = a.copy()
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a_old
            break
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            step *= 0.9995
        else:
            stall += 1
            if stall > 50:
                step *= 0.8
                beta = 0.5
            if stall > 200:
                a = best_a * (1 + 0.05 * np.random.randn(n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                step = 2.0 / (T_cur * np.sqrt(n))
                beta = 0.5
                stall = 0
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSD it{it}: obj={best_obj:.8f}, T={T_cur:.1f}, step={step:.2e}, t={elapsed:.1f}s")
    
    return best_a


def hybrid_optimize(a, time_budget=200.0, verbose=False):
    """

    and Nesterov gradient steps.
    """

    s = float(np.sum(a))

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    grad_step_budget = 5.0  # seconds per gradient phase
    lp_cooldown = 0  # iterations since last LP
    grad_step = 1.0 / np.sqrt(n)
    
    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Decide: LP or gradient?
        use_lp = (lp_cooldown >= 3 and remaining > 30)
        
        if use_lp and HAS_CVXPY:
            # LP step
            g_lp, M_lp = solve_lp_direction(a)
            if g_lp is not None:
                # Line search
                a_new, obj_new = golden_section_line_search(a, g_lp, compute_obj, max_iter=40)
                if obj_new < best_obj - 1e-12:

                    a = a_new
                    if verbose:

                lp_cooldown = 0
            else:
                lp_cooldown = 10  # don't retry LP for a while
        else:
            # Gradient step (batch of small steps)
            grad_end = time.monotonic() + min(grad_step_budget, remaining * 0.3)
            T = 50.0
            while time.monotonic() < grad_end:
                conv_a = compute_conv(a)
                grad = smoothed_max_gradient(conv_a, a, T)
                a_old = a.copy()
                a_new = a - grad_step * grad
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:

def coordinate_shave(a, time_budget=30.0):
    """
    Coordinate shaving: identify peak-contributing pairs and move mass
    to reduce the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find contributing pairs for this peak
        pairs = []
        for i in range(max(0, k_star - n + 1), min(n, k_star + 1)):
            j = k_star - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        
        pairs.sort(reverse=True)
        
        if not pairs or pairs[0][0] < 1e-12:
            break
        
        improved = False
        for contrib, i, j in pairs[:10]:
            if time.monotonic() - start > time_budget:
                break
            
            # Try moving mass from i to a non-contributing position
            non_contrib = [k for k in range(n) 
                         if k not in range(max(0, k_star-n+1), min(n, k_star+1))]
            
            for target in non_contrib[:3]:
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * min(a[i], a[j]) if i != j else frac * a[i]
                    if delta < 1e-12:
                        continue
                    if i != j:
                        a[i] -= delta * 0.5
                        a[j] -= delta * 0.5
                        a[target] += delta
                    else:
                        a[i] -= delta
                        a[target] += delta
                    
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        if i != j:
                            a[i] += delta * 0.5
                            a[j] += delta * 0.5
                        else:
                            a[i] += delta
                        a[target] -= delta
                    a = best_a.copy()
                
                if improved:
                    break
        
        if not improved:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal (edges)
        x = np.arange(n)
        a = np.exp(-((x - 0)**2) / (0.1*n)**2) + np.exp(-((x - (n-1))**2) / (0.1*n)**2)
        a = a / a.sum()
    elif mode == 3:
        # Bimodal (center)
        x = np.arange(n)
        center = (n-1)/2
        a = np.exp(-((x - center)**2) / (0.2*n)**2)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Sinc-like
        x = np.linspace(-3, 3, n)
        a = np.sinc(x / 3.0)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Random smooth
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)
