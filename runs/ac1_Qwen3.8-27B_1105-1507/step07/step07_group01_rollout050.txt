def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    s = float(np.sum(a))

        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""

            def obj_at(v):

                max_affected = float(np.max(affected_vals))
                max_total = max(m_oth, max_affected)
                new_s = s - ai + v
                if new_s < 1e-10:
                    return np.inf
                return 2.0 * n * max_total / (new_s * new_s)
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = obj_at(c)
            fd = obj_at(d)
            
            for _ in range(40):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_at(d)
            
            v_opt = (lo + hi) / 2.0
            obj_opt = obj_at(v_opt)
            
            if obj_opt < best_obj - 1e-14:
                a[i] = v_opt
                s = s - ai + v_opt
                # Update conv incrementally (or recompute)
                improved = True
            elif v_opt < 1e-12 and ai > 1e-8:
                a[i] = 0.0
                s = s - ai
                improved = True
        
        if improved:
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s < 1e-10:
                break
            a = a / s
            obj_now = compute_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
                stall = 0
                if verbose and sweep % 5 == 0:
                    print(f"        CoordDes sweep {sweep}: obj={best_obj:.8f}")
            else:
                a = best_a.copy()
                stall += 1
        else:
            stall += 1
            if stall > 5:
                break

    return best_a

def smooth_max_gradient_descent(a, time_budget=60.0, T_init=0.001, T_min=0.0001):
    """Gradient descent with smooth max and BB step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    v = np.zeros(n)  # momentum
    momentum = 0.7
    lr = 0.01
    stall = 0
    prev_grad = None
    prev_x = None

    max_iters = 10000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        
        # Gradient: grad[i] = sum_j weights[i+j] * a[j]
        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        # BB step size
        if prev_grad is not None and prev_x is not None:
            s_vec = a - prev_x
            y_vec = grad - prev_grad
            sy = float(np.dot(s_vec, y_vec))
            yy = float(np.dot(y_vec, y_vec))
            ss = float(np.dot(s_vec, s_vec))
            if yy > 1e-12 and sy > 1e-12:
                lr_bb = ss / sy
                lr = max(0.001, min(0.1, lr_bb))
        
        prev_grad = grad.copy()
        prev_x = a.copy()

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break

        if it > 100:
            T = max(T_min, T_init * (0.995 ** it))

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        a = np.cumsum(rng.uniform(0, 1, n))
        return a / np.sum(a)
    elif mode == 3:
        center = n // 2
        width = max(1, n // 5)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a += 0.5 * np.exp(-((x - center * 0.7) ** 2) / (2 * (width // 2 + 1) ** 2))
        return a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        return a / np.sum(a)
    elif mode == 5:
        x = np.arange(n) / max(1, n - 1)
        a = 1 - np.abs(2 * x - 1)
        return a / np.sum(a)
    else:
        return rng.dirichlet(np.ones(n))

def multi_peak_gradient_descent(a, time_budget=40.0, num_peaks=5):
    """
    Gradient descent targeting the top-k peaks simultaneously.
    This creates a flatter autocorrelation profile.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.01
    v = np.zeros(n)
    momentum = 0.7
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top-k peak bins
        sorted_idx = np.argsort(conv)[::-1]
        top_idx = sorted_idx[:num_peaks]
        
        # Weighted gradient from top peaks
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        
        grad_centered = grad - np.mean(grad)
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break

    return best_a
