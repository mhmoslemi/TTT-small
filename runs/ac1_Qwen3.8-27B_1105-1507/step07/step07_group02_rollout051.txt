from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def soft_max_obj(a, temp=0.01):
    """Smooth approximation of the objective using log-sum-exp."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    # soft max approximation
    max_conv = float(np.max(conv))
    log_sum_exp = max_conv + temp * np.log(np.sum(np.exp((conv - max_conv) / temp)))
    return 2.0 * n * log_sum_exp

    """Optimize using softened objective with scipy L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()

        if time.monotonic() - start > time_budget:
            break
        
        def objective(x):
            a_try = np.exp(x)  # ensures positivity
            a_try = a_try / np.sum(a_try)
            return soft_max_obj(a_try, temp)
        
        # Start from log of current best
        a_clipped = np.maximum(best_a, 1e-10)

                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
        except Exception:
            pass

        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find all positions achieving (near) max
        threshold = max_val * (1 - 1e-6)
        max_positions = np.where(conv >= threshold)[0]
        
        # Subgradient: average over all max positions
        grad = np.zeros(n)
        for k in max_positions:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(max_positions)
        
        # Center the gradient (for the constraint sum a = 1)
        grad_centered = grad - np.sum(a * grad)  # project to tangent space

        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.maximum(a_new, 1e-15)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        
        eta *= decay
    
    return best_a

    #                       = a[j] * a[k-j] (if 0<=k-j<n and k != 2j) [since a[k-j] when k-j=j is a[j]]
    # Wait, let me be more careful.
    
    # conv[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
    # The terms involving a[j] are:
    #   i = j: a[j] * a[k-j] (valid if 0 <= k-j < n)
    #   k-i = j, i.e., i = k-j: a[k-j] * a[j] (valid if 0 <= k-j < n)
    # If j != k-j: both terms present, total = 2*a[j]*a[k-j]
    # If j == k-j (k = 2j): only one term, total = a[j]^2
    
    # So old_contribution to conv[k]:
    #   if 0 <= k-j < n and k != 2j: 2*a[j]*a[k-j]
    #   if k == 2j: a[j]^2
    #   else: 0
    
    # Similarly for new_contribution with x_new replacing a[j].
    
    # The change in conv[k]:
    #   if 0 <= k-j < n and k != 2j: 2*(x_new - x_old)*a[k-j]
    #   if k == 2j: x_new^2 - x_old^2
    #   else: 0
    
    # New sum: s_new = s - x_old + x_new
    # Objective: 2*n*max(conv_new) / s_new^2
    
    # We only need max(conv_new)
    # conv_new[k] = full_conv[k] + delta[k]
    # where delta[k] is as above
    
    # To find max(conv_new), we need to check all k where delta[k] != 0
    # plus the max of full_conv[k] where delta[k] = 0
    
    # k where delta[k] != 0: k = i+j for i in [0, n), i.e., k in [j, j+n-1]
    # Plus k = 2j (already included if 2j is in [j, j+n-1], which it is since j <= 2j <= j+n-1 for n>=2)
    
    # So we need to check k in [j, j+n-1] ∩ [0, 2n-2]
    
    # Additionally, the max of full_conv outside this range is unchanged.
    
    k_start = max(0, j)
    k_end = min(conv_len - 1, j + n - 1)
    
    max_outside = 0.0
    if k_start > 0:
        max_outside = max(max_outside, float(np.max(full_conv[:k_start])))
    if k_end < conv_len - 1:
        max_outside = max(max_outside, float(np.max(full_conv[k_end+1:])))
    
    max_inside = max_outside
    for k in range(k_start, k_end + 1):
        i = k - j
        if 0 <= i < n:
            if i == j:  # k = 2*j
                delta = x_new * x_new - x_old * x_old
            else:
                delta = 2 * (x_new - x_old) * a[i]
        else:
            delta = 0.0
        val = full_conv[k] + delta
        if val > max_inside:
            max_inside = val
    
    s_new = s - x_old + x_new
    if s_new < 0.01:
        return np.inf
    return 2.0 * n * max_inside / (s_new * s_new)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent,
mirror descent, and multi-start exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def fast_coord_descent(a, time_budget=60.0, max_sweeps=500):
    """
    Fast coordinate descent with incremental convolution updates.
    For each coordinate j, minimize the objective by changing only a[j].
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    full_conv = compute_conv(a)
    conv_len = 2 * n - 1
    start = time.monotonic()
    
    sweep = 0
    while sweep < max_sweeps:
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for j in range(n):
            x_old = a[j]
            
            # Find range of k affected by changing a[j]
            k_start = j
            k_end = min(conv_len - 1, j + n - 1)
            
            # Precompute max of full_conv outside the affected range
            max_outside = 0.0
            if k_start > 0:
                max_outside = max(max_outside, float(np.max(full_conv[:k_start])))
            if k_end < conv_len - 1:
                max_outside = max(max_outside, float(np.max(full_conv[k_end + 1:])))
            
            # Try to find better x
            # The objective as a function of x:
            # f(x) = 2n * max_k [full_conv[k] + delta_k(x)] / (s - x_old + x)^2
            # where delta_k(x) = 2*(x-x_old)*a[k-j] for k in [j, j+n-1], k != 2j
            #                   x^2 - x_old^2 for k = 2j
            #                   0 otherwise
            
            # For efficiency, let's evaluate at a few candidate x values
            # and pick the best
            
            # Candidate: x = 0 (set coordinate to zero)
            # Candidate: x = x_old (no change, skip)
            # Candidate: x that makes the affected conv values equal (equalization)
            
            # Simple approach: try x values around x_old
            candidates = [0.0, x_old * 0.1, x_old * 0.3, x_old * 0.5,
                         x_old * 0.7, x_old * 1.3, x_old * 1.5, x_old * 2.0,
                         s / n]  # uniform value
            
            best_x = x_old
            best_f = 2.0 * n * float(np.max(full_conv)) / (s * s)  # current objective
            
            for x_try in candidates:
                if x_try < 0 or x_try == x_old:
                    continue
                s_new = s - x_old + x_try
                if s_new < 0.01:
                    continue
                
                # Compute max of new convolution
                max_new = max_outside
                for k in range(k_start, k_end + 1):
                    i = k - j
                    if 0 <= i < n:
                        if i == j:
                            delta = x_try * x_try - x_old * x_old
                        else:
                            delta = 2.0 * (x_try - x_old) * a[i]
                    else:
                        delta = 0.0
                    val = full_conv[k] + delta
                    if val > max_new:
                        max_new = val
                
                f_try = 2.0 * n * max_new / (s_new * s_new)
                if f_try < best_f:
                    best_f = f_try
                    best_x = x_try
            
            # Refine with a smaller grid around best_x if it's not a boundary
            if 0 < best_x < s * 0.5 and best_x != x_old:
                # Narrow search
                for x_try in [best_x * 0.8, best_x * 0.9, best_x * 1.1, best_x * 1.2]:
                    s_new = s - x_old + x_try
                    if s_new < 0.01 or x_try < 0:
                        continue
                    max_new = max_outside
                    for k in range(k_start, k_end + 1):
                        i = k - j
                        if 0 <= i < n:
                            if i == j:
                                delta = x_try * x_try - x_old * x_old
                            else:
                                delta = 2.0 * (x_try - x_old) * a[i]
                        else:
                            delta = 0.0
                        val = full_conv[k] + delta
                        if val > max_new:
                            max_new = val
                    f_try = 2.0 * n * max_new / (s_new * s_new)
                    if f_try < best_f:
                        best_f = f_try
                        best_x = x_try
            
            if abs(best_x - x_old) > 1e-12:
                # Update
                a[j] = best_x
                # Update full_conv incrementally
                for k in range(k_start, k_end + 1):
                    i = k - j
                    if 0 <= i < n:
                        if i == j:
                            full_conv[k] += best_x * best_x - x_old * x_old
                        else:
                            full_conv[k] += 2.0 * (best_x - x_old) * a[i]
                
                s = s - x_old + best_x
                # Renormalize? No, let's keep the sum unchanged for now
                # Actually, we should renormalize to keep the problem well-defined
                # But for coordinate descent, it's standard to not renormalize
                # until the end of the sweep
                
                if best_f < best_obj * (1 - 1e-10):
                    best_obj = best_f
                    improved = True
        
        # After each sweep, renormalize
        s = float(np.sum(a))
        if s > 0:
            a = a / s
            full_conv = compute_conv(a)  # Recompute after renormalization
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        sweep += 1
        
        # Check for convergence
        if sweep > 20 and not improved:
            break
    
    return best_a

def mirror_descent(a, time_budget=60.0, max_iters=5000):
    """Mirror descent (exponential weights) for simplex-constrained optimization."""
    n = len(a)
    a = np.maximum(a, 1e-10).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 1.0
    best_eta = 1.0
    stall = 0
    
    for t in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find near-max positions (within 1e-4 of max)
        threshold = max_val - 1e-4 * max_val
        max_positions = np.where(conv >= threshold)[0]
        
        # Subgradient: for each max position k, the gradient w.r.t. a[j] is a[k-j]
        grad = np.zeros(n)
        for k in max_positions:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                grad[i] += a[k - i]
        grad /= len(max_positions)
        
        # Project gradient to tangent space of simplex (sum a = 1)
        # The constraint is sum a = 1, so we subtract the weighted average
        grad_proj = grad - np.sum(a * grad)
        
        # Mirror descent update
        a_new = a * np.exp(-eta * grad_proj)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            best_eta = eta
            # Slightly increase eta
            eta = min(eta * 1.01, 10.0)
        else:
            stall += 1
            if stall > 50:
                # Restart with smaller eta
                eta = max(0.1, best_eta * 0.5)
                a = best_a * np.exp(np.random.default_rng(t).normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
            else:
                a = best_a  # Roll back
                eta *= 0.95
    
    return best_a
