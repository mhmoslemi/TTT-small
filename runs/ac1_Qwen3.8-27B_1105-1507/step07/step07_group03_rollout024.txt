def spectral_optimize(a, n_iters, time_budget, relax_schedule='decay'):
    """Improved spectral (Gerchberg-Saxton) optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute spectrum
        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-15)
        phase = A / mag_safe
        
        # Target: flat magnitude = 1 (since we normalized sum to 1, DC = 1)
        # Actually, for a normalized sequence (sum=1), the DC component is 1.
        # The target flat spectrum should have |A(0)| = 1 and |A(f)| = sqrt(n) for f != 0?
        # No, let's just use uniform target.
        target_mag = np.ones(L)
        target_mag[0] = 1.0  # DC is fixed by normalization
        # Actually, let's use target_mag = sqrt(L/n) * ones(L) to match the energy
        # sum |A(f)|^2 = L * sum a[i]^2 (by Parseval, with the convention used)
        # For flat spectrum: |A(f)|^2 = sum|A|^2 / L = L * sum(a^2) / L = sum(a^2)
        # So target_mag = sqrt(sum(a^2))
        # But this might not be achievable with a >= 0
        # Let's just use ones for simplicity
        target_mag = np.ones(L)
        
        # Adaptive relaxation
        if relax_schedule == 'decay':
            t = it / max(1, n_iters)
            relax = 0.8 * (1 - t) + 0.2 * t  # Start at 0.8, end at 0.2
        else:
            relax = 0.5
        
        A_new = (1 - relax) * A + relax * (target_mag * phase)
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    # Perturb and reset
                    rng = np.random.default_rng(hash((it, n)) % (2**32))
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
                else:
                    a = best_a.copy()
    
    return best_a

def subgradient_optimize(a, n_iters, time_budget):
    """Subgradient optimization with adaptive temperature and backtracking."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    beta = 0.8
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient at y
        conv_y = compute_conv(y)
        max_val = float(np.max(conv_y))
        
        # Get all peaks above threshold
        threshold = 0.9 * max_val
        peak_indices = np.where(conv_y >= threshold)[0]
        
        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(conv_y))])
        
        # Adaptive temperature
        temp = max_val * 0.1 * (1 + 0.5 * np.exp(-it / 100))
        
        # Compute weights
        peak_vals = conv_y[peak_indices]
        weights = np.exp((peak_vals - max_val) / temp)
        weights /= weights.sum()
        
        # Compute weighted subgradient
        # grad[i] = sum_k w_k * y[k-i] (for valid k-i)
        # This is a cross-correlation
        grad = np.zeros(n)
        for wi, ki in zip(weights, peak_indices):
            ki = int(ki)
            # For this peak at position ki:
            # grad[i] += wi * y[ki - i] for 0 <= ki - i < n
            # => i ranges from max(0, ki-n+1) to min(ki, n-1)
            i_start = max(0, ki - n + 1)
            i_end = min(ki, n - 1) + 1
            if i_end > i_start:
                grad[i_start:i_end] += wi * y[ki - (i_end - 1):ki - (i_start - 1)][::-1]
        
        # Backtracking line search
        momentum = beta * v + (1 - beta) * grad
        alpha = 0.1
        for _ in range(10):
            y_new = y - alpha * momentum
            y_new = np.maximum(y_new, 0)
            s_new = float(np.sum(y_new))
            if s_new > 1e-10:
                y_new = y_new / s_new
                obj_new = compute_obj(y_new)
                if obj_new < best_obj:
                    break
            alpha *= 0.5
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = y_new.copy()
        
        # Update momentum
        v = momentum * beta  # simplify
        y = y_new
        
        # Decay momentum
        beta = max(0.5, beta * 0.999)
    
    return best_a

def lp_polish(a, time_budget):

import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix B (2n-1 x n) such that B @ g = conv(a, g)."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        B[k_start:k_end, i] = a[:k_end - k_start]
    return B


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g (normalized) or None."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    B = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [B @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.MOSEK]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([B, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def spectral_optimize(a, n_iters, time_budget):
    """Gerchberg-Saxton style spectral flattening with adaptive relaxation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-15)
        phase = A / mag_safe

        # Adaptive relaxation: start aggressive, become conservative
        t_frac = float(it) / max(1, n_iters)
        relax = 0.7 * (1.0 - t_frac) + 0.15 * t_frac

        target_mag = np.ones(L)
        A_new = (1.0 - relax) * A + relax * (target_mag * phase)

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 25:
                    seed_val = (it * 7919 + n * 104729) % (2**31)
                    rng = np.random.default_rng(seed_val)
                    a = best_a * np.exp(rng.normal(0, 0.06, n))
                    a = np.maximum(a, 1e-10)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
                else:
                    a = best_a.copy()
        else:
            a = best_a.copy()

    return best_a


def subgradient_optimize(a, n_iters, time_budget):
    """Mirror descent with softmax-weighted multi-peak subgradient."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.1
    beta = 0.82
    v = np.zeros(n)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_y = compute_conv(a)
        max_val = float(np.max(conv_y))

        # Get peaks above adaptive threshold
        threshold = 0.85 * max_val
        peak_mask = conv_y >= threshold
        peak_indices = np.nonzero(peak_mask)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(conv_y))])
        elif len(peak_indices) > 50:
            # Keep only top 50
            peak_vals = conv_y[peak_indices]
            top_k = np.argsort(peak_vals)[-50:]
            peak_indices = peak_indices[top_k]

        # Adaptive softmax temperature
        peak_vals = conv_y[peak_indices]
        spread = float(peak_vals[-1] - peak_vals[0]) if len(peak_vals) > 1 else 0.01
        temp = max(0.05 * max_val, 0.3 * spread)
        weights = np.exp((peak_vals - peak_vals[-1]) / temp)
        weights /= weights.sum()

        # Compute weighted subgradient: grad[j] = sum_k w_k * a[k-j]
        grad = np.zeros(n)
        for wi, ki in zip(weights, peak_indices):
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                idx = ki - js
                grad[j_start:j_end + 1] += wi * a[idx]

        # Nesterov momentum
        v = beta * v + (1.0 - beta) * grad

        # Backtracking line search for mirror descent step
        best_alpha = alpha
        improved = False
        for _ in range(8):
            a_try = a * np.exp(-alpha * v)
            a_try = np.maximum(a_try, 1e-15)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_alpha = alpha
                    improved = True
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                    a = a_try
                    break
            alpha *= 0.5

        if not improved:
            alpha *= 0.7

        # Periodically boost alpha to escape
        if it % 500 == 499:
            alpha = min(alpha * 1.3, 0.3)

        # Decay momentum slightly
        beta = max(0.6, beta * 0.9995)

    return best_a


def lp_polish(a, time_budget):
    """LP-based polishing with adaptive line search and restarts."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-
10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp(a)
        if g_lp is None:
            break

        # Adaptive line search
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0 < last_t < 1:
            extra = [max(0.01, last_t * 0.6), max(0.01, last_t * 0.9), min(1.0, last_t * 1.3)]
            t_candidates = sorted(set(t_candidates + [t for t in extra if 0 < t <= 1.0]))

        best_t_obj = np.inf
        best_t_val = 0.1
        best_a_local = a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 1e-15)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
        else:
            stall += 1
            if stall > 8:
                seed_val = (it * 31 + n * 97) % (2**31)
                rng = np.random.default_rng(seed_val)
                scale = 0.02 + 0.005 * min(stall // 8, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-10)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def mirror_descent_refine(a, n_iters, time_budget):
    """Pure mirror descent (no momentum) with careful step sizing."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.15

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        top_k = 10
        top_idx = np.argsort(conv)[-top_k:]
        top_vals = conv[top_idx]

        temp = max(0.03 * max_val, 0.2 * float(top_vals[-1] - top_vals[0]))
        weights = np.exp((top_vals - top_vals[-1]) / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                idx = ki - js
                grad[j_start:j_end + 1] += wi * a[idx]

        # Mirror descent update
        a_new = a * np.exp(-alpha * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.95
        else:
            alpha *= 0.5

        if it % 300 == 299:
            alpha = min(alpha * 1.2, 0.25)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        alloc = min(remaining * 0.25, 50.0)
        a_f4 = mirror_descent_refine(best_seq, 30000, alloc)
        obj = compute_obj(a_f4)
        status = " **" if obj < best_score else ""
        print(f"  Final MD: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_f4.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
