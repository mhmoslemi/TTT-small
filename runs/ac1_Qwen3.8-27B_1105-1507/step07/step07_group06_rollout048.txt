def compute_c_vector(a, w):
    """
    Compute c[m] = sum_j a[j] * w[m+j] for m=0..n-1 using FFT.
    a: length n, w: length 2n-1.
    """
    n = len(a)
    w_rev = w[::-1].copy()
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    A = rfft(a, N)
    W_rev = rfft(w_rev, N)
    conv_aw = irfft(A * W_rev, N)[:3 * n - 2]
    c = conv_aw[n - 1:2 * n - 1][::-1]
    return c


def smooth_obj_grad_theta(theta, tau, n):
    """
    Compute smoothed objective and gradient w.r.t. log-parameters theta.
    a = softmax(theta), obj = 2n * smooth_max(conv(a), tau).
    Returns (obj_val, grad_theta).
    """
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    a = exp_t / np.sum(exp_t)

    conv = compute_conv(a)
    conv_max = np.max(conv)
    exp_vals = np.exp(tau * (conv - conv_max))
    Z = np.sum(exp_vals)
    smooth_max = conv_max + np.log(Z) / tau
    obj_val = 2.0 * n * smooth_max

    w = tau * exp_vals / Z

    c = compute_c_vector(a, w)
    mean_c = float(np.dot(c, a))
    grad_theta = 4.0 * n * a * (c - mean_c)

    return obj_val, grad_theta

def adam_smoothed(a_init, time_budget, tau_stages, lr_start=0.05, lr_end=0.001,
                  beta1=0.9, beta2=0.999, eps=1e-8, max_iter=500):
    """
    Adam optimization with smoothed objective and tau annealing.
    Uses log-softmax parameterization.
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0

    for tau in tau_stages:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        stage_start = time.monotonic()
        stage_iter = 0
        no_improve = 0
        best_obj_stage = best_obj

        while True:
            elapsed = time.monotonic() - start
            stage_elapsed = time.monotonic() - stage_start
            if remaining < 5 or stage_elapsed > max(remaining * 0.6, 2.0):
                break
            if stage_iter > max_iter:
                break

            t += 1
            stage_iter += 1

            obj_val, grad = smooth_obj_grad_theta(theta, tau, n)

            lr = lr_start * (lr_end / lr_start) ** (t / max(1000, max_iter * len(tau_stages)))
            lr = max(lr, 1e-5)

            m_adam = beta1 * m_adam + (1 - beta1) * grad
            v_adam = beta2 * v_adam + (1 - beta2) * grad * grad
            m_hat = m_adam / (1 - beta1 ** t)
            v_hat = v_adam / (1 - beta2 ** t)
            theta -= lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = np.exp(theta - np.max(theta))
            a_new = a_new / np.sum(a_new)

            obj_new = compute_obj(a_new)
            if np.isfinite(obj_new) and obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 50:
                # Perturb to escape
                rng = np.random.default_rng(t * 7 + n)
                theta += rng.normal(0, 0.02, n)
                no_improve = 0

    return best_a, best_obj


def adam_direct(a_init, time_budget, tau_stages, lr_start=0.02, lr_end=0.001,

            v_adam = beta2 * v_adam + (1 - beta2) * grad_proj * grad_proj
            m_hat = m_adam / (1 - beta1 ** t)
            v_hat = v_adam / (1 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)

                a = a / np.sum(a)
                m_adam *= 0.5

                no_improve = 0

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            if it % 5 == 0:
                rng = np.random.default_rng(it * 104729 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s

    return best_a


def peak_shift_heuristic(a, time_budget=30.0):
    """
    Peak-shifting heuristic: identify peak contributors and move mass away.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        # conv[k_star] = sum_j a[j]*a[k_star - j]
        contributors = []
        for j in range(max(0, k_star - n + 1), min(k_star + 1, n)):
            other = k_star - j
            if 0 <= other < n:
                contributors.append((j, other, a[j] * a[other]))
        contributors.sort(key=lambda x: -x[2])

        # Try moving mass from top contributor to reduce peak
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.15]:
            if improved:
                break
            for j, other, _ in contributors[:5]:
                src = j if a[j] >= a[other] else other
                # Find a "safe" destination: position with low a that's far from peak
                dest_candidates = []
                for d in range(n):
                    if d == src:
                        continue
                    # Check if dest would contribute to peak
                    if 0 <= k_star - d < n:
                        continue  # would contribute to same peak
                    dest_candidates.append((d, a[d]))
                if not dest_candidates:
                    dest_candidates = [(d, a[d]) for d in range(n) if d != src]
                dest_candidates.sort(key=lambda x: x[1])

                for dest, _ in dest_candidates[:3]:
                    delta = frac * a[src]
                    if delta >= a[src] * 0.9:
                        continue
                    a[src] -= delta
                    a[dest] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[dest] -= delta
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random exploration
            if it % 10 == 0:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    delta = 0.01 * np.max(a) * rng.uniform(0.1, 1.0)
                    src = i if a[i] > a[j] else j
                    dest = j if a[i] > a[j] else i
                    if delta < a[src] * 0.5:
                        a[src] -= delta
                        a[dest] += delta
                        a = np.maximum(a, 0)
                        obj_new = compute_obj(a)
                        if obj_new < best_obj - 1e-13:
                            best_obj = obj_new
                            best_a = a.copy()
                            improved = True
                        else:
                            a[src] += delta
                            a[dest] -= delta
            if not improved and it > 200:
                break

    return best_a


def subgradient_method(a_init, time_budget=60.0):
    """
    Subgradient method on the simplex with adaptive step size.
    Uses exact subgradient of max(conv).
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    step_size = 0.1
    best_obj_run = best_obj
    stuck = 0

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[m] = 2*a[k_star - m] for valid k_star - m
        grad = np.zeros(n)
        for m in range(n):
            idx = k_star - m
            if 0 <= idx < n:
                grad[m] = 2.0 * a[idx]

        # Project to simplex tangent
        grad_proj = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_proj)
        if grad_norm < 1e-12:
            break

        step = step_size / grad_norm
        a_new = a - step * grad_proj
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        obj_new = compute_obj(a_new)
        if np.isfinite(obj_new) and obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            step_size = min(step_size * 1.05, 0.5)
            stuck = 0
        else:
            stuck += 1
            step_size *= 0.95

        if stuck > 50:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-10)
            a = a / np.sum(a)
            step_size = 0.1
            stuck = 0

        if step_size < 1e-4:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length using interp."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining Adam, LP alternating, peak-shifting,
    subgradient method, population diversification, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Adam (log-softmax) from parent ===
    print("\n=== Phase 1: Adam log-softmax ===")
    for trial in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if trial == 0:
            a_init = best_seq.copy()
        elif trial == 1:
            a_init = best_seq * np.exp(rng.normal(0, 0.05, n_parent))
            a_init = np.maximum(a_init, 0)
            a_init = a_init / np.sum(a_init)
        else:
            shift = rng.uniform(0.1, 0.3)
            ramp = np.linspace(1 - shift, 1 + shift, n_parent)
            a_init = best_seq * ramp
            a_init = np.maximum(a_init, 0)
            a_init = a_init / np.sum(a_init)

        alloc = min(remaining * 0.3, 120.0)
        tau_stages = [10.0, 30.0, 100.0, 300.0, 800.0]
        a_opt, obj_opt = adam_smoothed(a_init, time_budget=alloc,
                                        tau_stages=tau_stages,
                                        lr_start=0.05, lr_end=0.001)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_parent = len(best_seq)
            print(f"  Trial {trial}: obj={best_score:.8f} **")
        else:
            print(f"  Trial {trial}: obj={obj_opt:.8f}")

    # === Phase 2: Adam (direct simplex) ===
    print("\n=== Phase 2: Adam direct simplex ===")
    remaining = deadline - time.monotonic()

        alloc = min(remaining * 0.2, 80.0)

            best_seq = a_opt.copy()

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
