**Another new idea: "Multiplicative weight update" (like in game theory / online learning)**

The idea from the "multiplicative weights" algorithm:

- For each iteration, identify the "most violating" constraint (peak lag)
- Update: a_new[i] = a[i] * exp(-eta * grad[i]) / Z

Peak-shaving + LP polishing + multiplicative updates for minimizing
2*n*max(conv(a,a))/(sum(a))^2 with non-negative a.
Key innovations: O(n) gradient peak shaving with top-k peaks, multiplicative weight
updates preserving non-negativity, fast LP via scipy linprog, and adaptive method switching.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def compute_obj_and_conv(a):
    """Compute objective and convolution together."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, None
    a_norm = a / s
    b = compute_conv(a_norm)
    obj = 2.0 * n * float(np.max(b))
    return obj, b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver_name in ['ECOS', 'SCS', 'CLARABEL']:
            try:
                import cvxpy as cp2
                solver = getattr(cp2, solver_name, None)
                if solver is None:
                    continue
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s2 = float(np.sum(g_val))
                    if s2 > 1e-10:
                        return g_val / s2
            except Exception:
                continue
    except Exception:
        pass
    return None


def linprog_lp(c, A_ub, b_ub, A_eq, b_eq, bounds):
    """Wrapper around scipy linprog."""
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:len(c) - 1]
    return None


def peak_shaving(a, time_budget=60.0, top_k=3, verbose=False):
    """
    Fast peak shaving via subgradient descent on top-k peaks.
    Gradient of b[k*] w.r.t. a[i] = 2*a[k*-i] if 0<=k*-i<n else 0.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    eta = 0.05
    momentum = 0.0
    v = np.zeros(n)
    stall = 0

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution to find peaks
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Get top-k peak lags
        if top_k == 1:
            peak_indices = [int(np.argmax(b))]
        else:
            top_vals = np.sort(b)[::-1][:top_k]
            threshold = top_vals[-1] * 0.8
            peak_indices = [int(i) for i in np.where(b >= threshold)[0]][:top_k]
            if not peak_indices:
                peak_indices = [int(np.argmax(b))]

        # Combined gradient
        grad = np.zeros(n)
        for k_star in peak_indices:
            ks = int(k_star)
            # grad[i] += 2*a[ks-i] for valid i
            i_min = max(0, ks - (n - 1))
            i_max = min(n - 1, ks)
            for i in range(i_min, i_max + 1):
                j = ks - i
                grad[i] += 2.0 * a[j]

        # Project to sum=0
        grad -= np.mean(grad)

        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        # Evaluate
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 0.5)
            momentum = min(momentum + 0.002, 0.8)
        else:
            stall += 1
            eta *= 0.96
            v *= 0.3
            if stall > 300:
                # Perturb and restart
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0.0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.05
                momentum = 0.0
                stall = 0
            if eta < 1e-9:
                break

        if verbose and it % 5000 == 0:
            print(f"            PS it{it}: obj={best_obj:.8f}, eta={eta:.5f}, stall={stall}")

    return best_a


def multiplicative_reduction(a, time_budget=45.0, verbose=False):
    """
    Multiplicative weight update: a_new[i] = a[i] * exp(-eta * grad[i]).
    Naturally preserves non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 1e-12).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    eta = 0.3
    stall = 0

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))

        # Gradient of peak
        grad = np.zeros(n)
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        for i in range(i_min, i_max + 1):
            j = k_star - i
            grad[i] = 2.0 * a[j]
        grad -= np.mean(grad)

        # Multiplicative update
        factor = np.exp(-eta * grad)
        a_new = a * factor
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.003, 2.0)
        else:
            stall += 1
            eta *= 0.97
            if stall > 200:
                # Perturb
                rng_local = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng_local.normal(0, 0.05, n))
                a = np.maximum(a, 1e-12)
                a = a / np.sum(a)
                eta = 0.3
                stall = 0
            if eta < 1e-6:
                break

        if verbose and it % 5000 == 0:
            print(f"            MU it{it}: obj={best_obj:.8f}, eta={eta:.4f}")

    return best_a


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0.0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 10:
                break

    return best_a


def coordinate_descent_peak(a, time_budget=30.0):
    """
    Coordinate descent: for each i, find optimal a[i] that minimizes max(conv).
    Since b[k] is quadratic in a[i], we can solve a 1D problem.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute: for each coordinate i, which lags does it affect?
    # b[k] = sum_j a[j]*a[k-j]. a[i] appears in b[k] when j=i (k-i valid) or k-j=i (k-i valid).
    # So a[i] affects b[k] for k in [i, i+n-1] (as first factor) and k in [i, i+n-1] (as second factor).
    # Actually b[k] involves a[i] if 0<=i<n and 0<=k-i<n, i.e., i <= k <= i+n-1.

    for sweep in range(2000):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                return best_a

            # Try different values of a[i]
            # b[k] = a[i]*a[k-i] + (other terms involving a[i] as second factor: a[k-i]*a[i])
            # = 2*a[i]*a[k-i] + rest[k] for k where both i and k-i are valid
            # Actually: b[k] includes a[i]*a[k-i] if 0<=k-i<n, and a[k-i]*a[i] if 0<=k-i<n
            # These are the same term counted once (since multiplication is commutative in the sum)
            # Wait no: b[k] = sum_j a[j]*a[k-j]. The term with j=i is a[i]*a[k-i].
            # The term with k-j=i, i.e., j=k-i, is a[k-i]*a[i].
            # If i != k-i (i.e., 2i != k), these are two different terms in the sum.
            # If 2i = k, it's a single term a[i]^2.

            # For a fixed a[i] = x, b[k] = x*(a[k-i] + a[k-i]) + rest = 2*x*a[k-i] + rest
            # (when 0<=k-i<n and k-i != i)
            # or b[k] = x^2 + rest (when k-i == i)

            # This is getting complex. Let's just do a simple line search.
            current_val = a[i]
            # Try: current, current*0.5, current*1.5, current + 0.01/n, current - 0.01/n
            candidates = [current_val, current_val * 0.5, current_val * 1.5,
                          max(0, current_val - 0.005 / n), current_val + 0.005 / n]
            # Also try 0
            candidates.append(0.0)

            best_candidate = current_val
            best_candidate_obj = compute_obj(a)

            for cv in candidates:
                if cv < 0 or cv > 10:
                    continue
                a[i] = cv
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = current_val
                    continue
                a_norm = a / s_new
                obj_try = compute_obj(a_norm)
                if obj_try < best_candidate_obj:
                    best_candidate_obj = obj_try
                    best_candidate = cv

            a[i] = best_candidate
            if abs(best_candidate - current_val) > 1e-10:
                improved = True

        # Renormalize after full sweep
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()

        if not improved:
            break

    return best_a


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Two distant peaks
        p1, p2 = n // 5, 4 * n // 5
        a = np.zeros(n)
        a[p1] = 1.0
        a[p2] = 1.0
        a = a + rng.uniform(0, 0.3, n)
        a = a / np.sum(a)
    elif mode == 6:
        # Exponential decay from center
        center = n // 2
        a = np.exp(-np.abs(np.arange(n) - center) / (n * 0.2))
        a = a / np.sum(a)
    else:
        # Random smooth
        a = rng.uniform(0, 1, 5)
        x = np.linspace(0, 1, 5)
        a_interp = np.interp(np.linspace(0, 1, n), x, a)
        a = a_interp + 0.01
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    # Use numpy interp (linear) - good enough
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def fft_gradient_descent(a, time_budget=60.0, mode='energy', verbose=False):
    """FFT-based gradient descent (energy or peak mode)."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    stall = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        if mode == 'energy':
            # Gradient of sum(b^2) w.r.t. a[i]
            a_flip = a[::-1]
            faf = rfft(a_flip, N)
            fb = rfft(b, N)
            c_full = irfft(faf * fb, N)[:3 * n - 2]
            c = c_full[n - 1:2 * n - 1]
        else:  # peak
            k_star = int(np.argmax(b))
            c = np.zeros(n)
            i_min = max(0, k_star - (n - 1))
            i_max = min(n - 1, k_star)
            for i in range(i_min, i_max + 1):
                c[i] = 2.0 * a[k_star - i]

        c = c - np.mean(c)
        v = momentum * v - eta * c
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.005, 3.0)
            momentum = min(momentum + 0.001, 0.92)
        else:
            stall += 1
            eta *= 0.97
            v *= 0.5
            if stall > 300:
                rng_local = np.random.default_rng(it + 999)
                a = best_a * np.exp(rng_local.normal(0, 0.04, n))
                a = np.maximum(a, 0.0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.7
                stall = 0
            if eta < 1e-8:
                break

        if verbose and it % 5000 == 0:
            print(f"            GD({mode}) it{it}: obj={best_obj:.8f}")

    return best_a


def spectral_shape(a, time_budget=30.0):
    """
    IFTA-style: try to flatten the power spectrum to reduce time-domain peaks.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    alpha = 0.7
    stall = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        from scipy.fft import fft, ifft
        fa = fft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Generalized flattening: mag_new = mag * (1-alpha) + alpha * mag_mean
        mag_mean = np.mean(mag)
        mag_new = (1 - alpha) * mag + alpha * mag_mean

        # Also try raising the floor
        mag_floor = np.percentile(mag, 20)
        mag_new = np.maximum(mag_new, mag_floor * 0.5)

        fa_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(fa_new, N))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            alpha *= 0.9
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            alpha = max(alpha * 0.95, 0.1)
            if stall > 500:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: peak shaving + LP + multiplicative + coordinate descent + multi-n.
    Starts from parent sequence, applies multiple optimization methods in sequence,
    explores different n values, and uses random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization with n={n_best}, obj={best_score:.8f}")

    def remaining_time():
        return max(0, deadline - time.monotonic())

    # === Phase 1: Peak Shaving (top-k=3) ===
    print(f"\n=== Phase 1: Peak Shaving top-3 ===")
    if remaining_time() > 30:
        alloc = min(remaining_time() * 0.15, 120.0)
        a_ps = peak_shaving(best_seq, time_budget=alloc, top_k=3, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  ** PS: {best_score:.8f} **")
        else:
            print(f"  PS: {obj_ps:.8f}")

    # === Phase 2: Multiplicative Reduction ===
    print(f"\n=== Phase 2: Multiplicative Reduction ===")
    if remaining_time() > 20:
        alloc = min(remaining_time() * 0.1, 60.0)
        a_mu = multiplicative_reduction(best_seq, time_budget=alloc, verbose=True)
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            print(f"  ** MU: {best_score:.8f} **")
        else:
            print(f"  MU: {obj_mu:.8f}")

    # === Phase 3: LP Alternating ===
    print(f"\n=== Phase 3: LP Alternating ===")
    if remaining_time() > 30:
        alloc = min(remaining_time() * 0.2, 180.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: {best_score:.8f} **")
        else:
            print(f"  LP: {obj_lp:.8f}")

    # === Phase 4: Peak Shaving again (top-1, finer) ===
    print(f"\n=== Phase 4: Peak Shaving top-1 (fine) ===")
    if remaining_time() > 20:
        alloc = min(remaining_time() * 0.1, 60.0)
        a_ps2 = peak_shaving(best_seq, time_budget=alloc, top_k=1)
        obj_ps2 = compute_obj(a_ps2)
        if obj_ps2 < best_score:
            best_score = obj_ps2
            best_seq = a_ps2.copy()
            print(f"  ** PS fine: {best_score:.8f} **")
        else:
            print(f"  PS fine: {obj_ps2:.8f}")

    # === Phase 5: FFT GD (energy) ===
    print(f"\n=== Phase 5: FFT GD Energy ===")
    if remaining_time() > 20:
        alloc = min(remaining_time() * 0.12, 80.0)
        a_gd = fft_gradient_descent(best_seq, time_budget=alloc, mode='energy')
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  ** GD: {best_score:.8f} **")
        else:
            print(f"  GD: {obj_gd:.8f}")

    # === Phase 6: Spectral Shaping ===
    print(f"\n=== Phase 6: Spectral Shaping ===")
    if remaining_time() > 15:
        alloc = min(remaining_time() * 0.05, 30.0)
        a_ss = spectral_shape(best_seq, time_budget=alloc)
        obj_ss = compute_obj(a_ss)
        if obj_ss < best_score:
            best_score = obj_ss
            best_seq = a_ss.copy()
            print(f"  ** SS: {best_score:.8f} **")
        else:
            print(f"  SS: {obj_ss:.8f}")

    # === Phase 7: Peak Shaving + MU combo ===
    print(f"\n=== Phase 7: PS + MU combo ===")
    if remaining_time() > 30:
        alloc1 = min(remaining_time() * 0.08, 50.0)
        a_c1 = peak_shaving(best_seq, time_budget=alloc1, top_k=5)
        obj_c1 = compute_obj(a_c1)
        if obj_c1 < best_score:
            best_score = obj_c1
            best_seq = a_c1.copy()

        if remaining_time() > 20:
            alloc2 = min(remaining_time() * 0.06, 40.0)
            a_c2 = multiplicative_reduction(best_seq, time_budget=alloc2)
            obj_c2 = compute_obj(a_c2)
            if obj_c2 < best_score:
                best_score = obj_c2
                best_seq = a_c2.copy()
                print(f"  ** Combo: {best_score:.8f} **")

    # === Phase 8: Multi-n exploration ===
    print(f"\n=== Phase 8: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        n_best - 100, n_best - 200,
        600, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 3000 and n != n_best]
    n_options = n_options[:8]

    for n_try in n_options:
        if remaining_time() < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick peak shaving on new n
        alloc = min(remaining_time() * 0.08, 45.0)
        a_opt = peak_shaving(a_try, time_budget=alloc, top_k=3)
        obj_try = compute_obj(a_opt)

        # If promising, do LP polish
        if obj_try < best_score + 0.05 and remaining_time() > 30:
            alloc2 = min(remaining_time() * 0.05, 30.0)
            a_opt2 = lp_alternating(a_opt, time_budget=alloc2)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 9: Random restarts ===
    print(f"\n=== Phase 9: Random Restarts ===")
    n_curr = len(best_seq)
    n_restarts = max(2, int(remaining_time() / 50.0))
    n_restarts = min(n_restarts, 6)

    for r in range(n_restarts):
        if remaining_time() < 25:
            break

        a_r = random_init(n_curr, rng)
        alloc = min(remaining_time() * 0.1, 40.0)
        a_opt = peak_shaving(a_r, time_budget=alloc, top_k=3)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r}: {best_score:.8f} **")
        else:
            print(f"  Restart {r}: {obj_r:.8f}")

    # === Phase 10: Final polish ===
    print(f"\n=== Phase 10: Final Polish ===")
    if remaining_time() > 60:
        # PS with patience
        alloc = min(remaining_time() * 0.2, 100.0)
        a_f1 = peak_shaving(best_seq, time_budget=alloc, top_k=3)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  ** Final PS: {best_score:.8f} **")

    if remaining_time() > 30:
        # LP
        alloc = min(remaining_time() * 0.2, 60.0)
        a_f2 = lp_alternating(best_seq, time_budget=alloc)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  ** Final LP: {best_score:.8f} **")

    if remaining_time() > 20:
        # MU
        alloc = min(remaining_time() * 0.15, 40.0)
        a_f3 = multiplicative_reduction(best_seq, time_budget=alloc)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  ** Final MU: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
