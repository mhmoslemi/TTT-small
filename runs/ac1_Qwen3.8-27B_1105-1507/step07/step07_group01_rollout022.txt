    fa = rfft(a, size)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))

        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

def simplex_project(v):
    n = len(v)
    s = np.sum(v)
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def spectral_init(n, rng):
    """Initialize using a flat-spectrum design."""
    # Start with random phases and take IFFT to get a real sequence
    # This gives a sequence with roughly flat power spectrum
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    phases = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    spectrum = np.ones(size // 2 + 1) * np.exp(1j * phases)
    spectrum[0] = 1.0  # DC
    # Make it conjugate symmetric
    full_spectrum = np.zeros(size, dtype=complex)
    full_spectrum[:size // 2 + 1] = spectrum
    full_spectrum[size // 2 + 1:] = np.conj(spectrum[1:(size // 2 + 1)][::-1])
    a_full = np.fft.ifft(full_spectrum).real
    a = np.abs(a_full[:n])
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

"""

Key improvements over previous: (1) CLARABEL-first LP solving for speed,
(2) multi-peak Adam gradient descent with adaptive top-k, (3) coordinate descent

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = np.sum(v)
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_and_weights(conv, T, top_k=None):
    """Compute smooth max value and softmax weights, optionally over top-k peaks."""
    if top_k is not None and top_k < len(conv):
        k_vals = np.argsort(conv)[-top_k:]
        weights_full = np.zeros_like(conv)
        vals = conv[k_vals]
        max_c = np.max(vals)
        shifted = (vals - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        w = exp_vals / total
        weights_full[k_vals] = w
        smooth_val = max_c + T * np.log(total)
        return smooth_val, weights_full
    else:
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        smooth_val = max_c + T * np.log(total)
        return smooth_val, weights


def compute_gradient(weights, a):
    """Compute gradient of smooth max w.r.t. a using convolution structure."""
    grad = np.correlate(weights, a, mode='valid')

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = ['CLARABEL', 'SCS', 'MOSEK', 'ECOS', 'PDLP', 'ECOS_BB']
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None

def multi_peak_gradient_descent(a, time_budget=60.0, T_init=0.001, T_min=0.00003, top_k=None):
    """Multi-peak gradient descent with Adam optimizer."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_obj = compute_obj(a)

    T = T_init
    lr = 0.02
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    stall = 0

    if top_k is None:
        top_k = max(5, n // 20)

    for it in range(8000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        smooth_val, weights = smooth_max_and_weights(conv, T, top_k=top_k)
        grad = compute_gradient(weights, a)
        grad_centered = grad - np.mean(grad)

        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:

                lr *= 0.8
                if lr < 1e-7:
                    break
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t_adam = 0
                stall = 0
                # Also increase top_k to broaden the search
                top_k = min(n, top_k + max(2, n // 50))

        if it > 30:
            T = max(T_min, T_init * (0.997 ** it))

    return best_a


def coordinate_descent_peak(a, time_budget=30.0):
    """Coordinate descent that directly reduces peak contributions."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Compute contribution of each a[i] to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Find the element with the least contribution (candidate to increase)
        # and the one with most contribution (candidate to decrease)
        min_idx = int(np.argmin(contrib))
        max_idx = int(np.argmax(contrib))

        if contrib[max_idx] <= contrib[min_idx]:
            continue

        # Try moving mass from max_idx to min_idx
        delta = step * a[max_idx]
        a_try = a.copy()
        a_try[max_idx] -= delta
        a_try[min_idx] += delta
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            step *= 0.8
            if step < 1e-5:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_init(n, rng):
    """Initialize using a flat-spectrum (random phase) design."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    phases = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    spectrum = np.ones(size // 2 + 1) * np.exp(1j * phases)
    spectrum[0] = 1.0
    full_spectrum = np.zeros(size, dtype=complex)
    full_spectrum[:size // 2 + 1] = spectrum
    full_spectrum[size // 2 + 1:] = np.conj(spectrum[1:(size // 2 + 1)][::-1])
    a_full = np.fft.ifft(full_spectrum).real
    a = np.abs(a_full[:n])
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def structured_init(n, rng, mode=0):
    """Generate structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        return a / np.sum(a)
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        return a / np.sum(a)
    elif mode == 3:
        # Gaussian
        x = np.arange(n)
        center = n // 2
        width = max(1, n // 4)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        return a / np.sum(a)
    elif mode == 4:
        # Two-peak
        x = np.arange(n)
        a = np.exp(-((x - n // 3) ** 2) / (2 * (n // 8) ** 2)) + \
            np.exp(-((x - 2 * n // 3) ** 2) / (2 * (n // 8) ** 2))
        return a / np.sum(a)
    elif mode == 5:
        # Random smooth
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a - np.min(a) + 0.1
        return a / np.sum(a)
    else:
        return rng.dirichlet(np.ones(n))

    Main search function: combines LP alternating, multi-peak Adam gradient,
    coordinate descent, spectral/structured initializations, and multi-length
    exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(42)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                print(f"Loaded parent: n={len(parent)}, obj={compute_obj(parent):.9f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = compute_obj(parent)
    else:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.9f}")

    # === Phase 1: Intensive LP on parent ===
    print("\n=== Phase 1: LP Alternating on Parent ===")
    a = best_seq.copy()
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        alloc = min(remaining * 0.18, 100.0)
        a_new = lp_alternating(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP Round {round_idx}: obj={best_score:.9f}")
        else:
            print(f"  LP Round {round_idx}: obj={obj:.9f} (stalled)")
            break

    # === Phase 2: Multi-peak Gradient ===
    print("\n=== Phase 2: Multi-Peak Adam Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n = len(a)
        alloc = min(remaining * 0.12, 50.0)
        a_sm = multi_peak_gradient_descent(a, time_budget=alloc, T_init=0.002, T_min=0.00005, top_k=max(10, n // 10))
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            a = a_sm.copy()
            print(f"  Multi-Peak GD: obj={best_score:.9f}")
        else:
            print(f"  Multi-Peak GD: obj={obj_sm:.9f}")

        # Second pass with lower temperature
        if remaining > 40:
            alloc = min(remaining * 0.08, 30.0)
            a_sm2 = multi_peak_gradient_descent(a, time_budget=alloc, T_init=0.0005, T_min=0.00002, top_k=max(3, n // 30))
            obj_sm2 = compute_obj(a_sm2)
            if obj_sm2 < best_score:
                best_score = obj_sm2
                best_seq = a_sm2.copy()
                a = a_sm2.copy()
                print(f"  Multi-Peak GD (low T): obj={best_score:.9f}")

    # === Phase 3: Coordinate Descent ===
    print("\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.08, 20.0)
        a_cd = coordinate_descent_peak(a, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  Coord Descent: obj={best_score:.9f}")
        else:
            print(f"  Coord Descent: obj={obj_cd:.9f}")

    # === Phase 4: LP Polish ===
    print("\n=== Phase 4: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.15, 60.0)
        a_lp = lp_alternating(a, time_budget=alloc, verbose=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  LP Polish: obj={best_score:.9f}")

    # === Phase 5: Multi-length Exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(500, n_best - 200),
        max(500, n_best - 100),
        max(500, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 300,
        n_best + 500,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 2000]
    # Remove duplicates, keep sorted
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Optimize
        alloc = min(remaining * 0.12, 45.0)
        a_opt = optimize_sequence(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = len(a_opt)
            print(f"  ** n={n_try}: obj={best_score:.9f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.9f}")

    # === Phase 6: Diverse Initializations ===
    print("\n=== Phase 6: Diverse Initializations ===")
    n_best = len(best_seq)

    for trial in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        # Vary the length and initialization
        n_try = n_best + rng.integers(-150, 151)
        n_try = max(500, min(1800, n_try))

        # Different initialization strategies
        mode = trial % 6
        a_init = structured_init(n_try, rng, mode=mode)
        if mode == 0 and trial % 2 == 1:
            a_init = spectral_init(n_try, rng)

        # Optimize
        alloc = min(remaining * 0.1, 40.0)
        a_opt = optimize_sequence(a_init, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(a_opt)
            print(f"  Init {trial} (n={n_try}, mode={mode}): obj={best_score:.9f} **")
        else:
            print(f"  Init {trial} (n={n_try}, mode={mode}): obj={obj_opt:.9f}")

    # === Phase 7: Blend best with diverse candidates ===
    print("\n=== Phase 7: Blend Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for blend_trial in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            # Generate a diverse candidate
            n_try = n_best + rng.integers(-100, 101)
            n_try = max(500, min(1500, n_try))
            a_div = spectral_init(n_try, rng)

            # Resample best to same length
            a_best_resampled = resample_sequence(best_seq, n_try)

            # Try blends
            for t in [0.2, 0.3, 0.5]:
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                a_blend = (1 - t) * a_best_resampled + t * a_div
                a_blend = np.maximum(a_blend, 0)
                s_b = float(np.sum(a_blend))
                if s_b < 1e-10:
                    continue
                a_blend = a_blend / s_b

                alloc = min(remaining * 0.15, 25.0)
                a_opt = optimize_sequence(a_blend, time_budget=alloc, verbose=False)
                obj_opt = compute_obj(a_opt)

                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    n_best = len(a_opt)
                    print(f"  Blend {blend_trial} (t={t}, n={n_try}): obj={best_score:.9f} **")

    # === Phase 8: Final Intensive Polish ===
    print("\n=== Phase 8: Final Polish ===")
    a_final = best_seq.copy()
    for polish_round in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # LP
        alloc = min(remaining * 0.2, 50.0)
        a_lp = lp_alternating(a_final, time_budget=alloc, verbose=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Polish LP {polish_round}: obj={best_score:.9f}")

        # Gradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_f = len(a_final)
            alloc2 = min(remaining * 0.15, 25.0)
            a_sm = multi_peak_gradient_descent(a_final, time_budget=alloc2,
                                               T_init=0.0005, T_min=0.00001,
                                               top_k=max(3, n_f // 25))
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                a_final = a_sm.copy()
                print(f"  Polish GD {polish_round}: obj={best_score:.9f}")

        # Coord descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc3 = min(remaining * 0.15, 15.0)
            a_cd = coordinate_descent_peak(a_final, time_budget=alloc3)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a_final = a_cd.copy()
                print(f"  Polish CD {polish_round}: obj={best_score:.9f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")

    return result
