import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))

def smoothed_gradient(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*b[k])) where b=conv(a,a)."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]
    
    # Softmax weights
    T_b = T * b
    T_b_max = np.max(T_b)
    exp_tb = np.exp(T_b - T_b_max)
    w = exp_tb / np.sum(exp_tb)
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = 2 * conv(a_rev, w_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    w_rev = w[::-1]
    
    N2 = 1
    while N2 < n + (2*n-1) - 1:
        N2 <<= 1
    
    fa_rev = rfft(a_rev, N2)
    fw_rev = rfft(w_rev, N2)
    conv_full = irfft(fa_rev * fw_rev, N2)
    grad = 2.0 * conv_full[n-1:2*n-1]
    
    return grad, b

    """Compute grad = 2 * (diag_sum of a * w) via FFT.
    grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    """
    n = len(a)
    w_rev = w[::-1]  # length 2n-1
    # Full conv length: n + (2n-1) - 1 = 3n-2
    N = 1
    while N < 3 * n - 2:

    fa = rfft(a, N)
    fw = rfft(w_rev, N)
    conv_full = irfft(fa * fw, N)[:3*n-2]
    grad = 2.0 * conv_full[n-1:2*n-1][::-1]
    return grad

- If g[i] > 0, then (A^T y)[i] = 1

"""
Smoothed Nesterov GD with FFT-based O(n log n) gradients, LP polishing, and multi-scale search.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothing with temperature annealing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

# --- Core functions ---

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = np.sum(v)
    if s < 1e-15:
        return np.ones(n) / n
    # Duchi et al. algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad_fft(a, T):
    """
    Compute gradient of f_T(a) = (1/T)*log(sum_k exp(T*conv(a,a)[k])).
    Returns (gradient, b) where b = conv(a,a).
    """
    n = len(a)
    # FFT size for conv(a,a): need >= 2n-1
    N1 = 1
    while N1 < 2 * n - 1:
        N1 <<= 1
    
    fa = rfft(a, N1)
    b = irfft(fa * fa, N1)[:2 * n - 1]
    
    # Softmax weights
    Tb = T * b
    Tb_max = np.max(Tb)
    exp_Tb = np.exp(Tb - Tb_max)
    w_sum = np.sum(exp_Tb)
    if w_sum < 1e-300:
        # Degenerate: all weight on max
        w = np.zeros(2 * n - 1)
        w[np.argmax(b)] = 1.0
    else:
        w = exp_Tb / w_sum
    
    # Gradient via FFT:
    # grad[i] = 2 * sum_j a[j] * w[i+j]
    # = 2 * conv(a, w_rev)[2n-2-i]
    w_rev = w[::-1]
    # conv(a, w_rev) has length n + (2n-1) - 1 = 3n-2
    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1
    
    fa2 = rfft(a, N2)
    fw = rfft(w_rev, N2)
    conv_full = irfft(fa2 * fw, N2)[:3 * n - 2]
    grad = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    
    return grad, b

def smoothed_gd(a, time_budget, T_init=1.0, T_end=200.0, verbose=False):
    """
    Smoothed gradient descent with momentum and temperature annealing.
    Minimizes f_T(a) with annealing T: T_init -> T_end.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Parameters
    momentum = 0.5
    eta = 0.5
    v = np.zeros(n)
    stall = 0
    
    # Precompute FFT size
    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1
    
    max_iter = 100000
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        frac = it / max_iter
        T = T_init * (T_end / T_init) ** frac  # geometric annealing
        T = max(T, 0.5)
        
        # Compute gradient at current point
        grad, b = compute_smoothed_grad_fft(a, T)
        
        # Center gradient (remove mean to preserve sum)
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 2.0)
            momentum = min(momentum + 0.002, 0.9)
        else:
            stall += 1
            eta *= 0.95
            v *= 0.5
            if stall > 100:
                # Perturbation to escape
                rng_local = np.random.default_rng(12345 + it)
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = project_simplex(a)
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.5
                stall = 0
            if eta < 1e-8:
                break
        
        if verbose and it % 500 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, T={T:.1f}, eta={eta:.4f}, mom={momentum:.3f}")
    
    return best_a

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return None
    a = a / s
    
    # Build constraint matrix A (2n-1 x n)
    # A[k,i] = a[k-i] if 0<=k-i<n, else 0
    # For linprog: minimize c^T * x s.t. A_ub @ x <= b_ub, A_eq @ x == b_eq, bounds
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M -> c = [0,...,0, 1]
    # Constraints: A @ g - M <= 0 for each k -> A_ub[k, i] = A_orig[k,i], A_ub[k, n] = -1
    #              sum(g) = 1 -> A_eq
    #              g[i] >= 0, M >= 0
    
    # Build A_orig efficiently
    A_orig = np.zeros((2 * n - 1, n))
    for i in range(n):
        # A_orig[k, i] = a[k-i] for k=i, i+1, ..., min(i+n-1, 2n-2)
        max_k = min(i + n, 2 * n - 1)
        A_orig[i:max_k, i] = a[:max_k - i]
    
    # Set up linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_orig
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_val = float(np.sum(g_val))
            if s_val > 1e-10:
                return g_val / s_val
    except Exception as e:
        pass
    
    return None

def lp_polish(a, time_budget, verbose=False):
    """Alternate between LP and GD for polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        # LP step
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search: find best t in (0, 1)
        best_t_obj = best_obj
        best_t = 0
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            # Try short GD burst
            remaining = time_budget - (time.monotonic() - start)

                a_gd = smoothed_gd(a, time_budget=min(remaining * 0.3, 10.0), T_init=50, T_end=200)

                if obj_gd < best_obj:
                    best_obj = obj_gd
                    best_a = a_gd.copy()
                    a = a_gd.copy()
                    continue
            break
    
    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def random_init(n, rng):
    """Diverse random initialization."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        sig = max(n // 15, 2)
        a = np.exp(-((np.arange(n) - c1)**2) / (2*sig**2)) + np.exp(-((np.arange(n) - c2)**2) / (2*sig**2))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.7
        a = a / np.sum(a)
    elif mode == 6:
        # Random smooth (low-pass)
        a = rng.uniform(0, 1, n)
        # Simple moving average
        k = max(n // 20, 2)
        kernel = np.ones(k) / k
        from scipy.signal import convolve
        a = convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        # Log-normal
        a = rng.lognormal(0, 0.5, n)
        a = a / np.sum(a)
    return a
