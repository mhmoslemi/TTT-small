"""
Optimized step function search combining softmax-approximated gradient descent,
Fourier-mode parameterization, LP alternating, and diverse random restarts.
Target: minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective value (normalized)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_grad(a, tau):
    """
    Compute gradient of softmax-approximated max convolution.
    Returns (gradient, probability vector p, convolution).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    exp_c = np.exp(tau * (conv - max_c))
    p = exp_c / np.sum(exp_c)
    # grad[j] = 2 * sum_{d=0}^{n-1} a[d] * p[d+j]
    # Build Toeplitz-like matrix P where P[d,j] = p[d+j]
    idx = np.arange(n)
    P = p[idx[None, :] + idx[:, None]]  # shape (n, n), P[d,j] = p[d+j]
    grad = 2.0 * (P @ a)
    return grad, p, conv


def project_simplex(a):
    """Project onto the simplex {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_gradient_descent(a, time_budget=200.0, tau_start=10.0, tau_end=200.0,
                             verbose=False):
    """
    Nesterov-accelerated softmax gradient descent on the normalized sequence.
    Adaptively increases tau to sharpen the max approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.85
    step_size = 0.01
    x = a.copy()  # Nesterov lookahead

    it = 0
    stall_count = 0
    while time.monotonic() - start < time_budget:
        # Adaptive tau
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute gradient at lookahead point
        x_norm = project_simplex(x)
        grad, p, conv = softmax_grad(x_norm, tau)
        obj_current = compute_obj(x_norm)

        # Nesterov update
        momentum = beta * momentum - step_size * grad
        new_a = a + momentum
        new_a = project_simplex(new_a)

        # Nesterov lookahead
        t_k = 2.0 / (it + 2.0)
        x = new_a + (1 - t_k) * (new_a - a)
        x = np.maximum(x, 0)

        a = new_a
        it += 1

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall_count = 0
            # Slightly increase step

        else:
            stall_count += 1
            if stall_count > 50:
                # Restart from best with perturbation

                scale = 0.01 * min(stall_count / 10, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = project_simplex(a)
                x = a.copy()
                momentum = np.zeros(n)
                step_size = 0.01

            print(f"    SGD: it={it}, obj={best_obj:.8f}, tau={tau:.1f}, step={step_size:.4f}, t={elapsed:.0f}s")

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers

    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None

def fourier_mode_optimize(a, time_budget=100.0, n_modes=8, verbose=False):
    """
    Optimize in a low-dimensional Fourier parameterization.

    The anti-symmetric sine modes naturally reduce center convolution peak.
    """

    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Build basis functions (anti-symmetric sines)
    j = np.arange(n)
    basis = np.zeros((n_modes, n))
    for m in range(n_modes):
        basis[m] = np.sin((2 * m + 1) * np.pi * (j + 0.5) / n)

    # Optimize coefficients c
    c = np.zeros(n_modes)
    lr = 0.01
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Compute current a from parameters
        a_current = a * (1.0 + basis @ c)
        a_current = np.maximum(a_current, 0)
        sc = float(np.sum(a_current))
        if sc < 1e-10:
            break
        a_current = a_current / sc

        obj = compute_obj(a_current)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_current.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                c = rng.normal(0, 0.05, n_modes)
                stall = 0
                lr = 0.01
                continue

        # Compute gradient w.r.t. c
        conv = compute_conv(a_current)
        max_c = np.max(conv)
        # Use top peaks for subgradient
        k_max = np.argmax(conv)
        # grad_obj w.r.t. a[j] = 2 * a[k_max - j] (if valid)
        grad_a = np.zeros(n)
        for dm in range(-min(k_max, n - 1), min(2 * n - 1 - k_max, n)):
            jj = k_max - dm
            if 0 <= jj < n:
                grad_a[jj] += 2.0 * a_current[dm]
        # Actually: d(conv[k_max])/d(a[j]) = 2*a[k_max-j] if 0<=k_max-j<n
        grad_a = np.zeros(n)
        for jj in range(n):
            kk = k_max - jj
            if 0 <= kk < n:
                grad_a[jj] = 2.0 * a_current[kk]

        # Chain rule: d(a_current)/d(c_m) = a_basis[m] / sc (approximately, ignoring projection)
        grad_c = np.zeros(n_modes)
        for m in range(n_modes):
            grad_c[m] = np.sum(grad_a * basis[m] * a) / sc

        c -= lr * grad_c
        # Clip to avoid making a negative
        for m in range(n_modes):
            c[m] = np.clip(c[m], -0.5, 0.5)
        lr = max(lr * 0.999, 0.001)

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse random initializations."""
    choice = rng.integers(0, 5)
    if choice == 0:
        # Uniform with noise
        a = np.ones(n) * np.exp(rng.normal(0, 0.1, n))
    elif choice == 1:
        # Anti-symmetric perturbation of uniform
        j = np.arange(n)
        a = 1.0 + 0.3 * np.sin(np.pi * (j + 0.5) / n)
        a = np.maximum(a, 0.01)
    elif choice == 2:
        # Two-lobe
        center = n / 2
        a = 1.0 + 0.3 * np.exp(-2.0 * ((j - center * 0.3) ** 2) / n) - 0.3 * np.exp(-2.0 * ((j - center * 1.7) ** 2) / n)
        a = np.maximum(a, 0.01)
    elif choice == 3:
        # Power law decay from center
        center = (n - 1) / 2.0
        dist = np.abs(j - center) / (n / 2)
        a = 1.0 + 0.5 * (1.0 - dist) ** 2
    else:
        # Random smooth
        coeff = rng.normal(0, 0.1, min(n, 20))
        j = np.arange(n)
        a = np.ones(n)
        for m, c in enumerate(coeff):
            a += c * np.sin((2 * m + 1) * np.pi * (j + 0.5) / n)
        a = np.maximum(a, 0.01)

    a = a / np.sum(a)
    return a

            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    if remaining > 15:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 30.0)
        a_cd = polish_coordinate(a_final, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a_final = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_cd:.8f}")

    if remaining > 15:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.4, 30.0)
        a_last = softmax_gradient_descent(a_final, time_budget=alloc,
                                          tau_start=200.0, tau_end=500.0)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_last:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
