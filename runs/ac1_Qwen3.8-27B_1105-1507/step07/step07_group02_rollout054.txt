import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    a_norm = a / S
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses LP alternating (fictitious play), exact coordinate descent, and multi-start
exploration with multiple sequence lengths.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    a_norm = a / S
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=20.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns g/sum(g) or None."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    return None

def coord_descent(a, max_passes=100, time_budget=60.0):
    """
    Coordinate descent: for each coordinate, find the best value by evaluating
    at strategically chosen points (0, current, and break points).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    conv = compute_conv(a)
    conv = np.maximum(conv, 0)  # numerical safety

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()

    for p in range(max_passes):
        if time.monotonic() - start > time_budget:
            break

        any_improved = False

        for i in range(n):
            old_ai = a[i]
            S_base = S - old_ai

            # Affected k range: k where 0 <= k-i < n, i.e., i <= k <= i+n-1
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            # Compute c_k for affected positions
            ks = np.arange(k_min, k_max + 1)
            c_vals = np.zeros(len(ks))
            for idx in range(len(ks)):
                k = ks[idx]
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        c_vals[idx] = a[ki]  # self-term: a[i]*a[i], derivative = a[i]
                    else:
                        c_vals[idx] = 2.0 * a[ki]

            base_vals = conv[ks] - c_vals * old_ai

            # Unaffected max
            if k_min > 0 or k_max < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_min:k_max + 1] = False
                base_unaff_max = float(np.max(conv[mask]))
            else:
                base_unaff_max = 0.0

            # Generate candidate values for a[i]
            upper_bound = min(1.0, max(10.0 * max(a) / n, 5.0 / n, 2.0 * old_ai))
            candidates = [0.0, old_ai, upper_bound]

            # Add break points: where base_k1 + c_k1 * x = base_k2 + c_k2 * x
            # Only consider top few to keep it fast
            # Sort by c_vals to find adjacent crossings
            active_indices = np.where(c_vals > 1e-15)[0]
            if len(active_indices) > 1:
                # Find top contributors at x=0
                vals_at_0 = base_vals[active_indices]
                top_k = min(len(active_indices), 15)
                top_idx = active_indices[np.argsort(vals_at_0)[-top_k:]]

                for ii in range(len(top_idx)):
                    for jj in range(ii + 1, len(top_idx)):
                        i1, i2 = top_idx[ii], top_idx[jj]
                        dc = c_vals[i1] - c_vals[i2]
                        if abs(dc) > 1e-15:
                            x_cross = (base_vals[i2] - base_vals[i1]) / dc
                            if 0.0 <= x_cross <= upper_bound:
                                candidates.append(x_cross)

            # Also add some heuristic points
            candidates.append(upper_bound * 0.5)
            candidates.append(np.median(a) if len(a) > 0 else 0)

            # Evaluate all candidates
            best_ai = old_ai
            best_val = np.inf
            for cand in candidates:
                cand = max(0.0, min(upper_bound, cand))
                S_prime = S_base + cand
                if S_prime < 1e-10:
                    continue
                aff_max = float(np.max(base_vals + c_vals * cand))
                overall_max = max(aff_max, base_unaff_max)
                val = 2.0 * n * overall_max / (S_prime ** 2)
                if val < best_val:
                    best_val = val
                    best_ai = cand

            new_ai = best_ai
            if abs(new_ai - old_ai) < 1e-14:
                continue

            # Update
            delta = new_ai - old_ai
            a[i] = new_ai
            S += delta
            for idx in range(len(ks)):
                conv[ks[idx]] += c_vals[idx] * delta

            any_improved = True

        # Track best
        current_obj = 2.0 * n * float(np.max(conv)) / (S ** 2)
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()

        if not any_improved:
            break

    return best_a


def grad_subgradient_descent(a, time_budget=30.0):
    """
    Subgradient descent on the smoothed objective.
    Uses log-sum-exp smoothing of the max.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1 / n

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        conv = np.maximum(conv, 0)
        S = float(np.sum(a))

        # Smooth max with log-sum-exp (temperature)
        tau = 50.0 + 50.0 * (it / 200.0)
        shifted = conv / tau
        max_shifted = np.max(shifted)
        log_sum_exp = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))

        # Subgradient of smoothed max
        weights = np.exp(shifted - max_shifted)
        weights /= np.sum(weights)

        # Gradient of conv[k] wrt a[i]
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-8:
                continue
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        grad[i] += w * a[ki]
                    else:
                        grad[i] += w * 2.0 * a[ki]

        # Full subgradient of f = 2n * smoothed_max / S^2
        obj_smooth = 2.0 * n * tau * log_sum_exp / (S ** 2)
        full_grad = (2.0 * n / (S ** 2)) * grad - (4.0 * n * tau * log_sum_exp / (S ** 3))

        # Center gradient to preserve sum
        full_grad -= np.mean(full_grad)

        # Update
        a_new = a - lr * full_grad
        a_new = np.maximum(a_new, 0)
        S_new = float(np.sum(a_new))
        if S_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / S_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 1.0 / n)
        else:
            lr *= 0.95
            a = best_a.copy()
            if lr < 1e-8:
                rng = np.random.default_rng(it * 31 + 7)
                a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                lr = 0.1 / n

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Smooth random
        a = rng.uniform(0, 1, n)
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=max(1.0, n / 40.0))
        a = np.maximum(a, 0)
        a = a / np.sum(a) if np.sum(a) > 0 else np.ones(n) / n
    elif mode == 3:
        # Asymmetric: more mass on one side
        side = rng.integers(0, 2)
        weights = np.linspace(0.1, 1.0, n)
        if side == 1:
            weights = weights[::-1]
        a = weights / np.sum(weights)
    elif mode == 4:
        # Bimodal
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - c1) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - c2) / (n * 0.1)) ** 2)
        a = a / np.sum(a)
    else:
        # Near-constant with structure
        a = np.ones(n)
        a += 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating, coordinate descent, and
    subgradient methods with multiple initializations and sequence lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try parent sequence if available
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr
    except Exception:
        pass

    if parent is not None:
        parent_list = [float(x) for x in parent]
        score = evaluate_sequence(parent_list)
        if score < np.inf:
            best_score = score
            best_seq = parent.copy()
            print(f"Parent (n={len(parent)}): score={score:.8f}")

    # Establish baseline
    n_base = 800
    if best_seq is None or best_score > 2.0:
        seq_const = [1.0] * n_base
        score_const = evaluate_sequence(seq_const)
        if score_const < best_score:
            best_score = score_const
            best_seq = np.ones(n_base)
            print(f"Constant (n={n_base}): score={score_const:.8f}")

    if best_seq is None:
        best_seq = np.ones(n_base)
        best_score = evaluate_sequence([1.0] * n_base)

    print(f"Initial best: n={len(best_seq)}, score={best_score:.8f}")

    # Determine which n values to try
    n_best = len(best_seq)
    n_candidates = sorted(set([
        max(200, n_best - 200),
        n_best,
        n_best + 200,
        n_best + 500,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    n_candidates = [n for n in n_candidates if 100 <= n <= 2500]

    # Main optimization loop: alternate between LP and coord descent
    for n_try in n_candidates:
        elapsed = time.monotonic() - start_time
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Initialize for this n
        if n_try == n_best and best_seq is not None and len(best_seq) == n_try:
            a_init = best_seq.copy()
        elif parent is not None:
            a_init = resample_sequence(parent, n_try)
        else:
            a_init = np.ones(n_try) / n_try

        a_init = a_init / np.sum(a_init)

        print(f"\n--- Optimizing n={n_try} (remaining={remaining:.0f}s) ---")

        # Phase A: LP alternating (coarse)
        alloc_lp = min(120.0, remaining * 0.35)
        if alloc_lp > 15:
            a_lp = lp_alternating(a_init, time_budget=alloc_lp)
            ev_lp = evaluate_sequence([float(x) for x in a_lp])
            obj_lp = compute_obj(a_lp)
            print(f"  LP alt: internal={obj_lp:.8f}, eval={ev_lp:.8f}")
            if ev_lp < best_score:
                best_score = ev_lp
                best_seq = a_lp.copy()
                n_best = n_try
                a_init = a_lp.copy()

        # Phase B: Coordinate descent (fine)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc_cd = min(120.0, remaining * 0.35)
            a_cd = coord_descent(a_init, max_passes=200, time_budget=alloc_cd)
            ev_cd = evaluate_sequence([float(x) for x in a_cd])
            obj_cd = compute_obj(a_cd)
            print(f"  Coord desc: internal={obj_cd:.8f}, eval={ev_cd:.8f}")
            if ev_cd < best_score:
                best_score = ev_cd
                best_seq = a_cd.copy()
                n_best = n_try
                a_init = a_cd.copy()

        # Phase C: More LP alternating from improved point
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc_lp2 = min(60.0, remaining * 0.2)
            a_lp2 = lp_alternating(a_init, time_budget=alloc_lp2)
            ev_lp2 = evaluate_sequence([float(x) for x in a_lp2])
            print(f"  LP alt 2: eval={ev_lp2:.8f}")
            if ev_lp2 < best_score:
                best_score = ev_lp2
                best_seq = a_lp2.copy()
                n_best = n_try
                a_init = a_lp2.copy()

        # Phase D: More coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc_cd2 = min(80.0, remaining * 0.25)
            a_cd2 = coord_descent(a_init, max_passes=300, time_budget=alloc_cd2)
            ev_cd2 = evaluate_sequence([float(x) for x in a_cd2])
            print(f"  Coord desc 2: eval={ev_cd2:.8f}")
            if ev_cd2 < best_score:
                best_score = ev_cd2
                best_seq = a_cd2.copy()
                n_best = n_try

    # Random restarts with the best n
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n--- Random restarts (n={n_best}, remaining={remaining:.0f}s) ---")
        restart_time = min(30.0, remaining / 8.0)

        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < restart_time + 5:
                break

            a_r = random_init(n_best, rng)

            # Quick LP + CD
            alloc = min(restart_time, remaining * 0.12)
            a_opt = lp_alternating(a_r, time_budget=alloc * 0.5)
            a_opt = coord_descent(a_opt, max_passes=100, time_budget=alloc * 0.5)
            ev_r = evaluate_sequence([float(x) for x in a_opt])
            if ev_r < best_score:
                best_score = ev_r
                best_seq = a_opt.copy()
                n_best = len(a_opt)
                print(f"  Restart {r}: score={best_score:.8f} **")
            else:
                print(f"  Restart {r}: score={ev_r:.8f}")

    # Final intensive polish on the best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n--- Final polish (remaining={remaining:.0f}s) ---")
        a_final = best_seq.copy()
        a_final = a_final / np.sum(a_final)

        for round_idx in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # Alternating LP + CD
            alloc_half = min(remaining * 0.25, 40.0)
            a_lp = lp_alternating(a_final, time_budget=alloc_half)
            a_final = coord_descent(a_lp, max_passes=200, time_budget=alloc_half)
            ev_final = evaluate_sequence([float(x) for x in a_final])
            if ev_final < best_score:
                best_score = ev_final
                best_seq = a_final.copy()
                print(f"  Polish round {round_idx}: score={best_score:.8f} **")
            else:
                print(f"  Polish round {round_idx}: score={ev_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    n_result = len(best_seq)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val == np.inf or final_val > best_score + 0.01:
        # Fallback
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
