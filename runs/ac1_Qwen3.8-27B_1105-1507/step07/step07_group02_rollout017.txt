"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent and soft-max gradient.
Key innovation: fully vectorized coordinate descent that is guaranteed to decrease
the objective at each step, combined with soft-max gradient for fast initial convergence.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def coord_descent_sweep(a, conv, n):
    """
    One full sweep of coordinate descent.
    Optimizes each coordinate in turn, updating conv after each change.
    Returns updated a, conv, and the final objective value.
    """
    S = float(np.sum(a))
    order = np.arange(n)  # Could randomize
    
    for i in order:
        a_i = a[i]
        if a_i < 1e-15:
            # Still try to increase from 0
            pass
        
        k_min = i
        k_max = min(2 * n - 2, i + n - 1)
        
        if k_min > k_max:
            continue
        
        ks = np.arange(k_min, k_max + 1)
        kis = ks - i  # all in [0, n-1]
        
        # Unaffected max
        if k_min > 0 or k_max < 2*n-2:
            mask = np.ones(2*n-1, dtype=bool)
            mask[k_min:k_max+1] = False
            if np.any(mask):
                max_unaffected = conv[mask].max()
            else:
                max_unaffected = 0.0
        else:
            max_unaffected = 0.0
        
        a_kis = a[kis]
        base = conv[k_min:k_max+1].copy()
        is_center = (ks == 2 * i)
        
        # Generate candidate x values
        if a_i > 1e-10:
            lo = max(0, a_i * 0.3)
            hi = a_i * 2.0 + 1e-4
        else:
            lo = 0.0
            hi = S / n * 3.0
        
        num_xs = 200
        xs = np.linspace(lo, hi, num_xs)

        # Non-center: new_val = base + 2*(x - a_i) * a_kis
        # Center: new_val = base_center + x^2 - a_i^2

        if not np.all(is_center):
            non_center_mask = ~is_center
            base_nc = base[non_center_mask]  # shape (num_nc,)
            a_kis_nc = a_kis[non_center_mask]  # shape (num_nc,)
            # vals_nc[k, x] = base_nc[k] + 2*(x - a_i)*a_kis_nc[k]
            # = (base_nc - 2*a_i*a_kis_nc) + 2*x*a_kis_nc
            c_nc = base_nc - 2 * a_i * a_kis_nc  # shape (num_nc,)
            # max over k for each x:
            max_nc = (c_nc[:, None] + 2 * xs[None, :] * a_kis_nc[:, None]).max(axis=0)  # shape (num_xs,)
        else:
            max_nc = np.full(num_xs, -np.inf)
        
        if np.any(is_center):
            ci = np.where(is_center)[0][0]
            center_val = xs**2 + base[ci] - a_i**2  # shape (num_xs,)
        else:
            center_val = np.full(num_xs, -np.inf)
        
        max_new = np.maximum.reduce([
            np.full(num_xs, max_unaffected),
            max_nc,
            center_val

        if not np.any(valid):

        objs = 2 * n * max_new[valid] / (new_Ss[valid] ** 2)
        best_local = np.argmin(objs)
        best_x = xs[valid][best_local]
        best_obj = objs[best_local]
