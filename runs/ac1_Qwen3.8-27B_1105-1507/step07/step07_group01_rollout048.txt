            # linear_coeff[k] = coefficient of new_x in conv_k (for the linear part)
            # quad_coeff[k] = 1 if there's a quadratic term (k=2i), else 0
            
            # For j=i: contribution = x * a_{k-i} (where x is new a_i)
            #   Valid if 0 <= k-i < n
            # For j=k-i: contribution = a_{k-i} * x (where x is new a_i)
            #   Valid if 0 <= k-i < n and k-i != i
            # For j=i=k-i (k=2i): contribution = x * x = x^2 (only once)
            
            # So:
            # rest[k] = conv[k] - x_old * a[k-i] (if valid) - (a[k-i] * x_old if 0<=k-i<n and k-i!=i)
            #         = conv[k] - x_old * a[k-i] - a[k-i] * x_old (if k-i valid and k-i!=i)
            #         = conv[k] - 2*x_old*a[k-i] (if k-i valid and k-i!=i)
            #         = conv[k] - x_old^2 (if k-i == i, i.e., k=2i)
            #         = conv[k] - x_old*a[k-i] (if k-i valid and k-i!=i... wait this is the same as above)
            
            # Let me just compute it directly:
            conv_k = compute_conv(a)  # This is expensive! Let me avoid recomputing.
            
            # Actually, let me compute it once at the start of each pass and update incrementally.
            # Or better: compute rest and d directly.
            
            # rest[k] = conv[k] - (terms involving a[i])
            # Terms involving a[i] in conv[k]:
            #   From j=i: a[i] * a[k-i] (if 0<=k-i<n)
            #   From j=k-i: a[k-i] * a[i] (if 0<=k-i<n and k-i != i)
            
            a_k_minus_i = np.zeros(2 * n - 1)
            valid_ji = (j_k >= 0) & (j_k < n)
            a_k_minus_i[valid_ji] = a[j_k[valid_ji]]
            
            # Subtract j=i contribution: a[i] * a[k-i]
            rest = conv_k - x_old * a_k_minus_i
            # Subtract j=k-i contribution (if k-i != i)
            not_self = valid_ji & (j_k != i)
            rest[not_self] -= a[j_k[not_self]] * x_old
            
            # Now conv[k] as function of new x = a[i]:
            # = rest[k] + x * a[k-i] (from j=i) + a[k-i] * x (from j=k-i, if k-i != i)
            # = rest[k] + x * (a[k-i] + a[k-i]) (if valid and k-i != i)
            # = rest[k] + 2*x*a[k-i] (if valid and k-i != i)
            # = rest[k] + x^2 (if k=2i)
            # = rest[k] + x*a[k-i] (if valid and k-i == i... but that's k=2i, handled above)
            # = rest[k] (if k-i not valid)
            
            d = np.zeros(2 * n - 1)
            d[valid_ji & (j_k != i)] = 2 * a[j_k[valid_ji & (j_k != i)]]
            # For k=2i: quadratic term
            quad_k = (j_k == i)  # This is k=2i
            d[quad_k] = 0  # Will handle separately
            
            # Minimize max_k (rest[k] + d[k]*x + (x^2 if quad_k[k] else 0)) over x >= 0
            # Use ternary search (convex function)
            
            # Upper bound for x: current value * 3 + some margin
            x_lo = 0.0
            x_hi = max(3.0 * x_old + 1.0 / n, 0.1)
            
            # Check if x_hi is large enough
            # At x=0: f(0) = max(rest)
            # At x=x_old: f(x_old) should be max(conv) (sanity check)
            
            # Ternary search
            f0 = np.max(rest + d * x_lo + x_lo**2 * quad_k.astype(float))
            f_hi = np.max(rest + d * x_hi + x_hi**2 * quad_k.astype(float))
            
            # If f is decreasing at x_hi, extend
            if f_hi < f0:
                x_hi = x_hi * 2
                # Don't extend too much
                for _ in range(5):
                    f_hi2 = np.max(rest + d * x_hi + x_hi**2 * quad_k.astype(float))
                    if f_hi2 >= f_hi:
                        break
                    x_hi *= 1.5
                    f_hi = f_hi2
            
            # Now ternary search in [x_lo, x_hi]
            a1, a2 = x_lo, x_hi
            for _ in range(40):  # ~40 iterations gives precision ~1e-12
                    m1 = a1 + (a2 - a1) / 3
                    m2 = a2 - (a2 - a1) / 3
                    f1 = np.max(rest + d * m1 + m1**2 * quad_k.astype(float))
                    f2 = np.max(rest + d * m2 + m2**2 * quad_k.astype(float))
                    if f1 < f2:
                        a2 = m2
                    else:
                        a1 = m1
            
            x_new = (a1 + a2) / 2
            
            # Ensure improvement
            f_new = np.max(rest + d * x_new + x_new**2 * quad_k.astype(float))
            f_old = np.max(conv_k)
            
            if f_new < f_old - 1e-15:
                a[i] = x_new
                improved_any = True
            else:
                a[i] = x_new  # Still take the step (should be >= old)
        
        # Normalize
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and passes % 10 == 0:
            print(f"    CCD pass {passes}: obj={best_obj:.8f}")
        
        if passes > 5 and not improved_any:
            break
    
    return best_a
```

                break

            d[not_self] = 2 * a[j_k[not_self]]

            x_hi = max(3.0 * x_old + 2.0/n, 0.5)

            f_xold = np.max(rest + d * x_old + x_old**2 * quad_k)
            f_xhi = np.max(rest + d * x_hi + x_hi**2 * quad_k)
            if f_xhi < f_xold:
                for _ in range(3):
                    x_hi *= 2
                    f_xhi_new = np.max(rest + d * x_hi + x_hi**2 * quad_k)
                    if f_xhi_new >= f_xhi:
                        break
                    f_xhi = f_xhi_new
            
            lo, hi = 0.0, x_hi
            for _ in range(30):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                f1 = np.max(rest + d * m1 + m1**2 * quad_k)
                f2 = np.max(rest + d * m2 + m2**2 * quad_k)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

Now, the key question: will this actually improve over LP alternating? The answer is: it explores a different path in the landscape. LP alternating goes to a specific fixed point (the minimax equilibrium). Coordinate descent might find a different local optimum or converge to the same one.

To combine the benefits, I'll:
1. Run LP alternating first (fast convergence to a good point)
2. Run coordinate descent from that point (might improve slightly)
3. Run smooth-max gradient (fine polishing)
4. Try different n values

Actually, let me think about what else could help:

**Key insight**: The LP solution g* gives us the "best response" to a. The fixed point of the alternating algorithm satisfies: a is the best response to a (i.e., a minimizes max conv(a, g) over g, AND a is the best response in the LP sense). At a fixed point, the LP value equals max conv(a,a).

To improve, we need to break the symmetry. One way: **aturistic LP** where we solve the LP with a slightly perturbed a.

Another way: **use the dual of the LP**. The dual variables tell us which convolution bins are "active" (binding). We can use this to construct a weighted objective: minimize sum_k w_k * conv_k where w_k are the dual variables. This is a QP (quadratic objective, linear constraints).

**QP approach**: 
minimize sum_k w_k * conv_k(a)  subject to sum(a)=1, a>=0
where w_k are the dual variables from the inner LP.

This is a QP because conv_k(a) = a^T C_k a (quadratic in a). The objective is a^T (sum_k w_k C_k) a, which is a quadratic form.

Solving this QP gives a direction that simultaneously addresses all active constraints.

Let me implement this:

Alternative: Don't build Q explicitly. Just compute Q@a when needed for the QP solver.

Actually, for the QP solver, we can use the fact that the optimum satisfies the KKT conditions:
- 2*Q*a + lambda*1 + mu = 0 (where mu >= 0 is the slack for a >= 0)
- 1^T a = 1
- a >= 0, mu >= 0, a_i * mu_i = 0

This is equivalent to finding the point on the simplex closest to the solution of 2*Q*a = -lambda*1 (with projection).

For a simpler approach, I can just do projected gradient on the QP:
- Gradient of a^T Q a = 2*Q*a
- Q*a: (Q*a)[i] = sum_j Q[i,j] a[j] = sum_j (sum_k w_k * C_k[i,j]) a[j] = sum_k w_k * sum_j C_k[i,j] a[j] = sum_k w_k * a_{k-i} (if valid)
  = sum_k w_k * conv_of_a_and_indicator... 

Actually, (Q*a)[i] = sum_k w_k * (sum_j C_k[i,j] * a[j]) = sum_k w_k * a_{k-i} (where valid)

This is a convolution-like operation: (Q*a)[i] = (w * a)_i where * is some cross-correlation.

Specifically: (Q*a)[i] = sum_k w[k] * a[k-i] for valid k-i.
= sum_j w[i+j] * a[j] for valid i+j
= np.correlate(w, a, mode='full')[n-1:]... 

Let me just compute it as:
(Q*a)[i] = sum_{j=0}^{n-1} a[j] * w[i+j] (where i+j < 2n-1)

This is the "cross-correlation" of w and a, which can be computed via FFT!

So the QP gradient is: 2 * cross_correlation(w, a)

And we can do projected gradient descent on the simplex for this QP.

This is a nice improvement! Let me incorporate this.

```python
"""
Enhanced optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating, exact coordinate descent, QP-directed projection,
and smooth-max gradient descent with adaptive temperature.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    n = len(v)
    total = np.sum(v)
    if total <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_direction(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a / float(np.sum(a))
    
    from scipy.optimize import linprog
    
    # Build A: A[k,i] = a[k-i] if 0<=k-i<n
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = max(0, i)
        k_end = min(2 * n - 1, i + n)
        A[k_start:k_end, i] = a[:k_end - k_start]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
            # Get dual values (slack values) for the A_ub constraints
            # res.slack gives the slack of A_ub constraints
            # The active constraints are where slack ~ 0
            return g_val
    return None

def get_lp_dual_info(a):
    """Solve LP and return both primal g and dual (weights on active constraints)."""
    n = len(a)
    a = a / float(np.sum(a))
    
    from scipy.optimize import linprog
    
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = max(0, i)
        k_end = min(2 * n - 1, i + n)
        A[k_start:k_end, i] = a[:k_end - k_start]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
        # Dual variables for A_ub constraints (these are the weights w_k)
        # In linprog, res.ineqlin is the dual for A_ub (if available)
        w = np.zeros(2 * n - 1)
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            w = np.maximum(np.asarray(res.ineqlin, dtype=float), 0)
        return g_val, w
    return None, None

def qp_projected_gradient(a, w, time_budget=10.0):
    """
    Minimize a^T Q a (= sum_k w_k * conv_k(a)) on the simplex.
    Gradient: 2 * cross_corr(w, a)
    """
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / (2 * n * float(np.max(w)) + 1e-10)
    # Actually the Lipschitz constant of the gradient is related to the max eigenvalue of Q
    # Q[i,j] = sum_k w[k] * [i+j == k for valid]
    # Q@a: (Q*a)[i] = sum_j (sum_k w[k]*[i+j==k]) a[j] = sum_j w[i+j] a[j]
    # = cross_correlation(w, a)
    # The Lipschitz constant L = max eigenvalue of Q (or 2*max eigenvalue for the gradient of a^T Q a)
    # Upper bound: L <= 2 * sum_i (Q*a)_i * a_i / max(a_i^2)... complicated
    # Just use a conservative lr with backtracking
    
    lr = 0.1
    v = np.zeros(n)  # momentum
    momentum = 0.8
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute Q@a = cross_correlation(w, a)
        # (Q*a)[i] = sum_j w[i+j] * a[j] for j=0..n-1 where i+j < 2n-1
        # This is np.correlate(w, a, mode='full')[n-1:n-1+n]... let me verify
        # np.correlate(w, a, 'full')[k] = sum_j w[j] * a[j + k - (n-1)]
        # Hmm, let me just compute it directly for clarity
        qa = np.zeros(n)
        for i in range(n):
            # qa[i] = sum_j a[j] * w[i+j] for valid i+j
            j_range = np.arange(n)
            ik = i + j_range
            valid = ik < (2 * n - 1)
            qa[i] = np.sum(a[j_range[valid]] * w[ik[valid]])
        
        # Actually this is O(n^2) per iteration which is fine for n=800
        
        grad = 2.0 * qa
        grad_centered = grad - np.sum(a * grad)  # Project out the component along 1
        
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Maybe increase lr
            if it % 100 == 0 and lr < 0.5:
                lr *= 1.05
        else:
            lr *= 0.9
            v *= 0.5
            if lr < 1e-6:
                a = best_a.copy()
                break
    
    return best_a

def coordinate_descent(a, time_budget=60.0):
    """Exact coordinate descent with 1D convex minimization."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    k_arr = np.arange(2 * n - 1)
    passes = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        passes += 1
        conv = compute_conv(a)
        
        rng = np.random.default_rng(passes * 7919 + 13)
        perm = rng.permutation(n)
        
        improved = False
        for i in perm:
            if time.monotonic() - start > time_budget:
                break
            
            x_old = a[i]
            j_k = k_arr - i
            valid_ji = (j_k >= 0) & (j_k < n)
            not_self = valid_ji & (j_k != i)
            quad_k = (j_k == i).astype(float)
            
            a_kmi = np.zeros(2 * n - 1)
            a_kmi[valid_ji] = a[j_k[valid_ji]]
            
            rest = conv.copy()
            rest[valid_ji] -= x_old * a_kmi[valid_ji]
            rest[not_self] -= a[j_k[not_self]] * x_old
            
            d = np.zeros(2 * n - 1)
            d[not_self] = 2.0 * a[j_k[not_self]]
            
            # Ternary search for x >= 0 minimizing max(rest + d*x + x^2*quad_k)
            f_at_0 = np.max(rest)
            x_hi = max(2.0 * x_old + 2.0 / n, 0.5)
            f_at_hi = np.max(rest + d * x_hi + x_hi * x_hi * quad_k)
            
            if f_at_hi > f_at_0:
                # The minimum is in [0, x_hi], likely near 0
                x_hi = x_old + 1.0 / n
            else:
                # Extend if still decreasing
                for _ in range(3):
                    f_new = np.max(rest + d * x_hi + x_hi * x_hi * quad_k)
                    if f_new >= f_at_hi:
                        break
                    x_hi *= 1.5
                    f_at_hi = f_new
            
            lo, hi = 0.0, x_hi
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(rest + d * m1 + m1 * m1 * quad_k)
                f2 = np.max(rest + d * m2 + m2 * m2 * quad_k)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            x_new = (lo + hi) / 2.0
            a[i] = x_new
            if abs(x_new - x_old) > 1e-10:
                improved = True
        
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break
    
    return best_a

def smooth_max_gradient(a, time_budget=30.0, T_init=0.0003, T_min=0.00002):
    """Smooth max gradient descent with Nesterov momentum."""
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = T_init
    lr = 0.003
    v = np.zeros(n)
    momentum = 0.85
    stall = 0
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = np.clip((conv - max_c) / T, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        
        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)
        
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 50 == 0 and lr < 0.01:
                lr *= 1.03
        else:
            stall += 1
            if stall > 40:
                lr *= 0.7
                v *= 0.5
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break
        
        if it > 30:
            T = max(T_min, T_init * (0.998 ** it))
    
    return best_a

def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization."""
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        g = solve_lp_direction(a)
        if g is None:
            break
        
        # Try multiple step sizes
        best_t = 0.3
        best_t_obj = np.inf
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = ((1 - best_t) * a + best_t * g)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / float(np.sum(best_a))
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = np.sum(a_new)
    if s > 0:
        a_new = a_new / s
    return np.maximum(a_new, 0)
