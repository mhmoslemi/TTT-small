"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with warm-starting, linearized proximal LP,
adaptive mirror descent, structured initializations, and fine-tuning.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # Try cvxpy solvers first
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[0] = 1.0  # minimize M
        m_constr = 2 * n - 1
        A_ub = np.zeros((m_constr, n + 1))
        A_ub[:, 0] = -1.0  # -M + A@g <= 0
        A_ub[:, 1:] = A
        b_ub = np.zeros(m_constr)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] + [(0, None)] * n  # M >= 0, g >= 0
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            M_val = float(res.x[0])
            g_val = np.maximum(res.x[1:], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, M_val
    except Exception:
        pass

    return None, None


def solve_linearized_proximal(a, epsilon=0.01):
    """
    Linearized proximal method: linearize the convolution at current point a,
    solve LP for direction delta, then line search.
    Returns the best point found.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    conv = compute_conv(a)
    c = conv  # c[k] = conv_k(a)
    
    # Build the gradient matrices: grad_k[i] = a[k-i] if 0 <= k-i < n
    # We need: 2 * grad_k · delta <= M - c_k
    # i.e., c_k + 2*grad_k·delta <= M
    
    # Build LP:
    # minimize M
    # s.t. 2*(grad_k · delta) - M <= -c_k for all k
    #      sum(delta) = 0
    #      -a[i] <= delta[i] (to keep a+delta >= 0)
    #      -epsilon <= delta[i] <= epsilon (locality)
    
    m = 2 * n - 1  # number of convolution lags
    
    # Variables: M, delta[0], ..., delta[n-1]
    # Total: n+1 variables
    
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0  # minimize M
    
    # Inequality constraints: 2*grad_k·delta - M <= -c_k
    # For each k, grad_k is a vector of length n
    # grad_k[i] = a[k-i] if 0 <= k-i < n, else 0
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for k in range(m):
        A_ub[k, 0] = -1.0  # -M
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i + 1] = 2.0 * a[j]
        b_ub[k] = -c[k]
    
    # Locality constraints: -epsilon <= delta[i] <= epsilon
    # This adds 2n more constraints
    A_loc = np.zeros((2 * n, n + 1))
    b_loc = np.zeros(2 * n)
    for i in range(n):
        A_loc[2*i, i+1] = 1.0
        b_loc[2*i] = epsilon
        A_loc[2*i+1, i+1] = -1.0
        b_loc[2*i+1] = epsilon
    
    A_full = np.vstack([A_ub, A_loc])
    b_full = np.concatenate([b_ub, b_loc])
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] + [(-epsilon, epsilon)] * n
    # M is unbounded below (but the constraints will bound it)
    
    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            delta = res.x[1:]
            # Line search: find best t in [0, t_max]
            best_obj = compute_obj(a)
            best_point = a.copy()
            t_max = 1.0  # maximum step
            
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
                a_try = a + t * delta
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_point = a_try.copy()
            
            return best_point
    except Exception:
        pass
    
    return a


def mirror_descent_improved(a, n_iters=3000, time_budget=60.0,

    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    alpha = alpha_init
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Get top m peaks

        top_val = conv[top_idx]
        max_val = top_val[-1]
        
        # Softmax weights with adaptive temperature
        spread = max_val - top_val[0]

def chirp_init(n, alpha=1.0):
    """Zadoff-Chu inspired initialization (real, non-negative)."""
    i = np.arange(n)
    # Use magnitude of Zadoff-Chu-like sequence
    phase = 2 * np.pi * alpha * i * (i + 1) / (2 * n)
    zc = np.exp(1j * phase)
    # Take a non-negative "shadow": 1 + cos of phase (always in [0, 2])
    a = 1.0 + np.cos(phase)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def thue_morse_init(n):
    """Thue-Morse inspired initialization."""
    # Generate Thue-Morse sequence
    t = np.zeros(n, dtype=int)
    m = 1
    while m < n:
        t[m:min(2*m, n)] = 1 - t[:min(m, n)]
        m *= 2
    # Convert to non-negative with structure
    a = 1.0 + 0.5 * (2 * t - 1)  # values in {0.5, 1.5}
    # Smooth slightly
    kernel = np.ones(3) / 3
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            if s_try > 0:
                a_norm = a / s_try
                obj_try = compute_obj(a_norm)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
        
        if not improved:
            step_size *= 0.7
            if step_size < 1e-6:
                break
    
    return a / np.sum(a)
