from scipy.fft import rfft, irfft
import time

def _fft_conv(a, b):
    """Compute convolution of a and b using FFT."""
    la, lb = len(a), len(b)
    size = 1
    while size < la + lb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:la + lb - 1]

    """Compute autocorrelation conv(a, a)."""

def _fft_conv(a, b):

    size = 1
    while size < la + lb - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:la + lb - 1]

    return _fft_conv(a, a)

    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    an = a / s

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a, solver_list=None):
    """Solve LP: min M s.t. A@g<=M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if solver_list is None:
        solver_list = [cp.PDLP, cp.SCS, cp.ECOS]
    for s in solver_list:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except:
            continue
    try:
        from scipy.optimize import linprog
        c = np.zeros(n+1); c[-1] = 1.0
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = A; A_ub[:, n] = -1.0
        b_ub = np.zeros(2*n-1)
        A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)]*n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.status == 0:
            gv = np.maximum(res.x[:n], 0)
            gs = float(np.sum(gv))
            if gs > 1e-10:
                return gv / gs, float(res.x[n])
    except:
        pass
    return None, None

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

6. **Gradient-based LP alternative**: Instead of solving the full LP, use a few iterations of the game (SPDM) as an approximation. This is much faster and can be done thousands of times.

Let me focus on idea 6: replace the expensive LP with fast SPDM game iterations. This will allow many more outer iterations.

```python
def fast_lp_approx(a, n_iters=20):
    """Approximate LP solution using SPDM game iterations."""
    n = len(a)
    a = a / np.sum(a)
    g = a.copy()
    p = np.ones(2*n-1) / (2*n-1)
    eta = 0.5
    
    for _ in range(n_iters):
        # Update p: p ∝ p * exp(eta * conv(a,g))
        ag = _fft_conv(a, g)
        p = p * np.exp(eta * (ag - np.max(ag)))
        p = np.maximum(p, 1e-30)
        p /= np.sum(p)
        
        # Update g: g ∝ g * exp(-eta * (A^T @ p))
        atp = _fft_conv(a[::-1], p)[n-1:n-1+n]
        g = g * np.exp(-eta * (atp - np.min(atp)))
        g = np.maximum(g, 1e-30)
        g /= np.sum(g)
    
    return g
```

This is much faster than solving the full LP (no solver overhead) and can be done many more times. The quality depends on n_iters, but even 20 iterations should give a good approximation.

Let me also add a "dual ascent" component to the SMD:

In the SMD, the gradient is 2*(A^T @ w) where w = softmax(T*conv). We can think of w as the "dual" variable. If we also optimize over w (or use the LP dual to get a better w), we get a better gradient direction.

Here's the combined approach:
1. Run a few SPDM game iterations to get a good (g, p) pair
2. Use p as the weight for the SMD gradient (instead of softmax(T*conv))
3. Update a toward g (the primal LP solution)

This combines the best of both worlds: the game gives us both the direction (g) and the gradient weights (p).
