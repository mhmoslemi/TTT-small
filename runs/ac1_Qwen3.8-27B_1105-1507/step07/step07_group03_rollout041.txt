Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time
import math

def compute_conv_fft(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa, size).real[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def error_reduction(a, fft_size=None):
    """
    One ER iteration for flat spectrum.
    Enforces flat magnitude in frequency, non-negativity in time.
    Returns updated a and the distance (error metric).
    """
    n = len(a)
    if fft_size is None:
        fft_size = 1
        while fft_size < 2 * n:
            fft_size <<= 1
    
    # Time -> Frequency
    A = fft(a, fft_size)
    
    # Enforce flat magnitude in frequency
    mag_target = np.ones(fft_size)  # flat
    phase = A / np.maximum(np.abs(A), 1e-30)
    A_er = mag_target * phase
    
    # Frequency -> Time
    a_er = ifft(A_er, fft_size).real[:n]
    
    # Enforce non-negativity in time
    a_er = np.maximum(a_er, 0)
    s = np.sum(a_er)
    if s > 1e-10:
        a_er = a_er / s  # maintain normalization
    
    # Error metric: difference between desired and actual in time domain
    # (for RESCAL, we use the time domain error)
    # For ERROR, we track the frequency domain error
    A_check = fft(a_er, fft_size)
    error = np.sum((np.abs(A_check) - mag_target[:n])**2)  # approximate
    
    return a_er, error

    # So if |A[k]| = C, then n_eff * fft_size = fft_size * C^2 (roughly)
    # C = sqrt(sum|a|^2) approximately
    # Let's just use C = 1 (since A[0] = sum(a) = 1)
    
    A_oa = (1 - alpha) * A + alpha * phase  # phase has unit magnitude
    
    a_oa = ifft(A_oa, fft_size).real[:n]
    a_oa = np.maximum(a_oa, 0)
    s = np.sum(a_oa)
    if s > 1e-10:
        a_oa = a_oa / s
    
    return a_oa

def error_overlap_algorithm(a, n_er=100, n_oa=10, alpha=0.5, time_budget=60.0, verbose=False):
    """
    ERROR algorithm: Alternates between Error Reduction (ER) and Overlap-Add (OA).
    ER steps aggressively enforce constraints.
    OA steps gently blend to avoid local minima.
    """
    n = len(a)

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_er + n_oa):
        if time.monotonic() - start > time_budget:
            break
        
        if it % (n_er + n_oa) < n_er:
            # ER step
            a, _ = error_reduction(a, fft_size)
        else:
            # OA step
            a = overlap_add(a, alpha=alpha, fft_size=fft_size)

def subgradient_refine(a, n_iters=5000, time_budget=120.0, verbose=False):
    """
    Subgradient method with multi-peak weighting and adaptive step size.
    Uses Nesterov-like acceleration.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    
    # Multi-peak: consider top K peaks
    K = min(10, n // 5)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient at y
        conv_y = compute_conv_fft(y)
        
        # Find top K peaks
        top_idx = np.argsort(conv_y)[-K:]
        top_val = conv_y[top_idx]
        max_val = top_val[-1]
        
        # Softmax weights for peaks
        temp = max(0.005, (max_val - top_val[0]) * 0.3)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()
        
        # Compute subgradient (weighted sum of contributions)
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * y[j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size
        alpha_step = 1.0 / (np.sqrt(it + 1) * grad_norm)
        alpha_step = min(alpha_step, 0.3)
        
        # Nesterov update
        v = beta * v + (1 - beta) * (grad / grad_norm)
        y_new = y - alpha_step * v
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new > 1e-10:
            y_new = y_new / s_new
        
        # Update y (Nesterov)
        mu = 2.0 / (np.sqrt(it + 1) + 1)
        y_new_update = (1 - mu) * y_new + mu * y
        y = y_new_update
        
        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()
        
        if verbose and it % 500 == 0:
            print(f"    Subgrad it{it}: obj={obj:.8f}")
    
    return best_a

def mirror_descent_refine(a, n_iters=5000, time_budget=120.0, verbose=False):

    Mirror descent (exponential weights) with multi-peak subgradient.
    More stable than subgradient for this problem.

        for w, k in zip(weights, top_idx):
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            for i in range(lo, hi + 1):

                if 0 <= j < n:

        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={obj_new:.8f}, eta={eta:.4f}")
    
    return best_a

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. ERROR (Error Reduction + Overlap Add) for flat spectrum design
2. Structured Gaussian-mixture initialization
3. Mirror descent with multi-peak softmax weighting
4. Simulated annealing for local refinement
5. Aggressive multi-n exploration with warm starts
6. Final intensive polish with adaptive step sizes

import numpy as np
from scipy.fft import fft, ifft
import time
import math


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa, size).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def error_reduction_step(a, fft_size):
    """One Error Reduction step for flat spectrum."""
    n = len(a)
    A = fft(a, fft_size)
    phase = A / np.maximum(np.abs(A), 1e-30)
    a_er = ifft(phase, fft_size).real[:n]
    a_er = np.maximum(a_er, 0)
    s = np.sum(a_er)
    if s > 1e-10:
        a_er = a_er / s
    return a_er


def overlap_add_step(a, alpha, fft_size):
    """One Overlap-Add step for flat spectrum."""
    n = len(a)
    A = fft(a, fft_size)
    phase = A / np.maximum(np.abs(A), 1e-30)
    A_oa = (1.0 - alpha) * A + alpha * phase
    a_oa = ifft(A_oa, fft_size).real[:n]
    a_oa = np.maximum(a_oa, 0)
    s = np.sum(a_oa)
    if s > 1e-10:
        a_oa = a_oa / s
    return a_oa


def error_overlap_algorithm(a, n_er, n_oa, alpha, time_budget, fft_size=None):
    """
    ERROR algorithm: Alternates Error Reduction and Overlap-Add steps.
    More robust than pure GS for flat spectrum design.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if fft_size is None:
        fft_size = 1
        while fft_size < 2 * n:
            fft_size <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_er + n_oa):
        if time.monotonic() - start > time_budget:
            break
        if it < n_er:
            a = error_reduction_step(a, fft_size)
        else:
            a = overlap_add_step(a, alpha, fft_size)
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

    return best_a


def mirror_descent_refine(a, n_iters, time_budget, K=None):
    """Mirror descent (exponential weights) with multi-peak softmax subgradient."""
    n = len(a)
    if K is None:
        K = min(8, max(2, n // 5))
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.4
    consecutive_rejects = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_a = compute_conv_fft(a)
        top_idx = np.argsort(conv_a)[-K:]
        top_val = conv_a[top_idx]
        max_val = top_val[-1]

        temp = max(0.005, (max_val - top_val[0]) * 0.4)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            for i in range(lo, hi + 1):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * a[j]

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            consecutive_rejects = 0
        else:
            consecutive_rejects += 1
            if consecutive_rejects % 20 == 0:
                eta *= 0.95
            if consecutive_rejects % 200 == 0:
                eta = min(eta * 1.3, 0.8)

    return best_a


def simulated_annealing_refine(a, time_budget, T_init=0.02, T_min=0.001, n_perturb=3):
    """
    Simulated annealing: perturb random elements, accept with Metropolis criterion.
    Effective for escaping shallow local minima near the optimum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    total_iters = 30000
    current_T = T_init

    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break

        # Cooling schedule
        progress = it / total_iters
        current_T = T_init * (T_min / T_init) ** progress

        # Perturb n_perturb random elements
        a_try = a.copy()
        idx = rng.integers(0, n, n_perturb)
        for i in idx:
            scale = max(a_try[i], 1.0 / n) * 0.3
            a_try[i] = max(0.0, a_try[i] + rng.normal(0, scale))

        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        delta = obj_try - current_obj

        # Metropolis acceptance
        if delta < 0 or rng.random() < math.exp(-delta / max(current_T, 1e-10)):
            a = a_try
            current_obj = obj_try
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a.copy()

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def gaussian_mixture_init(n, n_components, rng):
    """
    Generate a sequence as a sum of Gaussian bumps.
    This provides structured initializations that are known to have
    reasonable autocorrelation properties.
    """
    a = np.zeros(n)
    total_weight = 0.0
    for _ in range(n_components):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(n * 0.05, n * 0.4)
        w = rng.uniform(0.5, 1.0)
        x = np.arange(n, dtype=np.float64)
        a += w * np.exp(-(x - mu) ** 2 / (2.0 * sigma ** 2))
        total_weight += w

    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    return a


def build_initial(n, mode, rng):
    """Build an initial sequence of length n with given mode."""
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = np.linspace(1.0, 0.2, n)
    elif mode == 2:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
        a = np.maximum(a, 0.01)
    elif mode == 3:
        x = np.arange(n) - n / 2.0
        sigma = n / 4.0
        a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    elif mode == 4:
        a = rng.uniform(0.5, 1.0, n)
        ks = max(3, n // 40)
        kernel = np.ones(ks) / ks
        a = np.convolve(a, kernel, mode='same')
    elif mode == 5:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** (-0.25)
    elif mode == 6:
        a = gaussian_mixture_init(n, 3, rng)
    elif mode == 7:
        a = gaussian_mixture_init(n, 5, rng)
    elif mode == 8:
        # "Edge-boosted" uniform: slightly larger at edges
        a = np.ones(n)
        edge_width = n // 10
        a[:edge_width] *= 1.3
        a[-edge_width:] *= 1.3
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a


def run_optimization_pipeline(a, deadline, verbose=False):
    """
    Run a sequence of optimization methods on a.
    Returns (best_sequence, best_objective).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 0.01:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Step 1: ERROR algorithm (fast global-ish)
    remaining = deadline - time.monotonic()
    if remaining > 15:
        budget = min(remaining * 0.12, 25.0)
        fft_sz = 1
        while fft_sz < 2 * n:
            fft_sz <<= 1
        a1 = error_overlap_algorithm(a, 150, 15, 0.5, budget, fft_sz)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            a = a1
        if verbose:
            print(f"      ERROR(a=0.5): {obj1:.8f}")

    # Step 2: ERROR with different parameters
    remaining = deadline - time.monotonic()
    if remaining > 10:
        budget = min(remaining * 0.08, 15.0)
        fft_sz = 1
        while fft_sz < 2 * n:
            fft_sz <<= 1
        a2 = error_overlap_algorithm(best_a, 50, 50, 0.3, budget, fft_sz)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            a = a2
        if verbose:
            print(f"      ERROR(a=0.3): {obj2:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        budget = min(remaining * 0.08, 15.0)
        fft_sz = 1
        while fft_sz < 2 * n:
            fft_sz <<= 1
        a3 = error_overlap_algorithm(best_a, 50, 50, 0.7, budget, fft_sz)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            a = a3
        if verbose:
            print(f"      ERROR(a=0.7): {obj3:.8f}")

    # Step 3: Mirror descent (primary local optimizer)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        budget = min(remaining * 0.30, 60.0)
        a4 = mirror_descent_refine(a, 15000, budget)
        obj4 = compute_obj(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            a = a4
        if verbose:
            print(f"      MD: {obj4:.8f}")

    # Step 4: Simulated annealing
    remaining = deadline - time.monotonic()
    if remaining > 15:
        budget = min(remaining * 0.2, 35.0)
        a5 = simulated_annealing_refine(a, budget, T_init=0.015, T_min=0.0005)
        obj5 = compute_obj(a5)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()
            a = a5
        if verbose:
            print(f"      SA: {obj5:.8f}")

    # Step 5: One more mirror descent pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        budget = min(remaining * 0.2, 30.0)
        a6 = mirror_descent_refine(best_a, 10000, budget)
        obj6 = compute_obj(a6)
        if obj6 < best_obj:
            best_obj = obj6
            best_a = a6.copy()
        if verbose:
            print(f"      MD2: {obj6:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines ERROR flat-spectrum, mirror descent, simulated annealing,
    Gaussian-mixture structured initialization, multi-n exploration, and final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={best_seq.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Optimize parent at native length ===
    print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
    a_opt, obj_opt = run_optimization_pipeline(best_seq, deadline, verbose=True)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        n_parent = len(best_seq)
    print(f"  Best after Phase 1: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    n_options = sorted(set([
        n_parent,
        n_parent - 100, n_parent - 50,
        n_parent + 50, n_parent + 100, n_parent + 200, n_parent + 400,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_parent) > 30]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_seq(best_seq, n_try)
        pipeline_budget = min(remaining * 0.25, 50.0)
        sub_deadline = time.monotonic() + pipeline_budget

        a_opt_n, obj_opt_n = run_optimization_pipeline(a_try, sub_deadline)
        status = " **" if obj_opt_n < best_score else ""
        print(f"  n={n_try}: {obj_opt_n:.8f}{status}")

        if obj_opt_n < best_score:
            best_score = obj_opt_n
            best_seq = a_opt_n.copy()

    # === Phase 3: Structured initializations (Gaussian mixtures) ===
    print(f"\n=== Phase 3: Structured Initializations ===")
    n_best = len(best_seq)

    for mode in [3, 6, 7, 8, 0, 1, 4]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if mode == 6:
            a_init = gaussian_mixture_init(n_best, 3, rng)
        elif mode == 7:
            a_init = gaussian_mixture_init(n_best, 5, rng)
        else:
            a_init = build_initial(n_best, mode, rng)

        obj_init = compute_obj(a_init)
        if obj_init > best_score * 1.5:
            continue  # Skip if too far from best

        sub_deadline = time.monotonic() + min(remaining * 0.15, 35.0)
        a_opt_i, obj_opt_i = run_optimization_pipeline(a_init, sub_deadline)
        status = " **" if obj_opt_i < best_score else ""
        print(f"  Mode {mode} (n={n_best}): {obj_opt_i:.8f}{status}")

        if obj_opt_i < best_score:
            best_score = obj_opt_i
            best_seq = a_opt_i.copy()

    # === Phase 4: Perturbations of best ===
    print(f"\n=== Phase 4: Perturbations ===")
    for pert_scale in [0.05, 0.1, 0.2]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        n_p = len(best_seq)
        a_pert = best_seq * np.exp(rng.normal(0, pert_scale, n_p))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / np.sum(a_pert)

        sub_deadline = time.monotonic() + min(remaining * 0.12, 30.0)
        a_opt_p, obj_opt_p = run_optimization_pipeline(a_pert, sub_deadline)
        status = " **" if obj_opt_p < best_score else ""
        print(f"  Perturb(scale={pert_scale}): {obj_opt_p:.8f}{status}")

        if obj_opt_p < best_score:
            best_score = obj_opt_p
            best_seq = a_opt_p.copy()

    # === Phase 5: Final intensive refinement ===
    print(f"\n=== Phase 5: Final Refinement ===")
    for rnd in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        budget = min(remaining * 0.25, 50.0)
        sub_deadline = time.monotonic() + budget
        a_final, obj_final = run_optimization_pipeline(best_seq, sub_deadline, verbose=(rnd == 0))

        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
        print(f"  Final round {rnd}: {best_score:.8f}")

    # === Phase 6: Simulated annealing squeeze ===
    print(f"\n=== Phase 6: SA Squeeze ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        budget = min(remaining * 0.4, 40.0)
        a_squeezed = simulated_annealing_refine(best_seq, budget, T_init=0.008, T_min=0.0003)
        obj_sq = compute_obj(a_squeezed)
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_squeezed.copy()
        print(f"  After SA: {best_score:.8f}")

        # One more MD pass on the SA result
        remaining = deadline - time.monotonic()
        if remaining > 15:
            budget2 = min(remaining * 0.3, 30.0)
            a_md_final = mirror_descent_refine(best_seq, 20000, budget2)
            obj_md_final = compute_obj(a_md_final)
            if obj_md_final < best_score:
                best_score = obj_md_final
                best_seq = a_md_final.copy()
            print(f"  After final MD: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
