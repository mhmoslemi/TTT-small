def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_peak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    # Fallback to HiGHS
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq,

        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))

    except Exception:
        pass
    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, left=0, right=0)
    return np.maximum(a_new, 0)


def smooth_max_obj(theta, tau, n):
    """Compute smoothed objective and gradient w.r.t. theta for softmax parameterization."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    a = exp_t / np.sum(exp_t)
    
    conv = compute_conv(a)
    conv_max = np.max(conv)
    obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)
    
    # Soft-argmax weights
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)
    
    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < 2 * n - 1 and end > j:
            c[j] = np.dot(w[j:end], a[:end - j])
    
    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta

    taus = np.geomspace(tau_start, tau_end, n_stages)
    
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        def obj_func(th):
            val, _ = smooth_max_obj(th, tau, n)
            return val
        
        def grad_func(th):
            _, g = smooth_max_obj(th, tau, n)
            return g
        
        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-6, 'maxfun': 400})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_obj


def softmax_from_theta(theta):
    """Compute softmax with numerical stability."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def subgradient_averaging(a_init, time_budget=30.0, tau=50.0, n_iter=500):
    """Subgradient method with averaging for the smoothed objective."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    a = a / np.sum(a)
    theta = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Polyak step size
    obj_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    avg_theta = theta.copy()
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        
        val, grad = smooth_max_obj(theta, tau, n)
        
        # Polyak step
        if val > obj_lb:
            step = (val - obj_lb) / (np.dot(grad, grad) + 1e-10)
            step = min(step, 1.0)  # cap step size
        else:
            step = 0.01
        
        theta = theta - step * grad
        avg_theta = avg_theta + theta  # accumulate
        
        # Periodically evaluate
        if it % 20 == 0:
            a_check = softmax_from_theta(avg_theta / (it + 1))
            a_check = np.maximum(a_check, 0)
            s = np.sum(a_check)
            if s > 1e-10:
                a_check = a_check / s
                obj_check = compute_obj(a_check)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a_check.copy()
        
        # Also check current theta
        if it % 50 == 0:
            a_check = softmax_from_theta(theta)
            a_check = np.maximum(a_check, 0)
            s = np.sum(a_check)
            if s > 1e-10:
                a_check = a_check / s
                obj_check = compute_obj(a_check)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a_check.copy()
    
    return best_a, best_obj

"""
Population-based LP alternating with range-minimizing LP, smoothed L-BFGS-B,
subgradient averaging, and multi-n exploration.
Key new ideas over previous:
1. Range-minimizing LP (minimize max-min of convolution) directly targets flatness
2. Subgradient averaging for global convergence
3. Population-based search with structured mutations and crossover
4. More aggressive multi-n exploration

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_peak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_range(a, lambda_range=1.0):
    """Solve LP: min M - lambda*m s.t. m <= conv(a,g)[k] <= M, sum(g)=1, g>=0."""

            M = cp.Variable()
            m = cp.Variable()

                A @ g <= M,
                A @ g >= m,
                cp.sum(g) == 1
            ]
            prob = cp.Problem(cp.Minimize(M - lambda_range * m), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

        A_ub[:2*n-1, :n] = A
        A_ub[:2*n-1, n] = -1.0
        A_ub[2*n-1:, :n] = -A
        A_ub[2*n-1:, n+1] = -1.0
        A_eq = np.zeros((1, n + 2))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

    """Compute softmax with numerical stability."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def smooth_max_obj_and_grad(theta, tau, n):

    a = softmax_from_theta(theta)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)
    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < 2 * n - 1 and end > j:
            c[j] = np.dot(w[j:end], a[:end - j])
    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta


def smooth_optimize(a_init, tau_start, tau_end, time_budget, max_iter=200):
    """Smoothed L-BFGS-B with tau annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    n_stages = 5
    taus = np.geomspace(tau_start, tau_end, n_stages)
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        def obj_func(th):
            val, _ = smooth_max_obj_and_grad(th, tau, n)
            return val

        def grad_func(th):
            _, g = smooth_max_obj_and_grad(th, tau, n)
            return g

        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-6, 'maxfun': 400})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass
    return best_a, best_obj


def lp_alternating(a, time_budget=60.0, use_range=False, lambda_range=1.0, verbose=False):
    """LP alternating with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    for it in range(300):
        if time.monotonic() - start > time_budget:
            break
        if use_range:
            g_lp = solve_lp_range(a, lambda_range)
        else:
            g_lp, _ = solve_lp_peak(a)
        if g_lp is None:
            break
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"        LP it{it}: obj={best_obj:.8f}")
        else:
            stall_count += 1
            if stall_count >= 5 and it % 3 == 0:
                rng = np.random.default_rng(hash((it * 7919 + n)) % (2**31))
                mut_type = it % 4
                if mut_type == 0:
                    shift = rng.integers(1, max(2, n // 15))
                    a = 0.6 * best_a + 0.4 * np.roll(best_a, shift)
                elif mut_type == 1:
                    a = best_a * np.exp(rng.normal(0, 0.04, n))
                elif mut_type == 2:
                    a = 0.5 * best_a + 0.5 * best_a[::-1]
                else:
                    ramp = np.linspace(0.8, 1.2, n)
                    a = best_a * ramp
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a
    return best_a


def subgradient_averaging(a_init, time_budget=30.0, tau=50.0, n_iter=500):
    """Subgradient method with Polyak averaging."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    obj_lb = 2.0 * n / (2.0 * n - 1.0)
    avg_theta = np.zeros(n)
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        val, grad = smooth_max_obj_and_grad(theta, tau, n)
        if val > obj_lb + 1e-6:
            step = (val - obj_lb) / (np.dot(grad, grad) + 1e-10)
            step = min(step, 0.5)
        else:
            step = 0.005
        theta = theta - step * grad
        avg_theta = avg_theta + theta
        if it % 25 == 0:
            a_check = softmax_from_theta(avg_theta / (it + 1))
            a_check = np.maximum(a_check, 0)
            s = np.sum(a_check)
            if s > 1e-10:
                a_check = a_check / s
                obj_check = compute_obj(a_check)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a_check.copy()
        if it % 50 == 0:
            a_check = softmax_from_theta(theta)
            a_check = np.maximum(a_check, 0)
            s = np.sum(a_check)
            if s > 1e-10:
                a_check = a_check / s
                obj_check = compute_obj(a_check)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a_check.copy()
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, left=0, right=0)
    return np.maximum(a_new, 0)


def apply_shift(a, shift):
    """Apply circular shift."""
    return np.roll(a, shift)


def apply_mirror_blend(a, blend=0.5):
    """Blend with reversed."""
    return blend * a + (1 - blend) * a[::-1]


def apply_gaussian_noise(a, rng, sigma=0.03):
    """Multiply by log-normal noise."""
    return a * np.exp(rng.normal(0, sigma, len(a)))


def apply_ramp(a, lo=0.7, hi=1.3):
    """Multiply by linear ramp."""
    return a * np.linspace(lo, hi, len(a))


def apply_exp_asym(a, rng):
    """Multiply by exponential asymmetry."""
    rate = rng.uniform(0.1, 0.5)
    return a * np.exp(rate * np.linspace(-1, 1, len(a)))


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP alternating (peak + range minimizing), smoothed L-BFGS-B,
    subgradient averaging, population-based search, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: LP Range-Minimizing Alternating
    print(f"\n=== Phase 1: LP Range-Minimizing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 100.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, use_range=True,
                              lambda_range=1.0, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            n_parent = len(best_seq)
            print(f"  LP Range (lambda=1): obj={best_score:.8f} **")
        else:
            print(f"  LP Range (lambda=1): obj={obj_lp:.8f}")

    # Phase 1b: LP Range with different lambda
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 50.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, use_range=True,
                              lambda_range=0.3, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP Range (lambda=0.3): obj={best_score:.8f} **")
        else:
            print(f"  LP Range (lambda=0.3): obj={obj_lp:.8f}")

    # Phase 2: Standard LP Peak
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.2, 70.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, use_range=False, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP Peak: obj={best_score:.8f} **")
        else:
            print(f"  LP Peak: obj={obj_lp:.8f}")

    # Phase 3: Smoothed L-BFGS-B
    print(f"\n=== Phase 3: Smoothed L-BFGS-B ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for trial in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            if trial == 0:
                a_init = best_seq.copy()
            elif trial == 1:
                shift = rng.integers(1, max(2, n_parent // 10))
                a_init = 0.7 * best_seq + 0.3 * apply_shift(best_seq, shift)
                a_init = a_init / np.sum(a_init)
            elif trial == 2:
                a_init = apply_mirror_blend(best_seq, 0.5)
                a_init = a_init / np.sum(a_init)
            else:
                a_init = apply_exp_asym(best_seq, rng)
                a_init = np.maximum(a_init, 0)
                a_init = a_init / np.sum(a_init)
            alloc = min(remaining * 0.18, 35.0)
            a_opt, obj_opt = smooth_optimize(a_init, tau_start=10.0, tau_end=500.0,
                                              time_budget=alloc, max_iter=150)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_parent = len(best_seq)
                print(f"  Smooth trial {trial}: obj={best_score:.8f} **")
            else:
                print(f"  Smooth trial {trial}: obj={obj_opt:.8f}")

    # Phase 4: Subgradient Averaging
    print(f"\n=== Phase 4: Subgradient Averaging ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for tau_sg in [30.0, 100.0, 300.0]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.12, 20.0)
            a_sg, obj_sg = subgradient_averaging(best_seq, time_budget=alloc, tau=tau_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  SG tau={tau_sg}: obj={best_score:.8f} **")
            else:
                print(f"  SG tau={tau_sg}: obj={obj_sg:.8f}")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600,
        max(300, n_cur - 100), n_cur - 50, n_cur, n_cur + 50, n_cur + 100,
        min(2000, n_cur + 200)
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try == n_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.08, 18.0)
        a_opt = lp_alternating(a_try, time_budget=alloc, use_range=True, lambda_range=0.7)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 6: Population-based refinement
    print(f"\n=== Phase 6: Population Refinement ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    if remaining > 60:
        # Build diverse population
        population = [best_seq.copy()]
        a_m1 = np.maximum(apply_shift(best_seq, max(1, n_best // 25)), 0)
        a_m1 = a_m1 / np.sum(a_m1)
        population.append(a_m1)

        a_m2 = np.maximum(apply_mirror_blend(best_seq, 0.4), 0)
        a_m2 = a_m2 / np.sum(a_m2)
        population.append(a_m2)

        a_m3 = np.maximum(apply_gaussian_noise(best_seq, rng, 0.05), 0)
        a_m3 = a_m3 / np.sum(a_m3)
        population.append(a_m3)

        a_m4 = np.maximum(apply_exp_asym(best_seq, rng), 0)
        a_m4 = a_m4 / np.sum(a_m4)
        population.append(a_m4)

        # Evolve
        for gen in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            new_population = []
            for i, a_pop in enumerate(population):
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break
                alloc = min(remaining * 0.12, 18.0)
                use_range = (i % 2 == 0)
                a_opt = lp_alternating(a_pop, time_budget=alloc, use_range=use_range,
                                        lambda_range=1.0)
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    n_best = len(best_seq)
                    print(f"  Gen {gen} pop {i}: obj={best_score:.8f} **")
                new_population.append((a_opt, obj_opt))
            # Crossover top two
            if len(new_population) >= 2:
                new_population.sort(key=lambda x: x[1])
                for blend in [0.3, 0.5, 0.7]:
                    a_cross = blend * new_population[0][0] + (1 - blend) * new_population[1][0]
                    a_cross = np.maximum(a_cross, 0)
                    s = float(np.sum(a_cross))
                    if s > 1e-10:
                        a_cross = a_cross / s
                        obj_cross = compute_obj(a_cross)
                        if obj_cross < best_score:
                            best_score = obj_cross
                            best_seq = a_cross.copy()
                            print(f"  Gen {gen} cross b={blend}: obj={best_score:.8f} **")
                        new_population.append((a_cross, obj_cross))
            # Keep top 6
            new_population.sort(key=lambda x: x[1])
            population = [x[0] for x in new_population[:6]]

    # Phase 7: Final intensive polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, use_range=True,
                              lambda_range=0.5, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP Range: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 30.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, use_range=False, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP Peak: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 25.0)
        a_opt, obj_opt = smooth_optimize(best_seq, tau_start=50.0, tau_end=800.0,
                                          time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final Smooth: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 15.0)
        a_sg, obj_sg = subgradient_averaging(best_seq, time_budget=alloc, tau=200.0)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
