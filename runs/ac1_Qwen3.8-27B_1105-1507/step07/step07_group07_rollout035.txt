5. Final subgradient polish with adaptive temperature
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        A[j:j + end, j] = a[:end]
    return A


def solve_lp(a, method='highs'):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=method)
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternate(a, max_iters=500, verbose=False):
    """LP alternation with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0

    for it in range(max_iters):
        g, M_lp = solve_lp(a)
        if g is None:
            break

        improved = False
        # Adaptive step sizes: try full step first, then halve
        t = 1.0
        for _ in range(8):
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                improved = True
                stall = 0
                break
            t *= 0.5

        if not improved:
            stall += 1
            if stall > 15:
                break

    if verbose:
        print(f"    LP alt: {it+1} iters, obj={best_obj:.10f}")
    return best_a


def peak_peel_lp(a, max_rounds=5, verbose=False):
    """
    Peak peeling: after LP convergence, iteratively tighten constraints
    on the current peaks to reduce them further.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for rnd in range(max_rounds):
        conv = compute_conv(a)
        max_val = conv.max()

        # Identify top peaks (within 95% of max)
        threshold = max_val * 0.95
        peak_indices = np.where(conv >= threshold)[0]

        if len(peak_indices) <= 1:
            break

        # Solve LP with tightened constraints on peak positions
        a_norm = a / np.sum(a)
        A = build_conv_matrix(a_norm)

        # For peak positions, use tighter bound: max_val * (1 - 0.05/(rnd+1))
        tighten = 1.0 - 0.03 / (rnd + 1)
        b_ub_tight = np.full(2 * n - 1, max_val)
        for k in peak_indices:
            b_ub_tight[k] = max_val * tighten

        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub_tight, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    g_val = g_val / s_g
                    for t in [1.0, 0.7, 0.5, 0.3]:
                        a_try = (1 - t) * a + t * g_val
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj = compute_obj(a_try)
                        if obj < best_obj - 1e-12:
                            best_obj = obj
                            best_a = a_try.copy()
                            a = a_try
                            break
        except Exception:
            pass

    if verbose:
        print(f"    Peak peel: obj={best_obj:.10f}")
    return best_a


def subgradient_polish(a, time_budget=30.0, verbose=False):
    """Subgradient method with adaptive step size on the max-conv objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1 / n

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient of max conv w.r.t. a[i]
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    grad[i] = 2.0 * a[j]
                else:
                    grad[i] = a[j]

        # Projected gradient step (stay on simplex)
        grad_centered = grad - grad.mean()
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 1.0 / n)
        else:
            lr *= 0.95
            if lr < 1e-6 / n:
                break

    if verbose:
        print(f"    Subgrad polish: obj={best_obj:.10f}")
    return best_a


def softmax_gradient_step(a, tau, lr):
    """One step of gradient descent on softmax-smoothed objective."""
    n = len(a)
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    # Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a[i]
    # = sum_k w[k] * d(conv[k])/d(a[i])
    # For each k, d(conv[k])/d(a[i]) = a[k-i] (if valid, i != k-i) or 2*a[k-i] (if i = k-i)
    # Sum over k: grad[i] = sum_k w[k]*a[k-i] + sum_k: 2i==k w[k]*a[i]
    # First term: cross-correlation
    # For efficiency with n~800, use direct computation
    grad = np.zeros(n)
    for i in range(n):
        total = 0.0
        for k in range(max(0, i), min(2 * n - 1, i + n)):
            j = k - i
            if 0 <= j < n:
                total += w[k] * a[j]
                if i == j:
                    total += w[k] * a[j]
        grad[i] = total

    # Center and normalize
    grad_centered = grad - grad.mean()
    a_new = a - lr * grad_centered
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    return a_new / s_new


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, mode, rng):
    """Generate diverse structured initializations."""
    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Smoothed random
        a = rng.uniform(0.1, 1.0, n)
        k = max(1, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a + 0.05 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 3:
        # Two-hump
        c1, c2 = n // 4, 3 * n // 4
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n ** 2 / 30)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n ** 2 / 30))
        a = a + 0.02 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 4:
        # Sinusoidal modulation
        freq = rng.uniform(2, 10)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 5:
        # Chirp-like (linearly varying frequency)
        t = np.arange(n) / n
        freq_start = rng.uniform(1, 3)
        freq_end = rng.uniform(5, 15)
        freq = freq_start + (freq_end - freq_start) * t
        phase = 2 * np.pi * np.cumsum(freq) / n
        a = 0.5 + 0.4 * np.sin(phase)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 6:
        # Binary-inspired (m-sequence like)
        # Use a simple LCG to create a pseudo-random ±1 sequence, map to non-negative
        x = rng.integers(0, 2, n)
        a = 0.5 + 0.5 * (2 * x - 1)  # maps 0->0, 1->1
        # Smooth slightly
        k = max(1, n // 100)
        if k > 1:
            kernel = np.ones(k) / k
            a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.001)
        a = a / np.sum(a)
    elif mode == 7:
        # Power law decay from center
        center = n // 2
        dist = np.abs(np.arange(n) - center) / (n / 2)
        alpha = rng.uniform(0.5, 3.0)
        a = (1.0 - dist) ** alpha
        a = a + 0.05 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 8:
        # Periodic with random period
        period = rng.integers(max(2, n // 20), max(3, n // 5))
        a = 1.0 + 0.3 * np.sin(2 * np.pi * np.arange(n) / period)
        a = np.maximum(a, 0.01)
        a = a + 0.05 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    else:
        # Random
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    return a


def try_lp_round(a, max_iters=300, verbose=False):
    """Complete LP round: alternation + peak peeling + subgradient."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # LP alternation
    a = lp_alternate(a, max_iters=max_iters, verbose=verbose)

    # Peak peeling
    a = peak_peel_lp(a, max_rounds=3, verbose=verbose)

    # More LP
    a = lp_alternate(a, max_iters=min(100, max_iters // 3), verbose=False)

    return a
