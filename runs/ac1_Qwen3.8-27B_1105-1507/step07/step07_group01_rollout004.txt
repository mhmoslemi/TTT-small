import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_fast(a)
    
    from scipy.optimize import linprog
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))

                return g_val / s, float(res.x[n])

        pass

def mirror_descent_peak_pressure(a, time_budget=50.0, verbose=False):
    """
    Mirror descent (exponential tilting) with top-k peak pressure.
    Updates: a[i] *= exp(-eta * pressure[i]) where pressure comes from top-k peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 5.0
    k_top = max(3, n // 20)
    stall = 0
    best_eta = eta

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top-k peak indices
        top_k_idx = np.argsort(conv)[-k_top:]
        # Weights: exponential decay from max
        top_k_vals = conv[top_k_idx]
        max_val = top_k_vals[-1]
        weights = np.exp((top_k_vals - max_val) * n * 0.5)
        weights = weights / np.sum(weights)

        # Compute pressure
        pressure = np.zeros(n)
        for idx, w in zip(top_k_idx, weights):
            k = int(idx)
            # d conv[k] / d a[i] = (a[k-i] + a[i]) if 0 <= k-i < n, else 0
            valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
            if len(valid_i) == 0:
                continue
            k_minus_i = k - valid_i
            pressure[valid_i] += w * (a[k_minus_i] + a[valid_i])

        # Mirror descent update
        tilted = a * np.exp(-eta * pressure)
        tilted = np.maximum(tilted, 1e-15)
        tilted = tilted / np.sum(tilted)

        obj_new = compute_obj(tilted)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = tilted.copy()
            a = tilted
            stall = 0
        else:
            stall += 1
            if stall > 20:
                eta *= 0.8
                if eta < 0.01:
                    eta = 5.0
                    k_top = max(3, n // 10)
                a = best_a.copy()
                stall = 0
        # Slowly increase eta if making progress
        if stall == 0 and it % 50 == 0 and eta < 20.0:
            eta = min(eta * 1.02, 20.0)
    
    return best_a

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    max_iters = 3000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        current_max = conv[k_star]

        # Find pairs that contribute to conv[k_star]
        # conv[k_star] = sum_i a[i]*a[k_star-i] for valid i
        improved = False
        for _ in range(3):  # Try a few random pairs per iteration
            # Get valid indices
            valid_i = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
            if len(valid_i) < 2:
                break
            # Pick a random valid pair (i, k_star-i) where both are in range
            i = valid_i[np.random.randint(len(valid_i))]
            j = k_star - i
            if j < 0 or j >= n or j == i:
                continue

            # Current values
            ai = a[i]
            aj = a[j]
            
            # Try to reduce a[i]*a[j] by transferring mass
            # If we move delta from i to j: new product = (ai-delta)*(aj+delta)
            # = ai*aj + delta*(ai-aj) - delta^2
            # This is minimized when delta = (ai-aj)/2 (if ai > aj) or (aj-ai)/2 (if aj > ai)
            # Constraint: delta > -ai (keep non-negative), delta < aj (keep j non-negative for first)
            # Actually we want to minimize the product while keeping ai+aj constant
            # Product = ai*aj - delta*(2*aj - ai) ... let me redo
            # (ai - d)(aj + d) = ai*aj + ai*d - aj*d - d^2 = ai*aj + d(ai-aj) - d^2
            # To minimize: d = (ai-aj)/2
            # So new values: ai' = (ai+aj)/2, aj' = (ai+aj)/2
            # This makes them equal, which minimizes the product!
            
            avg_val = 0.5 * (ai + aj)
            # But we need to be careful: making them equal reduces the product
            # but might increase other convolution terms
            alpha = 0.3  # Damping factor
            a[i] = a[i] + alpha * (avg_val - ai)
            a[j] = a[j] + alpha * (avg_val - aj)
            improved = True
        
        if not improved:
            # Try a different approach: reduce all contributors slightly
            valid_i = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
            if len(valid_i) > 2:
                # Reduce the largest contributor
                contributions = np.abs(a[valid_i] * a[k_star - valid_i])
                max_idx = valid_i[np.argmax(contributions)]
                a[max_idx] *= 0.98
                a = a / np.sum(a)
                improved = True

        if improved:
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
    
    return best_a


def nesterov_subgradient(a, time_budget=40.0, verbose=False):
    """
    Nesterov-accelerated subgradient method with Polyak averaging.
    Subgradient: direction to reduce the max peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov parameters
    beta = 0.0  # momentum
    x_prev = a.copy()
    x_curr = a.copy()
    
    # Polyak averaging
    x_avg = a.copy()
    cumulative_step = 1.0
    step_num = 0

    # Step size schedule
    eta0 = 0.02
    decay = 0.9995

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        y = x_curr + beta * (x_curr - x_prev)
        y = np.maximum(y, 0)
        y = y / np.sum(y)

        # Compute subgradient at y
        conv = compute_conv(y)
        # Find top-3 peaks for robustness
        top_idx = np.argsort(conv)[-3:]
        top_weights = np.ones(len(top_idx)) / len(top_idx)
        
        grad = np.zeros(n)
        for idx, w in zip(top_idx, top_weights):
            k = int(idx)
            valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
            if len(valid_i) == 0:
                continue
            k_minus_i = k - valid_i
            grad[valid_i] += w * (y[k_minus_i] + y[valid_i])

        # Center gradient (for simplex constraint)
        grad = grad - np.mean(grad)

        # Step size
        eta = eta0 / (1.0 + 0.01 * step_num)
        
        # Update
        x_prev = x_curr.copy()
        x_new = x_curr - eta * grad
        x_new = np.maximum(x_new, 0)
        s_new = float(np.sum(x_new))
        if s_new < 1e-10:
            x_new = a.copy()
        else:
            x_new = x_new / s_new

        # Polyak averaging
        step_size = eta
        x_avg = (cumulative_step * x_avg + step_size * x_new) / (cumulative_step + step_size)
        cumulative_step += step_size
        step_num += 1

        # Update beta (Nesterov)
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        beta = (beta_new - 1.0) / 2.0
        beta = min(beta, 0.5)

        x_curr = x_new

        # Track best
        obj_curr = compute_obj(x_curr)
        obj_avg = compute_obj(x_avg)
        if obj_curr < best_obj - 1e-12:
            best_obj = obj_curr
            best_a = x_curr.copy()
        if obj_avg < best_obj - 1e-12:
            best_obj = obj_avg
            best_a = x_avg.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def chirp_init(n, c=1.0):
    """Initialize with a chirp-like sequence for flat autocorrelation."""
    x = np.arange(n)
    # Linear chirp: phase = pi * c * (x/n)^2
    phase = np.pi * c * (x / n) ** 2
    # Take magnitude of inverse DFT of exp(i*phase)
    spec = np.exp(1j * phase)
    a = np.abs(np.fft.ifft(spec))
    a = a / np.sum(a)
    return a


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = np.abs(rng.normal(0, 1, n))
        a = a / np.sum(a)
    elif mode == 3:
        # Smooth via cumulative
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        a = a / np.sum(a)
    elif mode == 5:
        # Chirp
        a = chirp_init(n, c=rng.uniform(0.5, 2.0))
    elif mode == 6:
        # Bimodal
        center1 = n // 3
        center2 = 2 * n // 3
        width = max(1, n // 8)
        x = np.arange(n)
        a = np.exp(-((x - center1) ** 2) / (2 * width ** 2)) + \
            np.exp(-((x - center2) ** 2) / (2 * width ** 2))
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a
