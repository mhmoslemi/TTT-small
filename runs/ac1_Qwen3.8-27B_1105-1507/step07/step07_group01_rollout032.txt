        if time.monotonic() - start > time_budget:
            break

        k_star = np.argmax(conv)
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        a_new = a * np.exp(-eta * grad)
        a_new = a_new / np.sum(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    
    return best_a

import numpy as np
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    from scipy.fft import rfft, irfft
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def mwu_optimize(a, time_budget=60.0, eta_init=0.1, eta_min=0.001):
    """Multiplicative Weights Update for minimizing max convolution."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        valid_i = np.arange(n)
        j = k_star - valid_i
        mask = (j >= 0) & (j < n)
        grad[mask] = a[j[mask]]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        s = np.sum(a_new)
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            a = a_new  # still update for diversity
            stall += 1
            if stall > 50:
                eta *= 0.95
                if eta < eta_min:
                    break
                a = best_a.copy()
                stall = 0
        
        a = a_new
        # Anneal eta
        elapsed = time.monotonic() - start
        eta = max(eta_min, eta_init * (0.999 ** int(elapsed * 100)))
    
    return best_a

def coordinate_descent(a, max_passes=10, time_budget=30.0):
    """
    Coordinate descent: for each coordinate, find optimal value
    given all others fixed (with sum=1 constraint).
    """
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for _pass in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):

                break

            rest_sum = 1.0 - old_val

            #         = x * a[k-i] + rest_conv[k]
            # where rest_conv[k] = conv[k] - old_val * a[k-i]

            # i.e., x in [0, rest_sum] (since others are fixed and non-negative)

            # we need to adjust the others. The simplest is to just change a[i]

            candidates.extend([rest_sum * 0.01, rest_sum * 0.1, rest_sum * 0.2, rest_sum * 0.5])
            candidates = [c for c in candidates if 0 <= c <= 1.0]
            
            best_x = old_val
            best_local_obj = best_obj
            
            for x in candidates:
                a[i] = x
                # Renormalize others
                if x < 1.0 and n > 1:
                    # Scale others to sum to 1-x
                    others = np.delete(a, i)
                    os = np.sum(others)

                best_obj = best_local_obj
                best_a = a.copy()
                improved = True

            break

    if n == 0:
        return np.array([0.0])

    while size < 2 * n - 1:
        size <<= 1

    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

Actually, I realize I should be more careful about the gradient computation for the smooth descent. Let me verify the cross-correlation formula.

We have `conv = a * a` (full convolution), length `2n-1`.
`weights = softmax(conv / T)`, length `2n-1`.

The objective we're minimizing (smoothed) is:
`J(a) = T * log(sum_k exp(conv[k] / T))`

The gradient:
`dJ/da[i] = sum_k weights[k] * d(conv[k])/da[i]`
`= sum_k weights[k] * a[k-i]` (for valid `k-i`, i.e., `0 <= k-i < n`)

So `dJ/da[i] = sum_{k=max(i,0)}^{min(i+n-1, 2n-2)} weights[k] * a[k-i]`

Let `m = k - i`:
`= sum_{m=0}^{n-1} weights[m+i] * a[m]` (where `m+i < 2n-1`, i.e., `m < 2n-1-i`)

This is the cross-correlation of `weights` and `a` at lag `i`:
`grad[i] = (weights ⋆ a)[i] = sum_m weights[m+i] * a[m]`

Using FFT: `FFT(weights ⋆ a) = FFT(weights) * conj(FFT(a))`

So: `grad = ifft(FFT(weights, N) * conj(FFT(a, N)))[:n]`

where `N >= len(weights) + len(a) - 1 = (2n-1) + n - 1 = 3n - 2`.

"""
Step function optimizer using MWU, smooth gradient descent with FFT gradients,
and multi-n exploration. Designed for 1500s budget with adaptive time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def mwu_optimize(a, time_budget=60.0, eta_init=0.5, use_multi_peak=False, k_peaks=3):
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        
        if use_multi_peak and n >= 10:
            k = min(k_peaks, 2 * n - 1)
            top_indices = np.argsort(conv)[-k:]
            grad = np.zeros(n)
            for ki in top_indices:
                k_star = int(ki)
                idx = np.arange(n)
                j = k_star - idx
                mask = (j >= 0) & (j < n)
                grad[mask] += a[j[mask]]
            grad /= k
        else:
            k_star = int(np.argmax(conv))
            idx = np.arange(n)
            j = k_star - idx
            mask = (j >= 0) & (j < n)
            grad = np.zeros(n)
            grad[mask] = a[j[mask]]
        
        log_upd = -eta * np.clip(grad, 0, 100.0)
        a_new = a * np.exp(np.clip(log_upd, -50, 0))
        s = np.sum(a_new)
        a_new = a_new / s if s > 1e-15 else a.copy()
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0:
                eta = max(0.001, eta * 0.9)
            if stall > 2000 and eta < 0.005:
                break
        
        a = a_new
        it += 1
    
    return best_a

def smooth_grad_optimize(a, time_budget=60.0, T_init=0.01, T_min=0.0005, lr_init=0.05):
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    mom = 0.85
    stall = 0
    
    grad_size = 1
    while grad_size < 3 * n:
        grad_size <<= 1
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -100, 100)
        exp_v = np.exp(shifted)
        weights = exp_v / float(np.sum(exp_v))
        
        fw = rfft(weights, grad_size)
        fa = rfft(a, grad_size)
        grad = irfft(fw * np.conj(fa), grad_size)[:n]
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn
        
        v = mom * v - lr * grad
        a_new = simplex_project(a + v)
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 300 == 0:

                v = np.zeros(n)
                a_new = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break
        
        a = a_new
        T = max(T_min, T_init * (0.997 ** min(stall, 500)))
    
    return best_a

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def random_init(n, rng, mode=0):

        return rng.dirichlet(np.ones(n))
    elif mode == 1:
        a = np.abs(rng.normal(0, 1, n))

    elif mode == 2:

        a = np.maximum(a, 0)
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif mode == 3:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        return a / np.sum(a)
    elif mode == 4:
        center = n / 2
        width = max(1, n // 4)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        return a / np.sum(a)
    else:
        x = np.arange(n) / max(n - 1, 1)
        a = 1 - np.abs(2 * x - 1)
        return a / np.sum(a)
