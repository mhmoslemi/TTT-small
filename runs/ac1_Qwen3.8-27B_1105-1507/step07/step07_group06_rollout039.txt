    n = len(a)
    size = 1
    while size < 2 * n - 1:

import numpy as np

from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c[-1] = 1.0

    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

        if m_start <= m_end:

            best_obj = best_t_obj
            best_a = (1.0 - best_t) * best_a + best_t * g
            best_a = np.maximum(best_a, 0)
            s = float(np.sum(best_a))
            if s > 1e-10:
                best_a = best_a / s
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                # Structured perturbation to escape local minimum
                ptype = it % 4
                if ptype == 0:
                    # Mirror blend
                    alpha = rng.uniform(0.15, 0.45)
                    a_pert = (1.0 - alpha) * best_a + alpha * best_a[::-1].copy()
                elif ptype == 1:
                    # Circular shift
                    shift = int(rng.integers(1, max(2, n // 5)))
                    a_pert = np.roll(best_a, shift).copy()
                elif ptype == 2:
                    # Tilt
                    tilt = rng.uniform(0.2, 0.6)
                    ramp = np.linspace(1.0 - tilt, 1.0 + tilt, n)
                    a_pert = best_a * ramp
                else:
                    # Gaussian multiplicative
                    a_pert = best_a * np.exp(rng.normal(0, 0.06, n))

                a_pert = np.maximum(a_pert, 0)
                s = float(np.sum(a_pert))
                if s > 1e-10:
                    a_pert = a_pert / s
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj + 0.01:
                        best_a = a_pert
                        best_obj = obj_pert
                    else:
                        best_a = a_pert
                    stall = 0
                else:
                    stall = 0

        if verbose and it % 30 == 0:
            elapsed = time.monotonic()
            print(f"  LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, "
                  f"t={best_t:.3f}, stall={stall}, lp_t={lp_time:.3f}s")

    return best_a, best_obj
