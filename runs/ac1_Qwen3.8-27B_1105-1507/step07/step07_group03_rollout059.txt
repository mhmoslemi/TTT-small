But that's O(n log n) per block (FFT), so per sweep it's O(n * n log n) = O(n^2 log n). For n=800, that's about 800 * 7000 ≈ 5.6M operations per sweep. At 100M ops/sec, that's about 56ms per sweep. In 300 seconds, that's about 5000 sweeps. That should be plenty.

Actually, I can be smarter: I only need to recompute the parts of full_conv that are affected by the block update. But with FFT, it's easier to just recompute the whole thing.

Let me also consider: instead of recomputing full_conv with FFT, I can maintain it incrementally. When block b changes from a_old to a_new, the change in conv is:
delta_conv[k] = sum_{j in block} (a_new[j] - a_old[j]) * a[k-j] + sum_{j in block} a_old[j] * (a_new[k-j] - a_old[k-j])
Wait, that's not right either because both arguments change.

Actually: conv_new[k] = sum_i a_new[i] * a_new[k-i]
conv_old[k] = sum_i a_old[i] * a_old[k-i]
delta = conv_new - conv_old = sum_i (a_new[i]-a_old[i])*a_old[k-i] + sum_i a_old[i]*(a_new[k-i]-a_old[k-i]) + sum_i (a_new[i]-a_old[i])*(a_new[k-i]-a_old[k-i])

The first two terms involve only the changed block, so they're easy to compute. The third term is the "second order" correction which is small if the update is small.

For simplicity, let me just recompute full_conv with FFT after each block update. It's fast enough.

"""
Block LP alternating optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy HiGHS solver for fast LP solves with block decomposition.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def fft_flat_spectrum(a, n_iters=300, time_budget=30.0, relaxation=0.5):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe
        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 40:
                    rng = np.random.default_rng(it + 1000)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

    return best_a


def mirror_descent_fast(a, n_iters=3000, time_budget=60.0):
    """Fast mirror descent with FFT-based subgradient computation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = np.maximum(a, 1e-15) / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.05
    m_peaks = 8

    # Precompute FFT size
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[-m_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        temp = max(0.001, (max_val - top_val[0]) * 0.3)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        # Compute subgradient using FFT
        # grad[i] = sum_k w_k * a[k-i]
        # This is a weighted sum of shifted versions of a
        # grad = sum_k w_k * reverse_shifted_a(k)
        
        # Alternative: use the fact that conv = a * a, and the gradient
        # of conv[k] w.r.t. a[i] is a[k-i] (for valid indices)
        # So grad[i] = sum_k w_k * a[(k-i) mod ... ] ... 
        # Actually it's the linear convolution, not circular.
        
        # For linear convolution: conv[k] = sum_i a[i]*a[k-i]
        # d conv[k] / d a[i] = a[k-i] (if 0 <= k-i < n)
        # So grad[i] = sum_k w_k * a[k-i] (if 0 <= k-i < n)
        #             = sum_k w_k * a[|k-i|] ... no, it's a[k-i] specifically
        
        # grad[i] = sum_{k=i}^{i+n-1} w_k * a[k-i]  (only terms where 0<=k-i<n)
        #         = sum_{j=0}^{n-1} w_{i+j} * a[j]
        
        # This is a cross-correlation of w (padded) with a!
        # Let W = w padded to length 2n-1 (w only has values at top_idx)
        # grad[i] = sum_j W[i+j] * a[j]
        # This is the (2n-1 - i - 1)th element of conv(W, a[::-1])... 
        
        # Actually, let me just compute it directly for the few active peaks
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            # a[k-i] for i in [0,n): valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            i_min = max(0, ki - n + 1)
            i_max = min(n - 1, ki)
            if i_min <= i_max:
                grad[i_min:i_max+1] += w * a[i_max - (i_max - i_min):][::-1]
                # Wait, let me be more careful
                # For i in [i_min, i_max]: a[k-i]
                # k-i ranges from k-i_max to k-i_min
                # a[k-i_max : k-i_min + 1] reversed
                # k-i_max = ki - i_max, k-i_min = ki - i_min
                # So a[ki-i_max : ki-i_min+1]
                # And we assign to grad[i_min : i_max+1]
                # grad[i_min] = w * a[ki - i_max]
                # grad[i_max] = w * a[ki - i_min]
                # So grad[i_min:i_max+1] = w * a[ki-i_max : ki-i_min+1][::-1]
        
        # Hmm this is getting confusing. Let me just do it simply:
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            for i in range(max(0, ki-n+1), min(n, ki+1)):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * a[j]

        # Mirror descent step
        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.97
        else:
            alpha *= 0.5

        if it % 500 == 499:
            alpha = min(alpha * 1.05, 0.2)

    return best_a


def block_lp_optimize(a, max_time=300.0, verbose=False):
    """Alternating block LP with decreasing block sizes."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Block LP start: n={n}, obj={best_obj:.8f}")

    # Start with larger blocks, go to smaller
    block_sizes = [200, 100, 50, 25, 10]
    
    for block_size in block_sizes:
        if time.monotonic() - start > max_time:
            break
        
        n_blocks = (n + block_size - 1) // block_size
        stalls = 0
        max_stalls = 5
        rng = np.random.default_rng(hash((n, block_size)) % (2**32))

        for sweep in range(50000):
            if time.monotonic() - start > max_time:
                break
            if stalls > max_stalls:
                break

            improved = False
            
            # Random block order
            block_order = list(range(n_blocks))
            rng.shuffle(block_order)
            
            for b in block_order:
                if time.monotonic() - start > max_time:
                    break

                lo = b * block_size
                hi = min(n, lo + block_size)
                B = hi - lo

                a_block = a[lo:hi]
                block_sum = float(np.sum(a_block))

                if block_sum < 1e-12:
                    continue

                # Compute full conv
                full_conv = compute_conv(a)

                # Block contribution to conv
                block_contrib = np.zeros(2*n-1)
                for j in range(B):
                    gi = lo + j
                    ai = a[gi]
                    if ai < 1e-15:
                        continue
                    k_start = gi
                    k_end = min(gi + n, 2*n-1)
                    if k_start < 2*n-1:
                        block_contrib[k_start:k_end] += a[0:k_end-k_start] * ai

                fixed = full_conv - block_contrib

                # Build LP constraint matrix
                # Only include rows where block contributes
                k_min = max(0, lo)
                k_max = min(2*n-2, lo + B - 1 + n - 1)

                A_rows = []
                b_rows = []

                for k in range(k_min, k_max + 1):
                    row = np.zeros(B)
                    any_nonzero = False
                    for j in range(B):
                        ki = k - (lo + j)
                        if 0 <= ki < n:
                            val = a[ki]
                            if val > 1e-15:
                                row[j] = val
                                any_nonzero = True
                    if any_nonzero:
                        A_rows.append(row)
                        b_rows.append(-fixed[k])

                if not A_rows:
                    continue

                A_ub_block = np.array(A_rows)
                b_ub_arr = np.array(b_rows)

                # Add M column
                A_ub = np.hstack([A_ub_block, -np.ones((len(A_ub_block), 1))])

                A_eq = np.zeros((1, B + 1))
                A_eq[0, :B] = 1.0
                b_eq = np.array([block_sum])

                c = np.zeros(B + 1)
                c[B] = 1.0

                bounds = [(0, None)] * (B + 1)

                try:
                    res = linprog(c, A_ub=A_ub, b_ub=b_ub_arr,
                                  A_eq=A_eq, b_eq=b_eq,
                                  bounds=bounds, method='highs')
                except Exception:
                    continue

                if res.status == 0:
                    g_new = np.maximum(res.x[:B], 0)
                    gs = float(np.sum(g_new))
                    if gs > 1e-10:
                        g_new = g_new * (block_sum / gs)

                    a_new = a.copy()
                    a_new[lo:hi] = g_new
                    s_new = float(np.sum(a_new))
                    if s_new < 1e-10:
                        continue
                    a_new = a_new / s_new

                    obj_new = compute_obj(a_new)

                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        improved = True

            if not improved:
                stalls += 1

        if verbose:
            print(f"  Block size {block_size}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Block LP alternating optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy HiGHS solver for fast LP solves with adaptive block decomposition,
FFT flat-spectrum initialization, mirror descent polishing, and multi-n exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def fft_flat_spectrum(a, n_iters=300, time_budget=30.0, relaxation=0.5):
    """FFT-based flat spectrum iteration (Gerchberg-Saxton variant)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe
        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 40:
                    rng = np.random.default_rng(it + 2000)
                    a = best_a * np.exp(rng.normal(0, 0.04, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

    return best_a


def mirror_descent_fast(a, n_iters=3000, time_budget=60.0, m_peaks=8):
    """Fast mirror descent (exponential weights) with multi-peak subgradient."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = np.maximum(a, 1e-15) / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.05

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[-m_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        temp = max(0.001, (max_val - top_val[0]) * 0.3)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            i_min = max(0, ki - n + 1)
            i_max = min(n - 1, ki)
            if i_min <= i_max:
                grad[i_min:i_max + 1] += w * a[ki - i_max:ki - i_min + 1][::-1]

        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.97
        else:
            alpha *= 0.5

        if it % 500 == 499:
            alpha = min(alpha * 1.05, 0.2)

    return best_a


def solve_block_lp(a, lo, hi):
    """
    Solve LP for one block: min M s.t. conv constraints, sum(block)=block_sum, g>=0.
    Returns the optimal block values, or None on failure.
    """
    n = len(a)
    B = hi - lo
    block_sum = float(np.sum(a[lo:hi]))
    if block_sum < 1e-12:
        return None

    full_conv = compute_conv(a)

    # Block contribution to convolution
    block_contrib = np.zeros(2 * n - 1)
    for j in range(B):
        gi = lo + j
        ai = a[gi]
        if ai < 1e-15:
            continue
        k_start = gi
        k_end = min(gi + n, 2 * n - 1)
        if k_start < 2 * n - 1:
            block_contrib[k_start:k_end] += a[0:k_end - k_start] * ai

    fixed = full_conv - block_contrib

    # Build constraint matrix: only rows where block contributes
    k_min = max(0, lo)
    k_max = min(2 * n - 2, lo + B - 1 + n - 1)

    A_rows = []
    b_rows = []

    for k in range(k_min, k_max + 1):
        row = np.zeros(B)
        any_nz = False
        for j in range(B):
            ki = k - (lo + j)
            if 0 <= ki < n:
                val = a[ki]
                if val > 1e-14:
                    row[j] = val
                    any_nz = True
        if any_nz:
            A_rows.append(row)
            b_rows.append(-fixed[k])

    if not A_rows:
        return None

    A_ub_block = np.array(A_rows)
    b_ub_arr = np.array(b_rows)

    A_ub = np.hstack([A_ub_block, -np.ones((len(A_ub_block), 1))])
    A_eq = np.zeros((1, B + 1))
    A_eq[0, :B] = 1.0
    b_eq = np.array([block_sum])

    c = np.zeros(B + 1)
    c[B] = 1.0
    bounds = [(0, None)] * (B + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub_arr,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None

    if res.status == 0:
        g_new = np.maximum(res.x[:B], 0)
        gs = float(np.sum(g_new))
        if gs > 1e-10:
            g_new = g_new * (block_sum / gs)
            return g_new
    return None


def block_lp_optimize(a, max_time=300.0, verbose=False):
    """
    Alternating block LP with decreasing block sizes.
    Large blocks for coarse optimization, small blocks for fine-tuning.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Block LP start: n={n}, obj={best_obj:.8f}")

    block_sizes = [200, 100, 50, 25, 10, 5]

    for block_size in block_sizes:
        if time.monotonic() - start > max_time:
            break
        if block_size > n:
            continue

        n_blocks = (n + block_size - 1) // block_size
        stalls = 0
        max_stalls = 5
        rng = np.random.default_rng(hash((n, block_size, int(start_time % 10000))) % (2**32))

        sweep_count = 0
        for sweep in range(100000):
            if time.monotonic() - start > max_time:
                break
            if stalls > max_stalls:
                break

            improved = False
            block_order = list(range(n_blocks))
            rng.shuffle(block_order)

            for b in block_order:
                if time.monotonic() - start > max_time:
                    break

                lo = b * block_size
                hi = min(n, lo + block_size)

                g_new = solve_block_lp(a, lo, hi)
                if g_new is None:
                    continue

                a_new = a.copy()
                a_new[lo:hi] = g_new
                s_new = float(np.sum(a_new))
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new

                obj_new = compute_obj(a_new)

                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True

            sweep_count += 1
            if not improved:
                stalls += 1
            else:
                stalls = 0

        if verbose:
            elapsed = time.monotonic() - start
            print(f"  Block size {block_size}: obj={best_obj:.8f}, sweeps={sweep_count}, time={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        freqs = rng.uniform(0, 4, 8)
        phases = rng.uniform(0, 2 * np.pi, 8)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        c1, c2 = n // 3, 2 * n // 3
        w = max(2, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    else:
        a = rng.uniform(0, 1, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    return a
