"""
Smooth-max optimization with FFT gradients for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp approximation with adaptive temperature, LP refinement, and multi-n search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def gradient_smooth_max(a, w):
    """Gradient of sum_k w_k * (a*a)_k with respect to a.
    Result: g[j] = 2 * sum_k w_k * a_{k-j} for 0 <= k-j < n.
    This is 2 * cross_correlation(w, a).
    """
    n = len(a)
    size = 1
    while size < 2*n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    # Cross-correlation: (w cross a)[j] = sum_k w[k] * a[k-j]
    # = ifft(conj(fft(w)) * fft(a))
    g = irfft(np.conj(W) * A, size)[:n] * 2.0
    return g

def smooth_max_optimize(a0, time_budget=100.0, verbose=False):

    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam parameters
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    m = np.zeros(n)
    v = np.zeros(n)
    
    # Temperature schedule
    T = 5.0  # Start moderate
    T_mult = 1.002  # Slow increase
    T_max = 200.0
    
    stall = 0
    patience = 500
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Compute convolution
        conv = compute_conv(a)
        
        # Compute softmax weights
        logits = T * conv
        logits -= logits.max()
        w = np.exp(logits)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.ones(2*n-1) / (2*n-1)
        
        # Compute gradient
        g = gradient_smooth_max(a, w)
        
        # Center the gradient (project to tangent space of simplex)
        g = g - g.sum() * a / max(s, 1e-10)  # Hmm, this is not quite right
        # Actually for simplex constraint, we just need to ensure the step stays in the simplex
        # Let's just use the raw gradient and project after
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new

            best_a = a.copy()
