```python
    theta = (cssv[rho] - 1) / (rho + 1.0)
```

```python
def minimax_optimize(a_init, time_budget=60.0):
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, np.inf
    a = a / s
    
    L = 1
    while L < 3*n - 1:
        L <<= 1
    
    y = np.ones(2*n-1) / (2*n-1)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    outer_iter = 0
    inner_iters = 50
    step = 0.4
    tau_y = 5.0
    
    while time.monotonic() - start < time_budget:
        outer_iter += 1
        a_rev = a[::-1]
        
        # Inner FISTA
        a_curr = a.copy()
        a_prev = a.copy()
        t_f = 1.0
        
        for _ in range(inner_iters):
            z = a_curr + (t_f - 1) / (t_f + 1) * (a_curr - a_prev)
            # Don't project z (FISTA doesn't require it for the extrapolation point)
            # But clip to avoid numerical issues
            # Compute gradient at z
            z_rev = z[::-1]
            F_z = rfft(z_rev, L)
            F_y_cached = rfft(y, L)  # cache this
            grad = irfft(F_z * F_y_cached, L)[n-1:2*n-1]
            grad = 2.0 * grad
            
            # Projected gradient step
            a_new = project_simplex(a_curr - step * grad)
            
            t_f = (1 + np.sqrt(1 + 4 * t_f * t_f)) / 2
            a_prev = a_curr
            a_curr = a_new
        
        a = a_curr
        conv = compute_conv(a)
        obj = 2.0 * n * np.max(conv)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Update y
        tau_y = max(tau_y * 0.99, 0.5)
        conv_shifted = conv - np.max(conv)
        y = np.exp(conv_shifted / tau_y)
        y_sum = np.sum(y)
        if y_sum > 1e-10:
            y /= y_sum
        else:
            idx = np.argmax(conv)
            y = np.zeros(2*n-1)
            y[idx] = 1.0
    
    return best_a, best_obj
```

```python
def active_set_lp(a, K=20, time_budget=30.0):
    """LP with active set: minimize M s.t. conv(a,g)[k] <= M for top-K k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix for a given set of positions
    def solve_with_positions(positions):
        # A[g] = conv(a, g) at positions in 'positions'
        # conv(a,g)[k] = sum_j a[j] * g[k-j] (for valid j, k-j)
        # For each position k, the constraint is: sum_j a[j] * g[k-j] <= M
        # This is a linear constraint on g (and M)
        
        m = len(positions)
        # Variables: g[0..n-1], M
        # Constraints: for each k in positions: sum_j a[j]*g[k-j] - M <= 0
        #              sum(g) = 1
        #              g >= 0, M >= 0
        
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for i, k in enumerate(positions):
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    A_ub[i, j] = a[j]
            A_ub[i, n] = -1.0
        b_ub[:] = 0.0
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        return res
    
    # Start with top K peaks
    conv = compute_conv(a)
    positions = np.argsort(conv)[-K:]
    
    for iteration in range(50):
        res = solve_with_positions(positions)
        if res.status != 0:
            break
        
        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s < 1e-10:
            break
        g /= s
        M = res.x[n]
        
        # Check if all constraints are satisfied
        conv_ag = compute_conv_cross(a, g)  # Need this function
        max_violation = np.max(conv_ag) - M
        if max_violation < 1e-8:
            # All constraints satisfied, this is the optimal g
            return g, M
        
        # Add the most violated constraint
        k_violated = int(np.argmax(conv_ag))
        if k_violated not in positions:
            positions = np.append(positions, k_violated)
            if len(positions) > 100:
                # Keep only the worst 100
                ag_at_positions = np.array([conv_ag[k] for k in positions])
                positions = positions[np.argsort(ag_at_positions)][::-1][:100]
    
    return g, M
```

```python
def compute_cross_conv(a, b):
    """Compute full linear cross-convolution a * b."""
    n1, n2 = len(a), len(b)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n1 + n2 - 1]
```

```python
"""
Minimax optimization for minimum peak autocorrelation on the probability simplex.
Uses FFT-based Nesterov-accelerated subgradient with adaptive dual weights (softmax on peaks),
combined with active-set LP polishing, multi-n exploration, and diverse restarts.
Key innovation: the dual variable y tracks peak positions via adaptive softmax, and the
gradient Q@a is computed in O(n log n) via FFT cross-correlation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross_conv(a, b):
    """Full linear cross-convolution a*b via FFT."""
    n1, n2 = len(a), len(b)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n1 + n2 - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    """Project vector onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_Qa_fft(a, y, L, a_rev):
    """Compute Q*a where Q = sum_k y_k P_k, using FFT cross-correlation.
    Returns 2*h where h[i] = sum_j a[j]*y[i+j]."""
    n = len(a)
    F_a = rfft(a_rev, L)
    F_y = rfft(y, L)
    conv_res = irfft(F_a * F_y, L)
    h = conv_res[n - 1:2 * n - 1]
    return 2.0 * h


def minimax_nesterov(a_init, time_budget=60.0, step=0.4, tau_init=3.0,
                     tau_min=0.3, tau_decay=0.997, verbose=False):
    """
    Minimax optimization via Nesterov-accelerated subgradient.
    min_a max_k (a*a)[k] on the simplex.
    Uses FFT-based gradient computation and adaptive softmax dual.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, np.inf
    a = a / s

    L = 1
    while L < 3 * n - 1:
        L <<= 1

    y = np.ones(2 * n - 1) / (2 * n - 1)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    a_prev = a.copy()
    t_mom = 1.0
    it = 0
    y_update_interval = max(1, 10000 // max(n, 1))

    while time.monotonic() - start < time_budget:
        it += 1

        # Nesterov extrapolation
        beta = (t_mom - 1.0) / (t_mom + 1.0)
        a_extrap = a + beta * (a - a_prev)
        a_extrap = np.maximum(a_extrap, 0)
        s_e = float(np.sum(a_extrap))
        if s_e < 1e-10:
            a_extrap = a.copy()
        else:
            a_extrap = a_extrap / s_e

        # Compute gradient via FFT
        a_extrap_rev = a_extrap[::-1]
        grad = compute_Qa_fft(a_extrap, y, L, a_extrap_rev)

        # Projected gradient step
        a_new = project_simplex(a_extrap - step * grad)

        # Update momentum
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        a_prev = a
        a = a_new

        # Periodically compute objective and update dual
        if it % y_update_interval == 0:
            conv = compute_conv(a)
            obj = 2.0 * n * np.max(conv)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                if verbose and it % (y_update_interval * 10) == 0:
                    print(f"      Minimax it{it}: obj={best_obj:.8f}, tau={tau_init:.3f}")

            # Update dual variable y (softmax on peaks)
            tau_init = max(tau_init * tau_decay, tau_min)
            conv_shifted = conv - np.max(conv)
            with np.errstate(over='ignore'):
                y_new = np.exp(conv_shifted / tau_init)
            y_sum = float(np.sum(y_new))
            if y_sum > 1e-10 and np.isfinite(y_sum):
                y = y_new / y_sum
            else:
                # Fallback: top-10 uniform
                top_idx = np.argsort(conv)[-10:]
                y = np.zeros(2 * n - 1)
                y[top_idx] = 0.1

    # Final check
    conv = compute_conv(a)
    obj = 2.0 * n * np.max(conv)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def active_set_lp_solve(a, max_positions=30, max_iters=30, time_budget=30.0):
    """
    Active-set LP: find g minimizing max_k conv(a,g)[k] subject to sum(g)=1, g>=0.
    Uses constraint generation (starts with top-K, adds violated constraints).
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    a = a / float(np.sum(a))
    start = time.monotonic()

    conv_a = compute_conv(a)
    positions = np.argsort(conv_a)[-min(max_positions, 2 * n - 1)]

    g_best = None
    M_best = np.inf

    for i in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        m = len(positions)
        # Build LP: min M s.t. A_ub @ [g; M] <= 0, A_eq @ [g; M] = 1, g>=0, M>=0
        A_ub = np.zeros((m, n + 1))
        for idx, k in enumerate(positions):
            k = int(k)
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    A_ub[idx, j] = a[j]
            A_ub[idx, n] = -1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        c = np.zeros(n + 1)
        c[n] = 1.0

        try:
            res = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 5})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        g = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g))
        if s_g < 1e-10:
            break
        g = g / s_g
        M_val = float(res.x[n])

        # Check full feasibility
        conv_ag = compute_cross_conv(a, g)
        max_viol = float(np.max(conv_ag)) - M_val

        if max_viol < 1e-7:
            g_best = g
            M_best = M_val
            break

        # Add violated constraint
        k_viol = int(np.argmax(conv_ag))
        if k_viol not in [int(p) for p in positions]:
            positions = np.append(positions, k_viol)
            # Trim if too many
            if len(positions) > 80:
                vals = conv_ag[np.array(positions, dtype=int)]
                keep = np.argsort(vals)[-60:]
                positions = np.array(positions, dtype=int)[keep]
        else:
            # Already in set but still violated - increase M bound
            # Just accept current solution
            g_best = g
            M_best = M_val
            break

    if g_best is None:
        # Fallback: use LP solution even if not fully feasible
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g))
            if s_g > 1e-10:
                g_best = g / s_g
                M_best = float(res.x[n])

    return g_best, M_best

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```
