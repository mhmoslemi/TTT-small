import time
from scipy.fft import rfft, irfft

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))

def peak_subgrad_ffft(a, active_k):
    """Compute subgradient of max(conv(a,a)) using active peaks active_k."""
    n = len(a)
    w = np.zeros(2 * n - 1)
    for k in active_k:
        w[k] = 1.0
    w /= len(active_k)

    N = 1
    while N < n + 2 * n - 1:
        N <<= 1
    fa_rev = rfft(a[::-1], N)
    fw = rfft(w, N)
    conv_result = irfft(fa_rev * fw, N)
    return conv_result[n-1:2*n-1]

    n = len(a)

    fa_rev = rfft(a[::-1], N)
    fb = rfft(b, N)
    conv_result = irfft(fa_rev * fb, N)
    return 4.0 * conv_result[n-1:2*n-1]

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (exponentiated gradient) with adaptive peak subgradients,
energy-based smoothing, multi-length exploration, LP polishing, and stochastic refinement.
Key innovations: FFT-accelerated subgradient computation, adaptive active-peak selection,
dual gradient mixing, and intelligent multi-n search strategy.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def compute_conv_and_obj(a):
    """Compute both conv and objective together."""
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    b = compute_conv(a_norm)
    obj = 2.0 * n * float(np.max(b))
    return b, obj


def peak_subgrad_fft(a, active_k, N=None):
    """Compute subgradient of max(conv(a,a)) w.r.t. a using active peaks.
    Returns grad of shape (n,) such that grad[i] = sum_{k in active} a[k-i] / |active| (for valid k-i)."""
    n = len(a)
    w = np.zeros(2 * n - 1)
    w[active_k] = 1.0
    w /= len(active_k)

    if N is None:
        N = 1
        while N < n + 2 * n - 1:
            N <<= 1

    a_rev = a[::-1]
    fa_rev = rfft(a_rev, N)
    fw = rfft(w, N)
    conv_result = irfft(fa_rev * fw, N)
    return conv_result[n - 1:2 * n - 1]


def energy_grad_fft(a, b, N=None):
    """Compute gradient of sum(b[k]^2) w.r.t. a.
    grad[i] = 4 * sum_k b[k] * a[k-i] (for valid k-i)."""
    n = len(a)
    if N is None:
        N = 1
        while N < n + 2 * n - 1:
            N <<= 1

    a_rev = a[::-1]
    fa_rev = rfft(a_rev, N)
    fb = rfft(b, N)
    conv_result = irfft(fa_rev * fb, N)
    return 4.0 * conv_result[n - 1:2 * n - 1]


def mirror_descent(a, time_budget, mode='peak', eta0=0.005, verbose=False,
                   peak_fraction=0.1, use_energy_mix=0.0, seed=42):
    """
    Mirror descent (exponentiated gradient) for minimizing max(conv(a,a)) on the simplex.
    mode: 'peak' (subgradient of max), 'energy' (gradient of sum of squares), 'mixed'
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta0
    rng = np.random.default_rng(seed)

    N_conv = 1
    while N_conv < 2 * n - 1:
        N_conv <<= 1
    N_grad = 1
    while N_grad < 3 * n - 1:
        N_grad <<= 1

    stall = 0
    max_stall = 200

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        # Compute auto-convolution
        fa = rfft(a, N_conv)
        b = irfft(fa * fa, N_conv)[:2 * n - 1]
        b_max = float(np.max(b))

        if mode == 'peak':
            # Adaptive active peak selection
            # Use peaks within a dynamic threshold
            threshold = max(0.005 * b_max, b_max - 0.02 * b_max)
            active_k = np.where(b >= threshold)[0]
            if len(active_k) == 0:
                active_k = np.array([int(np.argmax(b))])
            # Limit to top fraction for focus
            if len(active_k) > max(1, int(peak_fraction * (2 * n - 1))):
                # Sort by b value and take top ones
                order = np.argsort(b[active_k])[::-1]
                n_keep = max(1, int(peak_fraction * (2 * n - 1)))
                active_k = active_k[order[:n_keep]]

            grad = peak_subgrad_fft(a, active_k, N_grad)
            # Scale by 2n for the full objective
            grad *= 2.0 * n

        elif mode == 'energy':
            grad = energy_grad_fft(a, b, N_grad)
            grad *= n / 2.0

        else:  # mixed
            threshold = 0.01 * b_max
            active_k = np.where(b >= b_max - threshold)[0]
            if len(active_k) == 0:
                active_k = np.array([int(np.argmax(b))])
            grad_peak = peak_subgrad_fft(a, active_k, N_grad) * 2.0 * n
            grad_energy = energy_grad_fft(a, b, N_grad)
            # Normalize energy grad to similar scale
            if np.max(np.abs(grad_energy)) > 1e-10:
                scale = np.max(np.abs(grad_peak)) / np.max(np.abs(grad_energy))
                grad_energy *= scale
            grad = (1.0 - use_energy_mix) * grad_peak + use_energy_mix * grad_energy

        # Center the gradient (remove mean to preserve sum approximately)
        grad = grad - np.mean(grad)

        # Mirror descent update
        update = -eta * grad
        update = np.clip(update, -30, 30)
        a_new = a * np.exp(update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        # Evaluate
        fa_new = rfft(a_new, N_conv)
        b_new = irfft(fa_new * fa_new, N_conv)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Adaptive step size: increase on success
            eta = min(eta * 1.003, 0.5)
        else:
            stall += 1
            eta *= 0.98
            if stall > max_stall:
                # Perturbation restart
                noise = rng.normal(0, 0.03, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                eta = eta0
                stall = 0
                max_stall = min(max_stall + 100, 500)
            if eta < 1e-12:
                break

        if verbose and it % 2000 == 0:
            print(f"      MD({mode}) it{it}: obj={best_obj:.8f}, eta={eta:.6f}")

    return best_a


def lp_polish(a, time_budget=30.0, verbose=False):
    """LP-based alternating optimization for final polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP: min M s.t. A@M_row <= M, sum(g)=1, g>=0
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'time_limit': time_budget / 50})
            if res.status == 0 and res.x is not None:
                g_val = np.maximum(res.x[:n], 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    g_val = g_val / s_g
                    # Try different blend factors
                    best_t_obj = np.inf
                    best_t = 0.3
                    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * a + t * g_val
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_t = t

                    if best_t_obj < best_obj - 1e-12:
                        best_obj = best_t_obj
                        a_new = (1 - best_t) * a + best_t * g_val
                        a_new = np.maximum(a_new, 0)
                        a_new = a_new / np.sum(a_new)
                        best_a = a_new.copy()
                        a = a_new
                        stall = 0
                        # Update A for next iteration
                        A = np.zeros((2 * n - 1, n))
                        for j in range(n):
                            r = min(n, 2 * n - 1 - j)
                            A[j:j + r, j] = a[:r]
                        if verbose:
                            print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
                    else:
                        stall += 1
                        if stall > 10:
                            break
                else:
                    break
            else:
                break
        except Exception:
            break

    return best_a


def stochastic_refine(a, time_budget, n_restarts=5, verbose=False):
    """Stochastic perturbation + short MD to escape local minima."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    per_restart = time_budget / max(n_restarts, 1)

    for r in range(n_restarts):
        if time.monotonic() - start > time_budget:
            break

        # Generate perturbation
        noise_scale = 0.05 + 0.05 * r / max(n_restarts - 1, 1)
        noise = rng.normal(0, noise_scale, n)
        a_try = best_a * np.exp(noise)
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / np.sum(a_try)

        # Short MD from perturbed point
        alloc = min(per_restart * 0.7, 20.0)
        a_opt = mirror_descent(a_try, time_budget=alloc, mode='peak', eta0=0.003)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"      Stochastic restart {r}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Smooth random
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a / a[-1] + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Cosine-modulated
        x = np.arange(n) / n
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x * rng.integers(1, 5)) + rng.uniform(0, 0.1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 4:
        # Periodic-like
        period = int(n / rng.integers(2, 6))
        a = np.ones(n)
        for i in range(n):
            a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * i / period + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    else:
        # Gaussian-windowed
        center = n // 2
        width = n / rng.uniform(2, 8)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2)) + 0.05
        a = a / np.sum(a)
    return a

            best_seq = a_mix.copy()
            print(f"  ** MD Mixed: obj={best_score:.8f} **")
        else:
            print(f"  MD Mixed: obj={obj_mix:.8f}")

    # === PHASE 3: MD Energy mode (smoother) ===
    print(f"\n=== Phase 3: MD Energy ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.15, 150.0)
        a_eng = mirror_descent(best_seq, time_budget=alloc, mode='energy', eta0=0.002)
        obj_eng = compute_obj(a_eng)
        if obj_eng < best_score:
            best_score = obj_eng
            best_seq = a_eng.copy()
            print(f"  ** MD Energy: obj={best_score:.8f} **")
        else:
            print(f"  MD Energy: obj={obj_eng:.8f}")

    # === PHASE 4: Second MD Peak round (finer) ===
    print(f"\n=== Phase 4: MD Peak Round 2 (finer) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 150.0)
        a_md2 = mirror_descent(best_seq, time_budget=alloc, mode='peak', eta0=0.002,
                               peak_fraction=0.05, verbose=True)
        obj_md2 = compute_obj(a_md2)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  ** MD Peak2: obj={best_score:.8f} **")
        else:
            print(f"  MD Peak2: obj={obj_md2:.8f}")

    # === PHASE 5: Stochastic Refinement ===
    print(f"\n=== Phase 5: Stochastic Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.12, 100.0)
        a_stoch = stochastic_refine(best_seq, time_budget=alloc, n_restarts=6, verbose=True)
        obj_stoch = compute_obj(a_stoch)
        if obj_stoch < best_score:
            best_score = obj_stoch
            best_seq = a_stoch.copy()
            print(f"  ** Stochastic: obj={best_score:.8f} **")
        else:
            print(f"  Stochastic: obj={obj_stoch:.8f}")

    # === PHASE 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr + 100, n_curr + 200, n_curr + 400,
        1200, 1500, 1800, 2000, 2400, 3000
    ]))
    n_options = [n for n in n_options if 400 <= n <= 3500 and n != n_curr]
    n_options = n_options[:5]  # Limit to 5 different n values

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Short MD on new n
        alloc = min(remaining * 0.25, 60.0)
        a_opt = mirror_descent(a_try, time_budget=alloc, mode='peak', eta0=0.004,
                               peak_fraction=0.1)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === PHASE 7: LP Polish ===
    print(f"\n=== Phase 7: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.20, 120.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === PHASE 8: Final MD polish ===
    print(f"\n=== Phase 8: Final MD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 100.0)
        a_final = mirror_descent(best_seq, time_budget=alloc, mode='peak', eta0=0.001,
                                 peak_fraction=0.03)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final MD: obj={best_score:.8f} **")

    # === PHASE 9: One more stochastic attempt ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_final2 = stochastic_refine(best_seq, time_budget=alloc, n_restarts=4)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  ** Final Stochastic: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
