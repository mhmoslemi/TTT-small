    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_grad_topk(a, K, power):
    """
    Compute gradient of sum of top-K convolution bins (raised to 'power').
    Returns gradient w.r.t. a (normalized, sum=1).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Get top-K bins
    top_k_idx = np.argsort(conv)[-K:]
    top_k_vals = conv[top_k_idx]
    
    # Weights: emphasize the largest bins
    weights = top_k_vals ** power
    
    # Gradient: grad[j] = 2 * sum_i w_i * a[k_i - j]
    grad = np.zeros(n)
    for ki, wi in zip(top_k_idx, weights):
        ki = int(ki)
        # a[ki - j] is non-zero when 0 <= ki-j <= n-1, i.e., ki-(n-1) <= j <= ki
        j_start = max(0, ki - (n - 1))
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            grad[j_start:j_end+1] += wi * a[ki - j_end:ki - j_start:-1]
    
    # Normalize weights
    if weights.sum() > 0:
        grad /= weights.sum()
    
    return grad, float(np.max(conv))


def sharpened_gradient_descent(a, time_budget=60.0, K=20, power=4.0, 
                                step_init=0.05, verbose=False):
    """
    Gradient descent on the simplex with top-K sharpened gradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = step_init
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        # Compute gradient
        grad, max_conv = compute_grad_topk(a, K, power)
        
        # Project gradient onto tangent space of simplex
        grad_proj = grad - np.mean(grad)
        
        # Take step
        a_new = a - step * grad_proj
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            step *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            step = min(step * 1.05, 0.5)
            stall_count = 0
        else:
            step *= 0.7
            stall_count += 1
        
        if verbose and stall_count % 50 == 0:
            print(f"        SGD: obj={best_obj:.8f}, step={step:.4f}, max_conv={max_conv:.6f}")
        
        if stall_count > 200:
            # Try to escape
            a = best_a * np.exp(np.random.default_rng(int(time.monotonic()*1000)).normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            step = step_init * 0.5
            stall_count = 0
    
    return best_a, best_obj


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):

        A_ub[:, n] = -1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def peak_shaving(a, time_budget=30.0):
    """
    Direct peak shaving: identify the peak bin and move mass to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find which pairs (i, j) contribute to the peak: i + j = k_star
        # The peak value is sum_i a[i]*a[k_star-i]
        # To reduce it, decrease a[i] for large a[i]*a[k_star-i]
        
        # Compute contribution of each a[j] to the peak
        contrib = np.zeros(n)
        j_start = max(0, k_star - (n - 1))
        j_end = min(n - 1, k_star)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[k_star - j]
        
        # Find the j with highest contribution (excluding j = k_star/2 if integer)
        # Move mass from high-contribution j to low-contribution j'
        
        # Get sorted contributions
        order = np.argsort(contrib)[::-1]
        
        improved = False
        for j_src in order[:10]:
            if contrib[j_src] < 1e-15:
                continue
            # Find best destination
            for j_dst in np.argsort(contrib)[:10]:
                if j_dst == j_src:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * a[j_src]
                    if delta > a[j_src] * 0.9:
                        continue
                    a[j_src] -= delta
                    a[j_dst] += delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    a = a / s
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a = best_a.copy()  # revert
                        break
        
        if not improved:
            # Random perturbation
            rng = np.random.default_rng(int(time.monotonic() * 10000))
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            if compute_obj(a) > best_obj * 1.01:
                break
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
