import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak_active(a, active_k=None, verbose=False):
    """
    Solve LP: min M s.t. (A@g)[k] <= M for active_k, sum(g)=1, g>=0.
    If active_k is None, use all constraints.
    Returns (g_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if active_k is None:
        active_k = list(range(2 * n - 1))
    else:
        active_k = list(active_k)
    
    # Build reduced constraint matrix
    A_full = build_conv_matrix(a_norm)
    A_red = A_full[active_k, :]
    m = len(active_k)
    
    # Try cvxpy first
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    y = cp.Variable(m, nonneg=True)
    constraints = [A_red @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get dual for active constraints
                    dual_y = None
                    try:
                        # Re-solve to get duals
                        prob2 = cp.Problem(cp.Minimize(M), constraints)
                        prob2.solve(solver=solver, verbose=False)
                        # Extract dual from the constraint
                    except Exception:
                        pass
                    return g_val, float(M.value), dual_y
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A_red
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n]), res.eqlin
    except Exception:
        pass
    
    return None, None, None


def solve_lp_regularized(a, lam=0.01, active_k=None):
    """
    Solve LP with entropy regularization:
    min M + lam * sum(g^2) s.t. (A@g)[k] <= M, sum(g)=1, g>=0
    This is a QP.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if active_k is None:
        active_k = list(range(2 * n - 1))
    
    A_full = build_conv_matrix(a_norm)
    A_red = A_full[active_k, :]
    m = len(active_k)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_red @ g <= M, cp.sum(g) == 1]
    objective = cp.Minimize(M + lam * cp.sum_squares(g))
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def lp_alternating_improved(a, time_budget=200.0, active_K=50, verbose=False):

    Improved LP alternating with active set management and adaptive step size.
    """

def subgradient_descent_improved(a, time_budget=100.0, verbose=False):
    """
    Subgradient method with Nesterov acceleration and top-K peak targeting.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    momentum = 0.9
    v = np.zeros(n)
    lr = 1.0 / np.sqrt(n)
    stall = 0
    K = 5  # Top-K peaks

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Top-K peaks with softmax weighting
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break
        weights = np.exp((peak_vals - max_val) * 3.0 / max_val)
        weights /= weights.sum()

        # Compute weighted subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            offset = k
            i_start = max(0, offset - (n - 1))
            i_end = min(n - 1, offset)
            if i_start <= i_end:
                j_start = offset - i_end
                j_end = offset - i_start
                g[i_start:i_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]

        # Nesterov step
        v = momentum * v + g
        a_try = a - lr * v
        a_try = project_simplex(a_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            stall = 0
            lr = max(1e-6, lr * 0.999)
            momentum = min(0.95, momentum + 0.0005)
        else:
            stall += 1
            if stall > 100:
                # Restart with perturbation
                rng = np.random.default_rng(it * 97 + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.8
                lr = max(1e-6, lr * 0.5)
                stall = 0
            else:
                lr = max(1e-6, lr * 0.97)

        it += 1

    return best_a


def smooth_sequence(a, sigma=1.0):
    """Apply Gaussian smoothing to the sequence."""
    n = len(a)
    if n < 5:
        return a.copy()
    # Create Gaussian kernel
    kernel_size = min(n, max(3, int(4 * sigma)))
    if kernel_size % 2 == 0:
        kernel_size += 1
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-x**2 / (2 * sigma**2))
    kernel /= kernel.sum()
    # Apply with 'same' padding (edge-preserving)
    padded = np.pad(a, kernel_size // 2, mode='edge')
    smoothed = np.convolve(padded, kernel, mode='valid')[:n]
    smoothed = np.maximum(smoothed, 0)
    s = float(np.sum(smoothed))
    if s > 1e-10:
        smoothed /= s
    return smoothed


def smoothing_optimize(a, time_budget=30.0):
    """Try different levels of smoothing and keep the best."""

def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce max peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False

        # Find top contributing pairs to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Find non-contributing positions
        contrib_set = set()
        for _, i, j in contributors:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_set]

        # Try moving mass from top contributors to non-contributors
        for contrib, i, j in contributors[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break
            for target in non_contrib[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = frac * a[i]
                    if a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try swapping mass between two contributors
            for ci in range(min(10, len(contributors))):
                for cj in range(ci + 1, min(10, len(contributors))):
                    _, i1, j1 = contributors[ci]
                    _, i2, j2 = contributors[cj]
                    delta = 0.05 * (a[i1] - a[i2])
                    if abs(delta) < 1e-12:
                        continue
                    a[i1] -= delta
                    a[i2] += delta
                    if a[i1] < 0 or a[i2] < 0:
                        a[i1] += delta
                        a[i2] -= delta
                        continue
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        contributors.sort(reverse=True)
                    else:
                        a[i1] += delta
                        a[i2] -= delta

            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    # Step 2: LP alternating (main workhorse)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_budget = min(remaining * 0.5, 50.0)
        a_lp = lp_alternating_improved(best_a, time_budget=lp_budget, active_K=50)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp
            if verbose:
                print(f"    LP: obj={best_obj:.8f}")

    # Step 3: Smoothing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_sm = smoothing_optimize(best_a, time_budget=min(remaining * 0.15, 10.0))
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_obj:
            best_obj = obj_sm
            best_a = a_sm
            if verbose:
                print(f"    Smooth: obj={best_obj:.8f}")

    # Step 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_cd = coordinate_descent_refine(best_a, time_budget=min(remaining * 0.2, 15.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd
            if verbose:
                print(f"    CD: obj={best_obj:.8f}")

    # Step 5: Final LP pass
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_lp2 = lp_alternating_improved(best_a, time_budget=min(remaining * 0.3, 20.0), active_K=30)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2
            if verbose:
                print(f"    LP2: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses improved LP alternating with active sets,
    subgradient methods, smoothing, coordinate descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_parent = 800
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 1: Intensive optimization on parent ===
    print(f"\n=== Phase 1: Intensive Optimization (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(remaining * 0.35, 400.0)

    if phase1_budget > 30:
        # Run full pipeline in two rounds
        for rnd in range(2):
            remaining = deadline - time.monotonic()
            alloc = min(phase1_budget / (2 - rnd), remaining * 0.6)
            if alloc < 15:
                break
            print(f"  Round {rnd+1}, budget={alloc:.0f}s")
            a_opt = full_pipeline(best_seq, time_budget=alloc, verbose=True)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** Improved: obj={best_score:.8f}")
            else:
                print(f"  No improvement: obj={obj_opt:.8f}")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 100),
        max(300, n_current - 50),
        n_current,
        n_current + 50,
        n_current + 100,
        n_current + 200,
        500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]

    # Sort by priority: closest to current first
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = set([n_current])
    for n_try in n_options[:10]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try in n_tried:
            continue
        n_tried.add(n_try)

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.2, 60.0)
        a_opt = full_pipeline(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Diverse restarts ===
    print(f"\n=== Phase 3: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Focus on asymmetric initializations
    restart_modes = [2, 4, 5, 6, 3, 7, 2, 4]  # More asymmetric modes
    n_restarts = min(len(restart_modes), max(2, int(remaining / 80)))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = restart_modes[r % len(restart_modes)]
        a_r = diverse_init(n_best, rng, mode=mode)

        # Blend with best solution
        for blend in [0.3, 0.5, 0.7]:
            a_mixed = (1 - blend) * a_r + blend * best_seq
            a_mixed = np.maximum(a_mixed, 0)
            s_m = float(np.sum(a_mixed))
            if s_m < 1e-10:
                continue
            a_mixed = a_mixed / s_m

            restart_budget = min(remaining * 0.15, 50.0)
            a_opt = full_pipeline(a_mixed, time_budget=restart_budget, verbose=False)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r} (mode={mode}, blend={blend}): obj={best_score:.8f} **")
                break
        else:
            # Try pure initialization
            restart_budget = min(remaining * 0.1, 40.0)
            a_opt = full_pipeline(a_r, time_budget=restart_budget, verbose=False)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r} (mode={mode}, pure): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): best obj={obj_r:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        # Multiple LP passes with decreasing active set
        for active_K in [30, 20, 10, 50, 80]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            lp_budget = min(remaining * 0.15, 30.0)
            a_lp = lp_alternating_improved(best_seq, time_budget=lp_budget, active_K=active_K)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP (K={active_K}): obj={best_score:.8f} **")
            else:
                print(f"  LP (K={active_K}): obj={obj_lp:.8f}")

    # Final smoothing + coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_sm = smoothing_optimize(best_seq, time_budget=min(remaining * 0.2, 10.0))
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Final smooth: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_cd = coordinate_descent_refine(best_seq, time_budget=min(remaining * 0.3, 15.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final = lp_alternating_improved(best_seq, time_budget=min(remaining * 0.4, 30.0), active_K=40)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
