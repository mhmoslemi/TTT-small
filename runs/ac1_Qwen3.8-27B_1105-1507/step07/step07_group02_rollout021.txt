    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build matrix A (2n-1 x n) where (A @ g)[:2n-1] = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        end = min(m, j + n)
        length = end - j
        A[j:end, j] = a[:length]
    return A


def solve_lp_for_g(a):
    """
    Solve LP: min M s.t. conv(a_hat, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_hat = a / s
    A = build_A(a_hat)
    m = 2 * n - 1

    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * (n + 1)

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs, max(0.0, float(res.x[n]))
    except Exception:
        pass

    try:
        import cvxpy as cp
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable(nonneg=True)
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None:
                    g = np.maximum(g_var.value, 0)
                    gs = float(np.sum(g))
                    if gs > 1e-10:
                        return g / gs, max(0.0, float(M_var.value))
            except Exception:
                continue
    except Exception:
        pass

    return None, None


def lp_alternate(a, time_budget=60.0, verbose=False, seed_offset=0):
    """LP alternating with adaptive step and structured perturbations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()
    t = 0.5
    stall = 0
    rng = np.random.default_rng(100000 + seed_offset)

    for it in range(400):
        if time.monotonic() - start > time_budget:
            break

        g, M_lp = solve_lp_for_g(a)
        if g is None:
            break

        t_candidates = sorted(set([
            0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0,
            max(0.01, t * 0.6), min(1.0, t * 1.4), t
        ]))

        best_t_obj = np.inf
        best_t_val = t
        best_a_try = best_a.copy()

        for t_try in t_candidates:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = obj_value(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t_try
                best_a_try = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a
            t = best_t_val
            stall = 0
            if verbose and (it % 15 == 0 or best_obj < 1.51):
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t:.3f}")
        else:
            stall += 1
            if stall >= 6:
                ptype = stall % 4
                if ptype == 0:
                    scale = min(0.12, 0.008 * stall)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                elif ptype == 1:
                    a = best_a.copy()
                    bs = max(10, n // 15)
                    si = int(rng.integers(0, max(1, n - bs)))
                    a[si:si + bs] *= np.exp(rng.normal(0, 0.06, bs))
                elif ptype == 2:
                    half = n // 2
                    pert = rng.normal(0, 0.025, half)
                    a = best_a.copy()
                    a[:half] *= (1.0 + pert)
                    a[half:] *= (1.0 - pert[:n - half])
                    a = np.maximum(a, 0)
                else:
                    a = best_a * np.exp(rng.normal(0, 0.15, n))

                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                if s2 > 1e-10:
                    a = a / s2
                    best_a = a.copy()
                    best_obj = obj_value(a)
                stall = 0

    return best_a


def smooth_grad_descent(a, time_budget=30.0):
    """Gradient descent on log-sum-exp smoothed peak of auto-convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()
    lr = 0.005
    T0 = 80.0

    for it in range(150):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        T_cur = max(30.0, T0 - it * 0.3)
        conv_shifted = conv - conv.max()
        exp_conv = np.exp(T_cur * conv_shifted)
        weights = exp_conv / exp_conv.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-8:
                continue
            lo = max(0, k - n + 1)
            hi = min(k + 1, n)
            for j in range(lo, hi):
                grad[j] += w * a[k - j]

        grad = grad - np.mean(grad)
        a_try = a - lr * grad
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        obj_try = obj_value(a_try)

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.8
            if lr < 1e-9:
                lr = 0.005

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random non-negative sequence."""
    mode = int(rng.integers(0, 6))
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        x = np.arange(n)
        c1 = n * rng.uniform(0.25, 0.4)
        c2 = n * rng.uniform(0.6, 0.75)
        w1 = n * rng.uniform(0.08, 0.2)
        w2 = n * rng.uniform(0.08, 0.2)
        a = np.exp(-(x - c1)**2 / (2 * w1**2)) + np.exp(-(x - c2)**2 / (2 * w2**2)) + 0.01
    elif mode == 3:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2)**2 + 0.1
    elif mode == 4:
        a = 1.0 / (1.0 + np.exp(-(np.arange(n) - n * 0.5) / (n * 0.15)))
        a = a + 0.05
    else:
        a = np.ones(n)
        for _ in range(3):
            c = n * rng.uniform(0.2, 0.8)
            w = n * rng.uniform(0.05, 0.15)
            a += np.exp(-(np.arange(n) - c)**2 / (2 * w**2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 0 else np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization using LP alternating, gradient descent,
    peak shaving, multi-length exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Phase 0: Get initial sequences from parent/random ---
    print("=== Phase 0: Initialization ===")
    inits = []

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000)
        if parent.size > 0 and np.isfinite(parent).all():
            sp = float(np.sum(parent))
            if sp > 0.01:
                pn = parent / sp
                sc = obj_value(pn)
                ev = evaluate_sequence([float(x) for x in pn])
                print(f"  Parent: n={len(pn)}, internal={sc:.8f}, evaluated={ev:.8f}")
                if ev < np.inf:
                    inits.append(pn)
                    best_score = min(best_score, ev)
                    best_seq = pn.copy()
                else:
                    inits.append(pn)
    except Exception:
        pass

    # Constant baseline
    n_const = 800
    const = np.ones(n_const) / n_const
    sc_c = evaluate_sequence([float(x) for x in const])
    print(f"  Constant n={n_const}: score={sc_c:.8f}")
    inits.append(const)
    best_score = min(best_score, sc_c)
    if best_seq is None or sc_c < best_score:
        best_score = sc_c
        best_seq = const.copy()

    # A few random inits
    for ri in range(4):
        a_r = random_init(n_const, rng)
        inits.append(a_r)

    print(f"  Initial best: score={best_score:.8f}, n={len(best_seq)}")

    # --- Phase 1: Optimize top candidates with LP alternating ---
    print(f"\n=== Phase 1: LP Alternating (budget ~600s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        # Sort inits by score
        scored_inits = []
        for a in inits:
            scored_inits.append((obj_value(a), a))
        scored_inits.sort()

        # Optimize the top 5 with generous time
        n_optimize = min(5, len(scored_inits))
        time_per = min(120.0, remaining * 0.6 / n_optimize)

        for idx in range(n_optimize):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            obj_init, a_init = scored_inits[idx]
            alloc = min(time_per, remaining * 0.5)
            print(f"  Optimizing init {idx} (init_obj={obj_init:.8f}), budget={alloc:.0f}s")
            a_opt = lp_alternate(a_init, time_budget=alloc, verbose=True, seed_offset=idx)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < np.inf and ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"    ** NEW BEST: score={best_score:.8f} **")
            else:
                print(f"    result: eval={ev:.8f}")

    # --- Phase 2: Smooth gradient descent on best ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 2: Gradient Descent (remaining={remaining:.0f}s) ===")
        a_g = best_seq / float(np.sum(best_seq))
        a_opt = smooth_grad_descent(a_g, time_budget=min(60.0, remaining * 0.3))
        ev = evaluate_sequence([float(x) for x in a_opt])
        if ev < np.inf and ev < best_score:
            best_score = ev
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: score={best_score:.8f} **")
        else:
            print(f"  Gradient result: {ev:.8f}")

    # --- Phase 3: Multi-length exploration ---
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        print(f"\n=== Phase 3: Multi-length (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 200),
            max(200, n_best - 100),
            n_best,
            min(2000, n_best + 100),
            min(2000, n_best + 200),
            400, 600, 1000, 1200, 1600
        ]))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_seq(best_seq, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try
            alloc = min(45.0, remaining * 0.15)
            a_opt = lp_alternate(a_try, time_budget=alloc, seed_offset=n_try)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < np.inf and ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: NEW BEST score={best_score:.8f} **")
            else:
                print(f"  n={n_try}: score={ev:.8f}")

    # --- Phase 4: Random restarts with diverse initializations ---
    remaining = deadline - time.monotonic()
    if remaining > 40 and best_seq is not None:
        print(f"\n=== Phase 4: Random Restarts (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_restarts = min(10, int(remaining / 25))

        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            rtype = r % 5
            if rtype == 0:
                a_r = random_init(n_best, rng)
            elif rtype == 1:
                a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                a_r = a_r / sr if sr > 0 else np.ones(n_best) / n_best
            elif rtype == 2:
                # Anti-symmetric perturbation of best
                half = n_best // 2
                pert = rng.normal(0, 0.05, half)
                a_r = best_seq.copy()
                a_r[:half] *= (1.0 + pert)
                a_r[half:] *= (1.0 - pert[:n_best - half])
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                a_r = a_r / sr if sr > 0 else np.ones(n_best) / n_best
            elif rtype == 3:
                n_r = max(300, min(2000, n_best + int(rng.integers(-300, 300))))
                a_r = resample_seq(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.04, len(a_r)))
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                a_r = a_r / sr if sr > 0 else np.ones(n_r) / n_r
            else:
                # Near-uniform with structured deviation
                n_r = n_best
                x = np.arange(n_r)
                a_r = 1.0 + 0.1 * np.sin(2 * np.pi * rng.uniform(1, 5) * x / n_r)
                a_r = np.maximum(a_r, 0)
                a_r = a_r / np.sum(a_r)

            alloc = min(25.0, remaining * 0.2)
            a_opt = lp_alternate(a_r, time_budget=alloc, seed_offset=1000 + r)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < np.inf and ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  Restart {r} (type={rtype}, n={len(a_opt)}): ** {best_score:.8f} **")
            else:
                print(f"  Restart {r} (type={rtype}, n={len(a_opt)}): {ev:.8f}")

    # --- Phase 5: Final intensive polish ---
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_seq is not None:
        print(f"\n=== Phase 5: Final Polish (remaining={remaining:.0f}s) ===")
        a_fin = best_seq / float(np.sum(best_seq))

        # Multiple LP rounds
        for round_idx in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.25, 40.0)
            a_pol = lp_alternate(a_fin, time_budget=alloc, seed_offset=5000 + round_idx)
            ev = evaluate_sequence([float(x) for x in a_pol])
            if ev < np.inf and ev < best_score:
                best_score = ev
                best_seq = a_pol.copy()
                a_fin = a_pol / float(np.sum(a_pol))
                print(f"  Polish {round_idx}: ** {best_score:.8f} **")
            else:
                print(f"  Polish {round_idx}: {ev:.8f}")

        # Gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_gp = smooth_grad_descent(a_fin, time_budget=min(remaining * 0.4, 30.0))
            ev = evaluate_sequence([float(x) for x in a_gp])
            if ev < np.inf and ev < best_score:
                best_score = ev
                best_seq = a_gp.copy()
                print(f"  Gradient polish: ** {best_score:.8f} **")
            else:
                print(f"  Gradient polish: {ev:.8f}")

        # One more short LP
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_last = best_seq / float(np.sum(best_seq))
            a_final = lp_alternate(a_last, time_budget=min(remaining * 0.5, 20.0), seed_offset=9999)
            ev = evaluate_sequence([float(x) for x in a_final])
            if ev < np.inf and ev < best_score:
                best_score = ev
                best_seq = a_final.copy()
                print(f"  Final LP: ** {best_score:.8f} **")
            else:
                print(f"  Final LP: {ev:.8f}")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
