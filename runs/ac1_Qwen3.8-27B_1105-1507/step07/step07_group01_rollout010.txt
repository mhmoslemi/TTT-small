import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        row_end = min(2 * n - 1, j + n)
        col_len = row_end - row_start
        A[row_start:row_end, j] = a[:col_len]
    return A

        a_d = (1 - (1 - phi)) * a + (1 - phi) * g_lp
        a_d = np.maximum(a_d, 0)
        sd = float(np.sum(a_d))
        if sd > 1e-10:
            a_d = a_d / sd
        obj_d = compute_obj(a_d)

        for _ in range(15):
            if hi - lo < 0.01:
                break
            if obj_c < obj_d:
                hi = lo + phi * (hi - lo) if False else (1 - phi)
                # Adjust
                t_new = phi * (hi - lo) + lo
                a_try = (1 - t_new) * a + t_new * g_lp
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < obj_c:
                        lo, a_c, obj_c = t_new, a_try.copy(), obj_try
                    else:
                        hi, a_d, obj_d = t_new, a_try.copy(), obj_try
            else:
                t_new = (1 - phi) * (hi - lo) + lo
                a_try = (1 - t_new) * a + t_new * g_lp
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < obj_d:
                        hi, a_d, obj_d = t_new, a_try.copy(), obj_try
                    else:
                        lo, a_c, obj_c = t_new, a_try.copy(), obj_try

        t_best = (lo + hi) / 2
        a_best_try = (1 - t_best) * a + t_best * g_lp
        a_best_try = np.maximum(a_best_try, 0)
        sb = float(np.sum(a_best_try))
        if sb > 1e-10:
            a_best_try = a_best_try / sb
            obj_best_try = compute_obj(a_best_try)

        if obj_best_try < best_obj - 1e-13:
            best_obj = obj_best_try
            best_a = a_best_try.copy()
            a = a_best_try
            last_t = t_best
            stall_count = 0
            # Update active set
            conv = compute_conv(a)
            max_c = np.max(conv)
            active_set = set(np.where(conv > 0.95 * max_c)[0].tolist())
        else:
            stall_count += 1
            if stall_count > 20:
                # Perturb
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                scale = 0.02
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0

    return best_a


def smooth_max_gradient(a, time_budget=40.0, T_init=0.0003, T_min=0.00002):
    """Nesterov-accelerated smooth max gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.003
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    for it in range(8000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        weights = smooth_max_weights(conv, T)
        grad = compute_smooth_grad(weights, a)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + 0.5 * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0 and lr < 0.015:
                lr = min(lr * 1.03, 0.015)
        else:
            stall += 1
            if stall > 50:
                lr *= 0.6
                if lr < 1e-6:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 30:
            T = max(T_min, T_init * (0.998 ** it))

    return best_a


def paired_coordinate_descent(a, time_budget=60.0):
    """
    Novel paired coordinate descent: pick pairs (i,j) to redistribute mass
    between, reducing the peak convolution value.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    step_size = 0.05
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find i that contributes most to conv[k_star]: a[k_star - i] large
        # and j that contributes least: a[k_star - j] small
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[j]
            else:
                contributions[i] = 0

        # Top contributors
        top_i = int(np.argmax(contributions))
        # Bottom contributors (where we want to add mass)
        contributions[top_i] = -1  # Exclude top
        bot_j = int(np.argmin(contributions))
        if contributions[bot_j] < 0:
            # All are top, pick random
            candidates = list(range(n))
            candidates.remove(top_i)
            bot_j = rng.choice(candidates)

        if top_i == bot_j:
            continue

        # Try moving mass from top_i to bot_j
        a_try = a.copy()
        for frac in [0.1, 0.2, 0.3, 0.5]:
            transfer = frac * a[top_i]
            a_try[top_i] = a[top_i] - transfer
            a_try[bot_j] = a[bot_j] + transfer
            if a_try[top_i] < 0:
                continue
            s_try = float(np.sum(a_try))
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                stall = 0
                break
        else:
            stall += 1
            if stall > 100:
                step_size *= 0.8
                stall = 0
                if step_size < 0.001:
                    break

    return best_a


def gs_polish(a, time_budget=30.0, n_gs_iters=3, blend=0.1):
    """
    Gerchberg-Saxton frequency domain polishing.
    Flattens |FFT(a)|^2 toward uniform while maintaining non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for outer in range(50):
        if time.monotonic() - start > time_budget:
            break

        a_work = a.copy()
        for _ in range(n_gs_iters):
            A = rfft(a_work, size)
            mag = np.abs(A)
            phase = A / (mag + 1e-30)

            # Target: flat magnitude (keep DC)
            mean_mag = np.mean(mag[1:]) if len(mag) > 1 else mag[0]
            target_mag = np.full_like(mag, mean_mag)
            target_mag[0] = mag[0]

            # Blend toward flat
            mag_new = (1 - blend) * mag + blend * target_mag
            # Ensure DC is preserved
            mag_new[0] = mag[0]

            A_new = mag_new * phase
            a_work = irfft(A_new, size)[:n]
            a_work = np.maximum(a_work, 0)
            s_w = float(np.sum(a_work))
            if s_w < 1e-10:
                break
            a_work = a_work / s_w

        obj_work = compute_obj(a_work)
        if obj_work < best_obj - 1e-13:
            best_obj = obj_work
            best_a = a_work.copy()
            a = a_work
            # Decrease blend for finer control
            blend = max(0.02, blend * 0.9)
        else:
            blend = max(0.02, blend * 0.85)
            if blend < 0.02:
                break

    return best_a


def chase_peak_subgrad(a, time_budget=30.0):
    """Subgradient descent targeting the peak bin with momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.03
    v = np.zeros(n)
    momentum = 0.85
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient of conv[k_star] w.r.t. a[i]
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        grad_centered = grad - np.mean(grad)

        v = momentum * v - step * grad_centered
        a_new = simplex_project(a + 0.5 * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 200 == 0 and step < 0.05:
                step = min(step * 1.02, 0.05)
        else:
            stall += 1
            if stall > 60:
                step *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if step < 1e-4:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate random non-negative sequence normalized to sum=1."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif mode == 3:
        center = n // 2
        width = max(1, n // 5)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n) * 2)
    return a

    Main search function. Combines LP alternating, smooth-max gradient,
    paired coordinate descent, GS polishing, chase-peak, and multi-n exploration.

    best_score = np.inf

    try:

        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.8f}")

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating (n={len(best_seq)}) ===")
    a = best_seq.copy()
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:

        alloc = min(remaining * 0.2, 250.0 / (round_idx + 1))
        a_new = lp_alternating(a, time_budget=alloc)

        if obj < best_score:

            best_seq = a_new.copy()

            print(f"  LP Round {round_idx}: obj={obj:.8f} (no improvement)")

    print(f"\n=== Phase 2: Smooth Max Gradient ===")

            a_sm2 = smooth_max_gradient(a, time_budget=alloc, T_init=0.0001, T_min=0.00001)
            obj_sm2 = compute_obj(a_sm2)
            if obj_sm2 < best_score:

                best_seq = a_sm2.copy()

    if remaining > 15:
