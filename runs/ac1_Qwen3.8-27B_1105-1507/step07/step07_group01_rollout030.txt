`d(b[i])/d(a[j]) = d(a[k-i])/d(a[j]) = 1 if k-i = j, i.e., i = k-j, else 0.`

So `d(conv_k)/d(a[j]) = a[k-j] + a[k-j] [if k-j is in [0,n-1]]`...

No wait. `b[j] = a[k-j]` (if `k-j` in range, else 0). And `sum_i a[i] * d(b[i])/d(a[j]) = a[k-j]` (if `k-j` in [0,n-1], else 0).

So `d(conv_k)/d(a[j]) = a[k-j] + a[k-j] = 2*a[k-j]` (if `k-j` in [0,n-1]).

But wait, if `k-j = j` (i.e., `j = k/2`), then both terms are the same, and we should only count it once? No! The formula `sum_i a[i]*b[i]` has the term `i=j` which is `a[j]*b[j] = a[j]*a[k-j]`. If `k-j = j`, this is `a[j]^2`, and its derivative is `2*a[j]`. And the "cross" term is also `a[k-j] = a[j]`. So total is `2*a[j] + a[j]`? No that's wrong.

Let me just do a simple example. `a = [x, y, z]`, `n=3`.
`conv_2 = a[0]*a[2] + a[1]*a[1] + a[2]*a[0] = xz + y^2 + zx = 2xz + y^2`.
`d(conv_2)/d(y) = 2y`. Using my formula: `a[k-j] = a[2-1] = a[1] = y`. And the "cross" term: `k-j = 1 = j`, so it's the same. Total: `2*a[1] = 2y`. ✓

`d(conv_2)/d(x)`: `a[2-0] = a[2] = z`. Cross: `k-j = 2`, and `a[2] = z`. Total: `2z`. From the expression `2xz + y^2`, derivative w.r.t. `x` is `2z`. ✓

OK so the formula is: `d(conv_k)/d(a[j]) = 2*a[k-j]` if `0 <= k-j < n`, else `0`.

Great. So for the subgradient method:
- Find `k* = argmax conv(a,a)`
- Compute subgradient: `s[j] = 2*a[k*-j]` (with zero padding)

Optimized minimizer for 2*n*max(conv(a,a))/(sum(a))^2 using LP alternating,
smooth-max gradient descent, and multi-scale exploration.
Improvements: faster LP via HiGHS, adaptive line search, multi-peak subgradient,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    size_bits = 2 * n - 1
    while size < size_bits:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2 * n - 1]
    return result

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        # For fixed j, k ranges from j to j+n-1
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize: [0, ..., 0, 1]
    # Constraints: A@g - M <= 0, sum(g) = 1, g >= 0, M free (but we can bound it)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: A@g - M <= 0 => [A, -1] @ [g, M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def line_search_blend(a, g, M_a, M_ag, M_g):
    """Find best t in [0,1] for blend (1-t)*a + t*g."""
    n = len(a)
    best_t = 0.0
    best_obj = np.inf
    for t in np.linspace(0, 1, 21):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        conv = compute_conv(a_try)
        obj = 2.0 * n * float(np.max(conv))
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj

def lp_alternating(a, time_budget, max_iters=500):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g = solve_lp(a)
        if g is None:
            break
        
        # Line search
        best_t, best_t_obj = line_search_blend(a, g, best_obj, 0, 0)
        
        if best_t_obj < best_obj - 1e-10:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            a_try = a_try / s_try
            best_a = a_try.copy()
            best_obj = best_t_obj
            a = a_try
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
                if stall > 20:  # This won't trigger but as safety
                    break
    
    return best_a

def smooth_max_gradient(a, time_budget, T_init=0.001, T_min=0.0001):
    """Smooth max gradient descent with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = T_init
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        
        # Gradient: grad[j] = sum_k weights[k] * d(conv_k)/d(a_j) = 2 * sum_k weights[k] * a[k-j]
        # = 2 * correlate(weights, a, 'valid')
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        
        # Project onto simplex tangent: center the gradient
        grad_centered = grad - np.mean(grad)
        
        # Nesterov momentum
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break
        
        T = max(T_min, T_init * (0.999 ** it))
    
    return best_a

def simplex_project(v):
    """Project v onto probability simplex."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def chase_peak_multi(a, time_budget, top_k=5):
    """Chase the top-K peaks with weighted subgradient."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.02
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    
    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = np.max(conv)
        
        # Weight top-K peaks
        sorted_idx = np.argsort(conv)[::-1]
        top_indices = sorted_idx[:top_k]
        top_vals = conv[top_indices]
        
        # Exponential weights based on proximity to max
        weights = np.exp(-(max_c - top_vals) / (0.01 * max_c + 1e-10))
        weights = weights / np.sum(weights)
        
        # Subgradient: sum of weighted gradients
        grad = np.zeros(n)
        for idx, w in zip(top_indices, weights):
            k = int(idx)
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    grad[j] += 2.0 * w * a[kj]
        
        grad_centered = grad - np.mean(grad)
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break
    
    return best_a

def bregman_minimax(a, time_budget):
    """Bregman iteration for min max_k conv(a,a)_k."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)
    
    # Initialize u as uniform
    u = np.ones(2*n-1) / (2*n-1)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        # Build B_u matrix
        Bu = np.zeros((n, n))
        for k in range(2*n-1):
            if u[k] > 1e-15:
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    Bu[i, j] += u[k]
        
        # Solve QP: min a^T Bu a s.t. sum(a) = 1, a >= 0
        try:
            import cvxpy as cp
            a_var = cp.Variable(n, nonneg=True)
            obj = cp.quad_form(a_var, cp.psd_wrap(Bu))
            constraints = [cp.sum(a_var) == 1]
            prob = cp.Problem(cp.Minimize(obj), constraints)
            prob.solve(solver='CLARABEL')
            if a_var.value is not None:
                a_qp = np.maximum(a_var.value, 0)
                a_qp = a_qp / np.sum(a_qp)
            else:
                break
        except Exception:
            break
        
        # Evaluate
        obj_qp = compute_obj(a_qp)
        
        # Update u: temperature-based
        conv = compute_conv(a_qp)
        T = max(0.001, 0.1 * np.max(conv) / (1 + it))
        u = np.exp(conv / T)
        u = u / np.sum(u)
        
        if obj_qp < best_obj - 1e-10:
            best_obj = obj_qp
            best_a = a_qp.copy()
    
    return best_a

Let me also think about a completely different angle: **what if we fix the convolution shape and solve for a?**

If we want `conv(a,a) = c` for some target vector `c` (flat), this gives us a system of quadratic equations:
`sum_i a_i * a_{k-i} = c_k` for k = 0, ..., 2n-2.

This is overdetermined (2n-1 equations, n unknowns). We can solve it as a least-squares problem. But the solution might not be non-negative.

Alternatively, we can use a "phase retrieval" approach: the convolution gives us the squared magnitude of the DFT (by Parseval). So if we want a flat autocorrelation, we want a flat DFT magnitude. This is the "flat-top" problem in signal processing.

For a flat DFT magnitude, we need `|A(omega)| = const` for all omega, where `A(omega) = sum_i a_i * exp(-i*omega*i)`. This is equivalent to a "Zadoff-Chu" or "CAZAC" (Constant Amplitude Zero Autocorrelation) sequence. But those are complex-valued.

For real non-negative sequences, the best we can do is approximate a flat autocorrelation. The LP alternating method is essentially doing this: it iteratively finds the sequence that minimizes the peak autocorrelation.

    """Golden section search for best t in blend (1-t)*a + t*g."""
    def f(t):
        blend = (1-t)**2 * c_a + 2*t*(1-t) * c_ag + t**2 * c_g
        return 2.0 * n * float(np.max(blend))
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~ 0.618
    a_t, b_t = 0.0, 1.0
    x1 = a_t + (1 - gr) * (b_t - a_t)
    x2 = a_t + gr * (b_t - a_t)
    f1, f2 = f(x1), f(x2)
    
    for _ in range(50):  # 50 iterations gives precision ~ 1e-10
        if f1 < f2:
            b_t = x2
            x2 = x1
            f2 = f1
            x1 = a_t + (1 - gr) * (b_t - a_t)
            f1 = f(x1)
        else:
            a_t = x1
            x1 = x2
            f1 = f2
            x2 = a_t + gr * (b_t - a_t)
            f2 = f(x2)

LP-alternating optimizer with golden-section line search, multi-n exploration,
smooth-max gradient polish, and adaptive restarts. Minimizes 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np

# Precomputed constants
GOLDEN = (np.sqrt(5.0) - 1.0) / 2.0

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """FFT-based full linear convolution of a with b."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([0.0])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Column j: A[k, j] = a[k-j] for valid k
        # k ranges from j to j+n-1 (clipped to 2n-2)
        k_end = min(2 * n - 1, j + n)
        A[j:k_end, j] = a[:k_end - j]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    if n < 2:
        return np.ones(1), 0.0
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:

    except Exception as e:
        pass
    return None, None

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
