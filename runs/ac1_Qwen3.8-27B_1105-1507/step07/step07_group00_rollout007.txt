In the continuous limit, if f is a function on [0,1], the autocorrelation is:

"""
Smoothed optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses softmax-smoothed max with FFT-based gradients, projected gradient descent
with Barzilai-Borwein step sizes, active-set LP alternating, and multi-scale
exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_max_and_grad(a, tau):
    """
    Compute smoothed max of conv(a,a) and its gradient w.r.t. a.
    Uses softmax approximation: (1/tau)*log(sum_k exp(tau*conv[k]))
    Gradient is computed via FFT.
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    
    # Softmax weights
    c = conv - np.max(conv)
    exp_c = np.exp(c * tau)
    Z = np.sum(exp_c)
    p = exp_c / Z  # probability weights over convolution bins
    
    smooth_val = np.max(conv) + np.log(Z) / tau
    
    # Gradient: g[i] = sum_{j=0}^{n-1} a[j] * p[i+j]
    # Using FFT: g = conv_fft(a, p_reversed)[n-1 : 2n-1][::-1]
    p_rev = p[::-1]  # length 2n-1
    # Convolution of a (length n) and p_rev (length 2n-1) has length 3n-2
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = rfft(a, size)
    fp = rfft(p_rev, size)
    conv_ap = irfft(fa * fp, size)[:3 * n - 2]
    
    g = conv_ap[n - 1 : 2 * n - 1][::-1].copy()
    
    return smooth_val, g

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def bb_step_size(x1, x2, g1, g2):
    """Barzilai-Borwein step size."""
    s = x2 - x1
    y = g2 - g1
    ss = np.dot(s, s)
    sy = np.dot(s, y)
    if ss < 1e-15:
        return 0.1
    if sy > 1e-15:
        return ss / sy
    elif sy < -1e-15:
        return ss / (-sy)
    else:
        return 0.1

def smoothed_projected_gradient(a, tau_start=10.0, tau_end=200.0, 
                                  time_budget=200.0, verbose=False):
    """
    Smoothed projected gradient descent with BB step size.
    Minimizes smoothed max of conv(a,a) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x_prev = a.copy()
    g_prev = None
    step = 1.0 / tau_start
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Anneal tau
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        smooth_val, g = smoothed_max_and_grad(a, tau)
        
        # BB step size
        if g_prev is not None:
            bb = bb_step_size(x_prev, a, g_prev, g)
            step = 0.5 * step + 0.5 * max(1e-6, min(bb, 2.0 * step))
        
        # Line search (backtracking)
        alpha = step
        found = False
        for _ in range(20):
            a_new = a - alpha * g
            a_new = project_simplex(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj + 1e-12:
                # Accept (allow small increases for exploration)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                x_prev = a.copy()
                a = a_new
                g_prev = g.copy()
                found = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if not found:
            # Stuck, try perturbation
            if it > 100:
                rng = np.random.default_rng(it * 7919 + n)
                scale = 0.05 + 0.05 * (it // 500)
                a = best_a * np.exp(rng.normal(0, min(scale, 0.5), n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                g_prev = None
                step = 1.0 / tau
            else:
                # Just continue with smaller step
                alpha = max(alpha, 1e-8)
                a_new = project_simplex(a - alpha * g)
                x_prev = a.copy()
                a = a_new
                g_prev = g.copy()
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    SPG it{it}: obj={best_obj:.8f}, tau={tau:.1f}, alpha={alpha:.6f}, t={elapsed:.1f}s")
        
        it += 1
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_active_lp(a, M_init, max_constraints=None, tol=1e-8):
    """
    Active-set LP: min M s.t. conv(a,g)[k] <= M for active k, sum(g)=1, g>=0.
    Uses constraint generation to add violated constraints.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Initial active set: top peaks
    conv_a = compute_conv(a)
    M_cur = float(np.max(conv_a))
    
    if max_constraints is None:
        max_constraints = min(n, 2 * n - 1)
    
    # Get initial active set (peaks above M_init * 0.99)
    threshold = M_init * 0.98
    active = np.where(conv_a >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(conv_a)[::-1][:10]
    if len(active) > max_constraints:
        active = np.argsort(conv_a)[::-1][:max_constraints]
    
    M = M_init
    g = None
    
    for it in range(50):
        # Build LP with current active constraints
        A_rows = np.zeros((len(active), n))
        for idx, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                A_rows[idx, i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
        
        # LP: min M s.t. A_rows @ g <= M, sum(g) = 1, g >= 0
        try:
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((len(active), n + 1))
            A_ub[:, :n] = A_rows
            A_ub[:, n] = -1.0
            b_ub = np.zeros(len(active))
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(0, None)]
            
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = np.maximum(res.x[:n], 0)
            M = float(res.x[n])
        except Exception:
            break
        
        if g is None:
            break
        
        g = g / np.sum(g) if np.sum(g) > 1e-15 else np.ones(n) / n
        
        # Check for violated constraints
        conv_ag = compute_conv_ag(a, g)
        violations = np.where(conv_ag > M + tol)[0]
        
        if len(violations) == 0:
            break
        
        # Add worst violators to active set
        worst = np.argsort(conv_ag[violations])[::-1][:5]
        active = np.unique(np.concatenate([active, violations[worst]]))
        if len(active) > max_constraints:
            # Keep top max_constraints by violation amount
            viol_amounts = conv_ag[active] - M
            order = np.argsort(viol_amounts)[::-1]
            active = active[order[:max_constraints]]
            # Also always include the current peak
            k_peak = int(np.argmax(conv_ag))
            if k_peak not in active:
                active = np.append(active, k_peak)
    
    return g, M

def compute_conv_ag(a, g):
    """Compute conv(a, g) via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]

def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with active set and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv_a = compute_conv(a)
        M_cur = float(np.max(conv_a))
        
        g_lp, M_lp = solve_active_lp(a, M_cur, max_constraints=min(50, n))
        if g_lp is None:
            break
        
        # Line search over t
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={best_t_val:.2f}, M_lp={M_lp:.8f}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                scale = 0.05
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
    
    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    t = x / max(n - 1, 1)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
    elif mode == 2:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = a + 0.1
    elif mode == 3:
        # Inverted raised cosine (U-shaped)
        a = 0.5 * (1 - np.cos(2 * np.pi * t))
        a = a + 0.1
    elif mode == 4:
        # Sinusoidal
        a = 0.5 * (1 + 0.5 * np.sin(2 * np.pi * 2 * t))
        a = a + 0.2
    elif mode == 5:
        # Chirp-like
        a = 0.5 + 0.5 * np.sin(2 * np.pi * (t + t**2))
        a = a + 0.2
    elif mode == 6:
        # Parabolic
        a = 1 - (2 * t - 1)**2
        a = a + 0.1
    else:
        a = rng.uniform(0.1, 1, n)
    return np.maximum(a, 0.01) / np.sum(np.maximum(a, 0.01))

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: try moving mass between pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Get contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)
        
        improved = False
        for contrib_val, i, j in contributors[:10]:
            if contrib_val < 1e-12:
                break
            # Try moving mass from i (or j) to a non-contributor
            non_contrib = [k for k in range(n) if k != i and k != j and 
                          not (k_star - k >= 0 and k_star - k < n)]
            
            if not non_contrib:
                continue
            
            # Pick a random target
            target = non_contrib[np.random.randint(len(non_contrib))]
            
            for frac in [0.01, 0.02, 0.05, 0.1]:
                delta = frac * a[i]
                if delta <= 0:
                    continue
                a[i] -= delta
                a[target] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    k_star = int(np.argmax(conv))
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[target] -= delta
                    break
        
        if not improved:
            # Try random perturbations
            rng = np.random.default_rng(it)
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                delta = 0.01 * a[i]
                a[i] -= delta
                a[j] += delta
                new_obj = compute_obj(a)
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = a.copy()
                else:
                    a[i] += delta
                    a[j] -= delta
            else:
                break
        
        it += 1
    
    return best_a
