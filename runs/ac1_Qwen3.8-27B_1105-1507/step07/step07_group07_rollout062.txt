```python
def compute_gradient_smoothed(y, w, n):
    """
    Compute grad[m] = 2 * sum_j w[m+j] * y[j] for m=0..n-1
    where w has length 2n-1 and y has length n.
    """
    # (w ⋆ y)[m] = (y_rev * w)[m + n - 1]
    y_rev = y[::-1]
    size = next_pow2(n + (2*n-1) - 1)  # = next_pow2(3n-2)
    fy_rev = rfft(y_rev, size)
    fw = rfft(w, size)
    conv = irfft(fy_rev * fw, size)
    cross = conv[n-1 : n-1+n]  # positions n-1 to 2n-2
    return 2.0 * cross
```

**Idea: Spectral equalization**

The autocorrelation C(k) can be seen as the "time-domain" representation. In the frequency domain, the DFT of C(k) (padded to length 2n-1) is |FFT(a)|^2 (for the linear convolution case, with appropriate padding).

Wait, that's for circular convolution. For linear convolution, if we zero-pad a to length N >= 2n-1, then:

So the peak of C is related to the inverse FFT of |FFT(a)|^2. To minimize the peak of C, we want to make |FFT(a)|^2 as flat as possible (in the relevant frequency range).

This suggests a **spectral shaping** approach:
1. Compute FFT(a)
2. Modify the magnitude to make |FFT(a)|^2 flatter
3. Take inverse FFT and project to non-negative simplex

But the constraint a >= 0 makes this tricky in the frequency domain (it's equivalent to a constraint on the phase).

Actually, here's a cleaner idea: **Water-filling in frequency domain**.

We want to minimize max_k C(k) where C = IFFT(|FFT(a)|^2). This is equivalent to making the power spectrum |FFT(a)|^2 as flat as possible (by Wiener-Khinchin, flat spectrum = flat autocorrelation).

The "flattest" spectrum subject to a >= 0 and sum(a) = 1 would be... well, it's a hard problem, but we can approximate it.

One approach: Iterate between:
1. Time domain: project a onto {a >= 0, sum = 1}
2. Frequency domain: equalize |FFT(a)| (make it as flat as possible)

This is like a "spectral equalization" algorithm.

For step 2, "equalize |FFT(a)|" means: for each frequency bin, set |A[f]| = c (constant). But we need to preserve the phase (otherwise the inverse FFT won't be real). Also, we need to handle the constraint that the result should be non-negative.

A simpler version: In each frequency bin, scale |A[f]| towards the mean. This is:
A_new[f] = A[f] * (mean(|A|) / |A[f]|)^(1-t)
where t controls the strength of equalization.

This is like a "spetral tilt" correction.

```python
def spectral_equalize(a, n_iter=100, blend=0.5):
    """Iteratively equalize the spectrum while maintaining non-negativity."""
    n = len(a)
    a = a.copy() / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    size = next_pow2(2 * n - 1)
    
    for it in range(n_iter):
        # Compute FFT
        A = np.fft.rfft(a, size)
        
        # Equalize magnitude
        mag = np.abs(A)
        mean_mag = np.mean(mag)
        # Blend towards flat spectrum
        new_mag = mag * (mean_mag / mag + 1e-10) ** blend
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, size)[:n]
        
        # Project to simplex
        a_new = simplex_project(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
    
    return best_a
```

```python
"""
Hybrid optimizer combining FISTA (accelerated proximal gradient), spectral equalization,
LP alternating, and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2
for non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    cond = u * arange > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(y, tau, n):
    """
    Compute gradient of tau*logsumexp(conv(y,y)/tau) w.r.t. y.
    Returns (grad, f_value).
    """
    conv_y = compute_conv(y)
    c_scaled = conv_y / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    f_val = tau * (c_max + np.log(sum_exp))

    # Gradient: grad[m] = 2 * sum_j w[m+j] * y[j]
    # Computed as (y_rev * w)[m + n - 1]
    y_rev = y[::-1]
    size = next_pow2(3 * n - 2)
    fy_rev = rfft(y_rev, size)
    fw = rfft(w, size)
    conv_result = irfft(fy_rev * fw, size)
    cross = conv_result[n - 1:n - 1 + n]
    grad = 2.0 * cross
    return grad, f_val


def fista_optimize(a_init, time_budget=60.0, verbose=False):
    """
    FISTA with log-sum-exp smoothing for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive temperature schedule
    taus = [30.0 / n, 15.0 / n, 7.0 / n, 4.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.25 / n, 0.125 / n]

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break

        x_k = a.copy()
        x_prev = a.copy()
        t_k = 1.0
        alpha = 0.8  # initial step size

        for it in range(5000):
            if time.monotonic() - start > time_budget:
                break

            # FISTA extrapolation
            if it == 0:
                y = x_k.copy()
            else:
                t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
                coef = (t_k - 1.0) / t_next
                y = x_k + coef * (x_k - x_prev)
                y = np.maximum(y, 0.0)
                t_k = t_next

            # Compute gradient at y
            grad, f_y = compute_smoothed_grad(y, tau, n)

            # Backtracking line search
            alpha_bt = alpha
            accepted = False
            for _ in range(30):
                z = y - alpha_bt * grad
                x_new = simplex_project(z)

                # Armijo condition
                f_new = compute_smoothed_obj(x_new, tau)
                directional = np.sum(grad * (x_new - y))
                if f_new <= f_y + 0.1 * directional:
                    accepted = True
                    break
                alpha_bt *= 0.5

            if not accepted:
                x_new = simplex_project(y - alpha_bt * grad)

            alpha = np.clip(alpha_bt * 1.25, 0.01, 2.0)
            x_prev = x_k
            x_k = x_new

            # Check improvement
            obj = compute_obj(x_k)
            if obj < best_obj:
                best_obj = obj
                best_a = x_k.copy()

            # Convergence check
            if it > 10 and np.max(np.abs(x_k - x_prev)) < 1e-12:
                break

        a = best_a.copy()
        if verbose:
            print(f"    FISTA tau={tau:.4f}: obj={best_obj:.8f}")

    return best_a


def compute_smoothed_obj(a, tau):
    """Compute tau * logsumexp(conv(a,a)/tau)."""
    conv = compute_conv(a)
    c = conv / tau
    c_max = np.max(c)
    return tau * (c_max + np.log(np.sum(np.exp(c - c_max))))


def spectral_equalize(a_init, time_budget=60.0, verbose=False):
    """
    Iterative spectral equalization: alternate between time-domain projection
    and frequency-domain magnitude flattening.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    size = next_pow2(2 * n - 1)

    blend_schedule = [0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05]

    for blend in blend_schedule:
        if time.monotonic() - start > time_budget:
            break

        for it in range(200):
            if time.monotonic() - start > time_budget:
                break

            # Frequency domain: equalize magnitude
            A = np.fft.rfft(a, size)
            mag = np.abs(A)
            mean_mag = np.mean(mag)
            if mean_mag < 1e-15:
                break

            # Soft equalization: blend towards flat
            ratio = mean_mag / (mag + 1e-15)
            new_mag = mag * (ratio ** blend)
            phase = np.angle(A)
            A_new = new_mag * np.exp(1j * phase)

            # Back to time domain
            a_new = np.fft.irfft(A_new, size)[:n]

            # Project to simplex
            a_new = simplex_project(a_new)

            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new

        if verbose:
            print(f"    Spectral blend={blend:.2f}: obj={best_obj:.8f}")

    return best_a


def peak_shave_momentum(a_init, time_budget=30.0):
    """
    Subgradient + momentum method targeting the peak directly.
    """
    n = len(a_init)

            g[m] = 2.0 * a[k_star - m]

        # Momentum update
        velocity = mu * velocity - eta * g

        obj = compute_obj(a_new)

            best_obj = obj

        else:
            a = best_a.copy()
            velocity *= 0.3
            eta *= 0.6
            if eta < 1e-7:

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=15.0):
    """

    try:

        A_ub[:, :n] = A
        A_ub[:, n] = -1.0

        A_eq[0, n] = 0.0

            s = float(np.sum(g_val))

    return None, None

def lp_alternate_optimize(a_init, time_budget=60.0, verbose=False):

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=None):
    """Generate diverse random initializations."""
    if mode is None:
        mode = rng.integers(0, 8)

    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Gaussian bump
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n / 2) / (n / 6)) ** 2)
        a += 0.01 * rng.random(n)
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2 + 0.05
    elif mode == 3:
        # Triangular
        a = 1.0 - np.abs(np.arange(n) - n / 2) / (n / 2 + 1)
        a += 0.02 * rng.random(n)
    elif mode == 4:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n * n / 40)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n * n / 40))
        a += 0.01 * rng.random(n)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1) / n
        a = x ** 0.5 + 0.1
    elif mode == 6:
        # Random smooth
        a = rng.random(n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
    else:
        # Edge-decayed
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = np.exp(-0.3 * np.abs(x - center) / n) + 0.05

    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    return a


def multi_peak_targeted(a_init, time_budget=30.0):
    """
    Target the top-3 peaks simultaneously using averaged subgradients.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    velocity = np.zeros(n)
    mu = 0.8
    eta = 0.03

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)

        # Get top-3 peak positions
        top_k = np.argsort(conv)[::-1][:3]
        weights = np.exp(conv[top_k] / (0.5 / n))
        weights = weights / np.sum(weights)

        # Weighted subgradient
        g = np.zeros(n)
        for wk, k in zip(weights, top_k):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                g[m] += wk * 2.0 * a[k - m]

        # Momentum
        velocity = mu * velocity - eta * g
        a_new = simplex_project(a + velocity)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.003, 0.1)
        else:
            a = best_a.copy()
            velocity *= 0.4
            eta *= 0.5
            if eta < 1e-7:
                break

    return best_a

            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.03, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 2:
            # Symmetric perturbation
            a_r = best_seq + 0.03 * rng.normal(0, 1, n_best)
            a_r = (a_r + a_r[::-1]) / 2.0
            a_r = np.maximum(a_r, 0.001)
            a_r = a_r / np.sum(a_r)
        elif r == 3:
            # Fresh diverse init
            a_r = diverse_init(n_best, rng, mode=r % 8)
        else:
            # Blended with diverse init
            a_fresh = diverse_init(n_best, rng)
            a_r = 0.7 * best_seq + 0.3 * a_fresh
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.15, 30.0)
        a_opt = fista_optimize(a_r, time_budget=alloc * 0.6)
        obj = compute_obj(a_opt)

        if obj < best_score + 0.005:
            a_opt2 = spectral_equalize(a_opt, time_budget=alloc * 0.3)
            obj2 = compute_obj(a_opt2)
            if obj2 < obj:
                obj = obj2
                a_opt = a_opt2

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (n={n_best}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")

    # === Phase 10: Final polish ===
    print(f"\n=== Phase 10: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a = fista_optimize(best_seq, time_budget=min(remaining * 0.3, 30.0), verbose=True)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Final FISTA: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.3, 25.0))
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Final LP: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
```
