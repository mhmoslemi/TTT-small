"""
Enhanced LP-Alternating optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Focuses on high-precision LP alternating with spectral initialization, adaptive
blending, and gradient polishing. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time
import math


def next_pow2(m):
    """Smallest power of 2 >= m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """
    Build matrix A (2n-1 x n) such that A @ g = conv(a, g).
    A[k, i] = a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min <= i_max:
            A[k, i_min:i_max + 1] = a[k - i_max:k - i_min + 1]
    return A


def solve_lp_given_a(a):
    """
    Solve: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) where g is normalized to sum 1.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    # LP variables: [g_0, ..., g_{n-1}, M]
    # Minimize: M (last variable)
    # Constraints: A @ g - M <= 0 (2n-1 inequalities)
    #              sum(g) = 1 (equality)
    #              g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # conv(a,g)[k] - M <= 0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-12:
                g_val = g_val / s
                return g_val, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternate_step(a, alpha=None):
    """
    One step of LP alternating.
    Given a (sum=1), solve LP to get optimal g, then blend.
    Returns (new_a, new_obj, M_lp).
    """
    n = len(a)
    a = a / float(np.sum(a))
    
    g_lp, M_lp = solve_lp_given_a(a)
    if g_lp is None:
        return a, compute_obj(a), np.inf
    
    if alpha is None:
        # Line search for optimal alpha
        best_a = a.copy()
        best_obj = compute_obj(a)
        alphas = [0.3, 0.5, 0.7, 1.0, 0.2, 0.4, 0.6, 0.8, 0.9, 0.1]
        for t in alphas:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-12:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
        return best_a, best_obj, M_lp
    else:
        a_new = (1.0 - alpha) * a + alpha * g_lp
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-12:
            return a, compute_obj(a), M_lp
        a_new = a_new / s
        return a_new, compute_obj(a_new), M_lp


def lp_alternate_loop(a_init, time_budget, verbose=False):
    """
    Run LP alternating until time runs out or convergence.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        a_new, obj_new, M_lp = lp_alternate_step(a)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and (stall_count == 0 or best_obj < 1.6):
                elapsed = time.monotonic() - start
                print(f"    LP iter: obj={best_obj:.10f}, M_lp={M_lp:.8f}, t={elapsed:.1f}s")
        else:
            stall_count += 1
            if stall_count > 5:
                # Try with fixed alpha = 0.5
                a_pop = best_a.copy()
                for _ in range(3):
                    a_new, obj_new, _ = lp_alternate_step(a_pop, alpha=0.5)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        stall_count = 0
                        break
                if stall_count > 8:
                    break
    
    return best_a


def smoothed_objective_and_grad(x, tau):
    """
    Smoothed objective using log-sum-exp on convolution.
    x parameterizes a = exp(x) / sum(exp(x)).
    Returns (objective_value, gradient_wrt_x).
    """
    n = len(x)
    # Softmax parameterization for non-negativity and normalization
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    
    conv = compute_conv(a)
    
    # Log-sum-exp smoothing of max(conv)
    c_max = np.max(conv)
    exp_c = np.exp((conv - c_max) / tau)
    sum_exp = np.sum(exp_c)
    ls_max = c_max + tau * np.log(sum_exp)
    
    # Objective: 2*n*ls_max (since sum(a)=1)
    F = 2.0 * n * ls_max
    
    # Gradient: dF/dx_i
    # d a_j / d x_i = a_j * (delta_{ij} - a_i)
    # d conv[k] / d a_j = a[k-j] + a[j-k] (with bounds, considering full conv)
    # Actually for conv(a,a)[k] = sum_i a[i]*a[k-i]:
    #   d conv[k] / d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    #                     = a[k-j] + a[j-k] (symmetric)
    
    # Weight of each conv[k] in the soft-max
    w = exp_c / sum_exp
    
    # d F / d a_j = 2*n * sum_k w[k] * d conv[k] / d a_j
    # d conv[k] / d a_j = a[k-j] (if valid) + a[j-k] (if valid, and j!=k-j)
    # Actually: conv[k] = sum_i a[i] * a[k-i]
    # d conv[k] / d a[j] = a[k-j] (from i=j term) + a[j-k] (from k-i=j => i=k-j term)
    # But these are the same term if k-j = j... no, they're different unless k=2j.
    # More carefully:
    # conv[k] = sum_i a[i] * a[k-i]
    # Partial w.r.t. a[j]: appears when i=j (gives a[k-j]) and when k-i=j i.e. i=k-j (gives a[j])
    # So d conv[k]/d a[j] = a[k-j] (if 0<=k-j<n) + a[j] (if 0<=k-j<n, from the other side)
    # Wait no. Let me be very careful.
    # conv[k] = sum_i a[i] * a[k-i], where sum is over i such that 0<=i<n and 0<=k-i<n.
    # d/d a[j] of conv[k] = sum_i [delta_{ij} * a[k-i] + a[i] * delta_{k-i,j}]
    #                      = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n)  [when i=j and k-i=j]
    # Hmm, the second term: k-i=j means i=k-j. So the second contribution is a[k-j] when i=k-j is valid.
    # So d conv[k]/d a[j] = a[k-j] + a[k-j] = 2*a[k-j] (if 0<=k-j<n), unless k-j=j (i.e., k=2j)
    # in which case it's just a[j] (one term).
    
    # Simpler: use FFT-based cross-correlation
    # dF/dx_i = sum_j (dF/da_j) * (da_j/dx_i) = sum_j (dF/da_j) * a_j * (delta_{ij} - a_i)
    #         = a_i * (dF/da_i - sum_j dF/da_j * a_j)
    
    # dF/da_j = 2*n * sum_k w[k] * d conv[k]/d a[j]
    # d conv[k]/d a[j] = 2*a[k-j] (0<=k-j<n) - a[j]*(1 if k==2j else 0)
    # For simplicity, use: d conv[k]/d a[j] ≈ 2*a[k-j] (0<=k-j<n)
    # (The correction term is small and we can ignore it for the gradient)
    
    # Compute h[j] = sum_k w[k] * 2*a[k-j] (where k-j is in [0,n-1])
    # This is a cross-correlation: h = 2 * corr(w, a) shifted appropriately
    # h[j] = 2 * sum_k w[k] * a[k-j] for k-j in [0,n-1]
    # Let m = k-j, then k = j+m, so h[j] = 2 * sum_m w[j+m] * a[m] for j+m in [0,2n-2]
    
    # Use FFT: h = 2 * ifft(conj(fft(w, 2n-1)) * fft(a, 2n-1)) ... 
    # Actually, let's just do it directly for clarity (n is at most ~2000)
    
    size = next_pow2(2 * n - 1)
    fw = rfft(w, size)
    fa = rfft(a, size)
    # Cross-correlation: sum_k w[k] * a[k-j] = ifft(fft(w) * conj(fft(a)))[j]? 
    # No. Let's use: (w * a)[k] = sum_j w[j] * a[k-j] (cross-correlation)
    # We want h[j] = sum_k w[k] * a[k-j] = sum_m w[m+j] * a[m] (reindex m=k-j)
    # This is the cross-correlation of w and a at lag j.
    # Correlation(w, a)[j] = sum_m w[m] * a[m+j] ... hmm, convention matters.
    
    # Let's just compute it via the full convolution approach:
    # h[j] = sum_k w[k] * a[k-j] where 0<=k-j<n and 0<=k<2n-1
    # = sum_m w[m+j] * a[m] where m = k-j, 0<=m<n, 0<=m+j<2n-1
    # This is the (j)-th element of the "correlation" of w with a.
    
    # Using FFT: corr(j) = ifft(conj(FFT(w)) * FFT(a))[j] ... not quite.
    # Let's use: ifft(FFT(w, size) * conj(FFT(a, size))) gives circular cross-correlation.
    # For zero-padded vectors of length >= 2n-1, this equals the linear cross-correlation.
    
    cross = irfft(fw * np.conj(fa), size)[:2 * n - 1]
    # cross[j] = sum_k w[k] * a[(k-j) mod size]
    # For our zero-padded case with size >= 2n-1, this gives:
    # cross[j] = sum_k w[k] * a[k-j] (where a is zero outside [0,n-1])
    # = sum_{k=j}^{j+n-1} w[k] * a[k-j] (for j in [0, 2n-2])
    
    # We want h[j] = 2 * cross[j] for j in [0, n-1]
    # But cross is defined for j in [0, 2n-2]. We need the "positive lag" part.
    # Actually, cross[j] for j>=0 gives sum_k w[k]*a[k-j] which is what we want.
    
    h = 2.0 * cross[:n]  # h[j] = 2 * sum_k w[k]*a[k-j] for j=0..n-1
    
    # dF/da_j = h[j] (approximately, ignoring the self-term correction)
    # dF/dx_i = a_i * (h[i] - sum_j h[j]*a[j])
    
    grad_da = h
    sum_grad_a = np.dot(grad_da, a)
    grad = a * (grad_da - sum_grad_a)
    
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Optimize using L-BFGS-B on smoothed log-sum-exp objective with softmax parameterization.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Temperature schedule
    taus = [20.0 / n, 10.0 / n, 5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n]
    
    x = np.log(a) - np.log(a).max()
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_objective_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_objective_and_grad(xv, tau)
            return g
        
        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12,
                                    'maxls': 60})
            if res.fun < fun(x):
                x_opt = res.x - res.x.max()
                a_new = np.exp(x_opt)
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = x_opt
                    if verbose:
                        print(f"    grad tau={tau:.4f}: obj={best_obj:.10f}")
        except Exception:
            pass
    
    return best_a


def spectral_init(n, rng):
    """
    Initialize via spectral method: create a flat-ish power spectrum,
    take IFFT, clip to non-negative.
    """
    size = next_pow2(2 * n - 1)
    # Create a power spectrum that's flat in the middle and rolls off at edges
    freqs = np.arange(size // 2 + 1)
    # Raised cosine rolloff
    bandwidth = 0.3 + 0.4 * rng.random()  # random bandwidth between 0.3 and 0.7
    psd = 1.0
    for f in freqs:
        if f > 0:
            norm_f = f / (size // 2)
            if norm_f > (1 - bandwidth):
                t = (norm_f - (1 - bandwidth)) / bandwidth
                psd_val = 0.5 * (1 + np.cos(np.pi * t))
            else:
                psd_val = 1.0
            # Add some random variation
            psd_val *= (0.8 + 0.4 * rng.random())
    
    # But we need a REAL, non-negative time-domain signal
    # The IFFT of a real, non-negative PSD gives a real, even signal
    # We want it non-negative... this is the hard part.
    # Strategy: add a large DC component to ensure non-negativity
    
    # Actually, let's use a different approach:
    # Start with a random phase and the desired magnitude
    phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    phase[0] = 0  # DC is real
    mag = np.ones(size // 2 + 1)
    # Apply low-pass
    for i in range(1, len(mag)):
        t = i / (size // 2)
        if t > 0.5:
            mag[i] = 0.5 * (1 + np.cos(np.pi * (t - 0.5) / 0.5))
        # Add some randomness
        mag[i] *= (0.7 + 0.6 * rng.random())
    
    spectrum = mag * np.exp(1j * phase)
    # Make it Hermitian
    full_spectrum = np.zeros(size, dtype=complex)
    full_spectrum[:size // 2 + 1] = spectrum
    for i in range(1, size // 2 + 1):
        full_spectrum[size - i] = np.conj(spectrum[i])
    
    sig = irfft(full_spectrum)[:n]
    sig = np.maximum(sig, 0)
    s = np.sum(sig)
    if s < 1e-10:
        return np.ones(n) / n
    return sig / s


def cosine_window_init(n, rng):
    """Initialize with a cosine (Hann) window plus small noise."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x) ** 2
    a += 0.05 * rng.random(n)
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    return a


def tukey_window_init(n, rng, r=0.5):
    """Initialize with a Tukey window."""
    n_half = n // 2
    w = np.ones(n)
    for i in range(int(r * n_half) + 1):
        w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (r * n) - 1)))
        w[n - 1 - i] = w[i]
    w += 0.01 * rng.random(n)
    w = np.maximum(w, 0)
    w = w / np.sum(w)
    return w


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def exp_gradient_step(a, k_star, eta):
    """
    One exponentiated gradient step targeting the peak at k_star.
    Reduces a[i] where a[i]*a[k_star-i] contributes most to the peak.
    """
    n = len(a)
    # Gradient of conv[k_star] w.r.t. a[i] = a[k_star - i] (if valid)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
        j2 = i - k_star  # For the "other side" of the convolution
        if 0 <= j2 < n and j2 != j:
            grad[i] += a[j2]
    
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / np.sum(a_new)
    return a_new


def exp_gradient_optimize(a_init, time_budget=30.0):
    """Exponentiated gradient method targeting the maximum lag."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.5 / n
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        eta_new = eta
        improved = False
        for _ in range(10):
            a_new = exp_gradient_step(a, k_star, eta_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                eta_new *= 0.95
            else:
                eta_new *= 0.5
        
        if not improved:
            # Perturb and restart
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            eta = 0.3 / n
    
    return best_a


def coordinate_descent_optimize(a_init, time_budget=30.0, verbose=False):
    """
    Coordinate descent: at each step, adjust one coordinate to reduce the max.
    The max is at some lag k*. To reduce conv[k*], we can decrease a[i] for i
    that contributes most, while compensating by increasing other a[j].
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find the coordinate whose reduction most helps
        # conv[k_star] = sum_i a[i]*a[k_star-i]
        # Reducing a[i] by delta reduces conv[k_star] by delta * a[k_star-i]
        # But we need to maintain sum(a) = 1, so we redistribute.
        
        # Try: move mass from position i to position j
        # delta_conv[k_star] = -a[k_star-i]*a[i]/s + a[k_star-j]*a[j]/s - ... (complex)
        # Simpler: just try random pairs
        
        for _ in range(50):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Move small amount of mass from i to j (or vice versa)
            # Try both directions
            for direction in [1, -1]:
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    amount = direction * frac * min(a[i], a[j])
                    a_try = a.copy()
                    a_try[i] -= amount
                    a_try[j] += amount
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = np.sum(a_try)
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        break
        
        if verbose and (time.monotonic() - start) % 5 < 0.1:
            print(f"      CD: obj={best_obj:.10f}")
    
    return best_a
