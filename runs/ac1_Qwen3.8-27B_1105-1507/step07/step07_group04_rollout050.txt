import numpy as np

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def sparse_lp_minpeak(a, top_k=20, solver=None):
    """Solve LP with only top-K constraints for speed."""
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    # Get top-K lag indices
    k_indices = np.argsort(conv)[::-1][:top_k]
    
    # Build sparse constraint matrix for only these k values
    # conv(a, g)[k] = sum_j a[j] * g[k-j] (with bounds)
    A_rows = []
    for k in k_indices:
        k = int(k)
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            row[j] = a[k - j]
        A_rows.append(row)
    A = np.array(A_rows)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((len(k_indices), n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(len(k_indices))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    
    return None

def full_lp_minpeak(a, solver=None):
    """Solve LP with all constraints."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build full constraint matrix

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS]

def guided_mutation(a, rng, mode):

    elif mode == 2:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent: move mass between pairs to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

            break

        k_star = int(np.argmax(conv))
        improved = False

        # Find donors and recipients for the peak lag
        # For peak at k_star: conv[k_star] = sum_j a[j]*a[k_star-j]

                a[j] += delta

                else:
                    a[i] += delta
                    a[j] -= delta
                    break

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    a = best_seq.copy()
    n_best = len(a)

    # === Phase 1: Fast LP Alternating (350s) ===
    print(f"\n=== Phase 1: Fast LP Alternating (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    budget_1 = min(350.0, remaining * 0.3)

    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(budget_1 / (4 - round_idx), remaining * 0.45)
        top_k = 15 + round_idx * 5
        a_new = lp_alternating_fast(a, time_budget=alloc, top_k_start=top_k, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            n_best = len(a)
            print(f"  ** Phase 1 Round {round_idx}: obj={best_score:.8f}")
        else:
            print(f"  Phase 1 Round {round_idx}: obj={obj:.8f}")

    # === Phase 2: Multi-Start (300s) ===
    print(f"\n=== Phase 2: Multi-Start Exploration ===")
    remaining = deadline - time.monotonic()
    budget_2 = min(300.0, remaining * 0.25)

    # Generate diverse starting points
    starts = []
    starts.append(("parent", best_seq.copy()))
    starts.append(("mirror", best_seq[::-1].copy()))

    # Shift
    shift = n_best // 7
    starts.append(("shift", np.roll(best_seq, shift).copy()))

    # Frequency domain perturbation
    size = 1
    while size < 2 * n_best:
        size <<= 1
    fa = rfft(best_seq, size)
    mag = np.abs(fa) * np.exp(rng.normal(0, 0.05, size // 2 + 1))
    phase = fa / (np.abs(fa) + 1e-10)
    a_freq = np.maximum(irfft(mag * phase, size)[:n_best], 0)
    s = float(np.sum(a_freq))
    if s > 1e-10:
        starts.append(("freq", a_freq / s))

    # Gaussian perturbation
    a_gauss = np.maximum(best_seq * np.exp(rng.normal(0, 0.1, n_best)), 0)
    s = float(np.sum(a_gauss))
    if s > 1e-10:
        starts.append(("gauss", a_gauss / s))

    # Random uniform
    a_rand = np.ones(n_best) / n_best
    starts.append(("uniform", a_rand))

    # Random power law
    a_power = np.power(np.linspace(0.1, 1.0, n_best), 2.0)
    a_power = a_power / np.sum(a_power)
    starts.append(("power", a_power))

    per_start_budget = budget_2 / len(starts)

    for name, a_start in starts:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_start = a_start.copy()
        s = float(np.sum(a_start))
        if s > 1e-10:
            a_start = a_start / s

        alloc = min(per_start_budget, remaining * 0.4)
        a_opt = lp_alternating_fast(a_start, time_budget=alloc, top_k_start=15)
        obj = compute_obj(a_opt)

        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = len(a)
            print(f"  ** Start '{name}': obj={best_score:.8f}")
        else:
            print(f"  Start '{name}': obj={obj:.8f}")

    # === Phase 3: n-sweep (200s) ===
    print(f"\n=== Phase 3: n-Sweep ===")
    remaining = deadline - time.monotonic()
    budget_3 = min(200.0, remaining * 0.2)

    n_options = [400, 500, 600, 700, 800, 900, 1000, 1200, 1500]
    # Sort by proximity to current best
    n_options = [n for n in n_options if abs(n - n_best) > 50]
    n_options.sort(key=lambda x: abs(x - n_best))

    per_n_budget = budget_3 / min(6, len(n_options))

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(per_n_budget, remaining * 0.4)
        a_opt = lp_alternating_fast(a_try, time_budget=alloc, top_k_start=15)
        obj = compute_obj(a_opt)

        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = len(a)
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 4: Subgradient + Coordinate (150s) ===
    print(f"\n=== Phase 4: Subgradient + Coordinate ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_sg = min(60.0, remaining * 0.3)
        a_sg = subgradient_fast(best_seq, time_budget=alloc_sg, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score - 1e-12:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            n_best = len(a)
            print(f"  ** Subgradient: obj={best_score:.8f}")
        else:
            print(f"  Subgradient: obj={obj_sg:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_cd = min(50.0, remaining * 0.3)
        a_cd = coordinate_polish(best_seq, time_budget=alloc_cd)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-12:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            n_best = len(a)
            print(f"  ** Coordinate: obj={best_score:.8f}")
        else:
            print(f"  Coordinate: obj={obj_cd:.8f}")

    # === Phase 5: Final intensive LP (remaining) ===
    print(f"\n=== Phase 5: Final LP Polish ===")
    remaining = deadline - time.monotonic()

    # Multiple rounds with increasing top_k
    for tk in [20, 30, 50, 80]:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(50.0, remaining * 0.4)
        a_final = lp_alternating_fast(best_seq, time_budget=alloc, top_k_start=tk)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-12:
            best_score = obj_final
            best_seq = a_final.copy()
            a = a_final.copy()
            n_best = len(a)
            print(f"  ** Final LP (top_k={tk}): obj={best_score:.8f}")
        else:
            print(f"  Final LP (top_k={tk}): obj={obj_final:.8f}")

    # === Phase 6: Last chance mutations ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Last Chance ===")
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            mode = int(rng.integers(0, 5))
            a_mut = best_seq.copy()
            if mode == 0:
                a_mut = a_mut[::-1].copy()
            elif mode == 1:
                a_mut = a_mut * np.exp(rng.normal(0, 0.02 + 0.01 * r, n_best))
            elif mode == 2:
                # Local window modification
                center = int(rng.integers(0, n_best))
                width = max(2, n_best // 30)
                a_mut[center - width:center + width + 1] *= float(rng.uniform(0.6, 1.4))
            elif mode == 3:
                # Blend with uniform
                a_mut = 0.8 * a_mut + 0.2 * np.ones(n_best) / n_best
            else:
                # Frequency perturbation
                size = 1
                while size < 2 * n_best:
                    size <<= 1
                fa = rfft(a_mut, size)
                mag = np.abs(fa) * np.exp(rng.normal(0, 0.02, size // 2 + 1))
                phase = fa / (np.abs(fa) + 1e-10)
                a_mut = np.maximum(irfft(mag * phase, size)[:n_best], 0)

            a_mut = np.maximum(a_mut, 0)
            s_m = float(np.sum(a_mut))
            if s_m < 1e-10:
                continue
            a_mut = a_mut / s_m

            alloc = min(20.0, remaining * 0.35)
            a_opt = lp_alternating_fast(a_mut, time_budget=alloc, top_k_start=20)
            obj_opt = compute_obj(a_opt)

            if obj_opt < best_score - 1e-12:
                best_score = obj_opt
                best_seq = a_opt.copy()
                a = a_opt.copy()
                n_best = len(a)
                print(f"  ** Last chance {r} (mode={mode}): obj={best_score:.8f}")
            else:
                print(f"  Last chance {r} (mode={mode}): obj={obj_opt:.8f}")

    # Final coordinate polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd = coordinate_polish(best_seq, time_budget=min(15.0, remaining * 0.5))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-12:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** Final CD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0)
    result = [float(x) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
