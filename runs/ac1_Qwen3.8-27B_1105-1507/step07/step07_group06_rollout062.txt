def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Scale to project onto sum=1
    u = v / target_sum
    # Algorithm from Duchi et al. 2011
    rho = np.sort(u)[::-1]
    cssv = np.cumsum(rho)
    cond = (rho - (cssv - 1.0)) / np.arange(1, n + 1)
    ind = np.where(cond > 0)[0]
    if len(ind) == 0:
        return np.full(n, target_sum / n)
    t = (rho[ind[-1]] + (cssv[ind[-1]] - 1.0)) / (ind[-1] + 1)
    proj = np.maximum(u - t, 0)
    # Fix any numerical issues
    s = np.sum(proj)
    if s < 1e-15:
        return np.full(n, target_sum / n)
    return proj * target_sum


def pg_step(a, T, eta, N, fa):
    """
    One projected gradient step.
    Returns: (a_new, b, obj_val, fa_new)
    """
    n = len(a)
    # b = a * a
    b = irfft(fa * fa, N)[:2 * n - 1]
    
    # Softmax weights
    b_max = np.max(b)
    w = np.exp((b - b_max) / T)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= w_sum
    
    # Gradient: g_j = 2 * sum_m a_m * w_{m+j}
    fw = rfft(w, N)
    g = 2.0 * irfft(fw * np.conjugate(fa), N)[:n]
    
    # Projected update
    a_new = a - eta * g
    a_new = project_simplex(a_new)
    
    fa_new = rfft(a_new, N)
    return a_new, b, fa_new

    eta = eta_init
    start = time.monotonic()

    iter_count = 0
    stuck_count = 0
    T_current = T_start
    
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = elapsed / time_budget
        T_current = T_start * (T_end / T_start) ** frac
        
        # Compute step
        a_new, b, fa_new = pg_step(a, T_current, eta, N, fa)
        
        # Compute objective
        b_max = float(np.max(b))
        obj_new = 2.0 * n * b_max
        
        if obj_new < best_obj:
            # Accept
            best_obj = obj_new
            best_a = a_new.copy()
            stuck_count = 0
            # Try to increase eta
            eta = min(eta * 1.05, 5.0)
        else:
            stuck_count += 1
            # Reject and decrease eta
            eta *= 0.8
        
        # Update state
        # Use momentum
        g = 2.0 * irfft(rfft(np.exp((b - np.max(b)) / T_current) / np.sum(np.exp((b - np.max(b)) / T_current)), N) * np.conjugate(fa), N)[:n]

        # The pg_step already did the projection, so just update a and fa
        a = a_new
        fa = fa_new
        
        iter_count += 1
        
        # Periodic restart if stuck
        if stuck_count > 500 and stuck_count % 500 == 0:
            # Perturb from best
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            fa = rfft(a, N)
            eta = eta_init
            stuck_count = 0
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Try cvxpy solvers
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback to linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, time_budget=30.0):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            if iteration % 5 == 0:
                rng = np.random.default_rng(iteration * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
        
        iteration += 1
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

FFT-accelerated projected gradient descent (log-sum-exp smoothed) with momentum,

Key: gradient of conv-peak is a shifted copy of a, enabling O(n log n) per iteration.

def compute_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = v / target_sum
    rho = np.sort(u)[::-1]
    cssv = np.cumsum(rho)
    cond = (rho - (cssv - 1.0)) / np.arange(1, n + 1)
    ind = np.where(cond > 0)[0]
    if len(ind) == 0:
        return np.full(n, target_sum / n)
    t = (rho[ind[-1]] + (cssv[ind[-1]] - 1.0)) / (ind[-1] + 1)
    proj = np.maximum(u - t, 0)
    s = np.sum(proj)
    if s < 1e-15:
        return np.full(n, target_sum / n)
    return proj * target_sum


def normalize_nonneg(a):
    """Ensure non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def pg_descent(a_init, time_budget, T_start=5.0, T_end=300.0,
               beta=0.9, eta_init=0.3, seed=42):
    """
    Projected gradient descent with momentum on smoothed max-conv objective.
    Uses FFT for both convolution and gradient computation.
    """
    n = len(a_init)
    a = normalize_nonneg(a_init)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]
    best_obj = 2.0 * n * float(np.max(b))
    best_a = a.copy()

    v = np.zeros(n)
    eta = eta_init
    start = time.monotonic()
    stuck = 0
    rng = np.random.default_rng(seed)
    iter_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(elapsed / time_budget, 1.0)
        T = T_start * (T_end / T_start) ** frac
        eta = max(eta_init * (1.0 - 0.7 * frac), 0.01)

        # Compute b = a*a
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Softmax weights (temperature T)
        b_max = float(np.max(b))
        shifted = (b - b_max) / T
        # Clip for numerical stability
        shifted = np.maximum(shifted, -50.0)
        w = np.exp(shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= w_sum

        # Gradient: g_j = 2 * sum_m a_m * w_{m+j}
        fw = rfft(w, N)
        g = 2.0 * irfft(fw * np.conjugate(fa), N)[:n]

        # Momentum
        v = beta * v + (1.0 - beta) * g

        # Step + project
        a_new = a - eta * v
        a_new = project_simplex(a_new)

        # Evaluate (use smoothed obj for speed, exact for tracking)
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_exact = 2.0 * n * float(np.max(b_new))

        if obj_exact < best_obj:
            best_obj = obj_exact
            best_a = a_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 300:
                # Perturb from best to escape
                noise_scale = 0.05 + 0.1 * (stuck % 500) / 500.0
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = project_simplex(a)
                fa = rfft(a, N)
                v = np.zeros(n)
                stuck = 0
                eta = eta_init
                continue

        a = a_new
        fa = fa_new
        iter_count += 1

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization with line search."""
    n = len(a)
    a = normalize_nonneg(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iteration = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        if not improved:
            if iteration % 5 == 0:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = project_simplex(a)
        iteration += 1
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FFT-accelerated projected gradient (millions of iterations)
    + LP polish + multi-n exploration + diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Main Projected Gradient ===
    print(f"\n=== Phase 1: PG main (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.45, 550.0)
        a_opt, obj_opt = pg_descent(best_seq, alloc,
                                     T_start=3.0, T_end=200.0,
                                     beta=0.9, eta_init=0.3, seed=seed)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  PG main: obj={best_score:.8f} **")
        else:
            print(f"  PG main: obj={obj_opt:.8f}")

    # === Phase 2: Second PG (higher T, different momentum) ===
    print(f"\n=== Phase 2: PG explore ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 200.0)
        a_opt, obj_opt = pg_descent(best_seq, alloc,
                                     T_start=30.0, T_end=500.0,
                                     beta=0.95, eta_init=0.15, seed=seed + 7)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  PG explore: obj={best_score:.8f} **")
        else:
            print(f"  PG explore: obj={obj_opt:.8f}")

    # === Phase 3: LP Polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    if remaining > 20 and n_cur <= 1000:
        alloc = min(remaining * 0.12, 90.0)
        a_lp, obj_lp = lp_alternating(best_seq, time_budget=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = []
    for dn in [-300, -200, -100, -50, 50, 100, 200, 400, 600]:
        nt = n_cur + dn
        if 300 <= nt <= 3000:
            n_options.append(nt)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        if n_try == n_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 50.0)
        a_opt, obj_opt = pg_descent(a_try, alloc,
                                     T_start=5.0, T_end=200.0,
                                     beta=0.9, eta_init=0.3, seed=seed + n_try)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break
        init_type = r % 5
        if init_type == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.12, n_best))
        elif init_type == 1:
            ramp = np.linspace(0.2, 1.8, n_best)
            blend = rng.uniform(0.2, 0.8)
            a_r = blend * (best_seq / best_seq.sum()) + (1 - blend) * ramp / ramp.sum()
        elif init_type == 2:
            c1 = rng.uniform(0.15, 0.45)
            c2 = 1.0 - c1
            x = np.linspace(0, 1, n_best)
            a_r = 0.5 * np.exp(-0.5 * ((x - c1) / 0.12) ** 2) + \
                  0.5 * np.exp(-0.5 * ((x - c2) / 0.12) ** 2)
        elif init_type == 3:
            alpha = rng.uniform(0.3, 2.5)
            x = np.linspace(0.01, 1.0, n_best)
            a_r = x ** alpha if rng.uniform() < 0.5 else (1.0 - x) ** alpha
        else:
            # Blend of best with random smooth function
            freq = rng.uniform(1.0, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 1, n_best)
            wave = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
            blend = rng.uniform(0.3, 0.7)
            a_r = blend * (best_seq / best_seq.sum()) + (1 - blend) * wave / wave.sum()

        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r
        else:
            a_r = np.ones(n_best) / n_best

        alloc = min(remaining * 0.1, 35.0)
        a_opt, obj_opt = pg_descent(a_r, alloc,
                                     T_start=5.0, T_end=250.0,
                                     beta=0.9, eta_init=0.3, seed=seed + 200 + r)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (type={init_type}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_opt:.8f}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 50.0)
        a_opt, obj_opt = pg_descent(best_seq, alloc,
                                     T_start=20.0, T_end=800.0,
                                     beta=0.95, eta_init=0.1, seed=seed + 500)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final PG: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    if remaining > 15 and n_final <= 1000:
        alloc = min(remaining * 0.4, 50.0)
        a_lp, obj_lp = lp_alternating(best_seq, time_budget=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
