Alternative formulation: Instead of `min M s.t. A@x <= M, sum(x)=1, x>=0`, we can use:
`min (1^T x) s.t. (A@x)[k] <= M for all k, x >= 0`

Then the optimal value is `sum(x_opt)` and the normalized solution is `x_opt / sum(x_opt)`. The peak is `M / sum(x_opt)`.

**Let me also consider using a different approach entirely: successive convex approximation.**

The problem `min_a max_k (a*a)[k] s.t. a >= 0, sum(a) = 1` can be viewed as:
`min_a, M M s.t. (a*a)[k] <= M for all k, a >= 0, sum(a) = 1`

This is a QCQP (quadratically constrained quadratic program) with a linear objective. It's non-convex because the constraints `(a*a)[k] <= M` are not convex (they're non-convex quadratic constraints in a).

However, we can use DCP (conservative convex approximation): at a point `a_0`, the constraint `(a*a)[k] <= M` can be approximated by its first-order Taylor approximation:
`(a*a)[k] ≈ (a_0*a_0)[k] + 2 * sum_j a_0[k-j] * (a_j - a_0[j])`

This gives a convex (linear) approximation of the constraint. Using this, we get a sequence of LPs that converge to a local optimum.

This is essentially the same as the LP alternating method! Because the LP `min M s.t. (a_0 * g)[k] <= M, sum(g)=1, g>=0` is exactly the first-order approximation of the nonlinear feasibility problem.

OK so the LP alternating method is the right approach. The question is just how to implement it efficiently.

"""
LP-Alternating optimizer with aggressive active-set management, 
warm starts, FFT-based convolution, and multi-scale search.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_max = min(2 * n - 2, j + n - 1)
        A[j:k_max + 1, j] = a[:k_max - j + 1]
    return A


def solve_lp(a, A, active_mask=None, M_ub=None):
    """
    Solve LP: min M s.t. (A @ g)[k] <= M for k in active set, sum(g)=1, g>=0.
    """
    n = len(a)
    
    if active_mask is not None:
        K = int(np.sum(active_mask))
        A_active = A[active_mask]
    else:
        K = 2 * n - 1
        A_active = A
    
    # Variables: (g_0, ..., g_{n-1}, M)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A_active, -np.ones((K, 1))])
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            g = g / gs
            return g, float(res.x[n])
    
    return None, None


def lp_alternating_fast(a, time_budget=1000.0, verbose=True):
    """
    Fast LP alternating with active set management.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iteration = 0
    stall = 0
    
    # Build full constraint matrix
    A_full = build_conv_matrix(a)
    
    while time.monotonic() - start < time_budget:
        iteration += 1
        t_iter = time.monotonic() - start
        
        # Compute current convolution
        conv = compute_conv(a)
        M_current = float(np.max(conv))
        
        # Determine active set: constraints where (a*a)[k] > 0.95 * M_current
        # This gives us the "nearly active" constraints
        threshold = 0.90 * M_current
        active_mask = conv > threshold
        
        # Ensure at least a few constraints are active
        if np.sum(active_mask) < 10:
            # Take top 10
            top_idx = np.argsort(conv)[::-1][:10]
            active_mask = np.zeros(2*n-1, dtype=bool)
            active_mask[top_idx] = True
        
        # Solve LP with active constraints
        g_lp, M_lp = solve_lp(a, A_full, active_mask=active_mask)
        
        if g_lp is None:
            # Try with all constraints
            g_lp, M_lp = solve_lp(a, A_full, active_mask=None)
            if g_lp is None:
                break
        
        # Check if LP solution violates any inactive constraint
        if np.sum(active_mask) < 2 * n - 1:
            lp_conv = A_full @ g_lp
            max_inactive = 0
            for k in range(2 * n - 1):
                if not active_mask[k] and lp_conv[k] > M_lp + 1e-10:
                    max_inactive = max(max_inactive, lp_conv[k])
            if max_inactive > M_lp + 1e-10:
                # Need to re-solve with expanded active set
                active_mask = (conv > threshold) | (lp_conv > M_lp)
                g_lp, M_lp = solve_lp(a, A_full, active_mask=active_mask)
                if g_lp is None:
                    g_lp, M_lp = solve_lp(a, A_full, active_mask=None)
                    if g_lp is None:
                        break
        
        # Line search: try different t values
        best_t = 0.5
        best_t_obj = np.inf
        for t in [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        a_new = (1.0 - best_t) * a + best_t * g_lp
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Rebuild constraint matrix periodically
            if iteration % 10 == 0:
                A_full = build_conv_matrix(a)
        else:
            stall += 1
            if stall > 30:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 30, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                A_full = build_conv_matrix(a)
                stall = 0
        
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"  LP iter {iteration}: obj={best_obj:.8f}, t={t_iter:.0f}s, "
                  f"active={np.sum(active_mask)}, M_lp={M_lp:.6f}")

    return best_a


def mwu_optimize(a, time_budget=100.0, top_k=10):
    """Multiplicative Weight Update optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.05
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M_current = float(np.max(conv))
        k_indices = np.argsort(conv)[::-1][:top_k]
        
        # MWU gradient
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            d_start = k + 1 - j_end
            d_end = k - j_start + 1
            grad[j_start:j_end] += a[d_start:d_end][::-1]
        grad /= len(k_indices)
        
        # Update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(eta * 1.05, 0.5)
        else:
            no_improve += 1
            a = best_a.copy()
            eta = max(eta * 0.8, 0.005)
            if no_improve > 30:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                no_improve = 0
                eta = 0.05

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        # Try random pair swaps
        improved = False
        for _ in range(20):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = frac * a[i]
                if delta > a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                obj_try = compute_obj(a)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    break

        if not improved and it > 500:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
