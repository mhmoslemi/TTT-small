import numpy as np

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_obj_and_grad(x, tau):
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    n = len(a)
    a_norm = a / np.sum(a)
    
    try:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass
    
    try:
        A = build_conv_matrix(a_norm)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        if improved:
            stall = 0
            if verbose:
                print(f"      LP step: {best_obj:.10f}")
        else:
            stall += 1
            if stall > 8:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
    return best_a


def exp_grad_optimize(a_init, budget):
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 3.0 / n
    for it in range(20000):
        if time.monotonic() - start > budget:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        grad_k = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad_k[i] = 2.0 * a[j]
        a_new = a * np.exp(-eta * grad_k)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)
        ot = compute_obj(a_new)
        if ot < best_obj:
            best_obj = ot
            best_a = a_new.copy()
            a = a_new
            eta = max(eta * 0.998, 0.05 / n)
        else:
            eta *= 0.93
            if eta < 0.01 / n:
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                eta = 1.5 / n
    return best_a


def window_init(n, wtype, rng):
    x = np.linspace(0, 1, n)
    if wtype == 0:
        a = np.ones(n)
    elif wtype == 1:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif wtype == 2:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif wtype == 3:
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif wtype == 4:
        i = np.arange(n)
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / max(n - 1, 1)) + 0.08 * np.cos(4.0 * np.pi * i / max(n - 1, 1))
    elif wtype == 5:
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif wtype == 6:
        a = np.exp(-2.0 * (x - 0.5) ** 2) + 0.1
    elif wtype == 7:
        r = rng.random()
        a = np.exp(-r * 3.0 * (x - 0.5) ** 2) + 0.2
    else:
        a = rng.uniform(0.1, 1.0, n)
    a = np.maximum(a, 0.001)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)
    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): {obj:.10f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default (n=800): {best_score:.10f}")

    n_par = len(best_seq)
    print(f"\n--- Phase 1: Gradient on parent (n={n_par}) ---")
    rem = deadline - time.monotonic()
    if rem > 60:
        a_g = grad_optimize(best_seq, min(rem * 0.2, 150.0), True)
        o = compute_obj(a_g)
        if o < best_score:
            best_score = o
            best_seq = a_g.copy()
            print(f"  grad: {best_score:.10f}")

    print(f"\n--- Phase 2: Cosine search ---")
    rem = deadline - time.monotonic()
    if rem > 40:
        a_c = cosine_search(len(best_seq), min(rem * 0.08, 60.0), True)
        if a_c is not None:
            o = compute_obj(a_c)
            if o < best_score:
                best_score = o
                best_seq = a_c.copy()
                print(f"  cosine: {best_score:.10f}")
            else:
                print(f"  cosine best: {o:.10f}")

    print(f"\n--- Phase 3: Window initializations ---")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_w = len(best_seq)
        for wt in [1, 2, 3, 4, 5, 6]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            a_w = window_init(n_w, wt, rng)
            a_opt = grad_optimize(a_w, min(rem * 0.08, 30.0))
            o = compute_obj(a_opt)
            if o < best_score:
                best_score = o
                best_seq = a_opt.copy()
                print(f"  window {wt}: {best_score:.10f} **")
            else:
                print(f"  window {wt}: {o:.10f}")

    print(f"\n--- Phase 4: Exp gradient ---")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_e = exp_grad_optimize(best_seq, min(rem * 0.06, 30.0))
        o = compute_obj(a_e)
        if o < best_score:
            best_score = o
            best_seq = a_e.copy()
            print(f"  exp-grad: {best_score:.10f}")

    print(f"\n--- Phase 5: LP alternating ---")
    rem = deadline - time.monotonic()
    if rem > 60:
        a_l = lp_optimize(best_seq, min(rem * 0.15, 90.0), True)
        o = compute_obj(a_l)
        if o < best_score:
            best_score = o
            best_seq = a_l.copy()
            print(f"  LP: {best_score:.10f}")

    print(f"\n--- Phase 6: Multi-n exploration ---")
    n_cands = sorted(set([
        max(300, n_par - 200), max(300, n_par - 100), n_par,
        n_par + 100, n_par + 200, n_par + 300, n_par + 500,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_cands = [x for x in n_cands if 250 <= x <= 2500]
    for nt in n_cands:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        if nt == len(best_seq):
            a_t = best_seq.copy()
        else:
            a_t = resample_seq(best_seq, nt)
        alloc = min(rem * 0.1, 35.0)
        a_o = grad_optimize(a_t, alloc)
        o = compute_obj(a_o)
        if o < best_score:
            best_score = o
            best_seq = a_o.copy()
            n_par = nt
            print(f"  n={nt}: {best_score:.10f} **")
        else:
            print(f"  n={nt}: {o:.10f}")

    print(f"\n--- Phase 7: Intensify (n={len(best_seq)}) ---")
    rem = deadline - time.monotonic()
    if rem > 50:
        a_g2 = grad_optimize(best_seq, min(rem * 0.2, 80.0), True)
        o = compute_obj(a_g2)
        if o < best_score:
            best_score = o
            best_seq = a_g2.copy()
            print(f"  grad2: {best_score:.10f}")
    rem = deadline - time.monotonic()
    if rem > 40:
        a_l2 = lp_optimize(best_seq, min(rem * 0.15, 50.0), True)
        o = compute_obj(a_l2)
        if o < best_score:
            best_score = o
            best_seq = a_l2.copy()
            print(f"  LP2: {best_score:.10f}")

    print(f"\n--- Phase 8: Diverse restarts ---")
    nb = len(best_seq)
    rem = deadline - time.monotonic()
    n_rest = max(2, min(8, int(rem / 30)))
    for r in range(n_rest):
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.02, nb))
        elif r == 1:
            nr = nb + int(rng.integers(-200, 200))
            nr = max(350, min(1800, nr))
            a_r = resample_seq(best_seq, nr)
            a_r = a_r * np.exp(rng.normal(0, 0.02, len(a_r)))
        elif r == 2:
            a_r = window_init(nb, int(rng.integers(0, 8)), rng)
        elif r == 3:
            x = np.linspace(0, 1, nb)
            alpha = rng.uniform(1.0, 5.0)
            a_r = np.exp(-alpha * (x - 0.5) ** 2) + 0.15
        else:
            a_r = window_init(nb, int(rng.integers(0, 8)), rng)
        a_r = np.maximum(a_r, 0.001)
        a_r = a_r / np.sum(a_r)
        a_opt = grad_optimize(a_r, min(20.0, rem * 0.35))
        o = compute_obj(a_opt)
        if o < best_score:
            best_score = o
            best_seq = a_opt.copy()
            nb = len(best_seq)
            print(f"  restart {r}: {best_score:.10f} **")
        else:
            print(f"  restart {r}: {o:.10f}")

    print(f"\n--- Phase 9: Final polish ---")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_f = grad_optimize(best_seq, min(rem * 0.4, 40.0), True)
        o = compute_obj(a_f)
        if o < best_score:
            best_score = o
            best_seq = a_f.copy()
            print(f"  final grad: {best_score:.10f}")
    rem = deadline - time.monotonic()
    if rem > 12:
        a_fl = lp_optimize(best_seq, min(rem * 0.5, 20.0))
        o = compute_obj(a_fl)
        if o < best_score:
            best_score = o
            best_seq = a_fl.copy()
            print(f"  final LP: {best_score:.10f}")

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
