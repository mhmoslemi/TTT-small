def compute_conv(a):

    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):

    s = float(np.sum(a))

def build_conv_matrix(a):

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization for polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 10:
                break

    return best_a


def random_init(n, rng, mode=None):
    if mode is None:
        mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = np.maximum(a, 1e-6)
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    elif mode == 6:
        # Hann window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = np.maximum(a, 1e-6)
        a = a / np.sum(a)
    else:
        # Raised cosine with random width
        width = rng.uniform(0.3, 0.8)
        x = np.arange(n) / n
        a = np.cos(np.pi * (x - 0.5) / width) ** 2
        a = np.maximum(a, 1e-6)
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    # Quick Adam optimization

    return result
