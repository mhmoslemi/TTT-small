        return v

        j_end = k - i_start

import time

    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]

    """Build matrix A (m x n) such that A @ g = conv(a, g) at selected positions."""
    n = len(a)
    if selected_k is None:
        selected_k = list(range(2 * n - 1))
    A = np.zeros((len(selected_k), n))
    for row, k in enumerate(selected_k):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            A[row, i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return A

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((len(top_idx), n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(len(top_idx))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def solve_lp_full(a, verbose=False):
    """Solve full LP with all constraints."""
    n = len(a)
    a = a / np.sum(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build full constraint matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            A[k, i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def golden_section_line_search(a, direction, obj_current, time_limit=0.5):
    """
    Golden section search for best t in [0, 1] for a + t*(direction - a).
    Returns (best_t, best_obj, best_point).
    """
    start = time.time()
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    lo, hi = 0.0, 1.0
    # Initial points
    d = hi - lo
    x1 = hi - phi * d
    x2 = lo + phi * d
    
    def eval_t(t):
        pt = (1 - t) * a + t * direction
        pt = np.maximum(pt, 0)
        s = float(np.sum(pt))
        if s < 1e-10:
            return np.inf
        pt = pt / s
        return compute_obj(pt)
    
    f1 = eval_t(x1)
    f2 = eval_t(x2)
    
    best_t = 0.0
    best_f = obj_current
    if f1 < best_f:
        best_f = f1
        best_t = x1
    if f2 < best_f:
        best_f = f2
        best_t = x2
    
    iterations = 0
    while (hi - lo) > 1e-6 and time.time() - start < time_limit:
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            d = hi - lo
            x1 = hi - phi * d
            f1 = eval_t(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            d = hi - lo
            x2 = lo + phi * d
            f2 = eval_t(x2)
        
        if f1 < best_f:
            best_f = f1
            best_t = x1
        if f2 < best_f:
            best_f = f2
            best_t = x2
        
        iterations += 1
        if iterations > 50:
            break
    
    pt = (1 - best_t) * a + best_t * direction
    pt = np.maximum(pt, 0)
    s = float(np.sum(pt))
    if s < 1e-10:
        return 0.0, obj_current, a.copy()
    pt = pt / s
    return best_t, best_f, pt


def lp_alternating_momentum(a, time_budget=300.0, top_K=30, verbose=False):
    """
    LP alternating with:
    - Constraint-selected LP for speed
    - Golden section line search
    - Nesterov momentum
    - Adaptive K
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.time()
    
    # Nesterov momentum
    y = a.copy()
    t_k = 1.0
    momentum_effective = 0.0
    
    stall = 0
    K = top_K
    iter_count = 0
    use_full_lp_every = 10  # Use full LP every N iterations for correction
    
    while time.time() - start < time_budget:
        remaining = time_budget - (time.time() - start)
        if remaining < 2:
            break
        
        # Choose LP type
        use_full = (iter_count % use_full_lp_every == use_full_lp_every - 1)
        
        if use_full:
            g_lp, M_lp = solve_lp_full(y)
        else:
            g_lp, M_lp = solve_lp_fast(y, top_K=K)
        
        if g_lp is None:
            g_lp = y.copy()
        
        # Nesterov step: interpolate between y and momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y_next = y + beta * (y - a)
        y_next = np.maximum(y_next, 0)
        s_yn = float(np.sum(y_next))
        if s_yn > 1e-10:
            y_next = y_next / s_yn
        
        # Solve LP from y_next for a slightly different direction
        if not use_full:
            g_lp2, _ = solve_lp_fast(y_next, top_K=K)
            if g_lp2 is not None:
                g_lp = 0.5 * g_lp + 0.5 * g_lp2
                s_g = float(np.sum(g_lp))
                if s_g > 1e-10:
                    g_lp = g_lp / s_g
        
        # Golden section line search
        t_best, obj_try, a_try = golden_section_line_search(
            a, g_lp, best_obj, time_limit=min(0.3, remaining * 0.1)
        )
        
        iter_count += 1
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            t_k = t_new
            stall = 0
            # Adapt K: increase if making progress
            K = min(100, K + 1)
        else:
            stall += 1
            # Don't update a, just update y
            a = a  # stay
            t_k = t_new if stall < 20 else 1.0
            if stall > 40:
                # Perturb to escape
                rng = np.random.default_rng(iter_count * 6151 + n)
                scale = min(0.2, 0.02 + 0.01 * (stall // 40))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                y = a.copy()
                t_k = 1.0
                stall = 0
                K = max(20, K - 5)
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.time() - start
            print(f"    LP-M it{iter_count}: obj={best_obj:.8f}, K={K}, t={t_k:.2f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")
    
    return best_a


def subgradient_fista(a, time_budget=200.0, verbose=False):
    """
    FISTA with adaptive log-sum-exp smoothing for the max-conv objective.
    Uses all peaks with softmax weighting (adaptive temperature).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.time()
    
    y = a.copy()
    t_k = 1.0
    tau = 20.0  # smoothing parameter
    stall = 0
    it = 0
    
    while time.time() - start < time_budget:
        # Adaptive tau: start moderate, increase for sharper approximation
        progress = min(1.0, (time.time() - start) / time_budget)
        tau = 10.0 + 100.0 * progress ** 2
        
        # Compute smoothed gradient
        conv = compute_conv(y)
        max_c = np.max(conv)
        if max_c < 1e-15:
            break
        # Softmax weights (temperature = 1/tau)
        weights = np.exp((conv - max_c) * tau)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights /= w_sum
        
        # Gradient
        g = np.zeros(n)
        for k in range(len(weights)):
            if weights[k] < 1e-10:
                continue
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += weights[k] * y[j_start:j_end + 1][::-1]
        
        # Step size: backtracking
        L = tau * max_c * 2.0 + 1.0
        step = 1.0 / L
        
        # Try the step
        x_new = y - step * g
        x_new = project_simplex(x_new)
        
        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y_new = project_simplex(y_new)
        
        a_old = a.copy()
        a = x_new
        y = y_new
        t_k = t_new
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Restart
                y = a.copy()
                t_k = 1.0
                tau = max(10.0, tau * 0.5)
                stall = 0
                # Small perturbation
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                y = a.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.time() - start
            print(f"    FISTA it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={t_k:.1f}")
        
        it += 1
    
    return best_a


def random_phase_init(n, rng, num_trials=5):
    """Generate non-negative sequence with approximately flat spectrum."""
    best_a = None
    best_obj = np.inf
    
    for _ in range(num_trials):
        # Random phases in frequency domain
        phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
        X = np.exp(1j * phases)
        X[0] = 1.0  # DC: real positive
        if n % 2 == 0:
            X[-1] = 1.0  # Nyquist: real positive
        a = np.abs(irfft(X * n, n))
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
    return best_a


def taper_init(n, rng, mode=0):
    """Various structured initializations."""
    if mode == 0:
        # Sine taper (raised cosine)
        x = np.linspace(0, np.pi, n)
        a = (1 + np.cos(x)) / 2
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 1:
        # Gaussian centered
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2 / 4)
        a = a / a.sum()
    elif mode == 2:
        # Double-peaked
        x = np.linspace(0, 1, n)
        c1, c2 = 0.3, 0.7
        w = 0.1
        a = np.exp(-(x - c1) ** 2 / (2 * w ** 2)) + np.exp(-(x - c2) ** 2 / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(x))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Inverse Gaussian (U-shaped)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (1.0 + 4.0 * x ** 2)
        a = a / a.sum()
    elif mode == 5:
        # Random smooth
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amps = rng.uniform(0.2, 0.8, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, am in zip(freqs, phases, amps):
            a += am * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_polish(a, time_budget=30.0):
    """
    Fine coordinate descent: try moving small amounts of mass
    from peak-contributing positions to non-peak positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.time()
    conv = compute_conv(a)
    
    while time.time() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False
        
        # Find top contributors to peak
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                c = a[i] * a[j]
                contribs.append((c, i, j))
        contribs.sort(reverse=True)
        
        # Find non-contributors
        contrib_indices = set()
        for _, i, j in contribs:
            contrib_indices.add(i)
            if j != i:
                contrib_indices.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_indices]
        
        if not non_contrib:
            # All positions contribute; try redistributing among contributors
            non_contrib = [i for i, _ in enumerate(a) if a[i] < np.mean(a)]
        
        for c_val, i, j in contribs[:15]:
            if time.time() - start > time_budget:
                break
            if c_val < 1e-10:
                break
            for target in non_contrib[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = frac * a[i]
                    if delta < 1e-12 or a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            break
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):

        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))

                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

            obj_f = compute_obj(a_f)

            a_lp = lp_alternating_momentum(best_seq, time_budget=alloc, top_K=40, verbose=True)

            if obj_lp < best_score:

                best_seq = a_lp.copy()
                print(f"  LP chunk {chunk}: obj={best_score:.8f} **")
            else:
                print(f"  LP chunk {chunk}: obj={obj_lp:.8f}")
    
    # Phase 3: FISTA refinement on current best
    remaining = deadline - time.time()
    if remaining > 30:
        print(f"\n=== Phase 3: FISTA Refinement ===")
        fista2_budget = min(remaining * 0.15, 120.0)
        a_f2 = subgradient_fista(best_seq, time_budget=fista2_budget, verbose=True)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()

    print(f"\n=== Phase 4: Multi-n Exploration ===")

            a_rp = random_phase_init(n_best, rng, num_trials=3)
            a_r = 0.5 * a_rp + 0.5 * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:

            a_r = best_seq * np.exp(rng.normal(0, 0.1, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        
        restart_budget = min(remaining * 0.15, 80.0)
        a_opt = lp_alternating_momentum(a_r, time_budget=restart_budget, top_K=30, verbose=False)

    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")

    if remaining > 20:
        # LP polish
        lp_alloc = min(remaining * 0.3, 100.0)
        a_lp = lp_alternating_momentum(best_seq, time_budget=lp_alloc, top_K=50, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
    
    remaining = deadline - time.time()
    if remaining > 15:
        # FISTA polish
        fista_alloc = min(remaining * 0.25, 60.0)
        a_f = subgradient_fista(best_seq, time_budget=fista_alloc, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")
    
    remaining = deadline - time.time()
    if remaining > 10:
        # Coordinate polish
        cd_alloc = min(remaining * 0.3, 30.0)
        a_cd = coordinate_polish(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
    
    remaining = deadline - time.time()
    if remaining > 15:
        # One more LP round
        a_last = lp_alternating_momentum(best_seq, time_budget=min(remaining * 0.4, 40.0),
                                          top_K=60, verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")
    
    # Final coordinate polish
    remaining = deadline - time.time()
    if remaining > 5:
        a_cd2 = coordinate_polish(best_seq, time_budget=min(remaining * 0.5, 10.0))
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)
    
    elapsed = time.time() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
