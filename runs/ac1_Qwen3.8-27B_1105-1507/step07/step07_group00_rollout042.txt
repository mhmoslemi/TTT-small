Optimized step function search using LP alternating with constraint reduction,
adaptive subgradient methods, and diversification strategies.
Targets: minimize 2*n*max(conv(a,a))/(sum(a))^2
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a, top_k=None):
    """Build matrix A (m x n) such that A @ g = conv(a, g) for selected peaks."""
    n = len(a)
    if top_k is None:
        m = 2 * n - 1
    else:
        conv_a = compute_conv(a)
        top_idx = np.argsort(conv_a)[::-1][:top_k]
        m = len(top_idx)
    
    A = np.zeros((m, n))
    if top_k is None:
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a[:r]
    else:
        for idx, k in enumerate(top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                A[idx, i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return A

def solve_lp_fast(a, top_k=200, max_cons=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top_k peaks, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Get top peaks
    conv_a = compute_conv(a)
    if max_cons is not None:
        top_k = min(top_k, max_cons)
    top_idx = np.argsort(conv_a)[::-1][:top_k]
    
    # Build constraint matrix
    m = len(top_idx)
    A = np.zeros((m, n))
    for idx, k in enumerate(top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            A[idx, i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    
    # LP: min M s.t. A@g - M <= 0, sum(g) = 1, g >= 0, M >= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@g - M <= 0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(g) = 1
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                # Verify: compute actual max
                conv_g = compute_conv(a * g_val)  # This is conv(a, g) normalized
                # Actually we need conv(a, g) where g is the direction
                conv_check = np.zeros(2*n-1)
                for i in range(n):
                    if a[i] != 0:
                        conv_check[i:i+n] += a[i] * g_val
                M_actual = float(np.max(conv_check))
                return g_val, M_actual
    except Exception:
        pass
    
    return None, None

def subgradient_optimize(a, time_budget=100.0, topK=5, verbose=False):
    """
    Subgradient descent with top-K peak weighting and Barzilai-Borwein step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = a.copy()
    prev_g = None
    step_size = 0.1 / np.sqrt(n)
    it = 0
    stall = 0
    
    # Softmax temperature for peak weighting
    temperature = 5.0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 3:
            break
            
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Top-K peak selection with softmax weighting
        top_idx = np.argsort(conv)[::-1][:topK]
        peak_vals = conv[top_idx]
        weights = np.exp((peak_vals - peak_vals[0]) * temperature / max_val)
        weights /= weights.sum()

        # Compute weighted subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        # Barzilai-Borwein step
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-15:
                bb = ss / abs(sy)
                step_size = min(max(step_size * 0.7, 1e-6), bb * 2, 0.5 / n)
        
        prev_a = a.copy()
        prev_g = g.copy()

        # Projected gradient step
        a_new = a - step_size * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Gradually increase temperature (focus more on single peak)
            temperature = min(20.0, temperature * 1.01)
        else:
            stall += 1
            if stall > 50:
                # Reset and perturb
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.15, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                step_size = 0.1 / np.sqrt(n)
                prev_g = None
                prev_a = a.copy()
                temperature = 5.0
                stall = 0
            else:
                step_size *= 0.9
                temperature = max(1.0, temperature * 0.95)

        it += 1

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Front-loaded (exponential decay)
        a = np.exp(-np.linspace(0, 2, n))
        a = a / a.sum()
    elif mode == 3:
        # Back-loaded
        a = np.exp(-np.linspace(2, 0, n))
        a = a / a.sum()
    elif mode == 4:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Inverted triangle
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 6:
        # U-shaped (arcosine)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - 0.99 * x**2) + 0.1)
        a = a / a.sum()
    elif mode == 7:
        # Smooth random
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Linear ramp (front-loaded linear)
        a = np.linspace(1, 0.3, n)
        a = a / a.sum()
    elif mode == 9:
        # Quadratic front-loaded
        x = np.linspace(0, 1, n)
        a = (1 - x)**2 + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""
Optimized step function search using LP alternating with constraint reduction,
adaptive subgradient methods, multi-n exploration, and diversification strategies.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross convolution a*b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_fast(a, top_k=200):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top_k peaks, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)

    conv_a = compute_conv(a)
    top_idx = np.argsort(conv_a)[::-1][:top_k]

    m = len(top_idx)
    A = np.zeros((m, n))
    for idx, k in enumerate(top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            A[idx, i_start:i_end + 1] = a[i_start:i_end + 1][::-1]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                return g_val, float(res.x[n])
    except Exception:
        pass

    return g, M


def lp_alternating(a, time_budget=200.0, top_k=200, verbose=False):
    """
    LP alternating with adaptive line search and stall recovery.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    iter_count = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 5:
            break

        g_lp, M_lp = solve_lp_full(a, top_k=top_k)
        if g_lp is None:
            break

        # Adaptive line search
        if iter_count < 3:
            candidate_ts = [0.3, 0.5, 0.7, 1.0]
        else:
            base_ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
            if 0.05 < last_t < 1.0:
                extra = [last_t * 0.6, last_t * 0.8, last_t, last_t * 1.2, last_t * 1.5]
                base_ts = sorted(set(base_ts + [t for t in extra if 0 < t <= 1.0]))

        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in base_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and iter_count % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.3f}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 25:
                rng = np.random.default_rng(iter_count * 104729 + n)
                scale = min(0.5, 0.05 + 0.04 * (stall // 25))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                last_t = 0.5

    return best_a


def subgradient_optimize(a, time_budget=100.0, topK=5, verbose=False):
    """
    Subgradient descent with top-K softmax weighting and adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = a.copy()
    prev_g = None
    step_size = 0.05 / np.sqrt(n)
    it = 0
    stall = 0
    temperature = 3.0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 3:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Top-K with softmax weighting
        top_idx = np.argsort(conv)[::-1][:topK]
        peak_vals = conv[top_idx]
        weights = np.exp((peak_vals - peak_vals[0]) * temperature / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                g[i_start:i_end + 1] += w * a[i_start:i_end + 1][::-1]

        # Adaptive step size
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-15:
                bb = ss / abs(sy)
                step_size = min(max(step_size * 0.8, 1e-7), bb * 1.5, 0.3 / n)

        prev_a = a.copy()
        prev_g = g.copy()

        a_new = a - step_size * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            temperature = min(15.0, temperature * 1.02)
        else:
            stall += 1
            if stall > 40:
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                step_size = 0.05 / np.sqrt(n)
                prev_g = None
                prev_a = a.copy()
                temperature = 3.0
                stall = 0
            else:
                step_size *= 0.92
                temperature = max(1.0, temperature * 0.95)

        it += 1

    return best_a


def multiplicative_optimize(a, time_budget=80.0, topK=3, verbose=False):
    """
    Multiplicative (entropy mirror) descent with top-K peaks.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.5 / np.sqrt(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 3:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        top_idx = np.argsort(conv)[::-1][:topK]
        peak_vals = conv[top_idx]
        weights = np.exp((peak_vals - peak_vals[0]) * 5.0 / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                g[i_start:i_end + 1] += w * a[i_start:i_end + 1][::-1]

        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = max(1e-5, lr * 0.995)
        else:
            stall += 1
            if stall > 200:
                rng = np.random.default_rng(it * 17 + n)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(1e-5, lr * 0.5)
                stall = 0
            else:
                lr = max(1e-5, lr * 0.97)

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Front-loaded exponential
        a = np.exp(-np.linspace(0, 2.5, n))
        a = a / a.sum()
    elif mode == 3:
        # Back-loaded exponential
        a = np.exp(-np.linspace(2.5, 0, n))
        a = a / a.sum()
    elif mode == 4:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 5, n // 5]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Inverted triangle
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 6:
        # U-shaped
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - 0.98 * x ** 2) + 0.05)
        a = a / a.sum()
    elif mode == 7:
        # Smooth random
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Linear front-loaded
        a = np.linspace(1.0, 0.2, n)
        a = a / a.sum()
    elif mode == 9:
        # Quadratic front-loaded
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.05
        a = a / a.sum()
    elif mode == 10:
        # S-curve front-loaded
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(4.0 * (x - 0.4)))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def peak_equalize(a, time_budget=30.0):
    """
    Iteratively reduce the top peak by moving mass to positions that
    don't contribute to that peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    it = 0
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 2:
            break

        k_star = int(np.argmax(conv))

        # Find contributors to this peak
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)

        # Compute contribution of each position
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = k_star - i
            contrib[i] = a[i] * a[j]

        # Find non-contributors (positions that don't affect this peak)
        non_contrib = set(range(n))
        for i in range(i_start, i_end + 1):
            j = k_star - i
            non_contrib.discard(i)
            non_contrib.discard(j)
        non_contrib_list = sorted(non_contrib)

        if not non_contrib_list:
            break

        # Move mass from highest contributor to best non-contributor
        top_contributors = np.argsort(contrib)[::-1][:10]
        improved = False
        for i in top_contributors:
            if contrib[i] < 1e-10:
                continue
            for target in non_contrib_list[:5]:
                for frac in [0.03, 0.07, 0.15]:
                    delta = frac * a[i]
                    if delta <= 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-10:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
            if improved:
                break

        if not improved:
            break

        it += 1

    return best_a
