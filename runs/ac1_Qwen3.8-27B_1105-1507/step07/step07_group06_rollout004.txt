"""
Improved hybrid optimization: Aggressive LP alternating with active-set exploitation,
subgradient descent with momentum, and diverse restarts.
Key improvements over previous: (1) faster LP with precomputed matrix, (2) subgradient
with momentum for smoother convergence, (3) active-set-based perturbation, (4) 
more diverse initializations including exponential decay and ramp functions.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import linprog, minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(A, n):
    """Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0. A is precomputed."""
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def subgradient_momentum(a, time_budget=60.0):
    """
    Subgradient descent with momentum and top-k smoothing.
    Uses the fact that grad of max at argmax k is 2*a_reversed_shifted.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    lr = 0.05
    momentum = 0.85
    k_smooth = 3  # use top-k for smoothing

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Top-k smoothing for a better gradient estimate
        top_k_idx = np.argsort(conv)[-k_smooth:]
        # Weight by exponential of (conv - max) / temperature
        temp = 0.01
        weights = np.exp((conv[top_k_idx] - conv[top_k_idx[0]]) / temp)
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for ki in range(k_smooth):
            k_val = int(top_k_idx[ki])

            valid_end = min(n, k_val + 1)
            for i in range(valid_start, valid_end):
                grad[i] += weights[ki] * 2.0 * a[k_val - i]

        # Center gradient for simplex projection
        grad_centered = grad - np.mean(grad)

        # Momentum
        v = momentum * v + lr * grad_centered
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

            a = np.maximum(a - 0.01 * v, 0)
            a = a / np.sum(a)

        # Decay learning rate slowly
        if it % 500 == 0:
            lr *= 0.95

    return best_a


def coordinate_descent_fast(a, time_budget=30.0):
    """Fast coordinate descent focusing on peak-reducing moves."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Identify contributors to the peak
        # (a*a)[k_star] = sum_i a[i]*a[k_star-i]
        # Elements with large a[i] and a[k_star-i] contribute most
        contrib = np.zeros(n)
        valid_start = max(0, k_star - n + 1)
        valid_end = min(n, k_star + 1)
        for i in range(valid_start, valid_end):
            contrib[i] = a[i] * a[k_star - i]

        # Try moving mass from high-contribution elements to low ones
        top_contributors = np.argsort(contrib)[-20:]
        low_contributors = np.argsort(contrib)[:20]

        for _ in range(30):
            i = int(rng.choice(top_contributors))
            j = int(rng.choice(low_contributors))
            if i == j:
                continue
            delta = rng.uniform(0.005, 0.03) * a[i]
            if delta >= a[i]:
                continue
            a[i] -= delta
            a[j] += delta
            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a.copy()
                improved = True
            else:
                a[i] += delta
                a[j] -= delta

        if not improved and it % 50 == 0:
            # Random two-swap
            for _ in range(100):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = rng.uniform(0.01, 0.05) * min(a[i], a[j])
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
                if improved:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_diverse_init(n, choice, rng):
    """Generate diverse initial sequences with strong asymmetry."""
    if choice == 0:
        # Exponential decay
        alpha = rng.uniform(1.0, 4.0)
        a = np.exp(-alpha * np.arange(n) / (n - 1))
    elif choice == 1:
        # Linear ramp (increasing)
        slope = rng.uniform(1.0, 3.0)
        a = 1.0 + slope * np.arange(n) / (n - 1)
    elif choice == 2:
        # Bimodal: two peaks at different positions
        p1 = int(rng.uniform(0.1, 0.35) * n)
        p2 = int(rng.uniform(0.55, 0.85) * n)
        s1 = rng.uniform(0.05, 0.15) * n
        s2 = rng.uniform(0.05, 0.15) * n
        x = np.arange(n)
        a = np.exp(-(x - p1)**2 / (2*s1**2)) + 0.5 * np.exp(-(x - p2)**2 / (2*s2**2))
    elif choice == 3:
        # Power law
        p = rng.uniform(1.5, 4.0)
        a = (np.arange(n) + 1) ** (-p)
    elif choice == 4:
        # Smoothed step (asymmetric sigmoid)
        mid = rng.uniform(0.3, 0.7) * n
        width = rng.uniform(0.05, 0.2) * n
        a = 1.0 / (1.0 + np.exp((np.arange(n) - mid) / width))
    else:
        # Mixed: blend of decay and ramp
        alpha = rng.uniform(0.5, 2.0)
        decay = np.exp(-alpha * np.arange(n) / n)
        ramp = np.linspace(0.5, 1.5, n)
        blend = rng.uniform(0.3, 0.7)
        a = blend * decay + (1 - blend) * ramp
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: aggressive LP alternating + subgradient + multi-n + diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting optimization, target < {1.5027}")

    # === Phase 1: Quick n-sweep to find best length ===
    print(f"\n=== Phase 1: n-sweep ===")
    n_options = sorted(set([400, 500, 600, 700, n_parent, 900, 1000, 1200]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    n_best_init = n_parent
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        # Quick LP: just a few iterations
        a_opt = lp_alternating(a_try, time_budget=min(remaining * 0.05, 15.0))
        obj_opt = compute_obj(a_opt)
        print(f"  n={n_try}: obj={obj_opt:.8f}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best_init = n_try
            print(f"  ** new best at n={n_try}")

    n_cur = len(best_seq)
    print(f"Best after n-sweep: n={n_cur}, score={best_score:.8f}")

    # === Phase 2: Main LP alternating (primary method) ===
    print(f"\n=== Phase 2: Main LP Alternating ===")
    remaining = deadline - time.monotonic()
    main_budget = min(remaining * 0.50, 600.0)
    a_opt = lp_alternating(best_seq, time_budget=main_budget, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  LP main: obj={best_score:.8f} **")
    else:
        print(f"  LP main: obj={obj_opt:.8f}")

    # === Phase 3: Subgradient with momentum ===
    print(f"\n=== Phase 3: Subgradient + Momentum ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_budget = min(remaining * 0.20, 200.0)
        a_sg = subgradient_momentum(best_seq, time_budget=sg_budget)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 4: Second round of LP (on potentially improved sequence) ===
    print(f"\n=== Phase 4: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        lp2_budget = min(remaining * 0.25, 300.0)
        a_opt = lp_alternating(best_seq, time_budget=lp2_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  LP r2: obj={best_score:.8f} **")
        else:
            print(f"  LP r2: obj={obj_opt:.8f}")

    # === Phase 5: Diverse restarts (try to escape local minima) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        choice = r % 6
        a_r = generate_diverse_init(n_best, choice, rng)

        # Quick LP optimization from this init
        alloc = min(remaining * 0.08, 20.0)
        a_opt = lp_alternating(a_r, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r} (type={choice}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={choice}): obj={obj_opt:.8f}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_budget = min(remaining * 0.15, 60.0)
        a_cd = coordinate_descent_fast(best_seq, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 7: Final intensive LP ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        final_budget = min(remaining * 0.5, 120.0)
        a_final = lp_alternating(best_seq, time_budget=final_budget)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
