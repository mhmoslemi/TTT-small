"""
Optimized low-autocorrelation non-negative sequence finder.
Uses efficient LP alternating (HiGHS), smoothed gradient (L-BFGS-B with log-sum-exp),
subgradient with momentum, and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = a
    return A


def solve_lp_fast(a):
    """Solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0 using HiGHS directly."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None

    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)

    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]

    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """L-BFGS-B on log-sum-exp smoothed objective with adaptive temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = np.log(np.maximum(a, 1e-300))
    x = x - x.max()

    tus = [10.0 / n, 5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.25 / n, 0.1 / n, 0.05 / n]

    for idx, tau in enumerate(tus):
        if time.monotonic() - start > time_budget:
            break

        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_grad(xv, tau)
            return val

        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_grad(xv, tau)
            return g

        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-13,
                                    'maxls': 100})
            x_new = res.x - res.x.max()
            a_new = np.exp(x_new)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_new
                if verbose:
                    print(f"    grad tau={tau:.4f} (idx={idx}): obj={best_obj:.10f}")
            elif verbose:
                print(f"    grad tau={tau:.4f} (idx={idx}): obj={obj_new:.10f} (no improve)")
        except Exception as e:
            if verbose:
                print(f"    grad exception at tau={tau}: {e}")

    return best_a


def subgradient_optimize(a_init, time_budget=30.0):
    """Subgradient method with momentum on the exact max-conv objective."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    beta = 0.85
    lr = 0.5

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: d(conv[k*])/d(a[i]) = 2*a[k*-i] if 0 <= k*-i < n
        idx = np.arange(n)
        valid = (k_star - idx >= 0) & (k_star - idx < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[k_star - idx[valid]]

        # Project onto tangent space (sum(a) = 1)
        grad = grad - np.mean(grad)

        # Momentum
        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 2.0)
        else:
            lr *= 0.93
            if lr < 0.01:
                rng = np.random.default_rng(it * 31)
                a = best_a * (1.0 + 0.02 * rng.normal(0, 1, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                v = np.zeros(n)
                lr = 0.5

    return best_a


def cosine_init(n, rng):
    """Generate cosine-windowed initialization."""
    x = np.arange(n) / n
    alpha = rng.uniform(0.5, 2.5)
    beta = rng.uniform(0.1, 0.4)
    phase = rng.uniform(-1.0, 1.0)
    a = np.cos(alpha * (x - 0.5) * np.pi + phase) ** 2 + beta
    a = np.maximum(a, 0.01)
    a += 0.05 * rng.uniform(0, 1, n)
    a = a / np.sum(a)
    return a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating as primary optimizer, gradient polish,
    subgradient, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, score={best_score:.10f}")

    remaining = deadline - time.monotonic()
    print(f"Time budget: {remaining:.1f}s")

    # === Phase 1: LP Alternating (primary) ===
    print(f"\n=== Phase 1: LP Alternating (primary) ===")
    if remaining > 30:
        alloc = min(remaining * 0.35, 350.0)
        a_lp = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  After LP1: obj={best_score:.10f}")
        else:
            print(f"  LP1: no improvement (best stayed at {best_score:.10f})")

    # === Phase 2: Gradient Polish ===
    print(f"\n=== Phase 2: Gradient Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.15, 100.0)
        a_g = gradient_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_g)
        if obj < best_score:
            best_score = obj
            best_seq = a_g.copy()
            print(f"  After grad: obj={best_score:.10f}")

    # === Phase 3: Subgradient ===
    print(f"\n=== Phase 3: Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.08, 50.0)
        a_sg = subgradient_optimize(best_seq, time_budget=alloc)
        obj = compute_obj(a_sg)
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()
            print(f"  After subgrad: obj={best_score:.10f}")

    # === Phase 4: LP Polish on current best ===
    print(f"\n=== Phase 4: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 150.0)
        a_lp2 = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp2)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()
            print(f"  After LP2: obj={best_score:.10f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_candidates = sorted(set([
            n_best - 100, n_best - 50, n_best,
            n_best + 50, n_best + 100, n_best + 200,
            600, 800, 1000, 1200, 1400
        ]))
        n_candidates = [n for n in n_candidates if 400 <= n <= 2000]

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample(best_seq, n_try)

            alloc = min(remaining * 0.08, 25.0)
            # Quick gradient for this n
            a_opt = gradient_optimize(a_try, time_budget=alloc)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 6: Final LP + Gradient ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 60.0)
        a_f = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 30.0)
            a_g2 = gradient_optimize(best_seq, time_budget=alloc)
            obj = compute_obj(a_g2)
            if obj < best_score:
                best_score = obj
                best_seq = a_g2.copy()
                print(f"  Final grad: obj={best_score:.10f}")

    # === Phase 7: Random Restarts ===
    print(f"\n=== Phase 7: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_restarts = max(1, min(int(remaining / 30), 5))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.02, n_best))
        elif r == 1:
            a_r = cosine_init(n_best, rng)
        elif r == 2:
            a_r = 0.8 * best_seq + 0.2 * np.ones(n_best) / n_best
        elif r == 3:
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = np.maximum(a_r, 1e-10)
        else:
            a_r = 1.0 + 0.3 * np.sin(np.linspace(0, 4 * np.pi, n_best)) + 0.1 * rng.normal(0, 1, n_best)
            a_r = np.maximum(a_r, 0.01)

        a_r = np.maximum(a_r, 1e-10)
        a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.15, 20.0)
        a_opt = gradient_optimize(a_r, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
        print(f"\n=== FINAL: n={len(result)}, evaluated_score={val:.10f}, computed_score={best_score:.10f} ===")
    else:
        print(f"\n=== WARNING: evaluate_sequence returned inf! Fallback ===")
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")

    return result
