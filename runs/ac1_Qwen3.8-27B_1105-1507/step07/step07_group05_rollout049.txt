def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))

def project_simplex(x, target_sum=1.0):
    """Project x onto the simplex {a >= 0, sum(a) = target_sum}."""
    n = len(x)
    # Find lambda such that sum(max(0, x_i - lambda)) = target_sum
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    rho = np.where(x_sorted > 0)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    # Find largest rho such that x_sorted[rho] > (cumsum[rho] - target_sum) / (rho + 1)
    vals = (cumsum[:len(rho)+1] - target_sum) / np.arange(1, len(rho)+2)
    valid = x_sorted[:len(rho)+1] > vals
    if not np.any(valid):
        return np.full(n, target_sum / n)
    rho = np.max(np.where(valid)[0])
    lambda_ = (cumsum[rho] - target_sum) / (rho + 1)
    return np.maximum(x - lambda_, 0)

        return a
    a = a / s
    
    best_a = a.copy()

    while N < 2 * n - 1:
        N <<= 1

    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    lr = 0.1
    t = 0

    tau_min = 20.0

        if time.monotonic() - start > time_budget:
            break
        
        # Compute b = conv(a, a)
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        
        # Compute softmax weights (for log-sum-exp gradient)
        b_shifted = tau * (b - np.max(b))
        w = np.exp(b_shifted)
        w_sum = np.sum(w)
        w = w / w_sum
        
        # Compute gradient: grad_i = sum_k w_k * (a_{k-i} if 0 <= k-i < n)
        # This is a weighted convolution of w with a (reversed)
        a_flip = a[::-1]
        faf = rfft(a_flip, N)
        fw = rfft(w, 2*N)  # Need larger FFT to avoid wrap-around
        # Actually, let me think about this more carefully
        # We want grad_i = sum_k w_k * a_{k-i} for 0 <= k-i < n
        # This is the cross-correlation of w and a

        grad = grad - np.mean(grad)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

def build_conv_matrix(a):

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def project_simplex(x, target_sum=1.0):
    """Project x onto {a >= 0, sum(a) = target_sum}."""
    n = len(x)
    if n == 0:
        return np.array([])
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    positive = x_sorted > 0
    if not np.any(positive):
        return np.full(n, target_sum / n)
    idx = np.where(positive)[0]
    rho_candidates = idx + 1
    vals = (cumsum[rho_candidates - 1] - target_sum) / rho_candidates
    valid = x_sorted[rho_candidates - 1] > vals
    if not np.any(valid):
        return np.full(n, target_sum / n)
    rho = rho_candidates[np.max(np.where(valid)[0])]
    if rho == 0:
        return np.full(n, target_sum / n)
    lambda_ = (cumsum[rho - 1] - target_sum) / rho
    return np.maximum(x - lambda_, 0)


def gs_flat_spectrum(n, N, rng, n_iter=300, target_sum=1.0, soft_power=1.0):
    """Gerchberg-Saxton for flat spectrum initialization with soft projection."""
    a = rng.uniform(0, 1, n)
    a = project_simplex(a, target_sum)

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(n_iter):
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = fft(a_padded)

        mag = np.abs(A)
        mag_target = np.sqrt(np.sum(mag**2) / N)
        # Soft projection: blend current magnitude with target
        if soft_power < 1.0:
            new_mag = mag_target * np.power(mag / (mag_target + 1e-30), soft_power)
        else:
            new_mag = np.full_like(mag, mag_target)
        phases = np.angle(A)
        A_new = new_mag * np.exp(1j * phases)

        x = np.real(ifft(A_new))
        a_new = project_simplex(x[:n], target_sum)

        if np.max(np.abs(a_new - a)) < 1e-12:
            a = a_new
            break
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def gs_snap(a, N, strength=0.5):
    """Apply a single GS snap step: partially move spectrum toward flat."""
    n = len(a)
    a_padded = np.zeros(N)
    a_padded[:n] = a
    A = fft(a_padded)

    mag = np.abs(A)
    mag_target = np.sqrt(np.sum(mag**2) / N)
    phases = np.angle(A)
    # Soft: blend
    new_mag = (1 - strength) * mag + strength * mag_target
    A_new = new_mag * np.exp(1j * phases)

    x = np.real(ifft(A_new))
    a_new = project_simplex(x[:n], 1.0)
    return a_new


def lse_gradient_descent(a, time_budget=120.0, verbose=False, seed=0):
    """
    Adaptive log-sum-exp gradient descent with periodic GS snapping.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1

    # Adam parameters
    beta1 = 0.85
    beta2 = 0.999
    epsilon = 1e-10
    m = np.zeros(n)
    v = np.zeros(n)
    lr = 0.05
    t = 0

    # Adaptive temperature
    tau = 80.0
    tau_min = 30.0
    tau_decay = 0.9995
    tau_increase = 1.02

    stall = 0
    max_stall = 80
    gs_interval = 200  # do a GS snap every this many iterations
    gs_count = 0

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        # Compute b = conv(a, a)
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Compute softmax weights for log-sum-exp gradient
        b_max = np.max(b)
        b_shifted = tau * (b - b_max)
        # Clip for numerical stability
        b_shifted = np.maximum(b_shifted, -50)
        w = np.exp(b_shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = w / w_sum

        # Gradient: grad_i = sum_k w_k * a_{k-i} (cross-correlation)
        a_flip = a[::-1]
        faf = rfft(a_flip, N2)
        fw = rfft(w, N2)
        grad_full = irfft(faf * fw, N2)[:3 * n - 2]
        grad = grad_full[n - 1:2 * n - 1]

        # Center gradient (sum constraint)
        grad = grad - np.mean(grad)

        # Periodic GS snap
        gs_count += 1
        if gs_count >= gs_interval:
            gs_count = 0
            a_snapped = gs_snap(a, N, strength=0.3)
            # Blend with current
            a = 0.5 * a + 0.5 * a_snapped
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            obj_snap = compute_obj(a)
            if obj_snap < best_obj:
                best_obj = obj_snap
                best_a = a.copy()
            continue

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            m.fill(0)
            v.fill(0)
            t = 0
            continue
        a_new = a_new / s_new

        # Compute objective
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.005, 0.5)
            tau = max(tau * tau_decay, tau_min)
        else:
            stall += 1
            lr *= 0.97
            tau = min(tau * tau_increase, 300.0)
            if stall > max_stall:
                # Perturb from best
                noise = rng.normal(0, 0.02, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                m.fill(0)
                v.fill(0)
                t = 0
                lr = 0.05
                stall = 0
                max_stall = min(int(max_stall * 1.5), 500)
            if lr < 1e-12:
                break

        if verbose and it % 1000 == 0:
            print(f"            LSE-GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.CBC]:
            try:
                prob.solve(solver=solver, time_limit=time_limit)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass

    return None


def lp_refine(a, time_budget=60.0, verbose=False):
    """LP-based alternating refinement."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(30):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp = solve_lp_minpeak(a, time_limit=min(10, int(remaining) - 2))
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            if verbose and it % 3 == 0:
                print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate diverse initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5)**2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        a = np.zeros(n)
        for c in [n // 3, 2 * n // 3]:
            a += np.exp(-((np.arange(n) - c)**2) / max((n // 8)**2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Power law tail
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    elif mode == 6:
        # Symmetric bump
        x = np.arange(n) / max(n - 1, 1)
        a = np.exp(-8 * (x - 0.5)**2) + 0.3
        a = a / np.sum(a)
    else:
        # Multi-frequency
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.integers(2, 5) * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: GS spectral flattening init + adaptive LSE-GD + LP polishing + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
    except Exception:
        pass

    # Phase 1: Generate diverse initializations via GS
    print("=== Phase 1: GS Spectral Flattening Init ===")
    n_base = 800
    if parent is not None:
        n_base = len(parent)
        best_seq = parent.copy()
        best_score = compute_obj(best_seq)
        print(f"  Parent (n={n_base}): obj={best_score:.8f}")

    N = 1
    while N < 2 * n_base - 1:
        N <<= 1

    # GS initializations
    n_gs_tries = 6
    gs_candidates = []
    for i in range(n_gs_tries):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        rng_gs = np.random.default_rng(seed + 1000 + i)
        a_gs = gs_flat_spectrum(n_base, N, rng_gs, n_iter=200, target_sum=1.0,
                                 soft_power=0.8 if i % 2 == 0 else 1.0)
        obj_gs = compute_obj(a_gs)
        gs_candidates.append((obj_gs, a_gs))
        if obj_gs < best_score:
            best_score = obj_gs
            best_seq = a_gs.copy()
            print(f"  GS init {i}: obj={obj_gs:.8f} **NEW BEST**")
        else:
            print(f"  GS init {i}: obj={obj_gs:.8f}")

    # Sort GS candidates
    gs_candidates.sort(key=lambda x: x[0])
    if best_seq is None:
        a_init = diverse_init(n_base, rng)
        best_seq = a_init
        best_score = compute_obj(a_init)
        print(f"  Default init: obj={best_score:.8f}")

    # Phase 2: Main GD refinement from best
    print(f"\n=== Phase 2: LSE-GD Main (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 300.0)
    a_gd = lse_gradient_descent(best_seq, time_budget=alloc, verbose=True, seed=seed)
    obj_gd = compute_obj(a_gd)
    if obj_gd < best_score:
        best_score = obj_gd
        best_seq = a_gd.copy()
        print(f"  ** GD Main: obj={best_score:.8f} **")
    else:
        print(f"  GD Main: obj={obj_gd:.8f}")

    # Phase 3: LP polishing
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 120.0)
        a_lp = lp_refine(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # Phase 4: Second GD round (from improved point, different seed)
    print(f"\n=== Phase 4: LSE-GD Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 150.0)
        a_gd2 = lse_gradient_descent(best_seq, time_budget=alloc, seed=seed + 777)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            print(f"  ** GD Round 2: obj={best_score:.8f} **")
        else:
            print(f"  GD Round 2: obj={obj_gd2:.8f}")

    # Phase 5: Try second-best GS candidate
    if len(gs_candidates) > 1:
        print(f"\n=== Phase 5: Second GS Candidate ===")
        remaining = deadline - time.monotonic()
        if remaining > 60:
            a_alt = gs_candidates[1][1].copy()
            # Quick GD
            alloc = min(remaining * 0.15, 60.0)
            a_alt_opt = lse_gradient_descent(a_alt, time_budget=alloc, seed=seed + 500)
            obj_alt = compute_obj(a_alt_opt)
            if obj_alt < best_score:
                best_score = obj_alt
                best_seq = a_alt_opt.copy()
                print(f"  ** Alt GS: obj={best_score:.8f} **")
            else:
                print(f"  Alt GS: obj={obj_alt:.8f}")

    # Phase 6: Multi-n exploration
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr - 100, n_curr + 100, n_curr + 200, n_curr + 400,
        n_curr + 600, 1000, 1200, 1500, 1800, 2000
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500 and n != n_curr]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Quick GS snap + GD
        N_try = 1
        while N_try < 2 * n_try - 1:
            N_try <<= 1
        a_try = gs_snap(a_try, N_try, strength=0.5)
        a_try = project_simplex(a_try, 1.0)

        alloc = min(remaining * 0.15, 60.0)
        a_opt = lse_gradient_descent(a_try, time_budget=alloc, seed=seed + n_try)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 7: Final LP polish + GD
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # GD
        alloc = min(remaining * 0.3, 60.0)
        a_f1 = lse_gradient_descent(best_seq, time_budget=alloc, seed=seed + 999)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  ** Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP
        alloc = min(remaining * 0.4, 45.0)
        a_f2 = lp_refine(best_seq, time_budget=alloc)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

    # Phase 8: A few random restarts with GS if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 8: Random GS Restarts ===")
        n_restarts = min(4, int(remaining / 40))
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_r = diverse_init(len(best_seq), rng)
            N_r = 1
            while N_r < 2 * len(best_seq) - 1:
                N_r <<= 1
            a_r = gs_flat_spectrum(len(best_seq), N_r, rng, n_iter=100, target_sum=1.0)
            alloc = min(remaining * 0.25, 40.0)
            a_r_opt = lse_gradient_descent(a_r, time_budget=alloc, seed=seed + 2000 + r)
            obj_r = compute_obj(a_r_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r_opt.copy()
                print(f"  ** Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
