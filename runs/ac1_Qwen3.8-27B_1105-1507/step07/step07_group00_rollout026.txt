def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_peak(a, k):
    """Subgradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def subgradient_topK(a, K=5):
    """Weighted subgradient from top-K peaks."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n), top_idx[0], conv
    weights = np.exp((peak_vals - max_val) * 5.0 / max_val)
    weights /= weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        g += w * subgradient_at_peak(a, int(k))
    return g, int(top_idx[0]), conv


def nesterov_subgradient(a, time_budget=100.0, K=5, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov parameters
    t_k = 1.0
    y = a.copy()  # extrapolated point
    a_prev = a.copy()

    # Initial learning rate
    lr0 = 1.0 / np.sqrt(n)
    stall = 0
    patience = 50

    it = 0
    while time.monotonic() - start < time_budget:
        # Compute subgradient at y
        g_y, k_star, conv_y = subgradient_topK(y, K=K)

        # Nesterov step size: lr = lr0 / t_k
        lr = lr0 / t_k

        # Gradient step from y
        a_new = y - lr * g_y
        a_new = project_simplex(a_new)

        # Update Nesterov parameter
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Reset Nesterov
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.5)

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new

        it += 1
        if verbose and it % 500 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, t_k={t_k:.1f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak_scipy(a, A):

    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub = np.zeros((2 * n - 1, n + 1))

    A_ub[:, n] = -1.0

def mircover_method(a, time_budget=100.0, K=5, num_iters=200, verbose=False):
    """
    Mircover-style averaging of random subgradient steps.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    # Accumulator for averaging
    acc = np.zeros(n)
    count = 0
    lr = 2.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget and count < num_iters:
        # Random peak selection (not just top-K, but random among ALL peaks with probability ~ value)
        conv = compute_conv(a)
        # Softmax sampling
        temp = 0.1 * conv.max()
        probs = np.exp((conv - conv.max()) / temp)
        probs /= probs.sum()
        k = rng.choice(len(conv), p=probs)

        g = subgradient_at_peak(a, int(k))
        a_new = a - lr / (count + 1) * g
        a_new = project_simplex(a_new)

        acc += a_new
        count += 1

        # Periodically check averaged solution
        if count % 10 == 0:
            a_avg = acc / count
            s_avg = float(np.sum(a_avg))
            if s_avg > 1e-10:
                a_avg = a_avg / s_avg
                obj_avg = compute_obj(a_avg)
                if obj_avg < best_obj - 1e-14:
                    best_obj = obj_avg
                    best_a = a_avg.copy()

        a = a_new

    return best_a

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(min_lr, lr * 0.999)
            momentum = min(0.8, momentum + 0.002)
        else:
            stall += 1
            if stall > 100:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.15, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0
                lr = max(min_lr, lr * 0.3)
                stall = 0
            else:
                lr = max(min_lr, lr * 0.97)

        it += 1

    return best_a


def lp_alternating(a, time_budget=200.0, verbose=False):
    """
    Aggressive LP alternating with adaptive line search and escape mechanisms.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.4
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = solve_lp_maxsum(a, M_cur)
            if g_lp is None:
                break

        # Adaptive line search: focus around best_t
        ts = [best_t, best_t * 0.5, best_t * 1.5, 0.2, 0.5, 0.8, 1.0]
        ts = [t for t in ts if 0.01 < t <= 1.0]
        ts = list(set(ts))

        best_t_obj = np.inf
        best_t_val = best_t
        best_a_local = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 15:
                # Structured escape: combine best with fresh LP direction + noise
                noise_scale = min(0.3, 0.05 + 0.04 * (stall // 15))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.4

    return best_a


def smooth_optimize(a, time_budget=60.0, T=30.0, verbose=False):
    """
    Smooth optimization using log-sum-exp approximation with L-BFGS-B.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-8:
                g += w[k] * grad_conv_at_peak(x, k)
        g *= 2.0 * n
        return obj_val, g

    # Run L-BFGS-B in chunks to respect time budget
    max_iter = 200
    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 2, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Smooth: obj={best_obj:.8f}")
    except Exception:
        pass

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """
    Coordinate descent: move small amounts of mass to reduce the dominant peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find positions contributing to the peak
        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        # Find positions NOT contributing to this peak
        non_contrib = []
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n:
                non_contrib.append(i)

        if not non_contrib:
            break

        improved = False
        for ci, (val, i, j) in enumerate(contrib[:15]):
            if val < 1e-12 or ci >= 15:
                break
            for target in non_contrib[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = frac * a[min(i, j)]
                    if delta < 1e-15:
                        continue
                    src = min(i, j) if i == j else (i if a[i] > a[j] else j)
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            break
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2 + 0.05
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def enforce_symmetry(a):
    """Make sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    sym = np.zeros(n)
    for i in range(half):
        sym[i] = (a[i] + a[n - 1 - i]) / 2.0
        sym[n - 1 - i] = sym[i]
    s = sym.sum()
    if s > 1e-10:
        sym /= s
    return sym


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining LP alternating,
    Nesterov subgradient, mirror descent, smooth optimization, and
    multi-scale n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Nesterov subgradient (fast, ~120s) ===
    print(f"\n=== Phase 1: Nesterov Subgradient (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 120.0)
    if p1_budget > 15:
        a_ns = nesterov_subgradient(best_seq, time_budget=p1_budget, K=5, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f} (no improvement)")

    # === Phase 2: Mirror descent (fast, ~100s) ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 100.0)
    if p2_budget > 15:
        a_md = mirror_descent(best_seq, time_budget=p2_budget, K=5, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror: obj={best_score:.8f} **")
        else:
            print(f"  Mirror: obj={obj_md:.8f} (no improvement)")

    # === Phase 3: LP Alternating (main workhorse, ~600s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p3_total = min(remaining * 0.45, 600.0)

    # Split into rounds with increasing aggressiveness
    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p3_total / (n_rounds - rnd), remaining * 0.6)
        if alloc < 20:
            break

        # For later rounds, add small perturbation to avoid stagnation
        if rnd > 0:
            a_pert = best_seq * np.exp(rng.normal(0, 0.03 * rnd, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Smooth optimization (L-BFGS on log-sum-exp, ~120s) ===
    print(f"\n=== Phase 4: Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 120.0)
    if p4_budget > 15:
        # Try multiple T values
        for T_val in [20.0, 50.0, 100.0]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_budget = min(p4_budget / 3.0, remaining * 0.4)
            if t_budget < 10:
                continue
            a_sm = smooth_optimize(best_seq, time_budget=t_budget, T=T_val, verbose=True)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth T={T_val}: obj={best_score:.8f} **")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        400, 500, 600, 700, 800, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]
    # Sort by closeness to current
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40 or n_tried >= 6:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick optimization: Nesterov + LP
        alloc_ns = min(remaining * 0.1, 40.0)
        if alloc_ns > 10:
            a_try = nesterov_subgradient(a_try, time_budget=alloc_ns, K=3)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.1, 50.0)
        if alloc_lp > 15:
            a_try = lp_alternating(a_try, time_budget=alloc_lp)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.15, 60.0)
        # Quick LP polish
        a_opt = lp_alternating(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 30:
        lp_alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Nesterov polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        ns_alloc = min(remaining * 0.2, 40.0)
        a_ns = nesterov_subgradient(best_seq, time_budget=ns_alloc, K=7)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_alloc = min(remaining * 0.2, 25.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP round if time allows
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
