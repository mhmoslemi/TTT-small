"""
Optimized subgradient + LP alternating method for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: fast FFT-based subgradient with adaptive momentum, local LP refinement,
structured perturbations for escaping local minima, and multi-n exploration.
"""

import numpy as np
import time


def compute_conv(a):
    """FFT-based full linear convolution a*a. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = np.sum(v)
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def peak_subgradient(a, k_star):
    """Compute subgradient of conv[k_star] w.r.t. a. O(n)."""
    n = len(a)
    grad = np.zeros(n)
    # grad[i] = a[k_star - i] if 0 <= k_star - i < n
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = a[k_star - idx]
    return grad


def topk_subgradient(a, conv, top_k):
    """Compute averaged subgradient over top-k peaks."""
    n = len(a)
    if top_k == 1:
        k_star = int(np.argmax(conv))
        return peak_subgradient(a, k_star)
    top_idx = np.argpartition(conv, -top_k)[-top_k:]
    weights = np.zeros(len(conv))
    for rank, k in enumerate(sorted(top_idx, key=lambda x: -conv[x])):
        weights[k] = 1.0 / (rank + 1)
    weights /= weights.sum()
    # grad[i] = sum_k weights[k] * a[k-i] for valid k-i
    # Use correlate
    grad = np.correlate(weights, a, mode='valid')
    return grad


def smooth_max_subgradient(a, conv, T):
    """Compute smooth max subgradient using log-sum-exp weights."""
    n = len(a)
    max_c = conv.max()
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    weights = exp_vals / total
    grad = np.correlate(weights, a, mode='valid')
    return grad


def fast_subgradient(a, time_budget=60.0, top_k=1, T_smooth=0.0, lr_init=0.005, momentum_init=0.85):
    """
    Fast projected subgradient descent with momentum and adaptive step size.
    Primary local optimizer.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    stall = 0
    best_stall_lr = lr

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)

        if T_smooth > 0:
            grad = smooth_max_subgradient(a, conv, T_smooth)
        else:
            grad = topk_subgradient(a, conv, top_k)

        # Center gradient to preserve sum constraint
        grad -= np.mean(grad)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Very slowly increase learning rate
            lr = min(lr * 1.001, 0.05)
        else:
            stall += 1
            if stall % 5 == 0:
                lr *= 0.95
            if stall > 40:
                momentum = max(0.5, momentum - 0.05)
                if stall > 80:
                    v = np.zeros(n)
                    a = best_a.copy()
                    lr = max(lr * 0.5, 1e-6)
                    stall = 0
                if lr < 1e-7:
                    break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    try:
        import cvxpy as cp
    except ImportError:
        pass

    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in ['CLARABEL', 'MOSEK', 'SCS', 'ECOS', 'ECOS_BB']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue

        # Fallback to scipy linprog
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization with backtracking line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_t = 0.3

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Backtracking line search
        t = 1.0
        a_try = None
        for _ in range(15):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                t *= 0.5
                a_try = None
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                break
            t *= 0.5
            a_try = None

        if a_try is not None and obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            last_t = t
            stall_count = 0
        else:
            # Try a range of t values
            improved = False
            for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0,
                          max(0.01, last_t * 0.5), min(1.0, last_t * 1.5)]:
                a_try = (1.0 - t_try) * a + t_try * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    last_t = t_try
                    stall_count = 0
                    improved = True
                    break
            if not improved:
                stall_count += 1
                if stall_count > 20:
                    # Perturb to escape
                    rng = np.random.default_rng(hash((it, n, stall_count)) % (2**32))
                    scale = 0.02 + 0.01 * min(stall_count // 20, 5)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    stall_count = 0
                    last_t = 0.3

    return best_a


def kick_and_optimize(a, time_budget=30.0, rng=None):
    """
    Add structured noise (kick) to escape local minima, then re-optimize.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Try several kicks
    kick_budget = min(5.0, time_budget * 0.3)
    num_kicks = 5
    kick_time_per = kick_budget / num_kicks

    for k in range(num_kicks):
        if time.monotonic() - start > time_budget - 10:
            break

        # Different kick types
        kick_type = k % 3
        a_kicked = best_a.copy()

        if kick_type == 0:
            # Gaussian noise
            noise = rng.normal(0, 0.05, n)
            a_kicked = a_kicked * np.exp(noise)
        elif kick_type == 1:
            # Random coordinate swap
            i, j = rng.integers(0, n, 2)
            delta = 0.1 * a_kicked[i]
            a_kicked[i] -= delta
            a_kicked[j] += delta
        else:
            # Structured: anticorrelated perturbation at center
            center = n // 2
            width = max(1, n // 10)
            x = np.arange(n)
            pert = 0.05 * np.exp(-((x - center) ** 2) / (2 * width ** 2))
            a_kicked = a_kicked + pert * rng.choice([-1, 1], n)

        a_kicked = np.maximum(a_kicked, 0)
        s = float(np.sum(a_kicked))
        if s < 1e-10:
            continue
        a_kicked = a_kicked / s

        # Quick optimization
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        alloc = min(remaining * 0.3, 8.0)
        a_opt = fast_subgradient(a_kicked, time_budget=alloc, top_k=3, T_smooth=0.001)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        alloc = min(60.0, remaining * 0.15)
        a6 = kick_and_optimize(best_seq, time_budget=alloc, rng=rng)
        obj6 = compute_obj(a6)
