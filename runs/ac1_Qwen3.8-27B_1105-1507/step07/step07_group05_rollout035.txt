"""
Adaptive temperature smoothed gradient descent with LP polishing for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT for O(n log n) gradient computation,
adaptive softmax temperature for peak focusing, and periodic LP for precision.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_conv(a, b=None):
    """Linear convolution a*b via FFT."""
    if b is None:
        b = a
    n_a = len(a)
    n_b = len(b)
    N = 1
    while N < n_a + n_b - 1:
        N <<= 1
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:n_a + n_b - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    b = fft_conv(a_n)
    return 2.0 * n * float(np.max(b))


def compute_grad_peak(a, alpha):
    """
    Compute gradient of smoothed max objective.
    Returns (smoothed_value, gradient_vector).
    """
    n = len(a)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    
    b = fft_conv(a)  # length 2n-1
    
    # Numerically stable softmax weights
    b_max = float(np.max(b))
    shifted = alpha * (b - b_max)
    exp_s = np.exp(shifted)
    w_sum = float(np.sum(exp_s))
    w = exp_s / w_sum
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * (w * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    fw = rfft(w, N)
    far = rfft(a_rev, N)
    conv_full = irfft(fw * far, N)[:3 * n - 2]
    grad = 2.0 * conv_full[n - 1:2 * n - 1]
    
    lse = b_max + float(np.log(w_sum)) / alpha
    return lse, grad


def compute_grad_energy(a):

def solve_lp_subproblem(a, max_time=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix: A[i,k] = a[k-i] if 0<=k-i<n, else 0
    # conv(a,g)[i] = sum_k A[i,k] * g[k] <= M
    # A is (2n-1) x n
    
    # Use scipy linprog
    try:
        from scipy.optimize import linprog
        # Variables: g[0..n-1], M
        # Minimize M
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        # Inequality: A@x - M <= 0 for each row, i.e., A@x + (-1)*M <= 0
        # A_ub is (2n-1) x (n+1)
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        
        # Fill A_ub[:, :n] with the convolution matrix
        for k in range(n):
            # conv(a,g)[i] includes term a[i-k]*g[k] for i-k in [0,n-1]
            # So A[i,k] = a[i-k] for max(0, k-n+1) <= i <= min(k+n-1, 2n-2)
            # i ranges from k to k+n-1 (capped at 2n-2)
            i_start = k
            i_end = min(k + n, 2 * n - 1)
            # A[i, k] = a[i - k]
            for i in range(i_start, i_end):
                A_ub[i, k] = a[i - k]
        
        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time})
        
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None

def chirp_init(n):

    return a

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def mirror_descent_step(a, grad, eta):
    """Multiplicative update: a * exp(-eta * grad)."""
    a_new = a * np.exp(-eta * (grad - np.mean(grad)))
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s
