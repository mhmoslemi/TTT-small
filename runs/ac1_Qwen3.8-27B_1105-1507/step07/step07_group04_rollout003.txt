import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute aperiodic convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def get_subgradient(a, top_k=5):
    """
    Compute subgradient of max_k (a*a)[k] w.r.t. a.
    Returns the average gradient over the top-k peaks.
    """
    n = len(a)
    conv = compute_conv(a)
    k_indices = np.argsort(conv)[::-1][:top_k]
    grad = np.zeros(n)
    for k in k_indices:
        k = int(k)
        # d/d(a_i) (a*a)[k] = 2 * a_{k-i} if 0 <= k-i < n
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(k_indices)
    return grad, conv

def get_subgradient_fast(a, top_k=5):
    """
    Compute subgradient efficiently using FFT.
    The gradient of (a*a)[k] w.r.t. a is 2 * reverse(a)[k-:][:n] (with appropriate truncation).
    Actually, let me think about this differently.
    
    (a*a)[k] = sum_j a_j * a_{k-j}
    d/d(a_i) (a*a)[k] = 2 * a[k-i] (if 0 <= k-i < n)
    
    So the gradient vector g has g[i] = 2 * a[k-i] for the active k.
    This is essentially a reversed and shifted copy of a.
    """
    n = len(a)
    conv = compute_conv(a)
    k_indices = np.argsort(conv)[::-1][:top_k]
    grad = np.zeros(n)
    for k in k_indices:
        k = int(k)
        # g[i] += 2 * a[k-i] for i where 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a[k-i] for i in [i_start, i_end]
        # When i = i_start, k-i = k - i_start = k - max(0, k-n+1)
        # When i = i_end, k-i = k - min(n-1, k)
        j_start = k - i_end
        j_end = k - i_start
        grad[i_start:i_end+1] += 2.0 * a[j_start:j_end+1][::-1]
    grad /= len(k_indices)
    return grad, conv

def project_simplex(v):
    """Project v onto the simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last(u * np.arange(1, n + 1) > (cssv - 1))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_direction(a, timeout=5.0):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns g (normalized) or None.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build the matrix A such that A @ g = conv(a, g)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a (2n-1) x n matrix
    
    # Use scipy linprog
    from scipy.optimize import linprog
    
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Row k, col j: A[k, j] = a[k-j] if 0 <= k-j < n
        k_start = j  # k-j = 0 => k = j
        k_end = j + n - 1  # k-j = n-1 => k = j + n - 1
        A_mat[k_start:k_end + 1, j] = a[:n]
    
    # LP: min M
    # s.t. A @ g - M <= 0 (for all k)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', 
                      options={'time_limit': timeout})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    
    return None

Wait, building the A matrix for the LP is O(n^2) which is fine for n=800 (640K elements). But the LP itself has 2n-1 = 1599 constraints and n+1 = 801 variables. This should be solvable in a few seconds with HiGHS.

Let me also think about the Frank-Wolfe step more carefully.

**Frank-Wolfe step:**
Given `a` (in the simplex) and a subgradient `g`:
1. Find `i* = argmin_i g[i]`
2. `v = e_{i*}` (the vertex)
3. Line search: find `t` that minimizes `f((1-t)*a + t*v)` for `t in [0, 1]`
4. Update: `a = (1-t)*a + t*v`

The line search can be done by:
- Trying a few values of t: [0.1, 0.2, ..., 0.9, 1.0]
- Or using the "FW step size" formula: `t = 2/(k+2)` where k is the iteration number
- Or using exact line search (parabolic interpolation)

For the LP direction:
1. Solve LP to get `g_lp`
2. Line search: find `t` that minimizes `f((1-t)*a + t*g_lp)` for `t in [0, 1]`
3. Update: `a = (1-t)*a + t*g_lp`

Since both `a` and `g_lp` are in the simplex, the convex combination is also in the simplex. No projection needed!

This is a key advantage of this approach.

"""

def nesterov_subgradient(a, time_budget=100.0, top_k=5, verbose=False):

def coordinate_polish(a, time_budget=30.0):
    """
    Coordinate descent: try moving mass between pairs to reduce peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        k_star = int(np.argmax(conv))
        
        # Try moving mass from high-peak contributors to others
        # The peak at k_star is contributed by pairs (j, k_star-j)
        # We want to reduce this by moving mass away from these pairs
        
        # Find the largest contributor to the peak
        contribs = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        for i in range(i_start, i_end + 1):
            j = k_star - i
            contribs[i] = a[i] * a[j]
        
        # Move mass from the largest contributor to a random low position
        i_max = int(np.argmax(contribs))
        
        # Try a few fractions
        for frac in [0.05, 0.1, 0.2]:
            delta = frac * a[i_max]
            if delta <= 0 or delta >= a[i_max]:
                continue
            
            # Find a good target: position that doesn't contribute to the peak
            rng = np.random.default_rng(int(time.monotonic() * 100000 + i_max) % (2**31))
            targets = rng.integers(0, n, size=min(20, n))
            
            for j_target in targets:
                j_target = int(j_target)
                if j_target == i_max:
                    continue
                # Check if this target would create a new peak
                a_try = a.copy()
                a_try[i_max] -= delta
                a_try[j_target] += delta
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    conv = compute_conv(a)
                    improved = True
                    break
            if improved:
                break
        
        if not improved and it > 100:
            break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
