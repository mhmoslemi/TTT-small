Combines Gerchberg-Saxton spectral flattening, LP alternating, 
L-BFGS smooth-max optimization, chase-peak Nesterov, and multi-n exploration.
Key innovations: spectral domain initialization, L-BFGS with analytic gradients,

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_value_and_weights(conv, T):
    """Compute smooth max value and softmax weights."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    return smooth_val, weights


def compute_smooth_gradient(weights, a):
    """
    Compute gradient of smooth max w.r.t. a.
    grad[i] = 2 * sum_j weights[i+j] * a[j] for valid indices.
    """
    n = len(a)
    grad = np.correlate(weights, a, mode='valid') * 2.0
    return grad

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""

    # Try cvxpy first

        M = cp.Variable()

        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in ['CLARABEL', 'SCS', 'ECOS', 'MOSEK', 'HiGHS']:
            try:
                prob.solve(solver=solver)

                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))

            except Exception:
                continue
    except ImportError:
        pass
    
    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)

        a_local = a_local / sl
        conv = compute_conv(a_local)
        _, weights = smooth_max_value_and_weights(conv, T)
        grad = compute_smooth_gradient(weights, a_local)
        # Center gradient for sum constraint
        grad = grad - np.mean(grad)
        val = 2.0 * n * float(np.max(conv)) + 0.001 * T * float(np.sum(weights))
        # Actually use smooth value
        smooth_val, _ = smooth_max_value_and_weights(conv, T)
        val = 2.0 * n * smooth_val
        return val, grad
    
    # Use multiple L-BFGS runs with decreasing T
    temps = [T_init, T_init * 0.5, T_init * 0.25, T_init * 0.1, T_min]
    
    for T_curr in temps:
        if time.monotonic() - start > time_budget:
            break
        T = T_curr
        
        def obj(params):
            a_local = np.maximum(params, 0)
            sl = np.sum(a_local)
            if sl < 1e-10:
                return 1e10
            a_local = a_local / sl
            conv = compute_conv(a_local)
            sv, _ = smooth_max_value_and_weights(conv, T_curr)
            return 2.0 * n * sv
        
        def grad(params):
            a_local = np.maximum(params, 0)
            sl = np.sum(a_local)
            if sl < 1e-10:
                return np.zeros(n)
            a_local = a_local / sl
            conv = compute_conv(a_local)
            _, weights = smooth_max_value_and_weights(conv, T_curr)
            g = compute_smooth_gradient(weights, a_local)
            g = g - np.mean(g)
            return g
        
        try:
            from scipy.optimize import minimize as sp_minimize
            res = sp_minimize(obj, a, jac=grad, method='L-BFGS-B',
                             options={'maxiter': 200, 'gtol': 1e-8, 'maxfun': 500})
            if res.x is not None:
                a_try = simplex_project(res.x)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
        except Exception:
            pass
        
        # Manual gradient descent as backup
        if time.monotonic() - start > time_budget:
            break
        
        momentum = 0.8
        v = np.zeros(n)
        lr = 0.003
        for it in range(200):
            if time.monotonic() - start > time_budget:
                break
            conv = compute_conv(best_a)
            _, weights = smooth_max_value_and_weights(conv, T_curr)
            g = compute_smooth_gradient(weights, best_a)
            g = g - np.mean(g)
            v = momentum * v - lr * g
            a_new = simplex_project(best_a + v)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new
                if it % 50 == 0 and lr < 0.01:
                    lr = min(lr * 1.1, 0.01)
    
    return best_a


def chase_peak_nesterov(a, time_budget=30.0):
    """Chase-the-peak with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.01
    v = np.zeros(n)
    momentum = 0.85
    stall = 0
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of conv[k_star] w.r.t. a[i]
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        grad_centered = grad - np.mean(grad)
        
        v = momentum * v - step_size * grad_centered
        a_new = simplex_project(a + 0.5 * v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0 and step_size < 0.03:
                step_size = min(step_size * 1.02, 0.03)
        else:
            stall += 1
            if stall > 30:
                step_size *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if step_size < 1e-4:
                    break
    
    return best_a


def multi_peak_shave(a, time_budget=30.0):
    """
    Multi-peak shaving: identify top-K peaks and simultaneously reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = np.max(conv)
        
        # Find top K peaks (within 10% of max)
        threshold = max_val * 0.9
        peak_indices = np.where(conv >= threshold)[0]
        
        # Build subgradient from all peaks
        grad = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        
        grad /= len(peak_indices)
        grad_centered = grad - np.mean(grad)
        
        # Backtracking line search
        step = 0.01
        for _ in range(10):
            a_try = simplex_project(a - step * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                step = min(step * 1.2, 0.05)
                break
            step *= 0.5
        
        if step < 1e-5:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif mode == 3:
        # Bimodal
        center = n // 2
        width = max(1, n // 5)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a += 0.5 * np.exp(-((x - center * 0.6) ** 2) / (2 * (width // 2 + 1) ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        a = a / np.sum(a)
    elif mode == 5:
        # Triangle
        x = np.arange(n) / max(1, (n - 1))
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    elif mode == 6:
        # Smooth random
        a = np.convolve(rng.uniform(0, 1, n), np.ones(5)/5, mode='same')
        a = a / np.sum(a)
    elif mode == 7:
        # Two peaks
        c1, c2 = n // 4, 3 * n // 4
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * (n // 10) ** 2)) + np.exp(-((x - c2) ** 2) / (2 * (n // 10) ** 2))
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def support_compact(a, fraction=0.95):
    """Zero out the smallest elements to compact support."""
    n = len(a)
    k = int(n * fraction)
    if k >= n:
        return a
    idx = np.argsort(a)[:-k]
    a_new = a.copy()
    a_new[idx] = 0
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new
