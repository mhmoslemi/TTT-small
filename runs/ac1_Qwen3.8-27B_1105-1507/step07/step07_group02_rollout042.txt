    C = compute_conv(a)
    # Smooth max approximation
    max_C = C.max()
    log_sum_exp = max_C + np.log(np.sum(np.exp(C - max_C)))
    f = log_sum_exp / T
    
    # Gradient
    p = np.exp(C - max_C) / np.sum(np.exp(C - max_C))  # softmax
    grad = np.zeros_like(a)
    for k in range(len(C)):

            i_end = min(len(a) - 1, k)
            for i in range(i_start, i_end + 1):

    return f, grad

Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses smooth-max exponentiated gradient descent with annealing, p-norm gradient,
active-set LP refinement, multi-n exploration, and random restarts.
Key innovation: efficient FFT-based gradient computation for the smooth max of
the auto-convolution, combined with multiplicative (exponentiated) updates that
naturally preserve non-negativity.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    C = compute_conv(a_norm)
    return 2.0 * n * float(np.max(C))


def softmax_gradient(C, a, T):
    """
    Compute gradient of smooth-max approximation of max(C) w.r.t. a.
    Uses softmax weights p[k] = exp(C[k]/T) / Z.
    grad[i] = sum_k p[k] * a[k-i] (cross-correlation).
    Returns (smooth_value, p, grad).
    """
    n = len(a)
    max_C = np.max(C)
    scaled = (C - max_C) / T
    exp_C = np.exp(scaled)
    Z = np.sum(exp_C)
    p = exp_C / Z
    smooth_val = max_C + T * np.log(Z)

    # grad[i] = sum_j p[i+j] * a[j], computed via FFT
    size = 1
    while size < 3 * n:
        size <<= 1
    P = rfft(p, size)
    Af = rfft(a[::-1], size)
    conv_pa = irfft(P * Af, size)
    grad = conv_pa[n - 1:2 * n - 1]

    return smooth_val, p, grad


def pnorm_gradient(C, a, p):
    """
    Compute gradient of (sum C[k]^p) w.r.t. a.
    Returns (pnorm_value, grad).
    """
    n = len(a)
    Cp = C ** p
    weight = p * C ** (p - 1)  # d(C^p)/dC = p*C^(p-1)
    total = np.sum(Cp)

    # grad[i] = sum_k weight[k] * a[k-i]
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(weight, size)
    Af = rfft(a[::-1], size)
    conv_wa = irfft(W * Af, size)
    grad = conv_wa[n - 1:2 * n - 1]

    return total, grad

    """
    Smooth-max exponentiated gradient descent with geometric annealing.
    a[i] <- a[i] * exp(-eta * grad_centered[i] / T), then normalize.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_steps = min(400, max(50, int(time_budget * 3)))
    eta = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        C = compute_conv(a)
        max_C = C.max()
        if max_C < 1e-15:
            break

        # Anneal temperature: from smooth (T ~ max_C) to sharp (T ~ 0.01*max_C)
        progress = step / n_steps
        T_rel = 1.0 * (0.02) ** progress
        T = T_rel * max_C

        if T < 1e-10:
            T = 1e-10

        _, p, grad = softmax_gradient(C, a, T)
        grad_centered = grad - grad.mean()

        # Multiplicative update
        scale = eta * grad_centered / T
        scale = np.clip(scale, -50, 50)  # prevent overflow

        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-10:

        a_new = a_new / s_new

            c_obj[-1] = 1.0  # minimize M

            A_ub = np.zeros((K, m_vars))
            b_ub = np.zeros(K)
            for idx, k in enumerate(top_k):
                for j in range(n):
                    i = k - j
                    if 0 <= i < n:
                        A_ub[idx, j] = a[i]
                A_ub[idx, -1] = -1.0  # A_k g - M <= 0

            A_eq = np.zeros((1, m_vars))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])

            bounds = [(0, None)] * n + [(None, None)]

            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'time_limit': 10})
            if res.status == 0 and res.x is not None:
                g_val = np.maximum(res.x[:n], 0)
                gs = g_val.sum()
                if gs > 1e-10:
                    g_val = g_val / gs
                    # Line search for best blend
                    best_t = 0
                    best_t_obj = best_obj
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * a + t * g_val
                        a_try = np.maximum(a_try, 1e-30)
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_t = t
                    if best_t > 0 and best_t_obj < best_obj - 1e-12:
                        best_obj = best_t_obj
                        a_new = (1 - best_t) * a + best_t * g_val
                        a_new = np.maximum(a_new, 1e-30)
                        a_new = a_new / a_new.sum()
                        best_a = a_new.copy()
                        a = a_new
                        if verbose and it % 10 == 0:
                            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
                        continue
        except Exception:
            pass

        # If no improvement, try random perturbation
        rng = np.random.default_rng(it * 31 + n)
        a = best_a * np.exp(rng.normal(0, 0.05, n))
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        if compute_obj(a) < best_obj:
            best_obj = compute_obj(a)
            best_a = a.copy()

    return best_a


def full_lp_refine(a, time_budget=60.0, verbose=False):
    """Full LP refinement (all constraints) for smaller n."""
    n = len(a)
    if n > 500:
        return lp_active_set_refine(a, time_budget=time_budget, K=30, verbose=verbose)
    return lp_active_set_refine(a, time_budget=time_budget, K=min(n, 50), verbose=verbose)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([a.sum()])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multi_start_optimize(a, time_budget=30.0, verbose=False):
    """Run multiple optimization methods and return the best result."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Strategy 1: Exponentiated gradient
    remaining = time_budget - (time.monotonic() - start) - 10
    if remaining > 10:
        a1 = exponentiated_gradient_descent(best_a, time_budget=remaining, verbose=verbose)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1
            if verbose:
                print(f"      EG: {best_obj:.8f}")

    # Strategy 2: p-norm descent
    remaining = time_budget - (time.monotonic() - start) - 10
    if remaining > 10:
        a2 = pnorm_descent(best_a, time_budget=min(remaining, 30), p=10.0)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2
            if verbose:
                print(f"      P-norm: {best_obj:.8f}")

    # Strategy 3: LP refine
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a3 = full_lp_refine(best_a, time_budget=remaining, verbose=verbose)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3
            if verbose:
                print(f"      LP: {best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses smooth-max exponentiated gradient descent,
    p-norm descent, and active-set LP refinement to minimize the auto-convolution
    peak of non-negative sequences.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Establish baseline
    print("=== Phase 0: Baseline ===")
    for n_base in [400, 600, 800, 1000, 1200, 1500, 2000]:
        if time.monotonic() > deadline:
            break
        seq = [1.0] * n_base
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = np.ones(n_base)
        print(f"  Constant n={n_base}: score={score:.8f}")

    # Try parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr.copy()
                parent_list = [float(x) for x in parent]
                score = evaluate_sequence(parent_list)
                if score < np.inf:
                    print(f"  Parent (n={parent.size}): score={score:.8f}")
                    if score < best_score:
                        best_score = score
                        best_seq = parent.copy()
                else:
                    print(f"  Parent: INVALID")
    except Exception as e:
        print(f"  Parent not available: {e}")

    if best_seq is None:
        best_seq = np.ones(800)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"  Default constant n=800: score={best_score:.8f}")

    print(f"  Starting best: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: Primary optimization on parent/best
    if parent is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n=== Phase 1: Optimize parent (n={parent.size}) ===")
            a_p = parent / np.sum(parent)

            # Exponentiated gradient (main new method)
            alloc = min(180.0, remaining * 0.3)
            a_eg = exponentiated_gradient_descent(a_p, time_budget=alloc, verbose=True)
            obj_eg = compute_obj(a_eg)
            ev_eg = evaluate_sequence([float(x) for x in a_eg])
            print(f"  EG (parent): internal={obj_eg:.8f}, evaluated={ev_eg:.8f}")
            if ev_eg < best_score:
                best_score = ev_eg
                best_seq = a_eg.copy()

            # p-norm on EG result
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_pn = pnorm_descent(a_eg, time_budget=min(60.0, remaining * 0.2), p=10.0)
                ev_pn = evaluate_sequence([float(x) for x in a_pn])
                print(f"  P-norm (on EG): evaluated={ev_pn:.8f}")
                if ev_pn < best_score:
                    best_score = ev_pn
                    best_seq = a_pn.copy()

            # LP refine
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_lp = full_lp_refine(best_seq / np.sum(best_seq), time_budget=min(60.0, remaining * 0.25), verbose=True)
                ev_lp = evaluate_sequence([float(x) for x in a_lp])
                print(f"  LP (refine): evaluated={ev_lp:.8f}")
                if ev_lp < best_score:
                    best_score = ev_lp
                    best_seq = a_lp.copy()

    # Phase 2: Intensive refinement on current best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: Intensive refinement (n={len(best_seq)}) ===")
        a_best = best_seq.copy()
        s = a_best.sum()
        if s > 0:
            a_best = a_best / s

        # Round 1: EG with longer budget
        alloc = min(120.0, remaining * 0.3)
        a_r1 = exponentiated_gradient_descent(a_best, time_budget=alloc, verbose=True)
        ev_r1 = evaluate_sequence([float(x) for x in a_r1])
        print(f"  R1 EG: {ev_r1:.8f}")
        if ev_r1 < best_score:
            best_score = ev_r1
            best_seq = a_r1.copy()

        # Round 2: P-norm
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_r2 = pnorm_descent(best_seq / np.sum(best_seq), time_budget=min(60.0, remaining * 0.2), p=16.0)
            ev_r2 = evaluate_sequence([float(x) for x in a_r2])
            print(f"  R2 P-norm(p=16): {ev_r2:.8f}")
            if ev_r2 < best_score:
                best_score = ev_r2
                best_seq = a_r2.copy()

        # Round 3: LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_r3 = full_lp_refine(best_seq / np.sum(best_seq), time_budget=min(60.0, remaining * 0.2), verbose=True)
            ev_r3 = evaluate_sequence([float(x) for x in a_r3])
            print(f"  R3 LP: {ev_r3:.8f}")
            if ev_r3 < best_score:
                best_score = ev_r3
                best_seq = a_r3.copy()

        # Round 4: Another EG pass
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_r4 = exponentiated_gradient_descent(best_seq / np.sum(best_seq), time_budget=min(90.0, remaining * 0.3), verbose=False)
            ev_r4 = evaluate_sequence([float(x) for x in a_r4])
            print(f"  R4 EG: {ev_r4:.8f}")
            if ev_r4 < best_score:
                best_score = ev_r4
                best_seq = a_r4.copy()

    # Phase 3: Multi-n exploration
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Multi-n exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 200), max(200, n_best - 100),
            n_best + 100, n_best + 200, n_best + 300,
            500, 600, 700, 900, 1000, 1200, 1500, 2000, 2500
        ]))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)

            s_try = a_try.sum()
            if s_try > 0:
                a_try = a_try / s_try

            alloc = min(40.0, remaining * 0.12)
            a_opt = multi_start_optimize(a_try, time_budget=alloc)
            ev_try = evaluate_sequence([float(x) for x in a_opt])
            if ev_try < best_score:
                best_score = ev_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: score={best_score:.8f} **")
            else:
                print(f"  n={n_try}: score={ev_try:.8f}")

    # Phase 4: Random restarts
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: Random restarts ===")
        n_best = len(best_seq)
        restart_budget = min(30.0, remaining * 0.06)

        for r in range(10):
            remaining = deadline - time.monotonic()
            if remaining < restart_budget + 10:
                break

            init_type = r % 5
            if init_type == 0:
                # Perturbed best
                a_r = best_seq * np.exp(rng.normal(0, 0.15, n_best))
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()
            elif init_type == 1:
                # Near-constant with structure
                n_r = n_best
                x = np.arange(n_r)
                a_r = 1.0 + 0.3 * np.sin(2 * np.pi * x / n_r * rng.integers(2, 5))
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()
            elif init_type == 2:
                # Different n
                n_r = max(200, min(3000, n_best + int(rng.integers(-300, 500))))
                a_r = resample_sequence(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.08, len(a_r)))
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()
            elif init_type == 3:
                # Asymmetric ramp + noise
                n_r = n_best
                x = np.arange(n_r)
                a_r = np.exp(np.linspace(0, 1, n_r) * rng.uniform(0.5, 2.0))
                a_r += rng.uniform(0, 0.1, n_r)
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()
            else:
                # Bimodal
                n_r = n_best
                x = np.arange(n_r)
                c1 = n_r * rng.uniform(0.2, 0.4)
                c2 = n_r * rng.uniform(0.6, 0.8)
                w = n_r * rng.uniform(0.05, 0.2)
                a_r = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
                a_r = a_r / a_r.sum()

            a_opt = multi_start_optimize(a_r, time_budget=restart_budget)
            ev_r = evaluate_sequence([float(x) for x in a_opt])
            if ev_r < best_score:
                best_score = ev_r
                best_seq = a_opt.copy()
                print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): score={best_score:.8f} **")
            else:
                print(f"  Restart {r} (type={init_type}): score={ev_r:.8f}")

    # Phase 5: Final polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 5: Final polish (remaining={remaining:.0f}s) ===")
        a_final = best_seq.copy()
        s = a_final.sum()
        if s > 0:
            a_final = a_final / s

        for round_idx in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.25, 50.0)
            method = round_idx % 3
            if method == 0:
                a_pol = exponentiated_gradient_descent(a_final, time_budget=alloc, verbose=False)
            elif method == 1:
                a_pol = pnorm_descent(a_final, time_budget=alloc, p=12.0)
            else:
                a_pol = full_lp_refine(a_final, time_budget=alloc, verbose=False)

            ev_pol = evaluate_sequence([float(x) for x in a_pol])
            if ev_pol < best_score:
                best_score = ev_pol
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Polish {round_idx} (method={method}): score={best_score:.8f} **")
            else:
                print(f"  Polish {round_idx} (method={method}): score={ev_pol:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
