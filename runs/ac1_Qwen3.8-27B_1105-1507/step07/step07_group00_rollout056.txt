"""
Improved hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses enhanced LP alternating with dual-guided perturbation, adaptive subgradient,
peak flattening, and multi-scale exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def conv_matrix_apply(a, g):
    """Compute convolution of a and g efficiently."""
    return np.convolve(a, g)


def build_conv_matrix_rows(a, k_list):
    """Build rows of the conv matrix for specific k values.
    Row k: (a * g)[k] = sum_i a[i] * g[k-i] for valid i.
    So row k has non-zeros at positions j where a[k-j] is non-zero.
    Actually: (a*g)[k] = sum_j g[j] * a[k-j] for valid j.
    So the row for constraint k is: [a[k-0], a[k-1], ..., a[k-(n-1)]] padded.
    """
    n = len(a)
    m = len(k_list)
    rows = np.zeros((m, n))
    for idx, k in enumerate(k_list):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows[idx, j] = a[i]
    return rows


def solve_lp_for_g(a, M, time_limit=5.0):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    Uses scipy linprog for speed.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A such that A @ g <= M for all k
    # (a*g)[k] = sum_{j=0}^{n-1} a_norm[k-j] * g[j] (where k-j is valid)
    # So A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, only include "important" constraints (top peaks)
    conv_a = compute_conv(a_norm)
    # Include all constraints for correctness, but we can be smart about it
    
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    for j in range(n):
        # g[j] contributes to (a*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient a_norm[k-j] = a_norm[0..n-1]
        A[j:j+n, j] = a_norm
    
    # LP: max sum(g) s.t. A @ g <= M_vec, g >= 0
    # where M_vec is all M (same bound for all k)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    A_ub = A
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def solve_lp_minpeak(a, time_limit=5.0):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    for j in range(n):
        A[j:j+n, j] = a_norm
    
    # LP: min M s.t. A @ g <= M, sum(g) = 1, g >= 0
    # Variables: g[0..n-1], M
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: A @ g - M <= 0 for each constraint
    A_ub = np.zeros((n_constraints, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_constraints)
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def topk_subgradient(a, K=5):
    """Compute weighted subgradient at top-K peak positions."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    
    # Weights: softmax over top-K values
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n), top_idx
    
    temperature = 20.0
    weights = np.exp(temperature * (peak_vals - max_val) / max_val)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                g[j] += w * a[i]
    
    return g, top_idx


def mirror_descent_optimize(a, time_budget=100.0, verbose=False):
    """Mirror descent with adaptive top-K and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n) * 1.5
    min_lr = 1e-5
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        g, top_idx = topk_subgradient(a, K=5)
        
        v = momentum * v + g
        log_a = np.log(a)
        log_new = log_a - lr * v
        log_new -= np.max(log_new)
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = max(min_lr, lr * 0.999)
            momentum = min(0.85, momentum + 0.002)
        else:
            stall += 1
            if stall > 200:
                # Structured perturbation: shift mass from peak-contributing to non-contributing
                conv = compute_conv(a)
                k_star = int(np.argmax(conv))
                rng = np.random.default_rng(it * 7919)
                
                # Find peak contributors
                contribs = np.zeros(n)
                for j in range(n):
                    i = k_star - j
                    if 0 <= i < n:
                        contribs[j] = a[i]
                
                # Shift mass from high contributors to low contributors
                top_contrib = np.argsort(contribs)[::-1][:n//10]
                bot_contrib = np.argsort(contribs)[:n//10]
                shift = 0.01 * a[top_contrib].sum()
                a[top_contrib] -= shift / len(top_contrib)
                a[bot_contrib] += shift / len(bot_contrib)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                lr = max(min_lr, lr * 0.5)
                stall = 0
            else:
                lr = max(min_lr, lr * 0.97)
        
        it += 1
    
    return best_a


def lp_alternating_enhanced(a, time_budget=300.0, verbose=False):
    """
    Enhanced LP alternating with:
    - Better line search (golden section)
    - Dual-guided perturbation
    - Adaptive M target
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iter_count = 0
    
    while time.monotonic() - start < time_budget:
        # Solve LP to find best direction
        g_lp, M_lp = solve_lp_minpeak(a)
        
        if g_lp is None:
            # Fallback: use top-K subgradient direction
            g_dir, _ = topk_subgradient(a, K=5)
            g_dir = np.maximum(g_dir - np.mean(g_dir), 0)
            if g_dir.sum() > 1e-10:
                g_lp = g_dir / g_dir.sum()
            else:
                break
        
        # Golden section line search
        conv_a = compute_conv(a)
        M_cur = float(np.max(conv_a))
        
        # We want to find t that minimizes obj((1-t)*a + t*g_lp)
        # Use parabolic interpolation for efficiency
        t1, t2 = 0.3, 0.7
        f1 = compute_obj(np.maximum((1-t1)*a + t1*g_lp, 0))
        f2 = compute_obj(np.maximum((1-t2)*a + t2*g_lp, 0))
        
        # Try a few more points
        for t_try in [0.1, 0.2, 0.4, 0.5, 0.6, 0.8, 0.9]:
            f_try = compute_obj(np.maximum((1-t_try)*a + t_try*g_lp, 0))
            if f_try < min(f1, f2):
                if f_try < f1:
                    t2, f1, f2 = t1, t2, f1
                    t1, f1 = t_try, f_try
                else:
                    t1, f2 = t_try, f_try
        
        best_t = t1 if f1 < f2 else t2
        a_try = np.maximum((1-best_t)*a + best_t*g_lp, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try
        
        obj_try = compute_obj(a_try)
        iter_count += 1
        
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            stall = 0
            if verbose and iter_count % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, M_lp={M_lp:.6f}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 20:
                # Structured perturbation using dual info
                rng = np.random.default_rng(iter_count * 104729)
                conv = compute_conv(a)
                top_peaks = np.argsort(conv)[::-1][:10]
                
                # Randomly perturb the top-contributing elements
                for k in top_peaks[:3]:
                    if time.monotonic() - start > time_budget:
                        break
                    k = int(k)
                    # Find which j values contribute to this peak
                    contrib_js = []
                    for j in range(n):
                        i = k - j
                        if 0 <= i < n and a[j] * a[i] > 1e-10:
                            contrib_js.append(j)
                    
                    if not contrib_js:
                        continue
                    
                    # Pick a random contributor and shift mass
                    j_pick = contrib_js[rng.integers(len(contrib_js))]
                    # Find a non-contributor to shift to
                    all_js = set(range(n))
                    for j2 in contrib_js:
                        all_js.discard(j2)
                    non_contrib = list(all_js)
                    if not non_contrib:
                        continue
                    target = non_contrib[rng.integers(len(non_contrib))]
                    
                    delta = 0.005 * a[j_pick]
                    a[j_pick] -= delta
                    a[target] += delta
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
    
    return best_a
