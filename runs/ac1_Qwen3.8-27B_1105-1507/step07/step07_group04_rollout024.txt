"""
Adaptive minimax optimizer for step function coefficient sequences.
Combines smooth log-sum-exp minimax, QP/LP alternating, and animated-tau
gradient methods to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: QP-based alternating for smoother updates, animated-tau
smooth minimax for global convergence, and adaptive step sizing.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def gradient_smooth_max(a, tau):
    """
    Compute (objective, gradient) of smoothed minimax objective:
    f(a) = 2*n*tau*log(sum_k exp(conv(a,a)[k]/tau))
    Returns (f_value, gradient_array).
    """
    n = len(a)
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    shifted = (conv - c_max) / tau
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[int(np.argmax(conv))] = 1.0
        w_sum = 1.0
    else:
        weights /= w_sum

    obj = 2.0 * n * (c_max + tau * np.log(w_sum))

    # Gradient via cross-correlation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(weights, size)
    cross = irfft(np.conj(fa) * fw, size)[:n]
    grad = 4.0 * n * cross
    return obj, grad

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            if time.monotonic() - start > time_budget:
                break
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s = float(np.sum(a_pert))
            if s < 1e-10:
                continue
            a_pert /= s

            # Quick optimization
            a_opt = smooth_minmax_optimize(a_pert, time_budget=30)
            obj = compute_obj(a_opt)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a_opt.copy()
                print(f"      MultiStart scale={scale}: obj={best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multiple optimization strategies
    to find the best coefficient sequence.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except NameError:
        pass
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    a = best_seq.copy()
    n_parent = len(a)

    # === Phase 1: Smooth minimax (animated tau) ===
    print(f"\n=== Phase 1: Smooth Minimax (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(300.0, remaining * 0.25)
        a_new = smooth_minmax_optimize(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  SmoothMM: obj={best_score:.8f} **")
        else:
            print(f"  SmoothMM: obj={obj:.8f}")

    # === Phase 2: QP Alternating ===
    print(f"\n=== Phase 2: QP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(300.0, remaining * 0.25)
        a_new = qp_alternating(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  QP: obj={best_score:.8f} **")
        else:
            print(f"  QP: obj={obj:.8f}")

    # === Phase 3: LP Alternating (improved) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(300.0, remaining * 0.25)
        a_new = lp_alternating_improved(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj:.8f}")

    # === Phase 4: Subgradient with Nesterov ===
    print(f"\n=== Phase 4: Subgradient Nesterov ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(120.0, remaining * 0.15)
        a_new = subgradient_nesterov(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_options = sorted(set([
            max(200, n_parent - 100),
            max(200, n_parent - 50),
            n_parent,
            n_parent + 50,
            n_parent + 100,
            n_parent + 200,
            n_parent + 400,
        ]))
        n_options = [n for n in n_options if 100 <= n <= 3000]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try /= s_try

            # Quick optimization: smooth then LP
            alloc1 = min(30.0, remaining * 0.2)
            alloc2 = min(30.0, remaining * 0.2)

            if alloc1 > 10:
                a_try = smooth_minmax_optimize(a_try, time_budget=alloc1)
            if alloc2 > 10:
                a_try = lp_alternating_improved(a_try, time_budget=alloc2)

            obj_try = compute_obj(a_try)
            if obj_try < best_score - 1e-10:
                best_score = obj_try
                best_seq = a_try.copy()
                a = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Multi-start with perturbations ===
    print(f"\n=== Phase 6: Multi-Start Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(200.0, remaining * 0.4)
        a_new, obj_new = multi_start_search(best_seq, alloc, len(best_seq))
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP polish
        alloc = min(60.0, remaining * 0.3)
        a_new = lp_alternating_improved(best_seq, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Coordinate descent
        alloc = min(30.0, remaining * 0.3)
        a_new = coordinate_descent_polish(best_seq, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more smooth minimax pass
        alloc = min(40.0, remaining * 0.4)
        a_new = smooth_minmax_optimize(best_seq, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
