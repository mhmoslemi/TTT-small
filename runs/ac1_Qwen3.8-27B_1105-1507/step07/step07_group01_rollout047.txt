"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP alternating, peak-directed coordinate descent,
smooth-max gradient with momentum, and diverse multi-n exploration.
Key innovations: adaptive active-set management, dual-guided stepping,
peak-targeted coordinate updates, and aggressive line search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a is non-negative."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def build_active_rows(a, active_k):
    """Build only the active rows of the convolution matrix."""
    n = len(a)
    A = np.zeros((len(active_k), n))
    for i, k in enumerate(active_k):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[i, j] = a[k - j]
    return A


def solve_lp_max_sum(a, active_k, M_val):
    """Solve: maximize sum(g) s.t. conv(a,g)[k] <= M_val for k in active_k, g >= 0.
    Returns normalized g (sum=1) or None."""
    n = len(a)
    A = build_active_rows(a, active_k)
    c = -np.ones(n)
    b = np.full(len(active_k), M_val)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x, 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None


def solve_lp_full(a, M_val):
    """Solve LP with all 2n-1 constraints."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M_val)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x, 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None


def lp_alternating_active(a, time_budget, max_active=80, verbose=False):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    max_b = float(np.max(conv))
    active_k = sorted(np.argsort(conv)[-max_active:].tolist())

    it = 0

            if g is None:
                # Relax M

                continue

        best_t_obj = best_obj
        best_t = t_current
        best_a_local = best_a.copy()

            t_current = best_t

def peak_directed_descent(a, time_budget, verbose=False):
    """Coordinate descent that specifically targets the peak convolution bins.
    Reduces contributions to peak bins and increases elsewhere."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.01
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        max_b = float(np.max(conv))

        # Find top K peak bins
        K = max(5, min(20, n // 20))
        peak_indices = np.argsort(conv)[-K:]

        # Compute gradient: for each peak bin k, d(conv[k])/d(a[i]) = a[k-i]
        grad = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                grad[j] += a[k - j]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Center for simplex constraint
        grad_centered = grad - np.sum(grad) / n

        # Try multiple step sizes
        improved = False
        for ss in [step_size * 0.5, step_size, step_size * 2.0]:
            a_try = a - ss * grad_centered
            a_try = simplex_project(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            step_size *= 0.85
            if step_size < 1e-6:
                break
            # Try opposite direction (increase where gradient is low)
            a_try = simplex_project(a + step_size * 0.5 * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try

    return best_a


def smooth_max_gradient(a, time_budget, T_init=0.001, T_min=0.00005, verbose=False):
    """Smooth max (log-sum-exp) gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.01
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        max_c = np.max(conv)

        # Softmax weights
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400, 400)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient: grad[i] = sum_k weights[k] * a[k-i] (for valid k-i)
        # This is np.correlate(weights, a, mode='valid')
        grad = np.correlate(weights, a, mode='valid')

        # Center
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break

        # Temperature anneal
        T = max(T_min, T_init * (0.995 ** it))
        # Slowly increase lr
        if it % 100 == 0 and lr < 0.05:
            lr = min(lr * 1.02, 0.05)

    return best_a


def spread_heuristic(a, time_budget, verbose=False):
    """Heuristic that tries to spread mass away from peak-contributing positions.
    Identifies which a[i] values contribute most to the peak and reduces them,
    then redistributes to positions that contribute least to any peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    transfer_frac = 0.05

    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        peak_k = int(np.argmax(conv))

        # Find contributors to the peak
        j_start = max(0, peak_k - n + 1)
        j_end = min(peak_k, n - 1)
        contributors = list(range(j_start, j_end + 1))

        if not contributors:
            break

        # Contribution of each a[j] to the peak: a[j] * a[peak_k - j]
        contributions = np.array([a[j] * a[peak_k - j] for j in contributors])
        total_contrib = contributions.sum()
        if total_contrib < 1e-15:
            break

        # Fraction of total mass that each contributor represents
        contrib_frac = contributions / total_contrib

        # Find positions with least total contribution to ALL peak bins
        K_peaks = max(3, min(10, n // 30))
        peak_bins = np.argsort(conv)[-K_peaks:]

        # Total contribution of each position to all peak bins
        total_contrib_per_pos = np.zeros(n)
        for pk in peak_bins:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(pk, n - 1)
            for j in range(js, je + 1):
                total_contrib_per_pos[j] += a[j] * a[pk - j]

        # We want to move mass from high-contribution positions to low-contribution positions
        # Sort by total contribution (descending)
        sorted_pos = np.argsort(total_contrib_per_pos)[::-1]

        # Source: top positions by contribution
        n_source = max(1, n // 10)
        sources = sorted_pos[:n_source]
        # Destination: bottom positions
        n_dest = max(1, n // 10)
        dests = sorted_pos[-n_dest:]

        # Compute transfer amount
        source_mass = a[sources].sum() * transfer_frac
        if source_mass < 1e-10:
            break

        # Distribute equally to sources and dests
        per_source = source_mass / max(1, len(sources))
        per_dest = source_mass / max(1, len(dests))

        a_try = a.copy()
        for j in sources:
            a_try[j] = max(0, a_try[j] - per_source)
        for j in dests:
            a_try[j] += per_dest

        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try smaller transfer
            transfer_frac *= 0.8
            if transfer_frac < 0.001:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Bimodal
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w1 = max(2, n // 8)
        w2 = max(2, n // 8)
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
        a = a / np.sum(a)
    elif mode == 3:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2 + 0.1
        a = a / np.sum(a)
    elif mode == 4:
        # Geometric decay from center
        center = n / 2
        x = np.arange(n)
        a = np.exp(-np.abs(x - center) / (n * 0.3))
        a = a / np.sum(a)
    elif mode == 5:
        # Sparse spikes
        a = np.zeros(n)
        n_spikes = max(3, n // 30)
        spike_pos = rng.choice(n, size=n_spikes, replace=False)
        a[spike_pos] = 1.0
        a = a / np.sum(a)
        # Smooth slightly
        kernel = np.array([0.2, 0.3, 0.3, 0.2])
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    elif mode == 6:
        # Random smooth (low-frequency Fourier)
        freqs = np.arange(n)
        coeffs = rng.normal(0, 1, n)
        coeffs[1:] *= 1.0 / freqs[1:] ** 0.7
        a = np.abs(np.fft.irfft(coeffs * n, n))
        a = a + 0.01
        a = a / np.sum(a)
    elif mode == 7:
        # Triangular
        x = np.arange(n) / max(1, n - 1)
        a = 1.0 - np.abs(2 * x - 1)
        a = a + 0.05
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Intensive optimization of parent ===
    print("\n=== Phase 1: Intensive optimization of parent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_opt, obj_opt = optimize_single(best_seq, remaining * 0.35, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt
            print(f"  After Phase 1: obj={best_score:.8f}")

    # === Phase 2: Multi-n exploration ===
    print("\n=== Phase 2: Multi-n exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200),
        max(300, n_best - 100),
        n_best - 50,
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        400, 500, 600, 700, 800, 900, 1000, 1200, 1400, 1600,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]
    # Sort by distance from current best n
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.15, 60.0)
        a_opt, obj_opt = optimize_single(a_try, alloc)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Diverse random restarts ===
    print("\n=== Phase 3: Diverse random restarts ===")
    n_best = len(best_seq)

    restart_configs = [
        (n_best, 0), (n_best, 1), (n_best, 2),
        (n_best, 3), (n_best, 4), (n_best, 5),
        (n_best, 6), (n_best, 7),
        (max(400, n_best - 100), 2), (max(400, n_best - 100), 6),
        (min(1500, n_best + 100), 1), (min(1500, n_best + 100), 4),
        (min(1500, n_best + 200), 3), (max(300, n_best - 200), 5),
    ]

    for idx, (n_r, mode) in enumerate(restart_configs):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_r = diverse_init(n_r, rng, mode)
        alloc = min(remaining * 0.12, 45.0)
        a_opt, obj_opt = optimize_single(a_r, alloc)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt
            print(f"  ** Restart {idx} (n={n_r}, mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {idx} (n={n_r}, mode={mode}): obj={obj_opt:.8f}")

    # === Phase 4: Blend combinations ===
    print("\n=== Phase 4: Blend combinations ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Try blending best_seq with uniform
        for blend_n in [len(best_seq)]:
            a_unif = np.ones(blend_n) / blend_n
            for t in [0.1, 0.2, 0.3, 0.5]:
                a_blend = (1 - t) * best_seq + t * a_unif
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / np.sum(a_blend)
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_score:
                    best_score = obj_blend
                    best_seq = a_blend
                    print(f"  ** Blend with uniform (t={t}): obj={best_score:.8f} **")

    # === Phase 5: Final intensive polish ===
    print("\n=== Phase 5: Final intensive polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Multiple rounds of LP with decreasing active set
        for pol in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.3, 50.0)
            max_act = [100, 60, 30][pol]
            a_lp = lp_alternating_active(best_seq, alloc, max_active=max_act)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp
                print(f"  Polish LP {pol} (act={max_act}): obj={best_score:.8f}")

    # === Phase 6: Final peak-directed and smooth max ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_pd = peak_directed_descent(best_seq, min(remaining * 0.3, 20.0))
        obj_pd = compute_obj(a_pd)
        if obj_pd < best_score:
            best_score = obj_pd
            best_seq = a_pd
            print(f"  Final Peak: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_sm = smooth_max_gradient(best_seq, min(remaining * 0.3, 20.0),
                                   T_init=0.0003, T_min=0.00001)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm
            print(f"  Final Smooth: obj={best_score:.8f}")

    # Final LP polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_lp = lp_alternating_active(best_seq, min(remaining * 0.4, 30.0), max_active=80)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp
            print(f"  Final LP: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
