"""
Hybrid optimizer for non-negative low-autocorrelation step functions.
Uses momentum subgradient with BB step sizes, LP alternating, frequency-domain
diversification, and systematic n-exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_for_lag(a, k):
    """Compute subgradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    # g[i] = 2 * a[k - i] for i in [lo, hi]
    src_indices = k - np.arange(lo, hi + 1)
    g[lo:hi+1] = 2.0 * a[src_indices]
    return g


def weighted_subgradient(a, conv, top_k=5, temperature=0.01):
    """Compute weighted subgradient using top-k lags with softmax weights."""
    n = len(a)
    # Get top-k indices
    if top_k >= len(conv):
        top_indices = np.arange(len(conv))
    else:
        top_indices = np.argsort(conv)[::-1][:top_k]
    
    # Softmax weights
    vals = conv[top_indices]
    vals_scaled = vals / max(temperature, 1e-30)
    vals_scaled -= vals_scaled.max()
    weights = np.exp(vals_scaled)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w, k in zip(weights, top_indices):
        g += w * subgradient_for_lag(a, int(k))
    return g

    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    mu = 0.85
    step = 1.0 / n
    start = time.monotonic()
    
    a_prev = a.copy()
    g_prev = None
    g_norm = 0.0
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        k_star = int(np.argmax(conv))
        
        # Determine top-k adaptively
        conv_sorted = np.sort(conv)[::-1]
        if len(conv_sorted) > 1:
            gap = (conv_sorted[0] - conv_sorted[1]) / max(conv_sorted[0], 1e-30)
            if gap > 0.1:
                top_k = 1
            elif gap > 0.01:
                top_k = 3
            else:
                top_k = min(10, n // 20 + 1)
        else:
            top_k = 1
        
        if top_k == 1:
            g = subgradient_for_lag(a, k_star)
        else:
            g = weighted_subgradient(a, conv, top_k=top_k, temperature=peak_val * 0.05)
        
        # BB step size
        if g_prev is not None:
            s_vec = a - a_prev
            y_vec = g - g_prev
            ys = float(np.dot(y_vec, s_vec))
            ss = float(np.dot(s_vec, s_vec))
            yy = float(np.dot(y_vec, y_vec))
            if ss > 1e-20 and yy > 1e-20:

                    step_bb = ss / max(abs(ys), 1e-20)
                else:
                    step_bb = max(ys, 1e-20) / yy
                step = float(np.clip(step_bb, 0.05 / n, 5.0 / n))
        
        a_prev = a.copy()
        g_prev = g.copy()
        
        # Nesterov momentum
        v = mu * v - step * g
        a_try = a + v
        a_try = np.maximum(a_try, 0.0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = a.copy()
        else:
            a_try = a_try / s_try
        
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            a = a_try
            stall += 1
            if stall > 30:
                a = best_a.copy()
                v = np.zeros(n)
                mu = 0.85
                step = 1.0 / n
                stall = 0
                # Small perturbation

                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
        
        it += 1
    
    return best_a


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, use_cvxpy=True):
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if use_cvxpy:
        A = build_conv_matrix(a_norm)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M_var.value)
            except Exception:
                continue
    
    try:
        A = build_conv_matrix(a_norm)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternate_optimize(a, time_budget=60.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        best_trial = a.copy()
        best_trial_obj = best_obj
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0, 0.4, 0.6, 0.8]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_trial = a_try.copy()
        
        if best_trial_obj < best_obj - 1e-12:
            best_obj = best_trial_obj
            best_a = best_trial.copy()
            a = best_trial
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break
    
    return best_a


def freq_domain_perturb(a, rng, amount=0.1):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    
    # Add random phase perturbations to high-frequency components
    cutoff = size // 4
    phase_perturb = rng.uniform(-amount, amount, size // 2 + 1)
    phase_perturb[:cutoff] = 0  # Keep low frequencies
    phase_perturb *= np.abs(fa) > 1e-10  # Only where signal exists
    
    angles = np.angle(fa) + phase_perturb
    magnitudes = np.abs(fa) * np.exp(rng.normal(0, amount / 2, size // 2 + 1))
    
    fa_new = magnitudes * np.exp(1j * angles)
    a_new = irfft(fa_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic interpolation via scipy
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def gradient_optimize_smoothed(a_init, time_budget=60.0, verbose=False):
    """Smoothed gradient optimization with L-BFGS-B."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    taus = [5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n]
    x = np.log(a) - np.log(a).max()
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        xv_start = x - x.max()
        f_val, g_val = smoothed_obj_and_grad(xv_start, tau)
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g
        
        try:
            res = minimize(fun, xv_start, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-10})
            if res.x is not None:
                x_opt = res.x - res.x.max()
                a_new = np.exp(x_opt)
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = x_opt
        except Exception:
            pass
    
    return best_a


def smoothed_obj_and_grad(x, tau):
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def random_init(n, rng):
    mode = int(rng.integers(0, 5))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Flat spectrum (random phase)
        size = next_pow2(2 * n - 1)
        angles = rng.uniform(0, 2 * np.pi, size // 2 + 1)
        fa = np.exp(1j * angles)
        fa[0] = np.abs(fa[0])  # Keep DC real
        # Enforce Hermitian symmetry for real signal
        a_raw = irfft(fa, size)[:n]
        a = np.maximum(a_raw, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    elif mode == 3:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    else:
        # Triangular with noise
        a = 1.0 - np.abs(np.arange(n) - n / 2) / (n / 2 + 1)
        a = a + 0.05 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a
