    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    a = a / S  # Normalize
    
    best_a = a.copy()
    best_obj = computeObjective(a)
    start = time.monotonic()
    
    lr = 0.01
    momentum = 0.9
    vel = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        T = T_schedule[it] if it < len(T_schedule) else T_schedule[-1]
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = conv - max_conv
        w = np.exp(shifted / T)
        w_sum = float(np.sum(w))
        w = w / w_sum
        
        # Smooth objective
        obj_smooth = 2.0 * n * (max_conv + T * np.log(w_sum)) / S**2
        
        # Gradient
        corr = np.correlate(w, a, mode='valid')[:n]
        grad_conv = 2.0 * corr
        
        # Full gradient of F_smooth
        grad = 2.0 * n * grad_conv / (S * S) - 2.0 * obj_smooth * a / S
        
        # Gradient descent with momentum
        vel = momentum * vel - lr * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0)
        S_new = float(np.sum(a_new))
        if S_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / S_new
        
        # Check actual (non-smooth) objective
        obj_actual = computeObjective(a_new)
        
        if obj_actual < best_obj:
            best_obj = obj_actual
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.1)
        else:
            # Try smaller step
            a = best_a.copy()
            lr *= 0.8
    
    return best_a
