Advanced optimizer for low-peak self-convolution non-negative sequences.
Combines coordinate descent, adaptive smooth-max gradient, LP alternating,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog, minimize_scalar
import cvxpy as cp
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_grad_topk(a, k_top=5):
    """Compute objective and gradient using top-k smooth max."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf, np.zeros(n)
    a_norm = a / s
    conv = compute_conv(a_norm)
    
    # Top-k selection
    k = min(k_top, len(conv))
    top_k_indices = np.argpartition(conv, -k)[-k:]
    top_k_vals = conv[top_k_indices]
    
    # Smooth max on top-k with small temperature
    tau = 0.1 / n
    c_scaled = top_k_vals / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w_top = exp_c / sum_exp
    
    # Build full weight vector
    w = np.zeros(len(conv))
    w[top_k_indices] = w_top
    
    # Peak value
    peak = c_max * tau + tau * np.log(sum_exp)
    obj = 2.0 * n * peak  # since a_norm has sum=1
    
    # Gradient: d(peak)/da[i] = 2*sum_k w[k]*a_norm[k-i] (if valid)
    size = next_pow2(2 * n - 1)
    a_flip = a_norm[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    corr = irfft(fw * fa_flip, size)[:n]
    
    grad = 4.0 * n * corr  # d(2n*peak)/da[i] for normalized a
    
    return obj, grad


def compute_obj_grad_softmax(a, tau):
    """Compute objective and gradient using full softmax smoothing."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf, np.zeros(n)
    a_norm = a / s
    conv = compute_conv(a_norm)
    
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    
    peak = c_max * tau + tau * np.log(sum_exp)
    obj = 2.0 * n * peak
    
    size = next_pow2(2 * n - 1)
    a_flip = a_norm[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    corr = irfft(fw * fa_flip, size)[:n]
    
    grad = 4.0 * n * corr
    
    return obj, grad


def gradient_step(a, tau, steps=50, verbose=False):
    """Perform gradient descent on the smoothed objective."""
    n = len(a)
    a = np.maximum(a, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for step in range(steps):
        obj, grad = compute_obj_grad_softmax(a, tau)
        
        # Normalize gradient for step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Line search
        best_step_obj = best_obj
        best_step_a = a.copy()
        
        for lr in [0.1, 0.05, 0.02, 0.01, 0.005, 0.5, 1.0]:
            a_try = a - lr * grad / grad_norm
            a_try = np.maximum(a_try, 1e-30)
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()
        
        if best_step_obj < best_obj:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_step_a
            if verbose:
                print(f"    grad step {step}: obj={best_obj:.10f}")
        else:
            break
    
    return best_a


def coordinate_descent(a, max_sweeps=10, verbose=False):
    """Coordinate descent: optimize each element one at a time."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Precompute: for each i, which conv entries depend on a[i]
    # conv[k] = sum_j a[j]*a[k-j], so conv[k] depends on a[i] if k-i is a valid index
    # i.e., 0 <= k-i < n, so k ranges from i to i+n-1
    
    for sweep in range(max_sweeps):
        improved = False
        # Randomize order
        order = np.random.permutation(n)
        
        for i in order:
            # Save current value
            a_i = a[i]
            
            # Compute conv without a[i] contribution
            # conv[k] without a[i]: for each k, the contribution of a[i] is:
            #   a[i]*a[k-i] (if 0<=k-i<n and k-i != i)
            #   2*a[i]*a[k-i] if k-i == i (i.e., k=2i), but actually:
            #   conv[k] = sum_j a[j]*a[k-j], the terms involving a[i] are:
            #     j=i: a[i]*a[k-i] (if 0<=k-i<n)
            #     k-j=i => j=k-i: a[k-i]*a[i] (if 0<=k-i<n)
            #   So contribution is 2*a[i]*a[k-i] if 0<=k-i<n and k-i!=i
            #   Or a[i]^2 if k-i=i (k=2i)
            
            # Simpler: just evaluate the 1D function directly
            def obj_of_x(x):
                a[i] = x
                s_val = s - a_i + x
                if s_val < 0.01:
                    return np.inf
                a_norm = a / s_val
                conv = compute_conv(a_norm)
                return 2.0 * n * np.max(conv)
            
            # Find optimal x
            # The function is unimodal-ish, use bounded minimization
            x_max = max(2 * a_i, 2.0 / n * 10) if a_i > 0 else 2.0 / n * 5
            res = minimize_scalar(obj_of_x, bounds=(0.0, x_max), method='bounded',
                                   options={'maxiter': 50, 'xatol': 1e-8})
            
            if res.success:
                a[i] = max(0.0, res.x)
                s = float(np.sum(a))
                a = a / s
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
        if verbose:
            print(f"    coord sweep {sweep}: obj={best_obj:.10f}")
    
    # Restore
    a[i] = res.x if 'res' in dir() else a_i
    return best_a


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Try cvxpy first
    try:
        A = build_conv_matrix(a_norm)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s_val = float(np.sum(g_val))
                    if s_val > 1e-10:
                        return g_val / s_val, float(M_var.value)
            except Exception:
                continue
    except Exception:
        pass
    
    # Fallback: scipy linprog (only for smaller n)
    if n <= 500:
        try:
            A = build_conv_matrix(a_norm)
            c_vec = np.zeros(n + 1)
            c_vec[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(0, None)]
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(res.x[n])
        except Exception:
            pass
    
    return None, None


def lp_alternate_step(a):
    """One LP alternating step with backtracking line search."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    g_lp, M_lp = solve_lp_minpeak(a_norm)
    if g_lp is None:
        return a, compute_obj(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Backtracking line search
    t = 1.0
    for _ in range(20):
        a_try = (1.0 - t) * a_norm + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
        t *= 0.7
    
    return best_a, best_obj


def lp_alternate_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating with perturbation on stall."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        a_new, obj_new = lp_alternate_step(a)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 5:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                scale = min(0.05, 0.01 * stall)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
    
    return best_a


def freq_init(n, rng):
    """Frequency-domain inspired initialization."""
    mode = rng.integers(0, 5)
    
    if mode == 0:
        # Flat spectrum with random phases
        freq = np.ones(n)
        phases = rng.uniform(-np.pi, np.pi, n)
        a = np.real(irfft(freq * np.exp(1j * phases), 2 * n))[:n]
        a = np.maximum(a, 0)
        if np.sum(a) < 1e-10:
            a = np.ones(n)
        a = a / np.sum(a)
    elif mode == 1:
        # Low-pass filtered random
        a = rng.uniform(0, 1, n)
        # Apply low-pass in frequency domain
        size = next_pow2(n)
        fa = rfft(a, size)
        cutoff = int(size * 0.3)
        fa[cutoff:] *= 0.1
        a = np.real(irfft(fa, size))[:n]
        a = np.maximum(a, 0)
        if np.sum(a) < 1e-10:
            a = np.ones(n)
        a = a / np.sum(a)
    elif mode == 2:
        # Cosine window with perturbation
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a += 0.1 * rng.uniform(0, 1, n)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    elif mode == 3:
        # Two-peak Gaussian
        c1, c2 = n // 4, 3 * n // 4
        sigma = n / 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * sigma ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * sigma ** 2))
        a += 0.05 * rng.uniform(0, 1, n)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        # Thue-Morse inspired (alternating blocks)
        block = max(2, n // 64)
        for i in range(n):
            # Count trailing 1s in binary representation
            v = i // block
            a_tmp = 1.0 if bin(v).count('1') % 2 == 0 else 0.3
            a = np.zeros(n)
        a = np.array([1.0 if bin(i // block).count('1') % 2 == 0 else 0.3 for i in range(n)])
        a += 0.05 * rng.uniform(0, 1, n)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    
    return a


def multi_tau_gradient(a, time_budget=60.0, verbose=False):
    """Multi-scale gradient optimization with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive tau schedule: start large, decrease
    tau = 5.0 / n
    min_tau = 0.01 / n
    
    while time.monotonic() - start < time_budget:
        # Perform a few gradient steps at current tau
        a_new = gradient_step(a, tau, steps=20)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Decrease tau
            tau *= 0.7
            if verbose:
                print(f"    multi-grad tau={tau:.6f}: obj={best_obj:.10f}")
        else:
            # Try smaller tau
            tau *= 0.5
            if tau < min_tau:
                # Perturb and reset
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                tau = min_tau * 10
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def hybrid_optimize(a, time_budget=120.0, verbose=False):
    """Combined optimization: gradient + LP + coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    phase1_end = start + time_budget * 0.4
    phase2_end = start + time_budget * 0.7
    phase3_end = start + time_budget
    
    # Phase 1: Gradient
    if time.monotonic() < phase1_end:
        a_g = multi_tau_gradient(a, time_budget=phase1_end - time.monotonic(), verbose=verbose)
        obj = compute_obj(a_g)
        if obj < best_obj:
            best_obj = obj
            best_a = a_g.copy()
            if verbose:
                print(f"    After gradient: {best_obj:.10f}")
    
    # Phase 2: LP alternating
    remaining = phase2_end - time.monotonic()
    if remaining > 5:
        a_lp = lp_alternate_optimize(best_a, time_budget=min(remaining, 60.0))
        obj = compute_obj(a_lp)
        if obj < best_obj:
            best_obj = obj
            best_a = a_lp.copy()
            if verbose:
                print(f"    After LP: {best_obj:.10f}")
    
    # Phase 3: Coordinate descent + final gradient
    remaining = phase3_end - time.monotonic()
    if remaining > 10:
        # Coordinate descent
        a_cd = coordinate_descent(best_a, max_sweeps=min(10, int(remaining / 5)), verbose=verbose)
        obj = compute_obj(a_cd)
        if obj < best_obj:
            best_obj = obj
            best_a = a_cd.copy()
            if verbose:
                print(f"    After coord: {best_obj:.10f}")
        
        # Final gradient polish
        remaining = phase3_end - time.monotonic()
        if remaining > 10:
            a_f = multi_tau_gradient(best_a, time_budget=min(remaining, 30.0))
            obj = compute_obj(a_f)
            if obj < best_obj:
                best_obj = obj
                best_a = a_f.copy()
                if verbose:
                    print(f"    After final grad: {best_obj:.10f}")
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines multi-scale gradient, LP alternating, coordinate descent,
    frequency-domain initializations, and systematic n-exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Intensive optimization on parent ===
    print(f"\n=== Phase 1: Intensive Opt (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        a_opt = hybrid_optimize(best_seq, time_budget=min(remaining * 0.3, 180.0), verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 1 done: obj={best_score:.10f}")
    
    # === Phase 2: Wider n exploration ===
    print(f"\n=== Phase 2: N Exploration ===")
    n_candidates = sorted(set([
        max(400, n_parent - 200), max(400, n_parent - 100),
        n_parent,
        n_parent + 100, n_parent + 200,
        600, 700, 900, 1000, 1100, 1200, 1300
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1400]
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        alloc = min(remaining * 0.12, 50.0)
        a_opt = hybrid_optimize(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")
    
    # === Phase 3: Diverse restarts ===
    print(f"\n=== Phase 3: Diverse Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    n_restarts = max(4, int(remaining / 40))
    n_restarts = min(n_restarts, 12)
    
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        if r == 0:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            # Frequency init at best n
            a_r = freq_init(n_best, rng)
        elif r == 2:

            n_r = n_best + int(rng.integers(-150, 150))
            n_r = max(500, min(1300, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.02, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 3:

            a_r = (a_r + a_r[::-1]) / 2.0
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 4:
            # Frequency init at different n
            n_r = max(500, min(1300, n_best + int(rng.integers(-200, 200))))
            a_r = freq_init(n_r, rng)
        elif r == 5:
            # Concentrated center
            center = n_best // 2
            x = np.arange(n_best)

            a_r = a_r / np.sum(a_r)

            a_r += 0.05 * rng.uniform(0, 1, n_best)

            a_r = a_r / np.sum(a_r)

        if obj < best_score:

            a_cd = coordinate_descent(best_seq, max_sweeps=min(5, int(remaining / 10)))
            obj = compute_obj(a_cd)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()
                print(f"  Final coord: obj={best_score:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
