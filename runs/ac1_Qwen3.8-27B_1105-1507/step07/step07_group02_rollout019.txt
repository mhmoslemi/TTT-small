import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(a, target_sum=1.0):
    """Project onto {x: sum(x)=target_sum, x>=0}."""
    n = len(a)
    if n == 0:
        return a
    # Find threshold t such that sum(max(a_i - t, 0)) = target_sum
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv[-1] - target_sum))[0]
    if len(rho) == 0:
        t = (cssv[-1] - target_sum) / n
    else:
        rho = rho[-1]
        t = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(a - t, 0)

def soft_max_obj_and_grad(a, tau):
    """Compute soft-max objective and gradient for a (assumed sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    # Soft-max
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)  # for numerical stability
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    # Objective
    obj = 2.0 * n * (max_conv + np.log(np.sum(exp_shifted)) / tau)
    
    # Gradient: d(obj)/da[i] = 2*n * sum_k w[k] * d(conv[k])/da[i]
    # d(conv[k])/da[i] = 2*a[k-i] (if valid, with center correction)
    # This is a cross-correlation: grad = 2*n * 2 * (w cross-corr a)
    # With center correction for even k
    
    # Compute cross-correlation via FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a, size)
    # Cross-correlation: (w * a)[i] = sum_k w[k] * a[k-i]
    # In FFT: IFFT(FFT(w) * conj(FFT(a)))
    cross_corr = irfft(fw * np.conj(fa), size)
    
    # But we need to be careful about the indexing.
    # conv[k] = sum_i a[i]*a[k-i]

"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses LP alternating, projected gradient with soft-max, water-filling, and multi-n search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v, target=1.0):
    """Euclidean projection onto {x: sum(x)=target, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([target])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv[-1] - target)
    rho = np.argmax(~cond) if not cond.all() else n - 1
    if not cond.any():
        t = (cssv[-1] - target) / n
    else:
        t = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - t, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A
