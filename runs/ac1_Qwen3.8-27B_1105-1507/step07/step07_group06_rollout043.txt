import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def conv_gradient_at_k(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= k - j < n:
            g += a[k - j]
        if 0 <= j - k < n:
            g += a[j - k]
        grad[j] = g
    return grad

            idx2_end = hi2 - k

def conv_grad_at_lags(a, lags):

    grads = np.zeros((len(lags), n))

        # Second term: a[j-k] valid for j in [max(0,k), min(k+n-1,n-1)]
        lo2 = max(0, k)

def smooth_max(conv, tau):
    """Compute smooth max and weights (soft-argmax)."""
    conv_max = np.max(conv)
    scaled = (conv - conv_max) * tau
    scaled -= np.max(scaled)
    exp_scaled = np.exp(scaled)
    w = exp_scaled / np.sum(exp_scaled)
    val = conv_max + np.log(np.sum(exp_scaled)) / tau
    return val, w


def smoothed_obj_and_grad(a, tau):
    """
    Compute smoothed objective and gradient w.r.t. a (on simplex).
    Returns (obj_val, grad_a) where grad_a is the gradient projected to sum=0.

def adam_smoothed_descent(a_init, tau_schedule, n_steps, time_budget):

    Adam-optimized smoothed gradient descent on the simplex.

    a = a / np.sum(a)

    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for t in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        tau = tau_schedule[min(t, len(tau_schedule) - 1)]
        obj_val, grad = smoothed_obj_and_grad_fast(a, tau, top_k=8)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def build_conv_matrix_fixed_a(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_fixed_a(a)
    
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, time_budget=60.0, max_iter=100):
    """LP alternating with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stuck_count = 0
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = lp_minpeak(a)
        if g_lp is None:
            break
        
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count >= 5:
                # Perturb to escape
                rng = np.random.default_rng(it * 1000003 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stuck_count = 0
    
    return best_a, best_obj


def proposal_subgradient(a_init, time_budget=30.0, n_steps=500):
    """
    Subgradient method with adaptive step sizes (polyak-like).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.02
    
    for t in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        conv_max = conv[k_star]
        
        # Subgradient
        grad = np.zeros(n)
        lo1 = max(0, k_star - n + 1)
        hi1 = min(k_star, n - 1)
        if lo1 <= hi1:
            src_s = k_star - hi1
            src_e = k_star - lo1
            grad[lo1:hi1 + 1] = a[src_s:src_e + 1][::-1]
        if k_star > 0:
            lo2 = max(0, k_star)
            hi2 = min(k_star + n - 1, n - 1)
            if lo2 <= hi2:
                src2_s = lo2 - k_star
                src2_e = hi2 - k_star
                grad[lo2:hi2 + 1] += a[src2_s:src2_e + 1]
        if k_star == 0:
            grad[0] = 2.0 * a[0]
        
        # Project gradient
        grad = grad - np.mean(grad)
        
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            step_size *= 0.5
            continue
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            step_size = min(step_size * 1.02, 0.1)
        else:
            step_size *= 0.95
    
    return best_a, best_obj


def geometric_init(n, ratio):
    """Create geometric sequence a[j] = ratio^j, normalized."""
    a = ratio ** np.arange(n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def ramp_init(n, slope):
    """Create linear ramp sequence."""
    a = np.linspace(1.0, slope, n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def combined_init(n, ratio, slope, blend):
    """Blend geometric and ramp."""
    a_geo = geometric_init(n, ratio)
    a_ramp = ramp_init(n, slope)
    a = blend * a_geo + (1 - blend) * a_ramp
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def refine_sequence(a, time_budget=30.0):
    """
    Refine a sequence using multiplicative update + Adam + LP in sequence.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Step 1: Multiplicative pressure update
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a1, obj1 = multiplicative_pressure_update(
            best_a, n_iter=100, time_budget=min(remaining * 0.4, 15.0))
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
    
    # Step 2: Adam smoothed descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        tau_sched = np.geomspace(20, 200, 50)
        a2, obj2 = adam_smoothed_descent(
            best_a, tau_sched, n_steps=100,
            time_budget=min(remaining * 0.35, 12.0))
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
    
    # Step 3: LP alternating polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 500:
        a3, obj3 = lp_alternating(best_a, time_budget=min(remaining * 0.6, 
                                                          max(5.0, remaining - 5)),
                                   max_iter=50)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
    
    # Step 4: Final multiplicative polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a4, obj4 = multiplicative_pressure_update(
            best_a, n_iter=50, time_budget=min(remaining * 0.8, 10.0),
            alpha_init=0.2, alpha_end=0.05)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multiple optimization strategies
    with diverse initializations and progressive refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}", flush=True)
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = geometric_init(800, 1.005)
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}", flush=True)
    
    n_best = len(best_seq)
    
    # ============ Phase 1: Intensive refinement of parent ============
    print("\n=== Phase 1: Refine parent ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_ref, obj_ref = refine_sequence(best_seq, time_budget=min(remaining * 0.4, 120.0))
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            n_best = len(best_seq)
            print(f"  Refined: obj={best_score:.8f} **", flush=True)
        else:
            print(f"  Refined: obj={obj_ref:.8f}", flush=True)
    
    # ============ Phase 2: Diverse initializations ============
    print("\n=== Phase 2: Diverse initializations ===", flush=True)
    
    # Try different sequence lengths and initialization types
    n_options = [n_best]
    for delta in [-100, -50, 50, 100, 200]:
        nn = n_best + delta
        if 400 <= nn <= 2000:
            n_options.append(nn)
    n_options = sorted(set(n_options))
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        # Resample to target length
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s
        
        # Refine
        alloc = min(remaining * 0.25, 60.0)
        a_ref, obj_ref = refine_sequence(a_try, time_budget=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            n_best = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **", flush=True)
        else:
            print(f"  n={n_try}: obj={obj_ref:.8f}", flush=True)
    
    # ============ Phase 3: Geometric and ramp initializations ============
    print("\n=== Phase 3: Geometric/Ramp inits ===", flush=True)
    
    n_cur = len(best_seq)
    init_configs = [
        (geometric_init(n_cur, 1.5), "geo r=1.5"),

        (ramp_init(n_cur, 3.0), "ramp 3x"),
        (ramp_init(n_cur, 5.0), "ramp 5x"),

    ]
    
    # Also try reversed parent
    init_configs.append((best_seq[::-1].copy(), "reversed parent"))

    pert = best_seq * np.exp(rng.normal(0, 0.1, n_cur))
    pert = np.maximum(pert, 0)
    pert = pert / np.sum(pert)
    init_configs.append((pert, "perturbed parent"))
    
    for a_init, name in init_configs:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        alloc = min(remaining * 0.15, 40.0)
        a_ref, obj_ref = refine_sequence(a_init, time_budget=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()

        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **", flush=True)
        else:
            print(f"  LP: obj={obj_lp:.8f}", flush=True)
    
    # ============ Phase 6: Random restarts ============
    print("\n=== Phase 6: Random restarts ===", flush=True)
    
    n_best = len(best_seq)
    for r in range(8):

        if remaining < 25:
            break
        
        # Various perturbation strategies
        strategy = r % 4
        if strategy == 0:
            # Gaussian perturbation
            a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
        elif strategy == 1:
            # Add a geometric component
            a_r = 0.7 * best_seq + 0.3 * geometric_init(n_best, rng.uniform(1.5, 4.0))

        else:

            shift = rng.integers(50, n_best // 3)
            a_r = np.roll(best_seq, shift)
        
        a_r = np.maximum(a_r, 0)
        s = np.sum(a_r)
        if s > 1e-10:
            a_r = a_r / s
        
        alloc = min(remaining * 0.1, 25.0)
        a_ref, obj_ref = refine_sequence(a_r, time_budget=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  Restart {r} (strat={strategy}): obj={best_score:.8f} **", flush=True)

            print(f"  Restart {r} (strat={strategy}): obj={obj_ref:.8f}", flush=True)

    print("\n=== Phase 7: Final polish ===", flush=True)

    if remaining > 20:

        alloc = min(remaining * 0.3, 60.0)
        a_f1, obj_f1 = multiplicative_pressure_update(

        a_f2, obj_f2 = proposal_subgradient(best_seq, time_budget=alloc, n_steps=300)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final subgradient: obj={best_score:.8f}", flush=True)
    
    remaining = deadline - time.monotonic()
    if remaining > 10 and len(best_seq) <= 500:
        # Last LP
        alloc = min(remaining * 0.5, 30.0)

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===", flush=True)
    
    return result
