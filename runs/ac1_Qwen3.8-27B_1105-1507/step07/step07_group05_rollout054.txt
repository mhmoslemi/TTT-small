"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral flattening (GS-like) for quick initial improvement
2. Log-sum-exp gradient descent with adaptive cooling (main workhorse)
3. Subgradient peak chasing (periodic)
4. LP polishing for final precision
5. Multi-n exploration
Key innovation: log-sum-exp surrogate with FFT-based gradient computation,
combined with spectral domain operations for novel search directions.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import math


def next_power_of_2(n):
    """Return smallest power of 2 >= n."""
    return 1 << (n - 1).bit_length()


def fft_conv(a, b, N=None):
    """Linear convolution via FFT. If b is None, compute auto-convolution."""
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    if N is None:
        N = next_power_of_2(n1 + n2 - 1)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:n1 + n2 - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    N = next_power_of_2(2 * n - 1)
    fa = rfft(a_norm, N)
    b = irfft(fa * fa, N)[:2 * n - 1]
    return 2.0 * n * float(np.max(b))


def compute_b(a, N=None):
    """Compute auto-convolution of a (assumed normalized to sum=1)."""
    n = len(a)
    if N is None:
        N = next_power_of_2(2 * n - 1)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def spectral_flattening(a, time_budget=30.0, verbose=False):
    """
    GS-like iterative method: alternate between
    - Time domain: enforce non-negativity and normalization
    - Frequency domain: flatten |FFT(a)| magnitude
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = next_power_of_2(2 * n - 1)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Target: make |A[k]| as uniform as possible (except k=0 which must be 1)
    # Compute "ideal" target magnitude
    # sum(|A|^2)/N = ||a||^2 (Parseval with numpy convention: sum|x|^2 = 1/N * sum|X|^2)
    # So sum|A|^2 = N * ||a||^2
    # If |A[0]|=1 and |A[k]|=c for k!=0: 1 + (N-1)*c^2 = N*||a||^2
    # c = sqrt((N*||a||^2 - 1)/(N-1))

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        A = rfft(a, N)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        target_sq = (N * a_energy - 1.0) / max(N - 1, 1)
        target_sq = max(target_sq, 1e-10)
        target = np.sqrt(target_sq)

        # Set new magnitude: keep phase, replace magnitude
        # But keep A[0] real and positive (= sum(a) = 1)

        A_new[0] = 1.0  # sum(a) = 1

def lse_gradient_step(a, tau, N):
    """
    Compute gradient of L = tau * log(sum(exp(b/tau))) with respect to a.
    Returns gradient vector of length n.
    """
    n = len(a)
    b = compute_b(a, N)

    # Softmax weights
    b_scaled = b / tau
    b_max = float(np.max(b_scaled))
    exp_b = np.exp(b_scaled - b_max)
    s_w = exp_b / float(np.sum(exp_b))

    # Gradient: grad[i] = sum_k s_w[k] * (a[k-i] + a[i-k])
    # Part A: sum_k s_w[k] * a[k-i] = conv(a, s_w[::-1])[2n-2-i]
    # Part B: sum_k s_w[k] * a[i-k] = conv(a, s_w)[i]

    N2 = next_power_of_2(n + 2 * n - 1 - 1)  # n + (2n-1) - 1 = 3n-2

    fa = rfft(a, N2)
    fs = rfft(s_w, N2)
    fs_rev = rfft(s_w[::-1], N2)

    conv_as = irfft(fa * fs, N2)[:n + 2 * n - 2]  # conv(a, s_w), length 3n-2
    conv_as_rev = irfft(fa * fs_rev, N2)[:n + 2 * n - 2]  # conv(a, s_w[::-1]), length 3n-2

    # Part B[i] = conv(a, s_w)[i] for i=0..n-1
    part_b = conv_as[:n]

    # Part A[i] = conv(a, s_w[::-1])[2n-2-i] for i=0..n-1
    # So Part A = conv(a, s_w[::-1])[n-1 : 2n-1] reversed
    part_a = conv_as_rev[n - 1:2 * n - 1][::-1]

    grad = part_a + part_b
    return grad, b, s_w


def lse_gradient_descent(a, time_budget=200.0, verbose=False):
    """
    Log-sum-exp gradient descent with adaptive cooling.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = next_power_of_2(2 * n - 1)
    b = compute_b(a, N)

    # Initial temperature: large enough for smoothness
    tau = float(np.max(b)) / 3.0
    tau_min = float(np.max(b)) / 500.0

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(b))
    start = time.monotonic()

    eta = tau * n * 0.1
    eta_max = tau * n * 0.5
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    max_stall = 500

    for it in range(1000000):
        if time.monotonic() - start > time_budget:
            break

        # Compute gradient
        grad, b, s_w = lse_gradient_step(a, tau, N)

        # Center gradient (preserves sum)
        grad = grad - float(np.mean(grad))

        # Momentum
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        b_new = compute_b(a_new, N)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.005, eta_max)
            momentum = min(momentum + 0.0005, 0.9)

            # Cool temperature

                tau = max(tau * 0.998, tau_min)
                eta = tau * n * 0.1
                eta_max = tau * n * 0.5

            stall += 1
            eta *= 0.95
            v *= 0.5

                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = tau * n * 0.05
                momentum = 0.5
                tau = max(tau * 1.5, tau_min)
                stall = 0
                max_stall = min(max_stall + 200, 2000)

            if eta < 1e-12:
                break

        if verbose and it % 2000 == 0:
            print(f"            LSE-GD it{it}: obj={best_obj:.8f}, tau={tau:.6f}, eta={eta:.6f}")

def subgradient_chase(a, time_budget=30.0, verbose=False):

    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_b(a, N)))
    start = time.monotonic()

    eta = 0.1
    avg_a = a.copy()
    total_weight = 1.0

    # Track peak positions for momentum
    prev_k = -1

        if time.monotonic() - start > time_budget:
            break

        b = compute_b(a, N)
        k_star = int(np.argmax(b))

        # db[k_star]/da[i] = a[k_star - i] + a[i - k_star] (with bounds)

        for i in range(n):

                grad[i] += a[j]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]

        # Center
        grad = grad - float(np.mean(grad))

        # Step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        b_new = compute_b(a_new, N)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.002, 0.5)

            # Nesterov averaging
            total_weight += 1.0
            avg_a = (avg_a * (total_weight - 1) + a_new) / total_weight

        else:
            eta *= 0.98
            if eta < 1e-8:
                break

        # Periodically check the averaged solution
        if it % 100 == 0 and total_weight > 10:
            obj_avg = compute_obj(avg_a)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = avg_a.copy()
                a = avg_a.copy()

    # Check averaged solution at end
    if total_weight > 10:
        obj_avg = compute_obj(avg_a)
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = avg_a.copy()

    return best_a


def energy_gradient_descent(a, time_budget=60.0, verbose=False):
    """
    Energy (L2) gradient descent: minimize ||conv(a,a)||^2 via FFT.
    This is a smooth surrogate that helps escape sharp local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = next_power_of_2(3 * n - 2)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)
    stall = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Gradient of (1/2)||b||^2 = sum b[k]^2 / 2
        # d/d a[i] = sum_k b[k] * db[k]/da[i] = sum_k b[k] * (a[k-i] + a[i-k])
        # = conv(a, b) + conv(a, b[::-1]) (appropriate slices)

        # Use FFT: grad = conv(a, b) sliced + conv(a, b_rev) sliced
        fb = rfft(b, N)
        fb_rev = rfft(b[::-1], N)

        conv_ab = irfft(fa * fb, N)[:3 * n - 2]
        conv_ab_rev = irfft(fa * fb_rev, N)[:3 * n - 2]

        # Part B: conv(a,b)[i] for i=0..n-1
        part_b = conv_ab[:n]
        # Part A: conv(a,b_rev)[2n-2-i] for i=0..n-1
        part_a = conv_ab_rev[n - 1:2 * n - 1][::-1]

        grad = part_a + part_b
        grad = grad - float(np.mean(grad))

        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        b_new = irfft(rfft(a_new, N) ** 2, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.003, 3.0)
        else:
            stall += 1
            eta *= 0.97
            v *= 0.5
            if stall > 300:
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.7
                stall = 0
            if eta < 1e-10:
                break

        if verbose and it % 5000 == 0:
            print(f"            Energy-GD it{it}: obj={best_obj:.8f}")

    return best_a


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a = a / float(np.sum(a))

    # Build constraint matrix: A[k, i] such that (A @ g)[k] = conv(a, g)[k]
    # conv(a, g)[k] = sum_i a[i] * g[k-i] (with bounds)
    # For fixed k: sum_i A[k,i] * g[i] where A[k,i] = a[k-i] if 0<=k-i<n else 0
    #             + a[i-k] if 0<=i-k<n else 0
    # Wait, conv(a,g)[k] = sum_i a[i]*g[k-i] where 0<=i<n, 0<=k-i<n
    # = sum_i a[i]*g[k-i] for max(0,k-n+1) <= i <= min(k, n-1)
    # So A[k,i] = a[i] if max(0,k-n+1) <= i <= min(k,n-1) else 0

    # This is a Hankel-like matrix. For n up to ~1000, it's (2n-1) x n.
    # Use scipy linprog for smaller n, skip for very large n.

    if n > 1200:
        return None

    try:
        from scipy.optimize import linprog

        A_mat = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            A_mat[k, i_lo:i_hi + 1] = a[i_lo:i_hi + 1]

        # min M
        # s.t. A_mat @ g - M <= 0 for all k
        #      sum(g) = 1
        #      g >= 0, M >= 0

        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M

        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A_mat
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing: alternate between LP solution and current."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try different mixing parameters
        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose:
                print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 5:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = int(rng.integers(0, 8))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = float(np.sum(a))
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.3
        a = a / np.sum(a)
    elif mode == 5:
        # Triangular
        x = np.arange(n) / max(n - 1, 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    elif mode == 6:
        # Edge-concentrated
        x = np.arange(n) / max(n - 1, 1)
        a = np.exp(-10 * x) + np.exp(-10 * (1 - x)) + 0.01
        a = a / np.sum(a)
    else:
        # Random smooth
        freqs = rng.integers(1, 5, 4)
        phases = rng.uniform(0, 2 * np.pi, 4)
        amps = rng.uniform(0.5, 1.0, 4)
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0
        for f, ph, am in zip(freqs, phases, amps):
            a += am * np.cos(2 * np.pi * f * x + ph)
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        a = a / s if s > 0 else np.ones(n) / n
    return a


def spectral_perturb(a, rng, strength=0.3):
    """
    Perturb a in the frequency domain: add random phase noise to non-DC components.
    This explores structurally different but related sequences.
    """
    n = len(a)
    N = next_power_of_2(2 * n - 1)
    A = rfft(a, N)

    # Add random phase perturbation (keep DC)
    noise = rng.normal(0, strength, len(A))
    noise[0] = 0  # Don't perturb DC
    A_pert = A + 1j * noise * np.abs(A)

    a_new = irfft(A_pert, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies:
    1. Spectral flattening for fast initial improvement
    2. Log-sum-exp GD with cooling (primary method)
    3. Subgradient peak chasing
    4. Energy GD for smoothing
    5. LP polishing for precision
    6. Multi-n and spectral perturbation exploration
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    def update_best(a_new, label=""):
        nonlocal best_seq, best_score
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  ** {label}: obj={best_score:.8f} **")
            return True
        else:
            print(f"  {label}: obj={obj_new:.8f}")
            return False

    def remaining_time():
        return max(0, deadline - time.monotonic())

    # === Phase 1: Spectral Flattening (quick improvement) ===
    print(f"\n=== Phase 1: Spectral Flattening (n={n_best}) ===")
    t = remaining_time()
    if t > 30:
        alloc = min(t * 0.05, 30.0)
        a_sf = spectral_flattening(best_seq, time_budget=alloc, verbose=True)
        update_best(a_sf, "SpectralFlatten")

    # === Phase 2: Log-sum-exp GD (main workhorse) ===
    print(f"\n=== Phase 2: LSE Gradient Descent (n={n_best}) ===")
    t = remaining_time()
    if t > 60:
        alloc = min(t * 0.35, 400.0)
        a_lse = lse_gradient_descent(best_seq, time_budget=alloc, verbose=True)
        update_best(a_lse, "LSE-GD")

    # === Phase 3: Subgradient Peak Chase ===
    print(f"\n=== Phase 3: Subgradient Chase ===")
    t = remaining_time()
    if t > 30:
        alloc = min(t * 0.1, 60.0)
        a_sg = subgradient_chase(best_seq, time_budget=alloc)
        update_best(a_sg, "SubgradChase")

    # === Phase 4: Energy GD (smoothing) ===
    print(f"\n=== Phase 4: Energy GD ===")
    t = remaining_time()
    if t > 30:
        alloc = min(t * 0.1, 60.0)
        a_eg = energy_gradient_descent(best_seq, time_budget=alloc)
        update_best(a_eg, "EnergyGD")

    # === Phase 5: Second LSE round (from improved point, lower temp) ===
    print(f"\n=== Phase 5: LSE GD Round 2 ===")
    t = remaining_time()
    if t > 60:
        alloc = min(t * 0.2, 200.0)
        a_lse2 = lse_gradient_descent(best_seq, time_budget=alloc, verbose=True)
        update_best(a_lse2, "LSE-GD-R2")

    # === Phase 6: LP Polishing ===
    print(f"\n=== Phase 6: LP Polishing ===")
    t = remaining_time()
    if t > 60 and n_best <= 1000:
        alloc = min(t * 0.1, 80.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        update_best(a_lp, "LP")

    # === Phase 7: Spectral Perturbation + Optimize ===
    print(f"\n=== Phase 7: Spectral Perturbation ===")
    n_perturb = max(1, int(remaining_time() / 60.0))
    n_perturb = min(n_perturb, 8)

    for p in range(n_perturb):
        t = remaining_time()
        if t < 30:
            break
        strength = 0.1 + 0.05 * p
        a_pert = spectral_perturb(best_seq, rng, strength=strength)
        alloc = min(t * 0.15, 50.0)
        a_opt = lse_gradient_descent(a_pert, time_budget=alloc)
        update_best(a_opt, f"SpecPerturb-{p}")

    # === Phase 8: Multi-n Exploration ===
    print(f"\n=== Phase 8: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = [n_curr + 100, n_curr + 200, n_curr + 400, n_curr - 100,
                 1000, 1200, 1500, 2000]
    n_options = sorted(set([n for n in n_options if 400 <= n <= 2000 and n != n_curr]))
    n_options = n_options[:5]

    for n_try in n_options:
        t = remaining_time()
        if t < 40:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(t * 0.15, 50.0)
        a_opt = lse_gradient_descent(a_try, time_budget=alloc)
        update_best(a_opt, f"n={n_try}")

    # === Phase 9: Random Restarts ===
    print(f"\n=== Phase 9: Random Restarts ===")
    n_res = len(best_seq)
    n_restarts = max(1, int(remaining_time() / 50.0))
    n_restarts = min(n_restarts, 6)

    for r in range(n_restarts):
        t = remaining_time()
        if t < 30:
            break
        a_r = random_init(n_res, rng)
        alloc = min(t * 0.15, 40.0)
        a_opt = lse_gradient_descent(a_r, time_budget=alloc)
        update_best(a_opt, f"Restart-{r}")

    # === Phase 10: Final Polish (combined) ===
    print(f"\n=== Phase 10: Final Polish ===")
    t = remaining_time()
    if t > 30:
        alloc = min(t * 0.3, 100.0)
        a_final = lse_gradient_descent(best_seq, time_budget=alloc * 0.5, verbose=True)
        update_best(a_final, "Final-LSE")

    t = remaining_time()
    if t > 20 and len(best_seq) <= 1000:
        alloc = min(t * 0.3, 40.0)
        a_lp = lp_polish(best_seq, time_budget=alloc)
        update_best(a_lp, "Final-LP")

    t = remaining_time()
    if t > 20:
        alloc = min(t * 0.3, 30.0)
        a_sg = subgradient_chase(best_seq, time_budget=alloc)
        update_best(a_sg, "Final-Subgrad")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
