import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def simplex_project(v):
    n = len(v)
    v_sum = np.sum(v)
    if v_sum <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build A matrix
    A = build_conv_matrix(a)
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in ['MOSEK', 'CLARABEL', 'SCS', 'ECOS', 'PDLP']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass
    
    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def golden_section_line_search(a, g, obj_fn, tol=1e-10, max_iter=30):
    """Find t in [0,1] that minimizes obj((1-t)*a + t*g)."""
    a = a.astype(np.float64)
    g = g.astype(np.float64)
    
    def line_obj(t):
        a_t = (1 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        return obj_fn(a_t)
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # ~0.618
    
    a_lo, a_hi = 0.0, 1.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = line_obj(c)
    fd = line_obj(d)
    
    for _ in range(max_iter):
        if a_hi - a_lo < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = line_obj(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = line_obj(d)
    
    t_opt = (a_lo + a_hi) / 2
    return t_opt, line_obj(t_opt)

    n = len(a)

        return a
    a = a / s

    stall_count = 0

        if elapsed > time_budget:

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:

        # Golden section line search

        if obj_opt < best_obj - 1e-10:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = obj_opt
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and it % 20 == 0:

        else:
            stall_count += 1
            if stall_count > 20:

                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0

    return best_a

def smooth_max_gradient(a, time_budget=40.0, T_init=0.001, T_min=0.00001):
    """Gradient descent with smooth max."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        
        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0

                a = best_a.copy()
                v = np.zeros(n)

def resample_sequence(a, new_n):
    n = len(a)

    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def golden_line_search(a, g, max_iter=40):
    """Golden section search for optimal blend parameter t in [0,1]."""
    def line_obj(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        return compute_obj(a_t / s)
    
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = line_obj(c)
    fd = line_obj(d)
    
    for _ in range(max_iter):
        if hi - lo < 1e-8:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = line_obj(c)

            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = line_obj(d)

LP-alternating optimizer with golden-section line search (extended range),
dual LP directions, smooth-max gradient polish, multi-n exploration,
and adaptive restarts. Designed to push below 1.5027.

import numpy as np
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            x_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(x_val))
            if s > 1e-10:
                return x_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def golden_line_search(a, g, t_max=2.0, max_iter=50):
    """Golden section search for optimal blend t in [0, t_max]."""
    def line_obj(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        return compute_obj(a_t / s)

    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, t_max
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = line_obj(c)
    fd = line_obj(d)

    for _ in range(max_iter):
        if hi - lo < 1e-9:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = line_obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = line_obj(d)

    t_opt = (lo + hi) / 2.0
    return t_opt, line_obj(t_opt)


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP alternating with extended-line-search and dual LP directions."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        # Primary LP: fix a, optimize g
        g_lp, M_lp = solve_lp(a, time_budget=20.0)
        if g_lp is None:
            break

        candidates = [g_lp]

        # Dual LP: fix g_lp, optimize a direction
        if time.monotonic() - start < time_budget - 5:
            a_dual, M_dual = solve_lp(g_lp, time_budget=20.0)
            if a_dual is not None and not np.allclose(a_dual, a, atol=0.01):
                candidates.append(a_dual)

        best_line_obj = best_obj
        best_line_a = a

        for g_dir in candidates:
            t_opt, obj_opt = golden_line_search(a, g_dir, t_max=2.0)
            if obj_opt < best_line_obj - 1e-10:
                best_line_obj = obj_opt
                a_new = (1.0 - t_opt) * a + t_opt * g_dir
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                best_line_a = a_new

        if best_line_obj < best_obj - 1e-10:
            best_obj = best_line_obj
            best_a = best_line_a.copy()
            a = a_new
            stall = 0
            if verbose and it % 30 == 0:
                print(f"        LP it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall >= 12:
                rng = np.random.default_rng((it * 7919 + n * 104729 + stall) % (2**32))
                scale = 0.01 + 0.008 * min(stall // 12, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def smooth_max_grad(a, time_budget=40.0, T_init=0.001, T_min=0.000005):
    """Smooth-max (log-sum-exp) gradient descent with momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    lr = 0.008
    mu = 0.85
    v = np.zeros(n)
    stall = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        mc = np.max(conv)
        shifted = np.clip((conv - mc) / T, -500, 500)
        ev = np.exp(shifted)
        w = ev / np.sum(ev)

        grad = np.correlate(w, a, mode='valid')
        grad_c = grad - np.mean(grad)

        v = mu * v - lr * grad_c
        a_new = simplex_project(a + v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 200 == 0 and lr < 0.04:
                lr = min(lr * 1.03, 0.04)
        else:
            stall += 1
            if stall > 50:
                lr *= 0.7
                if lr < 1e-8:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 80:
            T = max(T_min, T_init * (0.997 ** it))

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(budget_s=1500):
    """
    Multi-phase optimizer: LP alternating with dual directions and extended
    line search, smooth-max gradient, multi-n exploration, final polish.
    """
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, None)
        if p.size > 0 and np.isfinite(p).all() and float(np.sum(p)) > 0.01:
            p = p / np.sum(p)
            best_score = compute_obj(p)
            best_seq = p.copy()
            print(f"Parent n={len(p)}: obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Init n=800: obj={best_score:.8f}")

    # === Phase 1: Intensive LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={len(best_seq)}) ===")
    a = best_seq.copy()
    for rd in range(5):
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        alloc = min(rem * 0.28, 200.0 + 50.0 * rd)
        a_new = lp_alternating(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP rd{rd}: obj={best_score:.8f}")
        else:
            print(f"  LP rd{rd}: obj={obj:.8f} (stalled)")
            if rd >= 3:
                break

    # === Phase 2: Smooth Max Gradient (multiple temperatures) ===
    print(f"\n=== Phase 2: Smooth Max Gradient ===")
    for T_i in [0.002, 0.0005, 0.0001]:
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        a_sm = smooth_max_grad(a, time_budget=min(35.0, rem * 0.25), T_init=T_i)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            a = a_sm.copy()
            print(f"  SmoothMax T={T_i}: obj={best_score:.8f}")
        else:
            print(f"  SmoothMax T={T_i}: obj={obj_sm:.8f}")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_best = len(best_seq)
    candidates = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100), max(300, n_best - 50),
        max(300, n_best - 25), n_best,
        min(1500, n_best + 25), min(1500, n_best + 50), min(1500, n_best + 100),
        min(1500, n_best + 200),
    ]))
    candidates.sort(key=lambda x: abs(x - n_best))

    for n_try in candidates:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        a_try = best_seq.copy() if n_try == n_best else resample(best_seq, n_try)
        alloc_lp = min(rem * 0.14, 18.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        rem = deadline - time.monotonic()
        if rem > 12 and obj_try < best_score + 0.005:
            a_opt2 = smooth_max_grad(a_opt, time_budget=min(8.0, rem * 0.15), T_init=0.0003)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt, obj_try = a_opt2, obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Final Intensive Polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    a_final = best_seq.copy()
    for pr in range(6):
        rem = deadline - time.monotonic()
        if rem < 12:
            break
        alloc = min(rem * 0.18, 30.0)
        a_lp = lp_alternating(a_final, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Polish {pr}: obj={best_score:.8f}")
        else:
            print(f"  Polish {pr}: obj={obj_lp:.8f} (stop)")
            break

    rem = deadline - time.monotonic()
    if rem > 10:
        a_sm = smooth_max_grad(a_final, time_budget=min(12.0, rem * 0.4), T_init=0.0003)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  FinalSmooth: obj={best_score:.8f}")

    # Validate
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - t0
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
