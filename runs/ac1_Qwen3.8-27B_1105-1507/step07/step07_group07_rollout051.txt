Optimized search for low-autocorrelation non-negative sequences.
Uses power-norm smoothing, LP alternating, subgradient polishing,
and multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def cross_correlate(w, a):
    """Compute (a ⋆ w)[i] = sum_j a[j] * w[j+i] for i=0..n-1 using FFT."""
    n = len(a)
    size = next_pow2(n + len(w) - 1)
    A = rfft(a, size)
    W = rfft(w, size)
    corr_full = irfft(W * np.conj(A), size)
    return corr_full[:n]


def power_obj_and_grad(a, p):
    """
    Power-norm smoothed objective and gradient.
    F(a) = (sum_k conv[k]^p)^(1/p) / (sum(a))^2
    """
    n = len(a)
    s = float(np.sum(a))
    conv = compute_conv(a)
    
    # Power norm of conv
    if p == 1:
        power_norm = float(np.sum(conv))
        grad_weight = np.ones(2 * n - 1)
        prefactor = 1.0
    else:
        power_norm = float(np.sum(conv**p)) ** (1.0/p)
        grad_weight = conv ** (p - 1)
        sum_wp = float(np.sum(grad_weight))
        prefactor = sum_wp ** (1.0/p - 1.0)
    
    F = power_norm / (s * s)
    
    # Gradient: dF/da[i] = prefactor * corr[i] / s^2 - 2*F*a[i]/s
    # where corr[i] = sum_k grad_weight[k] * a[k-i] (valid indices)
    # = cross_correlate(grad_weight, a)[i]
    corr = cross_correlate(grad_weight, a)
    grad = prefactor * corr / (s * s) - 2.0 * F * a / s
    
    return F, grad


def gradient_optimize_power(a_init, time_budget=60.0, verbose=False):
    """Optimize using L-BFGS-B with power-norm smoothing and adaptive p."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    p_list = [2, 5, 10, 20, 50, 100, 200, 500]
    
    x = np.log(a)  # work in log-space for non-negativity
    
    for p in p_list:
        if time.monotonic() - start > time_budget * 0.8:
            break
        
        def fun(xv):
            xv = xv - np.max(xv)
            av = np.exp(xv)
            return power_obj_and_grad(av, p)[0]

            av = np.exp(xv)
            F, g = power_obj_and_grad(av, p)
            # Chain rule: dF/dx = dF/da * da/dx = dF/da * a
            return g * av

def build_conv_matrix(a):

        A[j:j + r, j] = a[:r]
    return A

        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def subgradient_step(a, top_k=5):
    """
    Subgradient step on the true max objective.
    Uses top-k lags for a smoother subgradient.
    Returns new a (normalized).
    """
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    
    # Get top-k lags
    k_vals = np.argsort(conv)[-top_k:]
    weights = np.zeros(2 * n - 1)
    for k in k_vals:
        weights[k] = conv[k]
    w_sum = np.sum(weights)
    if w_sum > 0:
        weights = weights / w_sum
    
    corr = cross_correlate(weights, a)
    return corr


def subgradient_optimize(a_init, time_budget=30.0, top_k=5):
    """Nesterov-accelerated subgradient method."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    a_avg = a.copy()
    t = 1
    g_prev = np.zeros(n)
    
    while time.monotonic() - start < time_budget:
        a = np.maximum(a, 1e-10)
        a = a / np.sum(a)
        
        g = subgradient_step(a, top_k)
        g = g - np.sum(g) * a / np.sum(a)  # project onto tangent space
        
        # Nesterov momentum
        g_mom = 0.5 * (g + g_prev)
        g_prev = g_mom
        
        # Step size
        step = 1.0 / np.sqrt(t)
        a = a - step * g_mom
        a = np.maximum(a, 1e-10)
        a = a / np.sum(a)
        
        a_avg = a_avg + (a - a_avg) / (t + 1)
        t += 1
        
        if t % 100 == 0:
            obj = compute_obj(a_avg)
            if obj < best_obj:
                best_obj = obj
                best_a = a_avg.copy()
    
    obj = compute_obj(a_avg)
    if obj < best_obj:
        best_obj = obj
        best_a = a_avg.copy()
    
    return best_a


def lp_alternate_one_step(a):
    """One step of LP alternating."""
    n = len(a)
    if n > 500:
        # Skip for large n
        return a, compute_obj(a)
    
    a_norm = a / np.sum(a)
    g_lp, M_lp = solve_lp_minpeak(a_norm)
    if g_lp is None:
        return a, compute_obj(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_try = (1.0 - t) * a_norm + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a, best_obj


def make_tilted_uniform(n, alpha=0.0, rng=None):
    """Create a tilted uniform sequence."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(-1, 1, n)
    a = 1.0 + alpha * x
    if alpha != 0:
        a = a + 0.01 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def make_spectral_init(n, rng):
    """Initialize from a shaped spectrum."""
    size = next_pow2(n)
    # Create a spectrum with controlled shape
    freqs = np.arange(size // 2 + 1)
    # Low-pass with a shape parameter
    beta = rng.uniform(2, 10)
    spec = np.exp(-beta * (freqs / (size//2))**2)
    # Add random phase
    phase = rng.uniform(0, 2*np.pi, size // 2 + 1)
    # Take inverse FFT
    full_spec = np.zeros(size, dtype=complex)
    full_spec[:size//2+1] = spec * np.exp(1j * phase)
    full_spec[-(size//2):] = np.conj(full_spec[1:size//2][::-1]) if size > 2 else 0
    a = irfft(full_spec)[:n].real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)
