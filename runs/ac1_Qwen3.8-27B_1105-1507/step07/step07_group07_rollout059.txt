    idx = np.arange(n)

Multi-scale quadratic-penalty optimizer for low-autocorrelation non-negative sequences.
Key innovations over previous approach: (1) quadratic peak penalty with FFT-based
gradients for smooth optimization, (2) fast coordinate descent polishing, (3)
coarse-to-fine multi-scale initialization, (4) systematic n-exploration with
interpolation-based transfer.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def quad_penalty_obj_grad(x, T, lam):
    """
    Quadratic penalty: sum_k max(0, conv[k]-T)^2 + lam * sum_k conv[k]^2
    with a = exp(x), normalized. Returns (value, grad_wrt_x).
    """
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    a = a / s

    conv = compute_conv(a)
    n_conv = 2 * n - 1

    # Quadratic penalty on peaks above T
    w = np.maximum(conv - T, 0.0)
    val1 = float(np.sum(w * w))

    # Overall energy penalty
    val2 = lam * float(np.sum(conv * conv))

    total_val = val1 + val2

    # Gradient of val1: d/d a[i] sum_k w[k]^2 = 2 * sum_k w[k] * 2*a[k-i]
    # = 4 * sum_k w[k] * a[k-i] (with bounds)
    # This is 4 * cross_correlation(w, a_flipped)[i]
    a_flip = a[::-1]
    size = next_pow2(n_conv + n - 1)
    fw = rfft(w, size)
    faf = rfft(a_flip, size)
    corr1 = irfft(fw * faf, size)[:n]
    grad1 = 4.0 * corr1

    # Gradient of val2: d/d a[i] sum_k conv[k]^2 = 4 * sum_k conv[k] * a[k-i]
    fc = rfft(conv, size)
    corr2 = irfft(fc * faf, size)[:n]
    grad2 = 4.0 * lam * corr2

    grad_a = grad1 + grad2 - (grad1 + grad2).sum() * a / s  # project to sum-preserving

    # Chain rule: d/dx_i = a_i * d/d a_i
    grad_x = a * grad_a
    return total_val, grad_x


def quad_penalty_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Optimize using quadratic penalty with L-BFGS-B in log-parameterization.
    Sweeps T from high (near max) to low (near mean) for progressive flatness.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    a = a / np.sum(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    mean_c = float(np.mean(conv))

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # T schedule: from near max down to near mean
    n_temps = 8
    Ts = np.linspace(max_c * 0.95, mean_c * 1.05, n_temps)
    lams = [0.0] * n_temps
    lams[-3:] = [0.01, 0.005, 0.001]

    x = np.log(a) - np.log(a).max()

    for idx, (T, lam) in enumerate(zip(Ts, lams)):
        if time.monotonic() - start > time_budget:
            break

        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = quad_penalty_obj_grad(xv, T, lam)
            return val

            xv = xv - np.max(xv)
            _, g = quad_penalty_obj_grad(xv, T, lam)
            return g

        max_iter = max(50, int(200 * (time_budget - (time.monotonic() - start)) / time_budget))
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})

        if res.fun < fun(x) + 1e-12:
            x = res.x - res.x.max()
            a_new = np.exp(x)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    QP T={T:.6f} lam={lam}: obj={best_obj:.10f}")
            else:
                if verbose:
                    print(f"    QP T={T:.6f} lam={lam}: obj={obj_new:.10f} (no improve)")

    return best_a


def fast_coordinate_descent(a_init, sweeps=5, time_budget=30.0, verbose=False):
    """
    Fast coordinate descent: for each element, try a few candidate values
    and keep the best. Uses incremental conv update for speed.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = float(np.sum(a))
    if s < 1e-30:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(sweeps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        cur_obj = 2.0 * n * max_conv

        any_improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            a_i = a[i]
            if a_i < 1e-10:
                candidates = [0.0, 0.002 / n, 0.01 / n, 0.05 / n]
            else:
                spread = max(a_i * 0.4, 0.01 / n)
                candidates = [0.0, a_i * 0.5, a_i * 0.75, a_i * 1.25, a_i * 1.5, a_i + spread, a_i - spread]
                candidates = [max(0.0, c) for c in candidates]

            best_t = a_i
            best_local = compute_obj(a)

            for t in candidates:
                if abs(t - a_i) < 1e-12:
                    continue
                a_try = a.copy()
                a_try[i] = t
                s_try = float(np.sum(a_try))
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                c_try = compute_conv(a_try)
                obj_try = 2.0 * n * float(np.max(c_try))
                if obj_try < best_local - 1e-14:
                    best_local = obj_try
                    best_t = t
                    any_improved = True

            if best_t != a_i:
                a[i] = best_t
                s = float(np.sum(a))
                a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    CD sweep {sweep}: obj={best_obj:.10f}")
        elif not any_improved:
            if verbose:
                print(f"    CD sweep {sweep}: converged")
            break

    return best_a


def smooth_gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Smoothed max (log-sum-exp) gradient with L-BFGS-B in log-parameterization.
    Multi-temperature schedule.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def smooth_obj_grad(x, tau):
        x = x - np.max(x)
        a_local = np.exp(x)
        s = float(np.sum(a_local))
        a_local = a_local / s
        conv = compute_conv(a_local)
        c_scaled = conv / tau
        c_max = np.max(c_scaled)
        exp_c = np.exp(c_scaled - c_max)
        sum_exp = np.sum(exp_c)
        w = exp_c / sum_exp
        logsumexp_val = c_max + np.log(sum_exp)
        F = logsumexp_val

        # Gradient: dF/dx_i = (2/tau) * sum_k w[k]*a[k-i] - F * a_i / s
        size = next_pow2(2 * n - 1)
        a_flip = a_local[::-1]
        faf = rfft(a_flip, size)
        fw = rfft(w, size)
        corr = irfft(fw * faf, size)[:n]

        grad = (2.0 / tau) * corr - F * a_local
        return F, grad

    conv0 = compute_conv(a)
    max_c = float(np.max(conv0))
    x = np.log(a) - np.log(a).max()

    taus = [max_c * 0.5, max_c * 0.3, max_c * 0.15, max_c * 0.08, max_c * 0.04, max_c * 0.02]

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break

        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smooth_obj_grad(xv, tau)
            return val

        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smooth_obj_grad(xv, tau)
            return g

        max_iter = max(30, int(100 * (time_budget - (time.monotonic() - start)) / time_budget))
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})

        if res.fun < fun(x) + 1e-12:
            x = res.x - res.x.max()
            a_new = np.exp(x)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    SG tau={tau:.6f}: obj={best_obj:.10f}")

    return best_a


def mirror_descent_fast(a_init, time_budget=30.0, verbose=False):
    """Fast exponentiated gradient with top-K weighting."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 2.0
    K = 5  # track top K peaks

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Top-K weighted gradient
        topk_idx = np.argsort(conv)[::-1][:K]
        topk_vals = conv[topk_idx]
        topk_max = topk_vals[0]
        # Weight by softmax of top-K values
        w_k = np.exp((topk_vals - topk_max) / (topk_max * 0.1 + 1e-10))
        w_k = w_k / w_k.sum()

        g = np.zeros(n)
        for wi, ki in zip(w_k, topk_idx):
            ki = int(ki)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    g[i] += wi * 2.0 * a[j]

        g_max = np.max(g)
        if g_max < 1e-15:
            break

        a_new = a * np.exp(-eta * g / g_max)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta *= 1.005
        else:
            eta *= 0.97
            if eta < 0.1:
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                eta = 1.0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def downsample(a, factor):
    """Downsample by averaging blocks."""
    n = len(a)
    new_n = n // factor
    if new_n < 1:
        return a.copy()
    a_trim = a[:new_n * factor]
    return a_trim.reshape(new_n, factor).mean(axis=1)


def spectral_init(n, rng):
    """
    Generate a candidate via random-phase flat-spectrum initialization.
    Take IFFT of flat magnitude with random phases, threshold to non-negative.
    """
    size = next_pow2(2 * n - 1)
    mag = np.ones(size // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    phase[0] = 0  # DC term real
    if size % 2 == 0:
        phase[-1] = 0  # Nyquist real
    spectrum = mag * np.exp(1j * phase)
    # Make it conjugate symmetric for real IFFT
    a_complex = irfft(spectrum, size)
    a = np.maximum(a_complex[:n], 0)
    s = float(np.sum(a))
    if s < 0.01:
        # Retry with different phases
        phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
        phase[0] = 0
        if size % 2 == 0:
            phase[-1] = 0
        spectrum = mag * np.exp(1j * phase)
        a_complex = irfft(spectrum, size)
        a = np.maximum(a_complex[:n], 0)
        s = float(np.sum(a))
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def structured_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Asymmetric ramp
        a = 1.0 - x + 0.1 * rng.normal(0, 1, n)
    elif mode == 1:
        # Two Gaussian bumps (asymmetric)
        c1, c2 = 0.3, 0.7
        sig1, sig2 = 0.15, 0.12
        a = np.exp(-((x - c1) ** 2) / (2 * sig1 ** 2)) + 0.8 * np.exp(-((x - c2) ** 2) / (2 * sig2 ** 2))
        a += 0.05 * rng.normal(0, 1, n)
    elif mode == 2:
        # Cosine window (asymmetric)
        a = np.cos(x * np.pi * 0.7) ** 2 + 0.3 * np.cos(x * np.pi * 1.3) ** 2
        a += 0.05 * rng.normal(0, 1, n)
    elif mode == 3:
        # Linear chirp amplitude
        freq = rng.uniform(2, 8)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a += 0.05 * rng.normal(0, 1, n)
    elif mode == 4:
        # Power law
        p = rng.uniform(0.5, 3.0)
        a = (1 - x) ** p
        a += 0.05 * rng.normal(0, 1, n)
    else:
        # Sigmoid blend
        c = rng.uniform(0.3, 0.7)
        k = rng.uniform(5, 20)
        a = 1 / (1 + np.exp(-k * (x - c)))
        a += 0.05 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.001)
    return a / np.sum(a)


def lp_alternate_one(a):
    """One LP alternating step: solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    # Build constraint matrix A where A @ g = conv(a_norm, g)
    # conv(a, g)[k] = sum_j a[j]*g[k-j]
    # For fixed a, this is linear in g.
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # But we only need constraints for k = 0, ..., 2n-2

    from scipy.optimize import linprog

    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = a_norm[j]
        A_ub[k, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 5})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_polish(a, time_budget=30.0, verbose=False):
    """LP alternating with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        g_lp = lp_alternate_one(a)
        if g_lp is None:
            break

        improved = False
        for t in [0.5, 0.7, 0.9, 1.0, 0.3, 0.8]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimizer combining quadratic penalty gradients,
    coordinate descent, mirror descent, smooth gradient, and LP polishing.
    Explores multiple sequence lengths with multi-scale initialization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)

    def try_optimize(a, n_label, time_budget, methods=None):
        """Apply a sequence of optimization methods and return best result."""
        nonlocal best_seq, best_score
        if methods is None:
            methods = ['quad', 'smooth', 'cd', 'mirror']

        a_cur = np.maximum(a, 0).copy()
        a_cur = a_cur / np.sum(a_cur)
        local_best = a_cur.copy()
        local_best_obj = compute_obj(a_cur)

        for m in methods:
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            alloc = min(remaining * 0.3, time_budget)
            if alloc < 3:
                break

            if m == 'quad':
                a_new = quad_penalty_optimize(a_cur, time_budget=alloc, verbose=True)
            elif m == 'smooth':
                a_new = smooth_gradient_optimize(a_cur, time_budget=alloc, verbose=True)
            elif m == 'cd':
                a_new = fast_coordinate_descent(a_cur, sweeps=5, time_budget=alloc, verbose=True)
            elif m == 'mirror':
                a_new = mirror_descent_fast(a_cur, time_budget=alloc, verbose=True)
            elif m == 'lp':
                a_new = lp_polish(a_cur, time_budget=alloc, verbose=True)
            else:
                continue

            obj_new = compute_obj(a_new)
            if obj_new < local_best_obj - 1e-12:
                local_best_obj = obj_new
                local_best = a_new.copy()
                a_cur = a_new

        if local_best_obj < best_score:
            best_score = local_best_obj
            best_seq = local_best.copy()
            print(f"  ** {n_label}: obj={best_score:.10f} **")
        else:
            print(f"  {n_label}: best={local_best_obj:.10f} (no global improve)")

        return local_best

    # === Phase 1: Intensive optimization on parent n ===
    print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_opt = try_optimize(best_seq, f"parent_n={n_parent}", min(remaining * 0.3, 180),
                             methods=['quad', 'smooth', 'cd'])

    # === Phase 2: Quadratic penalty focused refinement ===
    print(f"\n=== Phase 2: Quadratic Penalty Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_q = quad_penalty_optimize(best_seq, time_budget=min(remaining * 0.2, 120), verbose=True)
        obj_q = compute_obj(a_q)
        if obj_q < best_score:
            best_score = obj_q
            best_seq = a_q.copy()
            print(f"  QP refine: obj={best_score:.10f}")

        # Coordinate descent polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_cd = fast_coordinate_descent(best_seq, sweeps=10, time_budget=min(remaining * 0.15, 60), verbose=True)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  CD polish: obj={best_score:.10f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_candidates = sorted(set([
        max(200, n_parent - 200), max(200, n_parent - 100),
        n_parent, n_parent + 50, n_parent + 100, n_parent + 200,
        400, 600, 900, 1100, 1300, 1600, 2000
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 2500]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.12, 60)
        a_opt = try_optimize(a_try, f"n={n_try}", alloc, methods=['quad', 'cd'])

    # === Phase 4: Diverse initializations at best n ===
    print(f"\n=== Phase 4: Diverse Inits (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    n_diverse = max(3, int(remaining / 40))
    n_diverse = min(n_diverse, 8)

    for d in range(n_diverse):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if d == 0:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.04, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif d == 1:
            # Spectral init
            a_r = spectral_init(n_best, rng)
        elif d == 2:
            a_r = structured_init(n_best, rng, mode=0)
        elif d == 3:
            a_r = structured_init(n_best, rng, mode=1)
        elif d == 4:
            a_r = structured_init(n_best, rng, mode=2)
        elif d == 5:
            # Blend of best and uniform
            a_r = 0.7 * best_seq + 0.3 * np.ones(n_best) / n_best
            a_r = a_r / np.sum(a_r)
        elif d == 6:
            # Asymmetric smoothing of best
            kernel = np.exp(-np.arange(-10, 11) ** 2 / 20)
            kernel = kernel / kernel.sum()
            a_r = np.convolve(best_seq, kernel, mode='same')
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            a_r = structured_init(n_best, rng, mode=d % 6)

        alloc = min(remaining * 0.15, 35)
        a_opt = try_optimize(a_r, f"diverse_{d}", alloc, methods=['quad', 'smooth', 'cd'])

    # === Phase 5: Coarse-to-fine at best n ===
    print(f"\n=== Phase 5: Coarse-to-Fine (target n={n_best}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Start from a small n and build up
        scales = [16, 32, 64, 128, 256, 512, n_best]
        scales = [s for s in scales if s <= n_best]
        if n_best not in scales:
            scales.append(n_best)

        a_coarse = best_seq.copy()
        # Downsample to smallest scale
        a_cur = downsample(a_coarse, n_best // scales[0])
        if len(a_cur) != scales[0]:
            a_cur = resample_sequence(a_coarse, scales[0])

        for scale_n in scales[1:]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_cur = resample_sequence(a_cur, scale_n)
            alloc = min(remaining * 0.2, 20)
            a_cur = try_optimize(a_cur, f"coarse_n={scale_n}", alloc, methods=['quad', 'cd'])

        if compute_obj(a_cur) < best_score:
            best_score = compute_obj(a_cur)
            best_seq = a_cur.copy()
            print(f"  Coarse-to-fine: obj={best_score:.10f}")

    # === Phase 6: LP polish ===
    print(f"\n=== Phase 6: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_lp = lp_polish(best_seq, time_budget=min(remaining * 0.2, 60), verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.10f}")

    # === Phase 7: Final intensive CD + quad ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Multiple CD sweeps
        a_f = fast_coordinate_descent(best_seq, sweeps=20, time_budget=min(remaining * 0.3, 60), verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_f2 = quad_penalty_optimize(best_seq, time_budget=min(remaining * 0.3, 40), verbose=True)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final QP: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_f3 = fast_coordinate_descent(best_seq, sweeps=10, time_budget=min(remaining * 0.4, 30), verbose=True)
            obj_f3 = compute_obj(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final CD2: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
