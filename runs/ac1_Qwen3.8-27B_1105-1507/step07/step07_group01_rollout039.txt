def simplex_project(v):
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)

        s = float(np.sum(g))
        if s > 1e-10:

        return g, float(res.x[n])
    return None, None

Optimized LP alternating with smooth-max polishing and multi-n search.
Key improvements: faster LP via structured matrices, adaptive step sizes,
QP direction blending, and systematic n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def compute_conv_ffta(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_ffta(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_weights(conv, T):
    """Compute softmax weights for smooth max."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    return exp_vals / total

def compute_grad_smooth(weights, a):
    """Gradient of smooth max w.r.t. a. grad[i] = sum_j weights[i+j]*a[j]."""
    n = len(a)
    # grad[i] = sum_j a[j] * weights[i+j] for valid i+j
    # This is the cross-correlation
    grad = np.correlate(weights, a, mode='valid')
    return grad

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    
    # Build convolution matrix (sparse)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(2*n-1, j+n)
        for i in range(end - start):
            rows.append(start + i)
            cols.append(j)
            vals.append(a[i])
    
    A_conv = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP formulation
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # A_ub: [A_conv | -1] @ [g; M] <= 0
    # We'll build this as a dense matrix for HiGHS
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A_conv.toarray()
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)]*n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            g = g / s
        return g, float(res.x[n])
    return None, None

def smooth_max_descent(a, time_budget, T_init=0.001, T_min=0.00005):
    """Smooth max gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = T_init
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv_ffta(a)
        weights = smooth_max_weights(conv, T)
        grad = compute_grad_smooth(weights, a)
        grad_centered = grad - np.mean(grad)
        
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                lr *= 0.7
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-6:
                    break
        
        T = max(T_min, T_init * (0.999 ** (int((time.monotonic()-start)*10))))
    
    return best_a

def qp_direction_step(a, n_steps=50, lr=0.05):
    """Take steps in the direction that minimizes sum of squares of conv."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for _ in range(n_steps):
        conv = compute_conv_ffta(a)
        # Gradient of sum(conv^2) w.r.t. g is 2*A^T*conv
        # A^T*conv at index j = sum_k A[k,j]*conv[k] = sum_i a[i]*conv[j+i]
        grad = 2.0 * np.correlate(conv, a, mode='valid')
        grad_centered = grad - np.mean(grad)
        
        a_new = simplex_project(a - lr * grad_centered)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a

def resample_sequence(a, new_n):
    """Resample via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
Optimized LP alternating with smooth-max polishing, multiplicative blending,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv_ffta(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_ffta(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_weights(conv, T):
    """Compute softmax weights for smooth max approximation."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = float(np.sum(exp_vals))
    return exp_vals / total


def compute_grad_smooth(weights, a):
    """Gradient of smooth max w.r.t. a via cross-correlation."""
    return np.correlate(weights, a, mode='valid')


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    # Build dense convolution matrix A where A[k,j] = a[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]

    # LP: minimize M (last variable)
    # Variables: [g_0,...,g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            g = g / s
        return g, float(res.x[n])
    return None, None


def line_search_blend(a, g, best_obj, max_evals=15):
    """Adaptive line search for blending a and g. Returns (best_a, best_obj, best_t)."""
    n = len(a)

    best_o = best_obj
    best_t = 0.0

    improved = False
    for t in t_candidates:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_o - 1e-14:

            best_t = t

        for t in t_refine:
            if t <= 0 or t >= 1.5:
                continue
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))

            obj_try = compute_obj(a_try)

def line_search_mult_blend(a, g, best_obj):

def targeted_perturbation(a, strength=0.02):

def smooth_max_descent(a, time_budget, T_init=0.001, T_min=0.00003):

    momentum = 0.88

    step = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv_ffta(a)
        weights = smooth_max_weights(conv, T)
        grad = compute_grad_smooth(weights, a)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 25:
                lr *= 0.7
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-6:
                    break

        step += 1
        T = max(T_min, T_init * (0.998 ** step))

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

            best_seq = a_lp2.copy()
            print(f"  LP2: obj={best_score:.10f}")
        else:
            print(f"  LP2: obj={obj_lp2:.10f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n{'='*50}")
    print(f"Phase 6: Multi-n Exploration")
    print(f"{'='*50}")
    n_current = len(best_seq)
    n_options = sorted(set([
        n_current - 100, n_current - 50, n_current - 25,
        n_current, n_current + 25, n_current + 50, n_current + 100,
        n_current + 150,
        600, 650, 700, 750,
    ]))
    n_options = [x for x in n_options if 400 <= x <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP
        alloc = min(remaining * 0.08, 20.0)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        # Smooth max polish if promising
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.003:
            alloc_sm = min(remaining * 0.05, 10.0)
            a_opt2 = smooth_max_descent(a_opt, time_budget=alloc_sm, T_init=0.0005, T_min=0.00002)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n{'='*50}")
    print(f"Phase 7: Final Polish")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP polish
        alloc = min(60.0, remaining * 0.4)
        a_final = lp_alternate(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.10f}")

        # Smooth max final
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sm = smooth_max_descent(best_seq, time_budget=min(30.0, remaining * 0.3),
                                      T_init=0.0002, T_min=0.00001)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Final SM: obj={best_score:.10f}")

        # One more LP on final result
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_lp = lp_alternate(best_seq, time_budget=min(20.0, remaining * 0.5))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Very Final LP: obj={best_score:.10f}")

    # === Final validation ===
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.array(result))

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL RESULT: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*50}")
    return result
