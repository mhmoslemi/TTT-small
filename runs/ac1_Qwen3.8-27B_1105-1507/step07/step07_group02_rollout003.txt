```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_max(conv, beta):
    """Smooth approximation to max using log-sum-exp."""
    m = np.max(conv)
    if beta < 1:
        return m
    scaled = beta * (conv - m)
    scaled = np.clip(scaled, -50, 0)
    return m + np.log(np.sum(np.exp(scaled))) / beta

def gradient_smooth_max(conv, a_norm, beta):
    """Gradient of smooth max of conv w.r.t. a_norm."""
    n = len(a_norm)
    m = np.max(conv)
    scaled = beta * (conv - m)
    scaled = np.clip(scaled, -50, 0)
    weights = np.exp(scaled)
    weights = weights / np.sum(weights)
    
    # grad[i] = 2 * sum_k weights[k] * a_norm[k-i] for valid k-i
    # = 2 * sum_{j=0}^{n-1} weights[i+j] * a_norm[j] (for i <= n-1)
    grad = np.zeros(n)
    for i in range(n):
        valid = slice(i, min(2*n-1, i+n))
        w_valid = weights[valid]
        a_valid = a_norm[:len(w_valid)]
        grad[i] = 2.0 * np.dot(w_valid, a_valid)
    
    # Center to preserve sum constraint (project onto sum=1 hyperplane)
    grad = grad - np.mean(grad)
    return grad
```

FISTA for our problem:
- We have objective f(a) = smooth_max(conv(a)) + indicator_{a>=0, sum(a)=1}(a)
- The proximal operator of the indicator is: project onto {a >= 0, sum(a) = 1}
- The prox of "non-negative and sum=1" is: sort the values, find the threshold, clip and renormalize

Actually, the projection onto {a >= 0, sum(a) = 1} is the "water-filling" algorithm:
1. Start with y = a - t*grad
2. Find the largest lambda such that sum(max(y - lambda, 0)) = 1
3. Return max(y - lambda, 0)

This can be done in O(n log n) using sorting.

```python
def project_nonneg_sum1(y):
    """Project y onto {a >= 0, sum(a) = 1}."""
    n = len(y)
    # Sort in descending order
    y_sorted = np.sort(y)[::-1]
    # Find the largest k such that y_sorted[k] > 1 - cumsum(y_sorted[:k])/k
    cumsum = np.cumsum(y_sorted)
    rho = np.arange(1, n+1)
    condition = y_sorted > (1 - cumsum) / rho
    k = np.sum(condition)
    if k == 0:
        return np.ones(n) / n
    theta = (cumsum[k-1] - 1) / k
    return np.maximum(y - theta, 0)

def fista_optimize(a, time_budget, beta_start=50, beta_end=2000, verbose=False):
    """FISTA optimization for smooth max of auto-convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t_k = 1.0
    a_old = a.copy()
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Adaptive beta
        progress = min(1.0, it / 200)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute gradient
        a_norm = a / np.sum(a)
        conv = compute_conv(a_norm)
        obj_smooth = smooth_max(conv, beta)
        
        grad = gradient_smooth_max(conv, a_norm, beta)
        
        # FISTA step
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        # Momentum
        y = a + (t_k - 1) / t_new * (a - a_old)
        y = np.maximum(y, 0)
        sy = np.sum(y)
        if sy > 0:
            y = y / sy
        
        # Step size (using Lipschitz constant estimate)
        # The Lipschitz constant of the gradient is roughly 4*beta*max(a) or so
        L = 4.0 * beta * np.max(a) + 1e-6
        step = 1.0 / L
        
        # Backtracking line search
        a_new = project_nonneg_sum1(y - step * grad)
        for _ in range(10):
            obj_new = smooth_max(compute_conv(a_new), beta)
            if obj_new <= obj_smooth + 0.5 * step * np.dot(grad, a_new - y):
                break
            step *= 0.5
            a_new = project_nonneg_sum1(y - step * grad)
        
        a_old = a
        a = a_new
        t_k = t_new
        it += 1
        
        # Update best (using exact objective)

        if exact_obj < best_obj:

            best_a = a.copy()
            if verbose and it % 20 == 0:
                print(f"    FISTA it{it}: obj={best_obj:.8f}, beta={beta:.0f}")
    
    return best_a
```

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses FISTA with adaptive smoothing, mirror descent, block coordinate descent,
multi-start exploration with parametric initializations, and multi-n search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a is normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def smooth_max(conv, beta):
    """Smooth approximation to max using log-sum-exp."""
    m = np.max(conv)
    if beta < 1:
        return m
    scaled = beta * (conv - m)
    scaled = np.clip(scaled, -50, 0)
    return m + np.log(np.sum(np.exp(scaled))) / beta


def gradient_smooth_max(conv, a_norm, beta):
    """Gradient of smooth max of auto-convolution w.r.t. a_norm. O(n^2)."""
    n = len(a_norm)
    m = np.max(conv)
    scaled = beta * (conv - m)
    scaled = np.clip(scaled, -50, 0)
    weights = np.exp(scaled)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros_like(conv)
        weights[int(np.argmax(conv))] = 1.0
        wsum = 1.0
    weights = weights / wsum

    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a_norm[j]
    grad = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        if k_start < 2 * n - 1:
            grad[i] = 2.0 * np.dot(weights[k_start:k_end], a_norm[:k_end - k_start])

    # Project onto sum=0 (to maintain normalization)
    grad = grad - np.mean(grad)
    return grad


def project_nonneg_sum1(y):
    """Project y onto {a >= 0, sum(a) = 1} via water-filling."""
    n = len(y)
    y_sorted = np.sort(y)[::-1]
    cumsum = np.cumsum(y_sorted)
    rho = np.arange(1, n + 1, dtype=np.float64)
    condition = y_sorted > (1.0 - cumsum) / rho
    k = int(np.sum(condition))
    if k == 0:
        return np.ones(n) / n
    theta = (cumsum[k - 1] - 1.0) / k
    return np.maximum(y - theta, 0.0)


def fista_optimize(a, time_budget, beta_start=30, beta_end=3000, verbose=False):
    """FISTA with adaptive smoothing for smooth max of auto-convolution."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t_k = 1.0
    a_old = a.copy()
    it = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, it / 300.0)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.7)

        a_norm = a / np.sum(a)
        conv = compute_conv(a_norm)
        obj_smooth = smooth_max(conv, beta)
        grad = gradient_smooth_max(conv, a_norm, beta)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + (t_k - 1.0) / t_new * (a - a_old)
        y = np.maximum(y, 0.0)
        sy = np.sum(y)
        if sy > 1e-10:
            y = y / sy

        a = a_new
        t_k = t_new
