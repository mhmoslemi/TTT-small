def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Try MOSEK first (fastest for this size)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

            s = float(np.sum(g_val))
            if s > 1e-10:

    except Exception:
        pass
    return None, None


def golden_section_line_search(a, g, n_iters=30):
    """
    Find optimal t in [0,1] minimizing obj((1-t)*a + t*g) using golden section search.
    Both a and g are assumed normalized (sum=1, non-negative).
    """
    def f(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)
    
    # Golden section search on [0, 1]
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = 1.0 - gr
    d = gr
    fc = f(c)
    fd = f(d)
    
    for _ in range(n_iters):
        if fc < fd:
            d = c
            fd = fc
            c = 1.0 - gr * (1.0 - c)
            fc = f(c)
        else:
            c = d
            fc = fd
            d = gr * (d)
            fd = f(d)
    
    t_opt = (c + d) / 2.0
    a_opt = (1.0 - t_opt) * a + t_opt * g
    a_opt = np.maximum(a_opt, 0)
    s = np.sum(a_opt)
    if s > 1e-10:
        a_opt = a_opt / s
    return a_opt, f(t_opt), t_opt

def smoothed_obj_and_grad(theta, tau, n):
    """Compute smoothed objective and gradient w.r.t. log-parameters theta."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    a = exp_t / np.sum(exp_t)
    
    conv = compute_conv(a)
    conv_max = np.max(conv)
    obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)
    
    # Soft-argmax weights
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)
    
    # c_j = sum_m w[j+m] * a[m]
    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < 2 * n - 1:
            c[j] = np.dot(w[j:end], a[:end - j])
    
    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta

            _, g = smoothed_obj_and_grad(th, tau, n)
            return g

def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_perturbation(a, rng, strength=0.1):
    """Apply random perturbation to escape local minima."""
    n = len(a)
    # Multiplicative noise
    noise = np.exp(rng.normal(0, strength, n))
    a_new = a * noise
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def structural_perturbation(a, rng):
    """Apply structural perturbation: zero out some elements, shift mass."""
    n = len(a)
    a_new = a.copy()
    
    # Randomly zero out 5-15% of elements
    n_zero = int(n * rng.uniform(0.05, 0.15))
    zero_idx = rng.choice(n, size=n_zero, replace=False)
    mass = np.sum(a_new[zero_idx])
    a_new[zero_idx] = 0
    
    # Redistribute mass randomly
    if mass > 0:
        targets = rng.choice(n, size=min(n_zero, n), replace=False)
        for i, t in enumerate(targets):
            a_new[t] += mass / len(targets)
    
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        obj_opt = compute_obj(a_opt)
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Init {i} (n={n_best}): obj={best_score:.8f} **")
        else:
            print(f"  Init {i}: obj={obj_opt:.8f}")
    
    # Phase 7: Final intensive polish on best
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    
    if remaining > 40:
        # LP with more iterations
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_alternating_improved(best_seq, time_budget=alloc, use_golden=True, max_iter=1000)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Directed peak reduction
        alloc = min(remaining * 0.2, 30.0)
        a_dp = directed_peak_reduction(best_seq, time_budget=alloc)
        obj_dp = compute_obj(a_dp)
        if obj_dp < best_score:
            best_score = obj_dp
            best_seq = a_dp.copy()
            print(f"  Final peak red: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more smooth pass
        alloc = min(remaining * 0.3, 30.0)
        a_opt, obj_opt = smooth_lbfgs(best_seq, tau_start=50.0, tau_end=1000.0,
                                       time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final smooth: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Final LP polish
        alloc = min(remaining * 0.4, 40.0)
        a_lp = lp_alternating_improved(best_seq, time_budget=alloc, use_golden=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    # Ensure valid output
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    
    result = [float(x) for x in best_seq]
    
    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
        print(f"  evaluate_sequence gave better: {best_score:.8f}")
    elif not np.isfinite(val):
        # Fallback
        result = [1.0] * 100
        best_score = 2.0
        print("  WARNING: Invalid result, using fallback")
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
