import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_toeplitz_conv(a):
    """Build matrix T (2n-1 x n) such that T @ g = conv(a, g)."""
    n = len(a)
    # T[k, j] = a[k-j] if 0 <= k-j < n else 0
    # Efficient construction using the fact that each column is a shifted version of a
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Column j: T[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        start = j
        end = min(2 * n - 1, j + n)
        T[start:end, j] = a[:end - start]
    return T


def solve_lp_minpeak(a, timeout_s=5.0):
    """
    Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    
    T = build_toeplitz_conv(a_norm)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize: M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: T @ g - M <= 0 for all k
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    start = time.monotonic()
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and time.monotonic() - start < timeout_s:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs, float(res.x[-1])
    
    return None, None

def momentum_gradient_descent(a, time_budget=60.0, verbose=False):
    """
    Subgradient descent with Nesterov momentum for minimizing the peak.
    The subgradient of max(conv(a,a)) w.r.t. a[i] is 2*a[k*-i] (if valid).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov lookahead
        a_look = a - beta * momentum
        a_look = np.maximum(a_look, 0)
        s_look = float(np.sum(a_look))
        if s_look > 1e-10:
            a_look = a_look / s_look
            conv = compute_conv(a_look)
        else:
            conv = compute_conv(a)
        
        k_star = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a_look[j]
        
        # Center to preserve sum
        grad = grad - np.mean(grad)
        
        # Update
        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.95
            a = best_a.copy()  # revert
            if lr < 1e-6:
                break
    
    return best_a
