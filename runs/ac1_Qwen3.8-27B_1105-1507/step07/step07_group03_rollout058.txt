- t[k] = c * exp(-(k-(n-1))²/(2σ²)) (Gaussian)

The optimal target is the one that's "closest" to achievable given the constraints a≥0, sum(a)=1.

Actually, I think the flat target is still the best because it directly minimizes the max. The issue is that the GS projection doesn't converge well. My softmax gradient approach should handle this better.

Let me also consider: what if I optimize in the frequency domain directly? The constraint a≥0 becomes a constraint on the inverse DFT. The objective max_k conv[k] becomes a constraint on the inverse DFT of |FFT(a)|² (with phase).

One more important consideration: **step size selection** for gradient descent. The objective is not smooth (it's the max of bilinear forms), so standard gradient descent theory doesn't directly apply. I'll use:
- Backtracking line search: try step α, if objective increases, try α/2, etc.
- Adaptive α: start large, decrease when stuck

"""
Optimization of 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive softmax-smoothed gradient descent with FFT-based gradients
2. Gerchberg-Saxton flat spectrum projection
3. LP with reduced active constraints
4. Multi-n exploration and random restarts
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def softmax_gradient_fft(a, beta):
    """Compute gradient of softmax-smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    weights = np.exp(beta * (conv - max_conv))
    weights /= weights.sum()
    
    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    # = 2 * (weights * a_reversed)[i + n - 1]
    b = a[::-1]
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    conv_result = irfft(rfft(weights, size) * rfft(b, size), size)
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    
    return grad, weights, max_conv, conv

def softmax_gradient_direct(a, beta):
    """Direct computation of softmax gradient (for small n or verification)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    weights = np.exp(beta * (conv - max_conv))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for j in range(n):
            k = i + j
            if k < 2 * n - 1:
                s += weights[k] * a[j]
        grad[i] = 2.0 * s
    
    return grad, weights, max_conv, conv

def gradient_descent_adaptive(a, time_budget=120.0, max_iters=5000):
    """Adaptive softmax gradient descent with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive beta: start high (focus on peak), decrease over time
    beta = 50.0
    lr = 0.5
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, weights, max_conv, conv = softmax_gradient_fft(a, beta)
        
        # Normalize gradient (it's already a subgradient of the smoothed objective)
        # The smoothed objective value is (1/beta)*log(sum exp(beta*conv[k]))
        # But we're minimizing the actual max, so we just use the gradient direction
        
        # Line search
        improved = False
        for attempt in range(15):
            step = lr / (1 + 0.1 * attempt)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                lr *= 0.5
                break
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                lr = min(lr * 1.05, 2.0)
                break
            elif obj_new < best_obj:
                # Accept smaller improvement
                a = a_new
                improved = True
                break
            else:
                lr *= 0.7
        
        if improved:
            stall = 0
        else:
            stall += 1
            lr *= 0.5
            if stall > 50:
                # Reset and re-focus
                beta = max(10.0, beta * 0.5)
                # Small perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
        
        # Slowly decrease beta
        if it % 100 == 0 and beta > 5.0:
            beta *= 0.95
    
    return best_a

def gs_flat_spectrum(a, n_iters=200, time_budget=60.0, relaxation=0.5):
    """Gerchberg-Saxton flat spectrum iteration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe
        
        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat
        
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    rng = np.random.default_rng(it)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
    
    return best_a

def lp_with_reduced_constraints(a, K=30, time_budget=60.0):
    """LP alternating with only top-K active constraints."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:

        top_idx = np.argsort(conv_a)[-K:]
        top_idx = sorted(top_idx.tolist())
        
        # Build reduced constraint matrix
        # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j
        # We need: for k in top_idx, conv(a, g)[k] <= M
        rows = []
        for k in top_idx:
            row = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    row[j] = a[idx]  # Wait, this is wrong
            # Actually conv(a, g)[k] = sum_j a[j] * g[k-j]
            # So the row for constraint k has row[j] = a[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
            row = np.zeros(n)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                row[j] = a[j]  # coefficient of g[k-j]... wait
            # conv(a,g)[k] = sum_j a[j] * g[k-j]
            # For a given k, the coefficient of g[i] is a[k-i] (when 0<=k-i<n)
            # So row[i] = a[k-i] if 0 <= k-i < n
            row = np.zeros(n)
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    row[i] = a[ki]
            rows.append(row)
        
        A_ub = np.array(rows)  # shape (K, n)
        
        # LP: min M s.t. A_ub @ g <= M, sum(g) = 1, g >= 0
        # Variables: g[0..n-1], M
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize M

        A_full[:, :n] = A_ub
        A_full[:, n] = -1.0  # A_ub @ g - M <= 0
        
        b_full = np.zeros(K)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    g_val = g_val / s_g

                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * a + t * g_val

                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)

                        best_obj = best_t_obj
                        a_new = (1 - best_t) * a + best_t * g_val
                        a_new = np.maximum(a_new, 0)
                        a_new = a_new / np.sum(a_new)
                        a = a_new
                        best_a = a_new.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        except Exception:

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

For a = [1, 2, 3], n=3:
conv = [1, 4, 10, 12, 9] (let me verify: 1*1=1, 1*2+2*1=4, 1*3+2*2+3*1=10, 2*3+3*2=12, 3*3=9) ✓

With beta large, max_conv = 12 (at k=3), weights ≈ [0, 0, 0, 1, 0] (mostly)

grad[0] = 2 * sum_j weights[0+j] * a[j] = 2 * (weights[0]*1 + weights[1]*2 + weights[2]*3) ≈ 0
grad[1] = 2 * sum_j weights[1+j] * a[j] = 2 * (weights[1]*1 + weights[2]*2 + weights[3]*3) ≈ 2*3 = 6
grad[2] = 2 * sum_j weights[2+j] * a[j] = 2 * (weights[2]*1 + weights[3]*2 + weights[4]*3) ≈ 2*2 = 4

So the gradient says: decrease a[1] and a[2], which makes sense since they contribute to the peak at k=3.

Let me verify with the FFT formula:
b = a[::-1] = [3, 2, 1]
weights (approximately) = [0, 0, 0, 1, 0]
(weights * b) linear convolution:
pos 0: 0*3 = 0
pos 1: 0*2+0*3 = 0
pos 2: 0*1+0*2+0*3 = 0
pos 3: 0*1+0*2+1*3 = 3
pos 4: 0*2+1*1+0*3 = 1
pos 5: 1*2+0*1 = 2

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def softmax_gradient_fft(a, beta):
    """
    Compute gradient of softmax-smoothed max of conv(a,a) using FFT.
    grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    where weights[k] = softmax(beta * conv[k])
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    weights = np.exp(beta * (conv - max_conv))
    weights /= weights.sum()

    # grad[i] = 2 * (weights * a_rev)[i + n - 1]
    b = a[::-1]
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    conv_result = irfft(rfft(weights, size) * rfft(b, size), size)
    grad = 2.0 * conv_result[n - 1:n - 1 + n]

    return grad, weights, max_conv, conv


def gradient_descent_adaptive(a, time_budget=120.0, max_iters=10000, use_mirror=True):

            if use_mirror:
                a_new = a * np.exp(-step * grad_norm)

def mirror_descent_multi(a, time_budget=120.0, max_iters=10000, m_peaks=5):
    """Mirror descent with multi-peak subgradient (exponential weights)."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

        top_idx = np.argsort(conv)[-m_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        temp = max(0.001, (max_val - top_val[0]) * 0.3)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        # Compute weighted subgradient via FFT

        # Mirror descent step
        grad_max = np.max(np.abs(grad)) + 1e-15

            alpha *= 0.95

        if it % 500 == 499:

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
