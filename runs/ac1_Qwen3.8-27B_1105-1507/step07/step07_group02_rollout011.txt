subgradient, and stochastic optimization with multi-n search.

from scipy.fft import next_fast_len, rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = next_fast_len(2*n - 1)
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = a/s
    conv = compute_conv(b)
    return 2.0*n*float(np.max(conv))

def solve_lp_for_g(a, max_time=5.0):
    """Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns g, M."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a/s

import numpy as np
from scipy.fft import next_fast_len, rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = a / s
    conv = compute_conv(b)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = max(0, 0)
        k_end = min(2 * n - 2, j + n - 1)
        if k_start <= k_end:
            indices = np.arange(k_start, k_end + 1)
            A[indices, j] = a[indices - j]
    return A


def solve_lp_for_g(a, max_time=10.0):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    # Try cvxpy first (faster for medium-sized problems)
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS]
        for solver in solvers:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def subgradient_step(a, alpha=0.1):
    """
    Take a subgradient step on the objective.
    Subgradient of max_k conv(a,a)[k] w.r.t. a is 2*R_{k*} a.
    Projected onto simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    b = a / s
    conv = compute_conv(b)
    k_star = int(np.argmax(conv))

    # Subgradient: grad_j = 2 * b[k_star - j] if 0 <= k_star - j < n
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * b[idx]

    # Center to preserve sum
    grad_centered = grad - np.mean(grad)

    a_new = b - alpha * grad_centered
    a_new = np.maximum(a_new, 0)
    sn = float(np.sum(a_new))
    if sn < 1e-10:
        return a
    return a_new / sn


def lp_alternating(a, time_budget=60.0, verbose=False, seed=0):
    """LP alternating optimization with adaptive step and perturbation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    last_t = 0.5
    max_iters = 200

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_for_g(a)
        if g_lp is None:
            break

        candidate_ts = [0.2, 0.35, 0.5, 0.7, 1.0]
        if 0 < last_t < 1:
            candidate_ts = sorted(set(candidate_ts + [max(0.05, last_t * 0.6), min(1.0, last_t * 1.4)]))

        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and it % 20 == 0:
                print(f"        alt it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall += 1
            if stall > 10:
                scale = 0.02 + 0.01 * min(stall // 10, 8)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                    a = best_a.copy()
                stall = 0

    return best_a


def gradient_optimize(a, time_budget=60.0, verbose=False):
    """Subgradient-based optimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.02
    rng = np.random.default_rng(12345)

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break

        a_new = subgradient_step(a, alpha=lr)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.03, 0.1)
        else:
            lr *= 0.9
            if lr < 1e-5:
                # Perturbation escape
                a_pert = best_a * np.exp(rng.normal(0, 0.04, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                    a = best_a.copy()
                lr = 0.02

    return best_a


def structured_init(n, mode, rng):
    """Generate structured initial sequences for various patterns."""
    if mode == 0:
        # Constant
        return np.ones(n) / n
    elif mode == 1:
        # Smooth random (Gaussian kernel smoothed)
        a = rng.uniform(0, 1, n)
        kernel_size = max(3, n // 100)
        x = np.arange(-kernel_size, kernel_size + 1)
        kernel = np.exp(-0.5 * (x / (kernel_size / 3.0)) ** 2)
        a = np.convolve(a, kernel / kernel.sum(), mode='same')
        a = np.maximum(a, 0)
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Two-peak pattern (antisymmetric around center)
        x = np.arange(n)
        c1 = n * 0.25 + rng.uniform(-0.05, 0.05) * n
        c2 = n * 0.75 + rng.uniform(-0.05, 0.05) * n
        w = n * 0.15
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif mode == 3:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.abs(np.cos(x)) + 0.3
        s = np.sum(a)
        return a / s
    elif mode == 4:
        # Periodic pattern with random phase
        period = max(3, n // rng.integers(10, 40))
        x = np.arange(n)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x / period + phase)
        a = np.maximum(a, 0.01)
        s = np.sum(a)
        return a / s
    elif mode == 5:
        # Linear ramp (asymmetric)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * x
        s = np.sum(a)
        return a / s
    elif mode == 6:
        # W-shaped (four peaks)
        x = np.arange(n)
        a = sum(np.exp(-((x - n * f) ** 2) / (2 * (n * 0.08) ** 2))
                for f in [0.15, 0.35, 0.65, 0.85])
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif mode == 7:
        # Random with heavy tail
        a = rng.pareto(2, n) + 0.1
        s = np.sum(a)
        return a / s
    elif mode == 8:
        # Bimodal antisymmetric (high at edges, low in middle)
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
        s = np.sum(a)
        return a / s
    else:
        # Pure random
        a = rng.uniform(0.1, 1, n)
        s = np.sum(a)
        return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):

        return sc

    # Phase 0: Establish baseline
    print("=== Phase 0: Baseline ===")
    for n_base in [400, 600, 800, 1000, 1200, 1500, 2000]:
        if time.monotonic() > deadline:
            break
        seq = [1.0] * n_base
        sc = evaluate_sequence(seq)
        print(f"  Constant n={n_base}: score={sc:.8f}")
        check_and_update(np.ones(n_base), f"const n={n_base}")

    # Try parent
    parent = None
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr.copy()
                n_parent = parent.size
                sc = check_and_update(parent, "parent")
                print(f"  Parent (n={n_parent}): score={sc:.8f}" if sc < np.inf else "  Parent: INVALID")
    except Exception as e:
        print(f"  Parent not available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_parent)
        best_score = evaluate_sequence([1.0] * n_parent)
        print(f"  Default constant (n={n_parent}): score={best_score:.8f}")

    print(f"  Current best: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: LP alternating on parent/best
    remaining = deadline - time.monotonic()
    if remaining > 60 and parent is not None:
        print(f"\n=== Phase 1: LP alternating on parent (n={n_parent}, budget={remaining:.0f}s) ===")
        a_p = parent / np.sum(parent)
        alloc = min(350.0, remaining * 0.35)
        a_lp = lp_alternating(a_p, time_budget=alloc, verbose=True, seed=seed)
        sc = check_and_update(a_lp, "LP on parent")
        if sc < np.inf:
            print(f"  LP on parent: score={sc:.8f}")

    # Phase 2: LP alternating on best with different seed
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: LP on best (n={len(best_seq)}, budget={remaining:.0f}s) ===")
        a_best = best_seq / np.sum(best_seq)
        alloc = min(200.0, remaining * 0.25)
        a_lp2 = lp_alternating(a_best, time_budget=alloc, verbose=True, seed=seed + 100)
        sc = check_and_update(a_lp2, "LP on best")
        if sc < np.inf:
            print(f"  LP on best: score={sc:.8f}")

    # Phase 3: Subgradient on best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: Subgradient (budget={remaining:.0f}s) ===")
        a_best = best_seq / np.sum(best_seq)
        alloc = min(100.0, remaining * 0.15)
        a_sg = gradient_optimize(a_best, time_budget=alloc)
        sc = check_and_update(a_sg, "subgradient")
        if sc < np.inf:
            print(f"  Subgradient: score={sc:.8f}")

    # Phase 4: Multi-n exploration with structured inits
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Multi-n exploration (budget={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(100, n_best - 200), max(100, n_best - 100),
            n_best,
            n_best + 100, n_best + 200,
            300, 500, 700, 900, 1100, 1400, 1700, 2000, 2500
        ]))

        per_n_budget = min(40.0, remaining / max(1, len(n_options)) * 0.7)

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < per_n_budget + 5:
                break

            # Resample best to this n
            if len(best_seq) == n_try:
                a_try = best_seq / np.sum(best_seq)
            else:
                a_try = resample_sequence(best_seq, n_try)
                s_t = float(np.sum(a_try))
                if s_t > 0:
                    a_try = a_try / s_t

            # Short LP optimization
            alloc = min(per_n_budget, remaining * 0.4)
            a_opt = lp_alternating(a_try, time_budget=alloc, seed=seed + n_try)
            sc = check_and_update(a_opt, f"multi-n {n_try}")
            if sc < np.inf:
                print(f"  n={n_try}: score={sc:.8f}")

    # Phase 5: Diverse initializations at best n
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Diverse inits (budget={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_inits = 12
        per_init = min(30.0, remaining / max(1, n_inits) * 0.6)

        for i in range(n_inits):
            remaining = deadline - time.monotonic()
            if remaining < per_init + 5:
                break

            # Vary n slightly and use different initialization modes
            n_try = n_best + int(rng.integers(-100, 100))
            n_try = max(100, min(3000, n_try))
            mode = i % 9
            a_init = structured_init(n_try, mode, rng)

            # Also try perturbation of best
            if i >= 4:
                a_pert = resample_sequence(best_seq, n_try)
                s_p = float(np.sum(a_pert))
                if s_p > 0:
                    a_pert = a_pert / s_p
                    a_pert = a_pert * np.exp(rng.normal(0, 0.08, n_try))
                    a_pert = np.maximum(a_pert, 0)
                    s_p2 = float(np.sum(a_pert))
                    if s_p2 > 0:
                        a_init = a_pert / s_p2

            alloc = min(per_init, remaining * 0.5)
            a_opt = lp_alternating(a_init, time_budget=alloc, seed=seed + 1000 + i)
            sc = check_and_update(a_opt, f"diverse {i} (mode={mode}, n={n_try})")
            if sc < np.inf:
                print(f"  Init {i} (mode={mode}, n={n_try}): score={sc:.8f}")

    # Phase 6: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Final polish (remaining={remaining:.0f}s) ===")
        a_final = best_seq.copy()
        s_f = float(np.sum(a_final))
        if s_f > 0:
            a_final = a_final / s_f

        for round_idx in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.3, 80.0)
            a_pol = lp_alternating(a_final, time_budget=alloc, verbose=False, seed=seed + 5000 + round_idx)
            sc = check_and_update(a_pol, f"polish {round_idx}")
            if sc < np.inf:
                print(f"  Polish {round_idx}: score={sc:.8f}")
                if sc < best_score:
                    a_final = a_pol / np.sum(a_pol)

        # Gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_gp = gradient_optimize(a_final, time_budget=min(remaining * 0.4, 40.0))
            sc = check_and_update(a_gp, "grad polish")
            if sc < np.inf:
                print(f"  Gradient polish: score={sc:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
