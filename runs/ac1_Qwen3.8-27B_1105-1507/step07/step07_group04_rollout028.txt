import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Ensure exact normalization
    s = np.sum(w)
    if s > 0:
        w /= s
    return w

def smooth_max_gradient(a, t):
    """Compute the gradient of the smooth max approximation of max(conv(a,a))."""
    n = len(a)
    c = compute_conv(a)  # length 2n-1
    # Softmax weights (shifted for numerical stability)
    tc = t * c
    tc = tc - np.max(tc)
    w = np.exp(tc)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        # Fallback: uniform weights (shouldn't happen with shift)
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    # Gradient: grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    grad = np.zeros(n)
    for j in range(n):
        # w[j:j+n] is always valid since j <= n-1 and 2n-1 - (j+n) = n-1-j >= 0
        grad[j] = 2.0 * np.dot(w[j:j+n], a)
    return grad, c

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses smooth-max gradient descent, LP alternating, MWU, and coordinate descent in a coordinated fashion
with annealing schedules and multi-n exploration to beat the target of 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = np.sum(w)
    if s > 1e-15:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def smooth_max_gradient(a, t):
    """Compute gradient of smooth max approximation of max(conv(a,a))."""
    n = len(a)
    c = compute_conv(a)
    tc = t * c
    tc = tc - np.max(tc)
    w = np.exp(tc)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(w[j:j + n], a)
    return grad, c


def smooth_max_descent(a, time_budget=100.0, t_start=50.0, t_end=50000.0,
                       eta_start=0.005, eta_end=0.0005, verbose=False):
    """Smooth max gradient descent with exponential annealing of t and eta."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, it / max(1, int(time_budget * 2.0)))
        t = t_start * (t_end / t_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        grad, c = smooth_max_gradient(a, t)
        grad -= np.mean(grad)

        # Adaptive step: scale by gradient norm
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            step = eta * grad / grad_norm * 0.01
        else:
            step = np.zeros(n)

        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        # Occasionally restart from best if stuck
        if no_improve > 200:
            a = best_a.copy()
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    SmoothMax: it={it}, obj={best_obj:.8f}, t={t:.0f}, elapsed={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n - j, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDLP, cp.SCS, cp.CLARABEL, cp.ECOS]
    if solver:
        solvers = [solver] + solvers

    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=100.0, verbose=False):

    n = len(a)

def mwu_gradient(a, k_indices):
    """Compute gradient of average conv at k_indices w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    K = len(k_indices)
    if K == 0:
        return grad
    for k in k_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        d_start = k + 1 - j_end
        d_end = k - j_start + 1
        grad[j_start:j_end] += a[d_start:d_end][::-1]
    grad /= K
    return 2.0 * grad


def mwu_optimize(a, time_budget=100.0, top_k=10, verbose=False):
    """MWU optimization targeting top-K convolution peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

    if s < 1e-10:

        max_b = float(np.max(conv))

        if top_k >= 2 * n - 1:
            k_indices = np.arange(2 * n - 1)
        else:
            k_indices = np.argsort(conv)[::-1][:top_k]

        obj_before = 2.0 * n * max_b
        batch = 20
        for _ in range(batch):
            if time.monotonic() - start > time_budget:
                break
            grad = mwu_gradient(a, k_indices)
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a /= s

        obj_new = compute_obj(a)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            if obj_new < obj_before:

        else:
            no_improve += 1

            if no_improve > 30:
                rng = np.random.default_rng(hash((n, no_improve, int(time.monotonic()*1000))) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))

                    obj_p = compute_obj(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            print(f"  Final SmoothMax: obj={best_score:.8f} **")
        else:
            print(f"  Final SmoothMax: obj={obj_smf:.8f}")

    if remaining > 15:
        # Coordinate descent
        alloc = min(20.0, remaining * 0.3)
        a_cd = coordinate_descent_polish(a_final, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a_final = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_cd:.8f}")

    # === Phase 7: More LP rounds if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 7: Extended LP ===")
        for ext_round in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(50.0, remaining * 0.3)
            a_ext = lp_alternating(best_seq, time_budget=alloc)
            obj_ext = compute_obj(a_ext)
            if obj_ext < best_score:
                best_score = obj_ext
                best_seq = a_ext.copy()
                print(f"  Ext LP {ext_round}: obj={best_score:.8f} **")
            else:
                print(f"  Ext LP {ext_round}: obj={obj_ext:.8f}")

    # === Phase 8: Random restarts from best ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(15):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            scale = 0.02 + 0.01 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr

            # Quick optimization
            alloc_total = min(remaining * 0.3, 20.0)
            a_opt = mwu_optimize(a_r, time_budget=alloc_total * 0.3, top_k=10)
            a_opt = lp_alternating(a_opt, time_budget=alloc_total * 0.7)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                if r % 5 == 0:
                    print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure non-negativity and bounded values
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with actual evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
