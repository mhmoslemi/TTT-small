"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 using
log-space (softmax parameterization) projected subgradient descent
with adaptive step sizes, combined with LP-based refinement.
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa, size).real[:2 * n - 1]


def compute_soft_subgradient(a, w):
    """
    Compute s_i = sum_k w[k] * a[k-i] for i=0,...,n-1.
    w has length 2n-1.
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    fw = fft(w, size)
    fa = fft(a, size)
    xc = ifft(fw * np.conj(fa), size).real
    # s_i = xc[size - i] for i >= 1, s_0 = xc[0]
    s = np.zeros(n)
    s[0] = xc[0]
    if n > 1:
        s[1:] = xc[size - 1 : size - n : -1]
    return s


def compute_obj_normalized(a):
    """Compute max_k c_k for a with sum(a)=1."""
    c = compute_conv(a)
    return float(np.max(c)), c


def softmax(theta):
    """Numerically stable softmax."""
    theta_shifted = theta - np.max(theta)
    exp_theta = np.exp(theta_shifted)
    return exp_theta / np.sum(exp_theta)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def optimize_logspace(a_init, n_steps=10000, beta=1.0, eta0=0.3, time_budget=60.0, verbose=False):
    """
    Optimize in log-space using softmax parameterization.
    Minimize max_k c_k where c = conv(a,a), a = softmax(theta), sum(a)=1.
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15).astype(np.float64)
    a_init = a_init / np.sum(a_init)
    
    theta = np.log(a_init)
    best_a = a_init.copy()
    best_score = np.max(compute_conv(a_init))
    
    eta = eta0
    start = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        a = softmax(theta)
        c = compute_conv(a)
        max_c = float(np.max(c))
        
        if max_c < best_score:
            best_score = max_c
            best_a = a.copy()
        
        # Weights: proportional to c_k (since sum(c) = 1 when sum(a) = 1)
        w = c  # w_k = c_k, already sums to 1
        w = w / np.sum(w)  # ensure normalization
        
        # Soft subgradient: s = sum_k w_k P_k a
        s = compute_soft_subgradient(a, w)
        c_bar = float(np.dot(w, c))
        
        # Gradient in log-space: 2 * a * (s - c_bar)
        grad = 2.0 * a * (s - c_bar)
        
        # Step
        theta_new = theta - eta * grad
        a_new = softmax(theta_new)
        c_new = compute_conv(a_new)
        max_c_new = float(np.max(c_new))
        
        if max_c_new < max_c:
            theta = theta_new
            eta = min(eta * 1.05, 2.0)
        else:
            eta *= 0.7
            if eta < 1e-6:
                # Restart with perturbation
                theta = np.log(np.maximum(a, 1e-15))
                theta += np.random.default_rng(step).normal(0, 0.1, n)
                eta = eta0
    
    return best_a


def optimize_subgradient(a_init, n_steps=50000, time_budget=60.0, verbose=False):

    Projected subgradient descent in the original space (simplex).
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_score = float(np.max(compute_conv(a)))
    
    eta = 1.0 / np.sqrt(n)
    start = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        c = compute_conv(a)
        max_c = float(np.max(c))
        
        if max_c < best_score:
            best_score = max_c
            best_a = a.copy()
        
        # Weights
        w = c / np.sum(c)
        
        # Subgradient: g = 2 * sum_k w_k P_k a
        s = compute_soft_subgradient(a, w)
        g = 2.0 * s

        # Projected gradient step
        a_new = project_simplex(a - eta_t * g)

        if compute_obj_normalized(a_new)[0] < max_c:
            a = a_new
        else:
            eta *= 0.9
    
    return best_a


def build_conv_matrix_a(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - n + 1)
        hi = min(n - 1, j)
        # A[j, i] = a[i] if i is in valid range for conv(a,g)[j]
        # conv(a,g)[j] = sum_i a[i]*g[j-i], valid when 0<=i<n and 0<=j-i<n
        # So i ranges from max(0, j-n+1) to min(n-1, j)
        for i in range(lo, hi + 1):
            A[j, i] = a[i]
    return A


def solve_lp_inner(a):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g (normalized to sum 1).
    """
    n = len(a)
    a = a / np.sum(a)
    
    try:
        from scipy.optimize import linprog
        # Variables: g[0..n-1], M
        # Minimize M
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Constraints: conv(a,g)[k] - M <= 0 for k=0,...,2n-2
        A_conv = np.zeros((2 * n - 1, n))
        for j in range(n):
            for k in range(2 * n - 1):
                i = k - j
                if 0 <= i < n:
                    A_conv[k, j] = a[i]
        
        A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = np.sum(g)
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    
    return None


def optimize_lp_alternating(a_init, time_budget=60.0, verbose=False):
    """LP alternating optimization."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        g = solve_lp_inner(a)
        if g is None:
            break
        
        # Line search
        best_t = 0.5
        best_t_score = np.inf
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            score = float(np.max(compute_conv(a_try)))
            if score < best_t_score:
                best_t_score = score
                best_t = t
        
        if best_t_score < best_score - 1e-12:
            best_score = best_t_score
            a = best_a + best_t * (g - a)  # hmm, this is wrong
            # Actually: a_new = (1-best_t)*a + best_t*g
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            a = a_new
            best_a = a_new.copy()
        else:
            # Perturb
            rng = np.random.default_rng(it * 7919)
            a_pert = a * np.exp(rng.normal(0, 0.05, n))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            a = a_pert
    
    return best_a
