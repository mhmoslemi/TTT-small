def compute_conv(a):
    n = len(a)
    if n == 1:

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

            a = a_new
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.85
            if lr < 1e-8:
                break
        
        # Anneal temperature
        temperature = max(0.001, temperature * 0.995)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses accelerated LP alternating with FFT-based gradient polishing,
temperature-annealed smoothed max, and multi-scale exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    from scipy.optimize import linprog

Wait, can max(conv) be less than 1/n for a non-negative sequence with sum=1?

By the Pigeonhole principle / averaging argument:
sum_k conv[k] = sum_k sum_i a[i]*a[k-i] = sum_i sum_j a[i]*a[j] = (sum a)^2 = 1

There are 2n-1 values of k, so the AVERAGE of conv[k] is 1/(2n-1).
Therefore, max(conv[k]) >= 1/(2n-1).

This gives a lower bound on the score: 2*n/(2n-1) ≈ 1 for large n.

def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def grad_smooth_descent(a, time_budget=30.0):

    lr = 0.05

        if time.monotonic() - start > time_budget:
            break

        # Softmax weights
        scaled = conv / temp
        ms = np.max(scaled)
        w = np.exp(scaled - ms)
        ws = np.sum(w)
        if ws < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(conv)] = 1.0
        else:
            w = w / ws

        a_rev = a[::-1]

        Ar = rfft(a_rev, size)
        conv_wa = irfft(W * Ar, size)
        grad = conv_wa[n - 1:2 * n - 1]

        grad -= np.mean(grad)

                a = a_new
                break

            break
        temp = max(0.001, temp * 0.99)

def resample(a, new_n):
    n = len(a)

    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
