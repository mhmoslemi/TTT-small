"""
Smooth LSE-based optimization with LP polishing for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothing, Adam optimizer, and adaptive temperature.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_obj_and_grad(a, tau):
    """
    Compute log-sum-exp smoothed objective and gradient w.r.t. a.
    Assumes a is on the simplex (sum=1, a>=0).
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    max_c = float(np.max(conv))
    
    # Softmax weights
    shifted = (conv - max_c) / tau
    exp_shifted = np.exp(shifted)
    Z = float(np.sum(exp_shifted))
    p = exp_shifted / Z  # softmax weights, length 2n-1
    
    # LSE value
    lse_val = max_c + tau * np.log(Z)
    
    # Gradient: g_a[i] = sum_k p[k] * d conv[k]/d a[i]
    # d conv[k]/d a[i] = 2*a[k-i] (if 0<=k-i<n and k!=2i), a[i] (if k=2i)
    # Approximation: g_a[i] = 2 * sum_j a[j] * p[i+j] - a[i] * p[2*i]
    g_a = np.zeros(n)
    for i in range(n):
        max_j = min(n, 2 * n - 1 - i)
        g_a[i] = 2.0 * float(np.dot(a[:max_j], p[i:i+max_j]))
        g_a[i] -= a[i] * p[2 * i]
    
    return lse_val, g_a


def lse_obj_and_grad_b(b, tau):
    """
    Compute LSE objective and gradient w.r.t. b (where a = softmax(b)).
    """
    # Compute a from b
    b_shifted = b - np.max(b)
    exp_b = np.exp(b_shifted)
    a = exp_b / float(np.sum(exp_b))
    
    # Get gradient w.r.t. a
    lse_val, g_a = lse_obj_and_grad(a, tau)
    
    # Chain rule: dF/db[j] = a[j] * (g_a[j] - sum_i g_a[i]*a[i])
    dot_ga_a = float(np.dot(g_a, a))
    g_b = a * (g_a - dot_ga_a)
    
    return lse_val, g_b


def smooth_optimize(a_init, time_budget=600.0, verbose=True):
    """
    Adam optimization on the log-parameterized simplex with LSE smoothing.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    a = a / np.sum(a)
    
    # Initialize b = log(a)
    b = np.log(a)
    
    # Initial objective
    init_conv = compute_conv(a)
    init_max = float(np.max(init_conv))
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 1.0 / np.sqrt(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    tau = init_max / 30.0
    min_tau = init_max / 500.0
    tau_decay = 0.97

    best_b = b.copy()
    best_a = a.copy()
    
    start = time.monotonic()
    stall = 0
    patience = 500
    
    while time.monotonic() - start < time_budget:
        t += 1
        
        # Compute objective and gradient in b-space
        f_val, g_b = lse_obj_and_grad_b(b, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g_b
        v = beta2 * v + (1 - beta2) * g_b**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Compute new a
        a_new = np.exp(b - np.max(b))
        a_new = a_new / float(np.sum(a_new))

                a = a / np.sum(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
                lr = 1.0 / np.sqrt(n)

        # Decay tau periodically
        if t % tau_decay_every == 0:
            tau = max(min_tau, tau * tau_decay)
        
        # Decay learning rate slowly

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - start

def compute_gradient_direct(a, top_k_positions):
    """
    Compute subgradient using specific peak positions.
    Returns gradient vector.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    
    # Softmax over top-k positions
    top_vals = conv[top_k_positions]
    shifted = (top_vals - top_vals[0]) / (max_c / 20)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w, k in zip(weights, top_k_positions):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if j != i:
                g[i] += 2.0 * w * a[j]
            else:
                g[i] += w * a[i]
    return g


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent(a_init, time_budget=200.0, verbose=True):
    """
    Subgradient descent with BB step size and top-k peak targeting.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / np.sum(a)
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    start = time.monotonic()
    step_size = 0.1 / np.sqrt(n)
    prev_a = a.copy()
    prev_g = None
    stall = 0
    K = 5
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:K]
        g = compute_gradient_direct(a, top_k)
        
        # BB step size
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-15:
                bb = ss / abs(sy)
                step_size = min(max(step_size * 0.5 + bb * 0.5, 1e-6), 1.0)
        
        # Projected step
        a_new = a - step_size * g
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            step_size *= 0.8
            if stall > 200:
                rng = np.random.default_rng(it * 13 + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                step_size = 0.1 / np.sqrt(n)
                prev_g = None
                stall = 0
        
        prev_a = a.copy()
        prev_g = g.copy()
        it += 1
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD it{it}: obj={best_obj:.8f}, step={step_size:.6e}, elapsed={elapsed:.1f}s")
    
    return best_a


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build the convolution matrix
    # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid indices
    # For fixed a, this is linear in g.
    # (A @ g)[k] = conv(a, g)[k]
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    from scipy.optimize import linprog
    
    # Variables: g[0], ..., g[n-1], M
    # Objective: minimize M
    # Constraints: A @ g - M <= 0 for all k, sum(g) = 1, g >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # A_ub @ [g, M] <= b_ub
    # conv(a,g)[k] - M <= 0
    A_conv = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_conv[k, j] = a[idx]
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - 0.5 * np.abs(x)
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.1))
    elif mode == 5:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)

"""
Alternative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses direct subgradient with top-peak gradient accumulation, BB step sizes,
adaptive temperature soft-max peak weighting, and iterative LP polishing.
Exploration across dimensions via FVS-inspired spectral concentration.
"""
import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_peak(a, k):
    """
    Gradient of conv(a, a)[k] w.r.t. a.
    d/d a[i] of sum_j a[j]*a[k-j] = 2*a[k-i] (when 2i != k), a[i] (when 2i == k).
    Returns vector of length n.
    """
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    for i in range(i_start, i_end + 1):
        j = k - i
        if j != i:
            g[i] = 2.0 * a[j]
        else:
            g[i] = a[i]
    return g


def soft_topK_grad(a, K=8, temperature=None):
    """
    Weighted gradient using soft-attention over top-K peaks of conv(a,a).
    Returns (gradient, max_conv_value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    # Soft attention weights over ALL peaks (favors top ones via temperature)
    if temperature is None:
        temperature = max_c / 5.0
    shifted = (conv - max_c) / temperature
    e = np.exp(shifted)
    weights = e / e.sum()
    # Accumulate gradient as sum_k weights[k] * grad_k
    # grad_k[i] = 2*a[k-i] (or a[i] if k=2i)
    # This is a weighted "blur" of a
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        gi = gradient_peak(a, k)
        g += w * gi
    return g, max_c


def lse_smoothed_grad(a, tau):
    """
    LSE-smoothed objective and its gradient w.r.t. a.
    F(a) = tau * log(sum_k exp(conv(a,a)[k] / tau))
    grad = sum_k p_k * grad_conv_k where p_k = softmax(conv/tau)
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / tau
    e = np.exp(shifted)
    Z = float(e.sum())
    p = e / Z
    lse_val = max_c + tau * np.log(Z)
    # grad_f = sum_k p_k * 2*a[k-i] (approx, with k=2i correction)
    g = np.zeros(n)
    for i in range(n):
        max_j = min(n, 2 * n - 1 - i)
        g[i] = 2.0 * float(np.dot(a[:max_j], p[i:i + max_j]))
        g[i] -= a[i] * p[2 * i]
    return lse_val, g


def subgrad_search(a0, time_budget, verbose=True):
    """
    Subgradient descent with BB step size, nonlinear step halving,
    and soft-topK gradient accumulation.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    prev_a = a.copy()
    prev_g = None
    step = 1.0 / np.sqrt(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        g, max_c = soft_topK_grad(a, K=8)
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-14 and abs(sy) > 1e-14:
                bb = ss / abs(sy)
                step = np.clip(0.5 * step + 0.5 * bb, 1e-6, 2.0)
        taken = False
        t = step
        for _ in range(10):
            a_try = a - t * g
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                taken = True
                stall = 0
                break
            t *= 0.5
        if not taken:
            stall += 1
            if stall > 300:
                rng = np.random.default_rng(1000 + it * 31 + n)
                scale = min(0.4, 0.04 + 0.03 * (stall // 300))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0.0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_g = None
                step = 1.0 / np.sqrt(n)
                stall = 0
        prev_a = a.copy()
        prev_g = g.copy()
        it += 1
        if verbose and it % 300 == 0:
            print("    subgrad it=%d obj=%.8f step=%.4e elapsed=%.1fs" % (
                it, best_obj, step, time.monotonic() - start))
    return best_a


def lse_optimize(a0, time_budget, verbose=True):
    """
    Adam on log-parametrized simplex for the LSE-smoothed objective.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-12).copy()
    a = a / a.sum()
    b = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    lr = 2.0 / np.sqrt(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    conv0 = compute_conv(a)
    max0 = float(np.max(conv0))
    tau = max0 / 25.0
    min_tau = max0 / 800.0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        it += 1
        _, g_a = lse_smoothed_grad(a, tau)
        dot = float(np.dot(g_a, a))
        g_b = a * (g_a - dot)
        m = beta1 * m + (1 - beta1) * g_b
        v = beta2 * v + (1 - beta2) * g_b * g_b
        mhat = m / (1 - beta1 ** it)
        vhat = v / (1 - beta2 ** it)
        b = b - lr * mhat / (np.sqrt(vhat) + eps)
        a_new = np.exp(b - b.max())
        a_new = a_new / a_new.sum()
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 600:
                rng = np.random.default_rng(7 + it * 17 + n)
                b = np.log(np.maximum(best_a, 1e-12)) + rng.normal(0, 0.05, n)
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                it = 0
                tau = min_tau * 4.0
                stall = 0
        if it % 400 == 0:
            tau = max(min_tau, tau * 0.9)
        lr = max(1e-5, lr * 0.999)
        if verbose and it % 1000 == 0:
            print("    lse it=%d obj=%.8f tau=%.3e lr=%.4e elapsed=%.1fs" % (
                it, best_obj, tau, lr, time.monotonic() - start))
    return best_a


def build_conv_matrix(a):
    """A @ g = conv(a, g); A has shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_minpeak(a, M_target=None):
    """
    LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0.
    M is given or set to current peak of conv(a,a).
    Returns (g_normalized, M_used) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    if M_target is None:
        M_target = float(np.max(compute_conv(a)))
    from scipy.optimize import linprog
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x, 0)
        s = float(g.sum())
        if s > 1e-10:
            return g / s, M_target
    return None, None


def lp_alternate(a0, time_budget, verbose=True):

    """

def spectral_init(n, rng, mode):
    """
    Construct initial sequences with diverse spectral shapes.
    mode 0: flat (uniform)
    mode 1: bandpass sine
    mode 2: FVS-ish (all-ones in a centered frequency window)
    mode 3: superposition of few sinusoids
    """
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        k0 = rng.integers(n // 6, 5 * n // 6)
        bw = max(2, n // 10)
        a = np.zeros(n)
        for k in range(max(0, k0 - bw), min(n, k0 + bw)):
            a[k] = 1.0
        a = np.real(irfft(rfft(a, 2 * n), 2 * n)[:n])
    elif mode == 2:
        # random + FFT -> magnitude -> IFFT -> |.| to keep nonnegative
        x = rng.normal(0, 1, n)
        F = rfft(x, 2 * n)
        bw = n // 12
        k0 = rng.integers(1, len(F) - 1)
        mag = np.abs(F)
        mask = np.zeros_like(mag)
        for k in range(max(0, k0 - bw), min(len(F), k0 + bw)):
            mask[k] = 1.0
        F2 = F * mask
        y = irfft(F2, 2 * n)[:n]
        a = np.abs(y)
    else:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for _ in range(3):
            f = rng.uniform(0.2, 3.0)
            p = rng.uniform(0, 2 * np.pi)
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
    a = np.maximum(a, 1e-4)
    a = a / a.sum()
    return a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

    n_cur = len(best)
    n_opts = sorted(set([n_cur - 100, n_cur - 50, n_cur + 50, n_cur + 100,
                         n_cur + 200, 600, 700, 900, 1000, 1200]))
    n_opts = [x for x in n_opts if 250 <= x <= 1500]
    for nt in n_opts[:5]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        if nt == n_cur:
            cand = best.copy()
        else:
            cand = resample(best, nt)
            s = cand.sum()
            cand = np.maximum(cand, 0.0)
            if s < 1e-10:
                cand = np.ones(nt) / nt
            else:
                cand = cand / s
        # quick LSE
        bL = min(rem * 0.35, 90.0)
        if bL > 15:
            aL = lse_optimize(cand, bL, verbose=False)
            cand = aL
        # quick LP
        rem = deadline - time.monotonic()
        bP = min(rem * 0.35, 70.0)
        if bP > 10:
            aP = lp_alternate(cand, bP, verbose=False)
            cand = aP
        o = compute_obj(cand)
        if o < best_obj:
            best_obj, best = o, cand
            print("phase5 n=%d -> %.8f **" % (nt, best_obj))
        else:
            print("phase5 n=%d -> %.8f" % (nt, o))

    # --- Phase 6: spectral rescape from fresh inits, seeded with best ---
    rem = deadline - time.monotonic()
    n_use = len(best)
    for mode in [1, 2, 3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        init = spectral_init(n_use, rng, mode)
        a = 0.6 * init + 0.4 * best
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        bL = min(rem * 0.3, 80.0)
        if bL > 15:
            a = lse_optimize(a, bL, verbose=False)
        rem = deadline - time.monotonic()
        bP = min(rem * 0.3, 50.0)
        if bP > 10:
            a = lp_alternate(a, bP, verbose=False)
        o = compute_obj(a)
        if o < best_obj:
            best_obj, best = o, a
            print("phase6 mode=%d -> %.8f **" % (mode, best_obj))
        else:
            print("phase6 mode=%d -> %.8f" % (mode, o))

    # --- Phase 7: final LP + subgrad polish ---
    rem = deadline - time.monotonic()
    if rem > 40:
        a = lp_alternate(best, min(rem * 0.5, 120.0), verbose=True)
        o = compute_obj(a)
        if o < best_obj:
            best_obj, best = o, a
            print("phase7 lp -> %.8f **" % best_obj)
    rem = deadline - time.monotonic()
    if rem > 20:
        a = subgrad_search(best, min(rem * 0.7, 80.0), verbose=True)
        o = compute_obj(a)
        if o < best_obj:
            best_obj, best = o, a
            print("phase7 subgrad -> %.8f **" % best_obj)

    result = finish()
    print("FINAL n=%d score=%.8f elapsed=%.1fs" % (
        len(result), best_obj, time.monotonic() - t0))
    return result
