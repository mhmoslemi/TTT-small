        obj = compute_obj(a_new)

            best_a = a_new.copy()
        a = a_new
    
    return best_a

multi-peak smooth gradient descent, QP projection, simulated annealing,

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        constrained_bins = [int(i) for i in idx]

        c[-1] = 1.0
        # Constraints: A_k @ g - M <= 0 for each constrained bin k

            A_ub[i, n] = -1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            mid = (lo + hi) / 2
            a_try = (1.0 - mid) * a + mid * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = mid
            # Also check a few nearby points
            for dt in [0.05, -0.05, 0.1, -0.1]:
                t2 = mid + dt
                if 0.01 <= t2 <= 1.0:
                    a_try2 = (1.0 - t2) * a + t2 * g_lp
                    a_try2 = np.maximum(a_try2, 0)
                    s_try2 = float(np.sum(a_try2))
                    if s_try2 < 1e-10:
                        continue
                    a_try2 = a_try2 / s_try2
                    obj_try2 = compute_obj(a_try2)
                    if obj_try2 < best_t_obj:
                        best_t_obj = obj_try2
                        best_t = t2
            if obj_try < best_t_obj:
                lo = mid
            else:
                hi = mid

        if best_t_obj < best_obj - 1e-13:
            a_try = (1.0 - best_t) * a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
            best_obj = best_t_obj
            best_a = a_try.copy()
            a = a_try
            stall_count = 0
            if verbose and iter_count % 20 == 0:
                print(f"        LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.4f}")
        else:
            stall_count += 1
            if stall_count > 50:
                # Perturb to escape local minimum
                rng = np.random.default_rng(hash((iter_count, n, stall_count)) % (2**32))
                scale = 0.005 + 0.005 * min(stall_count // 50, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0

    return best_a


def pnorm_continuation(a, time_budget=60.0, verbose=False):
    """
    p-norm continuation: minimize ||conv||_p for decreasing p.
    Large p approximates the max. Starting from p=2 (smooth) and increasing
    helps avoid local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Schedule: p goes from 2 to very large
    p_values = [2, 3, 4, 6, 10, 20, 50, 100, 200, 500, 1000]

    for p in p_values:
        if time.monotonic() - start > time_budget:
            break
        budget_this_p = max(5.0, time_budget / len(p_values))
        p_start = time.monotonic()

        lr = 0.005
        momentum = 0.8
        v = np.zeros(n)
        stall = 0

        while time.monotonic() - p_start < budget_this_p:
            conv = compute_conv(a)
            # Gradient of sum(conv^p) w.r.t. a[i]
            if p == 1:
                weights = np.ones_like(conv)
                pw = 1.0
            else:
                weights = conv ** (p - 1)
                pw = float(p)
            weights = weights / (np.sum(weights) + 1e-30)
            grad = np.correlate(weights, a, mode='valid')
            grad = grad - np.mean(grad)

            v = momentum * v - lr * grad
            a_new = simplex_project(a + v)
            obj_new = compute_obj(a_new)

            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                if lr < 0.02:
                    lr = min(lr * 1.02, 0.02)
            else:
                stall += 1
                if stall > 20:
                    lr *= 0.7
                    v = np.zeros(n)
                    a = best_a.copy()
                    stall = 0
                    if lr < 1e-6:
                        break

        if verbose:
            print(f"        p={p}: obj={best_obj:.8f}")

    return best_a


def multi_peak_gradient(a, time_budget=60.0, top_k=5, verbose=False):
    """
    Gradient descent targeting top-K convolution bins simultaneously.
    Uses softmax weights over the top-K bins for a smooth approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 0.01  # Temperature for softmax over top-k
    lr = 0.003
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        # Softmax weights over ALL bins (with temperature control)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)

        grad = np.correlate(weights, a, mode='valid')
        grad = grad - np.mean(grad)

        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 30 == 0 and lr < 0.015:
                lr = min(lr * 1.03, 0.015)
        else:
            stall += 1
            if stall > 40:
                lr *= 0.75
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break

        # Anneal temperature
        if it > 30:
            T = max(0.001, 0.01 * (0.998 ** it))

    return best_a


def qp_projection(a, time_budget=60.0, verbose=False):
    """
    QP projection: find a point close to current a that has lower max convolution.
    Uses bisection on the target M to find the feasible region boundary.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    try:
        import cvxpy as cp
    except ImportError:
        return a

    current_M = float(np.max(compute_conv(a)))

    # Bisection: find the minimum M such that there exists g with
    # conv(a,g) <= M, sum(g)=1, g>=0, and ||g-a||^2 < some bound
    # Actually, let's just do iterative QP: project a onto {conv <= M'} for decreasing M'

    M_target = current_M * 0.99
    iterations = 0
    max_iters = 50

    while iterations < max_iters and time.monotonic() - start < time_budget:
        if M_target < current_M * 0.5:
            break

        A = build_conv_matrix(a)
        conv_a = compute_conv(a)
        top_idx = int(np.argmax(conv_a))

        # QP: min ||g - a||^2 s.t. A[top_idx] @ g <= M_target, sum(g)=1, g>=0
        # But this only constrains one bin. Let's constrain top 3.
        top_3 = np.argsort(conv_a)[-3:]

        g = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(g) == 1]
        for k in top_3:
            constraints.append(A[int(k), :] @ g <= M_target)

        prob = cp.Problem(cp.Minimize(cp.sum_squares(g - a)), constraints)

        solved = False
        for solver in ['CLARABEL', 'SCS', 'ECOS']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        obj_g = compute_obj(g_val)
                        if obj_g < best_obj - 1e-13:
                            best_obj = obj_g
                            best_a = g_val.copy()
                            a = g_val.copy()
                            current_M = float(np.max(compute_conv(a)))
                            M_target = min(M_target * 0.98, current_M * 0.995)
                            solved = True
                            break
                        else:
                            # Try less aggressive target
                            M_target = (M_target + current_M) / 2
                            break
            except Exception:
                continue

        if not solved:
            M_target = (M_target + current_M) / 2
            if current_M - M_target < 1e-8:
                break

        iterations += 1

    return best_a


def simulated_annealing_optimize(a, time_budget=120.0, verbose=False):
    """
    Simulated annealing with structured perturbations.
    Perturbs the sequence multiplicatively and accepts based on Metropolis criterion.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    T_init = 0.05
    T_min = 0.0001
    cooling_rate = 0.995

    T = T_init
    perturb_scale = 0.05

    while time.monotonic() - start < time_budget:
        # Generate perturbation
        noise = rng.normal(0, perturb_scale, n)
        a_perturbed = a * np.exp(noise)
        a_perturbed = np.maximum(a_perturbed, 0)
        s_p = float(np.sum(a_perturbed))
        if s_p < 1e-10:
            T *= cooling_rate
            continue
        a_perturbed = a_perturbed / s_p

        obj_perturbed = compute_obj(a_perturbed)

        # Metropolis acceptance
        delta = obj_perturbed - current_obj
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            a = a_perturbed
            current_obj = obj_perturbed
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
                if verbose:
                    print(f"        SA: obj={best_obj:.8f}, T={T:.6f}")
        else:
            # Occasionally do a small LP step from current best
            if rng.random() < 0.1:
                g_lp, _ = solve_lp_direction(best_a, top_k=1)
                if g_lp is not None:
                    for t in [0.1, 0.2, 0.3, 0.5]:
                        a_try = (1 - t) * best_a + t * g_lp
                        a_try = np.maximum(a_try, 0)
                        st = float(np.sum(a_try))
                        if st > 1e-10:
                            a_try = a_try / st
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                current_obj = obj_try
                                a = a_try
                                break

        T *= cooling_rate
        perturb_scale = max(0.01, 0.05 * T / T_init)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a / np.sum(a)
    elif mode == 3:
        center = n // 2
        width = max(1, n // 5)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a += 0.3 * np.exp(-((x - center * 0.5) ** 2) / (2 * (width // 3 + 1) ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.5
        a = a / np.sum(a)
    elif mode == 5:
        x = np.arange(n) / (n - 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    elif mode == 6:
        # Two humps
        c1, c2 = n // 3, 2 * n // 3
        w = max(1, n // 10)
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    elif mode == 7:
        # Sine-modulated
        x = np.arange(n)
        a = 0.5 + 0.3 * np.sin(2 * np.pi * x / n * rng.integers(2, 8))
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 60.0  # Leave 60s for final validation

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_best = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                n_best = len(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={n_best}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_best) / n_best
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_best}): obj={best_score:.8f}")

    # === Phase 1: LP Alternating with multi-peak ===
    print(f"\n=== Phase 1: LP Alternating (n={n_best}) ===")
    a = best_seq.copy()

    # First: standard LP with top_k=1
    remaining = deadline - time.monotonic()
    alloc = min(backup_budget := 200.0, remaining * 0.2)
    a_new = lp_alternating(a, time_budget=alloc, verbose=True, top_k=1)
    obj = compute_obj(a_new)
    if obj < best_score:
        best_score = obj
        best_seq = a_new.copy()
        a = a_new
        print(f"  LP top1: obj={best_score:.8f}")

    # Second: LP with top_k=3
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(150.0, remaining * 0.15)
        a_new = lp_alternating(a, time_budget=alloc, verbose=True, top_k=3)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP top3: obj={best_score:.8f}")

    # Third: LP with top_k=5
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(100.0, remaining * 0.1)
        a_new = lp_alternating(a, time_budget=alloc, verbose=True, top_k=5)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP top5: obj={best_score:.8f}")

    # === Phase 2: p-norm continuation ===
    print(f"\n=== Phase 2: p-norm Continuation ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(120.0, remaining * 0.12)
        a_pn = pnorm_continuation(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_pn)
        if obj < best_score:
            best_score = obj
            best_seq = a_pn.copy()
            a = a_pn
            print(f"  p-norm: obj={best_score:.8f}")
        else:
            print(f"  p-norm: obj={obj:.8f} (no improvement)")

    # === Phase 3: Multi-peak gradient ===
    print(f"\n=== Phase 3: Multi-Peak Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(100.0, remaining * 0.1)
        a_mp = multi_peak_gradient(a, time_budget=alloc, top_k=5, verbose=True)
        obj = compute_obj(a_mp)
        if obj < best_score:
            best_score = obj
            best_seq = a_mp.copy()
            a = a_mp
            print(f"  Multi-peak: obj={best_score:.8f}")
        else:
            print(f"  Multi-peak: obj={obj:.8f} (no improvement)")

    # === Phase 4: QP Projection ===
    print(f"\n=== Phase 4: QP Projection ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(80.0, remaining * 0.08)
        a_qp = qp_projection(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_qp)
        if obj < best_score:
            best_score = obj
            best_seq = a_qp.copy()
            a = a_qp
            print(f"  QP: obj={best_score:.8f}")
        else:
            print(f"  QP: obj={obj:.8f} (no improvement)")

    # === Phase 5: Simulated Annealing ===
    print(f"\n=== Phase 5: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(120.0, remaining * 0.1)
        a_sa = simulated_annealing_optimize(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_sa)
        if obj < best_score:
            best_score = obj
            best_seq = a_sa.copy()
            a = a_sa
            print(f"  SA: obj={best_score:.8f}")
        else:
            print(f"  SA: obj={obj:.8f} (no improvement)")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        n_best - 50, n_best - 25,
        n_best, n_best + 25, n_best + 50, n_best + 100, n_best + 200,
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 2000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2500]

    # Sort by proximity to current best n first
    n_options_sorted = sorted(n_options, key=lambda x: abs(x - n_best))

    for n_try in n_options_sorted:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP alternating
        alloc_lp = min(remaining * 0.08, 25.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp, top_k=1)
        obj_try = compute_obj(a_opt)

        # Multi-peak polish
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_try < best_score + 0.01:
            alloc_mp = min(remaining * 0.05, 15.0)
            a_opt2 = multi_peak_gradient(a_opt, time_budget=alloc_mp)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        # p-norm polish
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_try < best_score + 0.005:
            alloc_pn = min(remaining * 0.04, 12.0)
            a_opt3 = pnorm_continuation(a_opt, time_budget=alloc_pn)
            obj_try3 = compute_obj(a_opt3)
            if obj_try3 < obj_try:
                a_opt = a_opt3
                obj_try = obj_try3

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Random Restarts with Different Initializations ===
    print(f"\n=== Phase 7: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        init_mode = r % 8
        n_r = n_best + int(rng.integers(-50, 51))
        n_r = max(300, min(1500, n_r))
        a_r = random_init(n_r, rng, mode=init_mode)

        # Quick optimization: LP + multi-peak
        alloc_r = min(remaining * 0.08, 25.0)
        a_opt = lp_alternating(a_r, time_budget=alloc_r * 0.5, top_k=1)

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = multi_peak_gradient(a_opt, time_budget=min(alloc_r * 0.4, remaining * 0.3))

        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_final = best_seq.copy()

        # Multiple LP rounds
        for polish_round in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            top_k = polish_round % 3 + 1
            alloc = min(remaining * 0.2, 40.0)
            a_lp = lp_alternating(a_final, time_budget=alloc, top_k=top_k)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_final = a_lp.copy()
                print(f"  Polish LP (top{k}): obj={best_score:.8f}")

        # p-norm final
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_pn = pnorm_continuation(a_final, time_budget=min(remaining * 0.15, 25.0))
            obj_pn = compute_obj(a_pn)
            if obj_pn < best_score:
                best_score = obj_pn
                best_seq = a_pn.copy()
                print(f"  Final p-norm: obj={best_score:.8f}")

        # Multi-peak final
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_mp = multi_peak_gradient(a_final, time_budget=min(remaining * 0.15, 20.0))
            obj_mp = compute_obj(a_mp)
            if obj_mp < best_score:
                best_score = obj_mp
                best_seq = a_mp.copy()
                print(f"  Final multi-peak: obj={best_score:.8f}")

        # QP final
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_qp = qp_projection(a_final, time_budget=min(remaining * 0.15, 20.0))
            obj_qp = compute_obj(a_qp)
            if obj_qp < best_score:
                best_score = obj_qp
                best_seq = a_qp.copy()
                print(f"  Final QP: obj={best_score:.8f}")

    # === Phase 9: Final fine-tuning with very small perturbations ===
    print(f"\n=== Phase 9: Fine-tuning ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_tune = best_seq.copy()
        n = len(a_tune)
        rng2 = np.random.default_rng(seed + 999)

        for ft_iter in range(2000):
            if time.monotonic() > deadline:
                break
            # Small random perturbation
            scale = 0.002
            noise = rng2.normal(0, scale, n)
            a_try = a_tune * np.exp(noise)
            a_try = np.maximum(a_try, 0)
            s_t = float(np.sum(a_try))
            if s_t < 1e-10:
                continue
            a_try = a_try / s_t
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                a_tune = a_try

        # Also try LP from the fine-tuned point
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_lp = lp_alternating(a_tune, time_budget=min(remaining * 0.5, 30.0), top_k=3)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Fine-tune LP: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.array(result))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
