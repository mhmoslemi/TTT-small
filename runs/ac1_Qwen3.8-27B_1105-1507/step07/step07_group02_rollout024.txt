**Method: "Peak reduction" via direct adjustment**

1. Compute c = conv(a, a), find peak positions k* (and near-peak positions)
2. For each peak position k*, the contribution of a[j] is a[j]*a[k*-j]
3. Reduce a[j] in proportion to its "responsibility" for the peak: a[j] * a[k*-j]
4. Redistribute the reduced mass to positions that don't contribute to the peak

This is like a "minimax" adjustment: reduce the peak by shaving off the most responsible elements and putting the mass where it doesn't hurt.

Let me formalize:
- Let P be the set of peak positions (c[k] >= max_c - epsilon)
- For each j, the "pressure" is p[j] = sum_{k in P} a[k-j] (sum of coefficients that would increase the peak if a[j] increases)
- Update: a[j] <- a[j] * (1 - eta * p[j] / max(p)) (multiplicative reduction)
- Normalize: a <- a / sum(a)

This is a form of "multiplicative weights" or "follow-the-regularized-leader" for this problem.

Actually, this is very similar to the **Exponential Weights** algorithm for online learning, or the **Hedge** algorithm.

def multiplicative_weight_update(a, eta=0.1, n_peaks=10):
    c = conv(a, a)
    max_c = max(c)
    # Find top n_peaks positions
    top_k = np.argsort(c)[-n_peaks:]
    P = set(top_k)
    
    # Compute pressure
    pressure = np.zeros(len(a))
    for k in P:
        for j in range(len(a)):
            idx = k - j
            if 0 <= idx < len(a):
                pressure[j] += a[idx]
    
    # Multiplicative update
    max_p = max(pressure)
    if max_p > 0:
        a = a * np.exp(-eta * pressure / max_p)
    
    # Normalize
    a = a / sum(a)
    return a
```

- Fixed g, optimize a (LP, by symmetry)

This is a "best response" or "coordinate descent" type method.

The convergence of this method is not guaranteed (it's not a convex problem), but it often works well in practice.

```python
"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses multiplicative weight descent, projected gradient, and LP alternating optimization.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def multiplicative_weight_descent(a, n_iters=5000, eta=0.5, n_peaks=5, verbose=False):
    """
    Multiplicative weight descent for minimizing max auto-convolution.
    a should be normalized (sum = 1, a >= 0).
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iters):
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Find top peak positions
        if n_peaks >= len(conv):
            peak_positions = np.arange(len(conv))
        else:
            peak_positions = np.argsort(conv)[-n_peaks:]
        
        # Compute pressure: for each j, sum of a[k-j] over peak positions k
        pressure = np.zeros(n)
        for k in peak_positions:
            # a[k-j] is valid for j in [max(0, k-n+1), min(k, n-1)]
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min <= j_max:
                # pressure[j] += a[k-j] for j in [j_min, j_max]
                # k-j ranges from k-j_max to k-j_min
                idx_min = k - j_max
                idx_max = k - j_min
                pressure[j_min:j_max+1] += a[idx_min:idx_max+1]
        
        # Multiplicative update
        max_p = float(np.max(pressure))
        if max_p > 1e-10:
            a = a * np.exp(-eta * pressure / max_p)
        
        # Normalize
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        
        # Check improvement
        obj = 2.0 * n * max_c  # This is the old max_c, not the new one
        # Actually compute new obj
        conv_new = compute_conv(a)
        obj_new = 2.0 * n * float(np.max(conv_new))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
        
        if verbose and it % 100 == 0:
            print(f"    MW it{it}: obj={obj_new:.8f}")
        
        # Early stopping
        if obj_new < 1.001 * (2.0 * n / (2.0 * n - 1)):
            break
        if it > 100 and obj_new > best_obj * 1.01:
            break
    
    return best_a

def lp_alternate(a, n_iters=50, verbose=False):
