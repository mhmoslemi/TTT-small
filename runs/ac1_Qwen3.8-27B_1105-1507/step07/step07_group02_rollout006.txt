def compute_conv(a):

    fa = rfft(a, size)

def subgradient_descent(a, time_budget, lr_start=0.05):
    """
    Subgradient method: at each step, find the argmax of conv and take a step

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:

    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_start
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0
        grad = np.zeros(n)
        # a[k_star - i] for i = 0, ..., n-1 where 0 <= k_star - i < n
        # i.e., k_star - n + 1 <= i <= k_star
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        for i in range(i_min, i_max + 1):
            grad[i] = 2.0 * a[k_star - i]
        
        # Center the gradient to approximately preserve sum
        grad_centered = grad - np.mean(grad)
        
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.1, 0.5)
            stall = 0
        else:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a_pert = a * np.exp(rng.normal(0, 0.1, n))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

            if stall > 10:

                scale = 0.02 + 0.02 * min(stall // 10, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0
    
    return best_a

def block_descent(a, time_budget=30.0, block_size=50):
    """
    Block coordinate descent: optimize a block of coordinates at a time.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # For each block, optimize the block coordinates
    # Keep others fixed
    # The objective within a block is: 2n * max_k (base_k + sum_{i in block} a[i] * a[k-i]) / (S_base + sum_{i in block} a[i])^2
    # where base_k is the convolution contribution from non-block coordinates
    
    # This is still complex. Let's use a simpler approach:
    # Just do local perturbation within blocks
    
    rng = np.random.default_rng(12345)
    while time.monotonic() - start < time_budget:
        # Pick a random block
        start_idx = rng.integers(0, max(1, n - block_size))
        end_idx = min(n, start_idx + block_size)
        block = a[start_idx:end_idx].copy()
        S_rest = s - np.sum(block)
        
        best_block = block.copy()
        best_block_obj = np.inf
        
        # Try several modifications to the block
        for trial in range(20):
            if trial == 0:
                block_try = block.copy()
            elif trial == 1:
                # Flatten within block
                block_try = np.full(end_idx - start_idx, np.mean(block))
            elif trial == 2:
                # Inverse flatten (more contrast)
                mean = np.mean(block)
                block_try = mean * (1 + 0.5 * (block - mean) / (np.std(block) + 1e-10))
                block_try = np.maximum(block_try, 0)
            else:
                block_try = block * np.exp(rng.normal(0, 0.1, end_idx - start_idx))
                block_try = np.maximum(block_try, 0)
            
            S_try = S_rest + np.sum(block_try)
            if S_try < 1e-10:
                continue
            
            a_try = a.copy()
            a_try[start_idx:end_idx] = block_try
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            
            if obj_try < best_block_obj:
                best_block_obj = obj_try
                best_block = block_try.copy()
        
        if best_block_obj < best_obj - 1e-12:
            a[start_idx:end_idx] = best_block
            s = float(np.sum(a))
            a = a / s
            best_a = a.copy()
            best_obj = best_block_obj
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses smoothed gradient descent as the primary optimizer,
    with LP alternating and other methods for refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Phase 0: Baselines
    print("=== Phase 0: Baselines ===")
    for n_base in [400, 600, 800, 1000, 1200, 1500]:
        if time.monotonic() > deadline:
            break
        seq = [1.0] * n_base
        score = evaluate_sequence(seq)
        print(f"  Constant n={n_base}: score={score:.8f}")
        if score < best_score:
            best_score = score
            best_seq = np.ones(n_base)
    
    # Try parent
    parent = None
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr
                n_parent = parent.size
                parent_list = [float(x) for x in parent]
                score = evaluate_sequence(parent_list)
                print(f"  Parent (n={n_parent}): score={score:.8f}")
                if score < best_score:
                    best_score = score
                    best_seq = parent.copy()
    except Exception as e:
        print(f"  Parent not available: {e}")
    
    if best_seq is None:
        best_seq = np.ones(800)
        best_score = evaluate_sequence([float(x) for x in best_seq])
    
    print(f"  Current best: n={len(best_seq)}, score={best_score:.8f}")
    
    # Phase 1: Smoothed gradient descent on parent/best
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: Smoothed GD (remaining={remaining:.0f}s) ===")
        a_work = best_seq.copy()
        s = float(np.sum(a_work))
        a_work = a_work / s
        
        # Run smoothed GD with multiple beta schedules
        schedules = [
            (5.0, 30.0, 0.1, 0.999),
            (10.0, 100.0, 0.05, 0.998),
            (20.0, 200.0, 0.03, 0.997),
            (1.0, 20.0, 0.2, 0.9995),
        ]
        
        for si, (bs, be, lr_s, lr_d) in enumerate(schedules):
            remaining = deadline - time.monotonic()

                break
            alloc = min(120.0, remaining * 0.25)
            a_opt = smoothed_grad_descent(a_work, time_budget=alloc, 
                                           beta_start=bs, beta_end=be,
                                           lr_start=lr_s, lr_decay=lr_d)

            print(f"  Schedule {si} (beta {bs}->{be}, lr={lr_s}): eval={ev:.8f}")
            if ev < best_score:

                a_work = a_opt / np.sum(a_opt)
    
    # Phase 2: Subgradient descent
    remaining = deadline - time.monotonic()
    if remaining > 30:
