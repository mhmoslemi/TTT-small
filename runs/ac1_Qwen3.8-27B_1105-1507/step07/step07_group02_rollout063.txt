            last_improve = it

            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            consec_no_improve += 1
            if consec_no_improve > 10:
                # Perturbation to escape
                rng = np.random.default_rng(last_improve * 31 + it)
                scale = 0.03 + 0.02 * min(consec_no_improve // 10, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a /= a.sum()
                best_obj = objective(a)
                best_a = a.copy()
                consec_no_improve = 0

    return best_a


def gradient_optimize(a, time_budget=30.0, verbose=False):
    """Gradient descent optimization with adaptive learning rate."""
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()
    lr = 0.1
    T = 0.05

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break

        a_new = gradient_step(a, lr=lr, T=T)
        obj_new = objective(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.03, 1.0)
            # Occasionally reduce T for sharper focus
            if it % 50 == 0 and T > 0.01:
                T *= 0.8
        else:
            lr *= 0.7
            if lr < 1e-5:
                # Try perturbation
                rng = np.random.default_rng(it * 7 + 3)
                a_pert = best_a * np.exp(rng.normal(0, 0.03, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
                obj_pert = objective(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    a = a_pert
                else:
                    a = best_a.copy()
                lr = 0.1
                T = max(0.01, T * 1.2)

    return best_a


def local_search(a, time_budget=30.0, verbose=False):
    """Stochastic local search with random perturbations."""
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Random perturbation: modify a random subset of elements
        n_mod = rng.integers(1, max(2, n // 100) + 1)
        indices = rng.integers(0, n, n_mod)
        scales = np.exp(rng.normal(0, 0.1, n_mod))

        a_try = best_a.copy()
        a_try[indices] *= scales
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try /= s_try

        obj_try = objective(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def gen_tilted_cosine(n, alpha=0.5):
    """Generate a tilted cosine initialization."""
    i = np.arange(n)
    a = 1.0 + alpha * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_skewed_exp(n, beta=1.0):
    """Generate a skewed exponential initialization."""
    i = np.arange(n)
    a = np.exp(-beta * i / n)
    return a / a.sum()


def gen_polynomial(n, p=2.0):
    """Generate a polynomial profile initialization."""
    i = np.arange(n)
    x = i / (n - 1)
    a = x ** p + 0.1
    return a / a.sum()


def gen_two_bump(n, c1=0.3, c2=0.7, w=0.15):
    """Generate a two-bump Gaussian initialization."""
    i = np.arange(n)
    x = i / (n - 1)
    a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + 0.3 * np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    return a / a.sum()


def gen_sinusoidal(n, k=2, phase=0.0):
    """Generate a sinusoidal modulation initialization."""
    i = np.arange(n)
    x = i / n
    a = 1.0 + 0.3 * np.cos(2 * np.pi * k * x + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def combined_optimize(a, time_budget=120.0, verbose=False):
    """
    Combined optimization: alternate between LP, gradient, and local search.
    """
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5.0:
            break

        # Phase: LP alternation (bulk of time)
        lp_time = min(remaining * 0.5, 30.0)
        a_lp = lp_alternate(a, time_budget=lp_time, verbose=verbose)
        obj_lp = objective(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5.0:
            break

        # Phase: Gradient refinement
        grad_time = min(remaining * 0.3, 20.0)
        a_grad = gradient_optimize(a, time_budget=grad_time)
        obj_grad = objective(a_grad)
        if obj_grad < best_obj:
            best_obj = obj_grad
            best_a = a_grad.copy()
            a = a_grad

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5.0:
            break

        # Phase: Local search
        ls_time = min(remaining * 0.2, 10.0)
        a_ls = local_search(a, time_budget=ls_time)
        obj_ls = objective(a_ls)
        if obj_ls < best_obj:
            best_obj = obj_ls
            best_a = a_ls.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses combined LP alternation, gradient descent,
    and local search with diverse initializations and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Phase 0: Get parent and establish baseline
    print("=== Phase 0: Baseline ===")
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr
                parent_list = [float(x) for x in parent]
                score = evaluate_sequence(parent_list)
                if score < np.inf:
                    print(f"  Parent (n={parent.size}): score={score:.8f}")
                    best_score = score
                    best_seq = parent.copy()
    except Exception as e:
        print(f"  Parent not available: {e}")

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"  Default constant (n={n0}): score={best_score:.8f}")

    print(f"  Current best: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: Optimize parent with combined method
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
        a_p = best_seq / np.sum(best_seq)

        alloc = min(remaining * 0.4, 300.0)
        a_opt = combined_optimize(a_p, time_budget=alloc, verbose=True)
        ev = evaluate_sequence([float(x) for x in a_opt])
        print(f"  Combined: score={ev:.8f}")
        if ev < best_score:
            best_score = ev
            best_seq = a_opt.copy()
            print(f"  ** New best: score={best_score:.8f} **")

    # Phase 2: Try diverse initializations at the best n
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 2: Diverse initializations (n={n_best}) ===")

        inits = [
            ("tilted_cos_0.3", gen_tilted_cosine(n_best, 0.3)),
            ("tilted_cos_0.5", gen_tilted_cosine(n_best, 0.5)),
            ("tilted_cos_0.7", gen_tilted_cosine(n_best, 0.7)),
            ("skewed_exp_0.5", gen_skewed_exp(n_best, 0.5)),
            ("skewed_exp_1.0", gen_skewed_exp(n_best, 1.0)),
            ("skewed_exp_2.0", gen_skewed_exp(n_best, 2.0)),
            ("poly_1.0", gen_polynomial(n_best, 1.0)),
            ("poly_2.0", gen_polynomial(n_best, 2.0)),
            ("poly_0.5", gen_polynomial(n_best, 0.5)),
            ("two_bump", gen_two_bump(n_best, 0.3, 0.7, 0.15)),
            ("two_bump2", gen_two_bump(n_best, 0.2, 0.8, 0.2)),
            ("sinusoidal_2", gen_sinusoidal(n_best, 2, 0)),
            ("sinusoidal_3", gen_sinusoidal(n_best, 3, np.pi / 4)),
            ("sinusoidal_5", gen_sinusoidal(n_best, 5, np.pi / 2)),
            ("reversed_parent", best_seq[::-1].copy()),
        ]

        # Also add perturbed parent
        for k in range(3):
            pert = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            pert = np.maximum(pert, 0)
            pert /= pert.sum()
            inits.append((f"perturbed_{k}", pert))

        alloc_each = min(remaining / len(inits), 40.0)
        for name, a_init in inits:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_try = a_init.copy()
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try /= s_try

            obj_init = objective(a_try)
            alloc = min(alloc_each, remaining * 0.3)
            a_opt = combined_optimize(a_try, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  {name}: init_obj={obj_init:.6f} -> score={ev:.8f} **")
            else:
                print(f"  {name}: init_obj={obj_init:.6f} -> score={ev:.8f}")

    # Phase 3: Multi-n exploration
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 3: Multi-n exploration ===")

        n_options = sorted(set([
            max(200, n_best - 200),
            max(200, n_best - 100),
            n_best + 100,
            n_best + 200,
            n_best + 400,
            500, 1000, 1200, 1500, 2000,
        ]))
        n_options = [n for n in n_options if 100 <= n <= 3000]

        alloc_each = min(remaining / max(1, len(n_options)), 60.0)
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_try = resample(best_seq, n_try)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try /= s_try

            alloc = min(alloc_each, remaining * 0.4)
            a_opt = combined_optimize(a_try, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  n={n_try}: score={ev:.8f} **")
            else:
                print(f"  n={n_try}: score={ev:.8f}")

    # Phase 4: Intensive final polish on best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 4: Final polish (n={n_best}, remaining={remaining:.0f}s) ===")

        a_final = best_seq / np.sum(best_seq)

        # Multiple rounds of combined optimization with decreasing time
        for round_idx in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.3, 60.0)
            a_pol = combined_optimize(a_final, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_pol])
            if ev < best_score:
                best_score = ev
                best_seq = a_pol.copy()
                a_final = a_pol / np.sum(a_pol)
                print(f"  Round {round_idx}: score={best_score:.8f} **")
            else:
                print(f"  Round {round_idx}: score={ev:.8f}")
                a_final = a_pol / np.sum(a_pol) if np.sum(a_pol) > 0 else a_final

        # Final local search
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_ls = local_search(a_final, time_budget=min(remaining * 0.5, 30.0))
            ev = evaluate_sequence([float(x) for x in a_ls])
            if ev < best_score:
                best_score = ev
                best_seq = a_ls.copy()
                print(f"  Local search: score={best_score:.8f} **")

    # Phase 5: Quick gradient polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_g = gradient_optimize(best_seq / np.sum(best_seq), time_budget=min(remaining * 0.5, 20.0))
        ev = evaluate_sequence([float(x) for x in a_g])
        if ev < best_score:
            best_score = ev
            best_seq = a_g.copy()
            print(f"  Gradient polish: score={best_score:.8f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
