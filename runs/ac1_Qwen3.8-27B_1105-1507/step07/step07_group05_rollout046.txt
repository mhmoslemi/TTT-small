"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses softmax gradient descent with annealing, LP polishing, and coordinate descent.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize_scalar


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2*n-1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    b = compute_conv(a_n)
    return 2.0 * n * float(np.max(b))


def softmax_gd(a, time_budget=60.0, verbose=False):
    """
    Softmax-weighted subgradient descent with annealing.
    Anneals temperature from high to low for global→local transition.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    
    # Temperature annealing schedule
    tau_start = max(best_obj / (2*n) * 0.1, 1e-6)
    tau_end = max(best_obj / (2*n) * 0.001, 1e-9)
    
    eta = 0.5
    momentum = 0.3
    v = np.zeros(n)
    stall = 0
    max_stall = 300
    
    a_flip = a[::-1]
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1.0))
        # Anneal temperature
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute b = conv(a, a)
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2*n-1]
        b_max = np.max(b)
        
        # Softmax weights
        w = np.exp((b - b_max) / tau)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros(2*n-1)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum
        
        # Update a_flip if a changed
        a_flip = a[::-1]
        
        # Compute gradient: grad[i] = sum_k w[k] * a[k-i] (valid k-i)
        # = conv(a_flip, w)[n-1 : 2n-1]
        wp = np.zeros(N)
        wp[:2*n-1] = w
        faf = rfft(a_flip, N)
        fw = rfft(wp, N)
        corr = irfft(faf * fw, N)
        grad = corr[n-1:2*n-1]
        
        # Center
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new
        
        # Evaluate
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2*n-1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-14:
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.005, 2.0)
            momentum = min(momentum + 0.0005, 0.8)
        else:
            stall += 1
            eta *= 0.97
            v *= 0.8
            if stall > max_stall:
                # Perturbation restart
                rng = np.random.default_rng(hash((it, n)) % 2**32)
                a = best_a * np.exp(rng.normal(0, 0.02 + 0.03 * progress, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.3
                stall = 0
                max_stall += 100
            if eta < 1e-10:
                break
    
    return best_a


def lp_polish(a, time_budget=60.0, verbose=False):
    """
    LP-based alternating optimization.
    Iteratively solves: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    Then blends: a_new = (1-t)*a + t*g
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    # Build convolution matrix A (2n-1) x n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    
    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        # Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0, M>=0
        # Variables: [g_0, ..., g_{n-1}, M]
        c = np.zeros(n+1)
        c[-1] = 1.0  # minimize M
        
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0  # A@g - M <= 0
        b_ub = np.zeros(2*n-1)
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0  # sum(g) = 1
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0 or res.x is None:
            break
        
        g = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g))
        if s_g < 1e-10:
            break
        g = g / s_g
        
        # Line search over blending parameter t
        best_t_obj = np.inf
        best_t = 0.0
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 1.0]:
            a_try = (1-t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-13 and best_t > 0:
            a_new = (1-best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = compute_obj(a_new)
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Rebuild A matrix
            for j in range(n):
                r = min(n, 2*n-1-j)
                A[j:j+r, j] = a[:r]
            A[:,:] = 0
            for j in range(n):
                r = min(n, 2*n-1-j)
                A[j:j+r, j] = a[:r]
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.10f}, t={best_t:.2f}, M={res.fun:.8f}")
        else:
            stall += 1
            if stall > 5:
                break
    
    return best_a
