Let m = n - k + n - 1 = 2n - 1 - k + n... this is getting confusing.

Let me just use a different approach. I'll compute the cross-correlation using FFT directly:

```python
def cross_correlation(p, a):
    """Compute cc[i] = sum_k p[k] * a[k-i] for i = 0, ..., len(p)-1."""
    M = len(p)
    N = len(a)
    L = 1
    while L < M + N - 1:
        L <<= 1
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    # Linear cross-correlation: IFFT(conj(P) * A)
    cc = np.fft.irfft(np.conj(P) * A, L)
    # cc[i] = sum_k p[k] * a[k-i] (circular, but with zero padding it's linear)
    # For i = 0, 1, ..., M-1: cc[i] is what we want (with a zero-padded to length L)
    return cc[:M]
```

Wait, I need to be more careful. The linear cross-correlation of p (length M) and a (length N) is:
cc[i] = sum_k p[k] * a[k-i] for i = -(N-1), ..., M-1

The total length is M+N-1.

Using FFT:
- Pack p into first M positions of length-L array
- Pack a into first N positions of length-L array
- IFFT(conj(FFT(p_padded)) * FFT(a_padded)) gives the circular cross-correlation
- With L >= M+N-1, this equals the linear cross-correlation
- The result has the "negative lag" part at the end (wrapped around)

So: cc_full = IFFT(conj(FFT(p, L)) * FFT(a, L))
cc_full[0:M] gives cc for i = 0, ..., M-1 (positive lags)
cc_full[L-N+1:L] gives cc for i = -(N-1), ..., -1 (negative lags)

For our case: M = 2n-1 (length of p), N = n (length of a).
We want cc[i] for i = 0, ..., n-1 (since we only need the gradient for a_0, ..., a_{n-1}).
cc[i] = sum_k p[k] * a[k-i] = sum_{m=0}^{n-1} p[m+i] * a[m]

This is cc_full[i] for i = 0, ..., n-1.

So: grad[i] = 2*cc_full[i] - p[2*i]*a[i] (if 2*i < 2n-1)

This is O(n log n) using FFT! Great.

Now, for the optimization:
- We minimize f(a) = soft_max(conv(a,a), t) subject to sum(a)=1, a>=0.
- Use projected gradient: a_new = proj(a - lr * grad)
  where proj clips to non-negative and rescales to sum 1.

Let me also add momentum for faster convergence.

**Additional idea: Use the fact that the problem is scale-invariant.**
If we scale a by c, conv(ca, ca) = c^2 * conv(a,a), sum(ca) = c*sum(a).
obj(ca) = 2n * c^2 * max(conv) / (c^2 * sum)^2 = same as obj(a).
So the objective is scale-invariant. We can work with sum(a)=1 without loss of generality.

"""
Optimized step function search using soft-max gradient descent with FFT-based
gradient computation, LP alternating refinement, and multi-n exploration.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def softmax_gradient(a, t):
    """
    Compute gradient of soft_max(conv(a,a), t) w.r.t. a.
    Uses FFT for efficiency.
    """
    n = len(a)
    conv = compute_conv(a)
    mc = np.max(conv)
    exp_conv = np.exp(t * (conv - mc))
    p = exp_conv / np.sum(exp_conv)
    
    # Compute cross-correlation cc[i] = sum_k p[k] * a[k-i] for i=0,...,n-1
    M = 2 * n - 1  # length of p
    N = n  # length of a
    L = 1
    while L < M + N - 1:
        L <<= 1
    P = rfft(p, L)
    A = rfft(a, L)
    cc = irfft(np.conj(P) * A, L)
    # cc[i] for i=0,...,n-1 is what we need
    cc_relevant = cc[:n]
    
    # grad[i] = 2*cc[i] - p[2*i]*a[i] (if 2*i < M)
    grad = 2.0 * cc_relevant.copy()
    for i in range(n):
        k = 2 * i
        if k < M:
            grad[i] -= p[k] * a[i]
    
    return grad, conv, mc

def softmax_objective(a, t):
    n = len(a)
    conv = compute_conv(a)
    mc = np.max(conv)
    exp_conv = np.exp(t * (conv - mc))
    return mc + np.log(np.sum(exp_conv)) / t

def project_to_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s

def softmax_gd(a, t_start=10.0, t_end=100.0, n_iters=5000, lr_start=0.01, lr_end=0.001, time_budget=200.0, verbose=True):
    """
    Soft-max gradient descent with adaptive temperature and learning rate.
    """
    n = len(a)
    a = project_to_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    v = np.zeros(n)  # velocity for momentum
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive temperature
        frac = it / max(n_iters, 1)
        t = t_start * (t_end / t_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute gradient
        conv = compute_conv(a)
        mc = float(np.max(conv))
        exp_conv = np.exp(t * (conv - mc))
        p = exp_conv / np.sum(exp_conv)
        
        # Cross-correlation via FFT
        M = 2 * n - 1
        L = 1
        while L < M + n - 1:
            L <<= 1
        P = rfft(p, L)
        A = rfft(a, L)
        cc = irfft(np.conj(P) * A, L)[:n]
        
        grad = 2.0 * cc
        for i in range(n):
            k = 2 * i
            if k < M:
                grad[i] -= p[k] * a[i]
        
        # Center gradient (project to sum=0 hyperplane)
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = beta * v - lr * grad
        a_new = project_to_simplex(a + v)
        
        a = a_new
        obj = 2.0 * n * mc_new if 'mc_new' in dir() else None
        # Actually recompute
        conv_new = compute_conv(a)
        obj_new = 2.0 * n * float(np.max(conv_new))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"      SoftMax GD: it={it}, t={t:.1f}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    try:
        from scipy.optimize import linprog
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception as e:
        print(f"LP error: {e}")
    
    return None, None

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
