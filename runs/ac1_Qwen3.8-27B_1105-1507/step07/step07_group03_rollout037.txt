The optimal compromise is what we're searching for.

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based alternating optimization as the primary engine, combined with
FFT-accelerated mirror descent, flat-spectrum initialization, multi-n exploration,
and simulated annealing for escaping local optima.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog


def fft_conv_self(a):
    """Compute conv(a, a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = rfft(a, L)
    return irfft(fa * fa, L)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fft_conv_self(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    return A


def solve_lp_fast(a_norm):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M) or (None, None)."""
    n = len(a_norm)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(max(res.x[n], 0))
    except Exception:
        pass
    return None, None

    """Mirror descent (exponential) with multi-peak subgradient."""
    n = len(a)

    stall = 0

            break

        conv = fft_conv_self(a)

        top_val = conv[top_idx]

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent loaded: n={best_seq.size}, obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: n=800, obj={best_score:.8f}")

    n_current = len(best_seq)

    # Phase 1: Quick mirror descent for initial refinement
    print(f"\n=== Phase 1: Mirror Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.08, 50.0)
        a_md = mirror_descent(best_seq, n_iters=5000, time_budget=alloc, K=5, alpha_init=0.12)
        obj = compute_obj(a_md)
        print(f"  MD(K=5): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()
            n_current = len(best_seq)

    # Phase 2: LP alternating - main workhorse (long)
    print(f"\n=== Phase 2: LP Alternating (n={n_current}, long) ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alloc = min(remaining * 0.30, 300.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  LP long: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            n_current = len(best_seq)

    # Phase 3: Mirror descent with different K
    print(f"\n=== Phase 3: MD with K=7 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.08, 50.0)
        a_md2 = mirror_descent(best_seq, n_iters=4000, time_budget=alloc, K=7, alpha_init=0.10)
        obj = compute_obj(a_md2)
        print(f"  MD(K=7): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md2.copy()
            n_current = len(best_seq)

    # Phase 4: LP again
    print(f"\n=== Phase 4: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(remaining * 0.20, 180.0)
        a_lp2 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp2)
        print(f"  LP2: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()
            n_current = len(best_seq)

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 150), max(200, n_best - 100), max(200, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        500, 600, 800, 1000, 1200, 1400
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)

        alloc1 = min(15.0, remaining * 0.08)
        a_opt = mirror_descent(a_try, n_iters=2000, time_budget=alloc1, K=5, alpha_init=0.12)

        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc2 = min(30.0, remaining * 0.2)
            a_opt = lp_alternating(a_opt, time_budget=alloc2)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try

    # Phase 6: Random restarts with different initializations
    print(f"\n=== Phase 6: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if r % 3 == 0:
            a_r = flat_spectrum_init(n_best, rng)
        elif r % 3 == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.04, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            a_r = smooth_random_init(n_best, rng)

        alloc1 = min(15.0, remaining * 0.12)
        a_opt = mirror_descent(a_r, n_iters=2000, time_budget=alloc1, K=5, alpha_init=0.12)

        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc2 = min(25.0, remaining * 0.3)
            a_opt = lp_alternating(a_opt, time_budget=alloc2)

        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r}: obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(best_seq)

    # Phase 7: Final intensive LP polish
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.4, 180.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_final)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()

    # Phase 8: Final mirror descent polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 50.0)
        a_final2 = mirror_descent(best_seq, n_iters=3000, time_budget=alloc, K=3, alpha_init=0.08)
        obj = compute_obj(a_final2)
        print(f"  Final MD(K=3): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_final2.copy()

    # One more short LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.5, 80.0)
        a_last = lp_alternating(best_seq, time_budget=alloc)
        obj = compute_obj(a_last)
        print(f"  Last LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_last.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
