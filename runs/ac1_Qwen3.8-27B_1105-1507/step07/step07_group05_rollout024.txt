    # For m = i+n-1: = sum_k w[i+n-1-k] * a[n-1-k]
    # Let l = n-1-k: k = n-1-l: = sum_l w[i+n-1-(n-1-l)] * a[l] = sum_l w[i+l] * a[l]
    # Yes! So grad[i] = (4n/beta) * (w * a_rev)[i+n-1]
    
    # Let me just compute it with numpy convolve for clarity, or use FFT.
    # For n=800, direct computation would be O(n^2) = 640K which is fine.
    # But let's use FFT for larger n.
    
    # Using FFT: conv(w, a_rev) has length (2n-1) + n - 1 = 3n-2
    # We want elements at indices n-1 to 2n-2.
    
    N3 = 1
    while N3 < 3 * n - 1:
        N3 <<= 1
    fw = rfft(w, N3)
    fa_rev = rfft(a_rev, N3)
    conv_result = irfft(fw * fa_rev, N3)[:3*n-2]
    
    grad = (4.0 * n / beta) * conv_result[n-1:2*n-1]
    
    return grad

def nesterov_entropic_gd(a, time_budget, beta_start=10.0, beta_end=100.0, verbose=False):
    """Nesterov accelerated gradient descent on entropic-smoothed peak objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    N3 = 1
    while N3 < 3 * n - 1:
        N3 <<= 1
    
    # Nesterov parameters
    y = a.copy()
    t = 1.0
    lr = 0.01
    
    stall = 0
    max_stall = 200
    beta = beta_start
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov step: y = x + (t_prev/(t+1)) * (x - x_prev)
        t_new = 1 + np.sqrt(1 + 4 * t * t) / 2
        momentum = t / t_new
        x_prev = a.copy()
        y = a + momentum * (a - x_prev)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            break
        y = y / s_y
        t = t_new
        
        # Compute gradient at y
        grad = entropic_gradient(y, beta, N3)
        
        # Project gradient onto sum=0 hyperplane (to maintain normalization)
        grad = grad - np.mean(grad)
        
        # Gradient step
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        # Line search (simple)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            # Adaptive learning rate
            if it % 100 == 0 and lr < 1.0:
                lr *= 1.02
            # Adaptive beta (increase over time for sharper peak)
            if it % 200 == 0:
                beta = min(beta * 1.5, beta_end)
        else:
            stall += 1
            lr *= 0.9
            if stall > max_stall:
                # Reset with perturbation
                rng = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                y = a.copy()
                t = 1.0
                lr = 0.01
                beta = beta_start
                stall = 0
                max_stall = min(max_stall + 100, 800)
            if lr < 1e-10:
                break
        
        if verbose and it % 500 == 0:
            print(f"            Nesterov it{it}: obj={best_obj:.8f}, lr={lr:.6f}, beta={beta:.1f}")
    
    return best_a

def multi_peak_subgradient(a, K=5, lr=0.01, time_budget=60.0, verbose=False):
    """Subgradient method using top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    alpha = lr
    stall = 0
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        
        # Get top K peaks
        top_k = np.argsort(b)[::-1][:K]
        
        # Average subgradient from top K peaks
        grad = np.zeros(n)
        for k_star in top_k:
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= K
        grad = grad - np.mean(grad)
        
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            alpha = min(alpha * 1.01, 0.5)
        else:
            stall += 1
            alpha *= 0.95
            if stall > 100:
                break
            if alpha < 1e-8:
                break
    
    return best_a

def structured_perturbation(a, rng):
    """Add structured sinusoidal perturbations to escape local minima."""
    n = len(a)
    mode = rng.integers(0, 4)
    if mode == 0:
        # Low frequency sinusoid
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a_new = a * (1 + 0.1 * pert)
    elif mode == 1:
        # High frequency
        freq = rng.uniform(5.0, 20.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a_new = a * (1 + 0.05 * pert)
    elif mode == 2:
        # Random smooth perturbation
        base = rng.normal(0, 1, n)
        # Smooth it
        kernel = np.exp(-np.arange(-10, 11)**2 / 10)
        kernel /= kernel.sum()
        smooth = np.convolve(base, kernel, mode='same')
        a_new = a * (1 + 0.05 * smooth)
    else:
        # Gaussian perturbation
        a_new = a * np.exp(rng.normal(0, 0.05, n))
    
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 0:
        a_new = a_new / s
    return a_new

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # We want to find tau such that sum(max(v - tau, 0)) = target_sum
    # This is equivalent to finding the (n-1)-th order statistic of v adjusted.
    # Standard algorithm:
    u = v.copy()
    total = target_sum
    # Sort in descending order
    r = np.sort(u)[::-1]
    cssv = np.cumsum(r)
    rho = np.nonzero(r * np.arange(1, n + 1) > (cssv - total))[0]

    rho = rho[-1]

    return np.maximum(u - theta, 0)

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FFT-based energy GD warmup, (2) Nesterov entropic smoothed gradient descent
with adaptive temperature, (3) aggressive LP alternating with HiGHS, (4) multi-peak
subgradient polishing, (5) structured sinusoidal perturbations for escape, (6) multi-n
exploration. All helper functions are top-level.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def project_simplex(v, target_sum=1.0):
    """Project vector v onto simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = v.copy()
    r = np.sort(u)[::-1]
    cssv = np.cumsum(r)
    idx = np.arange(1, n + 1)
    rho_mask = r * idx > (cssv - target_sum)
    if not np.any(rho_mask):
        return np.full(n, target_sum / n)
    rho = np.max(np.where(rho_mask)[0])
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(u - theta, 0.0)


def entropic_gradient(a, beta, N3):
    """
    Compute gradient of entropic-smoothed peak objective:
    f_beta(a) = (2n/beta) * log(sum_k exp(beta * (a*a)[k]))
    Returns gradient of f_beta with respect to a.
    """
    n = len(a)
    # Compute b = conv(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]

    # Softmax weights
    b_max = np.max(b)
    w = np.exp(beta * (b - b_max))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum

    # grad[i] = (4n/beta) * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = (4n/beta) * (a_rev conv w)[i + n - 1]
    a_rev = a[::-1]
    fw = rfft(w, N3)
    fa_rev = rfft(a_rev, N3)
    conv_result = irfft(fw * fa_rev, N3)[:3 * n - 2]
    grad = (4.0 * n / beta) * conv_result[n - 1:2 * n - 1]
    return grad


def energy_gradient(a, N3):
    """
    Compute gradient of (1/2)||conv(a,a)||^2 (energy objective).
    grad[i] = 2 * sum_k b[k] * a[k-i] (if valid) = 2 * (a_rev conv b)[i+n-1]
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]

    a_rev = a[::-1]
    fb = rfft(b, N3)
    fa_rev = rfft(a_rev, N3)
    conv_result = irfft(fb * fa_rev, N3)[:3 * n - 2]
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def nesterov_entropic_gd(a, time_budget=100.0, beta_start=5.0, beta_end=200.0,
                          lr_init=0.005, verbose=False):
    """Nesterov accelerated gradient descent on entropic-smoothed peak objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N3 = 1
    while N3 < 3 * n - 1:
        N3 <<= 1

    # Nesterov parameters
    y = a.copy()
    x_prev = a.copy()
    t_nesterov = 1.0
    lr = lr_init
    beta = beta_start

    stall = 0
    max_stall = 300

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov extrapolation
        t_new = 1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov) / 2.0
        momentum = t_nesterov / t_new
        y = a + momentum * (a - x_prev)
        x_prev = a.copy()
        t_nesterov = t_new

        # Ensure y is in feasible set
        y = project_simplex(y, 1.0)

        # Compute gradient at y
        grad = entropic_gradient(y, beta, N3)
        # Center gradient (to maintain sum constraint)
        grad = grad - np.mean(grad)

        # Gradient step with projection
        a_new = y - lr * grad
        a_new = project_simplex(a_new, 1.0)

        # Evaluate
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slowly increase lr if improving
            if it % 50 == 0 and lr < 0.5:
                lr *= 1.01
            # Increase beta over time (sharper approximation to max)

                beta = min(beta * 1.3, beta_end)
        else:
            stall += 1
            lr *= 0.93
            if stall > max_stall:
                # Escape: perturb best solution
                rng_local = np.random.default_rng(it * 7 + 31)
                a = best_a * np.exp(rng_local.normal(0, 0.04, n))
                a = project_simplex(a, 1.0)
                y = a.copy()
                x_prev = a.copy()
                t_nesterov = 1.0
                lr = lr_init
                beta = beta_start
                stall = 0
                max_stall = min(max_stall + 150, 1000)
            if lr < 1e-12:
                break

        if verbose and it % 500 == 0:
            print(f"              NE it{it}: obj={best_obj:.8f}, lr={lr:.6f}, beta={beta:.1f}")

def lp_alternating(a, time_budget=120.0, verbose=False):
    """LP-based alternating optimization with careful step sizes."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()

    start = time.monotonic()
    stall = 0

    for it in range(500):

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t = 0.3
        best_t_obj = np.inf
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:

            best_obj = best_t_obj
            best_a = (1.0 - best_t) * a + best_t * g_lp

            stall = 0

def multi_peak_subgradient(a, K=5, lr_init=0.005, time_budget=60.0, verbose=False):
    """Subgradient method using average of top-K peak subgradients."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    alpha = lr_init
    stall = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Get top K peaks
        if K < len(b):
            top_k = np.argsort(b)[::-1][:K]
        else:
            top_k = np.arange(len(b))

        # Average subgradient
        grad = np.zeros(n)
        for k_star in top_k:
            j_start = max(0, k_star - n + 1)
            j_end = min(n, k_star + 1)
            for i in range(j_start, j_end):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= float(len(top_k))
        grad = grad - np.mean(grad)

        a_new = a - alpha * grad
        a_new = project_simplex(a_new, 1.0)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0 and alpha < 0.3:
                alpha *= 1.02
        else:
            stall += 1
            alpha *= 0.96
            if stall > 150:
                break
            if alpha < 1e-10:
                break

        if verbose and it % 2000 == 0:
            print(f"              MP it{it}: obj={best_obj:.8f}, alpha={alpha:.6f}")

    return best_a


def structured_perturbation(a, rng):
    """Add structured perturbations to escape local minima."""
    n = len(a)
    mode = rng.integers(0, 5)
    if mode == 0:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a_new = a * (1.0 + 0.08 * pert)
    elif mode == 1:
        freq = rng.uniform(3.0, 15.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a_new = a * (1.0 + 0.04 * pert)
    elif mode == 2:
        # Smooth random perturbation
        base = rng.normal(0, 1, n)
        kernel_size = max(3, n // 50)
        kernel = np.exp(-np.arange(-kernel_size, kernel_size + 1) ** 2 / (kernel_size // 2 + 1))
        kernel /= kernel.sum()
        smooth = np.convolve(base, kernel, mode='same')
        a_new = a * (1.0 + 0.05 * smooth)
    elif mode == 3:
        # Bump perturbation
        center = rng.integers(0, n)
        width = rng.uniform(n // 20, n // 5)
        pert = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        sign = rng.choice([-1, 1])
        a_new = a * (1.0 + sign * 0.1 * pert)
    else:
        a_new = a * np.exp(rng.normal(0, 0.04, n))

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    # Phase 6: Multi-peak subgradient (exploits multiple concurrent peaks)
    print(f"\n=== Phase 6: Multi-Peak Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 50.0)
        a6 = multi_peak_subgradient(best_seq, K=5, lr_init=0.003, time_budget=alloc)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  ** Multi-peak: obj={best_score:.8f} **")
        else:
            print(f"  Multi-peak: obj={obj6:.8f}")

    # Phase 7: Structured perturbations + re-optimization
    print(f"\n=== Phase 7: Structured Perturbations ===")
    remaining = deadline - time.monotonic()
    n_perts = max(2, int(remaining / 50.0))
    n_perts = min(n_perts, 5)

    for p in range(n_perts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_p = structured_perturbation(best_seq, rng)
        alloc = min(remaining * 0.15, 40.0)
        a_opt = nesterov_entropic_gd(a_p, time_budget=alloc,
                                      beta_start=10.0, beta_end=150.0, lr_init=0.004)
        # Quick LP polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_opt = lp_alternating(a_opt, time_budget=min(remaining2 * 0.3, 20.0))

        obj_p = compute_obj(a_opt)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_opt.copy()
            print(f"  ** Pert {p}: obj={best_score:.8f} **")
        else:
            print(f"  Pert {p}: obj={obj_p:.8f}")

    # Phase 8: Multi-n exploration
    print(f"\n=== Phase 8: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr + 50, n_curr + 100, n_curr + 200,
        n_curr - 50, n_curr - 100,
        900, 1000, 1100, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_curr]
    n_options = n_options[:4]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 50.0)
        a_opt = nesterov_entropic_gd(a_try, time_budget=alloc,
                                      beta_start=5.0, beta_end=100.0, lr_init=0.005)
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_opt = lp_alternating(a_opt, time_budget=min(remaining2 * 0.25, 25.0))

        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 9: Final combined polish
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Final Nesterov with very high beta (sharp peak focus)
        alloc = min(remaining * 0.2, 40.0)
        a_f1 = nesterov_entropic_gd(best_seq, time_budget=alloc,
                                     beta_start=30.0, beta_end=500.0, lr_init=0.002)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  ** Final Nesterov: obj={best_score:.8f} **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 30.0)
            a_f2 = lp_alternating(best_seq, time_budget=alloc)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  ** Final LP: obj={best_score:.8f} **")

        # Final multi-peak
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.3, 15.0)
            a_f3 = multi_peak_subgradient(best_seq, K=3, lr_init=0.002, time_budget=alloc)
            obj_f3 = compute_obj(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  ** Final MP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
