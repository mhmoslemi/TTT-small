Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1

def simulated_annealing_polish(a, time_budget=60.0):
    """Small perturbations with simulated annealing acceptance."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    best_a = a.copy()
    start_time = time.monotonic()
    rng = np.random.default_rng(42)
    
    T = 0.01
    for it in range(10000):
        if time.monotonic() - start_time > time_budget:
            break
        
        # Perturb a random subset
        n_perturb = max(1, n // 100)
        idx = rng.integers(0, n, size=n_perturb)
        a_try = a.copy()
        a_try[idx] *= np.exp(rng.normal(0, 0.05, size=n_perturb))
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            continue
        a_try = a_try / s
        
        obj_try = compute_obj(a_try)
        delta = obj_try - best_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
        
        T *= 0.999
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
