import numpy as np
from scipy.fft import rfft, irfft
import time

def fft_conv(a, b=None):
    """FFT-based linear convolution. If b is None, computes a*a."""
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2. Assumes a is normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fft_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """Compute objective without assuming normalization."""
    n = len(a)

        return np.inf
    a_norm = a / s
    conv = fft_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, prev_g=None):
    """
    Solve LP: min M s.t. conv(a,g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def nesterov_mirror_descent(a, time_budget=200.0, verbose=False):
    """
    Nesterov-accelerated mirror descent on the simplex.
    Minimizes max_k conv(a,a)_k using top-K peak subgradients.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    y = a.copy()  # extrapolated point
    beta = 0.0   # momentum parameter, will be updated
    lr = 3.0 / (n + 1)  # initial learning rate
    K = 5  # number of peaks to target

    it = 0
    stall = 0
    while time.monotonic() - start < time_budget:
        # Compute subgradient at y
        conv_y = fft_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:K]
        peak_vals = conv_y[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Weighted subgradient
        weights = np.exp((peak_vals - max_val) * 3.0 / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * y[j_start:j_end + 1][::-1]

        # Mirror descent step on y
        log_y = np.log(np.maximum(y, 1e-15))
        log_y_new = log_y - lr * g
        log_y_new -= np.max(log_y_new)
        y_new = np.exp(log_y_new)
        y_new = y_new / y_new.sum()

        # Nesterov update
        beta = 2.0 / (np.sqrt(it + 10) + 2.0)
        a_new = (1.0 - beta) * y_new + beta * (y_new - (y - a) / max(beta, 1e-10))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        y = y_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Reset with perturbation
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                y = a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(1e-5, lr * 0.5)
                stall = 0
            else:
                lr = max(1e-6, lr * 0.99)

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    NMD it{it}: obj={best_obj:.10f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """
    Targeted coordinate descent: move mass from peak-contributing to non-contributing positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = fft_conv(a)

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False

        # Find top contributing pairs
        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        pairs.sort(reverse=True)

        # Find non-contributing positions
        contrib_set = set()
        for _, i, j in pairs:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_set and a[i] < a.max()]
        non_contrib.sort(key=lambda i: a[i])  # prefer small positions as targets

        for contrib_val, i, j in pairs[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < 1e-12:
                break
            for target in non_contrib[:8]:
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = frac * max(a[i], a[j])
                    src = i if a[i] >= a[j] else j
                    if a[src] - delta < 0:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = fft_conv(a)
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try the other peak
            conv_sorted = np.sort(conv)[::-1]
            if len(conv_sorted) > 1 and conv_sorted[1] > 0.9 * conv_sorted[0]:
                # Try second peak
                k_star2 = int(np.argsort(conv)[::-1][1])
                if k_star2 != k_star:
                    pairs2 = []
                    for i in range(n):
                        jj = k_star2 - i
                        if 0 <= jj < n:
                            pairs2.append((a[i] * a[jj], i, jj))
                    pairs2.sort(reverse=True)
                    for cv2, i, j in pairs2[:10]:
                        if cv2 < 1e-12:
                            break
                        for target in non_contrib[:5]:
                            for frac in [0.05, 0.1, 0.2]:
                                delta = frac * max(a[i], a[j])
                                src = i if a[i] >= a[j] else j
                                if a[src] - delta < 0:
                                    continue
                                a[src] -= delta
                                a[target] += delta
                                new_obj = compute_obj(a)
                                if new_obj < best_obj - 1e-14:
                                    best_obj = new_obj
                                    best_a = a.copy()
                                    conv = fft_conv(a)
                                    improved = True
                                    break
                                else:
                                    a[src] += delta
                                    a[target] -= delta
                                    break
                        if improved:
                            break

            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Inverted triangular
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # U-shaped (arc-sine-like)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()
    elif mode == 5:
        # Smooth random
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Polynomial (quadratic U)
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * x**2
        a = a / a.sum()
    elif mode == 7:
        # High-frequency oscillation
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.3 * np.sin(7 * x) + 0.1 * np.sin(13 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def parametric_search(n, time_budget=50.0):
    """Try parametric families to find good initializations."""
    best_a = np.ones(n) / n
    best_obj = compute_obj(best_a)

    x = np.linspace(0, 1, n)

    # Quadratic: a_i ∝ 1 + α*(2i/(n-1) - 1)^2
    for alpha in [0.1, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0]:
        a = 1.0 + alpha * (2 * x - 1) ** 2
        a = a / a.sum()
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Cubic
    for alpha in [0.5, 1.0, 2.0]:
        a = 1.0 + alpha * (2 * x - 1) ** 4
        a = a / a.sum()
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Exponential edges
    for sigma in [0.1, 0.2, 0.3, 0.5]:
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
        a = a / a.sum()
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Double exponential (U-shaped)
    for sigma in [0.05, 0.1, 0.15, 0.2, 0.3]:
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
        a = 1.0 / a  # invert to make U-shaped
        a = a / a.sum()
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines parametric initialization, Nesterov mirror descent,
    accelerated LP alternating, coordinate refinement, and multi-scale exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq, best_score = parametric_search(n_init, time_budget=30.0)
        print(f"Parametric init (n={len(best_seq)}): obj={best_score:.10f}")

    n_current = len(best_seq)

    # Phase 1: Nesterov Mirror Descent (fast, 150s)
    print(f"\n=== Phase 1: Nesterov Mirror Descent ===")
    remaining = deadline - time.monotonic()
    md_budget = min(remaining * 0.12, 150.0)
    if md_budget > 20:
        a_md = nesterov_mirror_descent(best_seq, time_budget=md_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  NMD: obj={best_score:.10f} **")
        else:
            print(f"  NMD: obj={obj_md:.10f} (no improvement)")

    # Phase 2: LP Alternating (main workhorse, ~600s)
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_total_budget = min(remaining * 0.45, 600.0)

    # Run LP alternating in rounds with increasing patience
    rounds = 4
    for rnd in range(rounds):
        remaining = deadline - time.monotonic()
        alloc = lp_total_budget / (rounds - rnd) * 0.9
        alloc = min(alloc, remaining * 0.6)
        if alloc < 30:
            break

        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd+1}: obj={best_score:.10f} **")
        else:
            print(f"  LP round {rnd+1}: obj={obj_lp:.10f} (no improvement)")

    # Coordinate refinement after LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        cd_budget = min(remaining * 0.05, 40.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD after LP: obj={best_score:.10f} **")

    # Another LP round after CD
    remaining = deadline - time.monotonic()
    if remaining > 60:
        lp2_budget = min(remaining * 0.2, 120.0)
        a_lp2 = lp_alternating(best_seq, time_budget=lp2_budget, verbose=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2: obj={best_score:.10f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.10f}")

    # Phase 3: Multi-n exploration
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 20, n_best - 40, n_best - 60,
        n_best + 20, n_best + 40, n_best + 60, n_best + 100,
        600, 700, 800, 900, 1000, 1100
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1200 and n != n_best]

    for n_try in n_options[:10]:
        remaining = deadline - time.monotonic()
        if remaining < 90:
            break

        a_try = resample_sequence(best_seq, n_try)
        obj_try = compute_obj(a_try)

        # Quick MD then LP
        md_alloc = min(remaining * 0.08, 40.0)
        if md_alloc > 15:
            a_md = nesterov_mirror_descent(a_try, time_budget=md_alloc)
            obj_md = compute_obj(a_md)
            if obj_md < obj_try:
                a_try = a_md
                obj_try = obj_md

        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.08, 60.0)
        if lp_alloc > 20:
            a_opt = lp_alternating(a_try, time_budget=lp_alloc, verbose=False)
            obj_opt = compute_obj(a_opt)
        else:
            a_opt = a_try
            obj_opt = obj_try

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.10f}")

    # Phase 4: Diverse restarts
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)

        # Blend with best (more weight on best for later restarts)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.15, 100.0)
        a_opt = lp_alternating(a_r, time_budget=restart_budget, verbose=False)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.10f}")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # NMD polish with more peaks
    if remaining > 30:
        md_alloc = min(remaining * 0.15, 60.0)
        a_md = nesterov_mirror_descent(best_seq, time_budget=md_alloc)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final NMD: obj={best_score:.10f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        lp_alloc = min(remaining * 0.25, 100.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_alloc = min(remaining * 0.2, 40.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.10f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more LP pass
        lp_alloc = min(remaining * 0.4, 50.0)
        a_last = lp_alternating(best_seq, time_budget=lp_alloc, verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
