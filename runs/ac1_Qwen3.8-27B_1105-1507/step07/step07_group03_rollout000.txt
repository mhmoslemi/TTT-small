Optimization combining FISTA with log-sum-exp smoothing, LP alternating,
population-based search, and multi-n exploration to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_gradient(a, t):
    """Compute gradient of LSE_t(conv(a,a)) with respect to a."""
    n = len(a)
    conv = compute_conv(a)
    # Numerical stability for softmax
    conv_shifted = conv - np.max(conv)
    p = np.exp(t * conv_shifted)
    p /= p.sum()
    # grad[j] = (2/t) * sum_k p[k] * a[k-j] * I(0<=k-j<n)
    # = (2/t) * sum_{m=0}^{n-1} p[j+m] * a[m]
    # Vectorized using Toeplitz matrix
    rows = np.arange(n)[:, None] + np.arange(n)[None, :]
    # rows has shape (n, n), rows[j, m] = j + m
    # We need p[rows[j,m]] * a[m] summed over m
    M = p[rows]  # (n, n) matrix, M[j, m] = p[j+m]
    grad = (2.0 / t) * (M @ a)
    return grad


def lse_value(a, t):
    """Compute LSE_t(conv(a,a))."""
    conv = compute_conv(a)
    conv_shifted = conv - np.max(conv)
    return np.max(conv) + np.log(np.sum(np.exp(t * conv_shifted))) / t

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a  # a[k-j] for k=j, j+1, ..., j+n-1
    return A


def solve_lp_highs(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0 using HiGHS."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix using scipy
    # A_ub @ x <= b_ub where x = [g_0, ..., g_{n-1}, M]
    # conv(a_norm, g)[k] - M <= 0 for all k
    # sum(g) = 1
    # g >= 0, M >= 0
    
    # conv(a_norm, g)[k] = sum_j a_norm[k-j] * g[j] (valid indices)
    # So the constraint is: sum_j a_norm[k-j] * g[j] - M <= 0
    
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n + 1))
    b_ub = np.zeros(n_cons)
    
    for k in range(n_cons):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a_norm[idx]
        A_ub[k, n] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c=np.zeros(n + 1)[:-1] + np.array([1.0]) * 0 + np.zeros(1),
                      A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        # Wait, the objective is minimize M, so c should be [0, 0, ..., 0, 1]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    # Fallback to cvxpy
    try:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_ub[:, :n] @ g - M <= 0, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except:
                continue
    except:
        pass
    
    return None, None
