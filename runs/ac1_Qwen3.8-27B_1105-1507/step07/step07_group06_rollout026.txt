"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines parametric initialization, smoothed L-BFGS-B, QP-based LP alternating
with Nesterov acceleration, multi-n exploration, and nonlinear peak deflation.
Key improvements over previous: QP-regularized LP updates, Nesterov acceleration,
parametric initializations, and more aggressive multi-n sweep.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import minimize, linprog, minimize_scalar


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2 (normalized to sum=1)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_val = float(np.sum(g_val))
            if s_val > 1e-10:
                return g_val / s_val, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_qp_update(a, lambda_reg):
    """
    Solve QP: min M + lambda*||g-a||^2 s.t. A@g <= M, sum(g)=1, g>=0.
    This is a convex QP that gives a regularized step.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s
    A = build_conv_matrix(a)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            objective = cp.Minimize(M + lambda_reg * cp.sum_squares(g - a))
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val
        except Exception:
            continue
    return None


def softmax_from_theta(theta):
    """Compute softmax with numerical stability."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def smoothed_obj_and_grad(theta, tau, n):
    """
    Compute smoothed objective and gradient w.r.t. log-parameters theta.
    a = softmax(theta), obj = 2n * smooth_max(conv(a), tau)
    """
    a = softmax_from_theta(theta)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)
    
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)
    
    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < 2 * n - 1:
            c[j] = np.dot(w[j:end], a[:end - j])
    
    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta


def smooth_lbfgs(a_init, tau_start, tau_end, time_budget, max_iter=200):
    """Smoothed L-BFGS-B optimization with tau annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    n_stages = 5
    taus = np.geomspace(tau_start, tau_end, n_stages)
    
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        stage_budget = min(remaining * 0.8, 80.0)
        
        def obj_func(th):
            val, _ = smoothed_obj_and_grad(th, tau, n)
            return val
        
        def grad_func(th):
            _, g = smoothed_obj_and_grad(th, tau, n)
            return g
        
        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-6, 'maxfun': 500})
            
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_obj


def nesterov_lp_alternating(a, time_budget, verbose=False):
    """
    LP alternating with Nesterov acceleration and adaptive step size.
    Forms QP: min M + lambda*||g-a||^2 s.t. A@g <= M, sum(g)=1, g>=0
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = a.copy()  # Nesterov momentum variable
    beta = 1.0
    lambda_reg = 0.01
    stall_count = 0
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        # Solve QP (or LP if lambda_reg is very small)
        if lambda_reg < 1e-8:
            g_opt, M_opt = solve_lp_minpeak(y)
            if g_opt is None:
                g_opt = best_a.copy()
        else:
            g_opt = solve_qp_update(y, lambda_reg)
            if g_opt is None:
                g_opt, M_opt = solve_lp_minpeak(y)
                if g_opt is None:
                    g_opt = best_a.copy()
        
        # Update with Nesterov
        beta_new = 0.5 * (1 + np.sqrt(1 + 4 * beta * beta))
        y_new = g_opt + (beta / beta_new) * (g_opt - y)
        y_new = np.maximum(y_new, 0)
        s_new = np.sum(y_new)
        if s_new > 1e-10:
            y_new = y_new / s_new
        else:
            y_new = g_opt.copy()
        
        # Evaluate
        obj_new = compute_obj(y_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = y_new.copy()
            stall_count = 0
            lambda_reg = max(lambda_reg * 0.9, 0.001)
        else:
            # Try direct step without Nesterov
            obj_direct = compute_obj(g_opt)
            if obj_direct < best_obj - 1e-12:
                best_obj = obj_direct
                best_a = g_opt.copy()
                stall_count = 0
                y_new = g_opt.copy()
            else:
                stall_count += 1
        
        y = y_new
        beta = beta_new
        
        # Adaptive: if stalled, perturb and reset
        if stall_count >= 15:
            rng = np.random.default_rng(it * 7919 + n)
            y = best_a * np.exp(rng.normal(0, 0.03, n))
            y = np.maximum(y, 0)
            s = np.sum(y)
            if s > 1e-10:
                y = y / s
            beta = 1.0
            lambda_reg = 0.05
            stall_count = 0
        
        if verbose and it % 10 == 0:
            print(f"        NLPA it{it}: obj={best_obj:.8f}, lambda={lambda_reg:.4f}")
    
    return best_a

def peak_deflation(a, time_budget):
    """
    Nonlinear optimization to reduce the peak convolution bin.
    Uses SLSQP to solve: min ||a' - a||^2 s.t. (a'*a')[k*] <= M*(1-delta), sum(a')=1, a'>=0
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for attempt in range(5):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        M = float(np.max(conv))
        k_star = int(np.argmax(conv))
        
        # Build the quadratic form matrix for (a'*a')[k_star]
        # (a'*a')[k_star] = sum_j a'_j * a'_{k_star - j}
        # = a'^T B a' where B[j, k_star-j] = 0.5 (symmetrized)
        B = np.zeros((n, n))
        for j in range(n):
            l = k_star - j
            if 0 <= l < n:
                B[j, l] = 0.5
                B[l, j] = 0.5
        
        target = M * (1 - 0.01)  # Try to reduce by 1%
        
        def obj_func(x):
            if time.monotonic() - start > time_budget:
                return np.inf
            return np.sum((x - best_a) ** 2)
        
        def grad_obj(x):
            return 2 * (x - best_a)
        
        # Constraint: x^T B x <= target, x >= 0, sum(x) = 1
        def constraint_quad(x):
            return target - x @ B @ x
        
        def constraint_sum(x):
            return np.sum(x) - 1
        
        cons = [
            {'type': 'ineq', 'fun': constraint_quad},
            {'type': 'eq', 'fun': constraint_sum}
        ]
        
        bounds = [(0, None)] * n
        
        try:
            res = minimize(obj_func, best_a, jac=grad_obj, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-10})
            
            if res.x is not None:
                a_new = np.maximum(res.x, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a


def lp_alternating_basic(a, time_budget, max_iter=200):
    """Basic LP alternating with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            if it % 10 == 0:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
    
    return best_a


def coordinate_descent(a, time_budget):
    """Coordinate descent targeting the peak bin."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        
        for i in range(0, n, max(1, n // 100)):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i:
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * a[i]
                    if delta > a[i] * 0.5:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
        
        if not improved:
            rng = np.random.default_rng(it * 31 + n)
            for _ in range(30):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.01 * a[i] * rng.uniform(0.1, 1.0)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining parametric init, smoothed L-BFGS-B, Nesterov LP alternating,
    peak deflation, multi-n exploration, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Parametric optimization (quick) ===
    print(f"\n=== Phase 1: Parametric Optimization ===")
    remaining = deadline - time.monotonic()
    
    param_n_values = [n_parent]
    if n_parent != 800:
        param_n_values.append(800)
    
    for n_p in param_n_values[:2]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.08, 25.0)
        a_param, obj_param = parametric_optimization(n_p, alloc)
        if obj_param < best_score:
            best_score = obj_param
            best_seq = a_param.copy()
            n_parent = n_p
            print(f"  Parametric n={n_p}: obj={best_score:.8f} **")
        else:
            print(f"  Parametric n={n_p}: obj={obj_param:.8f}")
    
    # === Phase 2: Smoothed L-BFGS-B ===
    print(f"\n=== Phase 2: Smoothed L-BFGS-B ===")
    for trial in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if trial == 0:
            a_init = best_seq.copy()
        elif trial == 1:
            a_init = best_seq * np.exp(rng.normal(0, 0.04, n_parent))
            a_init = np.maximum(a_init, 0)
            a_init = a_init / np.sum(a_init)
        elif trial == 2:
            shift = rng.uniform(0.1, 0.4)
            ramp = np.linspace(1 - shift, 1 + shift, n_parent)
            a_init = best_seq * ramp
            a_init = np.maximum(a_init, 0)
            a_init = a_init / np.sum(a_init)
        else:
            # Blend with parametric
            a_param, _ = parametric_optimization(n_parent, 5.0)
            blend = rng.uniform(0.3, 0.7)
            a_init = blend * best_seq + (1 - blend) * a_param
            a_init = a_init / np.sum(a_init)
        
        alloc = min(remaining * 0.2, 100.0)
        a_opt, obj_opt = smooth_lbfgs(a_init, tau_start=10.0, tau_end=500.0,
                                       time_budget=alloc, max_iter=150)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_parent = len(best_seq)
            print(f"  Trial {trial}: obj={best_score:.8f} **")
        else:
            print(f"  Trial {trial}: obj={obj_opt:.8f}")
    
    # === Phase 3: Nesterov LP Alternating (main heavy work) ===
    print(f"\n=== Phase 3: Nesterov LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.3, 250.0)
        a_nlp = nesterov_lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_nlp = compute_obj(a_nlp)
        if obj_nlp < best_score:
            best_score = obj_nlp
            best_seq = a_nlp.copy()
            n_parent = len(best_seq)
            print(f"  NLP-A: obj={best_score:.8f} **")
        else:
            print(f"  NLP-A: obj={obj_nlp:.8f}")
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 100), n_cur - 50, n_cur,
        n_cur + 50, n_cur + 100, n_cur + 200,
        n_cur + 400,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 3000]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if n_try == n_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        # Quick smoothed optimization
        alloc_smooth = min(remaining * 0.15, 35.0)
        a_opt, obj_opt = smooth_lbfgs(a_try, tau_start=20.0, tau_end=300.0,
                                       time_budget=alloc_smooth, max_iter=80)
        
        # Follow with NLP-A if promising
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc_lp = min(remaining * 0.15, 40.0)
            a_opt2 = nesterov_lp_alternating(a_opt, time_budget=alloc_lp)
            obj_opt = compute_obj(a_opt2)
            a_opt = a_opt2
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 5: Random restarts ===
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        init_type = r % 3
        if init_type == 0:
            slope = rng.uniform(0.5, 3.0)
            a_r = np.linspace(1.0, slope, n_best)
            a_r = a_r / np.sum(a_r)
        elif init_type == 1:
            rate = rng.uniform(0.5, 4.0)
            a_r = np.exp(np.linspace(0, rate, n_best))
            a_r = a_r / np.sum(a_r)
        else:
            blend = rng.uniform(0.3, 0.7)
            ramp = np.linspace(0.3, 1.7, n_best)
            a_r = blend * best_seq + (1 - blend) * ramp
            a_r = a_r / np.sum(a_r)
        
        alloc = min(remaining * 0.12, 30.0)
        a_opt, obj_opt = smooth_lbfgs(a_r, tau_start=15.0, tau_end=400.0,
                                       time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (type={init_type}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_opt:.8f}")
    
    # === Phase 6: Peak deflation ===
    print(f"\n=== Phase 6: Peak Deflation ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 60.0)
        a_pd = peak_deflation(best_seq, time_budget=alloc)
        obj_pd = compute_obj(a_pd)
        if obj_pd < best_score:
            best_score = obj_pd
            best_seq = a_pd.copy()
            print(f"  Peak deflation: obj={best_score:.8f} **")
        else:
            print(f"  Peak deflation: obj={obj_pd:.8f}")
    
    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_lp = nesterov_lp_alternating(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final NLP-A: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.3, 30.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 30.0)
        a_opt, obj_opt = smooth_lbfgs(best_seq, tau_start=50.0, tau_end=800.0,
                                       time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final Smooth: obj={best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
