"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses a combination of:
- LP alternating optimization (with symmetry exploitation)
- Subgradient descent with momentum
- Multi-start with diverse structured initializations
- n-sweep to find optimal length
- Simulated annealing for escape from local minima
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_work = a / s
    A = build_conv_matrix(a_work)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
        for solver in solvers:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass

    return None


def lp_alternate(a, time_budget=60.0, verbose=False):
    """LP alternating optimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    last_t = 0.3
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search over step sizes
        candidate_ts = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0.05 < last_t < 0.95:
            candidate_ts = sorted(set(candidate_ts + [max(0.05, last_t * 0.6), min(1.0, last_t * 1.4)]))

        best_t_obj = np.inf
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and stall == 0:
                pass
        else:
            stall += 1
            if stall > 8:
                # Perturb to escape
                rng_local = np.random.default_rng((int(time.monotonic() * 1000) + n) % (2**32))
                scale = 0.01 + 0.005 * min(stall // 8, 10)
                a = best_a * np.exp(rng_local.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0

    return best_a


def subgradient_optimize(a, time_budget=60.0, verbose=False):
    """
    Subgradient descent on the peak convolution with momentum and adaptive step.
    Uses the exact subgradient formula for non-negative sequences.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.002
    no_improve = 0
    rng = np.random.default_rng(12345)

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Find all near-max peak locations for robust gradient
        threshold = max_val * 0.995
        peak_mask = conv >= threshold
        n_peaks = int(np.sum(peak_mask))
        if n_peaks == 0:
            n_peaks = 1
            peak_mask[int(np.argmax(conv))] = True

        # Average subgradient over peak locations
        d = np.zeros(n)
        peak_indices = np.where(peak_mask)[0]
        for k_star in peak_indices:
            ks = int(k_star)
            lo = max(0, ks - n + 1)
            hi = min(n - 1, ks)
            for j in range(lo, hi + 1):
                d[j] += 2.0 * a[ks - j]
        d /= n_peaks

        # Center to preserve sum
        d = d - np.mean(d)

        # Momentum update
        momentum = beta * momentum + (1 - beta) * d
        momentum = momentum - np.mean(momentum)

        m_norm = np.linalg.norm(momentum)
        if m_norm < 1e-14:
            break
        direction = momentum / m_norm

        # Adaptive step: try a few sizes
        improved = False
        for step in [lr * 0.5, lr, lr * 2.0, lr * 4.0]:
            a_try = a - step * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                lr = min(lr * 1.1, 0.05)
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            lr *= 0.7
            if no_improve > 50:
                # Simulated annealing kick
                temp = 0.02 + 0.01 * min(no_improve // 50, 5)
                a = best_a * np.exp(rng.normal(0, temp, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    momentum = np.zeros(n)
                    lr = 0.002
                    no_improve = 0
                    if compute_obj(a) < best_obj:
                        best_obj = compute_obj(a)
                        best_a = a.copy()

    return best_a


def symmetric_lp_alternate(a, time_budget=60.0):
    """
    LP alternating with palindromic symmetry constraint.
    Only optimize the first half of the sequence.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Enforce symmetry
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = (a[i] + a[n - 1 - i]) / 2.0
        a_sym[n - 1 - i] = a_sym[i]
    s_sym = float(np.sum(a_sym))
    if s_sym > 1e-10:
        a = a_sym / s_sym

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 2:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Enforce symmetry on g
        g_sym = np.zeros(n)
        for i in range(half):
            g_sym[i] = (g_lp[i] + g_lp[n - 1 - i]) / 2.0
            g_sym[n - 1 - i] = g_sym[i]
        gs = float(np.sum(g_sym))
        if gs > 1e-10:
            g_sym = g_sym / gs
        else:
            break

        # Line search
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_sym
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 10:
                rng_local = np.random.default_rng(int(time.monotonic() * 10000) % (2**32))
                a = best_a + 0.02 * rng_local.normal(0, 1, n)
                a = np.maximum(a, 0)
                # Re-enforce symmetry
                for i in range(half):
                    avg = (a[i] + a[n - 1 - i]) / 2.0
                    a[i] = avg
                    a[n - 1 - i] = avg
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0

    return best_a


def structured_init(n, mode, rng):
    """Generate structured non-negative sequences for multi-start."""
    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
        return a / np.sum(a)
    elif mode == 2:
        # Gaussian bump
        x = np.arange(n)
        center = n / 2.0
        width = n / 4.0
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
        return a / np.sum(a)
    elif mode == 3:
        # Bimodal
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
        return a / np.sum(a)
    elif mode == 4:
        # Edge-peaked (U-shape)
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
        return a / np.sum(a)
    elif mode == 5:
        # Random smooth
        a = rng.uniform(0.5, 1.5, n)
        # Smooth with moving average
        k = max(3, n // 40)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        return a / np.sum(a)
    elif mode == 6:
        # Sine modulation
        x = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n)
        a = np.maximum(a, 0)
        return a / np.sum(a)
    elif mode == 7:
        # Power law
        x = np.arange(1, n + 1)
        a = x ** (-0.3)
        return a / np.sum(a)
    else:
        # Near-uniform with small noise
        a = np.ones(n) + rng.normal(0, 0.05, n)
        a = np.maximum(a, 0)
        return a / np.sum(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def polish_sequence(a, time_budget=30.0):
    """Apply a quick polish: short LP alternating + subgradient."""
    if time_budget < 5:
        return a
    alloc1 = time_budget * 0.5
    alloc2 = time_budget * 0.4
    a1 = lp_alternate(a, time_budget=alloc1)
    obj1 = compute_obj(a1)
    a2 = subgradient_optimize(a1, time_budget=alloc2)
    obj2 = compute_obj(a2)
    if obj2 < obj1:
        return a2
    return a1
