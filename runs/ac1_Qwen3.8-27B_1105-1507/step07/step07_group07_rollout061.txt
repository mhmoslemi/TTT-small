    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_peak_grad(a, k_star):
    """Gradient of conv(a,a)[k_star] w.r.t. a[i]."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]
    return g


def compute_soft_peak_grad(a, conv, tau):
    """Gradient of logsumexp(conv/tau)/tau w.r.t. a."""
    n = len(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.exp(c_scaled - c_max).sum()
    w = exp_c / sum_exp
    # g[i] = 2 * sum_k w[k] * a[k-i] / tau
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    g = 2.0 * corr / tau
    return g


def mirror_descent_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Mirror descent (exponentiated gradient) with adaptive temperature.
    Uses the actual peak subgradient for direct optimization.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Temperature schedule: from loose to tight
    tau_schedule = [5.0/n, 3.0/n, 2.0/n, 1.0/n, 0.5/n, 0.3/n, 0.2/n, 0.15/n, 0.1/n, 0.08/n]
    
    eta0 = 1.0 / n
    stall_count = 0
    
    for tau in tau_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        eta = eta0
        tau_time_start = time.monotonic()
        tau_budget = time_budget / len(tau_schedule)
        
        for it in range(50000):
            if time.monotonic() - start > time_budget:
                break
            if time.monotonic() - tau_time_start > tau_budget:
                break
            
            conv = compute_conv(a)
            curr_obj = 2.0 * n * float(np.max(conv))
            
            if curr_obj < best_obj:
                best_obj = curr_obj
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
            
            if stall_count > 200:
                eta *= 0.7
                if eta < 0.01 / n:
                    break
                stall_count = 0
            
            # Compute gradient
            if tau < 0.1 / n:
                # Use hard peak
                k_star = int(np.argmax(conv))
                g = compute_peak_grad(a, k_star)
            else:
                g = compute_soft_peak_grad(a, conv, tau)
            
            # Project gradient: subtract mean (for sum=1 constraint)
            g_centered = g - np.mean(g)
            
            # Mirror descent step
            a_new = a * np.exp(-eta * g_centered)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / np.sum(a_new)
            
            new_obj = compute_obj(a_new)
            if new_obj < curr_obj:
                a = a_new
                eta = min(eta * 1.01, 5.0 / n)
            else:
                eta *= 0.85
                if eta < 0.005 / n:
                    break
    
    return best_a


def fine_gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """
    L-BFGS-B on the smoothed objective with a tight temperature schedule.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    def fun_and_grad(xv, tau):
        xv = xv - np.max(xv)
        av = np.exp(xv)
        sv = float(np.sum(av))
        conv = compute_conv(av)
        c_scaled = conv / tau
        c_max = np.max(c_scaled)
        exp_c = np.exp(c_scaled - c_max)
        sum_exp = exp_c.sum()
        w = exp_c / sum_exp
        F = (c_max + np.log(sum_exp)) / (sv * sv)
        
        size = next_pow2(2 * n - 1)
        a_flip = av[::-1]
        fa_flip = rfft(a_flip, size)
        fw = rfft(w, size)
        conv_w_af = irfft(fw * fa_flip, size)
        corr = conv_w_af[n - 1:n - 1 + n]
        grad = 2.0 * corr / (tau * sv * sv) - 2.0 * F * av / sv
        return F, grad
    
    x = np.log(a) - np.log(a).max()
    
    # Tight temperature schedule
    taus = [2.0/n, 1.0/n, 0.5/n, 0.3/n, 0.2/n, 0.15/n, 0.1/n, 0.08/n, 0.06/n]
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = fun_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = fun_and_grad(xv, tau)
            return g
        
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 100})
        
        if res.fun < fun(x):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    fine-grad tau={tau:.4f}: obj={best_obj:.8f}")
    
    return best_a


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=10.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    if HAS_CVXPY and n <= 1200:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver_name in ['CLARABEL', 'MOSEK', 'ECOS', 'SCS', 'PDLP']:
            try:
                solver = getattr(cp, solver_name, None)
                if solver is None:
                    continue
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    
    # Fallback: scipy linprog (only for small n)
    if n <= 500:
        try:
            A = build_conv_matrix(a_norm)
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(0, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'time_limit': timeout})
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(res.x[n])
        except Exception:
            pass
    
    return None, None


def lp_alternate_step(a):
    """One LP alternating step."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_lp, M_lp = solve_lp_minpeak(a_norm, timeout=8.0)
    if g_lp is None:
        return a, compute_obj(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in [0.3, 0.5, 0.7, 0.9, 1.0, 0.4, 0.6, 0.8, 0.2, 0.1]:
        a_try = (1.0 - t) * a_norm + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a, best_obj


def lp_alternate_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating with perturbation on stall."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        a_new, obj_new = lp_alternate_step(a)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose:
                print(f"    LP step: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 5:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                scale = 0.02 + 0.02 * min(stall // 5, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
    
    return best_a


def u_shaped_init(n, alpha=1.0):
    """U-shaped initialization: large at edges, small in middle."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def cosine_window_init(n, power=2.0):
    """Cosine window initialization (inverted - U shaped)."""
    x = np.linspace(0, np.pi, n)
    a = np.abs(np.cos(x)) ** power + 0.1
    return a / np.sum(a)


def structured_init(n, rng):
    """Various structured initializations."""
    mode = rng.integers(0, 5)
    if mode == 0:
        return u_shaped_init(n, alpha=rng.uniform(0.5, 2.0))
    elif mode == 1:
        return cosine_window_init(n, power=rng.uniform(1.0, 4.0))
    elif mode == 2:
        # Periodic bumps
        freq = rng.integers(2, 8)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif mode == 3:
        # Power law edges
        x = np.linspace(0.01, 0.99, n)
        a = x ** (-0.3) + (1 - x) ** (-0.3) + 0.1
        return a / np.sum(a)
    else:
        # Random smooth
        a = np.ones(n)
        n_bumps = rng.integers(2, 6)
        for _ in range(n_bumps):
            c = rng.uniform(0, n)
            w = rng.uniform(n / 10, n / 3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-(np.arange(n) - c) ** 2 / (2 * w ** 2))
        a = np.maximum(a, 0.01)
        return a / np.sum(a)


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: mirror descent + fine gradient + LP alternating,
    with U-shaped initialization and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        best_seq = u_shaped_init(800)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped init (n=800): obj={best_score:.10f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Mirror descent on parent ===
    print(f"\n=== Phase 1: Mirror Descent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_md = mirror_descent_optimize(best_seq, time_budget=min(remaining * 0.2, 100.0), verbose=True)
        obj = compute_obj(a_md)
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()
            print(f"  After mirror descent: obj={best_score:.10f}")
        else:
            print(f"  Mirror descent: obj={obj:.10f} (no improve)")
    
    # === Phase 2: Fine gradient on parent ===
    print(f"\n=== Phase 2: Fine Gradient (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_fg = fine_gradient_optimize(best_seq, time_budget=min(remaining * 0.2, 100.0), verbose=True)
        obj = compute_obj(a_fg)
        if obj < best_score:
            best_score = obj
            best_seq = a_fg.copy()
            print(f"  After fine gradient: obj={best_score:.10f}")
        else:
            print(f"  Fine gradient: obj={obj:.10f} (no improve)")
    
    # === Phase 3: LP Alternating ===
    print(f"\n=== Phase 3: LP Alternating (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and n_parent <= 1000:
        a_lp = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.15, 80.0), verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  After LP: obj={best_score:.10f}")
        else:
            print(f"  LP: obj={obj:.10f} (no improve)")
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_candidates = sorted(set([
        max(300, n_parent - 100),
        max(300, n_parent - 50),
        n_parent,
        n_parent + 50,
        n_parent + 100,
        n_parent + 200,
        600, 700, 850, 950, 1000, 1100, 1200
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1400]
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        # Quick mirror descent
        alloc = min(remaining * 0.12, 35.0)
        a_opt = mirror_descent_optimize(a_try, time_budget=alloc * 0.6)
        obj_md = compute_obj(a_opt)
        
        # Fine gradient polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_opt2 = fine_gradient_optimize(a_opt, time_budget=min(alloc * 0.4, remaining2 * 0.3))
            obj_fg = compute_obj(a_opt2)
            if obj_fg < obj_md:
                a_opt = a_opt2
                obj_md = obj_fg
        
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_md:.10f}")
    
    # === Phase 5: Structured initializations at best n ===
    print(f"\n=== Phase 5: Structured Inits (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    
    for init_type in ['u_shaped', 'cosine', 'bumps', 'power_law']:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        
        if init_type == 'u_shaped':
            a_init = u_shaped_init(n_best, alpha=rng.uniform(0.5, 2.0))
        elif init_type == 'cosine':
            a_init = cosine_window_init(n_best, power=rng.uniform(1.0, 4.0))
        else:
            a_init = structured_init(n_best, rng)
        
        # Optimize
        alloc = min(remaining * 0.15, 40.0)
        a_opt = mirror_descent_optimize(a_init, time_budget=alloc * 0.6)
        obj = compute_obj(a_opt)
        
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_opt = fine_gradient_optimize(a_opt, time_budget=min(alloc * 0.4, remaining2 * 0.3))
            obj = compute_obj(a_opt)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** {init_type}: obj={best_score:.10f} **")
        else:
            print(f"  {init_type}: obj={obj:.10f}")
    
    # === Phase 6: Intensify on best ===
    print(f"\n=== Phase 6: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 60:
        # Mirror descent with tight tau
        alloc = min(remaining * 0.25, 100.0)
        a_md2 = mirror_descent_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_md2)
        if obj < best_score:
            best_score = obj
            best_seq = a_md2.copy()
            print(f"  Mirror descent 2: obj={best_score:.10f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Fine gradient with very tight tau
        alloc = min(remaining * 0.25, 80.0)
        a_fg2 = fine_gradient_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_fg2)
        if obj < best_score:
            best_score = obj
            best_seq = a_fg2.copy()
            print(f"  Fine gradient 2: obj={best_score:.10f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 1000:
        # LP polish
        alloc = min(remaining * 0.2, 60.0)
        a_lp2 = lp_alternate_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp2)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()
            print(f"  LP 2: obj={best_score:.10f}")
    
    # === Phase 7: Random restarts ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    n_restarts = min(5, max(2, int(remaining / 50)))
    
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            n_r = max(500, min(1200, n_best + int(rng.integers(-50, 50))))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 2:
            # Blend with U-shaped
            a_r = 0.7 * best_seq + 0.3 * u_shaped_init(n_best, rng.uniform(0.5, 2.0))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 3:

            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)
        else:

        alloc = min(remaining * 0.15, 35.0)

        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
