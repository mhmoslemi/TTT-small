def compute_conv(a):

    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)

        A[j:j+r, j] = a[:r]

def smooth_max_grad(a, tau=10.0):
    """Compute smoothed max of conv(a,a) and its gradient."""
    conv = compute_conv(a)
    # Log-sum-exp for numerical stability
    m = np.max(conv)
    shifted = conv - m
    weights = np.exp(tau * shifted) / np.sum(np.exp(tau * shifted))
    smooth_max = m + np.log(np.sum(np.exp(tau * shifted))) / tau
    
    # Gradient of smooth max w.r.t. a
    # d/d a[i] of (a*a)[k] = a[k-i] (if valid index)
    n = len(a)
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-10:
            continue
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += w * a[j]
    
    return smooth_max, grad

import numpy as np

from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

def golden_section_line_search(a, g, obj_current, n_evals=15):
    """
    Golden-section search for best t in [0,1] minimizing obj((1-t)*a + t*g).

    """

    def obj_at(t):
        at = (1.0 - t) * a + t * g
        at = np.maximum(at, 0)
        s = float(np.sum(at))
        if s < 1e-10:
            return np.inf
        at = at / s
        return compute_obj(at)

    obj_1 = obj_at(1.0)
    if obj_1 < obj_current:
        # Minimum might be at or near t=1
        lo, hi = 0.5, 1.0
    else:
        # Minimum is in [0, 1)
        obj_half = obj_at(0.5)
        if obj_half < obj_current:
            lo, hi = 0.0, 1.0
        else:
            obj_quarter = obj_at(0.25)
            if obj_quarter < obj_current:
                lo, hi = 0.0, 0.5
            else:
                return 0.0, obj_current

    # Golden section
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = obj_at(x1)
    f2 = obj_at(x2)
    evals = 3

    for _ in range(n_evals):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = obj_at(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = obj_at(x2)
        evals += 1

    best_t = 0.5 * (lo + hi)
    best_obj = min(f1, f2)
    # Check endpoints too
    f0 = obj_current
    best_t = 0.0
    best_obj = f0
    if f1 < best_obj:

    if f2 < best_obj:
        best_t, best_obj = x2, f2
    if obj_1 < best_obj:

def smoothed_gradient_optimize(a0, time_budget=60.0, verbose=False):
    """
    Optimize using smoothed max (log-sum-exp) subgradient.
    The gradient accounts for all near-peak convolution values simultaneously.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.02
    tau = 5.0  # Smoothness parameter (larger = closer to true max)

    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Log-sum-exp for smooth max
        m = float(np.max(conv))
        shifted = conv - m
        exp_shifted = np.exp(tau * shifted)
        weights = exp_shifted / np.sum(exp_shifted)
        smooth_max = m + np.log(np.sum(exp_shifted)) / tau

        # Gradient: grad[i] = sum_k weights[k] * a[k-i] (for valid indices)
        # This is the cross-correlation of weights and a
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-12:
                continue
            wk = weights[k]
            # For each i, if 0 <= k-i < n, then grad[i] += wk * a[k-i]
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                grad[i] += wk * a[k - i]

        # Center gradient to preserve sum
        grad = grad - np.mean(grad)

        # Step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.3)
            # Increase tau over time for sharper focus on the peak
            tau = min(tau * 1.01, 50.0)
        else:
            lr *= 0.9
            if lr < 1e-7:
                # Try perturbation

                        best_a = a_pert.copy()
                        a = a_pert
                    else:
                        a = best_a.copy()
                lr = 0.02
                tau = 5.0

    return best_a

def init_chirp(n, rng):
    # Real chirp (made non-negative)
    x = np.arange(n)
    freq = rng.uniform(0.5, 3.0)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x * x / (2 * n))
    a = np.maximum(a, 0.001)
    return a / np.sum(a)


def init_bimodal(n, rng):
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w = n * rng.uniform(0.05, 0.15)
    x = np.arange(n)
    a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    a += 0.01
    return a / np.sum(a)


def init_fourier_shaped(n, rng):
    """Initialize by shaping in frequency domain for low peak auto-correlation."""
    # Want |A(f)|^2 to have low-peak inverse FFT
    # Try a "raised cosine" shape in frequency domain
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    freqs = np.arange(size)
    # Raised cosine window in frequency
    beta = rng.uniform(0.5, 2.0)
    f_norm = freqs / size
    A_mag = 0.5 * (1 + np.cos(np.pi * np.abs(f_norm * 2 - 1) / 0.5))
    A_mag = np.clip(A_mag, 0, 1)
    # Add random phase
    phase = rng.uniform(0, 2 * np.pi, size)
    A = A_mag * np.exp(1j * phase)
    # Ensure Hermitian symmetry for real signal
    A[0] = np.abs(A[0])
    if size % 2 == 0:
        A[size // 2] = np.abs(A[size // 2])
    for i in range(1, size // 2 + (1 if size % 2 else 0)):
        A[size - i] = np.conj(A[i])
    a = np.fft.ifft(A).real[:n]
    a = np.maximum(a, 0.001)
    s = np.sum(a)
    if s < 1e-10:
        return init_constant(n)
    return a / s


def init_perturbed(parent, n, rng):
    """Resample parent to length n and add perturbation."""
    if len(parent) == n:
        a = parent.copy()
    else:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 0.001)
    a = a * np.exp(rng.normal(0, 0.05, n))
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return init_constant(n)
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating, smoothed gradient,
    and diverse initializations across multiple sequence lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Get parent if available
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr
                pscore = evaluate_sequence([float(x) for x in parent])
                if pscore < np.inf:
                    best_score = pscore
                    best_seq = parent.copy()
                    print(f"Parent (n={len(parent)}): score={pscore:.8f}")
                else:
                    parent = None
    except Exception:
        pass

    # Phase 0: Quick baseline
    if best_seq is None:
        a_base = np.ones(800)
        best_seq = a_base.copy()
        best_score = evaluate_sequence([1.0] * 800)
        print(f"Baseline constant n=800: score={best_score:.8f}")

    print(f"Starting best: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: Optimize parent/best with LP alternating (main optimization)
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: LP optimize best (n={n_best}, budget={remaining:.0f}s) ===")
        a_work = best_seq.copy()
        s = float(np.sum(a_work))
        if s > 0:
            a_work = a_work / s

        # Run LP alternating with most of the budget
        alloc1 = min(remaining * 0.5, 600.0)
        a_opt = lp_alternate(a_work, time_budget=alloc1, verbose=True)
        score = evaluate_sequence([float(x) for x in a_opt])
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            n_best = len(a_opt)
            print(f"  LP on best: score={score:.8f} **")
        else:
            print(f"  LP on best: score={score:.8f}")

        # Smoothed gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_sg = smoothed_gradient_optimize(best_seq / np.sum(best_seq),
                                              time_budget=min(remaining * 0.3, 120.0))
            score_sg = evaluate_sequence([float(x) for x in a_sg])
            if score_sg < best_score:
                best_score = score_sg
                best_seq = a_sg.copy()
                n_best = len(a_sg)
                print(f"  Smoothed grad: score={score_sg:.8f} **")
            else:
                print(f"  Smoothed grad: score={score_sg:.8f}")

    # Phase 2: Try diverse initializations at current best length
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Diverse inits at n={n_best} (budget={remaining:.0f}s) ===")
        inits = [
            ("constant", init_constant(n_best)),
            ("random", init_random(n_best, rng)),
            ("gauss", init_gaussian_window(n_best, rng)),
            ("chirp", init_chirp(n_best, rng)),
            ("bimodal", init_bimodal(n_best, rng)),
            ("fourier", init_fourier_shaped(n_best, rng)),
        ]
        if parent is not None:
            inits.append(("perturbed_parent", init_perturbed(parent, n_best, rng)))

        per_init_budget = min(remaining * 0.25, 60.0)
        for name, a_init in inits:
            remaining = deadline - time.monotonic()
            if remaining < per_init_budget + 5:
                break
            a_opt = lp_alternate(a_init, time_budget=per_init_budget)
            score = evaluate_sequence([float(x) for x in a_opt])
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                n_best = len(a_opt)
                print(f"  {name}: score={score:.8f} **")
            else:
                print(f"  {name}: score={score:.8f}")

    # Phase 3: Explore different lengths
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Multi-length exploration (budget={remaining:.0f}s) ===")
        n_options = sorted(set([
            max(200, n_best - 200),
            max(200, n_best - 100),
            n_best + 100,
            n_best + 200,
            400, 500, 700, 900, 1100, 1500, 2000
        ]))
        n_options = [n for n in n_options if 200 <= n <= 3000 and n != n_best]

        per_n_budget = min(remaining * 0.15, 45.0)
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < per_n_budget + 5:
                break

            # Initialize from best by resampling
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0.001)
            a_try = a_try / np.sum(a_try)

            a_opt = lp_alternate(a_try, time_budget=per_n_budget)
            score = evaluate_sequence([float(x) for x in a_opt])
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  n={n_try}: score={score:.8f} **")
            else:
                print(f"  n={n_try}: score={score:.8f}")

    # Phase 4: Intensive final polish on the best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 4: Final polish (remaining={remaining:.0f}s) ===")
        a_final = best_seq.copy()
        s = float(np.sum(a_final))
        if s > 0:
            a_final = a_final / s

        # Multiple rounds of LP with decreasing budget
        for round_idx in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.3, 60.0)
            a_pol = lp_alternate(a_final, time_budget=alloc, verbose=(round_idx == 0))
            score = evaluate_sequence([float(x) for x in a_pol])
            if score < best_score:
                best_score = score
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Polish round {round_idx}: score={score:.8f} **")
            else:
                print(f"  Polish round {round_idx}: score={score:.8f}")

        # Smoothed gradient final
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sg = smoothed_gradient_optimize(a_final, time_budget=min(remaining * 0.4, 30.0))
            score_sg = evaluate_sequence([float(x) for x in a_sg])
            if score_sg < best_score:
                best_score = score_sg
                best_seq = a_sg.copy()
                print(f"  Final smoothed grad: score={score_sg:.8f} **")
            else:
                print(f"  Final smoothed grad: score={score_sg:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
