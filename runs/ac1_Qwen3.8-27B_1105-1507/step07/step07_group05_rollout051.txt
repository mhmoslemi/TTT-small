"""
Adaptive Log-Sum-Exp Gradient Descent with Simplex Projection for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses Nesterov momentum, adaptive LSE parameter,
proper Euclidean simplex projection, LP polishing, and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective with sum(a)=1 normalization assumed."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = v.copy()
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_gradient(a, p, N_fft):
    """
    Compute LSE value and gradient for .
    LSE(a) = (1/p) * log(sum_k exp(p * conv(a,a)[k]))
    Returns (lse_value, gradient)
    """
    n = len(a)
    # Compute b = conv(a, a)
    fa = rfft(a, N_fft)
    b = irfft(fa * fa, N_fft)[:2 * n - 1]

    max_b = float(np.max(b))
    # Numerically stable LSE
    shifted = p * (b - max_b)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    W = float(np.sum(exp_shifted))
    lse_val = max_b + np.log(W) / p

    if W < 1e-300:
        return lse_val, np.zeros(n)

    # Weighted convolution: bw[k] = b[k] * w[k] / W
    bw = b * exp_shifted / W

    # Compute gradient: grad[i] = sum_k bw[k]*(a[k-i] + a[i-k])
    # Term 1: f1[i] = conv(bw, a_flip)[i + n - 1]
    # Term 2: f2[i] = conv(a, bw)[i]
    a_flip = a[::-1]

    # FFT size for convolutions of length (2n-1) + n - 1 = 3n-2
    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1

    fbw = rfft(bw, N2)
    fa_flat = rfft(a, N2)
    fa_flip = rfft(a_flip, N2)

    # conv(bw, a_flip) has length 3n-2, we need indices [n-1, 2n-2]
    conv_bw_af = irfft(fbw * fa_flip, N2)[:3 * n - 2]
    f1 = conv_bw_af[n - 1:2 * n - 1]

    # conv(a, bw) has length 3n-2, we need indices [0, n-1]
    conv_a_bw = irfft(fa_flat * fbw, N2)[:3 * n - 2]
    f2 = conv_a_bw[:n]

    grad = f1 + f2
    return lse_val, grad


def lse_gradient_descent(a, time_budget=100.0, p_start=30.0, p_end=300.0, verbose=False):
    """
    LSE-based gradient descent with Nesterov momentum and simplex projection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # FFT sizes
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1

    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)
    a_prev = a.copy()
    stall = 0
    max_stall = 300
    total_iters = 0

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        total_iters += 1
        # Adaptive p: geometric schedule
        progress = min(1.0, total_iters / 5000.0)
        p_cur = p_start * (p_end / p_start) ** progress
        p_cur = min(p_cur, 1000.0)

        # Nesterov lookahead
        y = a + momentum * (a - a_prev)
        y = project_simplex(y)

        # Compute gradient at y
        lse_val, grad = lse_gradient(y, p_cur, N_fft)

        # Center gradient (remove mean to preserve sum constraint approximately)
        grad = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - eta * grad
        a_new = y + v
        a_new = project_simplex(a_new)

        # Compute objective
        fa_new = rfft(a_new, N_fft)
        b_new = irfft(fa_new * fa_new, N_fft)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 3.0)
            momentum = min(momentum + 0.0005, 0.93)
        else:
            a_prev = a
            stall += 1
            eta *= 0.95
            v *= 0.5
            if stall > max_stall:
                # Perturb and restart
                rng_local = np.random.default_rng(total_iters)
                noise = rng_local.normal(0, 0.03, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.7
                stall = 0
                max_stall = min(max_stall + 100, 600)
            if eta < 1e-10:
                break

        # Periodic symmetrization
        if it % 500 == 499 and it > 0:
            a_sym = (a + a[::-1]) / 2.0
            s_sym = float(np.sum(a_sym))
            if s_sym > 1e-10:
                a_sym = a_sym / s_sym
            obj_sym = compute_obj(a_sym)
            if obj_sym < best_obj:
                best_obj = obj_sym
                best_a = a_sym.copy()
                a = a_sym.copy()
                a_prev = a.copy()
                v = np.zeros(n)

        if verbose and it % 500 == 0:
            remaining = time_budget - (time.monotonic() - start)
            print(f"            LSE-GD it{it}: obj={best_obj:.8f}, p={p_cur:.1f}, eta={eta:.4f}, rem={remaining:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    if n > 2000:
        return None  # Too large for LP
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    if n > 2000:
        return a
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"            LP it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 15:
                break

    return best_a


def spectral_init(n, rng):
    """Generate initialization with spectral shaping for flat auto-convolution."""
    mode = rng.integers(0, 5)
    if mode == 0:
        # Raised cosine (smooth, low-frequency concentrated)
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a + 0.1 * rng.uniform(0, 1, n)
    elif mode == 1:
        # Multi-scale oscillation
        a = np.ones(n)
        for freq in [2, 3, 5]:
            a += 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0)
    elif mode == 2:
        # Gaussian mixture
        a = np.zeros(n)
        for _ in range(rng.integers(2, 5)):
            center = rng.integers(0, n)
            width = rng.uniform(0.05, 0.2) * n
            a += np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
    elif mode == 3:
        # Inverse of a smooth window (peaked at edges)
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + (x - 0.5) ** 2 * 4)
    else:
        # Random with low-pass filter
        a = rng.uniform(0, 1, n)
        # Simple low-pass: moving average
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)

    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def refine_n(a, new_n, time_budget=30.0):
    """Optimize a sequence of new length starting from resampled a."""
    a_new = resample_sequence(a, new_n)
    s = float(np.sum(a_new))
    if s < 1e-10:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s
    return lse_gradient_descent(a_new, time_budget=time_budget, p_start=30, p_end=200)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Adaptive LSE gradient descent + LP polishing + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization with n={n_best}, score={best_score:.8f}")

    # Phase 1: Main LSE gradient descent (long, adaptive p)
    print(f"\n=== Phase 1: LSE-GD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.35, 400.0)
    a_gd = lse_gradient_descent(best_seq, time_budget=alloc, p_start=30, p_end=400, verbose=True)
    obj_gd = compute_obj(a_gd)
    if obj_gd < best_score:
        best_score = obj_gd
        best_seq = a_gd.copy()
        print(f"  ** LSE-GD: obj={best_score:.8f} **")
    else:
        print(f"  LSE-GD: obj={obj_gd:.8f} (no improvement)")

    # Phase 2: LP polishing
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60 and len(best_seq) <= 2000:
        alloc = min(remaining * 0.15, 120.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # Phase 3: Second LSE-GD round with different p schedule
    print(f"\n=== Phase 3: LSE-GD Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 150.0)
        a_gd2 = lse_gradient_descent(best_seq, time_budget=alloc, p_start=50, p_end=500)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            print(f"  ** LSE-GD2: obj={best_score:.8f} **")
        else:
            print(f"  LSE-GD2: obj={obj_gd2:.8f}")

    # Phase 4: Symmetrization + re-optimize
    print(f"\n=== Phase 4: Symmetrization ===")
    a_sym = (best_seq + best_seq[::-1]) / 2.0
    s_sym = float(np.sum(a_sym))
    if s_sym > 1e-10:
        a_sym = a_sym / s_sym
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  ** Symmetrized: obj={best_score:.8f} **")
            # Re-optimize from symmetric point
            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc = min(remaining * 0.1, 60.0)
                a_re = lse_gradient_descent(best_seq, time_budget=alloc, p_start=50, p_end=300)
                obj_re = compute_obj(a_re)
                if obj_re < best_score:
                    best_score = obj_re
                    best_seq = a_re.copy()
                    print(f"  ** Re-opt: obj={best_score:.8f} **")
        else:
            print(f"  Symmetrized: obj={obj_sym:.8f} (not better)")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr + 100, n_curr + 200, n_curr + 400, n_curr + 800,
        n_curr - 100, n_curr - 200,
        1000, 1200, 1500, 2000, 2500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 3000 and n != n_curr]
    n_options = n_options[:8]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.15, 50.0)
        a_opt = refine_n(best_seq, n_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 6: Random restarts with spectral init
    print(f"\n=== Phase 6: Random Restarts ===")
    n_use = len(best_seq)
    n_restarts = max(2, int((deadline - time.monotonic()) / 50.0))
    n_restarts = min(n_restarts, 10)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        # Try different lengths for diversity
        if r % 3 == 0:
            n_r = n_use
        elif r % 3 == 1:
            n_r = min(n_use + 200, 2500)
        else:
            n_r = max(n_use - 100, 500)

        a_r = spectral_init(n_r, rng)
        # Blend with best for warm start
        if n_r == n_use:
            a_r = 0.7 * a_r + 0.3 * best_seq
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

        alloc = min(remaining * 0.2, 40.0)
        a_opt = lse_gradient_descent(a_r, time_budget=alloc, p_start=30, p_end=250)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r} (n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}): obj={obj_r:.8f}")

    # Phase 7: Final combined polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_final = lse_gradient_descent(best_seq, time_budget=alloc, p_start=100, p_end=600)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final GD: obj={best_score:.8f} **")

    # Phase 8: Final LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 2000:
        a_lp2 = lp_polish(best_seq, time_budget=min(remaining * 0.4, 50.0))
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

    # Phase 9: Try a few more n values near the best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_best = len(best_seq)
        for dn in [50, -50, 100, -100]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            n_try = n_best + dn
            if n_try < 400 or n_try > 3000:
                continue
            alloc = min(remaining * 0.3, 25.0)
            a_try = refine_n(best_seq, n_try, time_budget=alloc)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** Fine-tune n={n_try}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
