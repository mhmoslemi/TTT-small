    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = np.fft.fft(a, L)
        # Gradient of sum_{k≠0} |A[k]|^4 w.r.t. a
        # d/da |A[k]|^4 = 4 * |A[k]|^2 * conj(A[k]) in freq domain
        # In time domain: 4 * ifft(|A|^2 .* conj(A) .* mask)
        mask = np.ones(L)
        mask[0] = 0  # Exclude DC
        grad_freq = 4.0 * (np.abs(A)**2 * np.conj(A) * mask)
        grad = np.fft.ifft(grad_freq).real[:n]
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        a_new = a - lr * v
        
        # Project onto a >= 0, sum(a) = 1
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        # Backtracking line search
        obj_new = compute_obj(a_new)
        if obj_new > best_obj * (1 + 0.01) and lr > 1e-8:
            lr *= 0.5
            a_new = a - lr * v
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            continue
        
        a = a_new
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
        
        # Occasionally increase learning rate
        if it % 1000 == 999:
            lr = min(lr * 1.2, 5.0)
    
    return best_a

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
New: L4 spectral gradient descent (minimizes sum|A|^4 = sum(conv^2) via FFT),
L4 mirror descent, improved peak-based mirror descent, LP alternating,

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    A = fft(a, L)
    return ifft(A * A, L).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2, assuming a>=0."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def l4_fft_gradient_descent(a, n_iters=15000, time_budget=120.0):
    """Minimize sum|A[k]|^4 (equiv to sum(conv^2)) using projected SGD with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    mask = np.ones(L)
    mask[0] = 0.0

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.3
    momentum = 0.9
    v = np.zeros(n)
    patience = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        grad = 4.0 * np.real(fft(mask * np.abs(A) ** 2 * np.conj(A)))[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            lr *= 0.5
            continue

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience % 500 == 0:
                lr *= 0.7
            if patience > 3000:
                # Perturb and reset
                rng = np.random.default_rng(it + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                lr = 0.3
                patience = 0
                best_obj = compute_obj(a)
                best_a = a.copy()

        if it % 5000 == 4999:
            lr = min(lr * 1.4, 1.5)

    return best_a


def l4_mirror_descent(a, n_iters=8000, time_budget=120.0):
    """Mirror descent (exponential) with L4 spectral gradient."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / np.sum(a)

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    mask = np.ones(L)
    mask[0] = 0.0

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.88
    v = np.zeros(n)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        grad = 4.0 * np.real(fft(mask * np.abs(A) ** 2 * np.conj(A)))[:n]
        grad -= np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad

        update = lr * v
        a_new = a * np.exp(-update)
        s_new = np.sum(a_new)
        if s_new > 1e-10:

            lr *= 0.97

        if it % 2000 == 1999:
            lr = min(lr * 1.3, 0.3)

    return best_a


def peak_mirror_descent(a, n_iters=8000, time_budget=120.0, n_peaks=8):
    """Mirror descent with weighted multi-peak subgradient."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.15
    momentum = 0.85
    v = np.zeros(n)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[-n_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        spread = max(0.005, (max_val - top_val[0]) * 0.8)
        weights = np.exp((top_val - max_val) / spread)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n, ki + 1)
            for i in range(lo, hi):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * a[j]

        grad -= np.mean(grad)
        v = momentum * v + (1.0 - momentum) * grad

        a_new = a * np.exp(-lr * v)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.97

        if it % 2000 == 1999:
            lr = min(lr * 1.25, 0.4)

    return best_a


def build_conv_matrix_rows(a_norm):
    """Return list of (row_index, col_indices, coefficients) for conv(a, g) rows."""
    n = len(a_norm)
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        cols = list(range(lo, hi))
        coeffs = [a_norm[k - i] for i in cols]
        rows.append((k, cols, coeffs))
    return rows


def solve_lp_minpeak(a, time_budget=30.0):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    # Build constraint matrix: (2n-1) x (n+1)
    # Variables: g[0],...,g[n-1], M
    # Constraint k: sum_i a_norm[k-i]*g[i] - M <= 0 for each k
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            A_ub[k, i] = a_norm[k - i]
        A_ub[k, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": time_budget})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, n_rounds=5, time_budget=120.0):

    n = len(a)

        # Line search on t

            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)

            if s_try < 1e-10:

        if best_t_obj < best_obj - 1e-13:

            a = np.maximum(a, 0)

def fft_flat_spectrum(a, n_iters=300, time_budget=60.0, relaxation=0.5):
    """Gerchberg-Saxton style flat spectrum iteration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-12)
        phase = A / mag_safe

        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1

                    rng = np.random.default_rng(it + n)

                    a = np.maximum(a, 0)

def coordinate_descent(a, time_budget=40.0):
    """Coordinate-wise refinement targeting the peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass away from peak contributors
        step = max(1, n // 100)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 5 * a[j] + 1e-6 and a[i] > 1e-5:
                for frac in [0.05, 0.1, 0.15]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            # Small random perturbation
            rng = np.random.default_rng(it * 31 + n)
            for _ in range(5):
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                if i1 == i2:
                    continue
                delta = 0.005 * max(a[i1], a[i2])
                a[i1] = max(0, a[i1] - delta)

                a = a / np.sum(a)
                new_conv = compute_conv(a)

                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True

        if not improved and it > 100:

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = int(rng.integers(0, 7))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Fourier features
        freqs = rng.uniform(0.5, 5, 6)
        phases = rng.uniform(0, 2 * np.pi, 6)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal Gaussian
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Periodic
        P = int(n // rng.integers(3, 8))
        if P < 2:
            P = 4
        a = np.ones(n)
        for i in range(n):
            a[i] = 1.0 + 0.5 * np.cos(2 * np.pi * i / P)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 5:
        # Smoothed random
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        # Power law decay then reflect
        a = 1.0 / (np.arange(1, n + 1) ** 0.3)
        a = np.concatenate([a, a[::-1]])[:n]
        a = a / np.sum(a)
    return a

            n_r = int(n_best + rng.integers(-200, 300))
            n_r = max(150, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.04, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 3:
            # Cosine window with perturbation
            x = np.linspace(0, np.pi, n_best)
            a_r = np.cos(x) ** 2 + 0.5 + rng.normal(0, 0.05, n_best)
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)
        else:
            # Asymmetric
            a_r = best_seq.copy()
            mid = n_best // 2
            a_r[:mid] *= 1.2 + rng.uniform(0, 0.3)
            a_r[mid:] *= 0.7 + rng.uniform(0, 0.2)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        # Optimize
        alloc1 = min(remaining * 0.1, 25.0)
        a_opt = l4_fft_gradient_descent(a_r, n_iters=3000, time_budget=alloc1)
        remaining = deadline - time.monotonic()
        if remaining < 20:
            try_update(a_opt)
            continue
        alloc2 = min(remaining * 0.08, 20.0)
        a_opt = peak_mirror_descent(a_opt, n_iters=3000, time_budget=alloc2, n_peaks=6)
        try_update(a_opt)

    # === Phase 9: Final Intensive Polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_final = best_seq.copy()

        # LP
        alloc = min(remaining * 0.25, 80.0)
        a_opt = lp_alternating(a_final, n_rounds=4, time_budget=alloc)
        try_update(a_opt)

        # L4 gradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.2, 60.0)
            a_opt = l4_fft_gradient_descent(a_final, n_iters=8000, time_budget=alloc)
            try_update(a_opt)

        # Peak mirror
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.2, 60.0)
            a_opt = peak_mirror_descent(a_final, n_iters=6000, time_budget=alloc, n_peaks=10)
            try_update(a_opt)

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.15, 40.0)
            a_opt = coordinate_descent(a_final, time_budget=alloc)
            try_update(a_opt)

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 25:
            alloc = min(remaining * 0.3, 50.0)
            a_opt = lp_alternating(best_seq, n_rounds=3, time_budget=alloc)
            try_update(a_opt)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
