import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv_fft(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp_fast(a):
    """Solve min M s.t. A@g <= M, sum(g)=1, g>=0 using HiGHS."""
    n = len(a)
    A = build_conv_matrix(a)
    c = np.zeros(n+1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*(n+1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s > 1e-10:
            return g/s, res.x[n]
    return None, None

def subgradient_at(a, conv, k_active):
    """Compute subgradient of max_k conv(a,a)[k] at a.
    k_active: indices of active bins."""
    n = len(a)
    grad = np.zeros(n)
    for k in k_active:
        # d conv[k] / d a[i] = a[k-i] if 0 <= k-i < n
        for i in range(n):

            if 0 <= j < n:
                grad[i] += a[j]
    # Normalize by number of active bins (equal weights)
    grad /= len(k_active)
    return 2 * grad  # factor of 2 from the quadratic

"""
Hybrid optimizer: LP alternating + Nesterov-accelerated subgradient + active-set refinement.
Key innovations: fast HiGHS LP, clean subgradient formula (2*shifted a), 
Nesterov O(1/t^2) acceleration, and adaptive active set tracking.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x[:n], 0)
        s = float(g.sum())
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_for_bin(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a.
    Formula: grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    # Valid range of i: max(0, k-n+1) <= i <= min(k, n-1)
    i_start = max(0, k - n + 1)
    i_end = min(k, n - 1)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def nesterov_subgradient(a0, time_budget=60.0, max_iters=20000, tol=1e-8):
    """
    Nesterov-accelerated subgradient method for minimizing max_k conv(a,a)[k]
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    start = time.monotonic()
    
    a_prev = a.copy()
    eta0 = 1.0  # initial step size
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov extrapolation
        momentum = (t - 1) / (t + 2.0)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        sy = float(y.sum())
        if sy < 1e-10:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Compute convolution and find active set
        conv_y = compute_conv(y)
        max_c = float(np.max(conv_y))
        
        # Active set: bins within 0.5% of max
        threshold = max_c * 0.995
        active = np.where(conv_y >= threshold)[0]
        
        # Compute weighted subgradient
        # Weights proportional to (conv_y[k] - second_max) for active bins
        if len(active) > 1:
            vals = conv_y[active]
            # Soft weights: emphasize the peak
            w = np.maximum(vals - vals[1] + 1e-10, 1e-10) if len(vals) > 1 else np.array([1.0])
            w = w / w.sum()
        else:
            w = np.array([1.0])
        
        grad = np.zeros(n)
        for idx, k in enumerate(active):
            grad += w[idx] * subgradient_for_bin(y, int(k))
        
        # Adaptive step size
        eta = eta0 / np.sqrt(t)
        
        # Update
        a_new = simplex_project(y - eta * grad)
        a_prev = a
        a = a_new
        
        # Evaluate
        conv_a = compute_conv(a)
        obj_a = float(np.max(conv_a))
        if obj_a < best_obj - 1e-10:
            best_obj = obj_a
            best_a = a.copy()
        
        # Convergence check
        if t > 100 and obj_a - best_obj < tol and t % 500 == 0:
            # Check if we've converged
            pass
    
    return best_a

def active_set_refine(a0, time_budget=30.0, max_iters=5000):
    """
    Refine using only the active set of convolution bins.
    At the optimum, the active bins should all be equal.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    start = time.monotonic()
    
    eta = 0.01
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Active set
        threshold = max_c * 0.99
        active = np.where(conv >= threshold)[0]
        
        # Equalize active bins: direction that reduces the max active bin
        # while increasing non-active bins
        grad = np.zeros(n)
        for k in active:
            grad += subgradient_for_bin(a, int(k))
        grad /= len(active)
        
        # Line search
        improved = False
        for t in [eta, eta * 0.5, eta * 0.1, eta * 2.0]:
            a_try = simplex_project(a - t * grad)
            conv_try = compute_conv(a_try)
            obj_try = float(np.max(conv_try))
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            eta *= 0.7
            if eta < 1e-6:
                break
        else:
            eta = min(eta * 1.1, 0.1)
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Hybrid optimizer: smooth-max gradient (via FFT-free correlate), LP alternating (HiGHS),
momentum subgradient, active-set refinement, and multi-n exploration.
Key innovation: clean O(n^2) gradient via np.correlate(p, a, 'valid'), aggressive

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x[:n], 0)
        s = float(g.sum())
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, T):
    """
    Compute gradient of T*log(sum exp(conv(a,a)/T)) w.r.t. a.
    Uses the identity: grad[i] = 2 * sum_j p[i+j] * a[j] = 2*correlate(p, a, 'valid')[i]
    where p = softmax(conv/T).

    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, p, float(max_c + T * np.log(exp_vals.sum()))

        a_new = simplex_project(a + v)

def active_set_refine(a0, time_budget=30.0, max_iters=5000):
    """
    Refine by equalizing active convolution bins.
    Uses subgradient of the max active bin with backtracking.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    eta = 0.005
    v = np.zeros(n)
    momentum = 0.8

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))

        # Active set within 0.5% of max
        threshold = max_c * 0.998
        active = np.where(conv >= threshold)[0]

        # Subgradient: weighted average of active bin gradients
        grad = np.zeros(n)
        for k in active:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(ki, n - 1)
            idx = np.arange(i_start, i_end + 1)
            grad[idx] += 2.0 * a[ki - idx]
        grad /= len(active)
        grad_centered = grad - np.mean(grad)

        # Momentum
        v = momentum * v - eta * grad_centered
        a_new = simplex_project(a + v)
        obj_new = float(np.max(compute_conv(a_new)))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.05)
        else:
            a = best_a.copy()
            v = np.zeros(n)
            eta *= 0.7
            if eta < 1e-7:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, np.maximum(a, 0))
