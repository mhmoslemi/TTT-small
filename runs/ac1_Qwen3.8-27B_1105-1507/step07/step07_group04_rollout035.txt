def solve_lp(a, warm_start=None):

    a = a / np.sum(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        try:
            g.value = warm_start
        except:
            pass
    
    for solver in [cp.PDLP, cp.SCS, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # grad at k_star: component i is 2*a[k_star - i] (if valid)
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)

            reflected = np.array([a[k_star - i] for i in indices])
        
        a_new = a.copy()
        a_new[indices] -= eps * reflected
        
        # Renormalize
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

def subgradient_nesterov(a, time_budget=100.0, top_k=10, verbose=False):
    """Nesterov-accelerated subgradient with proper projection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.7
    step = 0.01
    y = a.copy()  # Nesterov lookahead
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(y)
        k_indices = np.argsort(conv)[::-1][:top_k]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            indices = np.arange(j_start, j_end)
            grad[indices] += 2.0 * np.array([y[k - i] for i in indices])
        grad /= len(k_indices)
        
        # Center gradient
        grad = grad - np.mean(grad)
        
        # Momentum
        momentum = beta * momentum + grad
        a_new = a - step * momentum
        
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a
        
        # Nesterov update
        t_k = 2.0 / (np.sqrt(it) + 2.0)
        y = a_new + (1 - t_k) * (a_new - a)
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy
        
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive step
        if it % 200 == 199:
            step *= 0.95
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            max_frac = min(max_frac, a[indices[ii]] / (eps * reflected[ii]))

        if max_frac < 1.0:
            eps *= 0.5
            if eps < 1e-9:
                break

        a_new = a.copy()
        a_new[indices] -= eps * reflected
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"    Rake it={it}: obj={best_obj:.8f}, eps={eps:.5f}, {elapsed:.0f}s")
        else:
            eps *= 0.7
            if eps < 1e-9:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                eps = 0.005

    return best_a


def subgradient_method(a, time_budget=100.0, top_k=10, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.8
    step = 0.015
    y = a.copy()

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(y)
        k_indices = np.argsort(conv)[::-1][:top_k]

        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            idx = np.arange(j_start, j_end)
            grad[idx] += 2.0 * np.array([y[k - i] for i in idx])
        grad /= len(k_indices)
        grad = grad - np.mean(grad)

        momentum = beta * momentum + grad
        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        t_k = 2.0 / (np.sqrt(float(it)) + 2.0)
        y = a_new + (1 - t_k) * (a_new - a)
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        if it % 500 == 499:
            step *= 0.93

    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def symmetric_project(a):
    """Project a onto the symmetric subspace: a[j] = a[n-1-j]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return np.maximum(a_sym, 0)


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default (n=800): obj={best_score:.8f}")

    n0 = len(best_seq)
    a = best_seq.copy()

    # === Phase 1: Aggressive LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={n0}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(350.0, remaining * 0.28)
    a1 = lp_alternating(a, time_budget=alloc, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  LP1: {best_score:.8f} **")
    a = best_seq.copy()

    # === Phase 2: Raking Backtrack ===
    print(f"\n=== Phase 2: Raking Backtrack ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(150.0, remaining * 0.12)
        a2 = raking_backtrack(a, time_budget=alloc, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Rake: {best_score:.8f} **")
    a = best_seq.copy()

    # === Phase 3: LP again from improved point ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(200.0, remaining * 0.16)
        a3 = lp_alternating(a, time_budget=alloc, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  LP3: {best_score:.8f} **")
    a = best_seq.copy()

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.08)
        a4 = subgradient_method(a, time_budget=alloc, top_k=15, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Subgrad: {best_score:.8f} **")
    a = best_seq.copy()

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    explored_n = set()
    n_options = []
    for dn in [-100, -75, -50, -25, 0, 25, 50, 75, 100, 150, 200, 300]:
        n_try = n_best + dn
        if 100 <= n_try <= 3000 and n_try not in explored_n:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        explored_n.add(n_try)

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample(best_seq, n_try)

        # Quick LP optimization
        alloc = min(remaining * 0.12, 45.0)
        a_opt = lp_alternating(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # === Phase 6: Symmetric variant ===
    print(f"\n=== Phase 6: Symmetric Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_sym = len(best_seq)
        a_sym = symmetric_project(best_seq)
        s_sym = float(np.sum(a_sym))
        if s_sym > 1e-10:
            a_sym = a_sym / s_sym
            alloc = min(remaining * 0.1, 40.0)
            a_sym_opt = lp_alternating(a_sym, time_budget=alloc)
            obj_sym = compute_obj(a_sym_opt)
            if obj_sym < best_score:
                best_score = obj_sym
                best_seq = a_sym_opt.copy()
                print(f"  Symmetric: {best_score:.8f} **")
            else:
                print(f"  Symmetric: {obj_sym:.8f}")

    # === Phase 7: Random restarts from best ===
    print(f"\n=== Phase 7: Random Restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_best = len(best_seq)
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.02 + 0.01 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r < 1e-10:
                continue
            a_r = a_r / s_r

            alloc = min(remaining * 0.25, 25.0)
            a_opt = lp_alternating(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: {best_score:.8f} **")
            else:
                print(f"  Restart {r}: {obj_r:.8f}")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_final = lp_alternating(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
