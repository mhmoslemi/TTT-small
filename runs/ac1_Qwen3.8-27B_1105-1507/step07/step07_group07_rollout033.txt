import numpy as np

def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Convolution a*b via FFT."""
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """

    Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.

    best_obj = compute_obj(a)

    eta_min = 0.01 / n
    patience = 0

        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

                    eta = 1.0 / np.sqrt(n)
                    momentum = np.zeros(n)

def lbgfsb_optimize(a_init, time_budget=120.0, verbose=False):
    """
    L-BFGS-B optimization on log-parameterized sequence with adaptive temperature.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = np.log(a) - np.log(a).max()

    # Temperature schedule: start high, decrease geometrically
    n_temps = 8
    tau_start = 5.0 / n
    tau_end = 0.05 / n
    taus = np.geomspace(tau_start, tau_end, n_temps)

    for tau in taus:
        if time.monotonic() - start > time_budget - 5:
            break

        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val

        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g

        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12,
                                    'maxls': 50, 'iprint': -1})
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    L-BFGS tau={tau:.5f}: obj={best_obj:.10f}")
            else:
                if verbose:
                    print(f"    L-BFGS tau={tau:.5f}: obj={obj_new:.10f} (no improve)")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error: {e}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_min = j
        k_max = min(2 * n - 2, j + n - 1)
        vals = a[max(0, k_min - j):min(n, k_max - j + 1)]
        # Actually, (a*g)[k] = sum_j a[k-j]*g[j], so A[k,j] = a[k-j] if 0<=k-j<n
        for k in range(k_min, k_max + 1):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) using vectorization."""
    n = len(a)
    # A has 2n-1 rows and n columns
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For column j: A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1 (clipped to 2n-2)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, 2 * j + n)
        length = end - start
        A[start:end, j] = a[:length]
    return A


def solve_lp_minpeak(a, max_time=5.0):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)

    try:
        A = build_conv_matrix_fast(a_norm)
        # Variables: [g_0, ..., g_{n-1}, M]
        # Minimize M
        c = np.zeros(n + 1)
        c[-1] = 1.0
        # Constraints: A*g - M <= 0 (for each row)
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': max_time})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternate_step(a):
    """One step of LP alternating: solve LP, find best blend."""
    n = len(a)
    a_norm = a / np.sum(a)

    g_lp, M_lp = solve_lp_minpeak(a_norm, max_time=8.0)
    if g_lp is None:
        return a, compute_obj(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Adaptive line search
    for t in [1.0, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.9]:
        a_try = (1.0 - t) * a_norm + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def spectral_shaping(a_init, time_budget=30.0):

    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        # Compute power spectrum
        size = next_pow2(2 * n - 1)
        fa = rfft(a, size)
        power = np.abs(fa) ** 2

        # Target: flat spectrum (but keep DC component)
        target_power = np.full(size // 2 + 1, np.mean(power[1:]))
        target_power[0] = power[0]  # Keep DC

        # Interpolate between current and target
        alpha = 0.3
        new_power = (1 - alpha) * power + alpha * target_power

        # Keep phase of fa
        phase = np.angle(fa)
        new_fa = np.sqrt(new_power) * np.exp(1j * phase)

        # IFFT
        a_new = np.real(irfft(new_fa, size))[:n]

        # Soft threshold (maintain non-negativity)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multi_start_optimize(a_init, n, time_budget=30.0, seed=42):
    """Try multiple random perturbations and optimize each briefly."""
    rng = np.random.default_rng(seed)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)

    n_starts = max(3, int(time_budget / 5))
    per_start = time_budget / n_starts

    for i in range(n_starts):
        if time.monotonic() > time_budget_per_multi:
            break
        # Create perturbation
        scale = 0.1 + 0.1 * (i % 3)
        a_pert = best_a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / np.sum(a_pert)

        # Quick optimization
        a_opt = subgradient_method(a_pert, time_budget=per_start * 0.6)
        obj = compute_obj(a_opt)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines subgradient, L-BFGS-B, LP alternation, spectral shaping,
    and multi-n exploration to minimize the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
                # Also check with evaluate_sequence
                val = evaluate_sequence([float(x) for x in best_seq])
                if val < np.inf:
                    best_score_eval = val
                else:
                    best_score_eval = np.inf
                print(f"  evaluate_sequence: {best_score_eval:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
        best_score_eval = np.inf

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        best_score_eval = evaluate_sequence([float(x) for x in best_seq])
        print(f"Default init (n=800): obj={best_score:.10f}, eval={best_score_eval:.10f}")

    n_parent = len(best_seq)

    # === Phase 1: Subgradient method (fast, many iterations) ===
    print(f"\n=== Phase 1: Subgradient Descent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 90.0)
        a_sg = subgradient_method(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_sg)
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()
            print(f"  After subgradient: obj={best_score:.10f}")

    # === Phase 2: L-BFGS-B with smoothed max ===
    print(f"\n=== Phase 2: L-BFGS-B (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 150.0)
        a_lb = lbgfsb_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lb)
        if obj < best_score:
            best_score = obj
            best_seq = a_lb.copy()
            print(f"  After L-BFGS-B: obj={best_score:.10f}")

    # === Phase 3: Spectral shaping ===
    print(f"\n=== Phase 3: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.08, 30.0)
        a_spec = spectral_shaping(best_seq, time_budget=alloc)
        obj = compute_obj(a_spec)
        if obj < best_score:
            best_score = obj
            best_seq = a_spec.copy()
            print(f"  After spectral: obj={best_score:.10f}")

    # === Phase 4: LP Alternation ===
    print(f"\n=== Phase 4: LP Alternation (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)
        stall = 0
        for step in range(20):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_new, obj_new = lp_alternate_step(a_norm)
            if obj_new < best_score - 1e-10:
                best_score = obj_new
                best_seq = a_new.copy()
                a_norm = a_new / np.sum(a_new)
                stall = 0
                print(f"  LP step {step}: obj={best_score:.10f}")
            else:
                stall += 1
                if stall > 5:
                    # Perturb
                    a_norm = best_seq * np.exp(rng.normal(0, 0.05, n))
                    a_norm = np.maximum(a_norm, 0)
                    a_norm = a_norm / np.sum(a_norm)
                    stall = 0

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        n_best - 100, n_best - 50, n_best,
        n_best + 50, n_best + 100, n_best + 200,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1400]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick subgradient
        alloc = min(remaining * 0.1, 25.0)
        a_sg = subgradient_method(a_try, time_budget=alloc)
        obj = compute_obj(a_sg)

        # Quick L-BFGS
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.08, 20.0)
            a_lb = lbgfsb_optimize(a_sg, time_budget=alloc2)
            obj2 = compute_obj(a_lb)
            if obj2 < obj:
                obj = obj2
                a_sg = a_lb

        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 6: Intensify on best ===
    print(f"\n=== Phase 6: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # More L-BFGS with fine temperatures
        alloc = min(remaining * 0.25, 80.0)
        a_fine = lbgfsb_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_fine)
        if obj < best_score:
            best_score = obj
            best_seq = a_fine.copy()
            print(f"  Fine L-BFGS: obj={best_score:.10f}")

        # Subgradient polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.2, 40.0)
            a_sg2 = subgradient_method(best_seq, time_budget=alloc)
            obj = compute_obj(a_sg2)
            if obj < best_score:
                best_score = obj
                best_seq = a_sg2.copy()
                print(f"  SG polish: obj={best_score:.10f}")

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_norm = best_seq / np.sum(best_seq)
            for step in range(5):
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break
                a_new, obj_new = lp_alternate_step(a_norm)
                if obj_new < best_score - 1e-10:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_norm = a_new / np.sum(a_new)
                    print(f"  LP polish step {step}: obj={best_score:.10f}")

    # === Phase 7: Random restarts ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_restarts = min(5, max(2, int(remaining / 20)))
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if r == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
            elif r == 1:
                # Symmetric perturbation
                a_r = best_seq + 0.03 * rng.normal(0, 1, n_best)
                a_r = (a_r + a_r[::-1]) / 2.0
            elif r == 2:
                # Cosine window
                x = np.linspace(0, np.pi, n_best)
                a_r = np.cos(x / 2) ** 2 + 0.2
                a_r = a_r * np.exp(rng.normal(0, 0.05, n_best))
            elif r == 3:
                # Bimodal
                c1, c2 = n_best // 4, 3 * n_best // 4
                a_r = np.exp(-((np.arange(n_best) - c1) ** 2) / (n_best ** 2 / 30)) + \
                      np.exp(-((np.arange(n_best) - c2) ** 2) / (n_best ** 2 / 30))
            else:
                # Blended with uniform
                a_r = 0.5 * best_seq + 0.5 * np.ones(n_best) / n_best

            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r < 1e-10:
                continue
            a_r = a_r / s_r

            alloc = min(15.0, remaining * 0.5)
            a_opt = subgradient_method(a_r, time_budget=alloc)
            obj = compute_obj(a_opt)

            # Quick L-BFGS
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_opt2 = lbgfsb_optimize(a_opt, time_budget=min(10.0, remaining * 0.4))
                obj2 = compute_obj(a_opt2)
                if obj2 < obj:
                    obj = obj2
                    a_opt = a_opt2

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                n_best = len(a_opt)
                print(f"  Restart {r}: obj={best_score:.10f} **")
            else:
                print(f"  Restart {r}: obj={obj:.10f}")

    # === Phase 8: Final validation and polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        # Final subgradient
        alloc = min(remaining * 0.4, 20.0)
        a_final = subgradient_method(best_seq, time_budget=alloc)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final SG: obj={best_score:.10f}")

    # Validate with evaluate_sequence
    if best_seq is None:
        best_seq = np.ones(200) / 200

    # Ensure all values are valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        print(f"\n=== FINAL: n={len(result)}, eval_score={val:.10f}, internal={best_score:.10f} ===")
    else:
        print(f"\nWARNING: evaluate_sequence returned inf! Fallback.")
        result = [1.0] * 100
        val = evaluate_sequence(result)
        print(f"  Fallback score: {val}")

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")

    return result
