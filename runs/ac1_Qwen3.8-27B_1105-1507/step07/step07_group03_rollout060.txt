from scipy.fft import rfft, irfft, fft, ifft
from scipy.signal import correlate
from scipy.optimize import linprog
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak_scipy(a):
    """Solve LP using scipy linprog: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def solve_lp_minpeak_cvxpy(a):
    """Solve LP using cvxpy with multiple solvers."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.HIGHS, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def softmax_mirror_descent(a, n_iters=10000, time_budget=120.0, tau0=5.0, tau_max=200.0, alpha_init=0.1):
    """
    FFT-accelerated softmax-weighted mirror descent.
    Uses adaptive temperature annealing and backtracking step size.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = alpha_init
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive temperature: starts low (focus on peak), increases (flatter weighting)
        progress = min(1.0, it / (n_iters * 0.7))
        tau = tau0 + (tau_max - tau0) * progress

        conv = compute_conv(a)
        max_c = float(np.max(conv))

        # Softmax weights with numerical stability
        shifted = conv - max_c
        exp_shifted = np.exp(tau * shifted)
        w_sum = float(np.sum(exp_shifted))
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_shifted / w_sum

        # Compute gradient via correlation (FFT-accelerated internally)
        # grad[i] = sum_j a[j] * w[i+j]
        grad = correlate(w, a, mode='full')[:n]
        grad = np.maximum(grad, 0)

        # Normalize gradient for stable step size
        grad_norm = float(np.sum(grad))
        if grad_norm < 1e-15:
            grad = np.ones(n) / n
            grad_norm = 1.0
        else:
            grad = grad / grad_norm

        # Mirror descent step
        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                # Gradually increase alpha
                alpha = min(alpha * 1.01, alpha_init * 2.0)
            else:
                stall += 1
                alpha *= 0.97
                if stall > 50:
                    # Escape: perturb and reset
                    rng = np.random.default_rng(it * 13 + n)
                    scale = 0.02 + 0.01 * min(stall // 50, 3)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
                    alpha = alpha_init
        else:
            alpha *= 0.5

    return best_a


def top_k_mirror_descent(a, n_iters=5000, time_budget=120.0, k_top=20, alpha_init=0.15):
    """
    Focused mirror descent using top-k peaks with softmax weighting.
    Faster per iteration than full softmax.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = alpha_init
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top-k peak locations
        top_idx = np.argsort(conv)[-k_top:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        # Softmax weights over top-k
        temp = max(0.01, (max_val - top_val[0]) * 0.3)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        # Weighted gradient
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_idx):
            ki = int(k)
            # grad[i] += w_k * a[ki - i] for valid ki-i
            i_min = max(0, ki - (n - 1))
            i_max = min(n - 1, ki)
            if i_max >= i_min:
                j_indices = ki - np.arange(i_min, i_max + 1)
                grad[i_min:i_max + 1] += w_k * a[j_indices]

        # Normalize
        grad_sum = float(np.sum(grad))
        if grad_sum < 1e-15:
            continue
        grad = grad / grad_sum

        # Step
        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                alpha = min(alpha * 1.005, alpha_init * 1.5)
            else:
                stall += 1
                alpha *= 0.98
                if stall > 60:
                    rng = np.random.default_rng(it * 7 + n)
                    a = best_a * np.exp(rng.normal(0, 0.03, n))
                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
                    alpha = alpha_init
        else:
            alpha *= 0.5

    return best_a


def lp_alternating(a, time_budget=120.0, verbose=False):
    """LP alternating optimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak_scipy(a)
        if g_lp is None:
            g_lp, M_lp = solve_lp_minpeak_cvxpy(a)
        if g_lp is None:
            break

        # Adaptive line search
        candidate_ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0 < last_t < 1:
            candidate_ts = sorted(set(candidate_ts + [last_t * 0.5, last_t * 1.5]))
            candidate_ts = [t for t in candidate_ts if 0 < t <= 1.0]

        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"        LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng((it * 11 + n) % (2**32))
                scale = 0.02 + 0.01 * min(stall // 10, 4)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

    return best_a


def coordinate_refinement(a, time_budget=30.0):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        # Try redistributing mass away from peak contributors
        candidates = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n and i != j:
                candidates.append((i, j, a[i] * a[j]))
        candidates.sort(key=lambda x: -x[2])

        for i, j, _ in candidates[:30]:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-6:
                continue
            # Try moving some mass from i to a position that doesn't contribute to k_star
            # Find a good target: position t where (k_star - t) is out of [0, n-1]
            targets = []
            for t in range(n):
                if t == i:
                    continue
                kt = k_star - t
                if kt < 0 or kt >= n:
                    targets.append(t)
            if not targets:
                targets = [t for t in range(n) if t != i][:5]

            for t in targets[:3]:
                for frac in [0.05, 0.1, 0.15]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[t] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv)) / (float(np.sum(a)) ** 2)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[t] -= delta
                        break

        if not improved and it > 20:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=None):
    """Generate diverse initializations."""
    if mode is None:
        mode = rng.integers(0, 8)

    if mode == 0:
        # Flat
        a = np.ones(n) / n
    elif mode == 1:
        # Exponential decay
        gamma = rng.uniform(0.5, 3.0)
        x = np.arange(n) / n
        a = np.exp(-gamma * x)
        a = a / np.sum(a)
    elif mode == 2:
        # Polynomial
        p = rng.uniform(0.5, 4.0)
        x = np.arange(n) / n
        a = x ** p
        a = a / np.sum(a)
    elif mode == 3:
        # Gaussian bump
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.05, 0.3) * n
        x = np.arange(n)
        a = np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Two bumps
        c1 = rng.uniform(0.1, 0.4) * n
        c2 = rng.uniform(0.6, 0.9) * n
        w = rng.uniform(0.05, 0.2) * n
        x = np.arange(n)
        a = np.exp(-(x - c1) ** 2 / (2 * w ** 2)) + np.exp(-(x - c2) ** 2 / (2 * w ** 2))
        a = a / np.sum(a)
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / np.sum(a)
    elif mode == 6:
        # Brownian motion (smooth random)
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - a.min()
        a = a + 0.1
        a = a / np.sum(a)
    else:
        # Random Fourier
        n_freqs = rng.integers(3, 10)
        freqs = rng.uniform(0, 5, n_freqs)
        phases = rng.uniform(0, 2 * np.pi, n_freqs)
        amps = rng.uniform(0.1, 0.5, n_freqs)
        x = np.arange(n) / n
        a = np.ones(n) * 0.5
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)

    return a


def proposal_single(a, time_budget=60.0):
    """Run a sequence of optimization methods on a single sequence."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Step 1: Top-k mirror descent (fast)
    alloc = min(time_budget * 0.3, 20.0)
    a1 = top_k_mirror_descent(best_a, n_iters=3000, time_budget=alloc, k_top=15)
    obj1 = compute_obj(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()

    # Step 2: Full softmax mirror descent
    remaining = time_budget - (time_budget - alloc)
    alloc2 = min(remaining * 0.4, 20.0)
    if alloc2 > 5:
        a2 = softmax_mirror_descent(best_a, n_iters=5000, time_budget=alloc2)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Step 3: LP refinement
    remaining = time_budget - alloc - alloc2
    alloc3 = min(remaining * 0.4, 15.0)
    if alloc3 > 5:
        a3 = lp_alternating(best_a, time_budget=alloc3)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    return best_a, best_obj

    print(f"\n=== Phase 1: Main Optimization (n={n_parent}) ===")

                                        tau0=20.0, tau_max=500.0, alpha_init=0.08)
        obj = compute_obj(a_opt)
        print(f"  Final Softmax MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 50.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 30.0)
        a_cd = coordinate_refinement(best_seq, time_budget=alloc)
        obj = compute_obj(a_cd)
        print(f"  Final Coord: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 30.0)
        a_md = top_k_mirror_descent(best_seq, n_iters=5000, time_budget=alloc,
                                     k_top=30, alpha_init=0.1)
        obj = compute_obj(a_md)
        print(f"  Final Top-k MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Previous: 1.506362 ===")

    return result
