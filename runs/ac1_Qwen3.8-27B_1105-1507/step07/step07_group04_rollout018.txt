So the problem is non-convex. The LP alternating is a reasonable heuristic.

"""
Optimized solver for minimum peak autocorrelation problem.
Uses smooth gradient descent with annealing, LP-based best response,

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient(a, eps):
    """Gradient of 2n * LSE(conv(a,a)/eps) for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    exp_vals = np.exp((conv - max_val) / eps)
    weights = exp_vals / np.sum(exp_vals)
    
    # grad[i] = 4n * sum_k weights[k] * a[k-i] * I(0 <= k-i < n)
    grad = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        if k_start < k_end:
            grad[i] = np.dot(weights[k_start:k_end], a[:k_end - k_start])
    return 4.0 * n * grad


def smooth_gradient_fft(a, eps):
    """Faster gradient using FFT-based cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    exp_vals = np.exp((conv - max_val) / eps)
    weights = exp_vals / np.sum(exp_vals)
    
    # cross[i] = sum_j a[j] * weights[i+j] for valid i+j
    # = ifft(conj(fft(a, N)) * fft(weights, N))[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    A = rfft(a, N)
    W = rfft(weights, N)
    cross = irfft(A * np.conj(W), N)[:n]
    
    return 4.0 * n * cross


def gradient_descent_smooth(a, time_budget=100.0, eps_start=0.01, eps_end=0.001, verbose=False):
    """Projected gradient descent with Nesterov acceleration and epsilon annealing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov parameters
    x = a.copy()  # lookahead
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    total_steps = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(elapsed / time_budget, 1.0)
        
        # Anneal epsilon
        eps = eps_start * (eps_end / eps_start) ** progress
        
        # Compute gradient at lookahead point
        grad = smooth_gradient(x, eps)
        
        # Center gradient (remove mean to preserve sum=1 approximately)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov momentum
        momentum = beta * momentum - lr * grad_centered
        new_a = a + momentum
        
        # Project onto simplex
        new_a = project_simplex(new_a)
        
        # Check objective
        obj = compute_obj(new_a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = new_a.copy()
        
        # Update Nesterov lookahead
        t_k = 2.0 / (total_steps + 2.0)
        x = new_a + (1 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx
        
        a = new_a
        total_steps += 1
        
        # Adaptive learning rate
        if total_steps % 200 == 0:
            lr = max(lr * 0.9, 0.001)
        
        if verbose and total_steps % 500 == 0:
            print(f"    GD: step={total_steps}, obj={best_obj:.8f}, eps={eps:.6f}, lr={lr:.6f}, t={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    # Wait, this isn't right. Let me think again.
    # conv(a,g)[k] = sum_j a[j] * g[k-j] (for 0 <= k-j < n)
    # = sum_j a[k-j] * g[j] (for 0 <= k-j < n, i.e., j in [k-n+1, k] ∩ [0, n-1])
    # So A[k,j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build convolution matrix using structured approach."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Column j: A[k, j] = a[k-j] for k in [j, j+n-1]
        A[j:j+n, j] = a
    return A


def solve_lp_best_response(a, solver=None):
    """
    Given fixed a (normalized), find g >= 0, sum(g)=1 that minimizes max_k conv(a,g)[k].
    Returns (g, M_opt).
    """
    n = len(a)
    A = build_conv_matrix_fast(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers
    
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value).flatten(), 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])

        pass
    
    return None, None

def structured_init(n, seed=42):
    """Generate structured initial sequences."""
    rng = np.random.default_rng(seed)
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Raised cosine
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(rc / np.sum(rc))
    
    # Gaussian
    x = np.linspace(-1, 1, n)
    g = np.exp(-x**2 / (2 * 0.3**2))
    candidates.append(g / np.sum(g))
    
    # Exponential decay from center
    center = n // 2
    idx = np.arange(n)
    exp_dec = np.exp(-np.abs(idx - center) / (n * 0.1))
    candidates.append(exp_dec / np.sum(exp_dec))
    
    # Random (smooth)
    r = rng.random(n)
    # Smooth it
    kernel = np.ones(10) / 10
    r_smooth = np.convolve(r, kernel, mode='same')
    candidates.append(r_smooth / np.sum(r_smooth))
    
    # Triangular
    tri = 1 - np.abs(2 * x - 1)
    candidates.append(tri / np.sum(tri))
    
    # Cosine taper
    cos_t = np.sin(np.pi * x) ** 2
    candidates.append(cos_t / np.sum(cos_t))
    
    return candidates


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized solver for minimum peak autocorrelation.
Combines smooth gradient descent with epsilon annealing, LP-based best response,
active-set coordinate refinement, and multi-length exploration.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import linprog


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n <= 1:
        if n == 1:
            return np.array([1.0])
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient_fft(a, eps):
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    exp_vals = np.exp((conv - max_val) / eps)
    weights = exp_vals / np.sum(exp_vals)
    
    N = 1
    while N < 3 * n:
        N <<= 1
    A = rfft(a, N)
    W = rfft(weights, N)
    cross = irfft(np.conj(A) * W, N)[:n]
    
    return 4.0 * n * cross


def gradient_descent_smooth(a, time_budget=100.0, eps_start=0.01, eps_end=0.001, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    total_steps = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(elapsed / time_budget, 1.0)
        eps = eps_start * (eps_end / eps_start) ** progress
        
        grad = smooth_gradient_fft(x, eps)
        grad_centered = grad - np.mean(grad)
        
        momentum = beta * momentum - lr * grad_centered
        new_a = a + momentum
        new_a = project_simplex(new_a)
        
        obj = compute_obj(new_a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = new_a.copy()
        
        t_k = 2.0 / (total_steps + 2.0)
        x = new_a + (1 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx
        
        a = new_a
        total_steps += 1
        
        if total_steps % 200 == 0:
            lr = max(lr * 0.92, 0.002)
        
        if verbose and total_steps % 500 == 0:
            print(f"    GD: step={total_steps}, obj={best_obj:.8f}, eps={eps:.6f}, t={elapsed:.1f}s")
    
    return best_a
