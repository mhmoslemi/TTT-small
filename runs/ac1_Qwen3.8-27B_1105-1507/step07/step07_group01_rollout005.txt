"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, reduced_k=0):
    """
    Solve LP: min M s.t. (A@g)[k] <= M for all k, sum(g)=1, g>=0.
    If reduced_k > 0, only enforce top-k tightest constraints.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    n_constraints = 2 * n - 1

    if reduced_k > 0 and reduced_k < n_constraints:
        # Use only top-k constraints (where conv(a,a) is largest)
        conv_aa = A @ a
        top_idx = np.argsort(conv_aa)[-reduced_k:]
        A_use = A[top_idx, :]
        n_con = len(top_idx)
    else:
        A_use = A
        n_con = n_constraints

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_con, n + 1))
    A_ub[:, :n] = A_use
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_con)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            g = g / s
            # Verify M
            conv_ag = A @ g
            M = float(np.max(conv_ag))
            return g, M
    return None, None


def lp_alternating(a, time_budget=120.0, verbose=False):
    """
    LP alternating optimization with backtracking line search.
    Uses reduced LP after initial convergence for speed.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_t = 0.5
    reduced = False

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP (full initially, reduced after 50 iterations)
        if it < 50:
            g_lp, M_lp = solve_lp(a, reduced_k=0)
        else:
            g_lp, M_lp = solve_lp(a, reduced_k=min(100, n))

        if g_lp is None:
            break

        # Backtracking line search
        best_t_obj = best_obj
        best_t_val = 0.3
        best_a_local = best_a.copy()

        # Try geometrically spaced t values
        t_candidates = [min(1.0, last_t * 0.5), min(1.0, last_t * 0.75),
                        last_t, min(1.0, last_t * 1.5), min(1.0, last_t * 2.0),
                        0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        t_candidates = sorted(set(t for t in t_candidates if 0 < t <= 1.0))

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall_count = 0
            if verbose and it % 20 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall_count += 1
            # Switch to reduced LP after some stalling
            if stall_count > 20 and not reduced:
                reduced = True
            if stall_count > 50:
                # Perturb to escape
                rng = np.random.default_rng((it * 7919 + n) % (2**32))
                scale = 0.02 * min(1 + stall_count // 50, 5.0)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                stall_count = 0
                last_t = 0.5

    return best_a


def smooth_max_gradient_descent(a, time_budget=60.0, T_init=0.001, T_min=0.00005):
    """
    Gradient descent using smooth max (log-sum-exp) with Adam optimizer.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_hat = np.zeros(n)
    t_adam = 0

    T = T_init
    stall = 0

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient: grad[i] = sum_j weights[i+j] * a[j]
        grad = np.correlate(weights, a, mode='valid')

        # Center for simplex constraint
        grad_centered = grad - np.mean(grad)

        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad_centered
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_centered ** 2)
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat_corrected = v_hat / (1 - beta2 ** t_adam)
        a_new = a - lr * m_hat / (np.sqrt(v_hat_corrected) + eps)

        a_new = simplex_project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.5
                m = np.zeros(n)
                v_hat = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break

        # Temperature anneal
        if it > 100:
            T = max(T_min, T_init * (0.995 ** it))

    return best_a


def coordinate_descent_polish(a, time_budget=30.0):
    """
    Coordinate descent: for each coordinate, try small perturbations
    that reduce the maximum convolution value.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Try each coordinate
        for i in range(n):
            # Try moving mass from i to j (where j helps reduce the peak)
            # Simple: try decreasing a[i] and increasing a[j]
            # The gradient of conv[k_star] w.r.t. a[i] is a[k_star - i] (if valid)
            grad_i = 0.0
            j = k_star - i
            if 0 <= j < n:
                grad_i = a[j]

            if grad_i > 0:
                # Decreasing a[i] helps reduce conv[k_star]
                # But we need to add mass somewhere else
                # Try a few candidates for j
                for delta in [0.001, 0.005, 0.01]:
                    a_try = a.copy()
                    a_try[i] -= delta
                    if a_try[i] < 0:
                        continue
                    # Add to coordinate with smallest gradient contribution
                    # Try spreading to a coordinate far from the peak
                    j_add = (i + n // 2) % n
                    a_try[j_add] += delta
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        conv = compute_conv(a)
                        k_star = int(np.argmax(conv))
                        break

        if not improved and it > 100:
            break

    return best_a


def spectral_init(n, rng):
    """
    Initialize using flat power spectrum idea.
    Generate a sequence whose autocorrelation is close to flat.
    """
    # Start with random phase (gives flat power spectrum for circular case)
    x = np.zeros(2 * n)
    x[:n] = np.ones(n) / n
    # Add random perturbation
    x[:n] += rng.normal(0, 0.1, n)
    x = np.maximum(x, 0)
    s = np.sum(x)
    if s > 0:
        x = x / s
    return x


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence normalized to sum=1."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Edge-concentrated
        x = np.arange(n)
        a = np.exp(-0.5 * (x / (n * 0.15)) ** 2) + np.exp(-0.5 * ((x - (n-1)) / (n * 0.15)) ** 2)
        a = a / np.sum(a)
    elif mode == 3:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        a = a / np.sum(a)
    elif mode == 4:
        # Triangle
        x = np.arange(n) / max(1, n - 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    elif mode == 5:
        a = rng.dirichlet(np.ones(n) * 0.5)
    else:
        a = np.ones(n) / n
    return a

        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
