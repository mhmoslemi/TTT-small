```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n + 1) - (cssv - 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def compute_conv(a):
    """Compute auto-convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def convolve_ab(a, b):
    """Compute convolution of a and b using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1
    while size < n_out:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n_out]


def compute_obj(a):
    """Compute the objective 2n * max(conv(a,a)) for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.argmax(cond))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgrad_rows(a, g, K):
    """
    Compute the average subgradient of f(g) = max_k (a*g)_k at point g.
    Uses top-K peak positions for a smoothed subgradient.
    """
    n = len(a)
    conv = convolve_ab(a, g)
    if K >= len(conv):
        indices = np.arange(len(conv))
    else:
        indices = np.argsort(conv)[::-1][:K]
    sigma = np.zeros(n)
    for k in indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            d_start = k - j_end
            d_end = k - j_start
            sigma[j_start:j_end + 1] += a[d_start:d_end + 1][::-1]
    sigma /= float(len(indices))
    return sigma, conv

            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent: swap mass between pairs to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from positions that contribute to the peak
        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break
            # Find top contributors to peak k_star
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            contrib = a[j_start:j_end + 1] * a[k_star - j_end:k_star - j_start + 1][::-1]
            if len(contrib) == 0:
                break
            top_i = j_start + int(np.argmax(contrib))

            # Move small amount to a random other position
            others = [j for j in range(n) if j != top_i]
            if not others:
                break
            rng = np.random.default_rng(int(time.monotonic() * 100000) % (2**31))
            dest = int(rng.choice(others))

            for frac in [0.01, 0.02, 0.05]:
                delta = frac * a[top_i]
                if delta > a[top_i] * 0.9:
                    continue
                a[top_i] -= delta
                a[dest] += delta
                new_obj = 2.0 * n * float(np.max(compute_conv(a)))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[top_i] += delta
                    a[dest] -= delta
                    break

        if not improved and it > 100:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, rng):
    """Generate a structured initialization."""
    # Option 1: cosine-tapered uniform
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2.0) ** 2
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        return a / s

    # Option 2: random
    a = rng.random(n)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Fast Subgradient Alternating + LP Polish + Multi-Length Exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent_norm = parent_arr / s
                best_seq = parent_norm.copy()
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_norm.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Main Subgradient Alternating on parent length (400s) ===
    print(f"\n=== Phase 1: Subgradient Alternating (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(400.0, remaining * 0.25)

    a = best_seq.copy()
    for K_val in [3, 5, 10]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(phase1_budget / 3.0, remaining * 0.4)
        a_new = alternating_subgrad(a, time_budget=alloc, inner_iters=3000, K=K_val, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Subgrad K={K_val}: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad K={K_val}: obj={obj:.8f}")

    # === Phase 2: LP Polish (300s) ===
    print(f"\n=== Phase 2: LP Polish ===")
    remaining = deadline - time.monotonic()
    phase2_budget = min(300.0, remaining * 0.2)

    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(phase2_budget / max(3 - round_idx, 1), remaining * 0.5)
        a_new = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  LP Round {round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP Round {round_idx}: obj={obj:.8f}")

    # === Phase 3: More Subgradient with different params (300s) ===
    print(f"\n=== Phase 3: Extended Subgradient ===")
    remaining = deadline - time.monotonic()
    phase3_budget = min(300.0, remaining * 0.2)

    # Try with more inner iterations and higher K
    for K_val in [15, 20]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(phase3_budget / 2.0, remaining * 0.4)
        a_new = alternating_subgrad(best_seq, time_budget=alloc, inner_iters=5000, K=K_val)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  ExtendedSubgrad K={K_val}: obj={best_score:.8f} **")
        else:
            print(f"  ExtendedSubgrad K={K_val}: obj={obj:.8f}")

    # === Phase 4: Multi-length Exploration (300s) ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    phase4_budget = min(300.0, remaining * 0.2)

    n_options = sorted(set([
        max(100, n_parent - 200),
        max(100, n_parent - 100),
        n_parent,
        n_parent + 100,
        n_parent + 200,
        n_parent + 400,
        n_parent + 800,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 4000]

    # Remove duplicates and sort by priority (closer to parent first)
    n_options = list(dict.fromkeys(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: subgradient + LP
        alloc_sg = min(40.0, remaining * 0.3)
        alloc_lp = min(30.0, remaining * 0.2)

        a_opt = alternating_subgrad(a_try, time_budget=alloc_sg, inner_iters=2000, K=5)
        a_opt = lp_alternating(a_opt, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Random Restarts with Best Structure (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Random Restarts ===")
        n_best = len(best_seq)
        restart_budget = remaining * 0.15
        n_restarts = max(3, int(restart_budget / 20))

        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            scale = 0.02 + 0.01 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

            alloc = min(remaining * 0.25, 20.0)
            a_opt = alternating_subgrad(a_r, time_budget=alloc * 0.6, inner_iters=2000, K=5)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final Polish (remaining) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Final Polish (remaining={remaining:.0f}s) ===")

    if remaining > 30:
        # LP polish
        alloc = min(remaining * 0.4, 60.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    if remaining > 20:
        # Coordinate descent
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 30.0)
        a_cd = coordinate_polish(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    if remaining > 15:
        # One more subgradient round
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.5, 40.0)
        a_sg = alternating_subgrad(best_seq, time_budget=alloc, inner_iters=4000, K=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final Subgrad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
