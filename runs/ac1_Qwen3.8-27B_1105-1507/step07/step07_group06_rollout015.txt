def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))

        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return float(2.0 * n * max_b / (s * s))

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def init_ramp(n, direction=1, alpha=1.0):
    """Create a ramp initialization."""
    x = np.linspace(0, 1, n)
    if direction == 1:
        a = x ** alpha + 0.01
    else:
        a = (1 - x) ** alpha + 0.01
    return a / np.sum(a)


def init_exp(n, rate=1.0, direction=1):

    x = np.linspace(0, 1, n)
    if direction == 1:
        a = np.exp(rate * x)
    else:
        a = np.exp(rate * (1 - x))
    return a / np.sum(a)

"""
LP-Alternating optimization with diverse initializations, adaptive blending,
momentum, and multi-length exploration for minimizing auto-convolution peak.
Core idea: iteratively solve min M s.t. conv(a,g)[k]<=M, sum(g)=1, g>=0,
then blend a toward g with adaptive step size. Use multiple n values and
diverse structured initializations to avoid symmetric local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return float(2.0 * n * max_b / (s * s))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        row_end = min(2 * n - 1, j + n)
        A[row_start:row_end, j] = a[:row_end - row_start]
    return A


def solve_lp(a, use_cvxpy=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    if use_cvxpy:
        solvers_to_try = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
        for solver in solvers_to_try:
            try:
                g = cp.Variable(n, nonneg=True)
                M = cp.Variable()
                constraints = [A @ g <= M, cp.sum(g) == 1]
                prob = cp.Problem(cp.Minimize(M), constraints)
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_val = float(np.sum(g_val))
                    if s_val > 1e-10:
                        return g_val / s_val, float(M.value)
            except Exception:
                continue
        return None, None
    else:
        try:
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(0, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(res.x[n])
        except Exception:
            pass
        return None, None


def lp_alternating(a_init, time_budget=60.0, use_cvxpy=False, verbose=False):
    """LP alternating with momentum and adaptive line search."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy()
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = a.copy()
    momentum_beta = 0.3
    stagnant_count = 0
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp(a, use_cvxpy=use_cvxpy)
        if g_lp is None:
            break
        
        # Prevent over-concentration
        g_max = np.max(g_lp)
        if g_max > 10.0 / n:
            alpha_decon = min(1.0, 5.0 / (n * g_max))
            g_lp = alpha_decon * g_lp + (1.0 - alpha_decon) * np.ones(n) / n
            g_lp = g_lp / np.sum(g_lp)
        
        # Update momentum
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * g_lp
        
        # Line search
        best_t = 0.0
        best_t_obj = compute_obj(a)
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

                best_t_obj = obj_try
                best_t = t
        
        if best_t > 0 and best_t_obj < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * momentum
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            stagnant_count = 0
            if verbose and it % 50 == 0:
                print("      LP it{}: obj={:.8f}".format(it, best_obj))

            stagnant_count += 1

                    a_pert = a_pert / s_p
                    if compute_obj(a_pert) < best_obj + 0.05:

def resample(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return float(2.0 * n * max_b / (s * s))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        row_end = min(2 * n - 1, j + n)
        A[row_start:row_end, j] = a[:row_end - row_start]
    return A


def solve_lp(a, use_cvxpy=False):

    Returns (g_normalized, M_value) or (None, None)."""

                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

def resample(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

            best_score = obj_final
            best_seq = a_final.copy()
            print("  Final LP: obj={:.8f} **".format(best_score))
        else:
            print("  Final LP: obj={:.8f}".format(obj_final))
    
    # Phase 6: Peak targeting (direct mass redistribution)
    print("\n=== Phase 6: Peak Targeting ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        n = len(best_seq)
        a = best_seq.copy()
        
        for it in range(500):
            if time.monotonic() > start_time + budget_s - 25:
                break
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
            
            # Compute contribution of each index to the peak bin
            contrib = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contrib[i] += a[i] * a[j]
                    if i != j:
                        contrib[j] += a[i] * a[j]
            
            # Move mass from high-contrib to low-contrib
            sorted_idx = np.argsort(contrib)
            n_move = max(5, n // 40)
            bottom_idx = sorted_idx[:n_move]
            top_idx = sorted_idx[-n_move:]
            
            total_mass = np.sum(a[top_idx])
            if total_mass < 1e-10:
                break
            
            found = False
            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = frac * total_mass
                a_new = a.copy()
                a_new[top_idx] -= delta / len(top_idx)
                a_new[bottom_idx] += delta / len(bottom_idx)
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new < 1e-10:
                    break
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_score - 1e-12:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a = a_new
                    found = True
                    break
            
            if not found:
                # Try LP step instead
                use_cpx = n <= 600
                g_lp, _ = solve_lp(a, use_cvxpy=use_cpx)
                if g_lp is not None:
                    a_new = 0.5 * a + 0.5 * g_lp
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / np.sum(a_new)
                    obj_new = compute_obj(a_new)
                    if obj_new < best_score:
                        best_score = obj_new
                        best_seq = a_new.copy()
                        a = a_new
                    else:
                        break
        
        print("  Peak targeting: obj={:.8f}".format(best_score))
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
        elif not np.isfinite(val):
            result = [1.0] * 100
            best_score = 2.0
    except Exception:
        pass
    
    elapsed = time.monotonic() - start_time
    print("\n=== FINAL: n={}, score={:.8f}, time={:.1f}s ===".format(
        len(result), best_score, elapsed))
    
    return result
