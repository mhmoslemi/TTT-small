Optimized LP-alternating with simulated annealing, KKT polishing, and multi-n exploration.
Key improvements over previous: (1) adaptive step-size with full-update preference,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix_sparse(n):
    """Return a function that computes A @ x for the convolution matrix A (2n-1 x n)."""
    def matvec(x):
        return compute_conv(x)
    return matvec


def convolve_with_a(a, g):
    """Compute conv(a, g) where both have length n."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]


def solve_lp_minpeak(a, warm_start=None):

                    # In ECOS/SCS, duals are indexed by constraint
                    if prob.value is not None:
                        dual_y = None  # Extracting duals is solver-specific
                except Exception:
                    pass
                return g_val, float(M.value), dual_y
    except Exception:
        pass
    
    # Fallback: linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                return g_val, float(res.x[n]), None
    except Exception:
        pass
    
    return None, None, None


def lp_alternating_anneal(a_init, time_budget, seed=42, n_inner=200):
    """LP alternating with simulated annealing (noise injection)."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Annealing schedule: noise amplitude decays exponentially
    noise_start = 0.15
    noise_end = 0.001
    n_total = n_inner
    
    for it in range(n_total):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(1, n_total - 1)
        noise_amp = noise_start * (noise_end / noise_start) ** progress
        
        # Solve LP
        g_lp, M_lp, _ = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try line search with multiple step sizes (prefer large steps)
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        best_t = 0.0
        
        # Geometric step sizes, biased toward larger
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
                best_t = t
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            # Stall: inject noise to escape
            if noise_amp > 0.005:
                noise = np.exp(rng.normal(0, noise_amp, n))
                a = a * noise
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
            elif it % 20 == 0:
                # Try a larger perturbation
                noise = np.exp(rng.normal(0, 0.1, n))
                a = a * noise
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
    
    return best_a, best_obj


def kkt_polish(a_init, time_budget, n_sweeps=50):
    """Polish using KKT conditions: at optimum, for active lags K and weights w,
    sum_{k in K} w_k * a[k-j] = const for all j with a[j] > 0.
    We iteratively: identify active set, solve linear system, project to simplex.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Identify active set (within tolerance of the max)
        tol = M * 0.01
        active = np.where(conv >= M - tol)[0]
        
        if len(active) < 2:
            active = np.where(conv >= M * 0.95)[0]
        if len(active) < 2:
            break
        
        # Solve: for j with a[j] > 0, sum_{k in active} a[k-j] = const
        # This is a linear system. Use iterative method.
        # For simplicity, do a few Jacobi iterations on the system:
        # sum_{k in active} w_k * a[k-j] = lambda for all j
        # Use equal weights as initialization
        w = np.ones(len(active)) / len(active)
        
        for _ in range(5):
            # Compute current "gradient"
            grad = np.zeros(n)
            for idx, k in enumerate(active):
                for j in range(n):
                    if 0 <= k - j < n:
                        grad[j] += w[idx] * a[k - j]
            
            # Target: grad should be constant (= lambda) where a > 0
            support = a > 1e-10
            if not np.any(support):
                break
            lam = np.mean(grad[support])
            
            # Update a: move toward making grad constant
            # da[j] = eta * (lam - grad[j]) projected to simplex
            eta = 0.1
            a_new = a + eta * (lam - grad)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    eta *= 0.5
            else:
                break
        
        # Also try: directly reduce the peak
        k_star = int(np.argmax(compute_conv(a)))
        # Decrease a at positions that contribute most to conv[k_star]
        contributors = np.zeros(n)
        for j in range(n):
            if 0 <= k_star - j < n:
                contributors[j] = a[k_star - j]
        
        # Move mass from high contributors to low ones
        total_mass = float(np.sum(a))
        top_idx = np.argsort(contributors)[-max(1, n // 10):]
        bot_idx = np.argsort(contributors)[:max(1, n // 10)]
        
        for j in top_idx:
            for k in bot_idx[:3]:
                if j == k:
                    continue
                delta = 0.01 * a[j]
                a[j] -= delta
                a[k] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    obj_new = compute_obj(a)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                    else:
                        a[j] += delta
                        a[k] -= delta
                        a = a / float(np.sum(a))
    
    return best_a, best_obj


def smoothed_obj_and_grad(theta, tau, n):
    """Compute smoothed objective and gradient w.r.t. log-parameters theta."""
    a = np.exp(theta - np.max(theta))
    a = a / np.sum(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    
    # Log-sum-exp smoothing
    exp_arg = (conv - conv_max) * tau
    log_sum = np.log(np.sum(np.exp(exp_arg)))
    smooth_max = conv_max + log_sum / tau
    obj_val = 2.0 * n * smooth_max
    
    # Softmax weights over conv values
    w = tau * np.exp(exp_arg)
    w = w / np.sum(w)
    
    # c[j] = sum_k w[k] * a[k-j] (valid indices)
    c = np.zeros(n)
    for j in range(n):
        ks = np.arange(max(0, j), min(2 * n - 1, j + n))
        valid_a_idx = ks - j
        c[j] = np.dot(w[ks], a[valid_a_idx])
    
    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta


def smooth_lbfgs(a_init, tau_start, tau_end, time_budget, max_iter=100, seed=0):
    """Smoothed L-BFGS-B with tau annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-12)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    n_stages = 4
    taus = np.geomspace(tau_start, tau_end, n_stages)
    
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        def obj_func(th):
            val, _ = smoothed_obj_and_grad(th, tau, n)
            return val
        
        def grad_func(th):
            _, g = smoothed_obj_and_grad(th, tau, n)
            return g
        
        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-5, 'maxfun': 300})
            if result.x is not None:
                a_new = np.exp(result.x - np.max(result.x))
                a_new = a_new / np.sum(a_new)
                a_new = np.maximum(a_new, 0)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    theta = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_obj


def spectral_init(n, seed=0):
    """Initialize with a sequence that has approximately flat power spectrum.
    Use a chirp-like sequence (quadratic phase) which has flat magnitude spectrum,
    then take absolute value and normalize.
    """
    rng = np.random.default_rng(seed)
    x = np.arange(n) / n
    # Chirp: exp(i*pi*x^2) has flat magnitude spectrum
    # For non-negative, use |exp(i*pi*x^2) + c| for some c
    phase = np.pi * x * x
    # Real part gives cos, imaginary gives sin
    # Use a mix: a[x] |~ |cos(phase) + alpha*sin(phase)| or just use a random phase chirp
    # Better: use the magnitude of a random Fourier series
    kmax = n // 2
    coeffs = rng.normal(0, 1, kmax + 1) + 1j * rng.normal(0, 1, kmax + 1)
    coeffs[0] = np.abs(coeffs[0])
    # Inverse DFT
    a = np.zeros(n)
    for k in range(kmax + 1):
        a += coeffs[k] * np.exp(2j * np.pi * k * x)
    a = np.abs(a)
    a = a ** 2  # Square to make it smoother
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_fast(a_init, time_budget, max_rounds=200):
    """Fast coordinate descent: move mass between pairs to reduce peak conv."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]
        
        # Find the pair (i, j) that most reduces conv[k_star]
        # Moving mass from i to j changes conv[k_star] by:
        # delta_conv = -delta*a[k_star-i] + delta*a[k_star-j] (approximately, for small delta)
        # We want a[k_star-i] > a[k_star-j] and i has large a[i]
        
        improved = False
        # Try moving from high a[i] to positions where a[k_star - j] is small
        sorted_i = np.argsort(a)[::-1][:max(5, n // 50)]
        
        for i in sorted_i:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-8:
                continue
            # Best j: minimize a[k_star - j] (subject to validity)
            valid_j = []
            for j in range(n):
                if j == i:
                    continue
                ki = k_star - i
                kj = k_star - j
                if 0 <= ki < n and 0 <= kj < n:
                    # Moving from i to j: conv[k_star] changes by delta*(a[kj] - a[ki])
                    # We want a[kj] < a[ki]
                    valid_j.append((a[kj], j))
            
            if not valid_j:
                continue
            valid_j.sort()
            j_best = valid_j[0][1]
            
            # Try several fractions
            for frac in [0.05, 0.1, 0.02]:
                delta = frac * a[i]
                a[i] -= delta
                a[j_best] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j_best] -= delta
                        a = a / float(np.sum(a))
                        break
        
        if not improved:
            # Random perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-6:
                delta = 0.05 * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    obj_new = compute_obj(a)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                    else:
                        a[i] += delta
                        a[j] -= delta
                        a = a / float(np.sum(a))
            # If still no improvement after several rounds, break
            if rnd > 50 and not improved:
                # Larger random restart
                noise = np.exp(rng.normal(0, 0.05, n))
                a = best_a * noise
                a = np.maximum(a, 0)
                a = a / float(np.sum(a))
    
    return best_a, best_obj

        elif r == 3:
            # Blend with best
            a_r = 0.5 * best_seq + 0.5 * np.linspace(0.5, 1.5, n_restart)
        else:
            # Spectral
            a_r = spectral_init(n_restart, seed=seed + 500 + r)
        
        a_r = np.maximum(a_r, 0)
        a_r = a_r / np.sum(a_r)
        
        alloc = min(remaining * 0.1, 40.0)
        a_opt, obj_opt = lp_alternating_anneal(a_r, time_budget=alloc, seed=seed + 300 + r,
                                                n_inner=50)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_opt:.8f}")
    
    # Final: one more coordinate descent + LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 30.0)
        a_cd, obj_cd = coordinate_descent_fast(best_seq, time_budget=alloc)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        a_lp, obj_lp = lp_alternating_anneal(best_seq, time_budget=alloc,
                                              seed=seed + 5000, n_inner=50)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    # Ensure non-negative and reasonable range
    best_seq = np.maximum(best_seq, 0)
    s_final = float(np.sum(best_seq))
    if s_final < 0.01:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)
    
    result = [float(min(1000.0, x)) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
