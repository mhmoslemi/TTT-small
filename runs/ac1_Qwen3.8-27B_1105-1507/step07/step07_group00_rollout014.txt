def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    size_needed = 2 * n - 1
    while size < size_needed:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(n, a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M, dual_weights)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(n, a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get dual values for the inequality constraints

                    try:

                    except:
                        dual_vals = np.zeros(2 * n - 1)

"""
Optimizes non-negative sequences to minimize 2*n*max(self_conv)/(sum)^2.
Uses hybrid soft-subgradient descent with LP-guided peak weighting,
LP alternating with golden-section line search, spectral smoothing,
and multi-scale length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def fast_conv(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, w):
    """Cross-correlation: result[i] = sum_j a[j]*w[i+j], length n."""
    n = len(a)
    size = 1
    while size < n + len(w) - 1:
        size <<= 1
    F_a = fft(a, size)
    F_w = fft(w, size)
    cc = ifft(F_a * np.conj(F_w))
    return np.real(cc[:n])


def fast_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = fast_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_mat(n, a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, active_weights) or (None, None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_mat(n, a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    con1 = A @ g <= M
    con2 = cp.sum(g) == 1
    prob = cp.Problem(cp.Minimize(M), [con1, con2])

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=verbose)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    # Compute active set from the residual
                    conv_ag = A @ g_val
                    active = np.abs(conv_ag - M_val) < 1e-4 * max(M_val, 1e-10)
                    # Get dual values if available
                    try:
                        dual = np.abs(con1.dual_value)
                        if dual is not None and len(dual) == 2*n-1:
                            dsum = dual.sum()
                            if dsum > 1e-10:
                                active_weights = dual / dsum
                            else:
                                active_weights = active.astype(float)
                                active_weights /= active_weights.sum()
                        else:
                            active_weights = active.astype(float)
                            if active_weights.sum() > 0:
                                active_weights /= active_weights.sum()
                            else:
                                active_weights = np.ones(2*n-1) / (2*n-1)
                    except Exception:
                        active_weights = active.astype(float)
                        if active_weights.sum() > 0:
                            active_weights /= active_weights.sum()
                        else:
                            active_weights = np.ones(2*n-1) / (2*n-1)
                    return g_val, M_val, active_weights
        except Exception:
            continue

    # Fallback: scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                M_val = float(res.x[n])
                active_weights = np.ones(2*n-1) / (2*n-1)
                return g_val, M_val, active_weights
    except Exception:
        pass

    return None, None, None


def soft_sg_step(a, T, lr):
    """
    One soft-subgradient step with multiplicative updates.
    Returns (a_new, obj, max_conv).
    """
    n = len(a)
    conv = fast_conv(a)
    max_conv = float(np.max(conv))

    # Softmax weights over all peaks
    shifted = conv - max_conv
    w = np.exp(shifted / T)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        return a, max_conv, max_conv
    w /= w_sum

    # Compute subgradient via cross-correlation
    s = fast_cross_conv(a, w)

    # Multiplicative update
    a_new = a * np.exp(-lr * s)
    a_new = np.maximum(a_new, 1e-30)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new /= s_new
    else:
        a_new = a.copy()

    obj_new = fast_obj(a_new)
    return a_new, obj_new, max_conv


def golden_line_search(a, g, n_evals=10):
    """
    Golden section line search for t in [0,1] minimizing obj((1-t)*a + t*g).
    """
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    lo = 0.0
    hi = 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)

    a_c = (1 - c) * a + c * g
    a_c = np.maximum(a_c, 0)
    sc = float(np.sum(a_c))
    if sc < 1e-10:
        return a.copy(), np.inf
    a_c /= sc
    f_c = fast_obj(a_c)

    a_d = (1 - d) * a + d * g
    a_d = np.maximum(a_d, 0)
    sd = float(np.sum(a_d))
    if sd < 1e-10:
        return a.copy(), np.inf
    a_d /= sd
    f_d = fast_obj(a_d)

    for _ in range(n_evals - 2):
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - phi * (hi - lo)
            a_c = (1 - c) * a + c * g
            a_c = np.maximum(a_c, 0)
            sc = float(np.sum(a_c))
            if sc < 1e-10:
                break
            a_c /= sc
            f_c = fast_obj(a_c)
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + phi * (hi - lo)
            a_d = (1 - d) * a + d * g
            a_d = np.maximum(a_d, 0)
            sd = float(np.sum(a_d))
            if sd < 1e-10:
                break
            a_d /= sd
            f_d = fast_obj(a_d)

    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    sb = float(np.sum(a_best))
    if sb < 1e-10:
        return a.copy(), np.inf
    a_best /= sb
    f_best = fast_obj(a_best)
    return a_best, f_best


def hybrid_optimize(a, time_budget, T_init=0.5, T_max=5.0, lr_init=0.1,
                    lp_every=10, verbose=True):
    """
    Hybrid optimizer: soft subgradient steps + periodic LP steps.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a /= s

    best_a = a.copy()
    best_obj = fast_obj(a)
    start = time.monotonic()
    deadline = start + time_budget

    T = T_init
    lr = lr_init
    stall = 0
    it = 0
    n_evals_ls = 8

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 5:
            break

        if it % lp_every == lp_every - 1:
            # LP step
            g_lp, M_lp, aw = solve_lp_minpeak(a)
            if g_lp is not None:
                a_try, obj_try = golden_line_search(a, g_lp, n_evals_ls)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    stall = 0
                    lr = max(0.01, lr * 0.98)
                    T = min(T_max, T * 1.05)
                else:
                    stall += 1
                    if stall > lp_every * 3:
                        # Perturb
                        rng = np.random.default_rng(it * 7919 + n)
                        a = best_a * np.exp(rng.normal(0, 0.05, n))
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s > 1e-10:
                            a /= s
                        best_a = a.copy()
                        best_obj = fast_obj(a)
                        stall = 0
                        lr = max(0.01, lr * 0.7)
                        T = T_init
                    else:
                        lr = max(0.01, lr * 0.95)
            else:
                stall += 1
        else:
            # Soft subgradient step
            a_new, obj_new, _ = soft_sg_step(a, T, lr)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
                lr = max(0.005, lr * 0.999)
                T = min(T_max, T * 1.01)
            else:
                # Backtracking
                for trial in range(5):
                    lr_sm = lr * (0.5 ** trial)
                    a_new, obj_new, _ = soft_sg_step(a, T, lr_sm)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
                        lr = lr_sm
                        stall = 0
                        T = min(T_max, T * 1.01)
                        break
                else:
                    stall += 1
                    if stall > 50:
                        rng = np.random.default_rng(it * 31 + n)
                        a = best_a * np.exp(rng.normal(0, 0.03, n))
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s > 1e-fast_obj:
                            a /= s
                        best_a = a.copy()
                        best_obj = fast_obj(a)
