"""
Optimization via smoothed log-sum-exp objective with L-BFGS in softmax parameterization,
combined with LP alternation and coordinate refinement.
Key improvement: uses second-order (L-BFGS) information on a smoothed objective
in the unbounded simplex parameterization, which should converge faster than
subgradient methods used previously.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def softmax(x):
    """Numerically stable softmax."""
    x_max = np.max(x)
    e = np.exp(x - x_max)
    return e / np.sum(e)

def smooth_obj_and_grad(x, t):
    """
    Compute F_t(softmax(x)) and gradient w.r.t. x.
    F_t(a) = (1/t) * log(sum_k exp(t * conv(a,a)[k]))
    """
    a = softmax(x)
    n = len(a)
    conv = compute_conv(a)
    # Log-sum-exp
    max_c = np.max(conv)
    log_sum = max_c + np.log(np.sum(np.exp(t * (conv - max_c))))
    F = log_sum / t
    # p_k = exp(t*(conv[k]-max_c)) / sum(exp(t*(conv-max_c)))
    p = np.exp(t * (conv - max_c))
    p /= np.sum(p)
    # dF/da[i] = sum_j p[i+j] * a[j] (cross-correlation)
    # np.correlate(p, a, 'valid') gives c[k] = sum_j p[j+k]*a[j] for k=0..len(p)-len(a)
    dF_da = np.correlate(p, a, mode='valid')
    # Mirror gradient: dF/dx[j] = a[j] * (dF_da[j] - sum_i dF_da[i]*a[i])
    mean_val = np.dot(dF_da, a)
    grad = a * (dF_da - mean_val)
    return F, grad

def l_bfgs_smoothed(a_init, time_budget=100.0, t_start=20.0, t_end=500.0,

    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    x = np.log(a)  # Initial parameterization
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    n = len(a)
    # Divide time into phases with increasing t
    n_phases = 8
    t_values = np.linspace(np.log(t_start), np.log(t_end), n_phases)
    time_per_phase = time_budget / n_phases
    
    for phase in range(n_phases):
        if time.monotonic() - start >= time_budget:
            break
        t = np.exp(t_values[phase])
        phase_start = time.monotonic()
        phase_budget = min(time_per_phase, time_budget - (phase_start - start))
        
        def fun(x_vec):
            val, _ = smooth_obj_and_grad(x_vec, t)
            return val
        def grad(x_vec):
            _, g = smooth_obj_and_grad(x_vec, t)
            return g
        
        # Run L-BFGS with time limit
        x_best = x.copy()
        it_count = [0]

            it_count[0] += 1
            if time.monotonic() - phase_start > phase_budget:
                return True  # Stop
            return False
        
        try:

                x = res.x

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_for_g(a, M_target):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target, g >= 0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_target]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solvers = [cp.HIGHS, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    return None

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.HIGHS, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def lp_alternating_improved(a, time_budget=100.0, verbose=False):
    """
    Improved LP alternating with golden section line search and 
    adaptive perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iter_count = 0
    last_best_t = 0.5
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:

            M_cur = float(np.max(conv_a))

            if g_lp is None:
                break

        # Evaluate objective at several t values
        best_t_obj = np.inf
        best_t = 0.5

        t_candidates = np.unique(np.concatenate([
            [last_best_t + dt for dt in [-0.3, -0.2, -0.1, 0.0, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]],
            [0.1, 0.3, 0.5, 0.7, 1.0]
        ]))
        t_candidates = t_candidates[(t_candidates > 0.01) & (t_candidates <= 1.0)]
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()
        
        iter_count += 1
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_best_t = best_t
            stall = 0
            if verbose and iter_count % 5 == 0:

                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={elapsed:.1f}s")

def coordinate_refine(a, time_budget=30.0):
    """Paired coordinate descent: move mass to reduce peak convolution."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False
        
        # Find contributing positions
        contrib_list = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib_list.append((a[i] * a[j], i, j))
        contrib_list.sort(reverse=True)
        
        # Non-contributing positions
        contrib_positions = set()
        for _, i, j in contrib_list:
            contrib_positions.add(i)
            contrib_positions.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_positions]
        
        if not non_contrib:
            break
        
        for contrib, i, j in contrib_list[:30]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break
            for target in non_contrib[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            # Try moving from j instead
            for contrib, i, j in contrib_list[:10]:
                if time.monotonic() - start > time_budget:
                    break
                if i == j:
                    continue
                for target in non_contrib[:5]:
                    for frac in [0.01, 0.03, 0.05, 0.1]:
                        delta = frac * a[j]
                        a[j] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                            improved = True
                            break
                        else:
                            a[j] += delta
                            a[target] -= delta
                            break
                if improved:
                    break
            if not improved:
                break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Inverted triangle
        a = 1.0 - 2.0 * np.abs(x / (n - 1) - 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # U-shaped
        t = x / (n - 1)
        a = (2 * t - 1) ** 2 + 0.1
        a = a / a.sum()
    elif mode == 5:
        # Chirp-based: 1 + cos(f * x^2)
        f = rng.uniform(1, 5)
        t = x / (n - 1)
        a = 1.0 + np.cos(f * np.pi * t ** 2)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Multi-frequency
        freqs = rng.uniform(1, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        t = x / (n - 1)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * t + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Tail-heavy (edges large)
        t = x / (n - 1)
        a = (t - 0.5) ** 4 + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective in softmax parameterization (primary)

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2 * n - 1]
    np.maximum(result, 0, out=result)  # Clip tiny negatives from FFT
    return result


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_arr(x):
    """Numerically stable softmax."""
    x_max = np.max(x)
    e = np.exp(x - x_max)
    return e / np.sum(e)


def smooth_obj_and_grad(x, t):
    """
    Compute F_t(softmax(x)) and its gradient w.r.t. x.
    F_t(a) = (1/t) * log(sum_k exp(t * conv(a,a)[k]))
    approximates max_k conv(a,a)[k] for large t.
    """
    a = softmax_arr(x)
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    scaled = t * (conv - max_c)
    # scaled <= 0 always, so exp is safe
    p = np.exp(scaled)
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        # Degenerate: all mass on max
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    F = (max_c + np.log(p_sum)) / t
    
    # dF/da[i] = sum_j p[i+j] * a[j]
    # np.correlate(p, a, 'valid'): c[k] = sum_n p[n+k]*a[n], k=0..len(p)-len(a)
    dF_da = np.correlate(p, a, mode='valid')
    
    # Mirror gradient: dF/dx[j] = a[j] * (dF_da[j] - dot(dF_da, a))
    mean_val = float(np.dot(dF_da, a))
    grad = a * (dF_da - mean_val)
    return F, grad


def l_bfgs_smoothed(a_init, time_budget=100.0, t_start=20.0, t_end=500.0,
                     n_phases=6, verbose=False):
    """
    L-BFGS on smoothed objective with temperature annealing.
    x parameterizes a = softmax(x), eliminating simplex constraint.
    """
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    x = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    n = len(a)
    
    t_values = np.exp(np.linspace(np.log(t_start), np.log(t_end), n_phases))
    time_per_phase = time_budget / n_phases
    
    for phase in range(n_phases):
        phase_start = time.monotonic()
        remaining_phase = min(time_per_phase, time_budget - (phase_start - start))
        if remaining_phase < 2:
            break
        t = t_values[phase]
        
        def fun_vec(x_vec):
            val, _ = smooth_obj_and_grad(x_vec, t)
            return val
        
        def grad_vec(x_vec):
            _, g = smooth_obj_and_grad(x_vec, t)
            return g
        
        x_local = x.copy()
        maxfun = int(max(20, 500 * remaining_phase / time_budget))
        
        try:
            res = minimize(fun_vec, x_local, jac=grad_vec, method='L-BFGS-B',
                          options={'maxfun': maxfun, 'maxiter': 200,
                                   'gtol': 1e-10, 'ftol': 1e-12})
            if res.x is not None:
                a_try = softmax_arr(res.x)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    x = res.x.copy()
        except Exception:
            pass
        
        # Also try current x (in case L-BFGS didn't improve)
        a_cur = softmax_arr(x)
        obj_cur = compute_obj(a_cur)
        if obj_cur < best_obj:
            best_obj = obj_cur
            best_a = a_cur.copy()
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    phase {phase}: t={t:.1f}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    from scipy.sparse import csr_matrix
    n = len(a)
    rows = []
    cols = []
    data = []
    for j in range(n):
        for k in range(2 * n - 1):
            val = k - j
            if 0 <= val < n:
                rows.append(k)
                cols.append(j)
                data.append(a[val])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build dense matrix (for n <= 1500, this is manageable)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        A[j:j + end, j] = a[:end]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.HIGHS, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def coordinate_refine(a, time_budget=30.0):
    """Paired coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False
        
        # Contributors to peak at k_star
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)
        
        # Non-contributors
        cpos = set()
        for _, i, j in contribs:
            cpos.add(i)
            cpos.add(j)
        npos = [i for i in range(n) if i not in cpos]
        
        if not npos:
            break
        
        for contrib, i, j in contribs[:20]:
            if time.monotonic() - start > time_budget:
                break
            for target in npos[:5]:
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
