"""
Optimized step function search: LP alternating with golden-section line search,
softmax gradient, peak redistribution, and multi-scale exploration.
Targets obj < 1.5027 by improving upon parent (1.506336).
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        j_min = 0
        j_max = n - 1
        k_min = i + j_min
        k_max = i + j_max
        for j in range(j_min, j_max + 1):
            A[i + j, i] = a[j]
    return A

def solve_lp(a, prev_g=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt)."""
    import cvxpy as cp
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            if solver == cp.MOSEK:
                prob.solve(solver=solver, msdk_api=cp.MOSEK_API_THREADING,
                           msdk_params={'MSK_IPAR_INTPUMP': 0})
            elif solver == cp.CLARABEL:
                prob.solve(solver=solver, tol_gap_abs=1e-10, tol_gap_rel=1e-10)
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def softmax_subgradient(a, T=50.0):
    """Compute softmax-weighted subgradient of max_k (a*a)[k]."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n)
    weights = np.exp((conv - max_conv) * T)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-12:
            continue
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                g[i] += weights[k] * a[j]
    return g

def softmax_gradient_descent(a, time_budget=200.0, verbose=False):
    """Softmax gradient descent with adaptive temperature and Nesterov momentum."""
    n = len(a)

def golden_section_line_search(a, g, best_obj):
    """Golden section search for optimal t in a + t*(g - a) direction."""
    # We want to minimize compute_obj((1-t)*a + t*g) for t in [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    
    def f(t):
        x = (1 - t) * a + t * g
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            return np.inf
        x = x / s
        return compute_obj(x)
    
    fc = f(c)
    fd = f(d)
    
    for _ in range(15):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    
    t_best = (lo + hi) / 2
    x = (1 - t_best) * a + t_best * g
    x = np.maximum(x, 0)
    s = float(np.sum(x))
    if s < 1e-10:
        return a.copy(), np.inf
    x = x / s
    return x, compute_obj(x)

def peak_redistribute(a, time_budget=60.0):
    """Directly try to reduce the peak by redistributing mass."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    conv = compute_conv(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False
        
        # Find top contributors to peak
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)
        
        # Find non-contributing positions
        contrib_set = set()
        for _, i, j in contribs:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_set]
        
        if not non_contrib:
            break
        
        for c_val, i, j in contribs[:10]:
            if c_val < 1e-12:
                break
            for target in non_contrib[:5]:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    if time.monotonic() - start > time_budget:
                        break
                    delta = frac * a[i]
                    if a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:

        x = np.linspace(-1, 1, n)

    elif mode == 3:

        a = np.zeros(n)

    elif mode == 4:
