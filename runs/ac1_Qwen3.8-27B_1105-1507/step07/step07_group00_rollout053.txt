def compute_conv(a, b=None):
    """Full linear convolution a*b (or a*a if b is None)."""
    if b is None:

    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def multiplicative_weights_game(a, n_iters=500, eta=None):
    """
    Solve the game: min_g max_k conv(a,g)[k] s.t. sum(g)=1, g>=0
    Equivalently: find g that minimizes the peak of conv(a,g).
    
    Uses multiplicative weights (Hedge) algorithm.
    Returns (g_opt, V_estimate) where g_opt minimizes max_k conv(a,g)[k]
    and V_estimate is the game value.
    """
    n = len(a)
    a = a / np.sum(a)  # normalize
    
    if eta is None:
        eta = 1.0 / np.sqrt(np.log(2*n-1))  # standard choice
    
    # Initialize
    g = np.ones(n) / n
    lam = np.ones(2*n-1) / (2*n-1)
    
    V_est = 0.0
    best_g = g.copy()
    best_max = np.inf
    
    for t in range(n_iters):
        # Compute A @ g = conv(a, g)
        conv_ag = compute_conv(a, g)  # length 2n-1
        
        # Update lambda (row player, maximizing): lam[k] proportional to exp(eta * (A g)[k])
        max_val = np.max(conv_ag)
        lam_new = lam * np.exp(eta * (conv_ag - max_val))
        lam_new = lam_new / lam_new.sum()
        
        # Compute A^T @ lam = cross-correlation
        lam_total = lam_new + lam  # simple average? No, let's use the latest
        # Actually, let's compute A^T @ lam_new
        n_a = len(a)
        result = np.zeros(n_a)
        for j in range(n_a):
            result[j] = np.dot(a, lam_new[j:j+n_a])
        
        # Update g (column player, minimizing): g[j] proportional to exp(-eta * (A^T lambda)[j])
        min_val = np.min(result)
        g_new = g * np.exp(-eta * (result - min_val))
        g_sums = g_new.sum()
        if g_sums > 0:
            g_new = g_new / g_sums
        else:
            g_new = g.copy()
        
        # Track game value
        V_est = float(np.dot(lam_new, conv_ag))
        
        # Track best
        max_conv = float(np.max(conv_ag))
        if max_conv < best_max:
            best_max = max_conv
            best_g = g_new.copy()
        
        g = g_new
        lam = lam_new
    
    return best_g, V_est

"""
Hybrid optimization using exponentiated gradient, multiplicative weights game
for inner LP, and cvxpy refinement. Targets objective < 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

# --- Core computation functions ---

def get_fft_size(n):
    size = 1
    while size < n:
        size <<= 1
    return size

def compute_conv(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = get_fft_size(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_conv_normalized(a):
    """conv(a,a) with a already normalized (sum=1)."""
    return compute_conv(a)

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def gradient_conv_at_peak(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a.
    Result[m] = 2*a[k-m] if 0<=k-m<n, else 0."""
    n = len(a)
    g = np.zeros(n)
    m_start = max(0, k - (n - 1))
    m_end = min(n - 1, k)
    for m in range(m_start, m_end + 1):
        g[m] = 2.0 * a[k - m]
    return g

def gradient_conv_topK(a, K=3):
    """Compute weighted subgradient using top-K peaks."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n), int(top_idx[0])
    # Softmax weights
    scale = 5.0
    weights = np.exp((peak_vals - max_val) * scale / max_val)
    weights /= weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        g += w * gradient_conv_at_peak(a, int(k))
    return g, int(top_idx[0])

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# --- Multiplicative Weights Game ---

def solve_inner_game(a, n_iters=5000, eta=None):
    """
    Solve: min_g max_k conv(a,g)[k] s.t. sum(g)=1, g>=0
    via multiplicative weights. Returns (g_opt, game_value).
    """
    n = len(a)
    a = a / np.sum(a)
    a_rev = a[::-1].copy()
    
    if eta is None:
        eta = 1.0 / np.sqrt(np.log(max(2*n-1, 2)))
    
    g = np.ones(n) / n
    lam = np.ones(2*n-1) / (2*n-1)
    
    best_g = g.copy()
    best_max = np.inf
    
    for t in range(n_iters):
        # A @ g = conv(a, g)
        conv_ag = compute_conv(a, g)
        
        # Update lambda (maximizer): proportional to exp(eta * conv_ag)
        max_cv = np.max(conv_ag)
        lam = lam * np.exp(eta * (conv_ag - max_cv))
        lam_sum = lam.sum()
        if lam_sum > 0:
            lam = lam / lam_sum
        else:
            lam = np.ones(2*n-1) / (2*n-1)
        
        # A^T @ lam = cross-correlation = conv(a_rev, lam)[n-1:]
        correl = compute_conv(a_rev, lam)[:2*n-1]
        # Actually, conv(a_rev, lam) has length n + (2n-1) - 1 = 3n-2
        # We want correl[j] = sum_i a[i]*lam[i+j] = conv(a_rev, lam)[n-1+j]
        conv_full = compute_conv(a_rev, lam)
        correl = conv_full[n-1:n-1+n]
        
        # Update g (minimizer): proportional to exp(-eta * correl)
        min_c = np.min(correl)
        g = g * np.exp(-eta * (correl - min_c))
        g_sum = g.sum()
        if g_sum > 0:
            g = g / g_sum
        else:
            g = np.ones(n) / n
        
        # Track best
        max_conv = float(np.max(conv_ag))
        if max_conv < best_max:
            best_max = max_conv
            best_g = g.copy()
    
    return best_g, best_max

def solve_lp_minpeak_cvxpy(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0 using cvxpy."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build convolution matrix A (2n-1 x n)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        start = j
        length = min(n, 2*n-1-j)
        A[start:start+length, j] = a[:length]
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass
    
    return None, None
