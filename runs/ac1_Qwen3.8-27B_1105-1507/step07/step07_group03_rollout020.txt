4. Take a step in the negative subgradient direction
5. Project onto `{a >= 0, sum(a) = 1}` (water-filling, O(n log n))
6. Repeat

```python
def projected_subgradient_fista(a, n_iters=10000, time_budget=600):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    y = a.copy()
    v = np.zeros(n)
    beta = 0.8
    k_peaks = 10
    
    t_k = 1.0
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution at y
        conv_y = compute_conv(y)
        
        # Find top K peaks
        top_idx = np.argsort(conv_y)[-k_peaks:]
        top_val = conv_y[top_idx]
        max_val = top_val[-1]
        
        # Softmax weights for subgradient
        temp = max(1e-4, (max_val - top_val[0]) * 0.3)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            # d/da_j conv(a,a)[k] = a_{k-j} + a_{j+k} for valid indices
            for j in range(n):
                g1 = k - j
                g2 = j + k
                val = 0.0
                if 0 <= g1 < n:
                    val += y[g1]
                if 0 <= g2 < n:
                    val += y[g2]
                grad[j] += w * val
        
        # Nesterov step
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        beta_new = (t_k - 1) / t_new
        beta_new = np.clip(beta_new, 0, 1)
        
        v = beta_new * v + (1 - beta_new) * grad
        alpha = 2.0 / (1 + t_new)  # step size
        
        x_new = y - alpha * v
        x_new = project_onto_simplex(x_new)  # project onto {a>=0, sum(a)=1}
        
        # Evaluate
        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
        
        y = x_new
        t_k = t_new
    
    return best_a
```

The Euclidean projection onto this set can be computed in O(n log n) using the water-filling algorithm:

```python
def project_onto_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n+1)) > (cssv - 1))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

This gives us a fast projection step.

Let me also think about the "peaks" structure. The convolution `conv(a,a)` has 2n-1 entries. The max is typically at k=0 (the "zero lag" or "peak" of the autocorrelation). If the max is at k=0, then we want to minimize `sum_i a_i^2` subject to `sum(a)=1, a>=0`, which is achieved by the uniform distribution `a_i = 1/n`, giving objective 2.

But if we can make the max NOT at k=0, we can do better! The trick is to make the autocorrelation "side lobes" as small as possible while also keeping the peak at k=0 small.

For a "flat spectrum" sequence, `|FFT(a)|^2` is constant, which means the circular autocorrelation is a delta function (only the zero-lag is non-zero). For the linear autocorrelation, this translates to a roughly flat profile.

Actually, I think the key insight is:
- If the max of `conv(a,a)` is at the center (k ≈ n-1), then we're in a good regime
- If the max is at the edges (k=0 or k=2n-2), that's bad because those are "forced" to be large

For k=0: `conv(a,a)[0] = a[0]^2`. This is always small if a[0] is small.
For k=n-1: `conv(a,a)[n-1] = sum_i a[i]*a[i+n-1]`... wait, that's not right.

`conv(a,a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]`

For k=n-1: `conv(a,a)[n-1] = sum_{i=max(0,n-1-(n-1))}^{min(n-1,n-1)} a[i]*a[n-1-i] = sum_{i=0}^{n-1} a[i]*a[n-1-i]`

This is the "anti-diagonal" sum, which for a symmetric sequence equals `sum a[i]^2` (if a is symmetric about (n-1)/2).

For k=0: `conv(a,a)[0] = a[0]^2` (just the first element squared).

So the structure of the convolution is:
- k=0: a[0]^2 (small if a[0] small)
- k=1: a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]
- ...
- k=n-1: the "full overlap" sum
- ...
- k=2n-2: a[n-1]^2 (small if a[n-1] small)

The maximum is typically somewhere in the middle, unless the sequence has special structure.

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combined approach: FFT initialization, FISTA subgradient, LP alternating,
multi-n exploration, and coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

# Try to import cvxpy for faster LP
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1)
    rho = np.where(cond > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak_cvxpy(a):
    """Solve LP using cvxpy."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def solve_lp_minpeak_scipy(a):
    """Solve LP using scipy linprog."""
    n = len(a)
    A = build_conv_matrix(a)
    c = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    if HAS_CVXPY:
        result = solve_lp_minpeak_cvxpy(a)
        if result is not None:
            return result
    return solve_lp_minpeak_scipy(a)

def lp_alternating(a, n_iters=2000, time_budget=300.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it {it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
        
        g = solve_lp_minpeak(a)
        if g is None:
            break
        
        # Adaptive step size
        best_t = 0.5
        best_t_obj = np.inf
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * best_a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-10:
            a = (1.0 - best_t) * best_a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng((it * 13 + n) % (2**32))
                scale = 0.02 + 0.005 * min(stall // 30, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
    
    return best_a

def fista_subgradient(a, n_iters=100000, time_budget=120.0, k_peaks=10, verbose=False):
    """FISTA with multi-peak subgradient."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = a.copy()
    v = np.zeros(n)
    t_k = 1.0
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        if verbose and it % 10000 == 0 and it > 0:
            print(f"    FISTA it {it}: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")
        
        # Compute conv at y
        conv_y = compute_conv(y)
        
        # Top K peaks
        k_p = min(k_peaks, len(conv_y))
        top_idx = np.argsort(conv_y)[-k_p:]
        top_val = conv_y[top_idx]
        max_val = top_val[-1]
        
        # Softmax weights
        spread = max_val - top_val[0]
        if spread < 1e-12:
            weights = np.ones(k_p) / k_p
        else:
            temp = spread * 0.5
            weights = np.exp((top_val - max_val) / temp)
            weights /= weights.sum()
        
        # Vectorized gradient computation
        grad = np.zeros(n)
        for w, ki in zip(weights, top_idx):
            ki = int(ki)
            # Part 1: y[j+ki] for j = 0, ..., min(n-1-ki, n-1)
            if ki < n:
                end = min(n, n - ki)
                grad[:end] += w * y[ki:ki + end]
            # Part 2: y[ki-j] for j = max(0, ki-n+1), ..., min(n-1, ki)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            idx_start = max(0, ki - n + 1)
            idx_end = min(n - 1, ki)
            grad[j_start:j_end] += w * y[idx_start:idx_end + 1][::-1]
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = np.clip((t_k - 1.0) / t_new, 0.0, 1.0)
        v = beta * v + (1.0 - beta) * grad
        
        # Step size
        alpha = 2.0 / (1.0 + t_new)
        # Normalize: we want the step to be of order 1/n
        alpha = alpha * n / (1.0 + np.linalg.norm(grad) * np.linalg.norm(y))
        alpha = min(alpha, 1.0)
        
        x_new = y - alpha * v
        x_new = project_simplex(x_new)
        
        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                rng = np.random.default_rng(it + n)
                x_new = best_a * np.exp(rng.normal(0, 0.01, n))
                x_new = project_simplex(x_new)
                best_a = x_new.copy()
                best_obj = compute_obj(x_new)
                stall = 0
                t_k = 1.0
                v = np.zeros(n)
        
        y = x_new
        t_k = t_new
    
    return best_a

def fft_flat_spectrum(a, n_iters=200, time_budget=30.0, relaxation=0.5):
    """FFT-based flat spectrum iteration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    from scipy.fft import fft, ifft
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-15)
        phase = A / mag_safe
        
        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat
        
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    rng = np.random.default_rng(it + n)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def coordinate_descent(a, time_budget=30.0):
    """Coordinate-wise refinement."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        
        step = max(1, n // 100)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 10 * a[j] + 1e-6:
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    s_new = float(np.sum(a))
                    if s_new < 1e-10:
                        a[i] += delta
                        a[j] -= delta
                        break
                    a = a / s_new
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
        
        # Also try: reduce the peak element and distribute to random positions
        peak_el = int(np.argmax(a))
        if a[peak_el] > 2.0 / n:
            rng = np.random.default_rng(it)
            for _ in range(5):
                j = int(rng.integers(0, n))
                if j == peak_el:
                    continue
                delta = 0.05 * a[peak_el]
                a[peak_el] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
        
        if not improved and it > 20:
            break
    
    return best_a

def random_init(n, seed):
    """Generate random initialization."""
    rng = np.random.default_rng(seed)
    mode = int(rng.integers(0, 5))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        freqs = rng.uniform(0, 4, 8)
        phases = rng.uniform(0, 2 * np.pi, 8)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # Exponential decay from center
        x = np.arange(n)
        c = n / 2
        a = np.exp(-np.abs(x - c) / (n / 4))
        a = a / np.sum(a)
    else:
        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((np.arange(n) - c1)**2) / (2*w**2)) + \
            np.exp(-((np.arange(n) - c2)**2) / (2*w**2))
        a = a / np.sum(a)
    return a

def propose_candidate(seed=42, budget_s=1500):
    """Main search function."""
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent_arr = parent_arr / s
                obj = compute_obj(parent_arr)
                best_score = obj
                best_seq = parent_arr.copy()
                print(f"Parent (n={parent_arr.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default (n=800): obj={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    # Phase 1: Quick FFT + FISTA on parent
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: FFT + FISTA (n={n_parent}) ===")
        alloc = min(remaining * 0.08, 30.0)
        a = fft_flat_spectrum(best_seq, n_iters=300, time_budget=alloc, relaxation=0.5)
        obj = compute_obj(a)
        print(f"  FFT: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
        
        remaining = deadline - time.monotonic()
        if remaining > 60:
            alloc = min(remaining * 0.15, 120.0)
            a = fista_subgradient(best_seq, n_iters=200000, time_budget=alloc, k_peaks=10, verbose=True)
            obj = compute_obj(a)
            print(f"  FISTA: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
