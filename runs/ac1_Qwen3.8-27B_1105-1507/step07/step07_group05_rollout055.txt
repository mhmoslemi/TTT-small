"""
Log-sum-exp smoothed FISTA with LP alternation and adaptive temperature scheduling
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: proper smooth max approximation, Nesterov acceleration,
adaptive temperature, direct LP polishing, and coordinate descent final polish.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def proj_simplex(v):
    """Project v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    positive = u - cssv / np.arange(1, n + 1) > 0
    rho = int(np.sum(positive))
    if rho == 0:
        return np.ones(n) / n
    theta = cssv[rho - 1] / rho
    proj = np.maximum(v - theta, 0)
    return proj


def logsumexp_conv_gradient(a, T, N_fft):

        # Compute objective
        fa_new = rfft(x_new, N_fft)
        b_new = irfft(fa_new * fa_new, N_fft)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            x_prev = x
            x = x_new
            stall = 0
            eta = min(eta * 1.003, 3.0)
        else:
            x_prev = x
            x = x_new
            stall += 1
            eta *= 0.95

            if stall > 100:
                # Reset momentum and step
                x = best_a * (1 + 0.02 * np.random.default_rng(it).normal(0, 1, n))
                x = proj_simplex(np.maximum(x, 0))
                x_prev = x.copy()
                t_mom = 1.0
                eta = 0.5
                stall = 0

        # Update Nesterov parameter
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0

        if eta < 1e-6:
            break

        if verbose and it % 500 == 0:
            print(f"    LSE-FISTA it{it}: obj={best_obj:.8f}, T={T:.1f}, eta={eta:.4f}")

    return best_a


def direct_lp_polish(a, time_budget=60.0, verbose=False):
    """
    LP-based polishing: alternately solve min M s.t. conv(a,g)<=M, sum(g)=1, g>=0
    and project back.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Build convolution matrix A (2n-1 x n) where A @ g = conv(a, g)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:n]
        # Actually conv(a,g)[k] = sum_i a[i]*g[k-i], so for fixed g variable index j:
        # g[j] appears in conv[k] for k where 0 <= k-j < n, i.e., j <= k <= j+n-1
        # and the coefficient is a[k-j]
        # So A[k, j] = a[k-j] for j <= k <= j+n-1

    # Let me rebuild properly
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes a[i]*g[j] to conv[i+j] for i=0..n-1
        for i in range(n):
            A[i + j, j] = a[i]

    stall = 0
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        from scipy.optimize import linprog
        # LP: min M s.t. A@x <= M, sum(x)=1, x>=0, M>=0
        # Variables: [x_0, ..., x_{n-1}, M]
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize M

        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0  # A@x - M <= 0
        b_ub = np.zeros(2 * n - 1)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if res.status != 0 or res.x is None:
            break

        g = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g))
        if s_g < 1e-10:
            break
        g = g / s_g

        # Try different blend factors
        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 15:
                break

    return best_a


def coord_descent_polish(a, time_budget=30.0, verbose=False):
    """
    Coordinate descent: for each coordinate, find the best value that minimizes the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    for sweep in range(50):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for j in range(n):
            if time.monotonic() - start > time_budget:
                break

            old_val = a[j]
            # Try perturbing a[j]
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                trial = a.copy()
                trial[j] = old_val * factor
                trial = np.maximum(trial, 0)
                s_t = float(np.sum(trial))
                if s_t < 1e-10:
                    continue
                trial = trial / s_t
                fa_t = rfft(trial, N_fft)
                b_t = irfft(fa_t * fa_t, N_fft)[:2 * n - 1]
                obj_t = 2.0 * n * float(np.max(b_t))
                if obj_t < best_obj - 1e-12:
                    best_obj = obj_t
                    best_a = trial.copy()
                    a = trial.copy()
                    improved = True
                    break

        if not improved:
            break

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}")

    return best_a


def multi_peak_subgrad(a, time_budget=60.0, verbose=False):
    """
    Multi-peak subgradient: uses weighted combination of top-K peak subgradients.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N_fft)
        b = irfft(fa * fa, N_fft)[:2 * n - 1]

        # Get top K peak positions
        K = min(10, 2 * n - 1)
        top_idx = np.argsort(b)[::-1][:K]
        # Weights: exponential of (b[top_k] - b[max])
        weights = np.exp(b[top_idx] - b[top_idx[0]])
        weights = weights / np.sum(weights)

        # Combined subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            # Subgrad of conv_k w.r.t. a[j] = 2*a[k-j] if 0 <= k-j < n
            k_int = int(k)
            for j in range(n):
                idx = k_int - j
                if 0 <= idx < n:
                    grad[j] += w * 2.0 * a[idx]

        grad = grad - np.mean(grad)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = proj_simplex(a_new)

        fa_new = rfft(a_new, N_fft)
        b_new = irfft(fa_new * fa_new, N_fft)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 0.5)
        else:
            eta *= 0.97
            if eta < 1e-7:
                break

    return best_a


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 10) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.2
        a = a / np.sum(a)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.5
        a = a / np.sum(a)
    elif mode == 6:
        # Sawtooth
        x = np.arange(n) / n
        a = np.abs(x - 0.5) + 0.1
        a = a / np.sum(a)
    else:
        # Gaussian cluster
        center = n // 2
        a = np.exp(-((np.arange(n) - center) ** 2) / (n * n / 20))
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    """

    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # Phase 1: LSE-FISTA from parent (primary optimization)
    print(f"\n=== Phase 1: LSE-FISTA (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.30, 300.0)
    a_lse = lse_fista(best_seq, time_budget=alloc, T_start=3.0, T_end=80.0, verbose=True)
    obj_lse = compute_obj(a_lse)
    if obj_lse < best_score:
        best_score = obj_lse
        best_seq = a_lse.copy()
        print(f"  ** LSE-FISTA: obj={best_score:.8f} **")
    else:
        print(f"  LSE-FISTA: obj={obj_lse:.8f}")

    # Phase 2: LP Polishing
    print(f"\n=== Phase 2: LP Polishing ===")

        a_lp = direct_lp_polish(best_seq, time_budget=alloc, verbose=True)

            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # Phase 3: Multi-peak subgradient
    print(f"\n=== Phase 3: Multi-Peak Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 80.0)
        a_mp = multi_peak_subgrad(best_seq, time_budget=alloc, verbose=True)
        obj_mp = compute_obj(a_mp)
        if obj_mp < best_score:
            best_score = obj_mp
            best_seq = a_mp.copy()
            print(f"  ** MultiPeak: obj={best_score:.8f} **")
        else:
            print(f"  MultiPeak: obj={obj_mp:.8f}")

    # Phase 4: Second LSE-FISTA round (higher T for sharper peak focus)
    print(f"\n=== Phase 4: LSE-FISTA Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.20, 150.0)
        a_lse2 = lse_fista(best_seq, time_budget=alloc, T_start=20.0, T_end=200.0, verbose=True)
        obj_lse2 = compute_obj(a_lse2)
        if obj_lse2 < best_score:
            best_score = obj_lse2
            best_seq = a_lse2.copy()
            print(f"  ** LSE2: obj={best_score:.8f} **")
        else:
            print(f"  LSE2: obj={obj_lse2:.8f}")

    # Phase 5: Coordinate Descent Polish
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.10, 60.0)
        a_cd = coord_descent_polish(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # Phase 6: Multi-n exploration
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200),
        n_best + 100, n_best + 200, n_best + 400,

            break

    for r in range(n_restarts):

        alloc = min(remaining * 0.20, 60.0)

        if obj_r < best_score:

        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # Phase 8: Final combined polish
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # LP polish
        alloc = min(remaining * 0.3, 60.0)
        a_final = direct_lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

        # Multi-peak
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 40.0)
            a_mp = multi_peak_subgrad(best_seq, time_budget=alloc)
            obj_mp = compute_obj(a_mp)
            if obj_mp < best_score:
                best_score = obj_mp
                best_seq = a_mp.copy()
                print(f"  ** Final MP: obj={best_score:.8f} **")

        # CD polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 30.0)
            a_cd = coord_descent_polish(best_seq, time_budget=alloc)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  ** Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:

    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
