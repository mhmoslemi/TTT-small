import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):

    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver_timeout=10.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def softmax_from_theta(theta):
    """Compute softmax with numerical stability."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def smoothed_obj_and_grad(theta, tau, n):
    """Compute smoothed objective and gradient w.r.t. log-parameters theta."""
    a = softmax_from_theta(theta)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)
    
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)
    
    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < 2 * n - 1:
            c[j] = np.dot(w[j:end], a[:end - j])
    
    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta


def smooth_lbfgs(a_init, tau_start, tau_end, time_budget, max_iter=200):
    """Smoothed L-BFGS-B optimization with tau annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    n_stages = 6
    taus = np.geomspace(tau_start, tau_end, n_stages)
    
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        stage_budget = min(remaining * 0.85, 100.0)
        
        def obj_func(th):
            val, _ = smoothed_obj_and_grad(th, tau, n)
            return val
        
        def grad_func(th):
            _, g = smoothed_obj_and_grad(th, tau, n)
            return g
        
        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-7, 'maxfun': 800})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_obj


def peak_shaving(a, time_budget=60.0, max_rounds=200):
    """
    Greedy peak shaving: iteratively find the highest peak bin, then move mass
    from positions that contribute to that peak to positions that don't.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find positions that contribute most to the peak
        # b[k*] = sum_j a[j]*a[k*-j] (with bounds)
        contrib = np.zeros(n)
        for j in range(n):
            km = k_star - j
            if 0 <= km < n:
                contrib[j] = a[j] * a[km]
        
        # We want to reduce a[j] where contrib[j] is large
        # and increase a[j'] where a[k*-j'] is small (low contribution)
        # Score: how much reducing a[j] helps (high contrib) vs. cost at other peaks
        
        # Greedy: pick top contributor j_remove and a low-impact j_add
        top_contrib_idx = np.argsort(contrib)[::-1]
        low_impact_idx = np.argsort(contrib)[:n // 3]
        
        improved = False
        for j_remove in top_contrib_idx[:10]:
            if time.monotonic() - start > time_budget:
                break
            for j_add in low_impact_idx[:10]:
                if j_add == j_remove:
                    continue
                for frac in [0.02, 0.05, 0.08, 0.12]:
                    delta = frac * a[j_remove]
                    if delta <= 0 or delta >= a[j_remove]:
                        continue
                    a[j_remove] -= delta
                    a[j_add] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[j_remove] += delta
                        a[j_add] -= delta
                        break
            if improved:
                break
        
        if not improved:
            # Try random small moves to escape
            rng = np.random.default_rng(rnd * 7919 + n)
            found = False
            for _ in range(30):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * min(a[i], a[j]) * rng.uniform(0.5, 2.0)
                if a[i] < delta:
                    continue
                a[i] -= delta
                a[j] += delta
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    found = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
            if not found:
                break
    
    return best_a


def lp_alternating(a, time_budget=60.0, max_iters=200):
    """LP alternating optimization with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Adaptive line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        ts = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= 5:
                # Perturb to escape
                rng = np.random.default_rng(it * 104729 + n)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0
    
    return best_a


def simulated_annealing(a, time_budget=120.0, T_start=0.1, T_end=0.001):
    """Simulated annealing on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    total_steps = 5000
    for step in range(total_steps):
        if time.monotonic() - start > time_budget:
            break
        
        T = T_start * (T_end / T_start) ** (step / total_steps)
        
        # Random move: pick two positions, transfer mass
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(0.005, 0.05) * min(a[i], 0.1)
        if a[i] < delta:
            continue
        
        a[i] -= delta
        a[j] += delta
        a = np.maximum(a, 0)
        new_obj = compute_obj(a)
        
        delta_obj = new_obj - cur_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            a[i] += delta
            a[j] -= delta
    
    return best_a


def gs_spectral(a, time_budget=60.0, n_iters=200):
    """
    Gerchberg-Saxton style: try to flatten the spectrum of conv(a,a)
    by modifying in the Fourier domain and projecting back.
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    a = np.maximum(a, 0).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute spectrum of a (zero-padded)
        A = np.fft.fft(a, N)
        
        # Power spectrum (squared magnitude) corresponds to circular conv = linear conv
        power = np.abs(A) ** 2
        
        # Target: flatten the power spectrum
        # Compute target as geometric mean or median
        target = np.sqrt(np.median(power[power > 0])) if np.any(power > 0) else 1.0
        
        # Modify: A_new = A / |A| * target (preserving phase, flattening amplitude)
        mask = np.abs(A) > 1e-15
        A_new = A.copy()
        A_new[mask] = A[mask] / np.abs(A[mask]) * target
        A_new[~mask] = 0
        
        # Enforce conjugate symmetry (for real output)
        A_new[1:N//2] = A_new[1:N//2]
        A_new[N - np.arange(1, N//2)] = np.conj(A_new[1:N//2])
        if N % 2 == 0:
            A_new[N//2] = np.real(A_new[N//2])
        A_new[0] = np.real(A_new[0])
        
        # Inverse FFT
        a_full = np.fft.ifft(A_new).real
        
        # Keep only first n samples, clip to non-negative
        a_new = np.maximum(a_full[:n], 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Blend with current (to avoid too abrupt changes)
        blend = 0.3
        a_try = (1 - blend) * a + blend * a_new
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
            a = a_try
    
    return best_a


def coordinate_descent(a, time_budget=30.0):
    """Fine coordinate descent targeting the peak bin."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        
        # Try moving mass away from peak contributors
        step = max(1, n // 300)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i:
                for frac in [0.01, 0.02, 0.04, 0.06]:
                    delta = frac * a[i]
                    if delta > a[i] * 0.5:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
        
        if not improved:
            rng = np.random.default_rng(it * 31 + n)
            for _ in range(40):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * a[i] * rng.uniform(0.1, 1.0)
                if a[i] < delta:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.array([a[0]])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng, method=0):
    """Generate diverse initial sequences."""
    if method == 0:
        # Linear ramp
        slope = rng.uniform(0.5, 3.0)
        a = np.linspace(1.0, slope, n)
    elif method == 1:
        # Exponential
        rate = rng.uniform(0.5, 4.0)
        a = np.exp(np.linspace(0, rate, n))
    elif method == 2:
        # Cosine bump (asymmetric)
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.1, 0.4) * n
        x = np.arange(n)
        a = np.exp(-((x - center) / width) ** 2)
    elif method == 3:
        # Two-bump
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2)) * n
        w1, w2 = rng.uniform(0.05, 0.2, 2) * n
        x = np.arange(n)
        a = np.exp(-((x - c1) / w1) ** 2) + 0.7 * np.exp(-((x - c2) / w2) ** 2)
    elif method == 4:
        # Random smooth (low-pass filtered noise)
        a = rng.random(n)
        kernel_size = max(5, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    elif method == 5:
        # S-curve (sigmoid)
        x = np.linspace(0, 1, n)
        k = rng.uniform(5, 20)
        x0 = rng.uniform(0.3, 0.7)
        a = 1.0 / (1.0 + np.exp(-k * (x - x0)))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def optimize_at_n(a_init, time_budget, rng):
    """Run multi-method optimization at a fixed n, return best."""
    n = len(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    # Phase A: Smoothed L-BFGS-B
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        alloc = min(remaining * 0.3, 80.0)
        a_opt, obj_opt = smooth_lbfgs(a_init, tau_start=10.0, tau_end=500.0,
                                       time_budget=alloc, max_iter=150)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
    
    # Phase B: Peak shaving
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = min(remaining * 0.2, 40.0)
        a_opt = peak_shaving(best_a, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
    
    # Phase C: LP alternating
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = min(remaining * 0.3, 60.0)
        a_opt = lp_alternating(best_a, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
    
    # Phase D: GS spectral
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.15, 30.0)
        a_opt = gs_spectral(best_a, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
    
    # Phase E: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.15, 25.0)
        a_opt = coordinate_descent(best_a, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
    
    return best_a, best_obj

    if remaining > 30:
        # Multiple LP alternating rounds
        for round_num in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.15, 50.0)
            a_lp = lp_alternating(best_seq, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {round_num}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {round_num}: obj={obj_lp:.8f}")
    
    # Peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 50.0)
        a_ps = peak_shaving(best_seq, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Peak shaving: obj={best_score:.8f} **")
        else:
            print(f"  Peak shaving: obj={obj_ps:.8f}")
    
    # === Phase 5: Final diverse restarts ===
    print(f"\n=== Phase 5: Final Diverse Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        method = r % 6
        a_r = diverse_initialization(n_best, rng, method=method)
        # Blend with best to keep it in a good region
        blend = rng.uniform(0.2, 0.5)
        a_r = blend * best_seq + (1 - blend) * a_r
        a_r = np.maximum(a_r, 0)
        a_r = a_r / np.sum(a_r)
        
        alloc = min(remaining * 0.1, 40.0)
        a_opt, obj_opt = optimize_at_n(a_r, alloc, rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r} (method={method}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (method={method}): obj={obj_obj:.8f}")
    
    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final Fine-Tuning ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 20:
        alloc = min(remaining * 0.25, 50.0)
        a_opt, obj_opt = smooth_lbfgs(best_seq, tau_start=50.0, tau_end=1000.0,
                                       time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final smooth: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_ps = peak_shaving(best_seq, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Final PS: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
