Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast O(n) subgradient computation via shift-reverse formula,
mirror descent with adaptive step size, periodic LP refinement,
spectral initialization, and multi-n exploration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_subgrad_fast(a, top_lags, weights):
    """Compute weighted subgradient: grad[j] = sum_m w_m * 2*a[k_m - j] (valid range)."""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, top_lags):
        k = int(k)
        if k < 0 or k >= 2 * n - 1:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # a[k-j] for j in [j_min, j_max]
            # k-j ranges from k-j_max down to k-j_min
            idx_start = k - j_max
            idx_end = k - j_min + 1
            seg = a[idx_start:idx_end][::-1]
            grad[j_min:j_max + 1] += w * 2.0 * seg
    return grad

    if s < 1e-10:
        return a

    while L < 2 * n - 1:
        L <<= 1

    best_obj = compute_obj(a)

    stall = 0
    rel = relaxation

        if time.monotonic() - start > time_budget:
            break

        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-12)

        a_new = ifft(A_new).real[:n]

        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                # Slowly increase relaxation
                rel = min(rel * 1.01, 0.95)
            else:
                stall += 1
                rel = max(rel * 0.98, 0.3)
                if stall > 40:
                    rng = np.random.default_rng((it * 13 + n) % (2**32))
                    a = best_a * np.exp(rng.normal(0, 0.04, n))
                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
                    rel = relaxation

    return best_a


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    # Build constraint matrix: (2n-1) x n where A[k,j] = a_norm[k-j] if valid
    # conv(a_norm, g)[k] = sum_j a_norm[k-j] * g[j] (if 0<=k-j<n)
    # So A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0

    # Use scipy linprog
    from scipy.optimize import linprog

    m = 2 * n - 1
    # A_ub shape: (m, n+1), b_ub shape: (m,)
    # Constraints: sum_j A[k,j]*g[j] - M <= 0 for all k
    A_ub = np.zeros((m, n + 1))
    for j in range(n):
        # a_norm[k-j] is valid for k in [j, j+n-1]
        k_start = j
        k_end = min(m - 1, j + n - 1)
        # a_norm[k-j] for k in [k_start, k_end] is a_norm[0 : k_end-k_start+1]
        A_ub[k_start:k_end + 1, j] = a_norm[:k_end - k_start + 1]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    c = np.zeros(n + 1)
    c[n] = 1.0

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def lp_alternating_v2(a, time_budget=60.0, max_iters=30, verbose=False):
    """LP alternating with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        # Line search
        best_t_obj = np.inf
        best_t = 0.3
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall > 8:
                rng = np.random.default_rng((it * 7 + n) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """Fast coordinate refinement targeting peak reduction."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, L)
        conv = irfft(fa * fa, L)[:2 * n - 1]
        k_star = int(np.argmax(conv))

        # For the peak lag k_star, find positions to reduce
        # grad[j] = 2*a[k_star - j] (valid range)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)

        # Find j with largest grad (most contributes to peak)
        if j_min <= j_max:
            idx_start = k_star - j_max
            idx_end = k_star - j_min + 1
            seg = a[idx_start:idx_end][::-1]
            grad_seg = 2.0 * seg
            grad_full = np.zeros(n)
            grad_full[j_min:j_max + 1] = grad_seg

            # Reduce top contributors, add to non-contributors
            top_contrib = np.argsort(grad_full)[-3:]
            non_contrib = np.argsort(grad_full)[:3]

            improved = False
            for tc in top_contrib:
                for nc in non_contrib:
                    if tc == nc:
                        continue
                    for frac in [0.02, 0.05, 0.1]:
                        delta = frac * a[tc]
                        if a[tc] - delta < 0:
                            continue
                        a[tc] -= delta
                        a[nc] += delta
                        fa2 = rfft(a, L)
                        conv2 = irfft(fa2 * fa2, L)[:2 * n - 1]
                        obj2 = 2.0 * n * float(np.max(conv2))
                        if obj2 < best_obj - 1e-13:
                            best_obj = obj2
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[tc] += delta
                            a[nc] -= delta
                            break
                if improved:
                    break

        if not improved and it > 50:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_init(n, rng):
    """Initialize from a target flat spectrum."""
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    # Random phases with flat magnitude
    target_mag = np.ones(L) / np.sqrt(n)
    target_mag[0] = 1.0
    phases = np.exp(1j * rng.uniform(0, 2 * np.pi, L))
    # Symmetry for real signal
    phases[L - np.arange(1, L // 2 + 1)] = np.conj(phases[np.arange(1, L // 2 + 1)])

    A = target_mag * phases
    a = ifft(A).real[:n]
    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    return a


def random_init(n, rng, style=0):
    """Generate diverse random initializations."""
    if style == 0:
        a = np.ones(n) / n
    elif style == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / np.sum(a)
    elif style == 2:
        # Gaussian bumps
        centers = rng.uniform(0.2, 0.8, 2).astype(float) * n
        widths = rng.uniform(n * 0.05, n * 0.2, 2)
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    elif style == 3:
        # Anti-symmetric pair (high at edges)
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2 + 0.5
        a = a / np.sum(a)
    elif style == 4:
        # Spectral init
        a = spectral_init(n, rng)
    else:
        # Low-frequency dominant
        a = np.ones(n)
        for k in range(1, min(5, n // 4)):
            a += 0.3 * np.cos(2 * np.pi * k * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining fast mirror descent, LP, FFT, subgradient,
    coordinate refinement, multi-n exploration, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Quick spectral init + FFT to get a good starting point ===
    print("\n=== Phase 1: Spectral Init + FFT ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for style in [0, 2, 4]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_try = random_init(n_parent, rng, style)
            alloc = min(remaining * 0.1, 15.0)
            a_fft = fft_flat_spectrum_v2(a_try, n_iters=300, time_budget=alloc, relaxation=0.6)
            obj = compute_obj(a_fft)
            if obj < best_score:
                best_score = obj
                best_seq = a_fft.copy()
                print(f"  Style {style}: obj={obj:.8f} **")

    # === Phase 2: Main mirror descent (the workhorse) ===
    print(f"\n=== Phase 2: Mirror Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.3, 300.0)
        a_md = mirror_descent_v2(best_seq, n_iters=50000, time_budget=alloc, n_peaks=7, verbose=True)
        obj = compute_obj(a_md)
        print(f"  MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Only run LP if n is small enough
        if len(best_seq) <= 600:
            alloc = min(remaining * 0.2, 120.0)
            a_lp = lp_alternating_v2(best_seq, time_budget=alloc, max_iters=20, verbose=True)
            obj = compute_obj(a_lp)
            print(f"  LP: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()

    # === Phase 4: Nesterov subgradient ===
    print(f"\n=== Phase 4: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 90.0)
        a_ns = subgradient_prox(best_seq, n_iters=10000, time_budget=alloc, n_peaks=5)
        obj = compute_obj(a_ns)
        print(f"  Nesterov: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_ns.copy()

    # === Phase 5: Second mirror descent round ===
    print(f"\n=== Phase 5: Mirror Descent Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 120.0)
        a_md2 = mirror_descent_v2(best_seq, n_iters=50000, time_budget=alloc, n_peaks=9, verbose=False)
        obj = compute_obj(a_md2)
        print(f"  MD2: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md2.copy()

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        400, 500, 600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if abs(n_try - n_best) < 20:
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc1 = min(remaining * 0.15, 30.0)
        a_opt = mirror_descent_v2(a_try, n_iters=20000, time_budget=alloc1, n_peaks=7)
        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try

    # === Phase 7: Random restarts with different n ===
    print(f"\n=== Phase 7: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_restarts = max(3, int(remaining / 30))
    n_restarts = min(n_restarts, 15)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        n_r = int(n_best + rng.integers(-100, 100))
        n_r = max(300, min(1200, n_r))
        style = r % 6
        a_r = random_init(n_r, rng, style)

        alloc_md = min(remaining * 0.3, 25.0)
        a_opt = mirror_descent_v2(a_r, n_iters=10000, time_budget=alloc_md, n_peaks=7)
        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (n={n_r}, style={style}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_final = len(best_seq)

        # Quick LP if feasible
        if n_final <= 500 and remaining > 40:
            alloc = min(remaining * 0.25, 45.0)
            a_lp = lp_alternating_v2(best_seq, time_budget=alloc, max_iters=10)
            obj = compute_obj(a_lp)
            print(f"  Final LP: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()

        # Final MD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 45.0)
            a_md = mirror_descent_v2(best_seq, n_iters=30000, time_budget=alloc, n_peaks=11, verbose=False)
            obj = compute_obj(a_md)
            print(f"  Final MD: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_md.copy()

        # Coordinate refine
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.2, 20.0)
            a_cd = coordinate_refine(best_seq, time_budget=alloc)
            obj = compute_obj(a_cd)
            print(f"  Coord: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()

        # One more FFT
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.2, 20.0)
            a_fft = fft_flat_spectrum_v2(best_seq, n_iters=200, time_budget=alloc, relaxation=0.5)
            obj = compute_obj(a_fft)
            print(f"  Final FFT: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_fft.copy()

        # Last MD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.3, 25.0)
            a_md_last = mirror_descent_v2(best_seq, n_iters=20000, time_budget=alloc, n_peaks=7, verbose=False)
            obj = compute_obj(a_md_last)
            print(f"  Last MD: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_md_last.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:

    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
