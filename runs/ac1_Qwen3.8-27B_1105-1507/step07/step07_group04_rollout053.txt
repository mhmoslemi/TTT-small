But there's a subtlety: the LP `min M s.t. A(a) @ g <= M, sum(g)=1, g>=0` finds the best g for a fixed a. But the "true" next step in the alternating minimization would be to minimize `max_k conv(a, g)[k]` over g, which IS exactly what the LP does (since conv(a,g) is linear in g).

2. Update a ← (1-t)*a + t*g*

The correct comparison is:
- Current objective: `f(a) = max_k conv(a,a)[k]` (with sum(a)=1)
- New objective: `f(g*) = max_k conv(g*,g*[k]` (with sum(g*)=1)

If `f(g*) < f(a)`, we take the full step. Otherwise, we blend.

In practice, the LP alternating tends to converge, so the full step usually works or nearly works.

"""
LP-Alternating Optimizer with Multi-n Search and Subgradient Polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp(a):
    """Solve LP: min M s.t. A(a) @ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build A matrix: A[k, j] = a[k-j] if valid
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    # LP: min [0,...,0,1] @ [g, M]
    # s.t. [A, -1] @ [g, M] <= 0
    #      [1,...,1, 0] @ [g, M] = 1
    #      g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs
    return None

def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    stall = 0
    consecutive_improvements = 0
    
    while time.monotonic() - start < time_budget:
        g_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Try full step first (t=1)
        a_try = g_lp.copy()
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-14:
            # Full step works
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
            consecutive_improvements += 1
        else:
            # Try smaller steps
            improved = False
            for t in [0.5, 0.3, 0.1]:
                a_try = (1 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try

"""
LP-Alternating Optimizer with Multi-n Search, Subgradient Polish, and Structured Initializations.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences of length n.
Uses scipy HiGHS LP solver for the alternating step, with adaptive line search,
multi-n exploration, and Nesterov-momentum subgradient polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A(a) @ g <= M*1, sum(g)=1, g>=0, M>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, time_budget=100.0, verbose=False, over_relax=False):
    """
    LP alternating optimization with adaptive line search.
    Optionally uses over-relaxation for faster convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    last_t = 1.0
    prev_obj = best_obj
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Try candidates in order of preference
        candidates = []
        if over_relax and stall == 0 and it > 5:
            candidates.append(1.2)
        candidates.append(1.0)
        if 0 < last_t < 1.0:
            candidates.append(last_t)
        candidates.extend([0.7, 0.5, 0.3, 0.1])

        # Deduplicate and sort by preference
        seen = set()
        unique_candidates = []
        for t in candidates:
            t_round = round(t, 6)
            if t_round not in seen and 0 < t <= 1.5:
                seen.add(t_round)
                unique_candidates.append(t)

        best_t_obj = np.inf
        best_t_val = None
        best_a_local = None

        for t in unique_candidates:
            if t <= 1.0:
                a_try = (1.0 - t) * a + t * g_lp
            else:
                a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            last_t = best_t_val
            stall = 0
            if verbose and (it <= 5 or it % 20 == 0):
                print(f"    LP iter {it}: obj={best_obj:.10f}, t={best_t_val:.3f}, "
                      f"elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 20:
                # Perturb to escape local convergence
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                scale = 0.02 + 0.005 * min(stall // 20, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
                last_t = 1.0

    return best_a


def subgradient_optimize(a, time_budget=60.0, top_k=5):
    """Nesterov-momentum subgradient optimization for the non-convex objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.85
    step_size = 0.03
    x = a.copy()

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        k_indices = np.argsort(conv)[::-1][:top_k]

        # Compute subgradient: average of 2*a[k-i] over top-k positions
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx_start = k - (i_end - 1)
            idx_end = k - i_start + 1
            grad[i_start:i_end] += 2.0 * a[idx_start:idx_end][::-1]
        grad /= len(k_indices)

        # Center gradient (project to sum=0 for the sum(a)=1 constraint)
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        momentum = beta * momentum - step_size * grad_centered
        new_a = a + momentum
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            new_a = a.copy()
        else:
            new_a = new_a / s_new

        # Update Nesterov lookahead
        t_k = 2.0 / (it + 2.0)
        x = new_a + (1.0 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = new_a
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Decay step size geometrically
        if it % 500 == 499:
            step_size = max(step_size * 0.9, 0.002)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, kind='rectangular'):
    """Generate structured initial sequences."""
    if kind == 'rectangular':
        return np.ones(n) / n
    elif kind == 'triangular':
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
        return a / np.sum(a)
    elif kind == 'raised_cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-10:
            a /= s
        else:
            a = np.ones(n) / n
        return a
    elif kind == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-0.5 * x**2)
        return a / np.sum(a)
    elif kind == 'bipolar_smooth':
        # Smooth version of a bipolar pattern
        x = np.linspace(0, 2 * np.pi * n, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-10:
            a /= s
        return a
    else:
        return np.ones(n) / n
