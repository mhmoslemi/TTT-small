import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    n = len(a)
    conv = compute_conv(a)

        indices = np.array([k_star])
        weights = np.array([1.0])

        min_val = conv[sorted_idx[-1]]
        if max_val > min_val:

        else:
            weights = np.ones(top_m) / top_m
        weights = weights / weights.sum()
        indices = sorted_idx

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * w * a[k - i]
    return grad, conv
```

Wait, this loop is O(n*m) where m is top_m. For large n this could be slow. Let me use FFT or correlate instead.

For a single bin k, the gradient contribution is 2*a[k-i] for i in [max(0, k-n+1), min(n-1, k)]. This is just a shifted version of a.

For multiple bins with weights, it's a weighted sum of shifted versions of a.

I can compute this using np.correlate or by direct indexing:

```python
def subgradient_fast(a, active_bins, weights):
    """Compute weighted sum of subgradients for active bins."""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, active_bins):
        # grad[i] += 2*w*a[k-i] for valid i
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # a[k-i] for i in [lo, hi] is a[k-hi : k-lo+1] reversed... 
        # Actually: when i=lo, k-i = k-lo; when i=hi, k-i = k-hi
        # So a[k-i] for i=lo..hi is a[k-lo : k-hi+1] in reverse
        # Let's just do it directly
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi+1] += 2.0 * w * a[idx]
    return grad
```

This is O(n) per active bin, so O(n*m) total. For n=800 and m=5, that's 4000 operations per subgradient evaluation, which is fast.

Now let me think about the overall algorithm more carefully.

**Key insight**: The problem is convex (max of convex functions over a convex set). The projected subgradient method with Nesterov acceleration should converge, albeit slowly (O(1/sqrt(k)) for non-smooth, O(1/k) for accelerated).

But we can do better by exploiting the structure:
1. The subgradient is always a (weighted) shifted version of a
2. This means the update direction has a specific structure

**Alternative: Direct minimization via bisection on the max**

Here's an idea: We want to find min max_k conv[k] subject to sum(a)=1, a>=0.

Let M* be the optimal value. We can binary search on M:
- For a given M, check if there exists a >= 0, sum(a)=1, such that conv[k] <= M for all k.
- This is a feasibility LP: sum(a)=1, a>=0, A@a <= M*1 where A is the convolution matrix.
- The LP is: min 0 subject to A@a <= M*1, sum(a)=1, a>=0.
- This LP is feasible iff M >= M*.

So we can binary search on M and solve an LP at each step. But the LP has 2n-1 + 1 constraints and n variables, which for n=800 is about 1600 constraints and 800 variables. This is manageable for modern LP solvers.

Actually, this is essentially the same as the LP alternating approach, but instead of alternating, we're doing a direct bisection on M. The advantage is that we get a certificate of optimality (dual feasible solution).

But the LP might be slow. Let me think of a faster approach.

**Fast approach: Iterative water-filling / projected gradient**

Since the subgradient is simple (shifted a), we can implement a very fast projected gradient method:

```python
def projected_subgradient(a, budget_s, top_m=5):
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    gamma = 1.0  # initial step size
    t = 1.0
    y = a.copy()  # Nesterov momentum point
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.time() - start > budget_s:
            break
        
        # Compute active bins
        conv = compute_conv(y)
        max_val = conv.max()
        # Active bins: those within 5% of max
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        # Take top_m most active
        if len(active) > top_m:
            active = active[np.argsort(conv[active])[::-1][:top_m]]
        
        # Weights proportional to conv values
        weights = conv[active] / conv[active].sum()
        
        # Compute subgradient
        grad = subgradient_fast(y, active, weights)
        
        # Nesterov step
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t - 1) / t_new
        y_new = y + beta * (y - a_prev)
        
        # Projected gradient step
        a_new = simplex_project(y_new - (gamma / t_new) * grad)
        
        # Backtracking line search (simplified)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a_prev = a
        a = a_new
        y = a_new
        t = t_new
    
    return best_a
```

Hmm, but this might be slow because of the simplex projection at each step (O(n log n) due to sorting). For n=800, that's about 800*10 = 8000 operations per projection, which is fine.

Let me also think about the **Frank-Wolfe** approach, which avoids the projection:
- At each step, find v = argmin_{s in simplex} grad^T s = e_{argmin(grad)}
- Update: a ← (1-γ)*a + γ*v

This is O(n) per step (no sorting needed). Much faster!

But Frank-Wolfe can be slow to converge near the optimum. A hybrid approach:
- Use Frank-Wolfe for the bulk of the optimization
- Switch to projected gradient for the final polish

Let me also consider **accelerated Frank-Wolfe (SFWR - Strongly Convex Frank-Wolfe)** or **Away-sets Frank-Wolfe (AFW)**.

AFW is particularly nice: it maintains a working set S of vertices and projects onto the face of the simplex defined by S. This gives faster convergence near the optimum.

OK let me just write a comprehensive solution that combines:
1. Frank-Wolfe with away-sets (fast, no sorting)
2. Projected subgradient with Nesterov (for polish)
3. Coordinate descent (for very fine tuning)
4. Multi-n exploration
5. LP for final certification

Let me write the code:

```python
"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Frank-Wolfe with away-sets, Nesterov-accelerated projected subgradient,
coordinate descent, and LP-based verification.
Key innovations: away-set Frank-Wolfe for fast convergence, active-set
subgradient with adaptive bin selection, and exact coordinate descent.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def conv_subgradient_at_bin(a, k):
    """Compute gradient of conv[k] w.r.t. a. Returns array of length n."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] = 2.0 * a[idx]
    return grad


def get_active_bins(conv, top_m=5, rel_tol=0.1):
    """Get active bins (near max) with weights."""
    max_val = conv.max()
    threshold = max_val * (1 - rel_tol)
    active = np.where(conv >= threshold)[0]
    if len(active) > top_m:
        # Take the top_m
        sorted_active = active[np.argsort(conv[active])[::-1][:top_m]]
    else:
        sorted_active = active
    # Weights: proportional to conv value (normalized)
    w = conv[sorted_active].copy()
    w = w / w.sum()
    return sorted_active, w


def compute_weighted_subgradient(a, active_bins, weights):
    """Compute weighted sum of subgradients for active bins."""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, active_bins):
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            idx = k - np.arange(lo, hi + 1)
            grad[lo:hi + 1] += 2.0 * w * a[idx]
    return grad


def frank_wolfe_away(a, budget_s=60.0, top_m=5):
    """
    Frank-Wolfe with away-sets for minimizing max conv on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Working set for away-FW
    S = set()
    S.add(int(np.argmax(a)))  # Start with the largest coordinate
    S.add(int(np.argmin(a))) if n > 1 else None

    max_iters = 2000
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > budget_s:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        active_bins, weights = get_active_bins(conv, top_m=top_m, rel_tol=0.1)

        # Compute subgradient
        grad = compute_weighted_subgradient(a, active_bins, weights)

        # Linear oracle: min grad^T v over simplex = grad[argmin(grad)]
        v_idx = int(np.argmin(grad))

        # Away step: maximize grad^T v over face of S
        if len(S) > 1:
            s_list = list(S)
            s_arr = np.array(s_list)
            # Find s in S that maximizes grad[s]
            away_idx = s_list[int(np.argmax(grad[s_arr]))]
            # Coefficient for away step
            denom = grad[away_idx] - grad[v_idx]
            if denom > 1e-10:
                gamma_away = (a[away_idx] * (grad[away_idx] - grad[v_idx])) / (
                    (grad[away_idx] - grad.mean()) * a.sum() + 1e-10)
                # Simplified: just use a fraction
                gamma_away = min(0.5, a[away_idx] / (a.sum() + 1e-10) * 0.5)
            else:
                gamma_away = 0.0
        else:
            gamma_away = 0.0

        # Combine step and away step
        gamma_step = 2.0 / (it + 2)  # Standard FW step size
        a_new = a.copy()

        # Step towards v
        a_new += gamma_step * (np.eye(n)[v_idx] - a)

        # Away step
        if gamma_away > 0 and away_idx in S:
            a_new[away_idx] -= gamma_away
            if a_new[away_idx] < 0:
                a_new[away_idx] = 0
                S.discard(away_idx)

        # Normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Update working set
        if a_new[v_idx] > 0.01:
            S.add(v_idx)
        # Remove small elements from S
        S = set(s for s in S if a_new[s] > 0.005)
        if not S:
            S = {int(np.argmax(a_new))}

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # Still update to explore
            stall += 1
            if stall > 100:
                # Reset
                a = best_a.copy()
                S = {int(np.argmax(a))}
                stall = 0

    return best_a


def nesterov_projected_subgrad(a, budget_s=60.0, top_m=5):
    """
    Nesterov-accelerated projected subgradient method.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t = 1.0
    y = a.copy()  # Nesterov lookahead
    a_prev = a.copy()
    gamma = 10.0  # Learning rate for subgradient step

    max_iters = 5000
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > budget_s:
            break

        conv = compute_conv(y)
        max_val = float(np.max(conv))
        active_bins, weights = get_active_bins(conv, top_m=top_m, rel_tol=0.05)
        grad = compute_weighted_subgradient(y, active_bins, weights)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y_momentum = y + beta * (y - a_prev)

        # Projected gradient step
        step = gamma / t_new
        y_candidate = y_momentum - step * grad
        a_new = simplex_project(y_candidate)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                gamma = max(gamma * 0.8, 0.1)
                y = a_new
                a_prev = a_new
                a = a_new
                t = t_new
                stall = 0
                continue

        a_prev = a
        a = a_new
        y = a_new
        t = t_new

    return best_a


def coordinate_descent(a, budget_s=30.0):
    """
    Coordinate descent: for each coordinate, try to find a better value
    by 1D minimization of the max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(20):
        if time.monotonic() - start > budget_s:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - start > budget_s:
                break

            old_val = a[i]
            if old_val < 1e-8:
                continue

            # Try different values for a[i]
            # The optimal c minimizes max_k conv[k](c)
            # conv[k](c) is:
            #   If k-i is valid and k != 2i: 2*c*a[k-i] + C_k (linear in c)
            #   If k = 2i: c^2 + C_k (quadratic in c)
            #   Otherwise: C_k (constant)
            
            # Compute C_k = conv[k] - contribution of a[i]
            conv_current = compute_conv(a)
            max_current = float(np.max(conv_current))
            
            # Contribution of a[i] to conv[k]:
            # 2*old_val*a[k-i] if 0 <= k-i < n and k != 2i
            # old_val^2 if k = 2i
            # 0 otherwise
            
            # Try: c = 0, c = old_val * 0.5, c = old_val * 0.8, c = old_val * 1.2
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val * 1.25]
            # Also try the value that equalizes the top active bins
            top_k = int(np.argmax(conv_current))
            if 0 <= top_k - i < n and top_k != 2 * i:
                # conv[top_k](c) = 2*c*a[top_k-i] + C_top
                # C_top = conv_current[top_k] - 2*old_val*a[top_k-i]
                c_top = conv_current[top_k] - 2 * old_val * a[top_k - i]
                if a[top_k - i] > 1e-10:
                    # Try to make conv[top_k] = max of other bins
                    # For now, just try reducing
                    candidates.append(c_top * 0.5 / a[top_k - i] if a[top_k - i] > 0 else 0)
            
            best_c = old_val
            best_c_obj = best_obj
            
            for c in candidates:
                c = max(0.0, c)
                a_try = a.copy()
                a_try[i] = c
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                conv_try = compute_conv(a_try)
                obj_try = 2.0 * n * float(np.max(conv_try))
                if obj_try < best_c_obj:
                    best_c_obj = obj_try
                    best_c = c
            
            if best_c != old_val:
                a[i] = best_c
                s_total = a.sum()
                a = a / s_total
                obj_a = compute_obj(a)
                if obj_a < best_obj:
                    best_obj = obj_a
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None

    n = len(a)
    a = a / np.sum(a)
    
    # Build A matrix: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # conv(a, g)[k] = sum_j a[k-j] * g[j] (with valid indices)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] += a[k-j] * g[j]
        for k in range(j, min(2 * n - 1, j + n)):
            A[k, j] = a[k - j]
    
    # LP: min M subject to A@g <= M, sum(g) = 1, g >= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # Minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@g - M <= 0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None


def lp_polish(a, budget_s=30.0):
    """LP alternating polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(50):
        if time.monotonic() - start > budget_s:
            break

        result = solve_lp_minpeak(a)
        if result is None:
            break
        g_lp, M_lp = result

        # Line search
        best_t = 0.5
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            a = (1 - best_t) * best_a + best_t * g_lp
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_obj = best_t_obj
            best_a = a.copy()
        else:
            break

    return best_a


def spectral_init(n, rng):
    """Initialize using spectral considerations - try to make |FFT(a)|^2 flat."""
    # A flat power spectrum means the auto-correlation is a delta,
    # which means the auto-convolution is minimized.
    # We can try to construct a with approximately flat |FFT|.
    
    # Start with random and iterate a few times
    a = rng.uniform(0.1, 1.0, n)
    a = a / a.sum()
    
    for _ in range(20):
        fa = rfft(a, 2 * n)
        power = np.abs(fa) ** 2
        # Try to flatten the power spectrum
        # a_new should have flatter spectrum
        # Inverse: if we want |FFT(a_new)|^2 ≈ constant, then a_new ≈ IFFT(sqrt(const))
        # But we need a_new >= 0, so just use as a direction
        target = 1.0 / np.sqrt(np.maximum(power, 1e-10))
        target_n = len(target) // 2 + 1
        a_new = np.abs(irfft(target * np.exp(1j * np.angle(fa)), 2 * n))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            if compute_obj(a_new) < compute_obj(a):
                a = a_new
        else:
            break
    
    return a
