import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix T (2n-1 x n) such that T @ g = conv(a, g)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(2 * n - 1, n + j)
        T[:k_end, j] = a[:k_end]
    return T


def solve_lp_minpeak(a, verbose=False):

                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a.copy()
            last_t = best_t_val
            stall = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f} ({elapsed:.1f}s)")
        else:
            stall += 1
            if stall > 10:
                # Perturb to escape local minimum
                scale = min(0.02 + 0.01 * (stall // 10), 0.08)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                last_t = 0.5
                stall = 0

    return best_a


def mirror_descent(a, n_iters=3000, time_budget=60.0, top_k=8):
    """
    Mirror descent (exponential) with multi-peak subgradient and
    adaptive learning rate with backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = np.maximum(a, 1e-15) / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.3
    stall = 0
    rng = np.random.default_rng(67890)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top peaks with softmax weighting
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[sorted_idx]
        max_val = top_vals[0]
        min_top = top_vals[-1]

        temp = max(1e-6, (max_val - min_top) * 1.0)
        weights = np.exp((top_vals - max_val) / temp)
        weights /= weights.sum()

        # Weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            ki = int(k)
            # grad[j] += w * 2 * a[ki - j] if 0 <= ki-j < n
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_end > j_start:
                indices = np.arange(j_start, j_end)
                a_indices = ki - indices
                grad[j_start:j_end] += w * 2.0 * a[a_indices]

        # Mirror descent step
        a_new = a * np.exp(-eta * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                eta *= 0.9
                stall += 1
                if stall > 50:
                    # Perturb and reset
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    eta = 0.3
                    stall = 0
        else:
            eta *= 0.5

        # Periodically boost eta
        if it % 1000 == 999:
            eta = min(eta * 1.3, 1.0)

    return best_a


def combined_lp_md(a, time_budget=60.0, verbose=False):
    """
    Hybrid: alternate between LP steps and mirror descent steps.
    LP gives principled direction, MD gives fast local convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        if stall > 20:
            # Random restart from best
            rng = np.random.default_rng(it + n)
            a = best_a * np.exp(rng.normal(0, 0.04, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            best_a = a.copy()
            best_obj = compute_obj(a)
            stall = 0

        # LP step (every 3rd iteration for speed)
        if it % 3 == 0:
            g_lp, M_lp = solve_lp_minpeak(a)
            if g_lp is not None:
                # Try several step sizes
                best_t_obj = best_obj
                best_t = 0.2
                for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    a_try = (1.0 - t) * a + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_t = t

                a_new = (1.0 - best_t) * a + best_t * g_lp
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        stall = 0
                        if verbose and it % 30 == 0:
                            print(f"      LP+MD it{it}: obj={best_obj:.8f}")
                    else:
                        stall += 1

        # Mirror descent step
        else:
            conv = compute_conv(a)
            top_k = 5
            sorted_idx = np.argsort(conv)[::-1][:top_k]
            top_vals = conv[sorted_idx]
            max_val = top_vals[0]
            min_top = max(top_vals[-1], max_val - 1e-10)
            temp = max(1e-6, (max_val - min_top))
            weights = np.exp((top_vals - max_val) / temp)
            weights /= weights.sum()

            grad = np.zeros(n)
            for w, k in zip(weights, sorted_idx):
                ki = int(k)
                j_start = max(0, ki - n + 1)
                j_end = min(n, ki + 1)
                if j_end > j_start:
                    indices = np.arange(j_start, j_end)
                    grad[j_start:j_end] += w * 2.0 * a[ki - indices]

            eta_md = 0.15
            a_new = a * np.exp(-eta_md * grad)
            a_new = np.maximum(a_new, 1e-15)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1

    return best_a


def fft_flat_spectrum(a, n_iters=300, time_budget=30.0, relaxation=0.5):
    """FFT-based flat spectrum iteration."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(42)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = np.fft.fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-12)
        phase = A / mag_safe

        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    a = best_a * np.exp(rng.normal(0, 0.04, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 2:
        # Smooth random
        freqs = rng.uniform(0.5, 6.0, 6)
        phases = rng.uniform(0, 2 * np.pi, 6)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 3:
        # Bimodal
        c1, c2 = n // 3, 2 * n // 3
        w = max(1, n // 12)
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        return a / a.sum()
    elif mode == 5:
        # Random with smoothing
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # Anti-symmetric (high at edges, low in middle or vice versa)
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        # Geometric
        r = rng.uniform(0.95, 1.01)
        a = r ** np.arange(n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))
        return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP alternating, mirror descent, FFT flat spectrum,
    multi-n exploration, and random restarts with adaptive time allocation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Quick FFT flat spectrum to get a decent starting point ===
    print(f"\n=== Phase 1: FFT Flat Spectrum (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_fft = fft_flat_spectrum(best_seq, n_iters=400, time_budget=20.0, relaxation=0.5)
        obj = compute_obj(a_fft)
        print(f"  FFT relax=0.5: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fft.copy()

        a_fft2 = fft_flat_spectrum(best_seq, n_iters=200, time_budget=15.
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_fft2 = fft_flat_spectrum(best_seq, n_iters=200, time_budget=15.0, relaxation=0.3)
            obj2 = compute_obj(a_fft2)
            print(f"  FFT relax=0.3: obj={obj2:.8f}")
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_fft2.copy()

    # === Phase 2: LP Alternating (main optimization) ===
    print(f"\n=== Phase 2: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 200.0)
        a_lp = lp_alternating(best_seq, n_max_iters=500, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  LP round 1: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # === Phase 3: Combined LP+MD ===
    print(f"\n=== Phase 3: Combined LP+MD ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 100.0)
        a_hybrid = combined_lp_md(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_hybrid)
        print(f"  Combined LP+MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_hybrid.copy()

    # === Phase 4: Mirror Descent refinement ===
    print(f"\n=== Phase 4: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.12, 80.0)
        a_md = mirror_descent(best_seq, n_iters=5000, time_budget=alloc, top_k=10)
        obj = compute_obj(a_md)
        print(f"  MD (top_k=10): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # === Phase 5: More LP Alternating ===
    print(f"\n=== Phase 5: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 120.0)
        a_lp2 = lp_alternating(best_seq, n_max_iters=400, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp2)
        print(f"  LP round 2: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_options = []
        for delta in [-100, -50, -25, 25, 50, 100, 150, 200]:
            n_try = n_best + delta
            if 300 <= n_try <= 1500 and n_try != n_best:
                n_options.append(n_try)
        # Also try some fixed values
        for n_fix in [500, 600, 700, 900, 1000, 1100, 1200]:
            if 300 <= n_fix <= 1500 and abs(n_fix - n_best) > 30:
                n_options.append(n_fix)
        n_options = sorted(set(n_options))

        for n_try in n_options[:8]:  # Limit to 8 different n values
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            a_try = resample_sequence(best_seq, n_try)

            # Quick optimization: FFT + MD
            alloc1 = min(10.0, remaining * 0.05)
            a_opt = fft_flat_spectrum(a_try, n_iters=150, time_budget=alloc1, relaxation=0.5)
            alloc2 = min(15.0, remaining * 0.08)
            a_opt = mirror_descent(a_opt, n_iters=2000, time_budget=alloc2, top_k=8)
            alloc3 = min(20.0, remaining * 0.1)
            a_opt = lp_alternating(a_opt, n_max_iters=100, time_budget=alloc3)

            obj = compute_obj(a_opt)
            status = " ***" if obj < best_score else ""
            print(f"  n={n_try}: obj={obj:.8f}{status}")
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()

    # === Phase 7: Random restarts from diverse initializations ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    n_restarts = min(12, max(4, int(remaining // 30)))
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        mode = (r + seed) % 8
        n_r = n_best
        if r % 3 == 1:
            n_r = int(n_best + rng.integers(-100, 100))
            n_r = max(300, min(1300, n_r))
            if n_r == n_best:
                n_r = n_best + 50

        a_r = diverse_init(n_r, rng, mode)
        if r % 3 == 2:
            # Start from perturbed best
            a_best_resampled = resample_sequence(best_seq, n_r)
            a_r = 0.5 * a_r + 0.5 * a_best_resampled
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        # Optimize the restart
        alloc_fft = min(8.0, remaining * 0.15)
        a_opt = fft_flat_spectrum(a_r, n_iters=100, time_budget=alloc_fft, relaxation=0.5)
        alloc_md = min(10.0, remaining * 0.2)
        a_opt = mirror_descent(a_opt, n_iters=1500, time_budget=alloc_md, top_k=8)
        alloc_lp = min(15.0, remaining * 0.3)
        a_opt = lp_alternating(a_opt, n_max_iters=80, time_budget=alloc_lp)

        obj_r = compute_obj(a_opt)
        status = " ***" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}, n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_final = len(best_seq)
        a_final = best_seq.copy()

        # LP
        alloc = min(remaining * 0.35, 60.0)
        a_lp = lp_alternating(a_final, n_max_iters=300, time_budget=alloc, verbose=False)
        obj = compute_obj(a_lp)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            a_final = a_lp.copy()

        # MD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 30.0)
            a_md = mirror_descent(a_final, n_iters=3000, time_budget=alloc, top_k=12)
            obj = compute_obj(a_md)
            print(f"  Final MD: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_md.copy()
                a_final = a_md.copy()

        # Combined
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.4, 40.0)
            a_comb = combined_lp_md(a_final, time_budget=alloc, verbose=False)
            obj = compute_obj(a_comb)
            print(f"  Final Combined: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_comb.copy()

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.6, 30.0)
            a_last = lp_alternating(best_seq, n_max_iters=200, time_budget=alloc, verbose=False)
            obj = compute_obj(a_last)
            print(f"  Final LP2: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_last.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
