import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_max_and_weights(conv, beta):
    """Compute softmax-smoothed max and the softmax weights."""
    m = np.max(conv)
    exp_conv = np.exp((conv - m) * beta)
    sum_exp = np.sum(exp_conv)
    weights = exp_conv / sum_exp
    smooth_val = m + np.log(sum_exp) / beta
    return smooth_val, weights

def compute_gradient_smoothed(a, weights):
    """Compute gradient of sum_k w_k * conv(a,a)_k with respect to a."""
    # grad_j = 2 * sum_l a_l * w_{j+l}
    # This is 2 * correlate(weights, a, mode='valid')
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_smoothed_optimize(a, time_budget=600.0, beta_start=2.0, beta_end=50.0, 
                           lr=0.01, verbose=True):
    """
    Adam optimizer on the simplex, minimizing smoothed max of convolution.
    Beta is annealed from beta_start to beta_end.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    total_iters = 10000  # for beta annealing schedule
    
    while time.monotonic() - start < time_budget:
        # Anneal beta
        progress = min(1.0, t / total_iters)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute convolution and smooth max weights
        conv = compute_conv(a)
        smooth_val, weights = smooth_max_and_weights(conv, beta)
        
        # Compute gradient
        grad = compute_gradient_smoothed(a, weights)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            continue
        
        # Evaluate actual objective
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose and t % 500 == 0:
                elapsed = time.monotonic() - start
                print(f"    Adam it{t}: obj={best_obj:.8f}, beta={beta:.1f}, elapsed={elapsed:.1f}s")
        else:
            a = a_new.copy()  # still take the step, just don't update best
            # Occasionally reset if stuck
            if t % 2000 == 0:
                # Check if we've been stuck
                if compute_obj(a) > best_obj * 1.01:
                    a = best_a.copy()
    
    return best_a

"""
Hybrid optimization combining smoothed-gradient Adam, LP alternating, and multi-n exploration
to minimize 2*n*max(conv(a,a))/(sum(a))^2. Key innovations: softmax-smoothed max with beta
annealing, Adam optimizer with per-coordinate adaptive steps, and efficient FFT-based gradients.

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_max_weights(conv, beta):
    """Compute softmax weights over convolution values."""
    m = np.max(conv)
    shifted = (conv - m) * beta
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    return weights

def compute_grad_smoothed(a, weights):
    """
    Gradient of sum_k w_k * conv(a,a)_k w.r.t. a.
    grad_j = 2 * sum_l a_l * w_{j+l} = 2 * correlate(weights, a, mode='valid')
    """
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_smoothed_optimize(a, time_budget=500.0, beta_start=1.0, beta_end=100.0,
                           lr=0.005, verbose=True):
    """
    Adam optimizer on simplex minimizing smoothed max(conv(a,a)).
    Beta annealed from beta_start to beta_end over the run.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8

    stall_count = 0
    patience = 3000

    while time.monotonic() - start < time_budget:
        progress = min(1.0, t / 8000.0)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.7)

        conv = compute_conv(a)
        weights = smooth_max_weights(conv, beta)
        grad = compute_grad_smoothed(a, weights)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall_count = 0
            if verbose and t % 1000 == 0:
                elapsed = time.monotonic() - start
                print(f"    Adam it{t}: obj={best_obj:.8f}, beta={beta:.1f}, elapsed={elapsed:.1f}s")
        else:
            a = a_new.copy()
            stall_count += 1
            if stall_count > patience:
                # Perturb to escape local minimum
                rng = np.random.default_rng(t * 7919 + n)
                scale = min(0.4, 0.05 + 0.03 * (stall_count // patience))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall_count = 0
                patience = int(patience * 0.8)
                lr = max(0.001, lr * 0.7)

    return best_a

def multiplicative_descent(a, time_budget=200.0, verbose=False):
    """
    Multiplicative (exponential) updates: a_i *= exp(-lr * g_i), then renormalize.
    More stable than additive for non-negative variables.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    K = 5
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Top-K weights
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break
        softmax_w = np.exp((peak_vals - max_val) * 3.0 / max_val)
        softmax_w /= softmax_w.sum()

        weights = np.zeros(2 * n - 1)
        for w, k in zip(softmax_w, top_idx):
            weights[int(k)] = w

        grad = compute_grad_smoothed(a, weights)

        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * grad
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = max(0.1, lr * 0.998)
        else:
            stall += 1
            if stall > 500:
                rng = np.random.default_rng(int(stall) * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(0.1, lr * 0.5)
                stall = 0
            else:
                lr = max(0.05, lr * 0.99)

    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n - j, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def lp_alternating(a, time_budget=300.0, verbose=False):

    n = len(a)

        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in candidate_ts:

            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))

            obj_try = compute_obj(a_try)

def coordinate_descent_refine(a, time_budget=60.0):

        improved = False

        # Find top contributors to the peak
        contribs = []
        for i in range(n):

            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        # Non-contributing positions
        contributing_set = set()
        for _, i, j in contribs:
            contributing_set.add(i)
            contributing_set.add(j)
        non_contrib = [x for x in range(n) if x not in contributing_set]

        if not non_contrib:
            break

        for contrib_val, i, j in contribs[:30]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < 1e-12:
                break
            for target in non_contrib[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if delta < 1e-15:

                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
        if not improved:
            # Try symmetric move: reduce both i and j, add to target
            for contrib_val, i, j in contribs[:10]:
                if time.monotonic() - start > time_budget:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""

        a = rng.uniform(0, 1, n)

    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]

            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))

        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)

    elif mode == 4:

        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.05
        a = a / a.sum()
    elif mode == 6:
        # Multi-modal
        n_peaks = rng.integers(2, 5)
        centers = rng.integers(0, n, size=n_peaks)
        a = np.zeros(n)
        for c in centers:

            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    print(f"\n=== Phase 1: Accelerated Smoothed Gradient (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 50:
        a_opt = accelerated_smooth_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 result: obj={obj_opt:.8f} (no improvement)")

    # === Phase 2: Multiplicative Descent ===
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 30:
        print(f"\n=== Phase 2: Multiplicative Descent ===")
        a_opt = multiplicative_descent(best_seq, time_budget=p2_budget, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 2 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 result: obj={obj_opt:.8f}")

    # === Phase 3: LP Alternating ===
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.3, 400.0)
    if p3_budget > 30:
        print(f"\n=== Phase 3: LP Alternating ===")
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            alloc = min(p3_budget / (3 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a_opt = lp_alternating(best_seq, time_budget=alloc, verbose=True)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_opt:.8f}")

    # === Phase 4: Multi-n Exploration ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_current = len(best_seq)
    n_options = sorted(set([
        max(200, n_current - 100), n_current,
        n_current + 100, n_current + 200,
        600, 700, 800, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization
        alloc = min(remaining * 0.15, 60.0)
        half = alloc / 2
        if half > 15:
            a_opt = accelerated_smooth_optimize(a_try, time_budget=half, verbose=False)
            obj_opt = compute_obj(a_opt)
            if obj_opt < compute_obj(a_try):
                a_try = a_opt

        remaining = deadline - time.monotonic()
        if remaining > 20:
            lp_alloc = min(remaining * 0.1, 40.0)
            if lp_alloc > 10:
                a_lp = lp_alternating(a_try, time_budget=lp_alloc, verbose=False)
                obj_lp = compute_obj(a_lp)
                if obj_lp < compute_obj(a_try):
                    a_try = a_lp

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Diverse Restarts ===")
    n_best = len(best_seq)
    n_restarts = 4

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 100.0)
        a_opt = accelerated_smooth_optimize(a_r, time_budget=alloc * 0.6, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < compute_obj(a_r):
            a_r = a_opt

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_lp = lp_alternating(a_r, time_budget=min(remaining * 0.15, 40.0), verbose=False)
            obj_lp = compute_obj(a_lp)
            if obj_lp < compute_obj(a_r):
                a_r = a_lp

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final Polish ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Final Polish ===")

    if remaining > 30:
        md_alloc = min(remaining * 0.25, 100.0)
        a_md = accelerated_smooth_optimize(best_seq, time_budget=md_alloc, verbose=False)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Polish MD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        lp_alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Polish LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_alloc = min(remaining * 0.3, 40.0)
        a_cd = coordinate_descent_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Polish CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
