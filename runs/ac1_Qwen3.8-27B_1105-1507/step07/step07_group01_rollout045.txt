import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_weights(conv, T):
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    return exp_vals / total

def gradient_from_weights(weights, a):
    return np.correlate(weights, a, mode='valid')

def mirror_descent(a, time_budget=60.0, eta_init=0.1):
    """Mirror descent with negative entropy (multiplicative updates)."""
    n = len(a)
    a = np.maximum(a, 1e-10).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = np.max(conv)
        
        # Use smooth max for gradient
        T = max(1e-5, max_c * 0.01)
        weights = smooth_max_weights(conv, T)
        grad = gradient_from_weights(weights, a)
        
        # Center the gradient (to maintain sum)
        grad_centered = grad - np.mean(grad)

"""
Multi-strategy optimizer combining mirror descent (multiplicative updates),
LP alternating, chase-peak, symmetric constraint exploration, and structured
perturbations to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key new ideas: Boltzmann-style multiplicative updates, symmetry exploitation,
and frequency-targeted perturbations to escape local minima.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_weights(conv, T):
    """Compute softmax weights for smooth max approximation."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / total


def gradient_from_weights(weights, a):
    """Gradient of smooth max w.r.t. a using correlation."""
    return np.correlate(weights, a, mode='valid')


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        i_start = max(0, 0)
        i_end = min(2 * n - 1, j + n)
        for m in range(i_start, i_end):
            i = m - j
            if 0 <= i < n:
                A[m, j] = a[i]
    return A

    """
    Mirror descent with negative entropy (multiplicative/Boltzmann updates).
    a_new = a * exp(-beta * centered_grad), renormalized.
    This is fundamentally different from additive gradient descent.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = beta_init
    stall = 0
    T = max(1e-4, float(np.max(compute_conv(a))) * 0.05)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        T = max(1e-5, max_c * 0.02)
        weights = smooth_max_weights(conv, T)
        grad = gradient_from_weights(weights, a)
        grad_centered = grad - np.mean(grad)

        # Multiplicative update
        log_update = -beta * grad_centered
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)

            a = a / np.sum(a)
            beta *= 0.5
            stall += 10
            continue
        a_new = a_new / s_new

            best_a = a_new.copy()

def structured_perturb(a, rng, n_pert=None):
    """Add structured sinusoidal perturbation to escape local minima."""
    n = len(a)
    if n_pert is not None:
        n = n_pert
    x = np.arange(n)
    n_freqs = rng.integers(1, 4)
    perturbation = np.zeros(n)
    for _ in range(n_freqs):
        freq = rng.uniform(0.3, 8.0)
        phase = rng.uniform(0, 2 * np.pi)
        amplitude = rng.uniform(0.05, 0.25)
        perturbation += amplitude * np.sin(2 * np.pi * freq * x / n + phase)

    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def random_init(n, rng, mode=0):
    """Generate random non-negative sequence."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.5
        a = a / np.sum(a)
    elif mode == 3:
        x = np.arange(n) / (n - 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)

    else:
        # Log-normal
        a = np.exp(rng.normal(0, 0.5, n))

            best_seq = a_opt.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 7: Random Restarts ===
    print(f"\n=== Phase 7: Random Restarts ===")
    remaining = deadline - time.monotonic()
    for r in range(8):
        if remaining < 20:
            break
        remaining = deadline - time.monotonic()
        init_mode = r % 5
        n_r = n_best + int(rng.integers(-50, 51))
        n_r = max(500, min(1300, n_r))
        a_r = random_init(n_r, rng, mode=init_mode)

        a_opt = mirror_descent(a_r, time_budget=min(20, remaining * 0.1))
        obj_opt = compute_obj(a_opt)

        remaining = deadline - time.monotonic()
        if remaining > 10 and obj_opt < best_score + 0.02:
            a_opt = lp_alternating_v2(a_opt, time_budget=min(15, remaining * 0.2))
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_opt:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()

    polish_methods = [('lp', lp_alternating_v2, 0.25),
                      ('mirror', mirror_descent, 0.2),
                      ('chase', chase_peak_momentum, 0.15),
                      ('lp2', lp_alternating_v2, 0.2),
                      ('mirror2', mirror_descent, 0.15)]

    for method_name, method_fn, frac in polish_methods:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * frac, 50.0)
        if method_name.startswith('lp'):
            a_pol = method_fn(a_final, time_budget=alloc)
        elif method_name.startswith('mirror'):
            a_pol = method_fn(a_final, time_budget=alloc, beta_init=0.5)
        else:
            a_pol = method_fn(a_final, time_budget=alloc)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol.copy()
            print(f"  Polish {method_name}: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
