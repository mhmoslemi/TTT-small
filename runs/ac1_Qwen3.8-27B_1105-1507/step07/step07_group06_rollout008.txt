So the "active set" of g* is {j : (A_a^T y*)[j] = 1}.

This gives us a way to check if the current a is close to optimal: if the optimal g* (from the LP) is close to a, then a is close to a fixed point of the LP alternating, which suggests it's close to optimal.

**New idea: Use the dual to construct a certificate of optimality.**

If we can find y >= 0 such that A_a^T @ y <= 1 and sum(y) = M, and also (a*a)[k] <= M for all k where y[k] > 0, then we have a certificate that the objective is at most 2n*M.

This is useful for verifying solutions.

**Let me now code up the solution.**

My strategy:
1. Load parent sequence
2. Generate diverse initializations (U-shaped, W-shaped, frequency-domain)
3. For each, run FISTA-like subgradient optimization
4. Interleave with LP polishing (using Bregman updates)
5. Multi-scale: try different n values
6. Return the best

Let me write the code:

"""
Hybrid optimization: FISTA subgradient on simplex + LP alternating with Bregman updates
+ frequency-domain initialization + multi-scale exploration.
Focus on U-shaped sequences (large at edges, small in middle) to flatten autocorrelation.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize, linprog
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_of_max_conv(a, k_star):
    """Compute subgradient of (a*a)[k_star] w.r.t. a.
    Result is a vector of length n where result[j] = 2*a[k_star - j] if 0 <= k_star-j < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    return grad


def fista_simplex(a_init, time_budget=60.0, max_iter=5000):
    """
    FISTA-like accelerated subgradient method on the simplex.
    Minimizes f(a) = max_k (a*a)[k] subject to a in simplex.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = a.copy()  # FISTA momentum variable
    t = 1.0
    step_size = 1.0 / (4.0 * n)  # Initial step size (conservative)
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        grad = subgradient_of_max_conv(y, k_star)
        
        # Project gradient onto tangent space (sum=0)
        grad_proj = grad - np.mean(grad)
        
        # Gradient step
        z = y - step_size * grad_proj
        
        # Project onto simplex
        x = project_simplex(z)
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - a)
        t = t_new
        a = x
        
        # Evaluate
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive step size
        if it % 100 == 0 and it > 0:
            # Decrease step size if not improving
            step_size *= 0.95
        step_size = min(step_size, 1.0 / (2.0 * np.sqrt(n)))
    
    return best_a


def lp_solve_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build convolution matrix A (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    # Use linprog with HiGHS
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0  # (A@g)[k] - M <= 0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0  # sum(g) = 1
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 120})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    # Fallback to cvxpy
    try:
        solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
        for solver in solvers:
            try:
                g = cp.Variable(n, nonneg=True)
                M = cp.Variable()
                constraints = [A @ g <= M, cp.sum(g) == 1]
                prob = cp.Problem(cp.Minimize(M), constraints)
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass
    
    return None, None


def lp_bregman_alternate(a, time_budget=120.0):
    """
    LP alternating with Bregman (multiplicative/exponential) updates.
    Instead of linear interpolation a -> (1-t)*a + t*g, use:
    a_new[j] = a[j] * g[j]^(t/a[j]) (geometric interpolation)
    or simply a_new = normalize(a^alpha * g^(1-alpha))
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy()
    a = a / float(np.sum(a))
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    ema_obj = best_obj
    patience = 20
    no_improve_count = 0
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = lp_solve_minpeak(a)
        if g_lp is None:
            break
        
        # Try multiple mixing strategies
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        
        alpha_values = [0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alpha_values:
            # Geometric (Bregman) interpolation
            a_try = np.power(np.maximum(a, 1e-15), alpha) * np.power(np.maximum(g_lp, 1e-15), 1.0 - alpha)
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s > 1e-10:
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_a = a_try.copy()
        
        # Also try linear

            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s > 1e-10:
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            ema_obj = 0.9 * ema_obj + 0.1 * best_obj
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= patience:

                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                no_improve_count = 0
    
    return best_a


def freq_domain_init(n, c=1.0, p=1.0):
    """
    Generate U-shaped initialization using cosine.
    a[j] = (1 + c * cos(pi * j / (n-1)))^p
    For c=1, p=1: U-shaped, zero at center
    For c=0.5: mild U-shape
    """
    j = np.arange(n)
    if n > 1:
        a = (1.0 + c * np.cos(np.pi * j / (n - 1))) ** p
    else:
        a = np.array([1.0])
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def edge_conc_init(n, power=1.0):
    """
    Edge-concentrated initialization.
    a[j] proportional to (distance from center)^power
    """
    j = np.arange(n)
    center = (n - 1) / 2.0
    dist = np.abs(j - center) / center  # normalized distance, 0 to 1
    a = (dist + 0.01) ** power  # +0.01 to avoid exact zero
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def w_shape_init(n, c1=1.0, c2=0.5):
    """
    W-shaped initialization: combination of two U-shapes.
    """
    j = np.arange(n)
    a1 = 1.0 + c1 * np.cos(np.pi * j / (n - 1))
    a2 = 1.0 + c2 * np.cos(2.0 * np.pi * j / (n - 1))
    a = np.maximum(a1, 0) * np.maximum(a2, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def random_structured_init(n, rng):
    """Random structured initialization."""
    n_harmonics = rng.integers(1, 4)
    a = np.ones(n)
    for _ in range(n_harmonics):
        k = rng.integers(1, n // 2 + 1)
        amp = rng.uniform(-0.5, 0.8)
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    a = np.maximum(a, 0.001)
    a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_polish(a, time_budget=30.0):
    """
    Coordinate descent: try moving mass from peak contributors to reduce the max bin.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345 + n)
    
    iter_count = 0
    while time.monotonic() - start < time_budget:
        iter_count += 1
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find the j that contributes most to bin k_star
        # Contribution of a[j] to (a*a)[k_star] is a[j]*a[k_star-j]
        contribs = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contribs[j] = a[j] * a[idx]
        
        # Move mass from high-contributing j to low-contributing j'
        # that doesn't hurt k_star as much
        top_j = int(np.argmax(contribs))
        
        # Find a good target: a j' where a[k_star - j'] is small
        target_score = np.zeros(n)
        for j2 in range(n):
            idx = k_star - j2
            if 0 <= idx < n:
                target_score[j2] = -a[idx]  # prefer targets with small a[k_star-j']
            else:
                target_score[j2] = -1.0  # even better: out of range
        
        # Exclude top_j itself
        target_score[top_j] = -np.inf
        target_j = int(np.argmax(target_score))
        
        # Try moving some mass
        for frac in [0.01, 0.02, 0.05, 0.1]:
            delta = frac * a[top_j]
            if delta >= a[top_j]:
                continue
            a[top_j] -= delta
            a[target_j] += delta
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
                obj = compute_obj(a)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                else:
                    a[top_j] += delta
                    a[target_j] -= delta
                    a = np.maximum(a, 0)
                    a = a / float(np.sum(a))
                    break
            else:
                a[top_j] += delta
                a[target_j] -= delta
                break
        
        # Also try random moves
        if iter_count % 5 == 0:
            for _ in range(20):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * a[i] * rng.uniform(0.5, 2.0)
                if delta > a[i] * 0.5:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                    else:
                        a[i] += delta
                        a[j] -= delta
                        a = np.maximum(a, 0)
                        a = a / float(np.sum(a))
            
            if time.monotonic() - start > time_budget:
                break
    
    return best_a
