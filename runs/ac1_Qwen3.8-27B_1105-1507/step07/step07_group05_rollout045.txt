        A_ub = np.zeros((len(idx), n + 1))
        A_ub[:, :n] = A_sel
        A_ub[:, n] = -1.0
        b_ub = np.zeros(len(idx))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def flat_spectrum_init(n, rng, n_iter=200):
    """Gerchberg-Saxton style flat spectrum initialization."""
    a = rng.uniform(0.5, 1.5, n)
    a = a / np.sum(a)
    for _ in range(n_iter):
        A = fft(a)
        target_mag = np.sqrt(np.sum(a ** 2))
        mag = np.abs(A)
        phase = np.angle(A)
        # Set target: flat magnitude except DC
        A_target = np.zeros_like(A, dtype=complex)
        A_target[0] = A[0]  # preserve DC (sum = 1)
        for k in range(1, n):
            A_target[k] = target_mag * np.exp(1j * phase[k])
        a_new = np.real(ifft(A_target))
        a = np.maximum(a_new, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
            break
        a = a / s
    return a


def smoothed_max_gd(a, time_budget=50.0, T_init=50.0, T_min=5.0):
    """Gradient descent on smoothed max of conv(a_new, a) with adaptive temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)
    T = T_init
    stall = 0
    a_flip = a[::-1]

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Compute b = conv(a, a)
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Smoothed max gradient: w[k] = exp(T * b[k]) / sum(exp(T * b))
        b_max = np.max(b)
        w = np.exp(T * (b - b_max))
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient of weighted sum: c[i] = sum_k w[k] * a[k-i] (where valid)
        # This is conv(a_flip, w)[n-1 : 2n-1]
        fw = rfft(w, N)
        faf = rfft(a_flip, N)
        c_full = irfft(faf * fw, N)[:3 * n - 2]
        c = c_full[n - 1:2 * n - 1]

        c = c - np.mean(c)

        # Nesterov momentum
        v = momentum * v - eta * c
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            a_flip = a[::-1]
            stall = 0
            eta = min(eta * 1.003, 3.0)
            T = max(T * 0.995, T_min)
        else:
            stall += 1
            eta *= 0.98
            v *= 0.5
            if stall > 100:
                # Perturb and reset
                rng_local = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    a_flip = a[::-1]
                v = np.zeros(n)
                eta = 0.5
                T = T_init
                stall = 0
            if eta < 1e-8:
                break

    return best_a


def lp_alternating(a, time_budget=60.0, use_topk=False, top_k=400):
    """LP alternating with golden-section line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.5

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        if use_topk and n > 500:
            g_lp = solve_lp_topk(a, top_k)
        else:
            g_lp = solve_lp(a)

        if g_lp is None:
            break

        # Golden section line search for t
        phi = (np.sqrt(5) - 1) / 2  # ~0.618
        t_lo = 0.0
        t_hi = 1.0
        t1 = t_hi - phi * (t_hi - t_lo)
        t2 = t_lo + phi * (t_hi - t_lo)

        def eval_t(t):
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        f1 = eval_t(t1)
        f2 = eval_t(t2)

        for _ in range(15):
            if f1 < f2:
                t_hi = t2
                t2 = t1
                f2 = f1
                t1 = t_hi - phi * (t_hi - t_lo)
                f1 = eval_t(t1)
            else:
                t_lo = t1
                t1 = t2
                f1 = f2
                t2 = t_lo + phi * (t_hi - t_lo)
                f2 = eval_t(t2)

        t_opt = (t_lo + t_hi) / 2.0
        best_t = t_opt

        a_try = (1 - t_opt) * a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a


def coordinate_descent(a, time_budget=20.0):
    """Simple coordinate descent: try adjusting each coordinate in turn."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    b = compute_conv(a)

    for it in range(n * 5):
        if time.monotonic() - start > time_budget:
            break
        i = it % n
        best_local = best_obj
        best_val = a[i]
        for factor in [0.8, 0.9, 1.1, 1.2, 0.5, 1.5]:
            trial = best_a.copy()
            trial[i] = best_val * factor
            trial = np.maximum(trial, 0)
            s_tr = float(np.sum(trial))
            if s_tr < 1e-10:
                continue
            trial = trial / s_tr
            obj_tr = compute_obj(trial)
            if obj_tr < best_local - 1e-14:
                best_local = obj_tr
                best_val = best_a[i] * factor
        if best_val != a[i]:
            a[i] = best_val
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            if best_local < best_obj:
                best_obj = best_local
                best_a = a.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 7)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 10) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.2
        a = a / np.sum(a)
    elif mode == 5:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x ** 0.5
        a = a / np.sum(a)
    else:
        a = flat_spectrum_init(n, rng, n_iter=100)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: flat-spectrum init + smoothed-max GD + LP alternating with
    golden-section line search + coordinate descent + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = flat_spectrum_init(800, rng, n_iter=200)
        best_score = compute_obj(best_seq)
        print(f"Flat-spectrum init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # Phase 1: Smoothing GD to polish initial point
    print(f"\n=== Phase 1: Smoothed-Max GD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.2, 150.0)
    a_gd = smoothed_max_gd(best_seq, time_budget=alloc)
    obj_gd = compute_obj(a_gd)
    if obj_gd < best_score:
        best_score = obj_gd
        best_seq = a_gd.copy()
        print(f"  ** Smoothed GD: obj={best_score:.8f} **")
    else:
        print(f"  Smoothed GD: obj={obj_gd:.8f}")

    # Phase 2: Main LP alternating loop (bulk of time)
    print(f"\n=== Phase 2: LP Alternating (main) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.45, 600.0)
    a_lp = lp_alternating(best_seq, time_budget=alloc, use_topk=False)
    obj_lp = compute_obj(a_lp)
    if obj_lp < best_score:
        best_score = obj_lp
        best_seq = a_lp.copy()
        print(f"  ** LP: obj={best_score:.8f} **")
    else:
        print(f"  LP: obj={obj_lp:.8f}")

    # Phase 2b: Another round of smoothed GD from LP result
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 80.0)
        a_gd2 = smoothed_max_gd(best_seq, time_budget=alloc, T_init=20.0, T_min=3.0)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            print(f"  ** GD2: obj={best_score:.8f} **")

    # Phase 2c: LP again
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 120.0)
        a_lp2 = lp_alternating(best_seq, time_budget=alloc, use_topk=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** LP2: obj={best_score:.8f} **")

    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.05, 30.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # Phase 4: Multi-n exploration
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr - 100, n_curr + 100, n_curr + 200,
        n_curr + 400, n_curr - 200,
        1000, 1200, 1500, 600
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_curr]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Quick GP + LP
        alloc_gd = min(remaining * 0.1, 20.0)
        alloc_lp = min(remaining * 0.15, 40.0)
        a_opt = smoothed_max_gd(a_try, time_budget=alloc_gd, T_init=30.0, T_min=5.0)
        a_opt = lp_alternating(a_opt, time_budget=alloc_lp, use_topk=(n_try > 1000), top_k=300)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 5: Annealing restarts
    print(f"\n=== Phase 5: Annealing Restarts ===")
    n_curr = len(best_seq)
    n_restarts = max(2, int((deadline - time.monotonic()) / 50.0))
    n_restarts = min(n_restarts, 6)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Perturb the best solution
        noise_scale = 0.05 + 0.05 * r
        a_pert = best_seq * np.exp(rng.normal(0, noise_scale, n_curr))
        a_pert = np.maximum(a_pert, 0)
        s_p = float(np.sum(a_pert))
        if s_p > 1e-10:
            a_pert = a_pert / s_p

        alloc_gd = min(remaining * 0.15, 25.0)
        alloc_lp = min(remaining * 0.2, 30.0)
        a_opt = smoothed_max_gd(a_pert, time_budget=alloc_gd, T_init=20.0, T_min=3.0)
        a_opt = lp_alternating(a_opt, time_budget=alloc_lp, use_topk=(n_curr > 1000), top_k=300)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r} (scale={noise_scale:.2f}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (scale={noise_scale:.2f}): obj={obj_r:.8f}")

    # Phase 6: Final LP polish
    print(f"\n=== Phase 6: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 100.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, use_topk=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

        # One more CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.3, 15.0))
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  ** Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
