```python
def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_obj_and_grad(x, tau):

    x = x - np.max(x)
```

```python
"""
Optimized low-autocorrelation non-negative sequence search.
Combines spectral amplitude flattening (IFAA), adaptive-temperature L-BFGS-B on log params,
simplex-projected subgradient descent, and systematic multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(x, tau):
    """
    Smoothed objective and gradient for a = exp(x).
    Uses log-sum-exp smoothing of the max convolution.
    """
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))

    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)

    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]

    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(1.0 - u * np.arange(1, n + 1) > cssv)[0]
    if len(rho) == 0:
        rho = np.array([0])
    rho_val = rho[0]
    theta = cssv[rho_val] / (rho_val + 1.0)
    x = np.maximum(v - theta, 0.0)
    sx = np.sum(x)
    if sx < 1e-15:
        return np.full(n, target_sum / n)
    return x * (target_sum / sx)


def spectral_shaping(a, alpha=0.4, n_iter=200):
    """
    Iterative Fourier amplitude adjustment to flatten the autocorrelation spectrum.
    In the Fourier domain, autocorrelation ~ |A(f)|^2, so flattening |A| flattens autocorrelation.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    size = next_pow2(2 * n - 1)

    for _ in range(n_iter):
        A = fft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        # Compress dynamic range: lower alpha = flatter
        mag_new = mag ** alpha
        A_new = mag_new * phase
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a = a_new / s

        c_max = np.max(conv)

        # Soft-max weights
        w = np.exp((conv - c_max) * beta)
        w = w / np.sum(w)

        # Compute weighted cross-correlation (subgradient)
        size = next_pow2(2 * n - 1)
        a_flip = a[::-1]
        fa_flip = rfft(a_flip, size)
        fw = rfft(w, size)
        corr = irfft(fw * fa_flip, size)[n - 1:n - 1 + n]

        # Gradient of smoothed objective w.r.t. a (with sum=1 constraint via Lagrange)
        lagrange = 2.0 * float(np.dot(w, conv))  # derivative of sum constraint
        grad = 2.0 * corr - lagrange * a

        # Step
        a_new = a - step * grad
        a_new = simplex_project(a_new, 1.0)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            step = min(step * 1.01, 5.0 / n)
        else:
            stall += 1
            step *= 0.9
            if stall > 200:
                # Perturb and reset
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = a / np.sum(a)
                step = 2.0 / n
                stall = 0

        # Slowly increase beta (sharper subgradient)
        beta = min(beta * 1.001, 100.0)

    return best_a


def spectral_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Combined spectral shaping + gradient polishing.
    Tries multiple alpha values for spectral flattening.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    alphas = [0.5, 0.3, 0.2, 0.7, 0.4, 0.15]

    for alpha in alphas:
        if time.monotonic() - start > time_budget:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Spectral shaping
        a_spec = spectral_shaping(a, alpha, n_iter=300)
        obj_spec = compute_obj(a_spec)

        if obj_spec < best_obj:
            best_obj = obj_spec
            best_a = a_spec.copy()
            if verbose:
                print(f"      spectral a={alpha:.2f}: obj={best_obj:.10f}")

        # Polish with gradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_pol = gradient_optimize(a_spec, time_budget=min(remaining * 0.3, 20.0))
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_obj:
                best_obj = obj_pol
                best_a = a_pol.copy()
                if verbose:
                    print(f"      spectral+grad a={alpha:.2f}: obj={best_obj:.10f}")

        # Also try from original a with different alpha
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and alpha in [0.3, 0.5]:
            a_from_orig = spectral_shaping(a_init, alpha, n_iter=300)
            obj_so = compute_obj(a_from_orig)
            if obj_so < best_obj:
                best_obj = obj_so
                best_a = a_from_orig.copy()
                if verbose:
                    print(f"      spectral(from init) a={alpha:.2f}: obj={best_obj:.10f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 7)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.2
        a = a / np.sum(a)
    elif mode == 4:
        # Exponential edges
        d = np.arange(n) - n / 2
        a = np.exp(-0.5 * (d / (n * 0.3)) ** 2) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Smooth random
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = a + 0.01
        a = a / np.sum(a)
    else:
        # Spectral shaped random
        a = rng.uniform(0.1, 1.0, n)
        a = spectral_shaping(a, 0.4, n_iter=50)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining spectral shaping,
    adaptive L-BFGS-B, subgradient descent, and systematic n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)

    # === Phase 1: Gradient optimization on parent ===
    print(f"\n=== Phase 1: Gradient Optimization (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a1 = gradient_optimize(best_seq, time_budget=min(remaining * 0.18, 90.0), verbose=True)
        obj = compute_obj(a1)
        if obj < best_score:
            best_score = obj
            best_seq = a1.copy()
            print(f"  After grad1: obj={best_score:.10f}")

    # === Phase 2: Subgradient descent ===
    print(f"\n=== Phase 2: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a2 = subgradient_descent(best_seq, time_budget=min(remaining * 0.08, 30.0), verbose=True)
        obj = compute_obj(a2)
        if obj < best_score:
            best_score = obj
            best_seq = a2.copy()
            print(f"  After subgrad: obj={best_score:.10f}")

    # === Phase 3: Spectral shaping ===
    print(f"\n=== Phase 3: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a3 = spectral_optimize(best_seq, time_budget=min(remaining * 0.15, 60.0), verbose=True)
        obj = compute_obj(a3)
        if obj < best_score:
            best_score = obj
            best_seq = a3.copy()
            print(f"  After spectral: obj={best_score:.10f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_parent = len(best_seq)
    n_candidates = sorted(set([
        n_parent - 200, n_parent - 100, n_parent - 50, n_parent - 25,
        n_parent, n_parent + 25, n_parent + 50, n_parent + 100, n_parent + 200,
        n_parent + 400,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick spectral shaping
        a_spec = spectral_shaping(a_try, 0.35, n_iter=100)
        obj_spec = compute_obj(a_spec)

        # Gradient polish
        alloc = min(remaining * 0.1, 25.0)
        a_opt = gradient_optimize(a_spec, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 5: Intensify on best ===
    print(f"\n=== Phase 5: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        # More gradient
        a5 = gradient_optimize(best_seq, time_budget=min(remaining * 0.2, 80.0), verbose=True)
        obj = compute_obj(a5)
        if obj < best_score:
            best_score = obj
            best_seq = a5.copy()
            print(f"  Intensify grad: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Subgradient
            a5b = subgradient_descent(best_seq, time_budget=min(remaining * 0.15, 40.0))
            obj = compute_obj(a5b)
            if obj < best_score:
                best_score = obj
                best_seq = a5b.copy()
                print(f"  Intensify subgrad: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            # More spectral
            a5c = spectral_optimize(best_seq, time_budget=min(remaining * 0.15, 30.0))
            obj = compute_obj(a5c)
            if obj < best_score:
                best_score = obj
                best_seq = a5c.copy()
                print(f"  Intensify spectral: obj={best_score:.10f}")

    # === Phase 6: Random restarts ===
    print(f"\n=== Phase 6: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    n_restarts = max(3, int(remaining / 35))
    n_restarts = min(n_restarts, 10)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if r == 0:
            # Small perturbation of best
            a_r = best_seq * np.exp(rng.normal(0, 0.02, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            # Spectral-shaped random
            a_r = random_init(n_best, rng)
            a_r = spectral_shaping(a_r, 0.4, n_iter=100)
            a_r = a_r / np.sum(a_r)
        elif r == 2:
            # Different n
            n_r = n_best + int(rng.integers(-150, 150))
            n_r = max(400, min(1800, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.04, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 3:
            # Cosine window variant
            x = np.linspace(0, np.pi, n_best)
            a_r = (0.3 + 0.7 * np.cos(x) ** 2) + 0.05 * rng.normal(0, 1, n_best)
            a_r = np.maximum(a_r, 0.01)
            a_r = spectral_shaping(a_r, 0.5, n_iter=50)
            a_r = a_r / np.sum(a_r)
        elif r == 4:
            # Edge-enhanced
            d = np.abs(np.arange(n_best) - n_best / 2) / (n_best / 2)
            a_r = 1.0 - 0.5 * d ** 2 + 0.05 * rng.normal(0, 1, n_best)
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / np.sum(a_r)
        else:
            a_r = random_init(n_best, rng)

        # Combined optimization: spectral + gradient
        alloc = min(remaining * 0.12, 25.0)
        a_opt = spectral_shaping(a_r, 0.4, n_iter=100)
        a_opt = gradient_optimize(a_opt, time_budget=alloc)
        obj = compute_obj(a_opt)

        remaining = deadline - time.monotonic()
        if remaining > 10 and obj < best_score + 0.005:
            a_opt = gradient_optimize(a_opt, time_budget=min(remaining * 0.2, 15.0))
            obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_f = gradient_optimize(best_seq, time_budget=min(remaining * 0.4, 45.0), verbose=True)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final grad: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_f2 = subgradient_descent(best_seq, time_budget=min(remaining * 0.4, 25.0))
            obj = compute_obj(a_f2)
            if obj < best_score:
                best_score = obj
                best_seq = a_f2.copy()
                print(f"  Final subgrad: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_f3 = spectral_shaping(best_seq, 0.6, n_iter=200)
            a_f3 = gradient_optimize(a_f3, time_budget=min(remaining * 0.3, 15.0))
            obj = compute_obj(a_f3)
            if obj < best_score:
                best_score = obj
                best_seq = a_f3.copy()
                print(f"  Final spectral: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
```
