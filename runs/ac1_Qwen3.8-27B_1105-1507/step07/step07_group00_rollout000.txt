                lr = max(1e-5, lr * 0.97)
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")
        
        it += 1
    
    return best_a

def coordinate_descent_polish(a, time_budget=30.0):
    """Paired coordinate descent: move mass from peak contributors to others."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False
        
        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)
        
        non_contrib = set(range(n))
        for _, i, j in contributors:
            non_contrib.discard(i)
            non_contrib.discard(j)
        nc_list = list(non_contrib)
        
        if not nc_list:
            break
        
        for contrib, i, j in contributors[:30]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break
            for target in nc_list[:10]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            break
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
