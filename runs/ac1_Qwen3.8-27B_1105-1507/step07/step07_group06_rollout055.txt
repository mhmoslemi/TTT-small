def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def softmax_from_theta(theta):
    """Compute softmax with numerical stability."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def smoothed_obj_and_grad(theta, tau, n):
    """
    Compute smoothed objective and gradient w.r.t. log-parameters theta.
    a = softmax(theta), obj = 2n * smooth_max(conv(a), tau)
    Uses top-K smoothing for better conditioning.
    """
    a = softmax_from_theta(theta)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    
    # Log-sum-exp smoothing
    scaled = (conv - conv_max) * tau
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    lse = conv_max + np.log(np.sum(np.exp(scaled))) / tau
    obj_val = 2.0 * n * lse
    
    # Soft-argmax weights
    w = tau * np.exp(scaled)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum
    
    # c_j = sum_m w[j+m] * a[m] for valid m
    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < end:
            c[j] = np.dot(w[j:end], a[:end - j])
    
    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta

def smooth_lbfgs(a_init, tau_start, tau_end, time_budget, max_iter=200, topK=None):

    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    n_stages = 6
    taus = np.geomspace(tau_start, tau_end, n_stages)

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        if topK is not None:

                _, g = topk_smoothed_obj_and_grad(th, ts, topK, n)
                return g
        else:
            def obj_func(th):
                val, _ = smoothed_obj_and_grad(th, ts, n)
                return val

                _, g = smoothed_obj_and_grad(th, ts, n)
                return g
        
        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',

        except Exception:

    return best_a, best_obj

def diversify(a, rng, n_augs=5):
    """Generate diverse perturbations of a."""
    results = [a.copy()]
    n = len(a)
    
    for _ in range(n_augs):
        kind = rng.integers(0, 4)
        if kind == 0:
            # Gaussian noise in log-space
            a_new = a * np.exp(rng.normal(0, 0.1, n))
        elif kind == 1:
            # Local perturbation (Gaussian kernel)
            center = rng.integers(0, n)
            sigma = n * rng.uniform(0.05, 0.2)
            x = np.arange(n)
            kernel = np.exp(-0.5 * ((x - center) / sigma) ** 2)
            a_new = a * (1 + 0.2 * (kernel - kernel.mean()))
        elif kind == 2:
            # Shift (circular)
            shift = rng.integers(1, n // 4)
            a_new = np.roll(a, shift)
        else:
            # Scale a random subset
            mask = rng.random(n) < 0.3
            scale = rng.uniform(0.5, 2.0)
            a_new = a.copy()
            a_new[mask] *= scale
        
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
            results.append(a_new)
    
    return results


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

# Global parameters for callback functions (to avoid closures)
_GLOBAL_N = 0
_GLOBAL_TAU = 100.0
_GLOBAL_TOPK = None


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] ** 2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def _smoothed_value_and_weights(theta):
    """Internal helper: compute smooth max value and weights for current global params."""
    n = _GLOBAL_N
    tau = _GLOBAL_TAU
    topk = _GLOBAL_TOPK
    a = _softmax(theta)
    conv = compute_conv(a)
    conv_max = np.max(conv)

    if topk is not None and topk < len(conv):
        k_indices = np.argsort(conv)[-topk:]
        topk_vals = conv[k_indices]
        tk_max = np.max(topk_vals)
        scaled = (topk_vals - tk_max) * tau
        scaled = np.clip(scaled, -50, 50)
        lse = tk_max + np.log(np.sum(np.exp(scaled))) / tau
        w_topk = tau * np.exp(scaled)
        w_topk = w_topk / np.sum(w_topk)
        w = np.zeros(len(conv))
        for idx_in, idx_out in enumerate(k_indices):
            w[idx_out] = w_topk[idx_in]
    else:
        scaled = (conv - conv_max) * tau
        scaled = np.clip(scaled, -50, 50)
        lse = conv_max + np.log(np.sum(np.exp(scaled))) / tau
        w = tau * np.exp(scaled)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

    obj_val = 2.0 * n * lse
    return a, conv, w, obj_val


def _softmax(theta):
    """Numerically stable softmax."""
    s = theta - np.max(theta)
    e = np.exp(s)
    return e / np.sum(e)


def _obj_func(theta):
    """Objective function for scipy minimize (uses global params)."""
    a, conv, w, obj_val = _smoothed_value_and_weights(theta)
    return obj_val


def _grad_func(theta):
    """Gradient function for scipy minimize (uses global params)."""
    a, conv, w, obj_val = _smoothed_value_and_weights(theta)
    n = _GLOBAL_N

    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < end:
            c[j] = np.dot(w[j:end], a[:end - j])

    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return grad_theta


def _obj_grad_combined(theta):
    """Combined obj + grad for efficiency."""
    return _obj_func(theta), _grad_func(theta)


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_peak_shave(a, k_star, reduction):
    """
    LP: min M s.t. conv(a,g)[k_star] <= (1-reduction)*peak,
    conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    current_peak = conv[k_star]
    target = current_peak * (1.0 - reduction)

    A = build_conv_matrix(a)

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        n_rows = 2 * n - 1
        A_ub = np.zeros((n_rows, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(n_rows)

        # Add peak-shaving constraint: A[k_star] @ g <= target
        A_ub2 = np.zeros((n_rows + 1, n + 1))
        A_ub2[:n_rows, :] = A_ub
        A_ub2[n_rows, :n] = A[k_star]
        A_ub2[n_rows, n] = 0.0
        b_ub2 = np.concatenate([b_ub, [target]])

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub2, b_ub=b_ub2, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def solve_lp_multi_shave(a, top_k_bins, reduction):
    """
    LP: min M s.t. conv(a,g)[k] <= (1-reduction)*conv_a[k] for k in top_k_bins,
    conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a = a / np.sum(a)
    conv_a = compute_conv(a)

    A = build_conv_matrix(a)

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        n_rows = 2 * n - 1
        A_ub = np.zeros((n_rows, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(n_rows)

        # Add multi-peak constraints
        extra_A = np.zeros((len(top_k_bins), n + 1))
        extra_b = np.zeros(len(top_k_bins))
        for i, k in enumerate(top_k_bins):
            extra_A[i, :n] = A[k]
            extra_b[i] = conv_a[k] * (1.0 - reduction)

        A_ub2 = np.vstack([A_ub, extra_A])
        b_ub2 = np.concatenate([b_ub, extra_b])

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub2, b_ub=b_ub2, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def smooth_lbfgs(a_init, tau_start, tau_end, time_budget, max_iter=200, topk=None):
    """Smoothed L-BFGS-B with tau annealing using global params."""
    global _GLOBAL_N, _GLOBAL_TAU, _GLOBAL_TOPK

    n = len(a_init)
    _GLOBAL_N = n
    _GLOBAL_TOPK = topk

    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    n_stages = 6
    taus = np.geomspace(tau_start, tau_end, n_stages)

    for ts in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        _GLOBAL_TAU = ts

        try:
            result = minimize(_obj_func, theta, jac=_grad_func, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'gtol': 1e-9, 'maxfun': 300})
            if result.x is not None:
                a_new = _softmax(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def subgradient_descent(a, time_budget=20.0, tau=50.0):
    """
    Projected subgradient descent on simplex with soft-max weighting.
    Uses the true subgradient (soft-max approximation) directly in a-space.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.03
    lr_decay = 0.999

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        conv_max = np.max(conv)
        scaled = (conv - conv_max) * tau
        scaled = np.clip(scaled, -50, 50)
        w = np.exp(scaled)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Gradient in a-space: grad[j] = 2 * sum_k w[k] * a[k-j] (approx)
        grad_a = np.zeros(n)
        for j in range(n):
            total = 0.0
            for k in range(j, min(2 * n - 1, j + n)):
                m = k - j
                total += w[k] * a[m]
            grad_a[j] = 2.0 * total

        # Project to tangent space
        grad_proj = grad_a - np.mean(grad_a)
        grad_norm = np.linalg.norm(grad_proj)
        if grad_norm < 1e-15:
            break

        a_new = a - step * grad_proj / max(grad_norm, 1e-10)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
        step *= lr_decay

    return best_a, best_obj


def lp_alternating(a, time_budget=60.0, shave_mode='none', shave_param=0.05):
    """
    LP alternating with optional peak shaving.
    shave_mode: 'none', 'single', 'multi'
    shave_param: reduction fraction for shaving
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        if shave_mode == 'single':
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
            g_lp = solve_lp_peak_shave(a, k_star, shave_param)
            if g_lp is None:
                g_lp, _ = solve_lp_minpeak(a)
        elif shave_mode == 'multi':
            conv = compute_conv(a)
            top_k = int(np.argsort(conv)[-min(5, len(conv)):])[:3]  # top 3 peaks
            g_lp = solve_lp_multi_shave(a, top_k, shave_param)
            if g_lp is None:
                g_lp, _ = solve_lp_minpeak(a)
        else:
            g_lp, _ = solve_lp_minpeak(a)

        if g_lp is None:
            break

        # Fine line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= 5:
                rng = np.random.default_rng(it * 104729 + n)
                a = best_a * np.exp(rng.normal(0, 0.02 + 0.01 * stall_count, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diversify_population(a, rng, n_diverse=4):
    """Generate diverse perturbations of a."""
    results = [a.copy()]
    n = len(a)

    for i in range(n_diverse):
        kind = i % 5
        if kind == 0:
            a_new = a * np.exp(rng.normal(0, 0.08, n))
        elif kind == 1:
            center = rng.integers(0, n)
            sigma = n * rng.uniform(0.05, 0.15)
            x = np.arange(n)
            kernel = np.exp(-0.5 * ((x - center) / sigma) ** 2)
            a_new = a * (1.0 + 0.3 * (kernel - kernel.mean()))
        elif kind == 2:
            shift = rng.integers(1, max(2, n // 5))
            a_new = np.roll(a, shift)
        elif kind == 3:
            # Split and scale halves differently
            mid = n // 2
            a_new = a.copy()
            a_new[:mid] *= rng.uniform(0.7, 1.4)
            a_new[mid:] *= rng.uniform(0.7, 1.4)
        else:
            # Random smooth perturbation (low-pass filtered noise)
            noise = rng.normal(0, 0.1, n)
            # Simple moving average as low-pass
            k = max(1, n // 20)
            kernel = np.ones(k) / k
            noise = np.convolve(noise, kernel, mode='same')
            a_new = a * np.exp(noise)

        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            results.append(a_new / s)

    return results


def local_optimize(a, time_budget=30.0):
    """Apply a sequence of local optimization methods."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Step 1: Quick smoothed gradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_s, obj_s = smooth_lbfgs(a, tau_start=20, tau_end=300,
                                  time_budget=min(remaining * 0.25, 10), max_iter=60)
        if obj_s < best_obj:
            best_obj = obj_s
            best_a = a_s.copy()
            a = a_s.copy()

    # Step 2: Subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_sg, obj_sg = subgradient_descent(a, time_budget=min(remaining * 0.2, 8), tau=80)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            a = a_sg.copy()

    # Step 3: LP alternating with single peak shaving
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_lp = lp_alternating(a, time_budget=min(remaining * 0.3, 12),
                              shave_mode='single', shave_param=0.03)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp.copy()

    # Step 4: Final smoothed pass
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a_f, obj_f = smooth_lbfgs(a, tau_start=50, tau_end=500,
                                  time_budget=min(remaining * 0.3, 8), max_iter=40)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: population-based with diverse local optimization,
    LP peak-shaving, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: Optimize parent with long LP alternating
    print(f"\n=== Phase 1: Parent LP optimization ===")
    alloc = min(120, (deadline - time.monotonic()) * 0.15)
    a_lp = lp_alternating(best_seq, time_budget=alloc, shave_mode='single', shave_param=0.03)
    obj_lp = compute_obj(a_lp)
    if obj_lp < best_score:
        best_score = obj_lp
        best_seq = a_lp.copy()
        print(f"  Parent LP: obj={best_score:.8f} **")

    # Phase 2: Smoothed L-BFGS-B on parent
    print(f"\n=== Phase 2: Smoothed L-BFGS-B ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 60)
        a_s, obj_s = smooth_lbfgs(best_seq, tau_start=10, tau_end=1000,
                                  time_budget=alloc, max_iter=300)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
            n_parent = len(best_seq)
            print(f"  Smoothed: obj={best_score:.8f} **")

    # Phase 3: Population-based exploration
    print(f"\n=== Phase 3: Population Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Build diverse population
        population = [best_seq.copy()]
        population.extend(diversify_population(best_seq, rng, n_diverse=5))

        # Add structural diverse initializations
        n = n_parent
        structural_inits = [
            np.linspace(1, 3, n) / np.sum(np.linspace(1, 3, n)),
            np.exp(np.linspace(0, 3, n)) / np.sum(np.exp(np.linspace(0, 3, n))),
            (1 + 0.4 * np.sin(2 * np.pi * np.arange(n) / n * 2)) / n,
            np.concatenate([np.linspace(3, 0.5, n // 2), np.linspace(0.5, 2, n - n // 2)]) /
            np.sum(np.concatenate([np.linspace(3, 0.5, n // 2), np.linspace(0.5, 2, n - n // 2)])),
        ]
        for si in structural_inits:
            si = np.maximum(si, 0)
            s = np.sum(si)
            if s > 1e-10:
                population.append(si / s)

        pop_budget = min(200, remaining * 0.3)
        per_member = pop_budget / len(population)

        for i, pop_a in enumerate(population):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(per_member, remaining * 0.2)
            if alloc < 5:
                continue
            a_opt, obj_opt = local_optimize(pop_a, time_budget=alloc)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_parent = len(best_seq)
                print(f"  Pop {i}: obj={best_score:.8f} **")
            else:
                if obj_opt < best_score + 0.01:
                    print(f"  Pop {i}: obj={obj_opt:.8f} (close)")

    # Phase 4: Multi-n exploration (aggressive)
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 200), n_cur - 100, n_cur,
        n_cur + 100, n_cur + 200, n_cur + 500, n_cur + 1000,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 4000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 30)
        a_opt, obj_opt = local_optimize(a_try, time_budget=alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 5: Iterative refine-diversify cycle
    print(f"\n=== Phase 5: Iterative Refine-Diversify ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for iteration in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        cands = diversify_population(best_seq, rng, n_diverse=3)
        for i, cand in enumerate(cands):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.1, 15)
            if alloc < 3:
                continue
            a_opt, obj_opt = local_optimize(cand, time_budget=alloc)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_best = len(best_seq)
                print(f"  Iter {iteration} c{i}: obj={best_score:.8f} **")

    # Phase 6: Intensive LP with multi-peak shaving
    print(f"\n=== Phase 6: Intensive LP Multi-Shave ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80)
        a_lp2 = lp_alternating(best_seq, time_budget=alloc,
                               shave_mode='multi', shave_param=0.04)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Multi-shave LP: obj={best_score:.8f} **")

    # Phase 7: Final high-precision polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        # Very fine LP
        alloc = min(remaining * 0.3, 50)
        a_final_lp = lp_alternating(best_seq, time_budget=alloc,
                                    shave_mode='single', shave_param=0.01)
        obj_fl = compute_obj(a_final_lp)
        if obj_fl < best_score:
            best_score = obj_fl
            best_seq = a_final_lp.copy()
            print(f"  Fine LP: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Top-K smoothed optimization
        n_cur = len(best_seq)
        for topK in [3, 5]:
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            alloc = min(remaining * 0.3, 20)
            a_topk, obj_topk = smooth_lbfgs(best_seq, tau_start=50, tau_end=800,
                                            time_budget=alloc, max_iter=80, topk=topK)
            if obj_topk < best_score:
                best_score = obj_topk
                best_seq = a_topk.copy()
                print(f"  TopK={topK}: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        # Subgradient polish
        a_sg_final, obj_sg_final = subgradient_descent(best_seq,
                                                       time_budget=min(remaining * 0.4, 15),
                                                       tau=200)
        if obj_sg_final < best_score:
            best_score = obj_sg_final
            best_seq = a_sg_final.copy()
            print(f"  Subgrad final: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
