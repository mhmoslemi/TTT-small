        return np.ones(n) / n

"""
Optimized minimization of 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (log-sum-exp approximation) with tau annealing
2. Top-K subgradient with Nesterov momentum

4. Multi-dimension exploration and diverse restarts
5. Coordinate descent polishing
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = idx[cond]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / rho[-1]
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, g>=0, sum(g)=1. Returns (M_opt, g_opt)."""
    n = len(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

                sv = float(np.sum(g_val))
                if sv > 1e-10:

        except Exception:

    return None, None


def smooth_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)_k)) w.r.t. a.
    Returns (gradient, conv, peak_idx).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    w = np.exp(tau * (conv - max_conv))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum

    # grad_i = 2 * sum_j a[j] * w[i+j] for i=0..n-1
    v = np.zeros(n)
    for j in range(n):
        v += a[j] * w[j:j + n]
    return 2.0 * v, conv, int(np.argmax(conv))


def topk_subgradient(a, K=3):
    """Compute weighted subgradient targeting top-K peaks."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n), conv, int(top_idx[0])
    # Temperature-based weighting
    temp = 5.0
    weights = np.exp((peak_vals - max_val) * temp / max_val)
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
    return g, conv, int(top_idx[0])


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                stall = 0
                if verbose and it % 20 == 0:
                    elapsed = time.monotonic() - start
                    print(f"      LP it{it}: obj={best_obj:.8f} (t={best_t_val:.2f}), M_lp={M_lp:.8f}, t={elapsed:.1f}s")
            else:
                stall += 1
                if stall > 50:
                    # Check if we're at a fixed point
                    dist = float(np.sum(np.abs(g_lp - a)))
                    if dist < 1e-4:
                        # Near fixed point, perturb more aggressively
                        rng = np.random.default_rng(it * 31337 + n)
                        scale = min(0.5, 0.1 + 0.1 * (stall // 50))
                        a = best_a * np.exp(rng.normal(0, scale, n))
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s > 1e-10:
                            a = a / s
                        best_a = a.copy()
                        best_obj = compute_obj(a)
                        stall = 0
                        last_t = 0.5
                    else:
                        # Try a different perturbation
                        rng = np.random.default_rng(it * 104729 + n)
                        a = best_a + 0.01 * rng.standard_normal(n)
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s > 1e-10:
                            a = a / s
                        stall = 0
                        last_t = 0.5
                elif stall > 20:
                    # Try geometric interpolation
                    a Geo = np.power(np.maximum(a, 1e-15), 1-last_t) * np.power(np.maximum(g_lp, 1e-15), last_t)
                    a_geo = a_geo / a_geo.sum()
                    obj_geo = compute_obj(a_geo)
                    if obj_geo < best_obj - 1e-12:
                        best_obj = obj_geo
                        best_a = a_geo.copy()
                        a = a_geo.copy()
                        stall = 0

        it += 1

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """
    Coordinate descent: try moving mass between pairs of positions to reduce peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False

        # Find top contributors to the peak
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                c = a[i] * a[j]
                contribs.append((c, i, j))
        contribs.sort(reverse=True)

        # Find non-contributors
        contributor_set = set()
        for _, i, j in contribs:
            contributor_set.add(i)
            contributor_set.add(j)
        non_contrib = [i for i in range(n) if i not in contributor_set]

        if not non_contrib:
            break

        for contrib, i, j in contribs[:30]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break
            # Find best target
            for target in non_contrib[:10]:
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = frac * a[i]
                    if delta < 1e-15:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            for _ in range(50):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * a[i]
                if delta < 1e-15:
                    continue
                a[i] -= delta
                a[j] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x * x + 0.01))
        a = a / a.sum()
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w * w))
        a = a / a.sum()
    elif mode == 4:
        # Inverted triangle
        x = np.linspace(-1, 1, n)
        a = 1.0 - 0.7 * np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Smooth random
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amps = rng.uniform(0.2, 0.5, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, am in zip(freqs, phases, amps):
            a += am * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Tail-heavy
        x = np.linspace(0, 1, n)
        a = x ** 3 + (1 - x) ** 3 + 0.05
        a = a / a.sum()
    else:
        a = rng.exponential(1.0, n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines smooth gradient, top-K subgradient,
    LP alternating, multi-dimension exploration, and coordinate polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_a = None
    best_obj = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_a = parent_arr / s
                best_obj = compute_obj(best_a)
                print(f"Parent (n={parent_arr.size}): obj={best_obj:.8f}")
    except Exception:
        pass

    if best_a is None:
        n_init = 800
        best_a = np.ones(n_init) / n_init
        best_obj = compute_obj(best_a)
        print(f"Default init (n={n_init}): obj={best_obj:.8f}")

    n_best = len(best_a)
    print(f"Starting optimization with n={n_best}, obj={best_obj:.8f}")

    # === Phase 1: Smooth gradient with tau annealing ===
    print(f"\n=== Phase 1: Smooth Gradient (tau annealing) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        a_sg = smooth_gradient_optimize(best_a, time_budget=p1_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            print(f"  Smooth gradient: obj={best_obj:.8f} ** IMPROVED")
        else:
            print(f"  Smooth gradient: obj={obj_sg:.8f} (no improvement)")

    # === Phase 2: Top-K subgradient ===
    print(f"\n=== Phase 2: Top-K Subgradient ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 20:
        a_tk = topk_subgradient_optimize(best_a, time_budget=p2_budget, K=3, verbose=True)
        obj_tk = compute_obj(a_tk)
        if obj_tk < best_obj:
            best_obj = obj_tk
            best_a = a_tk.copy()
            n_best = len(best_a)
            print(f"  Top-K: obj={best_obj:.8f} ** IMPROVED")
        else:
            print(f"  Top-K: obj={obj_tk:.8f} (no improvement)")

    # === Phase 3: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.35, 500.0)

    # Run multiple rounds of LP
    for rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p3_budget / (3 - rnd), remaining * 0.6)
        if alloc < 20:
            break
        a_lp = lp_alternating_optimize(best_a, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            n_best = len(best_a)
            print(f"  LP round {rnd}: obj={best_obj:.8f} ** IMPROVED")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Multi-dimension exploration ===
    print(f"\n=== Phase 4: Multi-dimension Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        600, 700, 800, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]
    n_options = list(dict.fromkeys(n_options))  # deduplicate preserving order

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 6:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_sequence(best_a, n_try)

        # Quick optimization: LP alternating
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 15:
            a_opt = lp_alternating_optimize(a_try, time_budget=alloc, verbose=False)
            obj_opt = compute_obj(a_opt)
        else:
            a_opt = a_try
            obj_opt = compute_obj(a_opt)

        # If promising, do more optimization
        if obj_opt < best_obj + 0.05:
            remaining = deadline - time.monotonic()
            extra_alloc = min(remaining * 0.1, 60.0)
            if extra_alloc > 15:
                a_opt2 = smooth_gradient_optimize(a_opt, time_budget=extra_alloc, verbose=False)
                obj_opt2 = compute_obj(a_opt2)
                if obj_opt2 < obj_opt:
                    a_opt = a_opt2
                    obj_opt = obj_opt2

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            n_best = len(best_a)
            print(f"  ** n={n_try}: obj={best_obj:.8f} ** IMPROVED")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

        n_tried += 1

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_restarts = 4

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r + 2) % 7
        a_r = diverse_init(n_best, rng, mode=mode)

        # Blend with best to maintain quality
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_a
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        restart_budget = min(remaining * 0.15, 80.0)
        if restart_budget > 15:
            a_opt = lp_alternating_optimize(a_r, time_budget=restart_budget, verbose=False)
            obj_r = compute_obj(a_opt)
            if obj_r < best_obj:
                best_obj = obj_r
                best_a = a_opt.copy()
                n_best = len(best_a)
                print(f"  Restart {r} (mode={mode}): obj={best_obj:.8f} ** IMPROVED")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # More LP
    if remaining > 30:
        lp_alloc = min(remaining * 0.3, 120.0)
        a_lp = lp_alternating_optimize(best_a, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            print(f"  Final LP: obj={best_obj:.8f} ** IMPROVED")

    # Smooth gradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        sg_alloc = min(remaining * 0.25, 60.0)
        a_sg = smooth_gradient_optimize(best_a, time_budget=sg_alloc, verbose=False)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            print(f"  Final SG: obj={best_obj:.8f} ** IMPROVED")

    # Top-K polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        tk_alloc = min(remaining * 0.2, 40.0)
        a_tk = topk_subgradient_optimize(best_a, time_budget=tk_alloc, K=5, verbose=False)
        obj_tk = compute_obj(a_tk)
        if obj_tk < best_obj:
            best_obj = obj_tk
            best_a = a_tk.copy()
            print(f"  Final TK: obj={best_obj:.8f} ** IMPROVED")

    # Coordinate polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        cd_alloc = min(remaining * 0.25, 30.0)
        a_cd = coordinate_polish(best_a, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            print(f"  Final CD: obj={best_obj:.8f} ** IMPROVED")

    # One final LP pass
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_last = lp_alternating_optimize(best_a, time_budget=min(remaining * 0.5, 30.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last.copy()
            print(f"  Final LP2: obj={best_obj:.8f} ** IMPROVED")

    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    # Ensure non-negative and reasonable
    best_a = np.maximum(best_a, 0.0)
    s_final = float(np.sum(best_a))
    if s_final < 0.01:
        best_a = np.ones(100)
        s_final = 100.0

    result = [float(min(1000.0, max(0.0, x))) for x in best_a]

    # Validate with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        final_obj = val
    else:
        # Fallback
        result = [1.0] * 100
        final_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_obj:.8f}, time={elapsed:.1f}s ===")

    return result
