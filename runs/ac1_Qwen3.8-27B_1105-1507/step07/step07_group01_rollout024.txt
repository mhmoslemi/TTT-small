import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s*s)

    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    conv = compute_conv(a)
    best_obj = compute_obj(a)
    
    for pass_num in range(max_passes):
        improved = False
        for i in range(n):
            # Compute shifted a: a[k-i] for valid k
            # k ranges from max(0,i) to min(2n-2, i+n-1)
            k_start = i  # k-i = 0
            k_end = min(2*n-2, i+n-1)  # k-i = n-1
            # a_shifted[k] = a[k-i] for k in [k_start, k_end]
            
            # Evaluate g(delta) = 2n*max_k [conv[k] + 2*delta*a_shifted[k] + delta^2*[k==2i]] / (S+delta)^2
            
            def g(delta):
                if S + delta < 1e-10:
                    return np.inf
                # New conv
                c = conv.copy()
                c[k_start:k_end+1] += 2*delta*a[k_start-i:k_end+1-i]
                quad_k = 2*i
                if quad_k < 2*n-1:
                    c[quad_k] = conv[quad_k] + 2*a[i]*delta + delta*delta
                max_c = np.max(c)
                if max_c < 0:
                    max_c = 0
                return 2*n*max_c / (S+delta)**2
            
            # Golden section search
            lo = -a[i] + 1e-12
            hi = max(S, 1.0)  # upper bound
            
            # Check if we should extend hi
            g_lo = g(lo + 1e-6)
            g_hi = g(hi)
            while g_hi < g_hi * 0.99 and hi < 10*S + 10:  # this doesn't make sense
                break
            
            # Just use a reasonable upper bound
            hi = max(1.0, S)
            
            # Golden section
            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            fc1 = g(c1)
            fc2 = g(c2)
            
            for _ in range(40):
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    fc1 = g(c1)
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    fc2 = g(c2)
                if hi - lo < 1e-10 * max(1, abs(a[i])):
                    break
            
            delta = (lo + hi) / 2
            # Check if delta=0 is better
            g0 = g(0)
            if g(delta) > g0:
                delta = 0
            
            if abs(delta) > 1e-12:
                a[i] += delta
                a[i] = max(a[i], 0)
                S = float(np.sum(a))
                conv = compute_conv(a)
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    improved = True
        
        if verbose and pass_num % 10 == 0:
            print(f"  CD pass {pass_num}: obj={best_obj:.8f}")
        
        if not improved:
            break
    
    return a

"""
Coordinate descent + LP alternating optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Innovation: Exact 1D coordinate minimization via golden section search on the
convolution update formula, combined with LP alternating for escape from local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def coordinate_descend(a, max_passes=200, tol=1e-10, verbose=False, rng=None):
    """
    Coordinate descent to minimize 2n*max(conv(a,a))/(sum(a))^2.
    Uses exact 1D golden section search for each coordinate update.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv)) / (S * S)
    
    if rng is None:
        rng = np.random.default_rng(42)
    
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    for pass_num in range(max_passes):
        # Random order for this pass
        order = rng.permutation(n)
        max_improvement = 0.0
        
        for i in order:
            ai = a[i]
            if ai < 1e-15 and S < 1e-10:
                continue
                
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            quad_k = 2 * i
            has_quad = quad_k < 2 * n - 1 and quad_k >= k_lo and quad_k <= k_hi
            # a_slice = a[0 : k_hi - k_lo + 1]  (this is a[0 : min(n, 2n-1-i)])
            a_slice_len = k_hi - k_lo + 1
            a_slice = a[:a_slice_len]
            
            # Precompute for speed: the outside max
            if k_lo > 0:
                outside_left_max = np.max(conv[:k_lo])
            else:
                outside_left_max = 0.0
            if k_hi < 2 * n - 2:
                outside_right_max = np.max(conv[k_hi + 1:])
            else:
                outside_right_max = 0.0
            outside_max = max(outside_left_max, outside_right_max)
            
            def g_eval(delta):
                denom = S + delta
                if denom < 1e-12:
                    return np.inf
                # Max in range [k_lo, k_hi]
                vals = conv[k_lo:k_hi + 1] + 2.0 * delta * a_slice
                if has_quad:
                    idx = quad_k - k_lo
                    vals[idx] = conv[quad_k] + 2.0 * ai * delta + delta * delta
                max_in = float(np.max(vals))
                max_c = max(max_in, outside_max, 0.0)
                return 2.0 * n * max_c / (denom * denom)
            
            # Golden section search
            lo = -ai + 1e-14
            # Upper bound: don't make any element dominate
            hi = max(0.5 * S, 0.01)
            
            # Quick check: is delta=0 already optimal?
            g0 = 2.0 * n * float(np.max(conv)) / (S * S)
            g_lo = g_eval(lo + 0.01 * max(ai, 1e-6))
            g_hi = g_eval(hi)
            
            if g_lo >= g0 and g_hi >= g0:
                # Minimum is at delta ≈ 0, skip
                continue
            
            # Extend hi if needed
            if g_hi < g0:
                for _ in range(5):
                    hi *= 2
                    g_new = g_eval(hi)
                    if g_new >= g_hi:
                        break
                    g_hi = g_new
            
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            fc1 = g_eval(c1)
            fc2 = g_eval(c2)
            
            for _ in range(35):
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    fc1 = g_eval(c1)
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    fc2 = g_eval(c2)
                if hi - lo < 1e-12 * max(1.0, S):
                    break
            
            delta = (lo + hi) / 2.0
            g_delta = g_eval(delta)
            
            # Also check closest to 0
            if g_delta > g0:
                delta = 0.0
                g_delta = g0
            
            if abs(delta) > 1e-14:
                # Update conv incrementally
                conv[k_lo:k_hi + 1] += 2.0 * delta * a_slice
                if has_quad:
                    conv[quad_k] += delta * delta
                a[i] = ai + delta
                S += delta
                if g_delta < best_obj:
                    best_obj = g_delta
                    max_improvement = max(max_improvement, g0 - g_delta)
        
        if verbose and pass_num % 10 == 0:
            print(f"    CD pass {pass_num}: obj={best_obj:.10f}")
        
        if max_improvement < tol:
            break
    
    return a


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    a = a / np.sum(a)
    
    # Build convolution matrix A (2n-1 x n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in ['CLARABEL', 'MOSEK', 'SCS', 'ECOS', 'PDLP']:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):

    n = len(a)

        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    max_iters = 500
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search over t
        best_t = 0.3
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a_try = (1 - best_t) * a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            best_a = a_try.copy()
            a = a_try
            stall = 0
            if verbose and it % 20 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")

            stall += 1
            if stall > 20:

                a = best_a * np.exp(rng.normal(0, 0.02, n))

                stall = 0
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
