import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))

    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            s = float(np.sum(g_val))
            if s > 1e-10:

    return None, None

            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a
            a[j_min] += best_t
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            a = a / s
            no_improve_count = 0
            if best_t_obj < best_obj:
                best_obj = best_t_obj
                best_a = a.copy()
                if verbose and it % 10000 == 0:
                    print(f"    FW it={it}: obj={best_obj:.8f}")
        else:
            no_improve_count += 1
        
        # Stagnation: perturb
        if no_improve_count > stagnation_threshold:
            rng = np.random.default_rng(it + n)
            n_perturb = max(1, n // 100)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.05, n_perturb))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            a = a / s
            no_improve_count = 0
        
        it += 1
    
    return best_a, best_obj

def lp_alternating(a, time_budget=120.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose and it % 5 == 0:
                print(f"    LP alt it={it}: obj={best_obj:.8f}")
        else:
            # Perturb to escape
            if it % 5 == 0:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
    
    return best_a

def simulated_annealing(a, time_budget=200.0, T_start=0.02, T_end=0.001, verbose=False):
    """Simulated annealing on the sequence."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    total_iter = 0
    
    while (time.monotonic() - start) < time_budget:
        # Adaptive temperature
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_start * (T_end / T_start) ** progress
        
        # Generate perturbation: move mass between two random coordinates
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Amount to move: proportional to current value
        frac = rng.uniform(0.01, 0.1)
        delta = frac * a[i]
        if delta > a[i] * 0.9:
            delta = a[i] * 0.5
        
        a[i] -= delta
        a[j] += delta
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        a = a / s
        
        new_obj = compute_obj(a)
        total_iter += 1
        
        if new_obj < current_obj or rng.random() < np.exp(-(new_obj - current_obj) / T):
            current_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                if verbose and int((time.monotonic() - start) * 2) % 10 == 0:
                    print(f"    SA t={elapsed:.0f}s: best={best_obj:.8f}, iter={total_iter}")
        else:
            # Revert
            a[i] += delta
            a[j] -= delta
            a = np.maximum(a, 0)
            a = a / s
    
    return best_a

def simulated_annealing_gaussian(a, time_budget=200.0, sigma=0.01, verbose=False):
    """Simulated annealing with Gaussian multiplicative perturbations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(54321)
    
    while (time.monotonic() - start) < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = 0.01 * (0.001) ** progress
        sigma_t = sigma * (1.0 - 0.7 * progress)
        
        # Gaussian multiplicative perturbation on a subset
        n_perturb = max(5, n // 20)
        mask = rng.random(n) < (n_perturb / n)
        a_new = a.copy()
        a_new[mask] *= np.exp(rng.normal(0, sigma_t, mask.sum()))
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        new_obj = compute_obj(a_new)
        
        if new_obj < current_obj or rng.random() < np.exp(-(new_obj - current_obj) / T):
            current_obj = new_obj
            a = a_new
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a_new.copy()
                if verbose and int(elapsed) % 30 == 0:
                    print(f"    SA-G t={elapsed:.0f}s: best={best_obj:.8f}")
        # Else: keep a as is (don't revert, just don't accept)
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
