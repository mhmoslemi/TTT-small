For a general non-negative a, the convolution conv[k] for k=0,...,n-1 is the "autocorrelation" (or more precisely, the cross-correlation of a with itself).

Now, the key question is: what is the minimum possible value of max(conv) for a non-negative sequence of length n with sum 1?

This is a well-studied problem in signal processing (minimum peak sidelobe level). However, the standard problem usually allows complex or signed sequences. For non-negative sequences, the problem is different.

Let me think about it from a different angle. We want to minimize max_k conv[k] subject to sum a = 1, a >= 0.

By the sum constraint: sum_k conv[k] = (sum a)^2 = 1.
So the average is 1/(2n-1). The maximum must be at least the average: max >= 1/(2n-1).
This gives a lower bound on the objective: 2n/(2n-1), which approaches 1 as n grows.

For the uniform sequence, max = 1/n (at the center), objective = 2.

The center value conv[n-1] = sum a[i]*a[n-1-i]. For this to be < 1/n, we need the "cross terms" to be small. This happens when a[i] and a[n-1-i] are "anti-correlated" (one is large when the other is small).

                product = a[i] * a[j]
                if product > 0:
                    candidates.append((product, i, j))

        candidates.sort(reverse=True)

        for product, i, j in candidates[:10]:
            if time.monotonic() - start > time_budget:
                break
            # Try reducing a[i] (and/or a[j]) and adding to a random other position
            target = rng.integers(0, n)
            while target == i or target == j:
                target = rng.integers(0, n)

            for frac in [0.02, 0.05, 0.1]:
                delta = frac * a[i]
                if delta >= a[i]:
                    continue
                a[i] -= delta
                a[target] += delta
                # Recompute conv incrementally (approximate)
                # Full recompute for accuracy
                conv_new = compute_conv(a)
                obj_new = 2.0 * n * float(np.max(conv_new))
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    conv = conv_new
                    conv_max = float(np.max(conv))
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[target] -= delta

        if not improved:
            # Try random pair moves
            for _ in range(20):
                if time.monotonic() - start > time_budget:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * a[i] * rng.uniform(0.5, 1.5)
                if delta >= a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                conv_new = compute_conv(a)
                obj_new = 2.0 * n * float(np.max(conv_new))
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    conv = conv_new
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved and it > 100:
                # Break out of local minimum
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s_p = float(np.sum(a))
                if s_p > 1e-10:
                    a = a / s_p
                    conv = compute_conv(a)

    return best_a, best_obj


def smooth_lbfgs(a_init, time_budget, max_iter=150):
    """Smoothed L-BFGS-B with log-sum-exp and tau annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    n_stages = 4
    taus = np.geomspace(10.0, 500.0, n_stages)

    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        stage_budget = min(remaining * 0.7, 60.0)

        def obj_func(th):
            a_local = np.exp(th - np.max(th))
            a_local = a_local / np.sum(a_local)
            conv = compute_conv(a_local)
            m = np.max(conv)
            return 2.0 * n * (m + np.log(np.sum(np.exp((conv - m) * tau))) / tau)

        def grad_func(th):
            a_local = np.exp(th - np.max(th))
            a_local = a_local / np.sum(a_local)
            conv = compute_conv(a_local)
            m = np.max(conv)
            w = tau * np.exp((conv - m) * tau)
            w = w / np.sum(w)
            c = np.zeros(n)
            for j in range(n):
                end = min(2 * n - 1, j + n)
                c[j] = np.dot(w[j:end], a_local[:end - j])
            mean_c = np.dot(c, a_local)
            return 4.0 * n * a_local * (c - mean_c)

        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-6, 'maxfun': 300})
            if result.x is not None:
                a_new = np.exp(result.x - np.max(result.x))
                a_new = a_new / np.sum(a_new)
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj
