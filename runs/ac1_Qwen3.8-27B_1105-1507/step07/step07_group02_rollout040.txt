    n = len(a)
    m = 2*n - 1

    for j in range(n):
        # conv[a,x][k] = sum_i a[i]*x[k-i]
        # For fixed j (x[j]), it contributes to conv[k] when k-j is a valid index of a
        # i.e., k-j in [0, n-1], so k in [j, j+n-1]

"""

Strategy: LP alternating optimization with warm-starting, reduced active-set LPs,
KKT-condition-based polish, gradient refinement, and multi-n/multi-init exploration.
Key improvements over previous: faster LP via active-set reduction, KKT system solve,
adaptive tau smoothing for subgradient, and broader n exploration.

import numpy as np
from scipy.fft import rfft, irfft
import time


def convolve_full(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)

def obj_val(a):

    n = len(a)
    c = convolve_full(a)
    return 2.0 * n * float(np.max(c))


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_A(a):
    """Build dense matrix A (2n-1 x n) such that A @ x = convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, m - 1)
        A[k_lo:k_hi + 1, j] = a[0:k_hi + 1 - j]
    return A

                elif hasattr(res.ineqlin, 'mult') or isinstance(res.ineqlin, np.ndarray):
                    try:

def kkt_polish(a, time_budget=30.0, verbose=False):
    """
    Polish using KKT conditions: identify active set, solve linear system.
    At optimum, for active set A with weights w:
      2 * sum_{k in A} w_k * a[k-m] = lambda (for m in support)
      sum(a) = 1, a >= 0
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()

    for polish_round in range(50):
        if time.monotonic() - start > time_budget:
            break

        conv = convolve_full(a)
        M = float(np.max(conv))

        # Identify active set (within tolerance of max)
        tol = max(M * 0.01, 1e-10)
        active = np.where(conv >= M - tol)[0]

        if len(active) == 0:
            break
        if len(active) == 2 * n - 1:
            # Everything is active, can't reduce
            active = np.where(conv >= M - tol * 5)[0]
            if len(active) == 0:
                break

        # Use uniform weights over active set (or use LP dual if available)
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        # Build the linear system for KKT:
        # 2 * sum_k w_k * a[k-m] = lambda for all m where a[m] > 0
        # sum(a) = 1
        # a >= 0
        #
        # Let S = support(a) = {m : a[m] > epsilon}
        eps_support = 1e-8
        S = np.where(a > eps_support)[0]

        if len(S) == 0:
            break

        # Unknowns: a[0..n-1] and lambda => n+1 unknowns
        # Equations: |S| (KKT) + 1 (normalization) = |S|+1 equations
        # This is overdetermined if |S|+1 > n+1, which happens when |S| > n
        # |S| <= n always, so we have at most n+1 equations for n+1 unknowns

        num_eq = len(S) + 1
        num_unk = n + 1

        # Build system: G @ [a; lambda] = rhs
        # Row for m in S: 2*sum_k w_k * a[k-m] - lambda = 0
        # Row for normalization: sum(a) = 1
        G = np.zeros((num_eq, num_unk))
        rhs = np.zeros(num_eq)

        for idx, m in enumerate(S):
            for k in active:
                j = k - m
                if 0 <= j < n:
                    G[idx, j] += 2.0 * w[k]
            G[idx, n] = -1.0  # -lambda
            rhs[idx] = 0.0

        # Normalization row
        G[num_eq - 1, :n] = 1.0
        G[num_eq - 1, n] = 0.0
        rhs[num_eq - 1] = 1.0

        # Solve (least squares if overdetermined, direct if square)
        try:
            sol, residuals, rank, sv = np.linalg.lstsq(G, rhs, rcond=None)
            a_sol = np.maximum(sol[:n], 0)
            lam = sol[n]
        except Exception:
            break

        a_sol = normalize(a_sol)
        obj_sol = obj_val(a_sol)

        if obj_sol < best_obj - 1e-13:
            best_obj = obj_sol
            best_a = a_sol.copy()
            a = a_sol.copy()
            if verbose:
                print(f"    KKT polish {polish_round}: obj={best_obj:.10f}, "
                      f"|active|={len(active)}, |support|={len(S)}")
        else:
            # Try with tolerantly expanded/contracted active set
            tol = max(M * 0.05, 1e-8)
            active2 = np.where(conv >= M - tol)[0]
            if len(active2) != len(active):
                w = np.zeros(2 * n - 1)
                w[active2] = 1.0 / len(active2)
                # Rebuild and solve
                num_eq2 = len(S) + 1
                G2 = np.zeros((num_eq2, num_unk))
                rhs2 = np.zeros(num_eq2)
                for idx, m in enumerate(S):
                    for k in active2:
                        j = k - m
                        if 0 <= j < n:
                            G2[idx, j] += 2.0 * w[k]
                    G2[idx, n] = -1.0
                G2[num_eq2 - 1, :n] = 1.0
                rhs2[num_eq2 - 1] = 1.0
                try:
                    sol2, _, _, _ = np.linalg.lstsq(G2, rhs2, rcond=None)
                    a_sol2 = normalize(np.maximum(sol2[:n], 0))
                    obj2 = obj_val(a_sol2)
                    if obj2 < best_obj - 1e-13:
                        best_obj = obj2
                        best_a = a_sol2.copy()
                        a = a_sol2.copy()
                    else:
                        break
                except Exception:
                    break
            else:
                # Small perturbation to escape
                rng = np.random.default_rng(polish_round * 31 + n)
                a = normalize(best_a * np.exp(rng.normal(0, 0.01, n)))
                if polish_round > 30:
                    break

    return best_a


def gradient_refine(a, time_budget=30.0, verbose=False):
    """
    Subgradient/gradient refinement using smoothed max.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()

    tau = 5.0
    y = a.copy()
    t_mom = 1.0
    prev_a = a.copy()

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = convolve_full(y)
        M = float(np.max(conv))
        obj_y = 2.0 * n * M

        # Compute weights (smoothed max)
        shifted = conv - M
        max_shift = float(np.max(np.abs(shifted)))
        if max_shift < 1e-15:
            weights = np.full(len(conv), 1.0 / len(conv))
        else:
            scaled = tau * shifted / max_shift
            # Clip to avoid overflow
            scaled = np.clip(scaled, -50, 50)
            exp_vals = np.exp(scaled)
            weights = exp_vals / exp_vals.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for k in range(len(conv)):
            if weights[k] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = k - np.arange(lo, hi + 1)
                grad[lo:hi + 1] += 2.0 * weights[k] * y[idx]
        grad *= 2.0 * n

        # Center to preserve sum
        grad_centered = grad - np.mean(grad)
        grad_norm = float(np.sum(grad_centered ** 2))
        if grad_norm < 1e-20:
            break

        # Adaptive step size
        # Use Barra-Moreau type: step = (obj - obj_lower) / grad_norm
        # Lower bound heuristic: 2 (theoretical minimum for uniform)
        step = 2.0 / (grad_norm + 1e-10)
        step = min(step, 0.5 / (2.0 * n + 1))

        x = normalize(y - step * grad_centered)
        obj_x = obj_val(x)

        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x.copy()

        # FISTA momentum update
        t_new = (1 + np.sqrt(1 + 4 * t_mom ** 2)) / 2
        y = x + (t_mom - 1) / t_new * (x - prev_a)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = x.copy()
        else:
            y = y / ys
        prev_a = x
        t_mom = t_new

        # Increase tau periodically
        if it % 100 == 0 and it > 0:
            tau = min(tau * 1.2, 100.0)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Sinusoidal modulation
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 2:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            freq = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.05, 0.3)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Periodic blocks
        period = max(2, n // rng.integers(3, 10))
        block = rng.uniform(0.1, 1.0, min(period, n))
        n_blocks = n // period + 1
        a = np.tile(block, n_blocks)[:n]
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Gaussian bumps
        a = np.zeros(n)
        n_bumps = rng.integers(2, 6)
        for _ in range(n_bumps):
            c = rng.uniform(0, n)
            w = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(0.5, 2.0)
            a += amp * np.exp(-0.5 * ((np.arange(n) - c) / w) ** 2)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        # Smooth random
        a = rng.uniform(0, 1, n)
        # Smooth with moving average
        k = max(1, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP alternating, KKT polish, gradient refinement,
    and multi-n/multi-init exploration to minimize the objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Get parent sequence ---
    parent = None
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                n_parent = parent.size
                score_p = evaluate_sequence([float(x) for x in parent_arr])
                print(f"Parent (n={n_parent}): score={score_p:.10f}")
                if score_p < np.inf:
                    best_score = min(best_score, score_p)
                    best_seq = parent_arr.copy()
    except Exception as e:
        print(f"Parent not available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_parent)
        best_score = evaluate_sequence([1.0] * n_parent)
        parent = np.ones(n_parent) / n_parent
        print(f"Default uniform (n={n_parent}): score={best_score:.10f}")

    print(f"\n=== Starting optimization, initial best: n={len(best_seq)}, score={best_score:.10f} ===")

    # Phase 1: LP alternating on parent (main workhorse)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n--- Phase 1: LP alternating (budget={remaining:.0f}s) ---")
        alloc = min(remaining * 0.45, 600.0)
        a1 = lp_alternating(parent, time_budget=alloc, verbose=True)
        s1 = float(np.sum(a1))
        if s1 > 0:
            score1 = evaluate_sequence([float(x) for x in a1 * 1000])
            # Evaluate with original scale
            score1 = evaluate_sequence([float(x) for x in a1])
            print(f"  LP alt result: score={score1:.10f}")
            if score1 < best_score:
                best_score = score1
                best_seq = (a1 * np.max(parent)).copy()

    # Phase 2: KKT polish on best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n--- Phase 2: KKT polish (budget={min(remaining*0.15, 120.0):.0f}s) ---")
        a_best_norm = best_seq / np.sum(best_seq)
        alloc = min(remaining * 0.15, 120.0)
        a2 = kkt_polish(a_best_norm, time_budget=alloc, verbose=True)
        s2 = float(np.sum(a2))
        if s2 > 0:
            score2 = evaluate_sequence([float(x) for x in a2])
            print(f"  KKT polish result: score={score2:.10f}")
            if score2 < best_score:
                best_score = score2
                best_seq = a2.copy()

    # Phase 3: Gradient refinement
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n--- Phase 3: Gradient refinement ---")
        a_best_norm = best_seq / np.sum(best_seq)
        alloc = min(remaining * 0.1, 90.0)
        a3 = gradient_refine(a_best_norm, time_budget=alloc, verbose=True)
        score3 = evaluate_sequence([float(x) for x in a3])
        print(f"  Gradient refine result: score={score3:.10f}")
        if score3 < best_score:
            best_score = score3
            best_seq = a3.copy()

    # Phase 4: More LP alternating (second round on refined)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n--- Phase 4: LP alternating round 2 ---")
        a_best_norm = best_seq / np.sum(best_seq)
        alloc = min(remaining * 0.25, 300.0)
        a4 = lp_alternating(a_best_norm, time_budget=alloc, verbose=True)
        score4 = evaluate_sequence([float(x) for x in a4])
        print(f"  LP alt round 2 result: score={score4:.10f}")
        if score4 < best_score:
            best_score = score4
            best_seq = a4.copy()

    # Phase 5: Multi-n exploration
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n--- Phase 5: Multi-n exploration ---")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 200),
            max(200, n_best - 100),
            n_best + 50,
            n_best + 100,
            n_best + 200,
            300, 500, 700, 900, 1000, 1200, 1500, 1800, 2000
        ]))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq / np.sum(best_seq), n_try)

            a_try = normalize(a_try)

            # Quick LP optimization
            alloc = min(30.0, remaining * 0.1)
            a_opt = lp_alternating(a_try, time_budget=alloc)
            score_try = evaluate_sequence([float(x) for x in a_opt])
            if score_try < best_score:
                best_score = score_try
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: score={best_score:.10f} **")
            else:
                print(f"  n={n_try}: score={score_try:.10f}")

    # Phase 6: Diverse restarts
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n--- Phase 6: Diverse restarts ---")
        n_best = len(best_seq)
        n_restarts = min(10, int(remaining // 25))

        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_r = diverse_init(n_best, rng)
            alloc = min(25.0, remaining * 0.3)
            a_opt = lp_alternating(a_r, time_budget=alloc)
            score_r = evaluate_sequence([float(x) for x in a_opt])
            if score_r < best_score:
                best_score = score_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: score={best_score:.10f} **")
            else:
                print(f"  Restart {r}: score={score_r:.10f}")

    # Phase 7: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n--- Phase 7: Final polish (remaining={remaining:.0f}s) ---")
        a_final = best_seq / np.sum(best_seq)

        # KKT polish
        alloc1 = min(remaining * 0.3, 60.0)
        a_p1 = kkt_polish(a_final, time_budget=alloc1)
        s1 = evaluate_sequence([float(x) for x in a_p1])
        if s1 < best_score:
            best_score = s1
            best_seq = a_p1.copy()
            a_final = a_p1 / np.sum(a_p1)
            print(f"  KKT final: score={best_score:.10f} **")

        # LP final
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc2 = min(remaining * 0.4, 90.0)
            a_p2 = lp_alternating(a_final, time_budget=alloc2)
            s2 = evaluate_sequence([float(x) for x in a_p2])
            if s2 < best_score:
                best_score = s2
                best_seq = a_p2.copy()
                print(f"  LP final: score={best_score:.10f} **")

        # Gradient final
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_p3 = gradient_refine(best_seq / np.sum(best_seq),
                                   time_budget=min(remaining * 0.3, 30.0))
            s3 = evaluate_sequence([float(x) for x in a_p3])
            if s3 < best_score:
                best_score = s3
                best_seq = a_p3.copy()
                print(f"  Gradient final: score={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val >= np.inf:
        result = [1.0] * 800
        final_val = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_val:.10f}, time={elapsed:.1f}s ===")
    return result
