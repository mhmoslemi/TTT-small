    n = len(a)
    N = 2 * n - 1
    A = np.zeros((N, n))
    for j in range(n):
        A[j:j + n, j] = a[:n]
    return A


def lp_polish(a, max_iters=5, verbose=False):
    """LP alternating: solve min M s.t. conv(a,g)<=M, sum(g)=1, g>=0; blend."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(max_iters):
        A = build_conv_matrix(a)
        N = 2 * n - 1

        # LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((N, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(N)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s < 1e-10:
                break
            g_val /= s
        except Exception:
            break

        # Line search over blending parameter
        improved = False
        for t in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try /= s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True

        if not improved:
            break

        if verbose:
            print(f"    LP iter {it}: obj={best_obj:.10f}")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)
