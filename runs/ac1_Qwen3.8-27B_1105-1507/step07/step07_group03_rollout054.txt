import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = a.copy()  # Nesterov momentum point
    prev_y = a.copy()
    
    for t in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        max_val = float(np.max(conv_y))

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- FFT flat-spectrum initialization
- Nesterov-accelerated smooth gradient (log-sum-exp approximation)
- LP alternating optimization
- Multi-n exploration with resampling
- Final coordinate refinement

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(y, mu):
    """
    Compute gradient of f_mu(a) = (1/mu)*log(sum_k exp(mu*conv[k])) at point y.
    Returns (gradient, f_mu_value).
    """
    conv_y = compute_conv(y)
    # Numerically stable log-sum-exp
    max_c = float(np.max(conv_y))
    shifted = mu * (conv_y - max_c)
    exp_shifted = np.exp(shifted)
    log_sum_exp = max_c + np.log(float(np.sum(exp_shifted))) / mu
    p = exp_shifted / float(np.sum(exp_shifted))

    # Gradient: sum_k p[k] * 2*y[k-i] (for valid k-i)
    n = len(y)
    grad = np.zeros(n)
    # For each k, contributes 2*p[k]*y[k-i] to grad[i] for valid i
    # Vectorized: for each k, the valid i range is [max(0, k-(n-1)), min(k, n-1)]
    # grad[i] = 2 * sum_k p[k] * y[k-i] for 0<=k-i<n
    # This is 2 * (p * y) convolved at specific positions... actually it's simpler:
    # grad[i] = 2 * sum_{k: 0<=k-i<n} p[k] * y[k-i]
    # Let j = k-i, so k = i+j, j in [0, n-1]:
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * y[j]  (where p[i+j] is valid, i.e., i+j < 2n-1)
    # This is 2 * (cross-correlation of p and y)
    # Actually, let's just do it directly with the valid ranges
    for k in range(2 * n - 1):
        if p[k] > 1e-15:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                # grad[lo:hi+1] += 2*p[k] * y[k-hi : k-lo+1][::-1]
                seg_lo = k - hi
                seg_hi = k - lo
                grad[lo:hi + 1] += 2.0 * p[k] * y[seg_lo:seg_hi + 1][::-1]

    return grad, float(log_sum_exp)


def nesterov_smooth(a, n_iters=100000, time_budget=60.0, mu=50.0):
    """
    Nesterov-accelerated gradient descent on the smooth approximation f_mu.
    Uses log-sum-exp for smoothness.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    t_count = 0

    for t in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute gradient at y
        grad, f_val = smooth_gradient(y, mu)

        # Adaptive mu: increase as we converge (sharper approximation)
        # mu_adaptive = mu * (1 + t // 1000) capped
        # For now use fixed mu

        # Nesterov step size: for L-smooth convex, step = 2/(L*(t+1))
        # L <= 2*mu (rough bound)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-10:
            break
        step = 2.0 / (mu * grad_norm * (t + 10))
        step = min(step, 2.0 / (mu * (t + 1) + 1))
        step = min(step, 1.0)

        # Gradient step + projection
        x_new = project_simplex(y - step * grad)

        # Nesterov extrapolation
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = (1.0 + 1.0 / alpha) * x_new - (1.0 / alpha) * x
        y_new = project_simplex(y_new)

        # Track best using TRUE objective (not smooth approx)
        if t % 10 == 0 or t < 20:
            obj_new = compute_obj(x_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = x_new.copy()

        x = x_new
        y = y_new
        t_count += 1

    # Final evaluation
    obj_final = compute_obj(y)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = y.copy()

    return best_a


def subgradient_multi_peak(a, n_iters=50000, time_budget=60.0, m_peaks=5):
    """
    Multi-peak softmax subgradient with adaptive step size and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    beta = 0.85
    step = 0.05
    stall = 0

    for t in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Top m peaks with softmax weights
        m = min(m_peaks, 2 * n - 1)
        top_idx = np.argsort(conv)[-m:]
        top_val = conv[top_idx]
        range_v = max_val - float(top_val[0])
        if range_v < 1e-10:
            weights = np.ones(m) / m
        else:
            temp = max(range_v * 0.4, 1e-6)
            weights = np.exp((top_val - max_val) / temp)
            weights /= float(np.sum(weights))

        # Weighted gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            if lo <= hi:
                seg_lo = ki - hi
                seg_hi = ki - lo
                grad[lo:hi + 1] += w * 2.0 * a[seg_lo:seg_hi + 1][::-1]

        # Momentum
        v = beta * v + (1.0 - beta) * grad

        # Update
        a_new = a - step * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            step = min(step * 1.02, 0.1)
        else:
            stall += 1
            if stall > 20:
                step *= 0.7
                stall = 0
                if step < 0.001:
                    # Perturb
                    rng = np.random.default_rng(t)
                    a = best_a * np.exp(rng.normal(0, 0.03, n))
                    a = project_simplex(a)
                    best_obj = compute_obj(a)
                    best_a = a.copy()
                    step = 0.03
                    v = np.zeros(n)

        # Periodically increase beta
        if t % 2000 == 1999:
            beta = min(beta + 0.01, 0.95)

    return best_a


def fft_flat_spectrum(a, n_iters=200, time_budget=30.0, relaxation=0.5):
    """FFT-based flat spectrum iteration (Gerchberg-Saxton variant for linear conv)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe

        # Target: flat magnitude
        A_flat = phase
        A_new = (1.0 - relaxation) * A + relaxation * A_flat
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    rng = np.random.default_rng(it)
                    scale = 0.05 * (1 + stall // 50)
                    a = best_a * np.exp(rng.normal(0, min(scale, 0.15), n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

    return best_a

def lp_alternating(a, time_budget=60.0, verbose=False):

        A[i:i + n, i] = a

    max_iters = 100
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP: min M, s.t. A@g <= M, sum(g)=1, g>=0, M>=0
        # Variables: [g_0, ..., g_{n-1}, M], size n+1
        try:
            from scipy.optimize import linprog
            c = np.zeros(n + 1)
            c[-1] = 1.0  # minimize M
            # Inequalities: A@g - M <= 0 for each row
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            # Equality: sum(g) = 1
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0

            bounds = [(0, None)] * n + [(0, None)]

            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:

                    best_t_obj = np.inf

                    for t in candidate_ts:
                        a_try = (1.0 - t) * a + t * g_val
                        a_try = np.maximum(a_try, 0)

                    if best_t_obj < best_obj - 1e-10:

                        best_a = best_a_local.copy()
                        a = best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = int(rng.integers(0, 5))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        freqs = rng.uniform(0, 4, 8)
        phases = rng.uniform(0, 2 * np.pi, 8)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # U-shaped (high at ends)
        x = np.linspace(-1, 1, n)
        a = x ** 2 + 0.3
        a = a / np.sum(a)
    else:
        a = rng.uniform(0, 1, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a


def coordinate_refinement(a, time_budget=30.0):
    """Simple 2-coordinate swaps targeting the peak lag."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = float(conv[k_star])
        improved = False

        # Try moving mass from high contributors to low ones
        step = max(1, n // 300)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 2 * a[j] + 1e-6:
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * (a[i] - a[j])
                    if delta < 1e-8:
                        break
                    a[i] -= delta
                    a[j] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            # Random 2-swap
            rng = np.random.default_rng(it)
            for _ in range(5):
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                if i1 == i2:
                    continue
                delta = 0.01 * max(a[i1], a[i2], 1e-6)
                a[i1] -= delta
                a[i2] += delta
                if a[i1] < 0:
                    a[i1] = 0
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i1] += delta
                    a[i2] -= delta
                    continue
                a = a / s_new
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 100:
            break

    return best_a

            alloc2 = min(remaining * 0.08, 30.0)
            a_opt = nesterov_smooth(a_opt, n_iters=100000,
                                    time_budget=alloc2, mu=50.0)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try

    # === Phase 7: Random restarts ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    n_restarts = min(8, int(remaining // 20))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        rtype = r % 4
        if rtype == 0:
            a_r = random_init(n_best, rng)
        elif rtype == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.1, n_best))
            a_r = project_simplex(a_r)
        elif rtype == 2:
            n_r = int(n_best + rng.integers(-200, 200))
            n_r = max(500, min(2000, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = project_simplex(a_r)
        else:
            # Anti-symmetric: boost ends
            a_r = best_seq.copy()
            mid = n_best // 2
            a_r[:mid] *= 1.2
            a_r[mid:] *= 1.2
            a_r[mid - n_best // 8:mid + n_best // 8] *= 0.6
            a_r = project_simplex(a_r)

        alloc_fft = min(10.0, remaining * 0.1)
        a_opt = fft_flat_spectrum(a_r, n_iters=150,
                                  time_budget=alloc_fft, relaxation=0.5)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc_ns = min(20.0, remaining * 0.3)
            a_opt = nesterov_smooth(a_opt, n_iters=50000,
                                    time_budget=alloc_ns, mu=50.0)

        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (type={rtype}, n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # Coordinate refinement
    if remaining > 15:
        alloc = min(remaining * 0.15, 30.0)
        a_cd = coordinate_refinement(best_seq, time_budget=alloc)
        obj = compute_obj(a_cd)
        print(f"  Coord refine: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()

    # One more Nesterov pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 40.0)
        a_ns = nesterov_smooth(best_seq, n_iters=200000,
                               time_budget=alloc, mu=100.0)
        obj = compute_obj(a_ns)
        print(f"  Final Nesterov: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_ns.copy()

    # Final LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 50.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc)
        obj = compute_obj(a_lp)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_sp = subgradient_multi_peak(best_seq, n_iters=100000,
                                      time_budget=alloc, m_peaks=3)
        obj = compute_obj(a_sp)
        print(f"  Final Subgrad: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_sp.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
