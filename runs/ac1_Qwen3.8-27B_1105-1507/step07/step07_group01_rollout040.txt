    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)

        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)

    # conv(a,g)[k] = sum_j a[k-j]*g[j] for valid j
    # = sum_j A[k,j] * g[j]

    for j in range(n):

        A[j:j+n, j] = a

    #      sum(g) = 1

    A_ub[:, n] = -1.0  # A@g - M <= 0
    b_ub = np.zeros(2*n-1)

        else:

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Compute a*b cross-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]

    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_descent_lp(a, time_budget, verbose=False):
    """
    Subgradient method for the LP subproblem.
    Given a, find g that minimizes max_k (a*g)[k] s.t. sum(g)=1, g>=0.

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Cross-convolution a*b via FFT."""
    n = len(a)

    while size < 2 * n - 1:

    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]

    n = len(a)

        return np.inf

    return 2.0 * n * float(np.max(conv))

    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)

    for j in range(n):

    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = np.sum(v)
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.

    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
