The optimal strategy might be: make a[i] alternate between large and small values, so that at the center of the convolution (where a[i] pairs with a[n-1-i]), large values pair with small values.

For an alternating pattern a = [A, B, A, B, A, B, ...] with A > B:
- At even positions k: a[i] and a[k-i] have the same parity, so products are A*A or B*B.
- At odd positions k: a[i] and a[k-i] have different parity, so products are A*B.

The peak will be at an even position (where A*A products accumulate). To reduce this, we want A to be not too large.

The optimal ratio A/B depends on n. For the optimization, we can parameterize a[i] = (1 + r*(-1)^i) / (1 + 1/n * sum of r*(-1)^j) or something like that.

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses FFT-based gradient descent with log-sum-exp soft-max approximation,
cosine-modulated initializations, and multi-scale exploration.
Key insight: auto-convolution peak can be reduced below the uniform level
by introducing oscillatory structure that causes destructive interference
at the convolution peak positions.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_gradient(w, a):
    """
    Compute g[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i=0,...,n-1.
    This gives the gradient of sum_k w[k]*conv[k] with respect to a.
    The full gradient of soft_max(conv(a,a)) is 2*g.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # w has length 2n-1
    # g[i] = sum_j a[j]*w[i+j]
    # This equals conv(a, w_rev)[2n-2-i] where w_rev = w[::-1]
    w_rev = w[::-1]
    size = 1
    while size < n + (2 * n - 1) - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_full = irfft(fa * fw, size)
    # g[i] = conv_full[2n-2-i] for i=0,...,n-1
    # So g = conv_full[2n-2, 2n-3, ..., n-1]
    g = conv_full[2 * n - 2 : n - 2 : -1]
    return g


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def soft_max_grad_descent(a, n, time_budget, beta_start=5.0, beta_end=100.0,
                          max_iters=2000, verbose=False):
    """
    Gradient descent on soft-max of auto-convolution.
    Minimizes the peak of conv(a,a) subject to a>=0, sum(a)=1.
    """
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_v = 0.0  # first moment
    v_v = 0.0  # second moment
    t = 0
    lr = 0.05

    # Precompute FFT size for gradient
    grad_size = 1
    while grad_size < 3 * n - 2:
        grad_size <<= 1

    stall_count = 0
    best_stall_obj = best_obj

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Anneal beta
        progress = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute convolution
        conv = fft_convolve(a, a)

        # Compute softmax weights (numerically stable)
        bc = beta * conv
        bc -= bc.max()
        exp_bc = np.exp(bc)
        w = exp_bc / exp_bc.sum()

        # Compute gradient: 2 * sum_k w[k] * d(conv[k])/da[i]
        # = 2 * g[i] where g[i] = sum_j w[i+j]*a[j]
        g = compute_gradient(w, a)
        grad = 2.0 * g

        # Center gradient (to approximately preserve sum)
        grad = grad - np.mean(grad)

        # Adam update
        t += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * (grad * grad)
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        # Evaluate with true max (not soft max) for bookkeeping
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            lr = min(lr * 1.02, 0.3)
        else:
            stall_count += 1
            if stall_count % 50 == 0:
                lr *= 0.7
            if stall_count > 200:
                # Perturbation to escape local minimum
                scale = 0.02 + 0.01 * min(stall_count // 200, 5)
                a_pert = best_a * np.exp(np.random.default_rng(it * 31 + 7).normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                    a = best_a.copy()
                stall_count = 0
                lr = 0.05

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      SGD it={it}, beta={beta:.1f}, obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a * np.sum(best_a)  # return unnormalized (sum = original sum)

    n = len(a)

def init_cosine_mod(n, frequency, amplitude, rng):
    """Initialize with cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amplitude * np.cos(2 * np.pi * frequency * x / n)
    # Add small random perturbation
    a = a + rng.normal(0, 0.01, n)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def init_alternating(n, ratio, rng):
    """Initialize with alternating pattern."""
    x = np.arange(n)
    a = 1.0 + (ratio - 1) * 0.5 * (1 + (-1) ** x)
    a = a + rng.normal(0, 0.01, n)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def init_multi_freq(n, rng, n_freqs=3):
    """Initialize with sum of multiple cosine frequencies."""
    x = np.arange(n)
    a = np.ones(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, max(2, n // 4))
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def init_gaussian_window(n, rng):
    """Gaussian window initialization."""
    center = n / 2.0
    width = n * rng.uniform(0.2, 0.5)
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = a + 0.05
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([a.sum()])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
