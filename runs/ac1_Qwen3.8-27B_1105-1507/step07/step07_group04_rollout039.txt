Spectral gradient + LP alternating optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smooth-max gradient, primal-dual game via MWU, LP refinement,
and multi-initialization strategy with adaptive time allocation.

import numpy as np
from scipy.fft import rfft, irfft
import time

    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Compute linear convolution of a with a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    c = compute_conv(a_norm)
    return 2.0 * n * float(np.max(c))


def smooth_max_gradient_fft(a, T):
    """
    Compute gradient of smooth max of conv(a,a) via FFT cross-correlation.
    Returns (gradient, convolution).
    """
    n = len(a)
    c = compute_conv(a)
    c_max = np.max(c)
    exp_Tc = np.exp(T * (c - c_max))
    w = exp_Tc / np.sum(exp_Tc)

    # Cross-correlation: (a * w)[j] = sum_k w[k] * a[k-j]
    N = 1
    while N < 3 * n:
        N <<= 1
    Fa = np.fft.rfft(a, N)
    Fw = np.fft.rfft(w, N)
    cross = np.fft.irfft(np.conj(Fa) * Fw, N)
    grad = 2.0 * cross[:n]
    return grad, c

def mwu_optimize(a, time_budget, top_k=10, verbose=False):
    """
    Multiplicative Weight Update optimization targeting top-K peak positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.05
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        if top_k >= 2 * n - 1:
            k_indices = np.arange(2 * n - 1)
        else:
            k_indices = np.argsort(conv)[::-1][:top_k]

        # Compute gradient (average of shifted/flipped a for each peak)
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        grad /= len(k_indices)
        grad = grad - np.mean(grad)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.05, 1.0)
        else:
            stall += 1
            eta *= 0.85
            if stall > 30:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.04, n))

                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                eta = 0.05

        if verbose and (time.monotonic() - start) % 10 < eta:
            print(f"    MWU: obj={best_obj:.8f}, eta={eta:.4f}, t={time.monotonic()-start:.0f}s")

def build_conv_matrix(a):
    """Build (2n-1 x n) matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget, verbose=False):
    """LP alternating with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            a_pert = a * np.exp(rng.normal(0, 0.01, n))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            g_lp, M_lp = solve_lp_minpeak(a_pert)
            if g_lp is None:
                break

        # Line search
        best_t_obj = np.inf
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

    return best_a


def coordinate_descent_polish(a, time_budget):
    """Pairwise mass transfer to reduce the peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        k_star = int(np.argmax(conv))
        improved = False

        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() - start > time_budget:
                break
            j = (k_star - i) % n
            if j == i or a[i] < 1e-10:
                continue
            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    break

        if not improved and it > 100:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def duality_gap_estimate(a, y):
    """Estimate the duality gap between primal and dual objectives."""
    n = len(a)
    a = a / np.sum(a)
    primal = float(np.max(compute_conv(a)))
    # Dual: min_j (a * y)[j]
    N = 1
    while N < 3 * n:
        N <<= 1
    Fa = np.fft.rfft(a, N)
    Fy = np.fft.rfft(y, N)
    cross = np.fft.irfft(np.conj(Fa) * Fy, N)
    dual = float(np.min(cross[:n]))
    return primal - dual


def dual_mwu(a, n_iters=200, eta=0.1):
    """
    Solve the dual: max min_j (a*y)[j] s.t. y>=0, sum(y)=1
    using multiplicative weight updates.
    """
    n = len(a)
    a = a / np.sum(a)
    y = np.ones(2 * n - 1) / (2 * n - 1)

    N = 1
    while N < 3 * n:
        N <<= 1
    Fa = np.fft.rfft(a, N)

    for _ in range(n_iters):
        Fy = np.fft.rfft(y, N)
        cross = np.fft.irfft(np.conj(Fa) * Fy, N)
        j_star = int(np.argmax(cross[:n]))  # Actually we want argmin for dual
        j_star = int(np.argmin(cross[:n]))

        # Update y: boost positions that cover j_star
        for k in range(2 * n - 1):
            idx = k - j_star
            if 0 <= idx < n:
                y[k] *= np.exp(eta * a[idx])
        y /= np.sum(y)

    return y
