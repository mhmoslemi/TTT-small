import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_all_active(a, epsilon=0.01):
    """Compute subgradient using all bins within epsilon of max."""
    n = len(a)
    conv = compute_conv(a)
    max_val = conv.max()
    # Find all active bins
    active_threshold = max_val * (1 - epsilon)
    active_idx = np.where(conv >= active_threshold)[0]
    
    g = np.zeros(n)
    for k in active_idx:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += a[j_start:j_end + 1][::-1]
    
    g /= len(active_idx)
    return g, max_val, len(active_idx)

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def smooth_random_init(n, rng, n_freqs=5):
    """Generate a smooth random initialization using random Fourier features."""
    freqs = rng.integers(1, n_freqs + 1, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    amps = rng.uniform(0.1, 1.0, size=n_freqs)
    x = np.arange(n) / n
    a = np.ones(n)
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.sin(2 * np.pi * f * x + p)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Algorithm:
1. Multiple initialization strategies (window functions, smooth random)
2. Adaptive subgradient method with Nesterov acceleration
   - Uses all "active" convolution peaks (within epsilon of max)
   - Adaptive epsilon: starts broad, narrows over time
   - Backtracking line search
3. Frank-Wolfe with LP oracle and proper step size
4. Multi-n exploration to find optimal dimension
5. Iterative polishing
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgradient(a, epsilon=0.01):
    """
    Compute subgradient of max_k conv(a,a)[k] using all active peaks.
    
    Active peaks are those within (1-epsilon)*max of the maximum.
    Returns (gradient_vector, max_value, number_active).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = conv.max()
    
    if max_val < 1e-15:
        return np.zeros(n), 0.0, 0
    
    threshold = max_val * (1.0 - epsilon)
    active_mask = conv >= threshold
    n_active = int(np.sum(active_mask))
    
    if n_active == 0:
        # Fallback: just use the max
        k_star = int(np.argmax(conv))
        g = np.zeros(n)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            j_start = k_star - i_end
            j_end = k_star - i_start
            g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
        return g, max_val, 1
    
    # Accumulate subgradient from all active peaks
    g = np.zeros(n)
    for k in np.where(active_mask)[0]:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += a[j_start:j_end + 1][::-1]
    
    g /= n_active
    return g, max_val, n_active


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None

    min_lr = 1e-5

    stall = 0
    patience = 200

        eps = eps_start + (eps_end - eps_start) * progress

        g, max_val, n_active = compute_subgradient(y, epsilon=eps)

        a_new = (1.0 - beta) * y_trial + beta * a

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        y = y_trial
        t = t_new
        lr = max(min_lr, lr0 / np.sqrt(max(1, it)))

        # Restart if stalled
        if stall > patience:
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            scale = 0.03 + 0.02 * (stall // patience)
            a = best_a * np.exp(rng.normal(0, min(0.5, scale), n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            best_a = a.copy()
            best_obj = compute_obj(a)
            y = a.copy()
            t = 1.0
            lr = lr0
            stall = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, eps={eps:.4f}, n_act={n_active}, t={elapsed:.1f}s")

        it += 1

    return best_a


def frank_wolfe_lp(a, time_budget=200.0, verbose=False):
    """
    Frank-Wolfe algorithm with LP oracle.
    Minimizes max_k conv(a,a)[k] over the simplex.
    Uses adaptive step size with backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_count = 0
    stall = 0
    last_gamma = 0.5

    while time.monotonic() - start < time_budget:
        # Get LP solution (vertex v)
        v, M_lp = solve_lp_minpeak(a)
        if v is None:
            break

        # Generate candidate step sizes
        t_count += 1
        gamma_fw = 2.0 / (t_count + 2.0)  # Standard FW step size
        candidates = [gamma_fw, last_gamma, last_gamma * 0.5, last_gamma * 1.5, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        candidates = sorted(set([max(0.01, min(1.0, g)) for g in candidates]))

        # Evaluate all candidates
        best_gamma = gamma_fw
        best_trial_obj = np.inf
        best_trial = a.copy()

        for g in candidates:
            trial = (1.0 - g) * a + g * v
            s_t = float(np.sum(trial))
            if s_t < 1e-10:
                continue
            trial = trial / s_t
            obj_t = compute_obj(trial)
            if obj_t < best_trial_obj:
                best_trial_obj = obj_t
                best_trial = trial
                best_gamma = g

        if best_trial_obj < best_obj - 1e-13:
            best_obj = best_trial_obj
            best_a = best_trial.copy()
            a = best_trial
            last_gamma = best_gamma
            stall = 0
        else:
            stall += 1
            if stall > 15:
                # Perturb and restart
                rng = np.random.default_rng((t_count * 104729 + n) % (2**31))
                scale = 0.03 + 0.01 * (stall // 15)
                a = best_a * np.exp(rng.normal(0, min(0.4, scale), n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                t_count = 0
                last_gamma = 0.5

        if verbose and t_count % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW it{t_count}: obj={best_obj:.8f}, gamma={best_gamma:.4f}, stall={stall}, t={elapsed:.1f}s")

    return best_a


def window_init(n, wtype):
    """Generate a window function initialization."""
    if n <= 1:
        return np.ones(max(1, n)) / max(1, n)
    x = np.arange(n)
    m = (n - 1) / 2.0
    
    if wtype == 'rect':
        a = np.ones(n)
    elif wtype == 'tri':
        a = 1.0 - np.abs(x - m) / (m + 1e-10)
    elif wtype == 'hann':
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
    elif wtype == 'hamming':
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
    elif wtype == 'blackman':
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
    elif wtype == 'blackman_harris':
        a = (0.35875 - 0.48829 * np.cos(2*np.pi*x/(n-1))
             + 0.14128 * np.cos(4*np.pi*x/(n-1))
             - 0.01168 * np.cos(6*np.pi*x/(n-1)))
    elif wtype == 'flat_top':
        a = (0.21557895 - 0.41663158 * np.cos(2*np.pi*x/(n-1))
             + 0.277263158 * np.cos(4*np.pi*x/(n-1))
             - 0.083578947 * np.cos(6*np.pi*x/(n-1))
             + 0.006947368 * np.cos(8*np.pi*x/(n-1)))
    elif wtype == 'gauss':
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - m) / sigma) ** 2)
    elif wtype == 'inv_gauss':
        sigma = n / 5.0
        a = 1.0 / np.sqrt(1.0 + ((x - m) / sigma) ** 4)
    elif wtype == 'bir_rect':
        # Bivariate rectangular (raised cosine in center, drop at edges)
        a = np.ones(n)
        edge = n // 10
        for i in range(edge):
            a[i] = 0.5 * (1 + np.cos(np.pi * i / edge))
            a[n-1-i] = a[i]
    elif wtype == 'kaiser':
        alpha = 5.0
        a = np.i0(np.sqrt(1 - ((2.0 * x / (n - 1) - 1.0)) ** 2)) / np.i0(alpha)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def smooth_random(n, rng, n_harmonics=5, min_val=0.01):
    """Generate a smooth random initialization using random Fourier series."""
    freqs = rng.integers(1, n_harmonics + 1, size=n_harmonics)
    phases = rng.uniform(0, 2 * np.pi, size=n_harmonics)
    amps = rng.uniform(0.05, 0.5, size=n_harmonics)
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.sin(2 * np.pi * f * x + p)
    a = np.maximum(a, min_val)
    a = a / a.sum()
    return a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([a.sum()])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_single(a, time_budget=60.0, use_lp=True, verbose=False):
    """Run a combination of optimizers on a single starting point."""
    n = len(a)
    best = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Step 1: Nesterov subgradient (fast, good for broad exploration)
    nsd_budget = min(time_budget * 0.4, 40.0)
    if nsd_budget > 5:
        a1 = nesterov_subgrad(best, time_budget=nsd_budget, verbose=verbose)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best = a1
            best_obj = obj1
            if verbose:
                print(f"      NSD: {best_obj:.8f}")

    # Step 2: Frank-Wolfe with LP (slower, good for refinement)
    remaining = time_budget - (time.monotonic() - start)
    if use_lp and remaining > 10:
        fw_budget = min(remaining * 0.8, 50.0)
        a2 = frank_wolfe_lp(best, time_budget=fw_budget, verbose=verbose)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best = a2
            best_obj = obj2
            if verbose:
                print(f"      FW: {best_obj:.8f}")

    return best, best_obj
