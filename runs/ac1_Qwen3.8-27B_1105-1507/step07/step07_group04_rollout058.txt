- So we want |A(f)| to be constant (or as close as possible)
- This is the "flat spectrum" or "equispectral" design

The approach:
1. Compute current spectrum A(f) = FFT(a)
2. Compute target spectrum: |A_target(f)| = sqrt(average of |A(f)|^2) for all f

4. Project to non-negative: a_new = max(a_new, 0)
5. Renormalize
6. Iterate

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def spectral_equalization(a, n_iter=100, target_flatness=1.0):
    """
    Gerchberg-Saxton style spectral equalization.
    Tries to make the power spectrum as flat as possible.
    """
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        # Compute spectrum
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A = rfft(a, size)
        power = np.abs(A) ** 2
        
        # Target: flat power spectrum (preserving DC)
        # The DC component is |sum(a)|^2 = 1 (since sum=1)
        # We want all other frequency components to have the same power
        # But we need to be careful: the total energy is fixed by Parseval
        # sum |A(f)|^2 = N * sum a^2 (by Parseval)
        
        # Simple approach: target power for non-DC bins
        # = average power of non-DC bins
        if size // 2 > 1:
            non_dc_power = np.mean(power[1:])
            target_power = np.ones(size // 2 + 1)
            target_power[0] = power[0]  # preserve DC
            target_power[1:] = non_dc_power
            
            # Adjust magnitude, preserve phase
            phase = np.angle(A)
            new_magnitude = np.sqrt(target_power)
            A_new = new_magnitude * np.exp(1j * phase)
            
            # Back to time domain
            a_new = irfft(A_new, size)[:n]
            
            # Project to non-negative
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))

                a = a_new

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, solver=None):
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    return None, None

def time_domain_windowing(a, n_iter=50):
    """
    Iteratively apply a window that suppresses time-domain samples where
    the autocorrelation would create peaks.
    """
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        
        # Compute "importance" of each time-domain sample
        # Sample i contributes to conv[k] for k = i, i+1, ..., i+n-1
        # with weight a[i]
        # We want to reduce samples that contribute most to the peak
        peak_idx = int(np.argmax(conv))
        
        # Weight each sample by its contribution to the peak
        weights = np.zeros(n)
        for i in range(n):
            # Contribution to conv[peak_idx] is a[i] * a[peak_idx - i] (if valid)
            ki = peak_idx - i
            if 0 <= ki < n:
                weights[i] = a[i] * a[ki]
        
        w_max = float(np.max(weights))
        if w_max > 1e-15:
            weights = weights / w_max
        
        # Reduce high-weight samples, increase low-weight samples
        alpha = 0.05
        a_new = a * (1.0 - alpha * weights)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                improved = False
                for t in [0.3, 0.5, 0.7, 0.9]:
                    a_blend = (1.0 - t) * a + t * a_new
                    a_blend = np.maximum(a_blend, 0)
                    sb = float(np.sum(a_blend))
                    if sb < 1e-10:
                        continue
                    a_blend = a_blend / sb
                    obj_b = compute_obj(a_blend)
                    if obj_b < best_obj - 1e-12:
                        best_obj = obj_b
                        best_a = a_blend.copy()
                        a = a_blend
                        improved = True
                        break
                if not improved:
                    stall += 1
                    if stall > 20:
                        # Perturb and reset
                        rng = np.random.default_rng(hash((it, n)) % (2**31))
                        a = best_a * np.exp(rng.normal(0, 0.03, n))
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s > 1e-10:
                            a = a / s
                            best_a = a.copy()
                            best_obj = compute_obj(a)
                        stall = 0

    return best_a


def spectral_equalization_dual(a, n_iter=100):
    """
    Dual spectral equalization: work in frequency domain to find phases
    that produce non-negative time-domain signals with flat magnitude.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(n_iter):
        A = rfft(a, size)
        power = np.abs(A) ** 2

        if size // 2 > 1:
            # Target: geometric mean of current power and flat target
            non_dc_power = np.mean(power[1:])
            dc_power = power[0]

            # Blend between current and flat
            alpha = 0.3 + 0.4 * (it / max(n_iter, 1))
            target_power = np.ones(size // 2 + 1)
            target_power[0] = dc_power
            target_power[1:] = (1 - alpha) * power[1:] + alpha * non_dc_power

            phase = np.angle(A)
            new_mag = np.sqrt(np.maximum(target_power, 0))
            A_new = new_mag * np.exp(1j * phase)

            a_new = irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

    return best_a


def active_set_gradient(a, time_budget=60.0, top_fractions=None):
    """
    Projected gradient descent focusing on the active set (near-max conv indices).
    Uses adaptive step size and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.7
    step_size = 0.03

    if top_fractions is None:
        top_fractions = [0.1, 0.2, 0.3, 0.5, 1.0]

    frac_idx = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        obj = 2.0 * n * max_b

        # Active set: top fraction of conv values
        frac = top_fractions[frac_idx % len(top_fractions)]
        n_active = max(1, int(frac * (2 * n - 1)))
        active = np.argsort(conv)[::-1][:n_active]

        # Compute gradient: d(max of active conv[k]) / d a[i]
        # For each active k: d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<n)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(min(n, k + 1)):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += a[ki]

        # Center gradient (maintain sum=1)
        grad = grad - np.mean(grad)
        g_norm = float(np.sqrt(np.sum(grad**2)))
        if g_norm < 1e-15:
            break

        # Momentum update
        momentum = beta * momentum - step_size * grad / g_norm
        new_a = a + momentum

        # Project to simplex
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            new_a = a.copy()
        else:
            new_a = new_a / s_new

        a = new_a
        obj_new = 2.0 * n * float(np.max(compute_conv(a)))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            # Slightly increase step
            step_size = min(step_size * 1.02, 0.1)
        else:
            no_improve += 1
            step_size = max(step_size * 0.9, 0.001)
            if no_improve > 30:
                # Switch fraction and perturb
                frac_idx += 1
                rng = np.random.default_rng(no_improve * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                momentum = np.zeros(n)
                no_improve = 0

    return best_a


def soft_thresholding_descent(a, time_budget=60.0):
    """
    Iteratively soft-reduce elements that contribute most to the peak convolution.
    Different from hard zeroing - uses a smooth penalty.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    penalty = 0.05
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        peak_idx = int(np.argmax(conv))

        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        for i in range(n):
            ki = peak_idx - i
            if 0 <= ki < n:
                contrib[i] = a[i] * a[ki]

        c_max = float(np.max(contrib))
        if c_max < 1e-15:
            break

        # Soft penalty on high-contribution elements
        # Reduce elements that are "on the peak"
        weights = contrib / c_max
        a_new = a * (1.0 - penalty * weights)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slowly increase penalty
            penalty = min(penalty * 1.05, 0.3)
        else:
            # Try smaller penalty
            penalty = max(penalty * 0.8, 0.01)
            if penalty < 0.01:
                # Try targeting second peak
                conv_sorted = np.sort(conv)[::-1]
                if len(conv_sorted) > 5 and conv_sorted[5] > 0.9 * max_b:
                    # Multiple peaks, target them all
                    threshold = 0.9 * max_b
                    peak_indices = np.where(conv >= threshold)[0][:5]
                    for pk in peak_indices:
                        for i in range(n):
                            ki = int(pk) - i
                            if 0 <= ki < n:
                                contrib[i] += a[i] * a[ki]
                    c_max2 = float(np.max(contrib))
                    if c_max2 > 1e-15:
                        weights2 = contrib / c_max2
                        a_try = a * (1.0 - 0.03 * weights2)
                        a_try = np.maximum(a_try, 0)
                        s2 = float(np.sum(a_try))
                        if s2 > 1e-10:
                            a_try = a_try / s2
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try
                                a = a_try
                                penalty = 0.05
                                break
                        else:
                            break
                    else:
                        break

    return best_a


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver=None, perturb=0.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    if perturb > 0:
        c = perturb * np.random.default_rng(42).random(n)
        obj = cp.Minimize(M + c @ g)
    else:
        obj = cp.Minimize(M)
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(obj, constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers

    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue

    return None


def solve_lp_for_a(g, solver=None):
    """
    Solve LP: min M s.t. conv(g,a)[k] <= M for all k, sum(a)=1, a>=0.
    """
    n = len(g)
    g = g / np.sum(g)
    A = build_conv_matrix(g)

    a = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ a <= M, cp.sum(a) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS]
    if solver:
        solvers = [solver] + solvers

    for s in solvers:
        try:
            prob.solve(solver=s)
            if a.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a.value, 0)
                as_ = float(np.sum(a_val))
                if as_ > 1e-10:
                    return a_val / as_
        except Exception:
            continue

    return None


def lp_alternating_diverse(a, time_budget=150.0, n_diverse=3, verbose=False):
    """
    LP alternating with diverse solutions: try multiple LP optima by
    adding small random perturbations to break ties.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        # Try multiple diverse LP solutions
        best_g = None
        best_M_lp = np.inf
        
        for d in range(n_diverse):
            perturb = 0.0 if d == 0 else 1e-4 * (d + 1)
            g_candidate = solve_lp_minpeak(a, perturb=perturb)
            if g_candidate is not None:
                # Check quality
                conv_test = compute_conv((a + g_candidate) / 2)
                M_est = float(np.max(conv_test))
                if M_est < best_M_lp:
                    best_M_lp = M_est
                    best_g = g_candidate

        if best_g is None:
            break

        # Also solve for a given g (dual LP)
        a_from_g = solve_lp_for_a(best_g)
        
        # Choose better candidate
        candidates = [best_g]
        if a_from_g is not None:
            candidates.append(a_from_g)

        # Line search over all candidates
        for cand in candidates:
            best_t_obj = best_obj
            best_a_local = a.copy()
            
            ts = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
            for t in ts:
                a_try = (1.0 - t) * a + t * cand
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_a_local = a_try

            if best_t_obj < best_obj - 1e-12:
                best_obj = best_t_obj
                best_a = best_a_local.copy()
                a = best_a
                stall = 0
                if verbose:
                    elapsed = time.monotonic() - start
                    print(f"    LP-diverse: obj={best_obj:.8f}, t_elapsed={elapsed:.0f}s")
                break
        else:
            stall += 1
            if stall > 8:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 8, 4)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multiply_by_cosine(a, freq):
    """Multiply by a cosine window - creates structure that can flatten autocorrelation."""
    n = len(a)
    t = np.arange(n) / n
    window = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
    a_new = a * window
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def fade_edges(a, fade_frac=0.1):
    """Apply a fade at the edges of the sequence."""
    n = len(a)
    k = max(1, int(fade_frac * n))
    a = a.copy()
    fade = np.linspace(0, 1, k)
    a[:k] *= fade
    a[n-k:] *= fade[::-1]
    return a

                t = np.arange(n_best) / n_best
                a_r = best_seq * (0.7 + 0.3 * np.sin(2 * np.pi * rng.integers(1, 8) * t))
            elif strategy == "anti_pal":
                # Anti-palindromic perturbation
                a_r = best_seq.copy()
                for i in range(0, n_best // 2):
                    j = n_best - 1 - i
                    if rng.random() > 0.5:
                        a_r[i], a_r[j] = a_r[j], a_r[i]
            elif strategy == "edge_fade":
                a_r = fade_edges(best_seq, fade_frac=0.05 + 0.03 * (r % 3))
            else:
                a_r = best_seq * np.exp(rng.normal(0, scale, n_best))

            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
            else:
                continue

            # Quick optimization
            a_opt = spectral_equalization(a_r, n_iter=15)
            remaining = deadline - time.monotonic()
            if remaining > 8:
                a_opt = lp_alternating_diverse(a_opt, time_budget=min(10.0, remaining * 0.3), n_diverse=1)
            
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r} ({strategy}): obj={best_score:.8f} **")

    # === Phase 9: Final Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 9: Final Polish ===")
        # One more round of LP on the best
        alloc = min(remaining * 0.4, 30.0)
        a_final = lp_alternating_diverse(best_seq, time_budget=alloc, n_diverse=2)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  FinalPolish: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_f2 = spectral_equalization(best_seq, n_iter=30)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  FinalSpec: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
