adaptive log-sum-exp smoothing, frequency-domain relaxation, coordinate descent,
and multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def grad_of_max_conv(a, top_k=5):
    """
    Compute approximate gradient of max of conv(a,a) w.r.t. a.
    Uses top-k lags with softmax weighting.
    Returns (smoothed_max_value, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    k = min(top_k, len(conv))
    # Get top k indices
    top_idx = np.argsort(conv)[-k:]
    top_vals = conv[top_idx]
    
    temperature = 0.5 / n
    c_scaled = top_vals / temperature
    c_max = float(np.max(c_scaled))
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / np.sum(exp_c)
    
    smooth_max = temperature * (c_max + np.log(np.sum(exp_c)))
    
    # Gradient: d conv[m] / d a[p] = 2 * a[m-p] if 0 <= m-p < n
    grad = np.zeros(n)
    for m, wi in zip(top_idx, w):
        m = int(m)
        # For p in range where 0 <= m-p < n:
        #   p ranges from max(0, m-n+1) to min(n-1, m)
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            # a[m-p] for p in [lo, hi] is a[m-hi : m-lo+1] reversed
            segment = a[m - hi: m - lo + 1][::-1]
            grad[lo:hi + 1] += wi * 2.0 * segment
    
    return smooth_max, grad


def nesterov_projected_gradient(a_init, time_budget=60.0, verbose=False):
    """
    Nesterov-accelerated projected gradient descent on max(conv(a,a)).
    Constraint: a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    v = np.zeros(n)
    gamma = 0.9  # momentum
    eta = 1.0 / (2.0 * n)  # initial step size
    
    top_k = min(10, n)
    
    while time.monotonic() - start < time_budget:
        # Nesterov look-ahead
        a_look = a - gamma * v
        a_look = np.maximum(a_look, 0)
        s_look = float(np.sum(a_look))
        if s_look < 1e-10:
            a_look = a.copy()
        else:
            a_look = a_look / s_look
        
        # Compute gradient at look-ahead point
        _, grad = grad_of_max_conv(a_look, top_k=top_k)
        
        # Gradient step
        a_new = a - eta * grad
        
        # Project: a >= 0, sum(a) = 1
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Backtracking line search
        obj_curr = compute_obj(a)
        obj_new = compute_obj(a_new)
        
        tries = 0
        while obj_new > obj_curr and tries < 10:
            eta *= 0.5
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
                break
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            tries += 1
        
        # Update Nesterov state
        a_next = a_new
        t_curr = (1 + np.sqrt(1 + 4 * gamma * gamma)) / 2
        gamma_new = (1 + t_curr - 1) / t_curr
        v = (a - a_next) * (1 + gamma) / (1 + gamma_new) + gamma_new * v
        a = a_next
        gamma = gamma_new
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
            if verbose and best_obj < 1.51:
                print(f"    NG: obj={best_obj:.10f}, eta={eta:.6f}")
    
    return best_a


def lbfgs_smoothed(a_init, time_budget=60.0, verbose=False):
    """
    L-BFGS-B optimization with log-sum-exp smoothing of the max.
    Works on log-parameterization for automatic non-negativity.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = np.log(a) - np.log(a).max()
    
    # Adaptive temperatures
    taus = [5.0/n, 2.0/n, 1.0/n, 0.5/n, 0.3/n, 0.2/n, 0.1/n, 0.05/n]
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        if time.monotonic() - start > time_budget * 0.8:
            break
        
        def fun(xv):
            xv = xv - np.max(xv)
            aa = np.exp(xv)
            ss = float(np.sum(aa))
            aa = aa / ss
            conv = compute_conv(aa)
            c_scaled = conv / tau
            c_max = float(np.max(c_scaled))
            return c_max + np.log(np.sum(np.exp(c_scaled - c_max)))
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            aa = np.exp(xv)
            ss = float(np.sum(aa))
            aa = aa / ss
            conv = compute_conv(aa)
            c_scaled = conv / tau
            c_max = float(np.max(c_scaled))
            exp_c = np.exp(c_scaled - c_max)
            w = exp_c / np.sum(exp_c)
            
            # Gradient w.r.t. xv_i = 2 * corr[i] / (tau * ss) - 2 * F * aa[i] / ss
            # where corr[i] = sum_k w[k] * aa[k-i] (with bounds)
            size = next_pow2(2 * n - 1)
            # corr[i] = sum_m w[m] * aa[m-i] for valid m-i
            # This is the cross-correlation of w and aa
            w_pad = np.zeros(size)
            w_pad[:len(w)] = w
            fa = rfft(aa, size)
            fw = rfft(w_pad, size)
            # corr[i] = sum_m w[m] * aa[m-i] = (w * reverse(aa))[i] in circular sense
            # Actually: sum_m w[m] * aa[m-i] for m-i in [0,n-1]
            # = sum_j w[j+i] * aa[j] where j = m-i
            # This is the cross-correlation
            aa_flip = np.zeros(size)
            aa_flip[:n] = aa[::-1]
            faf = rfft(aa_flip, size)
            cross = irfft(fw * faf, size)
            corr = cross[n - 1: n - 1 + n]
            
            F = c_max + np.log(np.sum(exp_c))
            grad = 2.0 * corr / (tau * ss) - 2.0 * F * aa / ss
            return grad
        
        # Run L-BFGS-B
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxls': 50})
        
        x_opt = res.x - res.x.max()
        a_new = np.exp(x_opt)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    LBFGS tau={tau/n:.4f}: obj={best_obj:.10f}")
    
    return best_a


def freq_domain_relax(a_init, time_budget=30.0, verbose=False):
    """
    Frequency-domain relaxation (Gerchberg-Saxton variant) for non-negative sequences.
    Tries to flatten the power spectrum while maintaining non-negativity.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = next_pow2(2 * n - 1)
    
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        # Forward transform
        A = rfft(a, N)
        power = np.abs(A) ** 2
        
        # Flatten: reduce peaks in power spectrum
        mean_power = float(np.mean(power))
        # Soft flattening
        alpha = 0.1 + 0.1 * (it / 100.0)
        power_flat = np.where(power > mean_power, 
                              mean_power + (power - mean_power) * (1 - alpha),
                              power)
        # Ensure non-negative
        power_flat = np.maximum(power_flat, 0)
        
        # New magnitude
        new_mag = np.sqrt(power_flat)
        # Keep phase
        if it % 20 == 0:
            # Occasionally randomize phase for exploration
            phase = np.angle(A)
            if it > 20:
                rng = np.random.default_rng(it)
                phase_noise = rng.normal(0, 0.05, len(phase))
                phase = phase + phase_noise
        else:
            phase = np.angle(A)
        
        # Inverse transform
        A_new = new_mag * np.exp(1j * phase)
        b = irfft(A_new, N)[:n]
        
        # Non-negativity projection
        b = np.maximum(b, 0)
        s_b = float(np.sum(b))
        if s_b < 1e-10:
            break
        b = b / s_b
        
        obj = compute_obj(b)
        if obj < best_obj:
            best_obj = obj
            best_a = b.copy()
            if verbose and best_obj < 1.51:
                print(f"    FREQ it={it}: obj={best_obj:.10f}")
        else:
            # Relaxation: blend with previous
            b = 0.7 * a + 0.3 * b
            b = np.maximum(b, 0)
            s_b = float(np.sum(b))
            if s_b > 1e-10:
                b = b / s_b
        
        a = b
    
    return best_a


def coordinate_descent(a_init, time_budget=30.0, verbose=False):
    """
    Coordinate descent: optimize one element at a time.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute: for each coordinate i, the "background" convolution
    # conv[m] = 2*a[i]*a[m-i] + c_bg[m] where c_bg doesn't involve a[i]
    
    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            # Background: conv without a[i]
            a_bg = a.copy()
            a_bg[i] = 0
            c_bg = compute_conv(a_bg)
            
            # conv[m] with a[i] = x: c_bg[m] + 2*x*a[m-i] (for valid m-i)
            # We want to minimize max_m (c_bg[m] + 2*x*a[m-i])
            # subject to x >= 0 and sum constraint (approximately: reduce others)
            
            # For simplicity, try a few candidate values for x
            current_x = a[i]
            candidates = [0.0, current_x * 0.5, current_x * 0.7, current_x * 1.3,
                         current_x * 1.5, current_x * 0.3, current_x * 2.0]
            
            best_x = current_x
            best_local_obj = best_obj
            
            for x in candidates:
                a_try = a.copy()
                a_try[i] = x
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_x = x
            
            if best_x != current_x:
                a[i] = best_x
                a = a / np.sum(a)
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True
                else:
                    # Revert
                    a[i] = current_x
                    a = a / np.sum(a)
        
        if not improved:
            break
    
    return best_a, best_obj


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.exponential(1.0, n)
    elif mode == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.5 * np.cos(x) + 0.3 * rng.normal(0, 1, n)
        a = np.maximum(a, 0)
    elif mode == 3:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = (1 + np.cos(x)) / 2 + 0.1 * rng.normal(0, 1, n)
        a = np.maximum(a, 0)
    elif mode == 4:
        # Gaussian bump
        center = n // 2
        sigma = n / (4 + rng.uniform(0, 4))
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        a += 0.05 * rng.uniform(0, 1, n)
    elif mode == 5:
        # Alternating high/low
        a = np.ones(n)
        a[::2] = 1.0 + rng.uniform(0, 1) * 0.5
        a[1::2] = 1.0 - rng.uniform(0, 1) * 0.4
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Pseudo-random with correlation breaking
        a = rng.uniform(0.5, 1.5, n)
        # Smooth slightly
        kernel = np.ones(3) / 3
        a = np.convolve(a, kernel, mode='same')
    else:
        # Multi-scale
        a = np.ones(n)
        for freq in [int(n / 7), int(n / 13), int(n / 31)]:
            if freq > 1:
                a += 0.3 * rng.uniform(0, 1) * np.sin(2 * np.pi * np.arange(n) / freq + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    frequency-domain relaxation, coordinate descent, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf

        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():

            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    n_parent = len(best_seq) if best_seq is not None else 800
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        n_parent = 800
        print(f"Default init (n=800): obj={best_score:.10f}")
    
    # === Phase 1: Nesterov projected gradient on parent ===
    print(f"\n=== Phase 1: Nesterov PG (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_ng = nesterov_projected_gradient(best_seq, time_budget=min(remaining * 0.15, 90.0), verbose=True)
        obj = compute_obj(a_ng)
        if obj < best_score:
            best_score = obj
            best_seq = a_ng.copy()
            print(f"  After Nesterov: obj={best_score:.10f}")
        else:
            print(f"  Nesterov: obj={obj:.10f} (no improve)")
    
    # === Phase 2: L-BFGS-B smoothed ===
    print(f"\n=== Phase 2: L-BFGS-B Smoothed (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_lb = lbfgs_smoothed(best_seq, time_budget=min(remaining * 0.2, 120.0), verbose=True)
        obj = compute_obj(a_lb)
        if obj < best_score:
            best_score = obj
            best_seq = a_lb.copy()
            print(f"  After LBFGS: obj={best_score:.10f}")
        else:
            print(f"  LBFGS: obj={obj:.10f} (no improve)")
    
    # === Phase 3: Frequency domain relaxation ===
    print(f"\n=== Phase 3: Frequency Domain Relaxation ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fd = freq_domain_relax(best_seq, time_budget=min(remaining * 0.1, 60.0), verbose=True)
        obj = compute_obj(a_fd)
        if obj < best_score:
            best_score = obj
            best_seq = a_fd.copy()
            print(f"  After FreqDomain: obj={best_score:.10f}")
        else:
            print(f"  FreqDomain: obj={obj:.10f} (no improve)")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    n_best_current = len(best_seq)
    n_candidates = []
    for dn in [-100, -50, 0, 50, 100, 200, 300, 400]:
        n_try = n_best_current + dn
        if 400 <= n_try <= 2000:
            n_candidates.append(n_try)
    # Add some absolute values
    for n_abs in [600, 700, 800, 900, 1000, 1200, 1500]:
        if 400 <= n_abs <= 2000 and n_abs not in n_candidates:
            n_candidates.append(n_abs)
    n_candidates = sorted(set(n_candidates))
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if n_try == n_best_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        # Quick Nesterov
        alloc = min(remaining * 0.08, 25.0)
        a_opt = nesterov_projected_gradient(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")
    
    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    n_restarts = max(4, int(remaining / 40))
    n_restarts = min(n_restarts, 8)
    restart_budget = min(30.0, remaining / (n_restarts + 1))
    
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < restart_budget + 10:
            break
        
        if r == 0:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            # Different structure
            a_r = diverse_init(n_best, rng)
        elif r == 2:
            # Symmetric
            a_r = best_seq.copy()
            a_r = (a_r + a_r[::-1]) / 2.0
            a_r += 0.02 * rng.normal(0, 1, n_best)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 3:
            # Different n
            n_r = max(500, min(1500, n_best + int(rng.integers(-200, 200))))
            a_r = resample_sequence(best_seq, n_r)
            a_r += 0.03 * rng.normal(0, 1, n_r)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            a_r = diverse_init(n_best, rng)
            # Blend with best
            a_r = 0.5 * a_r + 0.5 * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        
        # Optimize
        a_opt = nesterov_projected_gradient(a_r, time_budget=restart_budget * 0.5)
        a_opt = lbfgs_smoothed(a_opt, time_budget=restart_budget * 0.4)
        obj = compute_obj(a_opt)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (n={n_best}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")
    
    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd, obj_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.2, 40.0))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  After CoordDesc: obj={best_score:.10f}")
        else:
            print(f"  CoordDesc: obj={obj_cd:.10f}")
    
    # === Phase 7: Final L-BFGS-B polish with very low temperature ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = lbfgs_smoothed(best_seq, time_budget=min(remaining * 0.3, 60.0), verbose=True)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final LBFGS: obj={best_score:.10f}")
        
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_ng2 = nesterov_projected_gradient(best_seq, time_budget=min(remaining * 0.3, 30.0))
            obj = compute_obj(a_ng2)
            if obj < best_score:
                best_score = obj
                best_seq = a_ng2.copy()
                print(f"  Final Nesterov: obj={best_score:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
