Optimization for minimum peak self-convolution sequence.
Uses: subgradient descent (exact peak), LP alternating, peak shaving,
multi-n exploration, and diverse restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import linprog


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_mat(a):
    """A[m,j] such that (A@g)[m] = conv(a,g)[m]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[m, j] = a[m-j] for valid m-j
        # m-j in [0, n-1] => m in [j, j+n-1]
        m_lo = max(0, j)
        m_hi = min(2 * n - 2, j + n - 1)
        A[m_lo:m_hi + 1, j] = a[0:m_hi - m_lo + 1]
    return A


def solve_lp(a, time_budget=30.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_mat(a)
    
    # Try cvxpy solvers
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def subgradient_descent(a, time_budget=60.0, n_peaks=1, step_init=0.05, verbose=False):
    """
    Subgradient descent on the simplex using exact peak subgradient.
    g[j] = sum over peak positions m of a[m-j] (if valid)
    Step: a -> a - alpha*(g - mean(g)), projected to simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = step_init
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        
        # Get top n_peaks positions
        if n_peaks == 1:
            peak_pos = np.array([np.argmax(conv)])
        else:
            # Use threshold: positions within 90% of max
            thresh = max_val * 0.9
            peak_pos = np.where(conv >= thresh)[0]
            if len(peak_pos) > 20:
                # Limit to top 20
                peak_vals = conv[peak_pos]
                top_idx = np.argsort(-peak_vals)[:20]
                peak_pos = peak_pos[top_idx]
        
        # Compute subgradient
        g = np.zeros(n)
        for m in peak_pos:
            # g[j] += a[m-j] if 0 <= m-j < n
            j_lo = max(0, m - n + 1)
            j_hi = min(n - 1, m)
            if j_lo <= j_hi:
                g[j_lo:j_hi + 1] += a[m - np.arange(j_lo, j_hi + 1)]
        
        # Project gradient to tangent space
        g = g - np.mean(g)
        
        # Take step
        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            alpha = min(alpha * 1.02, 0.5)
        else:
            a = a_new  # Still move (to escape)
            stall_count += 1
            if stall_count > 20:
                alpha *= 0.7
                stall_count = 0
                # Reset to best
                a = best_a.copy()
            if verbose and stall_count % 50 == 0:
                print(f"    SGD: alpha={alpha:.4f}, obj={best_obj:.8f}")
    
    return best_a, best_obj

    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))

        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose and it % 5 == 0:
                print(f"        LP it{it}: obj={best_obj:.8f}")
        else:
            if it % 15 == 0:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_shave_direct(a, time_budget=30.0):
    """
    Direct peak shaving: find the peak, identify top contributors,
    and move their mass to positions that minimize the global peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        m_star = int(np.argmax(conv))
        max_val = conv[m_star]
        
        # Find top contributors to the peak
        j_lo = max(0, m_star - n + 1)
        j_hi = min(n - 1, m_star)
        js = np.arange(j_lo, j_hi + 1)
        contribs = a[js] * a[m_star - js]
        
        # Top 5 contributors
        top_idx = np.argsort(-contribs)[:5]
        
        improved = False
        for idx in top_idx:
            j = js[idx]
            partner = m_star - j
            
            # Try moving mass from max(j, partner) to a "safe" position
            src = j if a[j] >= a[partner] else partner
            if src == partner and a[j] > a[partner]:
                src = j
            
            # Actually, move from the one with more mass
            if a[j] >= a[partner]:
                src = j
            else:
                src = partner
            
            # Target positions: try positions far from m_star
            # A position t is "safe" if a[m_star - t] is small
            # (this way, moving mass to t won't create a new peak at m_star)
            target_scores = np.minimum(a, np.maximum(a[m_star - (np.arange(n) - m_star + n) % n, 0] if False else 0, 0))
            # Simpler: target is where a[m_star - t] is small
            # (so that adding mass at t doesn't increase C[m_star] much)
            mt = m_star - np.arange(n)
            # a[m_star - t] for t in [0, n-1]: valid when 0 <= m_star - t < n
            partner_vals = np.zeros(n)
            valid = (mt >= 0) & (mt < n)
            partner_vals[valid] = a[mt[valid]]
            
            # We want to minimize: new peak after the move
            # First-order: C[m_star] changes by delta * (partner_vals[t] - a[m_star - src])
            # So we want partner_vals[t] < a[m_star - src]
            # Also want the destination to not create new peaks
            
            # Score: low partner_vals (good for not increasing m_star peak)
            # Plus: low a[t] (good for not creating new peaks at other positions)
            score = partner_vals + 0.5 * a
            # Find the best target (lowest score, excluding src)
            score[src] = np.inf
            # Also exclude positions that are too close to src (to avoid trivial moves)
            target = int(np.argmin(score))
            
            # Try the move
            for frac in [0.02, 0.05, 0.1]:
                delta = frac * a[src]
                if delta <= 0 or delta > a[src] * 0.5:
                    continue
                a[src] -= delta
                a[target] += delta
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[src] += delta
                    a[target] -= delta
                    break
        
        if not improved:
            # Escape: perturb
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
    
    return best_a

    best_score = np.inf

        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        # Default: use a ramp-like initialization
        n0 = 800
        best_seq = np.linspace(0.1, 1.0, n0)
        best_seq = best_seq / np.sum(best_seq)
        best_score = compute_obj(best_seq)

    print("\n=== Phase 1: Subgradient Descent ===")

        remaining = deadline - time.monotonic()
