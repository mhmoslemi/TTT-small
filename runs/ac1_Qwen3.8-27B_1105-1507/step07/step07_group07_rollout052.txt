So `∂/∂a[i] (conv(a,a))[k] = a[k-i] + a[k-i] = 2*a[k-i]` (when 0 <= k-i < n and 0 <= i < n).

But if i = k-i (k = 2i), then a[i] appears only once in the sum (as a[i]*a[i]), and the derivative is 2*a[i].

Either way, `∂/∂a[i] (conv(a,a))[k] = 2*a[k-i]` (with the convention that a[j] = 0 for j outside [0, n-1]).

Wait, that's not quite right. If k = 2i, then the term is a[i]*a[i], and its derivative is 2*a[i]. And 2*a[k-i] = 2*a[i]. So it's consistent.

If i ≠ k-i, then a[i] appears in two terms: a[i]*a[k-i] (when j=i) and a[k-i]*a[i] (when j=k-i). The derivative of the first is a[k-i], and the derivative of the second is a[k-i]. Total: 2*a[k-i]. ✓

So the gradient of `(conv(a,a))[k]` with respect to `a` is the vector `g` where `g[i] = 2*a[k-i]` (with a[j] = 0 for j outside [0,n-1]).

This is just `2 * reverse_shift(a, k)`.

For the top-k subgradient, if the top-k lags are k_1, ..., k_k, the subgradient is any convex combination of `2*reverse_shift(a, k_j)`.

A natural choice is the uniform average: `g[i] = (2/k) * sum_j a[k_j - i]`.

The update would be: `a_new[i] = max(0, a[i] - eta * g[i])`, then renormalize.

"""

Uses top-k multiplicative weights, improved smoothed gradient (L-BFGS-B),
LP alternating with MOSEK/HiGHS, adaptive multi-n search, and spectral

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_and_topk(a, k=5):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, [0]
    a_norm = a / s
    conv = compute_conv(a_norm)
    obj = 2.0 * n * float(np.max(conv))
    topk_indices = np.argsort(conv)[::-1][:k]
    return obj, topk_indices.tolist()


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=10.0):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # Try cvxpy solvers
    for solver_name in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            start = time.monotonic()
            prob.solve(solver=solver_name, time_limit=timeout)
            if time.monotonic() - start > timeout * 1.5:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback: scipy linprog with HiGHS
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def smoothed_obj_and_grad(x, tau):
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))

    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)

    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]

    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def one_smoothed_eval(x, tau):
    val, _ = smoothed_obj_and_grad(x, tau)
    return val


def one_smoothed_grad(x, tau):
    _, g = smoothed_obj_and_grad(x, tau)
    return g


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    taus = [20.0/n, 10.0/n, 5.0/n, 2.5/n, 1.5/n, 1.0/n, 0.5/n, 0.25/n, 0.1/n, 0.05/n]
    x = np.log(a) - np.log(a).max()

    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        maxiter = max(50, int(remaining * 3))
        maxiter = min(maxiter, 500)

        res = minimize(one_smoothed_eval, x, method='L-BFGS-B',
                       jac=one_smoothed_grad if False else None,
                       args=(tau,),
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-12, 'maxls': 100})

        # Use proper jac
        def _fun(xv):
            return one_smoothed_eval(xv, tau)

        def _fgrad(xv):
            return one_smoothed_grad(xv, tau)

        # Re-run with proper gradient
        del res
        try:
            res = minimize(_fun, x, method='L-BFGS-B', jac=_fgrad,
                           options={'maxiter': maxiter, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'maxls': 100})
        except Exception:
            break

        if res.fun < one_smoothed_eval(x, tau):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={best_obj:.10f}")
            else:
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={obj_new:.10f} (no improve)")

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    mode = int(rng.integers(0, 8))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = rng.uniform(0, 1, n)
        kernel = np.ones(max(3, n // 50)) / max(3, n // 50)
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    elif mode == 3:
        a = 1.0 - np.abs(np.arange(n) - n / 2) / (n / 2)
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 4:
        c1, c2 = n // 3, 2 * n // 3
        sigma = n / 5.0
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * sigma ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * sigma ** 2))
        a = a / np.sum(a)
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2 + 0.2
        a = a / np.sum(a)
    elif mode == 6:
        # Exponential decay from center
        c = n // 2
        a = np.exp(-np.abs(np.arange(n) - c) / (n / 4))
        a = a / np.sum(a)
    else:
        # Flat with structured noise

        a = np.maximum(a, 0.01)

    return a

            a_try = a_try * np.exp(rng.normal(0, 0.02, len(a_try)))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)

        alloc = min(remaining * 0.12, 50.0)
        a_opt = combined_optimize(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.10f} **", flush=True)
        else:
            print(f"  n={n_try}: obj={obj:.10f}", flush=True)

    # === Phase 5: Intensify on best ===
    print(f"\n=== Phase 5: Intensify (n={len(best_seq)}) ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Long gradient
        t5a = min(remaining * 0.25, 100.0)
        a5 = gradient_optimize(best_seq, time_budget=t5a, verbose=True)
        obj = compute_obj(a5)
        if obj < best_score:
            best_score = obj
            best_seq = a5.copy()
            print(f"  Grad: obj={best_score:.10f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Long MW
        t5b = min(remaining * 0.2, 60.0)
        a5 = topk_multiplicative_weights(best_seq, time_budget=t5b, verbose=True)
        obj = compute_obj(a5)
        if obj < best_score:
            best_score = obj
            best_seq = a5.copy()
            print(f"  MW: obj={best_score:.10f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Long LP
        t5c = min(remaining * 0.25, 80.0)
        a5 = lp_alternate_optimize(best_seq, time_budget=t5c)
        obj = compute_obj(a5)
        if obj < best_score:
            best_score = obj
            best_seq = a5.copy()
            print(f"  LP: obj={best_score:.10f}", flush=True)

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===", flush=True)
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    restart_budget = min(40.0, remaining * 0.06)
    n_restarts = max(4, int(remaining / (restart_budget + 5)))
    n_restarts = min(n_restarts, 12)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < restart_budget + 5:
            break

        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
        elif r == 1:
            n_r = n_best + int(rng.integers(-80, 80))
            n_r = max(450, min(1300, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.02, len(a_r)))
        elif r == 2:
            # Symmetric perturbation
            a_r = best_seq + 0.02 * rng.normal(0, 1, n_best)
            a_r = (a_r + a_r[::-1]) / 2.0
        elif r == 3:
            # Cosine-based
            x = np.linspace(0, np.pi, n_best)
            a_r = np.cos(x / 2) ** 2 + 0.15 + 0.03 * rng.normal(0, 1, n_best)
        elif r == 4:
            # Dual peak
            c1, c2 = n_best // 4, 3 * n_best // 4
            sigma = n_best / 6.0
            a_r = np.exp(-((np.arange(n_best) - c1) ** 2) / (2 * sigma ** 2)) + \
                  np.exp(-((np.arange(n_best) - c2) ** 2) / (2 * sigma ** 2))
        else:
            a_r = random_init(n_best, rng)

        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        a_opt = combined_optimize(a_r, time_budget=restart_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.10f} **", flush=True)
        else:
            print(f"  Restart {r}: obj={obj:.10f}", flush=True)

    # === Phase 7: Final fine polish ===
    print(f"\n=== Phase 7: Final Polish ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t7 = min(remaining * 0.3, 45.0)
        a7 = gradient_optimize(best_seq, time_budget=t7, verbose=True)
        obj = compute_obj(a7)
        if obj < best_score:
            best_score = obj
            best_seq = a7.copy()
            print(f"  Final grad: obj={best_score:.10f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 15:
        t7 = min(remaining * 0.3, 30.0)
        a7 = topk_multiplicative_weights(best_seq, time_budget=t7)
        obj = compute_obj(a7)
        if obj < best_score:
            best_score = obj
            best_seq = a7.copy()
            print(f"  Final MW: obj={best_score:.10f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 15:
        t7 = min(remaining * 0.4, 40.0)
        a7 = lp_alternate_optimize(best_seq, time_budget=t7)
        obj = compute_obj(a7)
        if obj < best_score:
            best_score = obj
            best_seq = a7.copy()
            print(f"  Final LP: obj={best_score:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===", flush=True)

    return result
