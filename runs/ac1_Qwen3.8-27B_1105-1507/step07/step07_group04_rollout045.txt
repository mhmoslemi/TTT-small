"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses: (1) Cutting-plane LP alternating, (2) Spectral initialization, (3) Nesterov subgradient,
(4) Multi-n exploration, (5) Adaptive peak weighting.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective value 2*n*max(conv)/(sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_of_conv_at_k(a, k):
    """Compute d(a*a)[k]/da as a vector of length n."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            if j == i:
                grad[i] += a[i]
            else:
                grad[i] += a[j]
    return grad


def build_conv_row(a, k):
    """Build the k-th row of the convolution matrix A where A @ g = (a*g)."""
    n = len(a)
    row = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            row[i] = a[j]
    return row


def lp_cutting_plane(a, max_iters=50, tol=1e-8):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0
    using cutting plane method (only add violated constraints).
    Returns (g_normalized, M_val) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)

    # Start with top peaks as initial constraints
    conv_aa = compute_conv(a)
    top_k = min(10, 2 * n - 1)
    k_indices = set(np.argsort(conv_aa)[::-1][:top_k].tolist())

    g = None
    M_val = None

    for it in range(max_iters):
        k_list = sorted(k_indices)
        A_rows = np.array([build_conv_row(a, k) for k in k_list])

        # Solve LP: min M s.t. A_rows @ g <= M, sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A_rows @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for solver in [cp.PDLP, cp.SCS, cp.CLARABEL, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g = np.maximum(g_var.value, 0)
                    gs = float(np.sum(g))
                    if gs > 1e-10:
                        g = g / gs
                    M_val = float(M_var.value)
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            # Fallback: uniform g
            g = np.ones(n) / n
            conv_ag = compute_conv(a) * 0 + np.array([np.sum(a[i] * g[(k-i) % n] if 0 <= k-i < n else 0 for i in range(n)) for k in range(2*n-1)])
            M_val = float(np.max(np.array([float(np.sum([a[i]*g[j] for i in range(n) if 0 <= k-i < n and (j := k-i)])) for k in range(2*n-1)])))
            break

        # Check if all constraints are satisfied
        conv_ag = compute_conv_ag(a, g)
        violations = np.where(conv_ag > M_val + tol)[0]

        if len(violations) == 0:
            break

        # Add violated constraints (limit addition rate)
        worst_violations = violations[np.argsort(conv_ag[violations])[::-1]][:5]
        k_indices.update(worst_violations.tolist())

        # Limit total number of constraints
        if len(k_indices) > 200:
            conv_ag_full = compute_conv_ag(a, g)
            k_indices = set(np.argsort(conv_ag_full)[::-1][:200].tolist())

    if g is None:
        return None, None
    return g, M_val


def compute_conv_ag(a, g):
    """Compute convolution of a and g (different lengths possible, but here both length n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]

def spectral_init(n, mode=0, rng=None):
    """Create initial sequences with specific spectral shapes."""
    if rng is None:
        rng = np.random.default_rng(42)

    if mode == 0:
        # Cosine bump
        x = np.arange(n)
        a = 1.0 + np.cos(2 * np.pi * x / n)
    elif mode == 1:
        # Sharper cosine
        x = np.arange(n)
        a = (1.0 + np.cos(2 * np.pi * x / n)) ** 2
    elif mode == 2:
        # Sine squared (smooth)
        x = np.arange(n)
        a = np.sin(np.pi * x / n) ** 2 + 0.1
    elif mode == 3:
        # Raised cosine
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n)) ** 0.5
    elif mode == 4:
        # Gaussian-like
        x = np.arange(n)
        center = n / 2
        sigma = n / 4
        a = np.exp(-(x - center) ** 2 / (2 * sigma ** 2)) + 0.05
    elif mode == 5:
        # Flat with small perturbation
        a = np.ones(n)
        a += 0.1 * np.cos(2 * np.pi * 2 * x / n)
        a += 0.05 * np.cos(2 * np.pi * 3 * x / n)
    else:
        # Random smooth
        x = np.arange(n)
        a = np.ones(n)
        for freq in range(1, 5):
            a += rng.uniform(0.05, 0.2) * np.cos(2 * np.pi * freq * x / n + rng.uniform(0, 2*np.pi))

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def nesterov_subgradient(a, time_budget=60.0, top_k=10, verbose=False):
    """Nesterov-accelerated subgradient method for the min-max problem."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.7
    step_size = 0.05
    a_prev = a.copy()

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        # Compute peaks
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        k_indices = np.argsort(conv)[::-1][:top_k]

        # Compute weighted subgradient (weight by excess over median)
        vals = conv[k_indices]
        med = np.median(vals)
        weights = np.maximum(vals - med, 0.01)
        weights = weights / np.sum(weights)

        # Gradient
        grad = np.zeros(n)
        for w, k in zip(weights, k_indices):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        grad[i] += w * a[i]
                    else:
                        grad[i] += w * a[j]

        # Center gradient (project to sum=0)
        grad_centered = grad - np.mean(grad)

        # Nesterov update
        momentum = beta * momentum - step_size * grad_centered
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = a.copy()
            sn = 1.0
        a_new = a_new / sn

        a_prev = a.copy()
        a = a_new

        # Nesterov extrapolation
        t_k = 2.0 / (it + 2.0)
        x_extrap = a_new + (1 - t_k) * (a_new - a_prev)
        x_extrap = np.maximum(x_extrap, 0)
        sx = float(np.sum(x_extrap))
        if sx > 1e-10:
            x_extrap = x_extrap / sx
            # Evaluate at extrapolated point for gradient computation next iteration
            # (optional: we could use this, but for simplicity we use a_new)

        obj = compute_obj(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step size
        if it % 50 == 49:
            step_size = max(step_size * 0.97, 0.005)

        if verbose and it % 1000 == 0:
            print(f"    Nesterov: obj={best_obj:.8f}, it={it}" , end="\r")

    return best_a


def subgradient_simple(a, time_budget=60.0, top_k=5, verbose=False):
    """Simple subgradient descent with adaptive step size."""
    n = len(a)

def pair_swap_polish(a, time_budget=30.0):
    """Try pairwise mass transfers to reduce the peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find which indices contribute most to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Try moving mass from high-contributor to low-contributor
        top_idx = int(np.argmax(contrib))
        bottom_candidates = np.where(contrib < np.mean(contrib))[0]
        if len(bottom_candidates) == 0:
            bottom_idx = (top_idx + n // 2) % n
        else:
            bottom_idx = int(bottom_candidates[rng.integers(len(bottom_candidates))])

        for frac in [0.05, 0.1, 0.2]:
            delta = frac * a[top_idx]
            if delta >= a[top_idx]:
                continue
            a[top_idx] -= delta
            a[bottom_idx] += delta
            obj_try = compute_obj(a)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a.copy()
                break
            else:
                a[top_idx] += delta
                a[bottom_idx] -= delta

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    # === Phase 3: Nesterov subgradient ===
    print(f"\n=== Phase 3: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(100.0, remaining * 0.2)
        a_ng = nesterov_subgradient(best_seq, time_budget=alloc, top_k=10, verbose=True)
        obj_ng = compute_obj(a_ng)
        if obj_ng < best_score:
            best_score = obj_ng
            best_seq = a_ng.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ng:.8f}")

    # === Phase 4: Simple subgradient on best ===
    print(f"\n=== Phase 4: Simple Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.15)
        a_sg = subgradient_simple(best_seq, time_budget=alloc, top_k=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = []
    for delta in [-200, -100, -50, 0, 50, 100, 200, 300, 500]:
        n_opt = n_best + delta
        if 100 <= n_opt <= 3000:
            n_options.append(n_opt)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        if n_try == n_best:
            continue  # Already optimized

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(40.0, remaining * 0.2)
        a_opt = lp_alternating_improved(a_try, time_budget=alloc)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_final = lp_alternating_improved(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 7: Pair swap polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_cs = pair_swap_polish(best_seq, time_budget=alloc)
        obj_cs = compute_obj(a_cs)
        if obj_cs < best_score:
            best_score = obj_cs
            best_seq = a_cs.copy()
            print(f"  Pair swap: obj={best_score:.8f} **")
        else:
            print(f"  Pair swap: obj={obj_cs:.8f}")

    # === Phase 8: Final random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            scale = 0.03 + 0.015 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

            alloc = min(remaining * 0.3, 20.0)
            a_opt = lp_alternating_improved(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
