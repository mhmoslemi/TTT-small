"""
Enhanced optimizer for low-autocorrelation non-negative sequences.
Uses multi-peak multiplicative weight shaving, log-sum-exp smoothed gradients,
spectral perturbations, and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def multi_peak_shave(a, time_budget=100.0, top_k=5, eta_init=None, verbose=False, seed=42):
    """
    Multi-peak multiplicative weight shaving.
    Targets top-k peaks simultaneously with adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if eta_init is None:
        eta_init = 3.0 / n

    eta = eta_init
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - start < time_budget and stall < max_stall:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj = 2.0 * n * max_conv

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top-k peaks
        k = min(top_k, len(conv))
        peak_indices = np.argsort(conv)[::-1][:k]
        peak_vals = conv[peak_indices]

        # Compute weighted pressure
        total_pv = peak_vals.sum()
        if total_pv < 1e-15:
            break
        weights = peak_vals / total_pv

        pressure = np.zeros(n)
        for w, ki in zip(weights, peak_indices):
            ki = int(ki)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if hi >= lo:
                pressure[lo:hi + 1] += w * a[ki - hi:ki - lo + 1][::-1]

        # Center pressure to maintain approximate sum
        affected = pressure > 0
        if np.any(affected):
            mean_p = pressure[affected].mean()
            pressure[affected] -= mean_p * 0.5

        # Multiplicative update
        a_new = a * np.exp(-eta * pressure)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < obj - 1e-15:
            a = a_new
            eta = min(eta * 1.005, 10.0 / n)
        else:
            eta *= 0.85
            if eta < 0.005 / n:
                # Perturb and restart
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                eta = eta_init * 0.5
                stall = 0

        if verbose and stall % 100 == 0:
            print(f"    MPS: obj={best_obj:.8f}, eta={eta:.6f}, stall={stall}")

    return best_a


def smoothed_obj_and_grad(x, tau):
    """Compute smoothed objective and gradient for a = exp(x)."""
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))

    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = float(np.sum(exp_c))
    w = exp_c / sum_exp

    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)

    # Gradient via cross-correlation
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]

    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_polish(a_init, time_budget=60.0, verbose=False):
    """L-BFGS-B on log-sum-exp smoothed objective with adaptive temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    taus = [5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
    x = np.log(a) - np.log(a).max()

    for tau in taus:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val

        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g

        remaining = time_budget - elapsed
        max_iter = max(50, int(remaining * 30))

        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxls': 40})

        if res.x is not None:
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    Grad tau={tau:.4f}: obj={best_obj:.10f}")

    return best_a


def spectral_perturb(a, strength=0.3, rng=None):
    """Apply soft spectral equalization as perturbation."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    L = next_pow2(2 * n - 1)
    a_pad = np.zeros(L)
    a_pad[:n] = a
    A = rfft(a_pad)
    mag = np.abs(A)
    phase = np.angle(A)
    target_mag = np.full_like(mag, float(np.mean(mag)))
    mag_safe = np.maximum(mag, 1e-10)

    A_new = new_mag * np.exp(1j * phase)

    a_new = np.maximum(a_new, 1e-15)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def u_shaped_init(n, alpha=1.0):
    """U-shaped (concave) initialization with more mass at edges."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def cosine_init(n, power=2):
    """Cosine window initialization."""
    x = np.linspace(0, np.pi, n)
    a = np.abs(np.cos(x)) ** power + 0.1
    return a / np.sum(a)


def random_diverse_init(n, rng):

    else:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)

    print(f"\n=== Phase 4: Spectral Perturbation ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for trial in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            strength = 0.2 + 0.1 * trial
            a_spec = spectral_perturb(best_seq, strength=strength, rng=rng)
            # Quick optimize
            alloc = min(remaining * 0.1, 40.0)
            a_opt = multi_peak_shave(a_spec, time_budget=alloc, top_k=5, verbose=False, seed=seed + 10 + trial)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Spec trial {trial} (s={strength:.1f}): obj={best_score:.10f} **")
            else:
                print(f"  Spec trial {trial} (s={strength:.1f}): obj={obj:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_parent = len(best_seq)
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200, n_parent + 300,
        n_parent - 200, n_parent + 400,
        600, 700, 800, 900, 1000, 1100, 1200, 1300
    ]))
    n_candidates = [x for x in n_candidates if 300 <= x <= 2000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick MPS
        alloc = min(remaining * 0.12, 50.0)
        a_opt = multi_peak_shave(a_try, time_budget=alloc, top_k=5, verbose=False, seed=seed + n_try)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 6: Intensify on best ===
    print(f"\n=== Phase 6: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # MPS with various top_k
        for k in [3, 5, 8, 12]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.1, 40.0)
            a_opt = multi_peak_shave(best_seq, time_budget=alloc, top_k=k, verbose=False, seed=seed + 100 + k)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  MPS k={k}: obj={best_score:.10f} **")

        # Gradient polish
        remaining = datetime_remaining(deadline)
        if remaining > 30:
            alloc = min(remaining * 0.15, 60.0)
            a_grad = gradient_polish(best_seq, time_budget=alloc, verbose=False)
            obj = compute_obj(a_grad)
            if obj < best_score:
                best_score = obj
                best_seq = a_grad.copy()
                print(f"  Final grad: obj={best_score:.10f} **")

    # === Phase 7: Diverse initializations ===
    print(f"\n=== Phase 7: Diverse Inits ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            if r == 0:
                a_r = u_shaped_init(n_best, alpha=rng.uniform(0.5, 2.0))
            elif r == 1:
                a_r = cosine_init(n_best, power=int(rng.integers(1, 6)))
            elif r == 2:
                a_r = random_diverse_init(n_best, rng)
            elif r == 3:
                # Perturbed best
                a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
                a_r = np.maximum(a_r, 1e-15)
                a_r = a_r / np.sum(a_r)
            elif r == 4:
                # Different n
                n_r = n_best + int(rng.integers(-200, 200))
                n_r = max(400, min(1500, n_r))
                a_r = resample_sequence(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.03, len(a_r)))
                a_r = np.maximum(a_r, 1e-15)
                a_r = a_r / np.sum(a_r)
            else:
                a_r = random_diverse_init(n_best, rng)

            alloc = min(remaining * 0.08, 30.0)
            a_opt = multi_peak_shave(a_r, time_budget=alloc, top_k=5, verbose=False, seed=seed + 200 + r)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                n_best = len(best_seq)
                print(f"  Diverse {r} (n={len(a_opt)}): obj={best_score:.10f} **")
            else:
                print(f"  Diverse {r} (n={len(a_opt)}): obj={obj:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_final = multi_peak_shave(best_seq, time_budget=alloc, top_k=5,
                                    eta_init=2.0 / len(best_seq), verbose=True, seed=seed + 500)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final MPS: obj={best_score:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 50.0)
            a_final2 = gradient_polish(best_seq, time_budget=alloc, verbose=False)
            obj = compute_obj(a_final2)
            if obj < best_score:
                best_score = obj
                best_seq = a_final2.copy()
                print(f"  Final Grad: obj={best_score:.10f} **")

        # One more MPS pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 30.0)
            a_final3 = multi_peak_shave(best_seq, time_budget=alloc, top_k=8,
                                         verbose=False, seed=seed + 600)
            obj = compute_obj(a_final3)
            if obj < best_score:
                best_score = obj
                best_seq = a_final3.copy()
                print(f"  Final MPS2: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result


def datetime_remaining(deadline):
    """Return remaining time until deadline."""
    return max(0.0, deadline - time.monotonic())
