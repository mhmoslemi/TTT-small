import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv_fft(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2*n-1]

def compute_conv_cross(a, b):
    """Compute linear convolution a*b via FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:2*n-1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv_fft(a_norm)
    return 2.0 * n * np.max(b)

def solve_lp_conv(a, M_init=None):
    """
    Solve LP: min M s.t. conv(a, g)_k <= M, sum(g)=1, g>=0.
    Returns (g, M).
    """
    n = len(a)
    # Build A matrix: A[k, i] = a[k-i] if 0 <= k-i < n
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # A[k, i] = a[k-i] for k = i, i+1, ..., i+n-1 (but capped at 2n-2)
        k_max = min(2*n-1, 2*i+n)
        for k in range(i, k_max):
            A[k, i] = a[k-i]
    
    # LP: min M
    # s.t. A @ g <= M  →  A @ g - M * 1 <= 0
    #      sum(g) = 1
    #      g >= 0, M >= 0

    # Variables: [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1]
    # A_ub = [A, -1_{2n-1}]
    # b_ub = 0
    # A_eq = [1_n^T, 0]
    # b_eq = [1]
    
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])

    if res.status == 0:

        M = max(res.x[n], 0)

        conv_ag = compute_conv_cross(a, g)
        M = np.max(conv_ag)
        return g, M

LP-based alternating optimization with acceleration for minimizing

acceleration, multi-n exploration, and gradient-based refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv_fft(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)

    """Compute linear convolution a*b via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:2 * n - 1]

    # LP: min M
    # s.t. A @ g - M <= 0  (2n-1 constraints)
    #      1^T @ g = 1
    #      g >= 0
    # Variables: x = [g_0, ..., g_{n-1}, M]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.column_stack([A, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g))
        if s_g > 1e-10:
            return g / s_g
    return None

def fft_gradient_descent(a, time_budget=60.0, verbose=False):
    """FFT-based gradient descent with adaptive parameters."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    stall = 0
    max_stall = 150

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Gradient of max(b) approximated by softmax-weighted
        b_shifted = b - np.max(b)
        w = np.exp(b_shifted)
        w_sum = np.sum(w)
        if w_sum > 1e-30:
            w = w / w_sum

        # c[i] = sum_m a[m] * w[m+i]
        a_flip = a[::-1]
        faf = rfft(a_flip, N)
        fw = rfft(w, N)
        N2 = 1
        while N2 < 2 * N - 1:
            N2 <<= 1
        # Actually, we need conv(a_flip, w) which has length n + (2n-1) - 1 = 3n-2
        # Let's use a different approach
        # c[i] = sum_j a[j] * w[i+j] for i=0..n-1
        # This is cross-correlation: c = correlate(a, w, mode='valid') approximately
        # Better: c[i] = (a_flip * w)[n-1+i] where * is conv
        # conv(a_flip, w)[k] = sum_j a_flip[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
        # Let m = n-1-j, so j = n-1-m
        # = sum_m a[m] * w[k-n+1+m]
        # We want c[i] = sum_m a[m] * w[i+m]
        # So we need k-n+1 = i, i.e., k = i+n-1
        # c[i] = conv(a_flip, w)[i+n-1]
        
        Nc = 1
        while Nc < n + (2*n-1) - 1:
            Nc <<= 1
        faf2 = rfft(a_flip, Nc)
        fw2 = rfft(w, Nc)
        c_full = irfft(faf2 * fw2, Nc)[:n + 2*n-2]
        c = c_full[n-1:2*n-1]

        c = c - np.mean(c)

        v = momentum * v - eta * c
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 3.0)
        else:
            stall += 1
            eta *= 0.97
            v *= 0.5
            if stall > max_stall:
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.7
                stall = 0
            if eta < 1e-10:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
LP-based alternating optimization with momentum acceleration for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses structured LP solving, adaptive blending,
multi-n exploration, and FFT-based gradient refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv_fft(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Compute linear convolution a*b via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None).
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a = a / s

    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.column_stack([A, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                      options={'presolve': True, 'time_limit': 20})
    except Exception:
        return None, None

        g = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g))
        if s_g > 1e-10:
            g = g / s_g
            conv_ag = compute_conv_cross(a, g)
            M = float(np.max(conv_ag))
            return g, M
    return None, None

            if stall > 150:
                break

    return best_a


def fft_gradient_descent(a, time_budget=60.0, verbose=False):
    """FFT-based gradient descent with softmax-weighted peak focusing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    Nc = 1
    while Nc < 3 * n - 1:
        Nc <<= 1

    a_flip = np.zeros(Nc)
    a_flip[:n] = a[::-1]

    stall = 0
    max_stall = 200

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        b_shifted = b - np.max(b)
        w = np.exp(b_shifted)
        w_sum = np.sum(w)
        if w_sum > 1e-30:
            w = w / w_sum

        fw = rfft(w, Nc)
        c_full = irfft(fa[:Nc] * 0 + rfft(a_flip, Nc) * fw, Nc)[:3 * n - 2]
        c = c_full[n - 1:2 * n - 1]

        c = c - np.mean(c)

        v = momentum * v - eta * c
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.008, 4.0)
            momentum = min(momentum + 0.0005, 0.92)
        else:
            stall += 1
            eta *= 0.97
            v *= 0.5
            if stall > max_stall:
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.7
                stall = 0
                max_stall = min(max_stall + 50, 400)
            if eta < 1e-10:
                break

    return best_a


def projected_subgradient(a, time_budget=30.0):
    """Projected subgradient method focusing on peak positions."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    eta = 0.02
    avg_a = np.zeros(n)
    num_avg = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        grad = grad - np.mean(grad)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        avg_a += a_new
        num_avg += 1
        a_avg = avg_a / num_avg

        fa_avg = rfft(a_avg, N)
        b_avg = irfft(fa_avg * fa_avg, N)[:2 * n - 1]
        obj_avg = 2.0 * n * float(np.max(b_avg))

        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()

        a = a_new
        eta = max(eta * 0.999, 0.001)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = int(rng.integers(0, 7))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    else:
        a = np.ones(n) / n
        mask = rng.random(n) < 0.5
        a[~mask] *= 0.1
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with acceleration + FFT GD + multi-n exploration.
    Strategy: LP is the primary workhorse (exploits convex structure),
    FFT GD provides smooth refinement, multi-n search finds optimal dimension.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization from n={n_best}, obj={best_score:.8f}")

    # Phase 1: Intensive LP Alternating (primary method)
    print(f"\n=== Phase 1: LP Alternating (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.35, 400.0)
    print(f"  Time budget: {alloc:.0f}s")
    a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
    obj_lp = compute_obj(a_lp)
    if obj_lp < best_score - 1e-12:
        best_score = obj_lp
        best_seq = a_lp.copy()
        print(f"  ** LP: obj={best_score:.8f} **")
    else:
        print(f"  LP: obj={obj_lp:.8f}")

    # Phase 2: FFT GD refinement (smooths out the LP solution)
    print(f"\n=== Phase 2: FFT GD Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.12, 100.0)
        a_gd = fft_gradient_descent(best_seq, time_budget=alloc, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score - 1e-12:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  ** GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # Phase 3: Second LP round (from improved point)
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.25, 200.0)
        a_lp2 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score - 1e-12:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.8f}")

    # Phase 4: Projected subgradient (different angle of attack)
    print(f"\n=== Phase 4: Projected Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 40.0)
        a_pg = projected_subgradient(best_seq, time_budget=alloc)
        obj_pg = compute_obj(a_pg)
        if obj_pg < best_score - 1e-12:
            best_score = obj_pg
            best_seq = a_pg.copy()
            print(f"  ** Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_pg:.8f}")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr + 200, n_curr + 400, n_curr + 600,
        n_curr - 200, n_curr - 400,
        600, 900, 1200, 1600
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and n != n_curr]
    n_options = n_options[:6]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        print(f"  Trying n={n_try}...")
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.22, 120.0)
        a_opt = lp_alternating(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_curr = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 6: Random restarts with LP
    print(f"\n=== Phase 6: Random Restarts ===")
    n_curr = len(best_seq)
    n_restarts = max(2, int((deadline - time.monotonic()) / 60.0))
    n_restarts = min(n_restarts, 8)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        a_r = random_init(n_curr, rng)
        alloc = min(remaining * 0.2, 80.0)
        a_opt = lp_alternating(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score - 1e-12:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # Phase 7: Final combined polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.35, 150.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-12:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 80.0)
        a_final_gd = fft_gradient_descent(best_seq, time_budget=alloc)
        obj_final_gd = compute_obj(a_final_gd)
        if obj_final_gd < best_score - 1e-12:
            best_score = obj_final_gd
            best_seq = a_final_gd.copy()
            print(f"  ** Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
