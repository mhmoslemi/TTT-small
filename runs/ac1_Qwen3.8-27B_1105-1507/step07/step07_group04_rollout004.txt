    # You know what, let me just use the direct formula with FFT:
    # g_m = sum_j a_j * w_{j+m} for m = 0, 1, ..., n-1
    # 
    # Define: A = a padded to length L (L >= 2n-1 + n = 3n-1 or so)
    #         W = w padded to length L
    # Then g = IFFT(FFT(A) * conj(FFT(W))) ... 
    #
    # Actually the standard cross-correlation formula:
    # (a ⋆ w)[m] = sum_n a[n] * w*[n+m] = IFFT(FFT(a) * conj(FFT(w)))[m]
    # For real signals: (a ⋆ w)[m] = IFFT(FFT(a) * FFT(w))[m] with conjugate... 
    #
    # I'm overcomplicating this. Let me just use np.correlate for correctness first,
    # then optimize with FFT if needed. For n=800, np.correlate should be fine.
    
    # Actually, the simplest correct approach:
    # g_m = sum_j a_j * w_{j+m} for m in [0, n-1]
    # This is the first n elements of the cross-correlation of a with w.
    # cross_corr(a, w) = np.correlate(w, a, mode='full') -- gives (w ⋆ a)[m] = sum_n w[n] * a[n+m]
    # Hmm, np.correlate(a, v) computes sum_n a[n+k] * v[n]
    # So np.correlate(w, a, mode='full')[n-1+m] = sum_j w[j+m] * a[j] for the right offsets
    
    # Let me just verify: np.correlate(w, a, 'full') has length 2n-1 + n-1 = 3n-2
    # Element [n-1+m] for m=0,...,n-1 gives sum_j w[(n-1+m)-j] ... no this is getting confusing.
    
    # Let me just do it directly with a loop for the gradient (it's O(n^2) per iteration
    # but n=800 so 640K operations, which is fast enough for a few thousand iterations).
    
    # Actually no, let me do it properly with FFT.
    # 
    # We want: g_m = sum_{j=0}^{n-1} a_j * w_{j+m} for m = 0, 1, ..., n-1
    #
    # Define the "reversed" w: wr[k] = w[L-1-k] where L = 2n-1 (length of w)
    # Then: sum_j a_j * w_{j+m} = sum_j a_j * wr[L-1-j-m]
    # Let i = L-1-m-j, so j = L-1-m-i:
    # = sum_i a_{L-1-m-i} * wr[i]
    # This is the (L-1-m)-th element of conv(a, wr)
    # 
    # conv(a, wr)[k] = sum_i a_i * wr[k-i]
    # So sum_i a_{L-1-m-i} * wr[i] -- this isn't quite conv(a,wr) in the standard form.
    
    # OK I think the cleanest approach for this specific problem is:
    # g_m = sum_j a_j * w_{j+m}
    # Let's define: extend a to length (2n-1) by zero-padding: a_ext[0..n-1] = a, a_ext[n..2n-2] = 0
    # Then g_m = sum_{j=0}^{2n-2} a_ext[j] * w_{j+m} (since a_ext[j]=0 for j>=n)
    # For m >= 0: w_{j+m} is nonzero for j+m in [0, 2n-2], i.e., j in [-m, 2n-2-m]
    #             g_m = sum_{j=0}^{min(n-1, 2n-2-m)} a[j] * w[j+m]
    
    # This is the cross-correlation of a_ext with w at lag -m (or equivalently, the m-th
    # "sliding dot product" of a with the shifted w).
    
    # The FFT trick: cross-correlation of x and h at lag m is:
    # (x ⋆ h)[m] = sum_n x[n] * h[n+m] = IFFT{X(f) * conj(H(f))}[m]  ... no
    # Actually: (x ⋆ h)[m] = sum_n x[n+m] * h[n] = IFFT{conj(X(f)) * H(f)} ... I keep getting confused.
    
    # Let me just use the direct computation. For n=800, the gradient computation
    # is sum over m=0..n-1 of sum over j=0..n-1, which is O(n^2) = 640K flops.
    # With numpy vectorization this is very fast (< 1ms).
    
    # g_m = sum_j a_j * w_{j+m}
    # We can write this as: for each j, a_j * w[j:j+n] contributes to g[0:n]
    # So: g = sum_{j=0}^{n-1} a_j * w[j:j+n]
    # This is a sum of shifted vectors weighted by a_j. We can do this with a matrix
    # or just a loop (which is fine for n=800).
    
    # Actually, the most efficient numpy approach:
    # Create a matrix M where M[m, j] = w[j+m] for valid (m,j)
    # Then g = M @ a
    # But building M is O(n^2) memory. For n=800 that's 640K floats = 5MB, fine.
    
    # Or even better: use the fact that this is a convolution!
    # g_m = sum_j a_j * w_{j+m} = (a * w_rev)_m where w_rev is appropriately defined
    
    # Let me define: w_full = w (length 2n-1)
    # a_full = a (length n)
    # g_m = sum_j a_full[j] * w_full[j+m] for m=0,...,n-1
    #
    # Consider: flip_a[j] = a_full[n-1-j] (length n)
    # Then: conv(flip_a, w_full)[n-1+m] = sum_j flip_a[j] * w_full[n-1+m-j]
    #   = sum_j a_full[n-1-j] * w_full[n-1+m-j]
    #   Let i = n-1-j: sum_i a_full[i] * w_full[m+i]
    #   = g_m!
    #
    # So g = conv(a[::-1], w)[n-1 : 2*n-1]
    # i.e., the middle portion of the convolution.
    
    # Let me verify: conv(a_rev, w)[k] = sum_j a_rev[j] * w[k-j]
    # a_rev[j] = a[n-1-j]
    # conv(a_rev, w)[n-1+m] = sum_j a[n-1-j] * w[n-1+m-j]
    # Let i = n-1-j (so j = n-1-i):
    # = sum_i a[i] * w[m+i]
    # = g_m ✓
    
    # So: g = convolve(a[::-1], w)[n-1 : 2*n-1]
    # Length of conv(a_rev, w) = n + (2n-1) - 1 = 3n-2
    # We want elements n-1 through 2n-2 (inclusive), which is n elements. ✓
    
    a_rev = a[::-1]
    # Compute conv(a_rev, w) using FFT
    conv_len = len(a_rev) + len(w) - 1
    size = 1
    while size < conv_len:
        size <<= 1
    fa = rfft(a_rev, size)
    fw = rfft(w, size)
    conv_result = irfft(fa * fw, size)[:conv_len]
    g = conv_result[n - 1: 2 * n - 1]
    
    return 4.0 * n * g, conv


def project_simplex(a):
    """Project a onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(a)
    # Sort in decreasing order
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)


def softmax_gradient_descent(a, time_budget=200.0, tau_init=50.0, tau_max=500.0, 
                              verbose=False):
    """
    Nesterov-accelerated softmax gradient descent on the simplex.
    Minimizes 2n * max(conv(a,a)) where a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_init
    momentum = np.zeros(n)
    beta = 0.9  # momentum coefficient
    
    # Adaptive step size
    step_size = 0.1 / np.sqrt(n)
    step_min = 1e-6
    step_max = 0.5
    
    # Nesterov
    y = a.copy()  # lookahead point
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Compute gradient at lookahead point y
        grad, conv_y = compute_gradient(y, tau)
        
        # Center the gradient (important for simplex projection)
        # The gradient already incorporates the 4n factor
        # We want to minimize 2n*max(conv) = 2n/tau * log(sum exp(tau*conv))
        # The gradient of this is 4n * (cross_corr) as computed above
        # But we need to account for the constraint sum(a) = 1
        # The projection handles this, but centering helps convergence
        
        # Nesterov update
        momentum = beta * momentum - step_size * grad
        a_new = a + momentum
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Nesterov lookahead
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1 - t_k) * (a_new - a)
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y /= sy
        
        a = a_new
        it += 1
        
        # Evaluate periodically
        if it % 5 == 0:
            obj = compute_obj(a)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                # Increase step size slightly on improvement
                step_size = min(step_size * 1.05, step_max)
            else:
                no_improve += 1
                if no_improve > 50:
                    # Decrease step size
                    step_size = max(step_size * 0.8, step_min)
                    no_improve = 0
                    # Increase tau to focus more on the peak
                    tau = min(tau * 1.2, tau_max)
        
        # Adaptive tau: increase over time to focus on the true maximum
        tau = min(tau * 1.001, tau_max)
        
        # Check time
        if time.monotonic() - start > time_budget:
            break
    
    return best_a


def build_conv_matrix_fixed(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    # conv(a, g)[k] = sum_i a_i * g[k-i]
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For column j: A[k, j] = a[k-j] for max(0,k-n+1) <= k <= min(2n-2, k)
        # a[k-j] is nonzero when 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_start = max(0, j)
        k_end = min(2 * n - 1, j + n)  # exclusive
        for k in range(k_start, k_end):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build convolution matrix more efficiently."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a[:n]
    # Wait, this isn't right. Let me redo.
    # conv(a, g)[k] = sum_j a[k-j] * g[j] (valid when 0 <= k-j < n)
    # So A[k, j] = a[k-j] if 0 <= k-j < n
    # For column j: k ranges from j to j+n-1, and A[k,j] = a[k-j]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def solve_lp_for_g(a, n_samples=None):
    """
    Solve LP: min M s.t. A(a) @ g <= M, sum(g) = 1, g >= 0
    where A(a) is the matrix such that A(a) @ g = conv(a, g).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_fast(a_norm)
    
    # Try cvxpy solvers
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternate(a, time_budget=100.0, verbose=False):
    """LP alternating: fix a, solve for g; fix g, solve for a; iterate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Solve for g given a
        g_lp, M_lp = solve_lp_for_g(a)
        if g_lp is None:
            break
        
        # Try different mixing parameters
        t_values = np.linspace(0.05, 1.0, 10)
        if 0 < last_good_t < 1:
            t_values = np.unique(np.concatenate([t_values, [last_good_t * 0.8, last_good_t, last_good_t * 1.2]]))
            t_values = t_values[(t_values > 0.01) & (t_values <= 1.0)]
        
        best_t = last_good_t
        best_a_try = a.copy()
        best_obj_try = best_obj
        
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_a_try = a_try.copy()
                best_t = t
        
        if best_obj_try < best_obj - 1e-12:
            best_obj = best_obj_try
            best_a = best_a_try.copy()
            a = best_a
            last_good_t = best_t
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb
                rng = np.random.default_rng(hash((it := 0)) % (2**31))
                # Actually use a proper seed
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                scale = 0.05 + 0.02 * min(stall - 10, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a /= s
                stall = 0
                last_good_t = 0.5

    return best_a


def spectral_flattening(a, time_budget=100.0, verbose=False):
    """
    Gerchberg-Saxton style: alternate between
    - Spectral domain: flatten |FFT(a)|^2
    - Time domain: project to non-negative
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        # Forward FFT
        A = rfft(a, size)
        magnitude = np.abs(A)
        phase = np.angle(A)
        
        # Flatten the magnitude (keep DC component)
        # The DC component is sum(a) = 1 (normalized)
        # We want |A[k]| to be as equal as possible for k >= 1
        if magnitude[0] > 1e-10:
            # Target: each |A[k]| for k>=1 should be sqrt(sum_{k>=1} |A[k]|^2 / (size/2))
            # By Parseval: sum |A[k]|^2 = size * sum a^2
            energy_ns = np.sum(magnitude[1:]**2)
            n_bins = size // 2
            target_mag = np.sqrt(energy_ns / max(n_bins - 1, 1))
        else:
            target_mag = magnitude[0]
        
        A[1:] = target_mag * np.exp(1j * phase[1:])
        # Keep DC
        A[0] = magnitude[0]
        
        # Inverse FFT
        a_new = irfft(A, size)[:n]
        
        # Project to non-negative, maintain sum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    Spectral iter {it}: obj={best_obj:.8f}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initializations(n, rng, num=5):
    """Generate diverse initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangular
    tri = np.abs(np.arange(n) - n/2 + 0.5) / (n * (n+1) / 4)
    inits.append(tri / np.sum(tri))
    
    # Random smooth
    for _ in range(num - 2):
        freq = rng.integers(1, 5)
        x = np.arange(n) / n
        sig = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
        sig = np.maximum(sig, 0.1)
        inits.append(sig / np.sum(sig))
    
    # Bimodal
    center = n // 2
    width = n // 4
    bimodal = np.exp(-((np.arange(n) - center * 0.3) / width)**2) + np.exp(-((np.arange(n) - center * 1.7) / width)**2)
    inits.append(bimodal / np.sum(bimodal))
    
    return inits
