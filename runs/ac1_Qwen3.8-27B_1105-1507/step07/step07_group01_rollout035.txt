Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smooth max (log-sum-exp) gradient descent with FFT
- Aggressive temperature annealing
- Simplex projection with momentum
- Multi-n exploration and structured perturbations
- Coordinate-wise local search for fine-tuning
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_weights(conv, T):
    """Compute softmax weights for smooth max approximation."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    return smooth_val, weights

def smooth_max_gradient(a, T):
    """Compute gradient of 2n * smooth_max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    _, weights = smooth_max_weights(conv, T)
    # grad[i] = 2 * sum_k weights[k] * a[k-i] for valid k-i
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad, conv

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_smooth_grad_conv(a, T):
    """Compute gradient of 2n*smooth_max(conv(a,a)) and conv values."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = np.clip((conv - max_c) / T, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    grad_inner = np.correlate(weights, a, mode='valid')
    grad = 2.0 * n * 2.0 * grad_inner  # 4n * grad_inner
    return grad, conv


def opt_smoothmax_pg(a, time_budget=30.0, T_init=0.0003, T_min=0.000005, lr_init=None):
    """
    Projected gradient descent with smooth max, Nesterov momentum,
    and adaptive temperature.
    """
    n = len(a)
    if lr_init is None:
        lr_init = 2.0 / n  # natural scale
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    best_lr = lr

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        grad, conv = get_smooth_grad_conv(a, T)
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        v = beta * v + grad_centered
        a_new = simplex_project(a - lr * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            best_lr = lr
            if stall == 0 and it % 500 == 0 and lr < lr_init * 3:
                lr = lr * 1.01
        else:
            stall += 1
            a = best_a.copy()
            if stall > 100:
                lr = max(lr * 0.7, best_lr * 0.3)
                v = np.zeros(n)
                stall = 0
                if lr < 1e-8:
                    break

        # Temperature annealing
        if it > 50:
            if it < 2000:
                T = max(T_min, T * 0.998)
            else:
                T = max(T_min, T * 0.9995)
        
        # Scale lr with T ratio
        lr = lr_init * max(0.2, min(3.0, T / T_init)) + 0.5 / n

    return best_a


def opt_smoothmax_mirror(a, time_budget=30.0, T_init=0.0003, T_min=0.000005, lr_init=None):
    """
    Mirror descent (exponential family) with smooth max.
    """
    n = len(a)
    if lr_init is None:
        lr_init = 1.0 / n
    a = np.maximum(a, 1e-15).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    lr = lr_init
    stall = 0

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        grad, conv = get_smooth_grad_conv(a, T)
        grad_centered = grad - np.mean(grad)

        # Mirror descent step
        log_a = np.log(a)
        log_a_new = log_a - lr * grad_centered
        log_a_new -= np.max(log_a_new)  # for numerical stability
        a_new = np.exp(log_a_new)
        s_new = np.sum(a_new)
        if s_new <= 0 or not np.isfinite(s_new):
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            a = best_a.copy()
            if stall > 80:
                lr = lr * 0.8
                stall = 0
                if lr < 1e-10:
                    break

        # Temperature anneal
        if it > 50:
            T = max(T_min, T * 0.998)
        lr = lr_init * max(0.3, min(2.0, T / T_init))

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_hann(n):
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    return a / np.sum(a)

def make_cosine_mod(n, freq=3.0):
    """Cosine modulation of uniform."""

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max (log-sum-exp) gradient
descent with FFT-based convolution, Nesterov momentum, aggressive temperature

local search. Uses mirror descent as an alternative geometry for the simplex.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_smooth_grad(a, T):
    """Compute gradient of 2n*smooth_max(conv(a,a)) and the convolution."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = np.clip((conv - max_c) / T, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    grad_inner = np.correlate(weights, a, mode='valid')
    grad = 4.0 * n * grad_inner
    smooth_val = max_c + T * np.log(total)
    return grad, conv, smooth_val
