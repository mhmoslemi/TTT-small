def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def lse_obj_and_grad(a, T):
    """
    Compute LSE-smoothed objective and gradient.
    Returns (lse_objective, grad) where objective = 2n * LSE(T, conv(a,a))
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = T * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    weights = exp_shifted / sum_exp
    lse_val = max_conv + float(np.log(sum_exp)) / T
    
    # Gradient: g[i] = sum_k weights[k] * a[k-i] (cross-correlation)
    # Using FFT: cross = ifft(FFT(weights) * conj(FFT(a)))
    N = 1
    while N < 3 * n:
        N <<= 1
    W = rfft(weights, N)
    A = rfft(a, N)
    cross = irfft(W * np.conj(A), N)
    grad = cross[:n]
    
    obj = 2.0 * n * lse_val
    grad_obj = 2.0 * n * grad
    return obj, grad_obj

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # For large n, restrict to top constraints to speed up
    conv_a = compute_conv(a)
    n_constraints = len(conv_a)
    
    # Use all constraints for n <= 1000
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.ECOS]
    
    for solver in solver_list:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    return a / a.sum()

Actually, I realize there might be an issue with my `lse_obj_and_grad` function. Let me double-check the cross-correlation computation.

We have:
- `weights` of length 2n-1 (the softmax over convolution values)
- `a` of length n
- We want: `g[i] = sum_k weights[k] * a[k-i]` where `a[k-i]` is 0 if `k-i` is out of [0, n-1]

Using FFT cross-correlation:
`cross[k] = sum_j x[j+k] * y[j]` (circular, with zero-padding for linear)

If I compute `ifft(FFT(weights, N) * conj(FFT(a, N)))`:
`result[k] = sum_j weights[(j+k) mod N] * a[j]`

For this to equal `g[k] = sum_j weights[j+k] * a[j]` (with weights[j+k] = 0 if out of range), I need:
- `j` ranges from 0 to n-1 (length of a)
- `j+k` ranges from k to k+n-1 (should be within [0, 2n-2] for weights to be non-zero)
- For k in [0, n-1]: j+k ranges from k to k+n-1, max is 2n-2. Good.
- No wraparound if N > 2n-2 + n-1 = 3n-3

So N >= 3n is sufficient (I used 3n, which is correct).

And `g[k] = result[k]` for k in [0, n-1]. This is what I compute. Good.

But wait, I need to be more careful about the FFT cross-correlation formula.

The standard circular cross-correlation is:
`(x ⊛ y)[k] = sum_n x[n+k] * conj(y[n])`

In frequency domain: `FFT(x ⊛ y) = X * conj(Y)`

So `ifft(FFT(weights) * conj(FFT(a)))` gives:
`result[k] = sum_n weights[n+k] * conj(a[n]) = sum_n weights[n+k] * a[n]` (since a is real)

And I want `g[i] = sum_k weights[k] * a[k-i]`.

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed Adam optimization (fast convergence on smoothed objective)
2. LP alternating (polishing)
3. Multi-n exploration
4. Diverse restarts with structured initializations
Key improvement: LSE smoothing provides a true gradient direction, enabling
much faster convergence than subgradient methods used previously.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_obj_and_grad(a, T):
    """
    Compute LSE-smoothed objective and gradient.
    Returns (lse_scaled_objective, gradient) where objective = 2n * LSE(T, conv(a,a)).
    Uses FFT for the cross-correlation in the gradient.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = T * (conv - max_conv)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    weights = exp_shifted / sum_exp
    lse_val = max_conv + float(np.log(sum_exp)) / T

    # Gradient: g[i] = sum_k weights[k] * a[k-i]
    # Computed via FFT cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    N = 1
    while N < 3 * n:
        N <<= 1
    W = rfft(weights, N)
    A = rfft(a, N)
    cross = irfft(W * np.conj(A), N)
    grad = cross[:n]

    obj = 2.0 * n * lse_val
    grad_obj = 2.0 * n * grad
    return obj, grad_obj


def adam_lse(a, time_budget=200.0, T_start=5.0, T_end=100.0, lr_init=None, verbose=False):
    """
    Adam optimizer on LSE-smoothed objective with true objective tracking.
    Uses backtracking line search and adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if lr_init is None:
        lr = 1.5 / np.sqrt(n)

    best_a = a.copy()
    best_obj = compute_obj(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    start = time.monotonic()
    stall = 0
    patience = 80

    while time.monotonic() - start < time_budget:
        t += 1
        # Adaptive temperature: ramp up over time
        progress = min(1.0, t / 800.0)
        T = T_start + (T_end - T_start) * progress

        _, grad = lse_obj_and_grad(a, T)

        # Normalize gradient for stability
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Direction (negative gradient with Adam scaling)
        direction = -m_hat / (np.sqrt(v_hat) + eps)
        # Normalize direction
        dnorm = float(np.linalg.norm(direction))
        if dnorm > 1e-15:
            direction = direction / dnorm

        # Backtracking line search
        found = False
        obj_best_local = best_obj
        a_best_local = a.copy()

        for scale in [lr, lr * 0.5, lr * 0.2, lr * 0.08]:
            a_try = a + scale * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < obj_best_local - 1e-12:
                obj_best_local = obj_try
                a_best_local = a_try.copy()
                found = True
                break

        if found:
            best_obj = obj_best_local
            best_a = a_best_local.copy()
            a = a_best_local
            stall = 0
            lr = max(1e-5, lr * 0.9997)
        else:
            stall += 1
            if stall > patience:
                # Random perturbation to escape local minimum
                rng = np.random.default_rng(t * 7919 + n)
                scale_pert = 0.02 + 0.015 * min(stall // patience, 5)
                a = best_a * np.exp(rng.normal(0, scale_pert, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(1e-5, lr * 0.4)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
            else:
                lr = max(1e-5, lr * 0.92)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      Adam-LSE it{t}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.6f}, el={elapsed:.1f}s")

    return best_a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog for reliability
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)

    while time.monotonic() - start < time_budget:
        t += 1
        # Nesterov lookahead
        if t > 1:
            momentum = 2.0 / (t + 5.0)
            y = a + momentum * (a - y)
        else:
            y = a.copy()

        _, grad = lse_obj_and_grad(y, T)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slowly decay learning rate
            lr = max(1e-5, lr * 0.9998)
        else:
            stall += 1
            if stall > 60:
                rng = np.random.default_rng(t * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                y = a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(1e-5, lr * 0.5)
                stall = 0
            else:
                lr = max(1e-5, lr * 0.95)

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"      Nesterov it{t}: obj={best_obj:.8f}, lr={lr:.6f}, el={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # Bimodal Gaussian
        a = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01)
    elif mode == 3:
        # Inverted triangle
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.01)
    elif mode == 4:
        # U-shaped (arc-sine like)
        a = 1.0 / (np.sqrt(4.0 * (x - 0.5) ** 2 + 0.01))
    elif mode == 5:
        # Smooth random (low frequency Fourier)
        freqs = rng.integers(1, 6, size=5)
        amplitudes = rng.uniform(0.1, 0.4, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        a = np.ones(n)
        for f, amp, p in zip(freqs, amplitudes, phases):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Rayleigh-distributed
        a = rng.rayleigh(1.0, n)
        a = np.maximum(a, 0.01)
    elif mode == 7:
        # Ripple / interference pattern
        f1 = rng.uniform(10, 30)
        f2 = rng.uniform(5, 15)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * f1 * x) + 0.2 * np.sin(2 * np.pi * f2 * x)
        a = np.maximum(a, 0.01)
    elif mode == 8:
        # Power-law heavy tails
        a = (x + 0.01) ** (-0.5) + ((1 - x) + 0.01) ** (-0.5)
    elif mode == 9:
        # Sparse bursts
        a = np.ones(n) * 0.1
        n_bursts = rng.integers(3, 8)
        for _ in range(n_bursts):
            center = rng.integers(0, n)
            width = rng.integers(2, max(3, n // 20))
            for i in range(max(0, center - width), min(n, center + width + 1)):
                a[i] += rng.uniform(0.5, 2.0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def subgradient_topK_step(a, K=5, lr=0.05, verbose=False):
    """
    Top-K subgradient step with line search.
    Targets the K largest convolution peaks simultaneously.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return a

    # Softmax weights over top peaks
    weights = np.exp((peak_vals - max_val) * 5.0 / max_val)
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

    gnorm = float(np.linalg.norm(g))
    if gnorm < 1e-15:
        return a
    g = g / gnorm

    # Line search
    for scale in [lr, lr * 0.5, lr * 0.2, lr * 0.1]:
        a_try = a - scale * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LSE-Adam, Nesterov, LP alternating,
    multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: LSE-Adam on parent ===
    print(f"\n=== Phase 1: LSE-Adam (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.18, 220.0)
    if p1_budget > 30:
        a1 = adam_lse(best_seq, time_budget=p1_budget, T_start=5, T_end=80, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1 Adam-LSE: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 Adam-LSE: obj={obj1:.8f} (best={best_score:.8f})")

    # Nesterov as complementary method
    remaining = deadline - time.monotonic()
    if remaining > 60:
        nest_budget = min(remaining * 0.08, 80.0)
        a_nest = nesterov_lse(best_seq, time_budget=nest_budget, T=40.0, verbose=True)
        obj_nest = compute_obj(a_nest)
        if obj_nest < best_score:
            best_score = obj_nest
            best_seq = a_nest.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_nest:.8f} (best={best_score:.8f})")

    # === Phase 2: LP Alternating (main polishing) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 400.0)
    if p2_budget > 30:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            alloc = min(p2_budget / (4 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a2 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
            obj2 = compute_obj(a2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj2:.8f} (best={best_score:.8f})")

    # === Phase 3: LSE-Adam high-T (sharper approximation) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: LSE-Adam high-T ===")
        p3_budget = min(remaining * 0.1, 100.0)
        a3 = adam_lse(best_seq, time_budget=p3_budget, T_start=20, T_end=200, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f} (best={best_score:.8f})")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 100), max(300, n_best - 50),
        n_best, n_best + 50, n_best + 100, n_best + 150, n_best + 200,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick Adam
        alloc = min(remaining * 0.08, 50.0)
        if alloc > 15:
            a_opt = adam_lse(a_try, time_budget=alloc, T_start=5, T_end=60)
            obj_opt = compute_obj(a_opt)
        else:
            a_opt = a_try
            obj_opt = compute_obj(a_try)

        # Quick LP if close
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.04, 30.0)
        if lp_alloc > 15 and obj_opt <= best_score + 0.02:
            a_opt = lp_alternating(a_opt, time_budget=lp_alloc)
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    n_restarts = 6

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = r % 10
        a_r = diverse_init(n_final, rng, mode=mode)
        # Blend with best for warm start
        blend = 0.3 + 0.1 * r
        blend = min(blend, 0.6)
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = a_r.sum()
        if s_r > 1e-10:
            a_r = a_r / s_r
        else:
            a_r = best_seq.copy()

        alloc = min(remaining * 0.12, 70.0)
        a_opt = adam_lse(a_r, time_budget=alloc, T_start=5, T_end=60)
        obj_opt = compute_obj(a_opt)

        remaining = deadline - time.monotonic()
        if remaining > 30 and obj_opt <= best_score + 0.01:
            lp_alloc = min(remaining * 0.08, 30.0)
            a_opt = lp_alternating(a_opt, time_budget=lp_alloc)
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_opt:.8f}")

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 30:
        a_lp = lp_alternating(best_seq, time_budget=min(remaining * 0.25, 80.0), verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    # Adam high-T polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_adam = adam_lse(best_seq, time_budget=min(remaining * 0.2, 50.0), T_start=30, T_end=200)
        obj_adam = compute_obj(a_adam)
        if obj_adam < best_score:
            best_score = obj_adam
            best_seq = a_adam.copy()
            print(f"  Final Adam: obj={best_score:.8f} **")
        else:
            print(f"  Final Adam: obj={obj_adam:.8f}")

    # Another LP round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp2 = lp_alternating(best_seq, time_budget=min(remaining * 0.3, 40.0))
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Top-K subgradient final pass
    remaining = deadline - time.monotonic()
    if remaining > 10:
        for _ in range(20):
            a_tk = subgradient_topK_step(best_seq, K=5, lr=0.02 / np.sqrt(len(best_seq)))
            obj_tk = compute_obj(a_tk)
            if obj_tk < best_score:
                best_score = obj_tk
                best_seq = a_tk.copy()
            else:
                break

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
