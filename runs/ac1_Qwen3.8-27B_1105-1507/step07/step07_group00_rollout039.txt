        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_val = float(np.sum(g_val))
            if s_val > 1e-10:
                return g_val / s_val, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating_improved(a, time_budget=200.0, verbose=False):
    """
    Improved LP alternating with adaptive mixing, dual-guided perturbation,
    and warm-start LP solves.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3
    iter_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Adaptive line search
        candidate_ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0 < last_t < 1:
            candidate_ts.append(last_t * 0.6)
            candidate_ts.append(last_t * 1.4)
        candidate_ts = sorted(set([t for t in candidate_ts if 0 < t <= 1.0]))

        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and iter_count % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.3f}, "
                      f"M_lp={M_lp:.6e}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(iter_count * 104729 + n)
                scale = min(0.2, 0.03 + 0.02 * (stall // 20))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_val = float(np.sum(a))
                if s_val > 1e-10:
                    a = a / s_val
                best_a = a.copy()
                best_obj = compute_obj_normalized(a)
                stall = 0
                last_t = 0.3

    return best_a


def spectral_perturb(a, rng, num_modes=3, strength=0.1):
    """
    Add small frequency-domain perturbations to escape local minima.
    Modifies the sequence by adding small sinusoidal components.
    """
    n = len(a)
    x = np.arange(n)
    a_new = a.copy()
    for _ in range(num_modes):
        freq = rng.integers(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = strength * rng.uniform(0.5, 1.0) / n
        a_new += amp * np.sin(2 * np.pi * freq * x / n + phase)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def coordinate_descent_polish(a, time_budget=30.0, verbose=False):
    """
    Coordinate descent: for each position, try moving mass to reduce the peak.
    Uses the insight that only positions in the peak's support matter.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    conv = compute_conv(a)
    start = time.monotonic()
    max_b = float(np.max(conv))
    mean_b = float(np.mean(conv))

    it = 0
    while time.monotonic() - start < time_budget:
        # Find the peak
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find positions that contribute to the peak
        # i contributes if 0 <= k_star - i < n
        contrib_idx = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib_idx.append((a[i] * a[j], i, j))
        contrib_idx.sort(reverse=True)

        # Find non-contributing positions
        non_contrib = [i for i in range(n)
                       if not (0 <= k_star - i < n)]

        improved = False
        for contrib, i, j in contrib_idx[:30]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break
            for target in non_contrib[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if i == target or delta < 1e-15:
                        continue
                    if a[i] - delta < 0:
                        continue
                    # Move mass from i to target
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_max = float(np.max(new_conv))
                    new_obj = 2.0 * n * new_max
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try small uniform perturbations
            rng = np.random.default_rng(it + n)
            a_try = a * np.exp(rng.normal(0, 0.01, n))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                conv = compute_conv(a)
            else:
                break

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] if a.size > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Smooth random
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Inverted triangle with noise
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a += rng.normal(0, 0.1, n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Geometric (increasing)
        a = np.power(1.1, np.arange(n))
        a = a / a.sum()
    elif mode == 5:
        # Bimodal
        c1, c2 = n // 3, 2 * n // 3
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * (n // 10) ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * (n // 10) ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Anti-symmetric (increasing) - minimizes center bin
        a = np.linspace(0.1, 1.0, n)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def nesterov_multiplicative_optimize(a, time_budget=200.0, verbose=False):
    """
    Nesterov-accelerated multiplicative updates with adaptive peak targeting.
    Uses a combination of single-peak and multi-peak subgradients.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    lr_min = 1e-5
    mu = 0.5  # Nesterov momentum
    y = a.copy()  # Nesterov extrapolation point
    v = np.zeros(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Compute gradient at the extrapolation point
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        conv_y = compute_conv(y)
        max_idx = int(np.argmax(conv_y))

        # Combined gradient: single peak + top-3 average
        g_peak = np.zeros(n)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        j_start = max_idx - i_end
        j_end = max_idx - i_start
        g_peak[i_start:i_end + 1] = 2.0 * y[j_start:j_end + 1][::-1]

        # Top-3 gradient
        top3 = np.argsort(conv_y)[::-1][:3]
        g_multi = np.zeros(n)
        for k in top3:
            k = int(k)
            s2 = max(0, k - (n - 1))
            e2 = min(n - 1, k)
            js = k - e2
            je = k - s2
            g_multi[s2:e2 + 1] += 2.0 * y[js:je + 1][::-1]
        g_multi /= 3.0

        g = 0.5 * g_peak + 0.5 * g_multi

        # Nesterov update
        v = mu * v + g
        y = a - lr * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = a.copy()

        # Multiplicative update from a using gradient
        a_new = multiplicative_step(a, g, lr)

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Nesterov coefficient update
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
            mu_new = (1.0 - t_k) / (1.0 + t_k)
            mu = max(0.3, min(0.8, mu + (mu_new - mu) * 0.1))
            lr = max(lr_min, lr * 0.9999)
        else:
            stall += 1
            if stall > 400:
                rng = np.random.default_rng(it * 13 + n)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj_normalized(a)
                v = np.zeros(n)
                y = a.copy()
                lr = 2.0 / np.sqrt(n)
                mu = 0.5
                stall = 0
            else:
                lr = max(lr_min, lr * 0.96)

        it += 1
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Nesterov it{it}: obj={best_obj:.8f}, lr={lr:.4e}, "
                  f"mu={mu:.3f}, elapsed={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines annealed full-spectrum subgradient,
    Nesterov multiplicative updates, LP alternating, spectral exploration,
    and coordinate descent. Uses multi-n sweep and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj_normalized(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Annealed full-spectrum subgradient ===
    print(f"\n=== Phase 1: Annealed Full-Spectrum Subgradient (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a1 = annealed_subgradient_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj1 = compute_obj_normalized(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Annealed subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Annealed subgrad: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Nesterov multiplicative ===
    print(f"\n=== Phase 2: Nesterov Multiplicative ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 30:
        a2 = nesterov_multiplicative_optimize(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj_normalized(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj2:.8f} (no improvement)")

    # === Phase 3: Target-level subgradient ===
    print(f"\n=== Phase 3: Target-Level Subgradient ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 120.0)
    if p3_budget > 20:
        a3 = target_level_optimize(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj_normalized(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Target-level: obj={best_score:.8f} **")
        else:
            print(f"  Target-level: obj={obj3:.8f} (no improvement)")

    # === Phase 4: LP Alternating (multiple rounds) ===
    print(f"\n=== Phase 4: LP Alternating ===")
    for rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining * 0.15, 150.0)
        if alloc < 20:
            break
        a4 = lp_alternating_improved(best_seq, time_budget=alloc, verbose=True)
        obj4 = compute_obj_normalized(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj4:.8f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100), max(400, n_current - 50),
        n_current, n_current + 50, n_current + 100, n_current + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]
    # Remove duplicates and sort by distance from current
    n_options = sorted(set(n_options), key=lambda x: abs(x - n_current))

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
            s_t = float(np.sum(a_try))
            if s_t > 1e-10:
                a_try = a_try / s_t

        obj_init = compute_obj_normalized(a_try)
        if verbose:
            print(f"  n={n_try}: init obj={obj_init:.8f}")

        # Quick optimize: annealed subgradient + short LP
        alloc1 = min(remaining * 0.12, 60.0)
        if alloc1 > 15:
            a_opt1 = annealed_subgradient_optimize(a_try, time_budget=alloc1)
            obj_opt1 = compute_obj_normalized(a_opt1)
            if obj_opt1 < obj_init:
                a_try = a_opt1
                obj_init = obj_opt1

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 40.0)
        if alloc2 > 10:
            a_opt2 = lp_alternating_improved(a_try, time_budget=alloc2)
            obj_opt2 = compute_obj_normalized(a_opt2)
            if obj_opt2 < obj_init:
                a_try = a_opt2
                obj_init = obj_opt2

        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: best obj={obj_init:.8f}")

    # === Phase 6: Spectral exploration + annealed subgradient ===
    print(f"\n=== Phase 6: Spectral Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_attempts = 4

    for att in range(n_attempts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        # Spectral perturbation
        a_pert = spectral_perturb(best_seq, rng, num_modes=4,
                                  strength=0.05 + 0.02 * att)
        # Blend with best
        blend = 0.3 + 0.1 * att
        a_pert = (1 - blend) * a_pert + blend * best_seq
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.15, 100.0)
        a_opt = annealed_subgradient_optimize(a_pert, time_budget=alloc)
        obj_opt = compute_obj_normalized(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Spectral {att}: obj={best_score:.8f} **")
        else:
            print(f"  Spectral {att}: obj={obj_opt:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_restarts = 3

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 80.0)
        a_opt = annealed_subgradient_optimize(a_r, time_budget=alloc)
        obj_r = compute_obj_normalized(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Annealed subgradient polish
    if remaining > 30:
        alloc = min(remaining * 0.25, 120.0)
        a_p = annealed_subgradient_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_p = compute_obj_normalized(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Final anneal: obj={best_score:.8f} **")

    # Nesterov polish
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.2, 80.0)
        a_p = nesterov_multiplicative_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_p = compute_obj_normalized(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 60.0)
        a_p = lp_alternating_improved(best_seq, time_budget=alloc, verbose=True)
        obj_p = compute_obj_normalized(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Coordinate descent polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_p = coordinate_descent_polish(best_seq, time_budget=alloc, verbose=True)
        obj_p = compute_obj_normalized(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final ann
