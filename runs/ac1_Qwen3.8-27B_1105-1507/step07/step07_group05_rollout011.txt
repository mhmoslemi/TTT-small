    n = len(a)
    m = 2 * n - 1
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0

    # and 0s on the left (for k < n-1) or right (for k >= n-1)
    # Actually it's like a "flipped" Toeplitz

    # First row (k=0): A[0, j] = a[0-j] which is a[0] for j=0, 0 otherwise
    #   So first row: [a[0], 0, 0, ..., 0]
    # Last row (k=2n-2): A[2n-2, j] = a[2n-2-j] which is valid for j = n-1, ..., 2n-2
    #   But j must be < n, so j = n-1: a[n-1]
    #   So last row: [0, 0, ..., 0, a[n-1]]
    # 
    # First column (j=0): A[k, 0] = a[k] for k = 0, ..., n-1; then 0
    #   So first column: [a[0], a[1], ..., a[n-1], 0, ..., 0]
    
    # Use scipy.linalg.toeplitz? No, this is not standard Toeplitz.
    # Let me just build it with numpy vectorization
    
    # Create index arrays
    k_idx = np.arange(m)[:, np.newaxis]  # (m, 1)
    j_idx = np.arange(n)[np.newaxis, :]  # (1, n)
    diff = k_idx - j_idx  # (m, n), diff[k,j] = k-j
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((m, n))

def solve_lp_direction(a):
    """
    Solve: min M s.t. conv(a, g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns optimal g.
    """
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)  # (m, n), A @ g = conv(a, g)
    
    # Variables: g[0..n-1], M => total n+1 variables
    # Minimize: M => c = [0, 0, ..., 0, 1]
    # Constraints: A @ g - M <= 0 => [A, -I_m] @ [g; M] <= 0
    #               sum(g) = 1 => [1, 1, ..., 1, 0] @ [g; M] = 1
    #               g >= 0, M >= 0
    
    from scipy.optimize import linprog
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.eye(m)])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    k_idx = np.arange(m)[:, np.newaxis]
    j_idx = np.arange(n)[np.newaxis, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((m, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns g or None."""

                stall = 0
                if verbose and it % 10 == 0:
                    print(f"    LP alt it{it}: obj={best_obj:.8f}, t={t:.3f}, diff={diff_norm:.6f}")
            else:
                stall += 1

        if stall > max_stall:
            # Perturb and restart
            rng_local = np.random.default_rng(it * 7 + 3)
            a = best_a * np.exp(rng_local.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            stall = 0
            max_stall += 10

    return best_a


def logsumexp_gradient(a, T):
    """Compute gradient of (1/T)*log(sum_k exp(T*conv(a,a)[k]))."""
    n = len(a)
    b = compute_conv(a)
    b_shifted = b - np.max(b)
    w = np.exp(T * b_shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum

    # grad[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * cross_correlation(w, a)[i]
    N = 1
    while N < len(w) + n - 1:
        N <<= 1
    W = np.zeros(N)
    W[:len(w)] = w
    A = np.zeros(N)
    A[:n] = a
    result = irfft(rfft(W, N) * np.conj(rfft(A, N)), N)
    grad = 2.0 * result[:n]
    return grad


def logsumexp_gd(a, time_budget=60.0, T_start=1.0, T_end=50.0, verbose=False):
    """Log-sum-exp smoothed gradient descent with temperature annealing and Adam."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam parameters
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t_step = 0
    lr = 0.5

    stall = 0
    max_stall = 100

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        # Anneal temperature
        progress = min(1.0, it / 50000)
        T = T_start + (T_end - T_start) * progress

        grad = logsumexp_gradient(a, T)
        # Project onto simplex tangent
        grad = grad - np.mean(grad)

        t_step += 1
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat_corr = m_hat / (1 - beta1 ** t_step)
        v_hat_corr = v_hat / (1 - beta2 ** t_step)

        a_new = a - lr * m_hat_corr / (np.sqrt(v_hat_corr) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.01, 2.0)
            if verbose and it % 1000 == 0:
                print(f"    LSE-GD it{it}: obj={best_obj:.8f}, T={T:.2f}, lr={lr:.4f}")
        else:
            stall += 1
            lr *= 0.98
            if stall > max_stall:
                # Reseed from best
                a = best_a.copy()
                m_hat = np.zeros(n)
                v_hat = np.zeros(n)
                lr = 0.5
                stall = 0
                max_stall += 50
            if lr < 1e-6:
                break

    return best_a


def subgradient_peak(a, time_budget=30.0, n_peaks=5, verbose=False):
    """Subgradient descent tracking multiple peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    momentum = np.zeros(n)
    mom_rate = 0.7

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv(a)
        max_b = np.max(b)

        # Find top n_peaks positions
        # Use softmax weighting with sharp temperature
        T = 20.0
        w = np.exp(T * (b - max_b))
        w = w / np.sum(w)

        # Gradient: 2 * sum_k w[k] * a[k-i]
        N = 1
        while N < len(w) + n - 1:
            N <<= 1
        W = np.zeros(N)
        W[:len(w)] = w
        A = np.zeros(N)
        A[:n] = a
        result = irfft(rfft(W, N) * np.conj(rfft(A, N)), N)
        grad = 2.0 * result[:n]
        grad = grad - np.mean(grad)

        momentum = mom_rate * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            momentum = np.zeros(n)
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.5)
        else:
            eta *= 0.97
            momentum *= 0.8
            if eta < 1e-8:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: LP alternation + log-sum-exp gradient + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"\nStarting optimization with n={n_best}, initial obj={best_score:.8f}")

    # === Phase 1: LP Alternation (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternation ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.35, 400.0)
    if alloc > 30:
        a_lp1 = lp_alternation(best_seq, time_budget=alloc, verbose=True)
        obj_lp1 = compute_obj(a_lp1)
        if obj_lp1 < best_score:
            best_score = obj_lp1
            best_seq = a_lp1.copy()
            print(f"  ** Phase 1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 LP: obj={obj_lp1:.8f} (no improvement)")

    # === Phase 2: Log-sum-exp gradient refinement ===
    print(f"\n=== Phase 2: Log-Sum-Exp Gradient ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.2, 180.0)
    if alloc > 20:
        a_lse = logsumexp_gd(best_seq, time_budget=alloc, T_start=2.0, T_end=80.0, verbose=True)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  ** Phase 2 LSE: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 LSE: obj={obj_lse:.8f}")

    # === Phase 3: LP Alternation round 2 (from improved point) ===
    print(f"\n=== Phase 3: LP Alternation Round 2 ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.15, 120.0)
    if alloc > 20:
        a_lp2 = lp_alternation(best_seq, time_budget=alloc, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** Phase 3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 LP: obj={obj_lp2:.8f}")

    # === Phase 4: Multi-peak subgradient ===
    print(f"\n=== Phase 4: Multi-Peak Subgradient ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.1, 60.0)
    if alloc > 15:
        a_sg = subgradient_peak(best_seq, time_budget=alloc, n_peaks=5, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** Phase 4 SG: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 SG: obj={obj_sg:.8f}")

    # === Phase 5: LP + LSE alternation (combined) ===
    print(f"\n=== Phase 5: Combined LP+LSE ===")
    remaining = deadline - time.monotonic()
    n_rounds = max(2, int(remaining / 40))
    for r in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        # LP step
        alloc_lp = min(remaining * 0.4, 40.0)
        if alloc_lp > 10:
            a_r = lp_alternation(best_seq, time_budget=alloc_lp)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"    Round {r} LP: {best_score:.8f}")
        # LSE step
        remaining = deadline - time.monotonic()
        alloc_lse = min(remaining * 0.4, 30.0)
        if alloc_lse > 10:
            a_r = logsumexp_gd(best_seq, time_budget=alloc_lse, T_start=5.0, T_end=100.0)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"    Round {r} LSE: {best_score:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")

    n_options = sorted(set([
        n_curr - 50, n_curr + 50, n_curr + 100, n_curr + 200,
        n_curr - 100, 700, 850, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_curr]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / np.sum(a_try)

        # Quick LP optimization
        alloc = min(remaining * 0.2, 35.0)
        opt = lp_alternation(a_try, time_budget=alloc)
        obj_try = compute_obj(opt)
        if obj_try < best_score:

        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.6, 80.0)
        a_final = lp_alternation(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

        # One more LSE pass
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.3, 25.0)
            a_final2 = logsumexp_gd(best_seq, time_budget=alloc2, T_start=10.0, T_end=200.0)
            obj_final2 = compute_obj(a_final2)
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  ** Final LSE: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
