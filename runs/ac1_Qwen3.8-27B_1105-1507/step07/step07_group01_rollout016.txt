import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_given_a(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)
    c = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    b_eq = np.array([1.0])

def lp_alternating_nesterov(a, time_budget=100.0, verbose=False):
    """LP alternating with Nesterov acceleration and adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    v = np.zeros(n)
    beta = 0.0  # will ramp up
    stall = 0
    last_t = 0.3
    n_iter = 0

    while time.monotonic() - start < time_budget:
        n_iter += 1
        g, M = solve_lp_given_a(a)
        if g is None:
            break

        # Nesterov extrapolation: blend based on momentum
        if beta < 0.3:
            beta = min(0.3, beta + 0.005)

        # Direction: from a toward g
        direction = g - a
        # Nesterov step: a + beta * v + (1-beta) * direction
        # Simplified: a_new = (1-t)*a + t*g with t modulated by momentum
        v = beta * v + (1 - beta) * direction
        effective_direction = v

        # Try multiple blend factors
        norm_dir = np.linalg.norm(effective_direction)
        if norm_dir < 1e-10:
            break

        # Normalize direction for step size
        base_t = last_t

        t_candidates.append(t_mom)

                last_t = 0.3

def subgradient_polish(a, time_budget=30.0):
    """Subgradient method targeting top peaks with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    top_k = 5
    patience = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)

        # Get top-k peaks
        top_indices = np.argsort(conv)[-top_k:]
        max_val = conv[top_indices[0]]
        # Weights: exponential based on gap to max
        gaps = max_val - conv[top_indices]
        weights = np.exp(-gaps / (0.001 * max_val + 1e-10))
        weights = weights / np.sum(weights)

        # Gradient via correlated
        w_full = np.zeros(len(conv))
        for i, idx in enumerate(top_indices):
            w_full[int(idx)] = weights[i]

        grad = np.correlate(w_full, a, mode='valid')
        grad_eff = grad - np.sum(conv * w_full)  # center by the smooth max value

        # Step
        a_new = a - step * grad_eff
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            step *= 0.5
            patience += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            step = min(step * 1.02, 0.15)
        else:
            patience += 1
            step *= 0.9
            a = best_a.copy()
            if patience > 50:
                top_k = min(top_k + 2, 20)
                patience = 0
                if top_k > 20:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample via linear interpolation on [0,1]."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, mode, rng):
    """Generate structured initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        # Two bumps
        c1, c2 = n // 3, 2 * n // 3
        w1, w2 = n // 8, n // 8
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
    elif mode == 2:
        # Three bumps
        c1, c2, c3 = n // 4, n // 2, 3 * n // 4
        w = n // 10
        a = (np.exp(-((x - c1) ** 2) / (2 * w ** 2)) +
             np.exp(-((x - c2) ** 2) / (2 * w ** 2)) +
             np.exp(-((x - c3) ** 2) / (2 * w ** 2)))
    elif mode == 3:
        # Wide Gaussian
        a = np.exp(-((x - n / 2) ** 2) / (2 * (n / 3) ** 2))
    elif mode == 4:
        # Cosine window
        a = np.cos(np.pi * x / n) ** 2 + 0.1
    elif mode == 5:
        # Random smooth
        a = rng.uniform(0.5, 1.0, n)
        # Smooth
        for _ in range(3):
            a = np.convolve(a, np.ones(5) / 5, mode='same')
    elif mode == 6:
        # Alternating pattern
        a = np.where(x % 2 == 0, 1.0, 0.5)
        a = np.convolve(a, np.ones(3) / 3, mode='same')
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
        a = a * n  # rescale
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n

        a_lp = lp_alternating_nesterov(a_final, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    # Subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 40.0)
        a_sg = subgradient_polish(a_final, time_budget=alloc)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f}")

    # Another LP after SG
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_lp2 = lp_alternating_nesterov(best_seq, time_budget=alloc)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Final LP2: obj={best_score:.8f}")

    # === Phase 5: Tiny n adjustments ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_adj = len(best_seq)
        for delta in [-10, -5, 5, 10, -3, 3]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            n_new = n_adj + delta
            if n_new < 200:
                continue
            a_adj = resample_sequence(best_seq, n_new)
            alloc = min(10.0, remaining * 0.2)
            a_opt = lp_alternating_nesterov(a_adj, time_budget=alloc)
            obj_adj = compute_obj(a_opt)
            if obj_adj < best_score:
                best_score = obj_adj
                best_seq = a_opt.copy()
                print(f"  Adj n={n_new}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
