    while p < m:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:

    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def exact_gradient(a, tol=1e-10):
    """
    Compute the exact (sub)gradient of f(a) = 2*n*max_k (a*a)[k] / (sum a)^2.
    Uses the active set of lags within tol of the maximum.
    Returns (objective_value, gradient).
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    obj = 2.0 * n * max_val / (S * S)
    
    # Active set: lags within tol of max
    active = np.where(conv >= max_val - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        # d conv[k] / da[i] = 2*a[k-i] if 0 <= k-i < n
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k_int - idx]
    
    # Average over active set
    grad /= len(active)
    
    # Full gradient of f = 2*n*conv[k]/S^2
    # df/da[i] = 2*n/S^2 * (dconv/da[i] - 2*conv[k]*a[i]/S)
    grad = 2.0 * n / (S * S) * (grad - 2.0 * max_val * a / S)
    
    return obj, grad


def log_space_gradient(x, tol=1e-10):
    """
    Gradient in log-space. a = exp(x - max(x)) / sum(exp(x - max(x))).
    Returns (obj, grad_x).
    """
    n = len(x)
    x_shift = x - np.max(x)
    a = np.exp(x_shift)
    a = a / np.sum(a)
    
    obj, grad_a = exact_gradient(a, tol)
    
    # Chain rule: a_i = exp(x_i - m) / S where m = max(x), S = sum(exp(x_j - m))
    # da_i/dx_i = a_i * (1 - a_i)  (since a_i = exp(x_i-m)/S, and S includes the i-th term)
    # da_i/dx_j = -a_i * a_j  (for i != j)
    # grad_x_i = grad_a_i * a_i * (1-a_i) + sum_{j!=i} grad_a_j * (-a_i * a_j)
    # = a_i * (grad_a_i * (1-a_i) - sum_{j!=i} grad_a_j * a_j)
    # = a_i * (grad_a_i - sum_j grad_a_j * a_j)
    
    dot = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

    while p < m:

    active = np.where(conv >= max_val - 1e-12)[0]
    return conv, max_val, active


def grad_of_conv_lag(a, k):
    """Compute d(conv(a,a))[k] / da for a specific lag k."""
    n = len(a)
    g = np.zeros(n)
    k_int = int(k)
    lo = max(0, k_int - n + 1)
    hi = min(n - 1, k_int)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k_int - idx]
    return g


def exact_subgradient(a, tol=1e-10):
    """
    Compute exact (sub)gradient of f(a) = 2*n*max_k conv[k] / (sum a)^2.
    Returns (objective, gradient, n_active).
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n), 0

    conv, max_val, active = compute_conv_and_max(a)
    # Widen active set slightly
    active = np.where(conv >= max_val - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])

    obj = 2.0 * n * max_val / (S * S)

    # Sum gradients from active lags
    grad_sum = np.zeros(n)
    for k in active:
        grad_sum += grad_of_conv_lag(a, k)
    grad_sum /= len(active)

    grad = (2.0 * n / (S * S)) * (grad_sum - 2.0 * max_val * a / S)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ b = conv(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,b)[k] <= M, sum(b)=1, b>=0.
    Returns (b_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # Variables: b[0..n-1], M (total n+1 variables)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: A @ b - M <= 0  =>  [A, -1] @ [b, M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            b_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(b_val))
            if s > 1e-10:
                return b_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

    best_a = a_norm.copy()
    best_obj = compute_obj_normalized(a_norm)

                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def diverse_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Sech shape
        x = np.linspace(-3, 3, n)
        a = 1.0 / np.cosh(x)
        a = a / np.sum(a)
    elif mode == 3:
        # Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
        a = a / np.sum(a)
    elif mode == 4:
        # Triangular
        a = 1.0 - np.abs(2 * np.arange(n) / n - 1)
        a = a / np.sum(a)
    elif mode == 5:
        # U-shaped (convex)
        x = np.arange(n) / n
        a = 0.3 + 0.7 * (2 * x - 1) ** 2
        a = a / np.sum(a)
    elif mode == 6:
        # Two-lobe
        a = (np.exp(-2 * ((np.arange(n) / n) - 0.25) ** 2) +
             np.exp(-2 * ((np.arange(n) / n) - 0.75) ** 2))
        a = a / np.sum(a)
    else:
        # Smooth random
        a = rng.uniform(0, 1, n)
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)

    return np.maximum(a, 1e-10)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines exact-gradient L-BFGS, LP alternating,
    and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            parent = parent_arr / np.sum(parent_arr)
            n_parent = len(parent)
        else:
            raise ValueError("Invalid parent")
    except Exception:
        parent = np.ones(800) / 800.0
        n_parent = 800

    best_a = parent.copy()
    best_score = compute_obj_normalized(parent)
    print(f"Initial (n={n_parent}): score={best_score:.10f}")

    def time_left():
        return deadline - time.monotonic()

    # === Phase 1: Gradient optimization at parent length ===
    print(f"\n=== Phase 1: Gradient Optimization (n={n_parent}) ===")
    if time_left() > 60:
        alloc = min(time_left() * 0.2, 120.0)
        a_opt = gradient_optimize(best_a, time_budget=alloc, verbose=True)
        obj = compute_obj_normalized(a_opt)
        if obj < best_score:
            best_score = obj
            best_a = a_opt.copy()
            print(f"  After gradient: score={best_score:.10f}")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating ===")
    if time_left() > 60:
        alloc = min(time_left() * 0.2, 120.0)
        a_opt = lp_alternate_optimize(best_a, time_budget=alloc, verbose=True)
        obj = compute_obj_normalized(a_opt)
        if obj < best_score:
            best_score = obj
            best_a = a_opt.copy()
            print(f"  After LP: score={best_score:.10f}")

    # === Phase 3: Gradient + LP combined polish ===
    print(f"\n=== Phase 3: Combined Polish ===")
    if time_left() > 60:
        a_cur = best_a.copy()
        for round_num in range(5):
            if time_left() < 30:
                break
            # Gradient step
            alloc_g = min(20.0, time_left() * 0.3)
            a_cur = gradient_optimize(a_cur, time_budget=alloc_g)
            obj = compute_obj_normalized(a_cur)
            if obj < best_score:
                best_score = obj
                best_a = a_cur.copy()

            # LP step
            if time_left() < 30:
                break
            alloc_l = min(15.0, time_left() * 0.3)
            a_cur = lp_alternate_optimize(a_cur, time_budget=alloc_l)
            obj = compute_obj_normalized(a_cur)
            if obj < best_score:
                best_score = obj
                best_a = a_cur.copy()
                print(f"  Round {round_num}: score={best_score:.10f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_a)
    n_candidates = sorted(set([
        max(300, n_best - 200),
        max(300, n_best - 100),
        n_best,
        n_best + 100,
        n_best + 200,
        600, 700, 900, 1000, 1100, 1200
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 1500]

    for n_try in n_candidates:
        if time_left() < 50:
            break

        if n_try == n_best:
            a_try = best_a.copy()
        else:
            a_try = resample_sequence(best_a, n_try)

        alloc = min(time_left() * 0.15, 40.0)
        if alloc < 10:
            alloc = 10.0

        a_opt = gradient_optimize(a_try, time_budget=alloc)
        obj = compute_obj_normalized(a_opt)

        if obj < best_score:
            best_score = obj
            best_a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: score={best_score:.10f} **")
        else:
            print(f"  n={n_try}: score={obj:.10f}")

    # === Phase 5: Intensify on best ===
    print(f"\n=== Phase 5: Intensify (n={len(best_a)}) ===")
    if time_left() > 60:
        a_cur = best_a.copy()
        for it in range(10):
            if time_left() < 20:
                break
            # Alternate gradient and LP with small budgets
            a_cur = gradient_optimize(a_cur, time_budget=min(15.0, time_left() * 0.2))
            obj = compute_obj_normalized(a_cur)
            if obj < best_score:
                best_score = obj
                best_a = a_cur.copy()

            if time_left() < 20:
                break
            a_cur = lp_alternate_optimize(a_cur, time_budget=min(10.0, time_left() * 0.2))
            obj = compute_obj_normalized(a_cur)
            if obj < best_score:
                best_score = obj
                best_a = a_cur.copy()
                print(f"  Iter {it}: score={best_score:.10f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    n_final = len(best_a)
    if time_left() > 80:
        n_restarts = min(5, max(2, int(time_left() / 20)))
        for r in range(n_restarts):
            if time_left() < 30:
                break

            if r == 0:
                a_r = best_a * np.exp(rng.normal(0, 0.02, n_final))
                a_r = np.maximum(a_r, 1e-30)
                a_r = a_r / np.sum(a_r)
            elif r == 1:
                n_r = n_final + int(rng.integers(-50, 50))
                n_r = max(400, min(1300, n_r))
                a_r = resample_sequence(best_a, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.02, n_r))
                a_r = np.maximum(a_r, 1e-30)
                a_r = a_r / np.sum(a_r)
            elif r == 2:
                a_r = diverse_init(n_final, rng)
            else:
                a_r = diverse_init(n_final, rng)

            alloc = min(20.0, time_left() * 0.3)
            a_opt = gradient_optimize(a_r, time_budget=alloc)
            obj = compute_obj_normalized(a_opt)
            if obj < best_score:
                best_score = obj
                best_a = a_opt.copy()
                n_final = len(best_a)
                print(f"  Restart {r}: score={best_score:.10f} **")
            else:
                print(f"  Restart {r}: score={obj:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if time_left() > 30:
        a_final = gradient_optimize(best_a, time_budget=min(time_left() * 0.4, 30.0), verbose=True)
        obj = compute_obj_normalized(a_final)
        if obj < best_score:
            best_score = obj
            best_a = a_final.copy()
            print(f"  Final gradient: score={best_score:.10f}")

    if time_left() > 20:
        a_final = lp_alternate_optimize(best_a, time_budget=min(time_left() * 0.3, 20.0))
        obj = compute_obj_normalized(a_final)
        if obj < best_score:
            best_score = obj
            best_a = a_final.copy()
            print(f"  Final LP: score={best_score:.10f}")

    # Final validation
    if best_a is None:
        best_a = np.ones(200) / 200.0

    n_out = len(best_a)
    print(f"\n=== FINAL: n={n_out}, score={best_score:.10f} ===")

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0 / 100.0] * 100
        best_score = evaluate_sequence(result)

    print(f"=== VERIFIED: n={len(result)}, score={best_score:.10f} ===")
    return result
