import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve min M s.t. (a*g)[k] <= M, sum g = 1, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    # Use scipy linprog (HiGHS)
    from scipy.optimize import linprog
    c = np.zeros(n+1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = np.sum(g)
        if gs > 1e-10:
            return g/gs, res.x[-1]
    return None, None

"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses LP-based alternating optimization with palindromic symmetry, multiple
initializations, adaptive step sizes, and subspace polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def compute_obj_normalized(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    return 2.0 * n * float(np.max(compute_conv(a)))


def solve_lp_for_g(a, max_time=5.0):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    
    # Build constraint matrix: (a*g)[k] = sum_j a[j]*g[k-j]
    # Coefficient of g[m] in (a*g)[k] is a[k-m] (when valid)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[j]  # g[k-j] has coefficient a[j]
    
    # Actually, let me redo this. 
    # (a*g)[k] = sum_j a[j] * g[k-j] where 0<=j<n and 0<=k-j<n
    # For g[m] (m = k-j, so j = k-m): valid when 0<=k-m<n, i.e., k-n+1<=m<=k
    # Coefficient of g[m] in (a*g)[k] is a[k-m] (when 0<=k-m<n)
    
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        for m in range(m_start, m_end + 1):
            A[k, m] = a[k - m]
    
    # LP: min M
    # s.t. A @ g - M <= 0  (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0, M free
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

                      options={'time_limit': max_time})
        if res.status == 0:

            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs, float(res.x[-1])
    except Exception:

def build_conv_matrix(a, n_constraints=None):
    """Build matrix A such that A @ g = conv(a, g)[:n_constraints]."""
    n = len(a)
    if n_constraints is None:
        n_constraints = 2 * n - 1
    
    A = np.zeros((n_constraints, n))
    for m in range(n):
        # g[m] appears in (a*g)[k] for k = m, m+1, ..., m+n-1
        # with coefficient a[k-m]
        k_start = m
        k_end = min(n_constraints - 1, m + n - 1)
        A[k_start:k_end+1, m] = a[:k_end - k_start + 1]
    
    return A

            A_red[k, i] = coeff
    
    # LP: min M
    # s.t. A_red @ h - M <= 0  (2n-1 constraints)
    #      sum(g) = 1  =>  2 * sum(h[0..half-2]) + (h[half-1] if n odd) = 1
    #      Actually: sum(g) = sum of all g[m] = 2*sum_{i=0}^{half-2} h[i] + (h[half-1] if n odd)
    #      Wait: g has n elements. If n is even: g[0..n/2-1] free, g[n/2..n-1] = mirror. sum = 2*sum(h)
    #      If n is odd: g[0..(n-1)/2] free (including middle), g[(n+1)/2..n-1] = mirror. sum = 2*sum(h[0..half-2]) + h[half-1]
    
    if n % 2 == 0:
        # n even: half = n/2, sum(g) = 2 * sum(h)
        # constraint: 2 * sum(h) = 1, i.e., sum(h) = 1/2
        sum_coeff = np.ones(half) * 2.0
    else:
        # n odd: half = (n+1)/2, sum(g) = 2*sum(h[0..half-2]) + h[half-1]
        sum_coeff = np.ones(half) * 2.0
        sum_coeff[-1] = 1.0  # middle element
    
    c = np.zeros(half + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A_red, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, half + 1))
    A_eq[0, :half] = sum_coeff
    b_eq = np.array([1.0])
    bounds = [(0, None)] * half + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        h = np.maximum(res.x[:half], 0)
        # Reconstruct full g
        g = np.zeros(n)
        for i in range(half):
            g[i] = h[i]
            g[n-1-i] = h[i]
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs, float(res.x[-1])
    return None, None
