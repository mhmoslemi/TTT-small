"""
Cutting-plane LP + smoothed mirror descent + spectral shaping for low-peak
non-negative convolution sequences. Minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_grad_bk(a, k):
    """Compute gradient of b[k] = conv(a,a)[k] w.r.t. a. Returns array of length n."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]
    return grad


def compute_grad_bk_fast(a, k):
    """Fast gradient computation using FFT for a single k."""
    n = len(a)
    # grad[i] = 2*a[k-i] for valid k-i
    valid_start = max(0, k - (n - 1))
    valid_end = min(n - 1, k)
    grad = np.zeros(n)
    if valid_start <= valid_end:
        indices = np.arange(valid_start, valid_end + 1)
        grad[indices] = 2.0 * a[k - indices]
    return grad


def softmax_grad_conv(a, tau):
    """Compute softmax-weighted gradient of smoothed peaked convolution.
    Returns (gradient_vector, conv_values, softmax_weights)."""
    n = len(a)
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / np.sum(exp_c)
    
    # Gradient: sum_k w[k] * grad_bk(a, k)
    # grad_bk(a,k)[i] = 2*a[k-i] (for valid k-i)
    # So grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * cross_correlation(w, a)[i]
    # cross_correlation(w,a)[i] = sum_k w[k] * a[k-i]
    
    size = next_pow2(3 * n)
    fw = rfft(w, size)
    fa = rfft(a, size)
    # We need sum_k w[k] * a[k-i] = (w correlated with a) at lag i
    # = irfft(conj(fft(a)) * fft(w)) -- but need to handle linearity
    # Actually: sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] (substituting j=k-i)
    # This is the circular cross-correlation of a and w at lag i (for circular)
    # For linear, we need zero-padding
    cross = irfft(np.conj(fa) * fw, size)
    grad = 2.0 * cross[:n]
    return grad, conv, w


def smoothed_obj_val(a, tau):

def cutting_plane_lp(a0, max_iters=30, tol=1e-8, verbose=False):
    """
    Cutting-plane method: iteratively linearize the quadratic constraints
    b[k] <= M and solve the resulting LP.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iters):
        # Compute current convolution and gradients
        b = compute_conv(a)
        M_current = float(np.max(b))
        
        # Build LP: minimize M s.t. sum_i grad_bk(a)[i] * a_i <= M + b[k] for all k
        # Plus: sum(a) = 1, a >= 0, M >= 0
        # Variables: a_0, ..., a_{n-1}, M (total n+1)
        
        # For efficiency, only include "active" constraints (near the peak)
        # But start with all constraints
        b_max = M_current
        # Select top constraints (to keep LP size manageable)
        threshold = b_max * 0.5
        active_k = np.where(b >= threshold)[0]
        
        if len(active_k) == 0:
            active_k = np.array([int(np.argmax(b))])
        
        # Build constraint matrix
        # For each active k: sum_i (2*a[k-i]) * x_i - M <= b[k]
        # i.e.: sum_i (2*a[k-i]) * x_i - M <= b[k]
        
        m_cons = len(active_k)
        A_ub = np.zeros((m_cons, n + 1))
        b_ub = np.zeros(m_cons)
        
        for idx, k in enumerate(active_k):
            grad_k = compute_grad_bk_fast(a, k)
            A_ub[idx, :n] = grad_k
            A_ub[idx, n] = -1.0
            b_ub[idx] = b[k]
        
        # Equality: sum(a) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        # Bounds: a_i >= 0, M >= 0
        bounds = [(0, None)] * n + [(0, None)]
        
        # Objective: minimize M
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        # Solve LP
        solved = False
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.status == 0:
                a_new = np.maximum(res.x[:n], 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    b_new = compute_conv(a_new)
                    M_new = float(np.max(b_new))
                    
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        if verbose:

                            if s_blend > 1e-10:
                                a_blend = a_blend / s_blend

                                if obj_blend < best_obj:
                                    best_obj = obj_blend
                                    best_a = a_blend.copy()
                                    a = a_blend
                                    if verbose:
                                        print(f"    CP iter {it} blend t={t}: obj={best_obj:.8f}")

            break

def spectral_shaping(a, alpha=0.7, iterations=20, verbose=False):
    """
    Spectral shaping: modify the magnitude spectrum to flatten the convolution.
    a -> IFFT(|FFT(a)|^alpha * phase(FFT(a))) -> project to non-negative simplex.
    """
    n = len(a)
    a_work = np.maximum(a, 0).copy().astype(np.float64)
    a_work = a_work / np.sum(a_work)
    
    best_a = a_work.copy()
    best_obj = compute_obj(a_work)
    
    size = next_pow2(2 * n - 1)
    
    for it in range(iterations):
        fa = fft(a_work, size)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # Compress dynamic range
        mag_new = mag ** alpha + 0.01 * np.mean(mag)
        
        fa_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(fa_new, size))[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a_work = a_new
        else:
            # Try different alpha
            for al in [0.5, 0.8, 0.9, 0.3]:
                mag_try = (np.abs(fa)) ** al + 0.01 * np.mean(np.abs(fa))
                fa_try = mag_try * np.exp(1j * phase)
                a_try = np.real(ifft(fa_try, size))[:n]
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a_work = a_try
                    alpha = al
            break
    
    if verbose:
        print(f"    Spectral shaping: obj={best_obj:.8f}")
    return best_a


def lbfgs_smoothed_polish(a_init, time_budget=30.0, verbose=False):
    """L-BFGS-B polish on smoothed objective with log parameterization."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    taus = [3.0 / n, 1.0 / n, 0.3 / n, 0.1 / n]
    x = np.log(a + 1e-30)
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        def fun(xv):
            xv_shift = xv - np.max(xv)
            av = np.exp(xv_shift)
            sv = np.sum(av)
            if sv < 1e-10:
                return np.inf
            av = av / sv
            conv = compute_conv(av)
            cs = conv / tau
            cm = np.max(cs)
            lse = cm + np.log(np.sum(np.exp(cs - cm)))
            return 2.0 * n * lse * tau
        
        def fgrad(xv):
            xv_shift = xv - np.max(xv)
            av = np.exp(xv_shift)
            sv = np.sum(av)
            av_norm = av / sv
            grad_full, conv, w = softmax_grad_conv(av_norm, tau)
            # Chain rule: d/dx_i = d/d a_i * da_i/dx_i
            # a_i = exp(x_i) * av_norm_i / exp(x_i) ... 
            # Actually a_i(x) = exp(x_i) / sum(exp(x))
            # da_i/dx_j = a_i * (delta_ij - a_j)
            # So grad_x_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
            # = a_j * (grad_a_j - sum_i grad_a_i * a_i)
            dot = np.dot(grad_full, av_norm)
            grad_x = av_norm * (grad_full - dot)
            return grad_x
        
        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-12})
            if res.x is not None:
                x_new = res.x - np.max(res.x)
                a_new = np.exp(x_new)
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = res.x
                    if verbose:
                        print(f"    LBF tau={tau:.4f}: obj={best_obj:.8f}")
        except Exception:
            pass
    
    return best_a


def projected_subgradient_nesterov(a_init, time_budget=60.0, verbose=False):
    """Nesterov-accelerated projected subgradient on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    x_prev = a.copy()
    x = a.copy()
    momentum = 0.0
    
    # Gradient bound (upper bound on ||grad||)
    # grad = 2*a_shifted, so ||grad||_inf <= 2*max(a) <= 2
    # ||grad||_2 <= 2*sqrt(sum(a^2)) <= 2
    G = 2.0
    
    for t in range(1, 50000):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov update for y
        if t > 1:
            new_momentum = (1 + np.sqrt(1 + 4 * momentum * momentum)) / 2
            beta = (1 - momentum) / new_momentum
            momentum = new_momentum
            y = x + beta * (x - x_prev)
        else:
            y = x.copy()
        
        # Compute subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        g = compute_grad_bk_fast(y, k_star)
        
        # Step size
        eta = G / (2.0 * np.sqrt(t + 1))
        
        # Update
        x_new = y - eta * g
        x_new = np.maximum(x_new, 0)
        s = np.sum(x_new)
        if s > 1e-15:
            x_new = x_new / s
        else:
            x_new = y.copy()
        
        obj_new = compute_obj(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = x_new.copy()
        
        x_prev = x
        x = x_new
        
        # Periodic check
        if t % 5000 == 0 and verbose:
            print(f"    PSM t={t}: obj={best_obj:.8f}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def active_set_lp(a0, verbose=False):
    """
    Refined cutting plane: only use constraints that are 'active' (near max).
    Multiple passes with widening threshold.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for threshold_frac in [0.8, 0.6, 0.4, 0.2, 0.0]:
        b = compute_conv(a)
        M = float(np.max(b))
        threshold = M * threshold_frac
        active_k = np.where(b >= threshold)[0]
        
        if len(active_k) < 5:
            # Ensure we have at least the top 5
            top_idx = np.argsort(b)[-5:]
            active_k = np.sort(top_idx)
        
        # Build and solve LP
        m_cons = len(active_k)
        A_ub = np.zeros((m_cons, n + 1))
        b_ub = np.zeros(m_cons)
        
        for idx, k in enumerate(active_k):
            grad_k = compute_grad_bk_fast(a, k)
            A_ub[idx, :n] = grad_k
            A_ub[idx, n] = -1.0
            b_ub[idx] = b[k]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 5})
            if res.status == 0:
                a_new = np.maximum(res.x[:n], 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
        except Exception:
            pass
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining cutting-plane LP, smoothed mirror descent,
    Nesterov projected subgradient, spectral shaping, and L-BFGS polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")
    
    n_parent = len(best_seq)
    
    def remaining():
        return deadline - time.monotonic()
    
    # === Phase 1: Cutting-plane LP (main workhorse) ===
    print(f"\n=== Phase 1: Cutting-Plane LP (n={n_parent}) ===")
    if remaining() > 60:
        a_cp = cutting_plane_lp(best_seq, max_iters=50, tol=1e-9, verbose=True)
        obj = compute_obj(a_cp)
        if obj < best_score:
            best_score = obj
            best_seq = a_cp.copy()
        else:
            print(f"  CP LP: obj={obj:.10f} (no improvement over {best_score:.10f})")
    
    # === Phase 2: Active-set LP refinement ===
    print(f"\n=== Phase 2: Active-Set LP (n={len(best_seq)}) ===")
    if remaining() > 40:
        a_as = active_set_lp(best_seq, verbose=True)
        obj = compute_obj(a_as)
        if obj < best_score:
            best_score = obj
            best_seq = a_as.copy()
        else:
            print(f"  Active-set: obj={obj:.10f}")
    
    # === Phase 3: Smoothed Mirror Descent ===
    print(f"\n=== Phase 3: Smoothed Mirror Descent (n={len(best_seq)}) ===")
    if remaining() > 40:
        a_md = mirror_descent_smoothed(best_seq, time_budget=min(remaining() * 0.15, 90.0), verbose=True)
        obj = compute_obj(a_md)
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()
        else:
            print(f"  Mirror descent: obj={obj:.10f}")
    
    # === Phase 4: Spectral Shaping ===
    print(f"\n=== Phase 4: Spectral Shaping (n={len(best_seq)}) ===")
    if remaining() > 30:
        for alpha in [0.6, 0.7, 0.8, 0.5]:
            if remaining() < 10:
                break
            a_spec = spectral_shaping(best_seq, alpha=alpha, iterations=30, verbose=False)
            obj = compute_obj(a_spec)
            if obj < best_score:
                best_score = obj
                best_seq = a_spec.copy()
                print(f"  Spectral alpha={alpha}: obj={best_score:.10f} **")
            else:
                print(f"  Spectral alpha={alpha}: obj={obj:.10f}")
    
    # === Phase 5: L-BFGS-B Polish ===
    print(f"\n=== Phase 5: L-BFGS-B Polish (n={len(best_seq)}) ===")
    if remaining() > 30:
        a_lbf = lbfgs_smoothed_polish(best_seq, time_budget=min(remaining() * 0.15, 60.0), verbose=True)
        obj = compute_obj(a_lbf)
        if obj < best_score:
            best_score = obj
            best_seq = a_lbf.copy()
        else:
            print(f"  L-BFGS: obj={obj:.10f}")
    
    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_current = len(best_seq)
    n_candidates = sorted(set([
        n_current - 100, n_current - 50, n_current,
        n_current + 50, n_current + 100, n_current + 200,
        n_current + 300,
        500, 600, 700, 900, 1000, 1100, 1200, 1400, 1600, 2000
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 3000]
    
    for n_try in n_candidates:
        if remaining() < 30:
            break
        
        if n_try == n_current:
            continue
        
        a_try = resample_sequence(best_seq, n_try)
        
        # Quick optimization: mirror descent + one CP step
        a_opt = mirror_descent_smoothed(a_try, time_budget=min(remaining() * 0.05, 20.0))
        if remaining() > 15:
            a_opt = cutting_plane_lp(a_opt, max_iters=10, tol=1e-8)
        
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")
    
    # === Phase 7: Intensify on best ===
    print(f"\n=== Phase 7: Intensify (n={len(best_seq)}) ===")
    if remaining() > 30:
        # Another round of CP LP
        a_cp2 = cutting_plane_lp(best_seq, max_iters=30, tol=1e-10, verbose=True)
        obj = compute_obj(a_cp2)
        if obj < best_score:
            best_score = obj
            best_seq = a_cp2.copy()
            print(f"  CP2: obj={best_score:.10f} **")
        
        # Mirror descent again
        if remaining() > 20:
            a_md2 = mirror_descent_smoothed(best_seq, time_budget=min(remaining() * 0.2, 60.0), verbose=True)
            obj = compute_obj(a_md2)
            if obj < best_score:
                best_score = obj
                best_seq = a_md2.copy()
                print(f"  MD2: obj={best_score:.10f} **")
    
    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    if remaining() > 20:
        # L-BFGS with very fine temperature
        a_f = lbfgs_smoothed_polish(best_seq, time_budget=min(remaining() * 0.3, 40.0), verbose=True)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final LBF: obj={best_score:.10f} **")
        
        # One more CP
        if remaining() > 10:
            a_f2 = cutting_plane_lp(best_seq, max_iters=20, tol=1e-10)
            obj = compute_obj(a_f2)
            if obj < best_score:
                best_score = obj
                best_seq = a_f2.copy()
                print(f"  Final CP: obj={best_score:.10f} **")
        
        # Active set with all constraints
        if remaining() > 10:
            a_f3 = active_set_lp(best_seq)
            obj = compute_obj(a_f3)
            if obj < best_score:
                best_score = obj
                best_seq = a_f3.copy()
                print(f"  Final AS: obj={best_score:.10f} **")
    
    # === Phase 9: Random perturbation + restart ===
    print(f"\n=== Phase 9: Perturbation Restarts ===")
    n_best = len(best_seq)
    if remaining() > 30:
        for r in range(5):
            if remaining() < 20:
                break
            
            if r == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            elif r == 1:
                a_r = best_seq * np.exp(rng.normal(0, 0.01, n_best))
            elif r == 2:
                # Add small random component
                a_r = best_seq + 0.01 * rng.dirichlet(np.ones(n_best) * 10)
            elif r == 3:
                # Symmetric perturbation
                a_r = best_seq + 0.02 * rng.normal(0, 1, n_best)
                a_r = (a_r + a_r[::-1]) / 2.0
            else:
                # Shift the distribution slightly
                shift = rng.integers(-20, 20)
                a_r = np.roll(best_seq, shift)
                a_r = np.maximum(a_r, 0)
            
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / np.sum(a_r)
            
            a_opt = mirror_descent_smoothed(a_r, time_budget=min(remaining() * 0.1, 25.0))
            if remaining() > 15:
                a_opt = cutting_plane_lp(a_opt, max_iters=15, tol=1e-9)
            
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.10f} **")
            else:
                print(f"  Restart {r}: obj={obj:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
