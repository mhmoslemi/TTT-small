import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value (normalized)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def soft_max(values, tau=0.01):
    """Compute soft maximum."""
    values = np.asarray(values)
    shifted = values - np.max(values)
    result = tau * np.log(np.sum(np.exp(shifted / tau))) + np.max(values)
    return float(result)

def soft_max_weights(conv, tau=0.01):
    """Compute weights for each lag based on soft-max."""
    shifted = conv - np.max(conv)
    weights = np.exp(shifted / tau)
    weights /= np.sum(weights)
    return weights

def mwu_gradient(a, weights):
    """Compute weighted gradient of (a*a)_k."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        # (B_k * a)_j = a[k-j] for valid j
        for j in range(j_start, j_end):
            grad[j] += weights[k] * a[k - j]
    return grad

def solve_lp(a, solver=None, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix A where (A @ g)[k] = conv(a,g)[k]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[j] * g[k-j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] = sum_j a[j] * g[k-j], so a[j] * g[m] contributes to conv[j+m]
        # So A[j+m, j] = a[j] for m = 0, ..., n-1
        for m in range(n):
            A[j + m, j] = a[j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    if warm_start is not None:
        g.value = warm_start
        M.value = compute_obj(a) / (2.0 * n)  # approximate
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    return None, None

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build A (2n-1 x n) where A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp_for_g(a, warm_g=None):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Try cvxpy with warm start
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    if warm_g is not None:
        g.value = warm_g
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except:
            continue
    
    # Fallback: scipy linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.array([[1.0] * n + [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    
    return None, None

def subgradient_update(a, top_k=20, eta=0.01):
    """One subgradient step targeting top-K peaks."""
    n = len(a)
    conv = compute_conv(a)
    k_indices = np.argsort(conv)[::-1][:top_k]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in k_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            grad[j] += a[k - j]
    grad /= top_k
    
    # Update
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new /= s
    return a_new

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def nesterov_subgradient(a, time_budget=100.0, top_k=20, eta_init=0.02):
    """Nesterov-accelerated subgradient method."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    a_prev = a.copy()
    eta = eta_init
    
    t = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        beta = (1.0 + np.sqrt(1.0 + 4.0 / (t * t))) / 2.0
        beta_prev = (1.0 + np.sqrt(1.0 + 4.0 / ((t-1) * (t-1)))) / 2.0 if t > 1 else 1.0
        
        if t > 1:
            y = a + (beta_prev - 1.0) / beta * (a - a_prev)
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy > 1e-10:
                y = y / sy
        else:
            y = a.copy()
        
        # Subgradient step at y
        conv = compute_conv(y)
        k_indices = np.argsort(conv)[::-1][:top_k]
        
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                grad[j] += y[k - j]
        grad /= top_k
        
        # Step
        a_new = y - eta * grad
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a_prev = a.copy()
        a = a_new
        
        # Adaptive eta
        if t % 100 == 0:
            eta = max(eta * 0.95, 0.001)
        
        # Perturbation if stuck
        if no_improve > 100:
            rng = np.random.default_rng(t * 7919)
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = project_simplex(a)
            a_prev = a.copy()
            eta = 0.02
            no_improve = 0
    
    return best_a
