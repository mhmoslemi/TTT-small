def subgradient_polish(a, time_budget=60.0, n_peaks=8):
    """Subgradient method with multi-peak weighted gradient and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Projected subgradient onto simplex {x: sum(x)=1, x>=0}
    def project_simplex(v):
        """Project v onto the probability simplex."""
        n_v = len(v)
        if np.sum(v) <= 0:
            return np.ones(n_v) / n_v
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.where(u * np.arange(1, n_v + 1) > (cssv - 1))[0]
        if len(rho) == 0:
            return np.ones(n_v) / n_v
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        w = np.maximum(v - theta, 0)
        s_w = float(np.sum(w))
        if s_w < 1e-15:
            return np.ones(n_v) / n_v
        return w / s_w

    momentum = np.zeros(n)
    beta = 0.7
    alpha_base = 0.3

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Get all bins within 5% of max
        threshold = max_val * 0.95
        active_bins = np.where(conv >= threshold)[0]
        if len(active_bins) == 0:
            active_bins = np.array([np.argmax(conv)])

        # Weight by proximity to max
        weights = np.exp(-(max_val - conv[active_bins]) / (0.01 * max_val + 1e-10))
        weights /= weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active_bins):
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * a[j]

        # Nesterov-like momentum
        momentum = beta * momentum + (1 - beta) * grad
        alpha = alpha_base / (1.0 + 0.01 * it)

        a_new = project_simplex(a - alpha * momentum)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            alpha_base *= 0.97
            if alpha_base < 0.01:
                break

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via Fourier method."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n * n)
    # Fourier resampling
    L = 1
    while L < max(n, new_n):
        L <<= 1
    A = fft(a, L)
    if new_n > n:
        # Zero-pad in frequency (interpolate)
        A_new = np.zeros(L, dtype=complex)
        idx = np.arange(L)
       房租 = (idx < (L + new_n) // 2) | (idx > (L - new_n) // 2)
        A_new[idx] = A[idx]
        a_new = ifft(A_new).real[:new_n]
    else:
        # Truncate
        A_new = np.zeros(L, dtype=complex)
        half = new_n // 2
        A_new[:half+1] = A[:half+1]
        A_new[L-half:] = A[L-half:L]
        a_new = ifft(A_new).real[:new_n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def random_init(n, rng):
    """Generate a structured random initial sequence."""
    mode = int(rng.integers(0, 6))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Fourier random
        freqs = rng.uniform(0.5, 5, 10)
        phases = rng.uniform(0, 2 * np.pi, 10)
        amps = rng.uniform(0.1, 0.4, 10)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 2:
        # Gaussian mixture
        centers = rng.uniform(0.1, 0.9, 3) * n
        widths = rng.uniform(0.05, 0.2, 3) * n
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-(x - c) ** 2 / (2 * w ** 2))
        a = np.maximum(a, 0.001)
        a = a / np.sum(a)
    elif mode == 3:
        # Periodic + noise
        freq = rng.uniform(2, 8)
        x = np.arange(n) / n
        a = 1 + 0.3 * np.cos(2 * np.pi * freq * x)
        a += 0.05 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 4:
        # Block structure
        n_blocks = int(rng.integers(3, 8))
        block_size = n // n_blocks
        amplitudes = rng.uniform(0.5, 1.5, n_blocks)
        a = np.zeros(n)
        for b in range(n_blocks):
            start = b * block_size
            end = min(n, (b + 1) * block_size)
            a[start:end] = amplitudes[b]
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    else:
        # Inverse-cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2 + 0.1
        a = a / np.sum(a)
    return a

    Main search combining LP alternating (HiGHS), subgradient polishing,
    mirror descent, FFT flat spectrum, coordinate refinement, multi-n
    exploration, and structured random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting optimization with n={n_parent}, score={best_score:.8f}")

    # Phase 1: LP alternating (primary optimizer)
    print(f"\n=== Phase 1: LP Alternating (main) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 400.0)
        a_lp = lp_alternating_v2(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  LP main: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # Phase 2: Subgradient polish
    print(f"\n=== Phase 2: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 120.0)
        a_sg = subgradient_polish(best_seq, time_budget=alloc, n_peaks=10)
        obj = compute_obj(a_sg)
        print(f"  Subgradient: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()

    # Phase 3: Mirror descent
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.12, 100.0)
        a_md = mirror_descent_v2(best_seq, time_budget=alloc, n_peaks=12)
        obj = compute_obj(a_md)
        print(f"  Mirror descent: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # Phase 4: Second LP round (refined)
    print(f"\n=== Phase 4: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 150.0)
        a_lp2 = lp_alternating_v2(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp2)
        print(f"  LP round 2: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        500, 600, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and abs(n - n_best) >= 30]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: LP + subgradient
        alloc_lp = min(remaining * 0.15, 40.0)
        a_opt = lp_alternating_v2(a_try, time_budget=alloc_lp)
        alloc_sg = min(remaining * 0.1, 20.0)
        a_opt = subgradient_polish(a_opt, time_budget=alloc_sg)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try

    # Phase 6: Random restarts
    print(f"\n=== Phase 6: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    n_restarts = min(8, int(remaining // 30))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Different initialization strategies
        init_type = r % 4
        if init_type == 0:
            a_r = random_init(n_best, rng)
        elif init_type == 1:
            # Perturbed best with Fourier noise
            freqs = rng.uniform(1, 20, 5)
            x = np.arange(n_best) / n_best
            noise = np.zeros(n_best)
            for f in freqs:
                noise += rng.uniform(0.02, 0.08) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
            a_r = best_seq + noise
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            # Different n
            n_r = int(n_best + rng.integers(-200, 200))
            n_r = max(400, min(1200, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            # Mirror of best
            a_r = best_seq[::-1].copy()
            a_r = a_r * np.exp(rng.normal(0, 0.03, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        # Optimize the restart
        alloc_lp = min(remaining * 0.25, 50.0)
        a_opt = lp_alternating_v2(a_r, time_budget=alloc_lp)
        alloc_sg = min(remaining * 0.15, 20.0)
        a_opt = subgradient_polish(a_opt, time_budget=alloc_sg)

        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # Phase 7: Final intensive polish on best
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()

    # LP
    if remaining > 40:
        alloc = min(remaining * 0.4, 100.0)
        a_lp = lp_alternating_v2(a_final, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            a_final = a_lp.copy()

    # Subgradient
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_sg = subgradient_polish(a_final, time_budget=alloc, n_peaks=15)
        obj = compute_obj(a_sg)
        print(f"  Final SG: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()
            a_final = a_sg.copy()

    # Coordinate refine
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_cd = coordinate_refine(a_final, time_budget=alloc)
        obj = compute_obj(a_cd)
        print(f"  Final CD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()

    # Mirror descent final
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 30.0)
        a_md = mirror_descent_v2(a_final, time_budget=alloc, n_peaks=15)
        obj = compute_obj(a_md)
        print(f"  Final MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 60.0)
        a_lp = lp_alternating_v2(best_seq, time_budget=alloc)
        obj = compute_obj(a_lp)
        print(f"  Final LP2: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
