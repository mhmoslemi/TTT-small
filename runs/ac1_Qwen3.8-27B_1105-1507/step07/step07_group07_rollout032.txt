from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)

        return np.array([a[0]*a[0]])
    size = next_pow2(2*n-1)
    fa = rfft(a, size)

        a_new = a_new / s_new
        a = a_new

    # LP: min M
    # s.t. A @ g - M <= 0  (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = g_val.sum()
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_polish(a, time_budget, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = lp_solve_minpeak(a)
        if g_lp is None:
            break

        a_new = a.copy()
        improved = False
        for t in [0.5, 0.3, 0.7, 1.0, 0.2, 0.8]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if improved:
            stall = 0
            if verbose:
                print(f"    lp step: obj={best_obj:.10f}")
        else:
            stall += 1
            if stall > 5:
                break

    return best_a


def gs_init(n, n_iter=200, rng=None):
    """
    Gerchberg-Saxton style initialization:
    Randomize phase, enforce flat magnitude, take positive part.
    Produces a sequence with near-flat spectrum.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    N = next_pow2(2 * n - 1)
    # Random phase
    phase = rng.uniform(-np.pi, np.pi, N // 2 + 1)
    phase[0] = 0  # DC is real

    # Target magnitude: flat (except DC)
    target_mag = np.ones(N // 2 + 1) / np.sqrt(n)
    target_mag[0] = 1.0  # DC = sum(a) = 1

    A = target_mag * np.exp(1j * phase)

    a = np.zeros(n)
    for it in range(n_iter):
        A = target_mag.copy() * np.exp(1j * np.angle(A))
        a_full = np.fft.ifft(A).real[:n]
        a = np.maximum(a_full, 0)
        s = a.sum()
        if s < 1e-10:
            # Reset with random phase
            phase = rng.uniform(-np.pi, np.pi, N // 2 + 1)
            phase[0] = 0
            A = target_mag * np.exp(1j * phase)
            continue
        a = a / s
        A = np.fft.fft(a, N)

    return a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_shrink(a, n_iter=200, alpha=0.005):
    """
    Direct peak-reduction heuristic: iteratively shrink elements that
    contribute to the current autocorrelation peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(n_iter):
        c = compute_conv(a)
        k_star = int(np.argmax(c))

        # Contribution of each a[i] to c[k_star]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[j]

        max_contrib = contrib.max()
        if max_contrib < 1e-12:
            break

        # Shrink contributing elements
        a_new = a * (1.0 - alpha * (contrib / max_contrib))
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            alpha *= 0.9
            if alpha < 1e-6:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Adam subgradient + top-k, GS init, LP polish, multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)

    # === Phase 1: GS frequency-domain init for diversity ===
    print(f"\n=== Phase 1: GS Init + Adam (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        for gs_try in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            gs_seq = gs_init(n_parent, n_iter=100 + 50 * gs_try, rng=rng)
            # Quick adam from GS init
            alloc = min(remaining * 0.15, 40.0)
            a_gs = adam_optimize(gs_seq, time_budget=alloc, top_k=3)
            obj = compute_obj(a_gs)
            if obj < best_score:
                best_score = obj
                best_seq = a_gs.copy()
                print(f"  GS init {gs_try}: obj={best_score:.10f} **")
            else:
                print(f"  GS init {gs_try}: obj={obj:.10f}")

    # === Phase 2: Adam with top-1 (hard max) ===
    print(f"\n=== Phase 2: Adam top-1 (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 200.0)
        a_opt = adam_optimize(best_seq, time_budget=alloc, top_k=1, lr_init=0.03, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Adam top-1: obj={best_score:.10f} **")
        else:
            print(f"  Adam top-1: obj={obj:.10f}")

    # === Phase 3: Adam with top-k smoothing (k=3,5) ===
    print(f"\n=== Phase 3: Adam top-k smoothing ===")
    for k in [3, 5]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining * 0.15, 120.0)
        a_opt = adam_optimize(best_seq, time_budget=alloc, top_k=k, lr_init=0.02, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Adam top-{k}: obj={best_score:.10f} **")
        else:
            print(f"  Adam top-{k}: obj={obj:.10f}")

    # === Phase 4: Peak shrink heuristic ===
    print(f"\n=== Phase 4: Peak Shrink ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_shrink = peak_shrink(best_seq, n_iter=500, alpha=0.003)
        obj = compute_obj(a_shrink)
        if obj < best_score:
            best_score = obj
            best_seq = a_shrink.copy()
            print(f"  Peak shrink: obj={best_score:.10f} **")
        else:
            print(f"  Peak shrink: obj={obj:.10f}")

    # === Phase 5: LP Polish ===
    print(f"\n=== Phase 5: LP Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60 and len(best_seq) <= 1200:
        alloc = min(remaining * 0.2, 120.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  LP polish: obj={best_score:.10f} **")
        else:
            print(f"  LP polish: obj={obj:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        n_best - 50, n_best, n_best + 50, n_best + 100,
        n_best + 200, n_best + 300,
        700, 800, 900, 1000, 1200, 1500
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)

        # Quick Adam
        alloc = min(remaining * 0.12, 60.0)
        a_opt = adam_optimize(a_try, time_budget=alloc, top_k=2, lr_init=0.02)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 7: Intensify on best ===
    print(f"\n=== Phase 7: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Adam top-1 with fine lr
        alloc = min(remaining * 0.25, 150.0)
        a_fine = adam_optimize(best_seq, time_budget=alloc, top_k=1, lr_init=0.01)
        obj = compute_obj(a_fine)
        if obj < best_score:
            best_score = obj
            best_seq = a_fine.copy()
            print(f"  Fine Adam: obj={best_score:.10f} **")
        else:
            print(f"  Fine Adam: obj={obj:.10f}")

    # === Phase 8: Final LP + Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40 and len(best_seq) <= 1200:
        alloc = min(remaining * 0.3, 80.0)
        a_lp2 = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp2)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj:.10f}")

    # === Phase 9: Final Adam polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 60.0)
        a_final = adam_optimize(best_seq, time_budget=alloc, top_k=1, lr_init=0.005)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final Adam: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    # Scale to avoid numerical issues in evaluation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        # Fallback
        result = [1.0 / 100] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
