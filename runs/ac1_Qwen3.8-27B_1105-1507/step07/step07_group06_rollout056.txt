def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Normalizes internally."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        A[j:j + n, j] = a[n - 1:0 * n - 1:-1][:n] if False else a[:n][::-1][n - j - 1:] if False else 0
    # Redo properly: conv(a,g)[k] = sum_j a[k-j]*g[j] for valid j
    # So A[k,j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Use linprog with HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_topk(a, K=3):
    """LP that minimizes weighted top-K peaks."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    conv_aa = compute_conv(a)
    
    # Weight: emphasize the current peaks
    sorted_idx = np.argsort(conv_aa)[::-1]
    weights = np.zeros(2 * n - 1)
    for i, idx in enumerate(sorted_idx[:K]):
        weights[idx] = 1.0 / (i + 1)
    
    # LP: min sum_k w[k]*conv(a,g)[k] + M * (max violation penalty)
    # Simplified: just minimize the weighted sum (LP)
    c_g = A.T @ weights  # c_g[j] = sum_k w[k]*A[k,j]
    
    # min c_g^T g s.t. sum(g)=1, g>=0
    # Solution: g = e_{argmin c_g} (vertex)
    # That's too degenerate. Add the M constraint.
    
    # Better: min M s.t. A@g <= M, sum(g)=1, g>=0 (same as before)
    # But with a different initial point suggestion
    return solve_lp_minpeak(a)


def smoothed_gradient(a, tau):
    """Compute gradient of 2n*smooth_max(conv(a), tau) w.r.t. a, with sum(a)=1 normalization."""
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    
    # Softmax weights
    exp_vals = np.exp(tau * (conv - conv_max))
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient: g[j] = 2n * tau * sum_k w[k] * d(conv[k])/da[j]
    # d(conv[k])/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # For efficiency, compute directly:
    # g[j] = 2n*tau * [sum_{m=0}^{n-1} w[j+m]*a[m] + sum_{m=0}^{n-1} w[j-m]*a[m]]
    #      (with w[k]=0 for k<0 or k>=2n-1)
    
    g = np.zeros(n)
    for j in range(n):
        # First sum: m from 0 to n-1, need j+m in [0, 2n-2]
        m_max1 = min(n - 1, 2 * n - 2 - j)
        if m_max1 >= 0:
            g[j] += np.dot(w[j:j + m_max1 + 1], a[:m_max1 + 1])
        # Second sum: m from max(1, j-(2n-2)) to n-1, need j-m in [0, 2n-2]
        m_min2 = max(1, j - (2 * n - 2))
        if m_min2 < n:
            g[j] += np.dot(w[j - n + 1:j], a[m_min2:n][::-1]) if j > 0 else 0
            # Let me redo this more carefully
            # We need w[j-m] * a[m] for m where 0 <= j-m < 2n-1
            # i.e., j-(2n-2) <= m <= j, and 0 <= m < n
            m_lo = max(0, j - (2 * n - 2))
            m_hi = min(n - 1, j)
            if m_hi >= m_lo:
                indices = np.arange(m_lo, m_hi + 1)
                g[j] += np.dot(w[j - indices], a[indices])
    
    # Subtract mean for simplex projection
    g = g - np.mean(g)
    return 2.0 * n * tau * g


def gradient_descent_smoothed(a_init, time_budget=60.0):
    """Gradient descent on smoothed max with adaptive step size."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Multi-stage tau annealing
    tau_schedule = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
    step_size = 0.3
    
    for tau in tau_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        tau_budget = min(10.0, (time_budget - (time.monotonic() - start)) * 0.8)
        tau_start = time.monotonic()
        
        while time.monotonic() - tau_start < tau_budget:
            grad = smoothed_gradient(a, tau)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-10:
                break
            
            # Line search
            improved = False
            for alpha in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1]:
                a_try = a - alpha * grad / grad_norm
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    step_size = min(step_size * 1.05, 2.0)
                    break
            
            if not improved:
                step_size *= 0.8
                if step_size < 0.001:
                    break
    
    return best_a, best_obj

def make_initialization(n, init_type, rng):
    """Create diverse asymmetric initializations."""
    if init_type == 0:
        # Linear ramp (increasing)
        a = np.linspace(1.0, 3.0, n)
    elif init_type == 1:
        # Linear ramp (decreasing)
        a = np.linspace(3.0, 1.0, n)
    elif init_type == 2:
        # Quadratic ramp
        a = np.linspace(1.0, 1.0, n)
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * t**2
    elif init_type == 3:
        # U-shape (high at edges)
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (2 * t - 1)**2
    elif init_type == 4:
        # Inverse U (high in middle)
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (1 - (2 * t - 1)**2)
    elif init_type == 5:
        # Exponential ramp
        rate = rng.uniform(1.0, 4.0)
        a = np.exp(np.linspace(0, rate, n))
    elif init_type == 6:
        # Sine-based asymmetry
        t = np.linspace(0, np.pi, n)
        a = 1.0 + np.sin(t) + 0.5 * np.cos(t)
        a = np.maximum(a, 0.1)
    elif init_type == 7:
        # Two-bump asymmetric
        t = np.linspace(0, 1, n)
        a = 1.0 + np.exp(-((t - 0.2) / 0.15)**2) + 0.5 * np.exp(-((t - 0.8) / 0.1)**2)
    elif init_type == 8:
        # Power law
        alpha = rng.uniform(0.3, 1.5)
        t = np.linspace(0.01, 1.0, n)
        a = t**alpha
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(3):
            freq = rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.linspace(0, 1, n)
            a += 0.3 * np.sin(2 * np.pi * freq * t + phase)
        a = np.maximum(a, 0.1)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            print(f"  Final Grad: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # One more LP
        alloc = min(remaining * 0.5, 30.0)
        a_lp = lp_alternating_improved(best_seq, time_budget=alloc, max_iters=100)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
