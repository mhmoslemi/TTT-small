def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)

    while size < 2 * n - 1:

    fa = rfft(a, size)

def compute_obj(a):

    n = len(a)
    s = float(np.sum(a))

    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))

def subgradient_at_peak(a):
    """
    Compute the subgradient of max(conv(a,a)) with respect to a.
    If k* = argmax(conv(a,a)), then the subgradient component for a[i] is
    sum over j such that i+j = k* of a[j] (with appropriate bounds).
    
    Actually, for conv(a,a)[k] = sum_{i+j=k} a[i]*a[j], the derivative w.r.t. a[i]
    is sum_{j: i+j=k, j valid} a[j].
    
    For the max, we pick the k* that maximizes, and the subgradient is the gradient
    at that particular k*.
    """
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    # Subgradient: for each i in [0, n-1], grad[i] = sum of a[j] where i+j = k_star and 0<=j<n
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
    
    return grad, conv, k_star

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    # Algorithm from Duchi et al. (2008)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

Uses top-k subgradient for smoother optimization, combined with exact LP polish.

from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_single(a):
    """Subgradient at the single peak."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        j = k_star - i
        grad[i] = a[j]
    return grad, conv, k_star

        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            j = k_star - i

        x_new = project_simplex(x_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = project_simplex(y_new)
        
        x = x_new
        y = y_new
        t = t_new
        
        chk = (k + 1)
        if chk % 100 == 0 or chk < 10:
            obj = compute_obj(x)
            if obj < best_obj:
                best_obj = obj
                best_a = x.copy()
                best_iter = k + 1
    
    obj = compute_obj(x)
    if obj < best_obj:
        best_obj = obj
        best_a = x.copy()
    
    return best_a, best_obj


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    # Fallback: linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None


def lp_alternating(a, time_budget, max_iter=500):
    """LP alternating with perturbation to escape local minima."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        best_t_obj = best_obj
        best_a_local = a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            # Perturb
            noise_scale = 0.03 * (1.0 + it * 0.01)
            a = best_a * np.exp(rng.normal(0, noise_scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
    
    return best_a


def smoothed_lbfgs(a_init, time_budget):
    """Smoothed L-BFGS-B with log-softmax parameterization."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / np.sum(a_init)
    
    from scipy.optimize import minimize
    
    def softmax_from_theta(theta):
        theta_shift = theta - np.max(theta)
        exp_t = np.exp(theta_shift)
        return exp_t / np.sum(exp_t)
    
    def smooth_obj_and_grad(theta, tau):
        a = softmax_from_theta(theta)
        conv = compute_conv(a)
        conv_max = np.max(conv)
        obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)
        
        w = tau * np.exp((conv - conv_max) * tau)
        w = w / np.sum(w)
        
        c = np.zeros(n)
        for j in range(n):
            end = min(2 * n - 1, j + n)
            if j < 2 * n - 1:
                c[j] = np.dot(w[j:end], a[:end - j])
        
        mean_c = np.dot(c, a)
        grad_theta = 4.0 * n * a * (c - mean_c)
        return obj_val, grad_theta
    
    theta = np.log(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    taus = [10.0, 30.0, 100.0, 300.0, 500.0]
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        def obj_func(th):
            val, _ = smooth_obj_and_grad(th, tau)
            return val
        
        def grad_func(th):
            _, g = smooth_obj_and_grad(th, tau)
            return g
        
        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': 100, 'gtol': 1e-7, 'maxfun': 300})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Nesterov-accelerated subgradient + aggressive LP alternating + geometric interpolation
+ smoothed L-BFGS-B + multi-length exploration.
Key improvements: top-k subgradient for smoother Nesterov updates, geometric interpolation

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_single(a):
    """Subgradient at the single peak of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        j = k_star - i
        grad[i] = a[j]
    return grad, conv, k_star


def subgradient_topk(a, k=3):
    """Subgradient using convex combination of top-k peak subgradients."""
    n = len(a)
    conv = compute_conv(a)
    if k >= 2 * n - 1:
        k = 2 * n - 1
    top_k = np.argpartition(conv, -k)[-k:]
    vals = conv[top_k]
    w = vals - vals.min() + 0.01
    w = w / w.sum()

    grad = np.zeros(n)
    for idx, wi in zip(top_k, w):
        k_star = int(idx)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            j = k_star - i
            grad[i] += wi * a[j]

    return grad, conv, int(top_k[0])


def nesterov_optimize(a_init, time_budget, topk=1, max_iter=200000, step_scale=1.0):

        else:
            grad, conv, _ = subgradient_topk(y, topk)

        x_new = y - step * grad
        x_new = project_simplex(x_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = project_simplex(y_new)

        x = x_new
        y = y_new
        t = t_new

        chk = k + 1
        if chk % 200 == 0 or chk < 10:

            if obj < best_obj:
                best_obj = obj
                best_a = x.copy()

    if obj < best_obj:
        best_obj = obj

    return best_a, best_obj


def mirror_descent(a_init, time_budget, topk=1, max_iter=100000, eta_scale=1.0):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()

    for k in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        if topk == 1:
            grad, conv, _ = subgradient_single(a)
        else:
            grad, conv, _ = subgradient_topk(a, topk)

        eta = eta_scale / np.sqrt(k + 1.0)

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-15:
            a_new = a_new / s_new
            a = a_new

        chk = k + 1
        if chk % 100 == 0 or chk < 10:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def lp_alternating(a, time_budget, max_iter=500, use_geometric=False):
    """LP alternating with adaptive perturbations and optional geometric interpolation."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)
    stuck_count = 0

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = a.copy()

        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            if use_geometric:
                a_try = np.power(np.maximum(a, 1e-15), 1.0 - t) * np.power(np.maximum(g_lp, 1e-15), t)
            else:
                a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count >= 3:
                noise_scale = 0.02 + 0.02 * min(stuck_count, 10)
                if rng.random() < 0.3:
                    a = best_a * np.exp(rng.normal(0, noise_scale, n))
                else:
                    shift = rng.uniform(0.05, 0.2)
                    ramp = np.linspace(1.0 - shift, 1.0 + shift, n)
                    a = best_a * ramp
                a = np.maximum(a, 1e-15)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stuck_count = 0

    return best_a


def smoothed_lbfgs(a_init, time_budget):
    """Smoothed L-BFGS-B with log-softmax parameterization and tau annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / np.sum(a_init)

    def softmax_from_theta(theta):
        theta_shift = theta - np.max(theta)
        exp_t = np.exp(theta_shift)
        return exp_t / np.sum(exp_t)

    def smooth_obj_and_grad(theta, tau):
        a = softmax_from_theta(theta)
        conv = compute_conv(a)
        conv_max = np.max(conv)
        obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)

        w = tau * np.exp((conv - conv_max) * tau)
        w = w / np.sum(w)

        c = np.zeros(n)
        for j in range(n):
            end = min(2 * n - 1, j + n)
            if j < 2 * n - 1:
                c[j] = np.dot(w[j:end], a[:end - j])

        mean_c = np.dot(c, a)
        grad_theta = 4.0 * n * a * (c - mean_c)
        return obj_val, grad_theta

    theta = np.log(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    taus = [5.0, 15.0, 50.0, 150.0, 400.0]
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        def obj_func(th):
            val, _ = smooth_obj_and_grad(th, tau)
            return val

        def grad_func(th):
            _, g = smooth_obj_and_grad(th, tau)
            return g

        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': 80, 'gtol': 1e-7, 'maxfun': 250})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget):
    """Targeted coordinate descent on peak-contributing coordinates."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Move mass from peak contributors to non-contributors
        contributors = []
        for i in range(max(0, k_star - n + 1), min(n, k_star + 1)):
            contributors.append(i)

        non_contributors = [i for i in range(n) if i not in set(contributors)]

        if non_contributors and contributors:
            for i in contributors[:min(20, len(contributors))]:
                if time.monotonic() - start > time_budget:
                    break
                j = non_contributors[rng.integers(0, len(non_contributors))]
                delta = 0.01 * a[i]
                if delta > 0.9 * a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta

        if not improved:
            # Random moves
            for _ in range(30):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.005 * a[i] * rng.uniform(0.5, 1.5)
                if delta > 0.8 * a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s_a = np.sum(a)
                if s_a < 1e-10:
                    a = best_a.copy()
                    break
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta

            if not improved:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining Nesterov subgradient, mirror descent, LP alternating,
    smoothed L-BFGS-B, coordinate descent, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: Nesterov with various configurations
    print(f"\n=== Phase 1: Nesterov / Mirror Descent ===")
    configs = [
        ("nested_topk1", lambda: nesterov_optimize(best_seq, 80.0, topk=1, step_scale=1.0)),
        ("nested_topk3", lambda: nesterov_optimize(best_seq, 80.0, topk=3, step_scale=1.0)),
        ("mirror_topk1", lambda: mirror_descent(best_seq, 60.0, topk=1, eta_scale=0.5)),
        ("mirror_topk3", lambda: mirror_descent(best_seq, 60.0, topk=3, eta_scale=0.3)),
    ]

    for name, fn in configs:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_opt, obj_opt = fn()
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  {name}: obj={best_score:.8f} **")
        else:
            print(f"  {name}: obj={obj_opt:.8f}")

    # Phase 2: LP alternating (primary, long run)
    print(f"\n=== Phase 2: LP Alternating (long) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.28, 250.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, use_geometric=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP (linear): obj={best_score:.8f} **")
        else:
            print(f"  LP (linear): obj={obj_lp:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc2 = min(remaining * 0.15, 100.0)
            a_lp2 = lp_alternating(best_seq, time_budget=alloc2, use_geometric=True)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP (geometric): obj={best_score:.8f} **")
            else:
                print(f"  LP (geometric): obj={obj_lp2:.8f}")

    # Phase 3: Smoothed L-BFGS-B
    print(f"\n=== Phase 3: Smoothed L-BFGS-B ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 100.0)
        a_opt, obj_opt = smoothed_lbfgs(best_seq, time_budget=alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  L-BFGS-B: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS-B: obj={obj_opt:.8f}")

    # Phase 4: Coordinate descent
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 50.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 200), n_cur - 100, n_cur - 50,
        n_cur, n_cur + 50, n_cur + 100, n_cur + 200, n_cur + 400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if n_try == n_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Nesterov on resampled
        alloc1 = min(remaining * 0.10, 40.0)
        a_opt1, obj1 = nesterov_optimize(a_try, time_budget=alloc1, topk=1)

        # LP polish
        alloc2 = min(remaining * 0.08, 30.0)
        a_opt2 = lp_alternating(a_opt1, time_budget=alloc2)
        obj2 = compute_obj(a_opt2)

        if obj1 <= obj2:
            best_local = obj1
            best_local_a = a_opt1
        else:
            best_local = obj2
            best_local_a = a_opt2

        if best_local < best_score:
            best_score = best_local
            best_seq = best_local_a.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={best_local:.8f}")

    # Phase 6: Random restarts with diverse initializations
    print(f"\n=== Phase 6: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    inits = []
    for slope in [1.5, 2.5, 4.0]:
        a_r = np.linspace(1.0, slope, n_best)
        inits.append(("ramp_s" + str(slope), a_r / np.sum(a_r)))
    for rate in [1.0, 2.5, 5.0]:
        a_r = np.exp(np.linspace(0, rate, n_best))
        inits.append(("exp_r" + str(rate), a_r / np.sum(a_r)))
    for blend in [0.3, 0.5, 0.7]:
        ramp = np.linspace(0.3, 1.7, n_best)
        ramp = ramp / np.sum(ramp)
        a_r = blend * best_seq + (1.0 - blend) * ramp
        inits.append(("blend_" + str(blend), a_r / np.sum(a_r)))
    # Gaussian bump (asymmetric)
    for center in [0.25, 0.4, 0.6]:
        x = np.linspace(0, 1, n_best)
        a_r = np.exp(-((x - center) / 0.15) ** 2)
        a_r = np.maximum(a_r, 1e-10)
        inits.append(("bump_c" + str(center), a_r / np.sum(a_r)))

    for idx, (name, a_r) in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        alloc = min(remaining * 0.08, 35.0)
        a_opt, obj_opt = nesterov_optimize(a_r, time_budget=alloc, topk=3)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  {name}: obj={best_score:.8f} **")
        else:
            print(f"  {name}: obj={obj_opt:.8f}")

    # Phase 7: Final intensive polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.35, 100.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, use_geometric=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.30, 60.0)
            a_opt, obj_opt = nesterov_optimize(best_seq, time_budget=alloc, topk=1, step_scale=1.0)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Final Nesterov: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.30, 50.0)
            a_opt, obj_opt = smoothed_lbfgs(best_seq, time_budget=alloc)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Final L-BFGS-B: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
