"""
Smoothed subgradient + LP alternating optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: (1) direct peak-focused subgradient using top-k peaks and 
log-sum-exp smoothing, (2) power-law initializations based on asymptotic analysis, 
(3) adaptive LP polishing, (4) multi-length exploration, (5) coordinate descent refinement.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective for normalized a (sum=1). Returns 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    p = a / s
    b = compute_conv(p)
    return 2.0 * n * float(np.max(b))


def compute_obj_normalized(p):
    """Compute objective assuming p is already normalized (sum=1)."""
    n = len(p)
    b = compute_conv(p)
    return 2.0 * n * float(np.max(b))


def normalize(p):
    """Normalize to sum=1, non-negative."""
    p = np.maximum(p, 0)
    s = float(np.sum(p))
    if s < 1e-15:
        return np.ones(len(p)) / len(p)
    return p / s


def topk_peak_gradient(p, b, k=5):
    """
    Compute the subgradient of max(b) using top-k peaks.
    Returns gradient w.r.t. p (length n).
    """
    n = len(p)
    # Find top-k peak positions
    if k >= len(b):
        peak_indices = np.arange(len(b))
    else:
        peak_indices = np.argpartition(b, -k)[-k:]
    
    grad = np.zeros(n)
    for k_idx in peak_indices:
        # grad[j] += 2 * p[k_idx - j] if 0 <= k_idx - j < n
        j_min = max(0, k_idx - n + 1)
        j_max = min(n - 1, k_idx)
        if j_min <= j_max:
            # p[k_idx - j] for j in [j_min, j_max]
            # When j = j_min, k_idx - j = k_idx - j_min (max value)
            # When j = j_max, k_idx - j = k_idx - j_max (min value)
            idx = k_idx - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += 2.0 * p[idx]
    
    grad /= k  # Average over top-k
    return grad


def lse_gradient(p, b, tau):
    """
    Compute gradient of smoothed objective (1/tau)*log(sum exp(tau*b)) using FFT.
    Returns gradient (length n), not multiplied by 2*n.
    """
    n = len(p)
    
    # Softmax weights
    b_shifted = b - np.max(b)
    w = np.exp(tau * b_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum
    
    # h[j] = sum_m w[m+j] * p[m] for m=0..n-1
    # Using FFT cross-correlation
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fw = rfft(w, N)
    fp = rfft(p, N)
    h_full = irfft(fw * np.conj(fp), N)[:n]
    
    return 2.0 * h_full


def smooth_subgrad_descent(p, time_budget=60.0, verbose=False):
    """
    Smoothed subgradient descent on the peak of self-convolution.
    Uses adaptive temperature and step size.
    """
    n = len(p)
    p = normalize(p.copy())
    best_p = p.copy()
    best_obj = compute_obj_normalized(p)
    start = time.monotonic()
    
    eta = 0.5
    tau = 10.0
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    max_stall = 50
    N_conv = 1
    while N_conv < 2 * n - 1:
        N_conv <<= 1
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution
        fp = rfft(p, N_conv)
        b = irfft(fp * fp, N_conv)[:2 * n - 1]
        
        # Compute gradient using LSE smoothing
        grad = lse_gradient(p, b, tau)
        
        # Also incorporate top-k peak gradient for focus
        if it > 20:
            grad_topk = topk_peak_gradient(p, b, k=3)
            grad = 0.5 * grad + 0.5 * grad_topk
        
        # Center gradient (to maintain normalization approximately)
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - eta * grad
        p_new = p + v
        p_new = np.maximum(p_new, 0)
        s = float(np.sum(p_new))
        if s < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        p_new = p_new / s

        b_new = irfft(fp_new * fp_new, N_conv)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

            best_obj = obj_new
            best_p = p_new.copy()
            p = p_new
            stall = 0

            tau = max(tau * 1.005, 50.0)  # Increase temperature (more focused)

            stall += 1
            eta *= 0.95
            v *= 0.5

            if stall > max_stall:

                p = best_p * np.exp(rng_local.normal(0, 0.03, n))

                v = np.zeros(n)
                eta = 0.5
                tau = 10.0

def build_conv_matrix_vec(a):
    """Build convolution matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        i_start = j
        i_end = min(2 * n - 1, j + n)
        A[i_start:i_end, j] = a[:i_end - j]
    return A


def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_vec(a)
    
    # Try cvxpy first
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    
    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None


def lp_alternating(p, time_budget=60.0, verbose=False):

    best_p = p.copy()

    start = time.monotonic()

    # Pre-build matrix
    A = build_conv_matrix_vec(p)

        if time.monotonic() - start > time_budget:
            break
        
        # Solve LP for optimal g given current a
        g_lp = solve_lp_minpeak(p)
        if g_lp is None:
            break
        
        # Line search: find best t in (1-t)*p + t*g_lp
        best_t = 0.5
        best_t_obj = np.inf
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            p_try = (1 - t) * p + t * g_lp
            p_try = normalize(p_try)
            obj_try = compute_obj_normalized(p_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:

            p = (1 - best_t) * p + best_t * g_lp

            A = build_conv_matrix_vec(p)
            if verbose and it % 5 == 0:

        else:
            stall += 1

def coordinate_descent(p, time_budget=30.0, verbose=False):

    """

        for j in range(n):
            if time.monotonic() - start > time_budget:

            # Save original value
            orig = p[j]

            best_val = orig
            best_obj_j = 2.0 * n * 10  # Large initial
            
            # Compute current b for reference
            fp = rfft(p, N_conv)
            b = irfft(fp * fp, N_conv)[:2 * n - 1]
            cur_max = float(np.max(b))
            
            # Try increasing and decreasing p[j]
            for delta in [0.5 * orig, 0.2 * orig, 0.1 * orig, -0.5 * orig, -0.2 * orig, -0.1 * orig]:
                p[j] = max(0.0, orig + delta)

                s = float(np.sum(p))
                if s < 1e-10:

                    continue
                p_tmp = p / s

                    # Don't renormalize p yet, just track
                    s_tmp = float(np.sum(p))
                    p = p / s_tmp if s_tmp > 0 else p_tmp
            
            p[j] = best_val

            else:
                p[j] = orig

            if verbose and sweep % 20 == 0:

def power_law_init(n, exponent, rng=None):
    """Generate power-law initialization a_j ∝ (j+1)^p."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** exponent
    return a / np.sum(a)


def exponential_init(n, rate, rng=None):
    """Generate exponential initialization."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(rate * j / n)
    return a / np.sum(a)


def uniform_init(n):
    return np.ones(n) / n


def resample_sequence(p, new_n):
    """Resample normalized sequence to new length."""
    n = len(p)
    if n == new_n:
        return p.copy()
    if n == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    return normalize(p_new)

        if obj1 < best_obj:
            best_obj = obj1
            best_p = p1
            p = p1
            if verbose:
                print(f"              SmoothSG: {best_obj:.8f}")
    
    # Phase 2: Top-k descent
    remaining = start + time_budget - time.monotonic()
    alloc = min(remaining * 0.2, 25.0)
    if alloc > 5:
        p2 = topk_descent(p, time_budget=alloc, verbose=verbose)
        obj2 = compute_obj_normalized(p2)
        if obj2 < best_obj:
            best_obj = obj2
            best_p = p2
            p = p2
            if verbose:
                print(f"              TopK: {best_obj:.8f}")
    
    # Phase 3: LP alternating
    remaining = start + time_budget - time.monotonic()
    if use_lp and alloc > 10:
        alloc_lp = min(remaining * 0.25, 40.0)
        if alloc_lp > 10:
            p3 = lp_alternating(p, time_budget=alloc_lp, verbose=verbose)
            obj3 = compute_obj_normalized(p3)
            if obj3 < best_obj:
                best_obj = obj3
                best_p = p3
                p = p3
                if verbose:
                    print(f"              LP: {best_obj:.8f}")
    
    # Phase 4: Coordinate descent polish
    remaining = start + time_budget - time.monotonic()
    alloc = min(remaining * 0.15, 20.0)
    if alloc > 5:
        p4 = coordinate_descent(p, time_budget=alloc, verbose=verbose)
        obj4 = compute_obj_normalized(p4)
        if obj4 < best_obj:
            best_obj = obj4
            best_p = p4
            if verbose:
                print(f"              CD: {best_obj:.8f}")
    
    return best_p


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Tries multiple initializations and lengths,
    using smoothed subgradient + LP + coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj_normalized(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n0 = 800
        best_seq = uniform_init(n0)
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    # === Phase 1: Optimize parent length ===
    n_parent = len(best_seq)
    print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 300.0)
    if alloc > 30:
        p_opt = full_optimize(best_seq, time_budget=alloc, verbose=True, use_lp=True)
        obj_opt = compute_obj_normalized(p_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = p_opt
            print(f"  ** Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_opt:.8f} (no improvement)")
    
    # === Phase 2: Power-law initializations ===
    print(f"\n=== Phase 2: Power-law inits (n={n_parent}) ===")
    p_values = [1.0, 1.5, 2.0, 3.0, 4.0, 5.0]
    for p_exp in p_values:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.12, 45.0)
        p_init = power_law_init(n_parent, p_exp)
        # Quick optimize
        p_opt = full_optimize(p_init, time_budget=alloc, verbose=False, use_lp=(alloc > 25))
        obj_opt = compute_obj_normalized(p_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = p_opt
            print(f"  ** p={p_exp}: obj={best_score:.8f} **")
        else:
            print(f"  p={p_exp}: obj={obj_opt:.8f}")
    
    # === Phase 3: Exponential inits ===
    print(f"\n=== Phase 3: Exponential inits (n={n_parent}) ===")
    for rate in [0.5, 1.0, 2.0, 3.0, 5.0]:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(remaining * 0.1, 35.0)
        p_init = exponential_init(n_parent, rate)
        p_opt = full_optimize(p_init, time_budget=alloc, verbose=False, use_lp=(alloc > 20))
        obj_opt = compute_obj_normalized(p_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = p_opt
            print(f"  ** rate={rate}: obj={best_score:.8f} **")
        else:
            print(f"  rate={rate}: obj={obj_opt:.8f}")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration ===")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        n_best - 100, n_best + 100, n_best + 200,
        n_best + 400, n_best + 800,
        600, 900, 1200, 1600, 2000
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 3000 and n != n_best]
    n_candidates = n_candidates[:8]  # Limit to 8 lengths
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining * 0.1, 50.0)
        
        # Resample best sequence to new length
        p_res = resample_sequence(best_seq, n_try)
        p_opt = full_optimize(p_res, time_budget=alloc, verbose=False, use_lp=(alloc > 25 and n_try <= 1200))
        obj_opt = compute_obj_normalized(p_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = p_opt
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 5: Random restarts ===
    print(f"\n=== Phase 5: Random restarts ===")
    n_curr = len(best_seq)
    n_restarts = max(2, int((deadline - time.monotonic()) / 45.0))
    n_restarts = min(n_restarts, 8)
    
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.12, 40.0)
        
        mode = rng.integers(0, 4)
        if mode == 0:
            p_init = power_law_init(n_curr, float(rng.uniform(0.5, 5.0)))
        elif mode == 1:
            p_init = exponential_init(n_curr, float(rng.uniform(0.5, 5.0)))
        elif mode == 2:
            # Bimodal
            c1, c2 = int(n_curr * rng.uniform(0.2, 0.4)), int(n_curr * rng.uniform(0.6, 0.8))
            p_init = np.zeros(n_curr)
            x = np.arange(n_curr)
            p_init += np.exp(-((x - c1)**2) / (n_curr * 0.02))
            p_init += np.exp(-((x - c2)**2) / (n_curr * 0.02))
            p_init = normalize(p_init)
        else:
            # Smooth random
            p_init = np.exp(rng.normal(0, 0.5, n_curr))
            # Smooth with a window
            kernel = np.ones(10) / 10
            p_init = np.convolve(p_init, kernel, mode='same')
            p_init = normalize(p_init)
        
        p_opt = full_optimize(p_init, time_budget=alloc, verbose=False, use_lp=False)
        obj_opt = compute_obj_normalized(p_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = p_opt
            print(f"  ** Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_opt:.8f}")
    
    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.6, 150.0)
        p_final = full_optimize(best_seq, time_budget=alloc, verbose=True, use_lp=True)
        obj_final = compute_obj_normalized(p_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = p_final
            print(f"  ** Final: obj={best_score:.8f} **")
        else:
            print(f"  Final: obj={obj_final:.8f}")
    
    # === Phase 7: One more LP pass if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 1200:
        alloc = min(remaining * 0.5, 60.0)
        p_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj_normalized(p_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = p_lp
            print(f"  ** Final LP: obj={best_score:.8f} **")
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
    
    # Convert to list of floats, ensuring non-negative and bounded
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
