"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient mirror descent with adaptive beta, LP alternating with
warm starts, peak balancing, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def smooth_gradient(a, beta):
    """
    Compute the gradient of the smooth approximation to max(conv(a,a)).
    Uses softmax weighting: grad[i] = sum_k softmax(beta*conv[k]) * a[k-i]
    Returns (gradient, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Softmax weights (numerically stable)
    shifted = beta * (conv - max_conv)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum > 0:
        weights /= w_sum
    
    # Compute gradient: grad[i] = sum_j weights[i+j] * a[j] for j=0..n-1
    # This is a sliding dot product
    grad = np.zeros(n)
    for i in range(n):
        # j ranges from 0 to n-1, so k=i+j ranges from i to i+n-1
        # Need 0 <= k <= 2n-2
        k_start = max(0, i)
        k_end = min(2 * n - 1, i + n)
        j_start = max(0, -i)  # always 0 since i >= 0
        j_end = min(n, 2 * n - 1 - i + 1)
        
        # More simply: for each i, grad[i] = dot(weights[i:i+n], a) but clipped
        # weights has length 2n-1, a has length n
        # k = i+j, j from 0 to n-1, so k from i to i+n-1
        # Need k <= 2n-2, so i+n-1 <= 2n-2, i.e., i <= n-1 (always true)
        # Need k >= 0, i.e., i >= 0 (always true)
        grad[i] = np.dot(weights[i:i+n], a)
    
    return grad, max_conv


def smooth_mirror_descent(a, n_iters=3000, time_budget=120.0, beta_start=10.0, beta_end=500.0):
    """
    Mirror descent with smooth gradient and adaptive beta.
    Beta increases from beta_start to beta_end over the iterations.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive beta: logarithmic schedule
        progress = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad, max_conv = smooth_gradient(a, beta)
        
        if np.all(grad == 0):
            break
        
        # Adaptive step size
        alpha = 2.0 / (it + 10)
        alpha = min(alpha, 0.5)
        
        # Mirror descent step
        a_new = a * np.exp(-alpha * grad / max_conv)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    # Try smaller step
                    a_try = a * np.exp(-alpha * 0.5 * grad / max_conv)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / np.sum(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
        
        if it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      Smooth MD it{it}: beta={beta:.1f}, obj={best_obj:.8f}, time={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # (conv(a,g))[k] = sum_j a[k-j] * g[j] for valid k-j
        # So A[k,j] = a[k-j] if 0 <= k-j < n
        for k in range(max(0, j), min(m, j + n)):
            A[k, j] = a[k - j]
    return A


def solve_lp_dual(a, time_limit=10.0):
    """
    Solve the dual LP to find the optimal g.
    Primal: min M s.t. A@g <= M, sum(g)=1, g>=0
    This is equivalent to: min M s.t. A@g - M*1 <= 0, sum(g)=1, g>=0
    
    We use scipy linprog on the primal directly (it's fast with HiGHS).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    m = 2 * n - 1
    
    # Primal: min c^T x where x = [g; M]
    # c = [0, ..., 0, 1] (n zeros then 1)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub @ x <= b_ub
    # Constraints: A_conv @ g - M <= 0  =>  [A_conv | -1] @ [g; M] <= 0
    A_ub = np.zeros((m, n + 1))
    for j in range(n):
        for k in range(max(0, j), min(m, j + n)):
            A_ub[k, j] = a_norm[k - j]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # A_eq @ x = b_eq
    # sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception as e:
        pass
    
    return None, None

def peak_balancing(a, time_budget=60.0):
    """
    Iteratively balance pairs that contribute to the peak lag.
    If peak is at lag k*, for each i, the term a[i]*a[i+k*] contributes.
    We reduce the larger of a[i], a[i+k*] and increase the smaller.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        improved = False
        # For each pair (i, i+k_star) that contributes to the peak
        valid_range = range(max(0, k_star - n + 1), min(n, n - k_star) if k_star > 0 else n)
        
        for i in range(n):
            j = k_star - i
            if 0 <= j < n and j != i:
                # Both a[i] and a[j] contribute to conf[k_star]
                if a[i] > a[j] * 1.1 and a[i] > 1e-6:
                    # Try to balance
                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        delta = frac * (a[i] - a[j])
                        a[i] -= delta
                        a[j] += delta
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                            break
        
        if not improved:
            # Try opposite direction: increase smaller, decrease larger more aggressively
            improved2 = False
            for i in range(0, n, 2):
                j = k_star - i
                if 0 <= j < n and j != i:
                    if a[j] > a[i] * 1.1 and a[j] > 1e-6:
                        for frac in [0.05, 0.1, 0.2]:
                            delta = frac * (a[j] - a[i])
                            a[j] -= delta
                            a[i] += delta
                            obj_try = compute_obj(a)
                            if obj_try < best_obj - 1e-13:
                                best_obj = obj_try
                                best_a = a.copy()
                                improved2 = True
                                break
                            else:
                                a[j] += delta
                                a[i] -= delta
                                break
                if time.monotonic() - start > time_budget:
                    break
            
            if not improved2 and not improved:
                # Random perturbation
                rng = np.random.default_rng(it + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
        
        if it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"      Peak bal it{it}: obj={best_obj:.8f}, k*={k_star}, time={elapsed:.1f}s")

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        freqs = rng.uniform(0, 6, 10)
        phases = rng.uniform(0, 2 * np.pi, 10)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # Random permutation of a smooth function
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 + 0.1
        a = a / np.sum(a)
        perm = rng.permutation(n)
        a = a[perm]
    elif mode == 4:
        # Piecewise constant
        n_blocks = rng.integers(3, 10)
        block_size = n // n_blocks
        a = rng.uniform(0.1, 1.0, n_blocks)
        a = np.repeat(a, block_size)[:n]
        a = a / np.sum(a)
    else:
        # Sparse-ish
        a = rng.uniform(0, 1, n)
        a[a < 0.3] = 0
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        a = a / np.sum(a)
    return a

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient mirror descent with adaptive beta, LP alternating,
peak balancing, spectral shaping, multi-n exploration, and local search.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def smooth_gradient(a, beta):
    """
    Compute gradient of smooth approximation to max(conv(a,a)) using softmax.
    grad[i] = sum_j softmax_k[beta*conv[k]] * a[k-i] (with valid bounds)
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    shifted = beta * (conv - max_conv)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum > 0:
        weights /= w_sum

    # grad[i] = sum_{j=0}^{n-1} weights[i+j] * a[j]
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    W = weights[idx]
    grad = W @ a

    return grad, max_conv

                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    accepted = True
        if not accepted:
            a_try = a * np.exp(-alpha * 0.3 * grad / max_conv)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try

        if it % 500 == 0 and it > 0:
            elapsed = time.monotonic() - start
            print(f"      SmoothMD it{it}: beta={beta:.1f}, obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def solve_lp_minpeak(a, time_limit=5.0):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    m = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    for j in range(n):
        k_lo = max(0, j)
        k_hi = min(m, j + n)
        A_ub[k_lo:k_hi, j] = a_norm[k_lo - j:k_hi - j]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating(a, time_budget=120.0, verbose=False, max_iters=200):
    """LP alternating with adaptive line search and perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        remaining = time_budget - (time.monotonic() - start)
        tl = min(5.0, max(0.5, remaining * 0.02))
        g_lp, M_lp = solve_lp_minpeak(a, time_limit=tl)
        if g_lp is None:
            stall += 1
            if stall > 8:
                break
            continue

        best_t = 0.5
        best_t_obj = np.inf
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t

        if best_t_obj < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 10 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng((it * 7919 + n) % (2**32))
                scale = 0.03 + 0.01 * min(stall // 10, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

        if it % 30 == 0 and it > 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, stall={stall}, t={elapsed:.1f}s")

    return best_a


def peak_balancing(a, time_budget=60.0):
    """Balance pairs contributing to the peak lag to reduce it."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        for i in range(n):
            j = k_star - i
            if 0 <= j < n and j != i:
                if a[i] > a[j] * 1.2 and a[i] > 1e-6:
                    delta = 0.15 * (a[i] - a[j])
                    a[i] -= delta
                    a[j] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] += delta
                        a[j] -= delta
                elif a[j] > a[i] * 1.2 and a[j] > 1e-6:
                    delta = 0.15 * (a[j] - a[i])
                    a[j] -= delta
                    a[i] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] += delta
                        a[i] -= delta

        if not improved:
            rng = np.random.default_rng(it * 31 + n)
            a = best_a * np.exp(rng.normal(0, 0.04, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            best_obj = compute_obj(a)

        if it % 30 == 0 and it > 0:
            elapsed = time.monotonic() - start
            print(f"      PeakBal it{it}: obj={best_obj:.8f}, k*={k_star}, t={elapsed:.1f}s")

    return best_a


def spectral_shaping(a, time_budget=30.0):
    """Try to flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-12)
        phase = A / mag_safe

        # Target: flatter spectrum
        A_target = phase * (0.8 + 0.2 * (mag / mag.max()))
        A_new = 0.7 * A + 0.3 * A_target
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

    return best_a


def local_search(a, time_budget=30.0):
    """Random perturbation local search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - start < time_budget:
        scale = 0.02 * (1.0 + 0.5 * np.sin(it * 0.1))
        a_try = best_a * np.exp(rng.normal(0, scale, n))
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        freqs = rng.uniform(0, 8, 12)
        phases = rng.uniform(0, 2 * np.pi, 12)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.25 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) + 0.2
        a = a / np.sum(a)
    elif mode == 4:
        n_blocks = rng.integers(3, 15)
        block_vals = rng.uniform(0.1, 1.0, n_blocks)
        a = np.repeat(block_vals, n // n_blocks + 1)[:n]
        a = a / np.sum(a)
    else:
        a = rng.uniform(0, 1, n)
        a = np.convolve(a, np.ones(7) / 7, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a
