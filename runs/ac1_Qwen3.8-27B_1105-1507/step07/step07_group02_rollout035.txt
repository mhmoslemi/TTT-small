        while L < (2*n-1) + n - 1:

        g_centered = g - np.mean(g)

        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-11:
            lr *= 0.5
            continue
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    
    return best_a

But the issue is that gradient descent on a non-smooth (or smoothed non-convex) function can get stuck. We need to combine it with other methods.

**Let me also implement a "push-pull" strategy:**

The idea is: identify where the convolution is above average and where it's below. Push mass from elements that contribute to high-convolution positions toward elements that contribute to low-convolution positions.

This is similar to the "Optimal Permutation" problem in graph theory.

**Another idea: Use the "mirror descent" or "multiplicative weights" approach.**

Update: `a_new[i] = a[i] * exp(-lr * g[i])` where `g[i]` is the gradient. Then normalize.

This naturally maintains non-negativity and is related to the exponential family.

def fast_conv(a):
    n = len(a)
    if n <= 1:
        return a * a.copy()
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = fast_conv(a)
    return 2.0 * n * np.max(conv) / (s * s)

def compute_grad_smoothed(a, tau=50.0):
    """Compute smoothed gradient of the objective (with sum=1 normalization)."""
    n = len(a)
    conv = fast_conv(a)
    max_c = np.max(conv)
    exp_c = np.exp(tau * (conv - max_c))
    w = exp_c / np.sum(exp_c)

    A = rfft(a, L)
