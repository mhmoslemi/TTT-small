For large beta, `f_beta(a) ≈ max(a*a)`. The gradient is:
`grad[i] = sum_k softmax_k * d(a*a)[k]/da[i] = sum_k softmax_k * 2*a[k-i] (if valid)`

This can be computed efficiently! Let `w_k = softmax_k` (the Boltzmann weights). Then:
`grad[i] = 2 * sum_k w_k * a[k-i] = 2 * (w * a)[i]`

where `*` denotes convolution! So the gradient is `2 * conv(w, a)` (truncated to length n).

This is great because we can compute the gradient in O(n log n) using FFT.

Now, we need to optimize `f_beta(a)` subject to `a >= 0, sum(a) = 1`. We can use:
- Projected gradient descent with adaptive step size
- Adam optimizer with projection
- Mirror descent (exponential weights) with projection

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def convolve_trunc(w, a):
    """Compute conv(w, a) truncated to length len(a)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a, size)
    result = irfft(fw * fa, size)[:n]
    return result

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search for tau
    lo, hi = 0.0, float(np.max(v))
    for _ in range(60):
        tau = (lo + hi) / 2
        s = np.sum(np.maximum(v - tau, 0))
        if s > 1:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2
    return np.maximum(v - tau, 0)

def smooth_max_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*conv))): which approximates max(conv).
    Returns (gradient, smoothed_value)."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    beta_conv = beta * conv
    max_bc = np.max(beta_conv)
    exp_bc = np.exp(beta_conv - max_bc)
    w = exp_bc / np.sum(exp_bc)
    # Gradient: 2 * conv(w, a) truncated to n
    grad = 2.0 * convolve_trunc(w, a)
    smooth_val = (np.max(beta_conv) + np.log(np.sum(exp_bc))) / beta
    return grad, smooth_val

def adam_smooth_max(a, n_iters=10000, beta=100.0, lr=0.01, time_budget=60.0):
    """Optimize using Adam on the smoothed max objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    
    # Adaptive beta schedule
    beta_start = beta
    beta_end = 500.0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        t += 1
        # Schedule beta
        progress = min(1.0, t / (n_iters * 0.5))
        cur_beta = beta_start + (beta_end - beta_start) * progress
        
        grad, smooth_val = smooth_max_gradient(a, cur_beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a_new = simplex_project(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def mirror_descent_smooth(a, n_iters=5000, beta=100.0, time_budget=60.0):
    """Mirror descent on smoothed max."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = np.maximum(a, 1e-15) / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.5
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, _ = smooth_max_gradient(a, beta)
        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.99
        else:
            alpha *= 0.5
    
    return best_a

smoothed max gradient with Adam optimizer as the primary method,
combined with FFT flat-spectrum iterations, LP alternating, and
multi-n exploration. Key innovation: efficient O(n log n) gradient
computation via FFT for the log-sum-exp smoothed objective with
adaptive beta scheduling and per-coordinate learning rates.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_smooth_grad(a, w):
    """Compute gradient of smoothed max: grad[i] = 2*sum_j w[i+j]*a[j]."""
    n = len(a)
    w_rev = w[::-1]
    conv_result = np.convolve(w_rev, a, mode='full')
    return 2.0 * conv_result[n - 1:2 * n - 1][::-1]


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} via bisection."""
    n = len(v)
    if n == 0:
        return v
    v_max = float(np.max(v))
    if v_max <= 0:
        return np.ones(n) / n
    lo = 0.0
    hi = v_max
    for _ in range(80):
        tau = (lo + hi) / 2.0
        s = float(np.sum(np.maximum(v - tau, 0.0)))
        if s > 1.0:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    result = np.maximum(v - tau, 0.0)
    s = float(np.sum(result))
    if s > 0:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def smooth_max_objective(a, beta):
    """Compute smoothed max and its gradient using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    beta_conv = beta * conv
    max_bc = float(np.max(beta_conv))
    exp_bc = np.exp(beta_conv - max_bc)
    w = exp_bc / float(np.sum(exp_bc))
    grad = compute_smooth_grad(a, w)
    smooth_val = (max_bc + float(np.log(float(np.sum(exp_bc))))) / beta
    return grad, smooth_val


def adam_smooth_max(a, n_iters=20000, beta_start=50.0, beta_end=2000.0,
                    lr=0.005, time_budget=120.0):
    """Adam optimization on log-sum-exp smoothed max of convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        t += 1
        progress = min(1.0, t / max(1, n_iters * 0.6))
        cur_beta = beta_start + (beta_end - beta_start) * progress * cur_beta

        grad, _ = smooth_max_objective(a, cur_beta)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if stall > 5000:
            rng = np.random.default_rng(t)
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            best_a = a.copy()
            best_obj = compute_obj(a)
            stall = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0

    return best_a


def adam_multipeak(a, n_iters=10000, time_budget=60.0, n_peaks=5):
    """Adam with multi-peak subgradient (faster than full softmax)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    lr = 0.01
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        t += 1
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[-n_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        spread = max(1e-6, max_val - top_val[0])
        weights = np.exp((top_val - max_val) / (0.1 * spread + 1e-10))
        weights = weights / float(np.sum(weights))

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            lo_i = max(0, ki - n + 1)
            hi_i = min(n - 1, ki)
            for i in range(lo_i, hi_i + 1):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * a[j]
        grad *= 2.0

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.9
                stall = 0

        a = a_new

    return best_a


def fft_flat_spectrum(a, n_iters=500, time_budget=30.0, relaxation=0.5):
    """FFT-based flat spectrum iteration (Gerchberg-Saxton variant)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe

        A_flat = phase
        A_new = (1.0 - relaxation) * A + relaxation * A_flat

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    rng = np.random.default_rng(it + n)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

    return best_a


def mirror_descent_smooth(a, n_iters=5000, beta=100.0, time_budget=60.0):
    """Mirror descent (exponential) on smoothed max."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = np.maximum(a, 1e-15) / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.3

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, _ = smooth_max_objective(a, beta)
        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.995
        else:
            alpha *= 0.5

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 2000:
        n = 2000
        a = a[:n]
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.MOSEK, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def lp_alternating(a, time_budget=60.0):
    """LP alternating with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(it * 13 + n)
                a = best_a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

    return best_a


def chirp_init(n, variant=0):
    """Generate chirp-based initialization."""
    x = np.arange(n)
    if variant == 0:
        a = (1.0 + np.cos(2.0 * np.pi * x * x / n)) / 2.0
    elif variant == 1:
        a = (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n))) / 2.0
    elif variant == 2:
        a = (1.0 + np.cos(2.0 * np.pi * x * x / n + np.pi / 4)) / 2.0
    else:
        a = np.abs(np.exp(1j * np.pi * x * x / n))
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s


def structured_init(n, variant=0):
    """Various structured initializations."""
    x = np.arange(n)
    if variant == 0:
        return np.ones(n) / n
    elif variant == 1:
        a = np.sin(np.pi * x / n) ** 2
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif variant == 2:
        c1, c2 = n // 3, 2 * n // 3
        w = n // 8
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif variant == 3:
        freqs = np.random.default_rng(variant).uniform(0, 5, 6)
        phases = np.random.default_rng(variant + 100).uniform(0, 2 * np.pi, 6)
        xn = x / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.cos(2 * np.pi * f * xn + p)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    else:
        a = np.cos(np.pi * x / n) ** 2 + 0.2
        return a / np.sum(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def peak_shaving(a, time_budget=20.0):
    """Targeted peak reduction: move mass from peak contributors."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        improved = False
        for _, i, j in contributors[:10]:
            if i == j or a[i] < 1e-8:
                continue
            # Try moving mass from position i to a position that doesn't contribute
            for frac in [0.02, 0.05, 0.1, 0.15]:
                delta = frac * a[i]
                if delta > a[i] * 0.5:
                    continue
                # Find a non-contributing position
                for new_i in range(n):
                    new_j = k_star - new_i
                    if 0 <= new_j < n and new_i != i and (a[new_i] < 0.1 * a[i]):
                        a[i] -= delta
                        a[new_i] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-10:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[new_i] -= delta
                            break
                if improved:
                    break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines smoothed-max Adam optimization, FFT flat spectrum,
    mirror descent, LP alternating, chirp initializations, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Try structured initializations at parent n ===
    print(f"\n=== Phase 1: Structured Inits (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    candidates = []

    # Add parent
    candidates.append((best_seq.copy(), "parent"))

    # Chirp variants
    for v in range(4):
        candidates.append((chirp_init(n_parent, v), f"chirp{v}"))

    # Other structured
    for v in range(5):
        candidates.append((structured_init(n_parent, v), f"struct{v}"))

    # Quick eval of all
    for idx, (cand, name) in enumerate(candidates):
        if time.monotonic() > deadline:
            break
        obj = compute_obj(cand)
        status = " **" if obj < best_score else ""
        print(f"  {name}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = cand.copy()

    # === Phase 2: Adam smooth max (primary optimizer) ===
    print(f"\n=== Phase 2: Adam Smooth Max (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.3, 300.0)
        a_opt = adam_smooth_max(best_seq, n_iters=50000, beta_start=30.0,
                                beta_end=2000.0, lr=0.003, time_budget=alloc)
        obj = compute_obj(a_opt)
        print(f"  Adam smooth max: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # Second round with different hyperparams
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 120.0)
        a_opt2 = adam_smooth_max(best_seq, n_iters=30000, beta_start=100.0,
                                 beta_end=5000.0, lr=0.002, time_budget=alloc)
        obj2 = compute_obj(a_opt2)
        print(f"  Adam smooth max (v2): obj={obj2:.8f}")
        if obj2 < best_score:
            best_score = obj2
            best_seq = a_opt2.copy()

    # === Phase 3: Adam multi-peak ===
    print(f"\n=== Phase 3: Adam Multi-Peak (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.2, 120.0)
        a_mp = adam_multipeak(best_seq, n_iters=10000, time_budget=alloc, n_peaks=5)
        obj = compute_obj(a_mp)
        print(f"  Adam multi-peak: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_mp.copy()

    # === Phase 4: FFT flat spectrum ===
    print(f"\n=== Phase 4: FFT Flat Spectrum (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_fft = fft_flat_spectrum(best_seq, n_iters=500, time_budget=alloc, relaxation=0.5)
        obj = compute_obj(a_fft)
        print(f"  FFT (relax=0.5): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fft.copy()

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.05, 30.0)
        a_fft2 = fft_flat_spectrum(best_seq, n_iters=300, time_budget=alloc, relaxation=0.3)
        obj = compute_obj(a_fft2)
        print(f"  FFT (relax=0.3): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fft2.copy()

    # === Phase 5: Mirror descent smooth ===
    print(f"\n=== Phase 5: Mirror Descent Smooth (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_md = mirror_descent_smooth(best_seq, n_iters=5000, beta=200.0, time_budget=alloc)
        obj = compute_obj(a_md)
        print(f"  Mirror descent: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # === Phase 6: LP Alternating ===
    print(f"\n=== Phase 6: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_for_lp = len(best_seq)
        # LP is expensive for large n, cap at 1000
        if n_for_lp > 1000:
            a_for_lp = resample_sequence(best_seq, 1000)
        else:
            a_for_lp = best_seq.copy()
        alloc = min(remaining * 0.2, 120.0)
        a_lp = lp_alternating(a_for_lp, time_budget=alloc)
        obj = compute_obj(a_lp)
        print(f"  LP (n={len(a_lp)}): obj={obj:.8f}")
        if obj < best_score and len(a_lp) == len(best_seq):
            best_score = obj
            best_seq = a_lp.copy()
        elif obj < best_score:
            # Resample back
            a_lp_full = resample_sequence(a_lp, len(best_seq))
            obj_full = compute_obj(a_lp_full)
            if obj_full < best_score:
                best_score = obj_full
                best_seq = a_lp_full.copy()

    # === Phase 7: Peak shaving ===
    print(f"\n=== Phase 7: Peak Shaving (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 30.0)
        a_ps = peak_shaving(best_seq, time_budget=alloc)
        obj = compute_obj(a_ps)
        print(f"  Peak shaving: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_ps.copy()

    # === Phase 8: Multi-n exploration ===
    print(f"\n=== Phase 8: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        400, 500, 600, 700, 800, 900, 1000, 1200, 1500,
        max(200, n_best - 100), max(200, n_best + 100),
        max(200, n_best - 50), max(200, n_best + 50),
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if abs(n_try - n_best) < 20:
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc1 = min(remaining * 0.08, 30.0)
        a_opt = adam_smooth_max(a_try, n_iters=20000, beta_start=50.0,
                                beta_end=1500.0, lr=0.003, time_budget=alloc1)
        alloc2 = min(remaining * 0.04, 15.0)
        a_opt = fft_flat_spectrum(a_opt, n_iters=200, time_budget=alloc2, relaxation=0.5)
        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Adam with high beta
        alloc = min(remaining * 0.3, 60.0)
        a_final = adam_smooth_max(best_seq, n_iters=30000, beta_start=500.0,
                                  beta_end=10000.0, lr=0.001, time_budget=alloc)
        obj = compute_obj(a_final)
        print(f"  Final Adam (high beta): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Mirror descent
        alloc = min(remaining * 0.2, 40.0)
        a_final = mirror_descent_smooth(best_seq, n_iters=5000, beta=1000.0, time_budget=alloc)
        obj = compute_obj(a_final)
        print(f"  Final MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Peak shaving again
        alloc = min(remaining * 0.3, 20.0)
        a_final = peak_shaving(best_seq, time_budget=alloc)
        obj = compute_obj(a_final)
        print(f"  Final Peak Shave: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
