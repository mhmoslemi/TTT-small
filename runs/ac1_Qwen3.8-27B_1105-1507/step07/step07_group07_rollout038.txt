Great! So the gradient of the peak is simple:
`∇_a max_k c[k] = 2 * a[k*-i]` (for each i, where k* is the argmax).

More precisely, if `k* = argmax_k c[k]`, then:
`∂(c[k*])/∂a[i] = 2*a[k*-i]` if `0 <= k*-i < n`, else 0.

This is a very simple gradient! It's just a shifted and possibly reversed copy of a.

This means we can do a simple gradient descent:
```
a_new[i] = a[i] - eta * 2 * a[k* - i]  (if 0 <= k*-i < n, else a[i] - eta * 0)
```

But we need to maintain `a >= 0` and `sum(a) = 1`. We can project onto this constraint set.

Even better: since the gradient is just a shifted copy of a, we can use this structure for faster optimization.

**Projected gradient method:**
1. Compute `c = conv(a, a)`, find `k* = argmax(c)`.
2. Compute gradient `g[i] = 2*a[k*-i]` (with zero padding).
3. Update: `a_new = a - eta * g`.
4. Project: clip to [0, inf], normalize to sum=1.
5. Repeat.

This is very fast! Each iteration is O(n log n) for the convolution (via FFT) plus O(n) for the update and projection.

Let me think about whether this will work well...

The issue is that the gradient only points in one direction (toward reducing the peak at k*). But reducing the peak at k* might increase the peak at another lag. So we need to be careful with the step size.

A better approach: use the "smoothed" version where we weight all lags by their softmax:
`f_smooth(a) = log(sum_k exp(c[k]/tau)) * tau` (soft-min aproximación)

The gradient of this is:
`∂f_smooth/∂a[i] = (1/s) * sum_k exp(c[k]/tau) * ∂c[k]/∂a[i]`
`= (1/s) * sum_k w[k] * 2*a[k-i]` (where w[k] = exp(c[k]/tau) / sum_j exp(c[j]/tau))

This is like a "soft" version of the gradient, weighting contributions from all lags.

This is exactly what the previous code's `smoothed_obj_and_grad` function does!

Actually, `c[k] = sum_j a[j]*a[k-j]`. The Hessian has:
`∂²c[k]/∂a[i]∂a[j] = δ(i,j') + δ(j, k-i)` where j' = k-j...

"""
Hybrid optimizer for low-peak non-negative autocorrelation sequences.
Combines projected gradient, smoothed L-BFGS, LP alternating, block coordinate
descent, and multi-resolution exploration to minimize 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

import sys

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    if abs(s - 1.0) > 1e-6:
        a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def norm_sum(a):
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    return a / s

def project_simplex(a):
    """Project a onto {x : x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.full_like(a, 1.0 / len(a))
    return a / s

def projected_gradient_step(a, k_star, eta):
    """One step of projected gradient descent on peak convolution."""
    n = len(a)
    # Gradient: g[i] = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0
    g = np.zeros(n)
    # For each i, k_star - i is in [0, n-1] when i is in [k_star-n+1, k_star]
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    if i_min <= i_max:
        indices = np.arange(i_min, i_max + 1)
        g[indices] = 2.0 * a[k_star - indices]
    
    a_new = a - eta * g
    return project_simplex(a_new)

def fast_projected_gradient(a_init, time_budget=30.0, eta0=None):
    """Fast projected gradient with backtracking line search."""
    n = len(a_init)
    a = norm_sum(np.maximum(a_init, 0).astype(np.float64)).copy()
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    if eta0 is None:
        eta0 = 0.1 / n
    eta = eta0
    start = time.monotonic()
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Gradient
        g = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            g[indices] = 2.0 * a[k_star - indices]
        
        # Backtracking line search
        a_try = a - eta * g
        for _ in range(20):
            a_try_p = project_simplex(a_try)
            obj_try = compute_obj_normalized(a_try_p)
            if obj_try < best_obj or obj_try < compute_obj_normalized(a):
                break

            best_obj = obj_try
            best_a = a_try_p.copy()
            a = a_try_p
            eta = min(eta * 1.05, 1.0 / n)  # slow increase
        else:
            eta *= 0.8
            if eta < 1e-6 / n:

def smoothed_obj_grad(x, tau):
    """
    Smoothed objective and gradient for a = exp(x), normalized.
    x is the log-parameter (minus max for stability).
    """
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    a = a / s  # normalize
    
    conv = compute_conv(a)
    
    # Softmax weights
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = float(np.sum(exp_c))
    w = exp_c / sum_exp
    
    # Objective: tau * log(sum exp(c/tau)) (approximately max)
    logsumexp_val = c_max + np.log(sum_exp)
    F = tau * logsumexp_val  # This is the smoothed max
    # But we want 2n * smoothed_max, and a is normalized
    F = 2.0 * n * F
    
    # Gradient: dF/dx_i
    # dF/da_i = 2n * sum_k w[k] * d(c[k])/da[i] = 2n * sum_k w[k] * 2*a[k-i]
    # da_i/dx_j = a_i * δ(i,j) - a_i * a_j / s (from normalization)
    # But since we normalized a = exp(x)/s, and the objective is scale-invariant,
    # we can just compute dF/da_i and use the chain rule for the exp parameterization.
    
    # Actually, since F is scale-invariant (F(a) = F(ca) for c>0), 
    # and a = exp(x) / sum(exp(x)), we have:
    # dF/dx_i = sum_j (dF/da_j) * (da_j/dx_i)
    # da_j/dx_i = δ(i,j)*a_i - a_i*a_j (after normalization)
    # So dF/dx_i = (dF/da_i)*a_i - a_i * sum_j (dF/da_j)*a_j
    
    # But since F is scale-invariant, sum_j (dF/da_j) * a_j = 0 (Euler's theorem)
    # So dF/dx_i = (dF/da_i) * a_i
    
    # dF/da_i = 2n * sum_k w[k] * 2*a[k-i] (with bounds)
    # Let's compute cross = sum_k w[k] * a[k-i]
    
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    cross_full = irfft(fw * fa_flip, size)
    # cross[i] = sum_k w[k] * a[k-i] (the relevant part)
    # cross_full has length 2n-1 (padded to size)
    # cross_full[k] = sum_j w[j] * a_flip[k-j] = sum_j w[j] * a[n-1-(k-j)]
    # We want cross[i] = sum_k w[k] * a[k-i]
    # Let's use a different approach
    
    # cross[i] = sum_k w[k] * a[k-i] for 0<=k-i<n
    # This is the cross-correlation of w and a
    # corr[i] = sum_k w[k] * a[k-i]
    # Using FFT: if we flip a, then corr = conv(w, a_flip) evaluated at appropriate indices
    
    # Actually, let me just compute it directly for clarity (n is at most ~2000)
    # cross[i] = sum_{k: 0<=k-i<n} w[k] * a[k-i]
    # = sum_{j: 0<=j<n} w[i+j] * a[j] (substituting j=k-i)
    
    # This is a cross-correlation. Let me use the FFT approach:
    # cross = ifft(fft(w_padded) * conj(fft(a_padded)))
    # or equivalently cross = conv(w, flip(a)) at appropriate offset
    
    # Let me just use the direct convolution approach:
    # cross[i] = sum_j w[i+j] * a[j] for j=0..n-1 (with w[k]=0 for k<0 or k>=2n-1)
    
    # Using FFT: cross = ifft(fft(w, size) * fft(flip(a), size))[n-1:n-1+n]
    # Wait, let me verify:
    # conv(w, flip(a))[m] = sum_k w[k] * flip(a)[m-k] = sum_k w[k] * a[n-1-(m-k)]
    # We want cross[i] = sum_k w[k] * a[k-i]
    # Let j = k-i, so k = i+j: cross[i] = sum_j w[i+j] * a[j]
    # conv(w, flip(a))[n-1+i] = sum_k w[k] * a[n-1-((n-1+i)-k)] = sum_k w[k] * a[k-i]
    # Yes! So cross[i] = conv(w, flip(a))[n-1+i]
    
    conv_w_af = irfft(fw * fa_flip, size)
    cross = conv_w_af[n - 1:n - 1 + n]
    
    # dF/da_i = 2n * 2 * cross[i] = 4n * cross[i]
    # dF/dx_i = dF/da_i * a_i = 4n * cross[i] * a[i]
    grad = 4.0 * n * cross * a
    
    return F, grad

def solve_lp_minpeak_fast(a, max_iters=100):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses iterative refinement with active set.
    Returns g_normalized.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build the constraint matrix: A[k, j] = a_norm[k-j] if 0<=k-j<n, else 0
    # (a_norm * g)[k] = sum_j a_norm[k-j] * g[j] (with a_norm[m]=0 for m<0 or m>=n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For each j, a_norm[k-j] is nonzero when 0 <= k-j < n, i.e., j <= k < j+n
        k_start = max(0, j)
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a_norm[0:k_end-k_start]
    
    # LP: min M s.t. A @ g <= M, sum(g) = 1, g >= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    # Using scipy linprog (HiGHS)
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'maxiter': 5000})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception as e:
        pass
    
    return None

def block_coordinate_descent(a_init, time_budget=60.0, block_size=50):
    """
    Block coordinate descent: optimize blocks of consecutive variables.
    Within each block, use the projected gradient method.
    """
    n = len(a_init)
    a = norm_sum(np.maximum(a_init, 0).astype(np.float64)).copy()
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    n_blocks = n // block_size + 1
    offsets = list(range(0, n, block_size))
    
    epoch = 0
    while time.monotonic() - start < time_budget:
        epoch += 1
        # Shuffle block order
        order = list(range(len(offsets)))
        rng.shuffle(order)
        
        for idx in order:
            if time.monotonic() - start > time_budget:
                break
            
            off = offsets[idx]
            bs = min(block_size, n - off)
            
            # Get current block
            a_block = a[off:off+bs].copy()
            
            # Optimize the block using projected gradient (keeping rest fixed)
            for it in range(200):
                conv = compute_conv(a)
                k_star = int(np.argmax(conv))
                
                # Compute gradient only for the block
                g_block = np.zeros(bs)
                for i in range(bs):
                    gi = off + i
                    k_minus_i = k_star - gi
                    if 0 <= k_minus_i < n:
                        g_block[i] = 2.0 * a[k_minus_i]
                
                # Projected gradient step on the block
                eta = 0.5 / (n * np.max(g_block) + 1)
                a[off:off+bs] = a_block - eta * g_block
                # Project the block: ensure non-negativity and re-normalize
                a[off:off+bs] = np.maximum(a[off:off+bs], 0)
                a = norm_sum(a)
                a_block = a[off:off+bs].copy()
            
            obj = compute_obj_normalized(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a

def symmetric_optimize(a_init, time_budget=30.0):
    """Optimize with palindromic constraint."""
    n = len(a_init)
    # Enforce symmetry
    a_sym = (a_init + a_init[::-1]) / 2.0
    a_sym = project_simplex(a_sym)
    
    # Optimize the half and mirror
    half = a_sym[:n//2 + 1].copy()
    best_half = half.copy()
    
    start = time.monotonic()
    eta = 0.1 / n
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        # Reconstruct full sequence
        h = len(half)
        a_full = np.zeros(n)
        a_full[:h] = half
        a_full[n-h+1:] = half[:n-h+1][::-1] if n % 2 == 0 else half[:n-h][::-1]
        # Simpler: 
        a_full = np.zeros(n)
        for i in range(h):
            a_full[i] = half[i]
            a_full[n-1-i] = half[i]
        a_full = project_simplex(a_full)
        
        conv = compute_conv(a_full)
        k_star = int(np.argmax(conv))
        
        # Gradient for the half
        g_half = np.zeros(h)
        for i in range(h):
            # Two positions: i and n-1-i
            for pos in [i, n-1-i]:
                if pos >= h and n % 2 == 0:
                    continue  # already counted
                km = k_star - pos
                if 0 <= km < n:
                    g_half[i] += 2.0 * a_full[km]
        
        eta = 0.1 / (n * (np.max(g_half) + 1e-10))
        half = np.maximum(half - eta * g_half, 0)
        s = float(np.sum(half))
        if s > 1e-10:
            half = half / s * (float(np.sum(a_full[:h])) / s + 1e-10)  # maintain approximation
            half = half / (1.0 + 1e-10)
    
    # Final symmetric reconstruction
    a_full = np.zeros(n)
    h = len(half)
    for i in range(h):
        a_full[i] = half[i]
        a_full[n-1-i] = half[i]
    a_full = project_simplex(a_full)
    
    obj = compute_obj_normalized(a_full)
    obj_orig = compute_obj_normalized(a_init)
    
    if obj < obj_orig:
        return a_full
    return project_simplex(a_init)

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(np.maximum(a_new, 0))

def diversity_restart(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
    elif mode == 3:
        # Raised cosine with random width
        center = n / 2
        width = n * rng.uniform(0.3, 0.8)
        x = np.linspace(-1, 1, n)
        a = np.maximum(0.5 + 0.5 * np.cos(np.pi * x / width), 0)
    elif mode == 4:
        # Random smooth (low-pass)
        a = rng.uniform(0, 1, n)
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    elif mode == 5:
        # Two peaks
        c1, c2 = int(n * rng.uniform(0.2, 0.4)), int(n * rng.uniform(0.6, 0.8))
        w = n * 0.1
        x = np.arange(n)
        a = np.exp(-((x-c1)**2)/(2*w*w)) + np.exp(-((x-c2)**2)/(2*w*w))
    elif mode == 6:
        # Tapered ends
        a = np.ones(n)
        taper = n // 10
        for i in range(taper):
            a[i] *= (i + 1) / taper
            a[n-1-i] *= (i + 1) / taper
    else:
        # Power law decay from center
        x = np.abs(np.arange(n) - n/2) / (n/2)
        a = (1 - x) ** rng.uniform(0.5, 3.0)
    
    return project_simplex(np.maximum(a + 0.01 * rng.uniform(0, 1, n), 0))

def multi_temperature_gradient(a_init, time_budget=120.0):
    """

    """

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    budget_a = min(time_budget * 0.2, 30.0)
    a_pg = fast_projected_gradient(a, time_budget=budget_a, eta0=0.5/n)
    obj_pg = compute_obj_normalized(a_pg)
    if obj_pg < best_obj:
        best_obj = obj_pg
        best_a = a_pg.copy()
        a = a_pg
        print(f"    PG: obj={best_obj:.8f}")
    
    # Phase B: Smoothed L-BFGS (fine)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:

        a_lbfgs = lbfgs_smoothed(a, time_budget=budget_b)

            best_obj = obj_lbfgs
            best_a = a_lbfgs.copy()
            a = a_lbfgs
            print(f"    L-BFGS: obj={best_obj:.8f}")
