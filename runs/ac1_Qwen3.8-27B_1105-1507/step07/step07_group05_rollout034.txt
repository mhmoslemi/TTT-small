import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    b = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - (n - 1))
        hi = min(2 * n - 1, j + n)
        for k in range(lo, hi):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build conv matrix A efficiently using vectorization."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        valid_k = np.arange(max(0, -(n - 1) + j), min(2 * n - 1, n + j))
        A[valid_k, j] = a[valid_k - j]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g (normalized, sum=1, nonnegative) or None.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s

    A = build_conv_matrix_fast(a)

    # Primal: min M
    # s.t. A@g - M <= 0  (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    # Variables: [g_0, ..., g_{n-1}, M] (n+1 total)

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M

        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g
    except Exception:
        pass

    return None

def logspace_gradient_descent(a, time_budget=120.0, verbose=False, T_max=50.0):
    """
    Log-parameterized gradient descent with Adam and temperature annealing.
    Parameterization: a = softmax(x), so a > 0 and sum(a) = 1 automatically.
    Uses log-sum-exp smoothing of the max function.
    """
    n = len(a)
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)

    # Initialize x from a: x = log(a) (up to a constant, since softmax is shift-invariant)
    x = np.log(a) - np.log(a).max()  # shift for numerical stability

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # Adam optimizer state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    t_adam = 0

    # FFT setup
    N = 1
    while N < 3 * n - 1:
        N <<= 1

    a_rev = a[::-1]

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        # Temperature schedule: start low (smooth), anneal to high (sharp)
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        # Cosine annealing from 1 to T_max
        T = 1.0 + (T_max - 1.0) * 0.5 * (1 - np.cos(np.pi * progress))

        # Compute conv(a, a)
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Log-sum-exp weights
        b_scaled = T * b
        b_max = np.max(b_scaled)
        w = np.exp(b_scaled - b_max)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
        # = 2 * conv(w, a_rev)[n-1+i]
        a_rev = a[::-1]
        fw = rfft(w, N)
        fa_rev = rfft(a_rev, N)
        h_full = irfft(fw * fa_rev, N)
        h = h_full[n - 1:n - 1 + n]
        grad_a = 2.0 * h

        # Convert to x-space gradient:
        # df/dx[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])
        grad_weighted_mean = np.dot(a, grad_a)
        grad_x = a * (grad_a - grad_weighted_mean)

        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back to a
        x_new_shifted = x_new - x_new.max()
        a_new = np.exp(x_new_shifted)
        a_new = a_new / np.sum(a_new)

        # Evaluate
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            x = x_new
            a = a_new
            # Adaptive lr: increase slightly on success
            lr = min(lr * 1.002, 3.0)
        else:
            # Decrease lr on failure
            lr *= 0.95
            if lr < 1e-10:
                break
            # Keep x and a at best point
            x = np.log(best_a) - np.log(best_a).max()
            a = best_a.copy()

        if verbose and it % 5000 == 0:
            print(f"    LogGD it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.4f}")

    return best_a


def peak_subgradient_descent(a, time_budget=30.0):
    """
    Subgradient method targeting the peak of convolution.
    Uses the steepest subgradient direction.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    eta = 0.1

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Find all positions achieving (near) the maximum
        b_max = np.max(b)
        peak_mask = b >= b_max - 1e-10 * b_max
        peak_indices = np.where(peak_mask)[0]

        # Subgradient: average of gradients at peak positions
        # grad[i] = 2 * mean_{k in peaks} a[k-i]
        grad = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad = 2.0 * grad / len(peak_indices)

        # Project onto tangent space (sum=1): center the gradient
        grad = grad - np.mean(grad)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.97
            if eta < 1e-8:
                # Try random perturbation
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                eta = 0.1

    return best_a


def monotone_init(n, rng, mode=0):
    """Generate monotone (increasing) initializations."""
    if mode == 0:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
    elif mode == 1:
        # Exponential ramp
        a = np.exp(np.linspace(0, 2, n))
    elif mode == 2:
        # Quadratic ramp
        x = np.linspace(0, 1, n)
        a = 0.1 + x ** 2
    elif mode == 3:
        # Power ramp
        x = np.linspace(0, 1, n)
        a = 0.1 + x ** 3
    elif mode == 4:
        # Sigmoid ramp
        x = np.linspace(-3, 3, n)
        a = 1.0 / (1.0 + np.exp(-x))
    else:
        # Random monotone
        increments = rng.uniform(0.1, 1.0, n)
        a = np.cumsum(increments)
    return a / np.sum(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initialization patterns."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Edge-concentrated (U-shaped)
        x = np.arange(n) / max(n - 1, 1)
        a = 0.1 + 4.0 * (x - 0.5) ** 2
        a = a / np.sum(a)
    elif mode == 3:
        # Monotone increasing
        return monotone_init(n, rng, mode=rng.integers(0, 5))
    elif mode == 4:
        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        a = np.exp(-((np.arange(n) - c1) ** 2) / max((n // 8) ** 2, 1))
        a += np.exp(-((np.arange(n) - c2) ** 2) / max((n // 8) ** 2, 1))
        a += 0.01
        a = a / np.sum(a)
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.1 + 0.5 * (1 + np.cos(x))
        a = a / np.sum(a)
    elif mode == 6:
        # Sparse peaks
        n_peaks = max(2, n // 100)
        a = np.zeros(n)
        for _ in range(n_peaks):
            c = rng.integers(0, n)
            width = max(1, n // 50)
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * width ** 2))
        a += 0.001
        a = a / np.sum(a)
    else:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x ** 0.5
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def combined_polish(a, time_budget=60.0, verbose=False):
    """Combined polish: alternate between LogGD and LP."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # Alternate between methods
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP step
        alloc_lp = min(remaining * 0.5, 30.0)
        a_lp = lp_polish(a, time_budget=alloc_lp)
        obj_lp = compute_obj_normalized(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp.copy()
            if verbose:
                print(f"    Polish LP: {best_obj:.8f}")

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LogGD step
        alloc_gd = min(remaining * 0.5, 30.0)
        a_gd = logspace_gradient_descent(a, time_budget=alloc_gd, T_max=30.0)
        obj_gd = compute_obj_normalized(a_gd)
        if obj_gd < best_obj:
            best_obj = obj_gd
            best_a = a_gd.copy()
            a = a_gd.copy()
            if verbose:
                print(f"    Polish GD: {best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: log-parameterized GD + intensive LP polishing + multi-restart.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0  # 30s buffer for safety

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent loaded (n={parent.size}): obj={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization with n={n_best}, obj={best_score:.8f}")

    # === Phase 1: Log-parameterized GD (main workhorse) ===
    print(f"\n=== Phase 1: Log-space GD with annealing (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 250.0)
        a_gd = logspace_gradient_descent(best_seq, time_budget=alloc, T_max=50.0, verbose=True)
        obj_gd = compute_obj_normalized(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            n_best = len(best_seq)
            print(f"  ** LogGD: obj={best_score:.8f} **")
        else:
            print(f"  LogGD: obj={obj_gd:.8f} (no improvement)")

    # === Phase 2: Intensive LP polishing ===
    print(f"\n=== Phase 2: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alloc = min(remaining * 0.3, 350.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj_normalized(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            n_best = len(best_seq)
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Second LogGD round (higher temperature for exploration) ===
    print(f"\n=== Phase 3: LogGD Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 150.0)
        a_gd2 = logspace_gradient_descent(best_seq, time_budget=alloc, T_max=30.0)
        obj_gd2 = compute_obj_normalized(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            n_best = len(best_seq)
            print(f"  ** LogGD2: obj={best_score:.8f} **")
        else:
            print(f"  LogGD2: obj={obj_gd2:.8f}")

    # === Phase 4: Peak subgradient ===
    print(f"\n=== Phase 4: Peak Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.08, 60.0)
        a_peak = peak_subgradient_descent(best_seq, time_budget=alloc)
        obj_peak = compute_obj_normalized(a_peak)
        if obj_peak < best_score:
            best_score = obj_peak
            best_seq = a_peak.copy()
            print(f"  ** PeakSGD: obj={best_score:.8f} **")
        else:
            print(f"  PeakSGD: obj={obj_peak:.8f}")

    # === Phase 5: Multi-n exploration with diverse initializations ===
    print(f"\n=== Phase 5: Multi-n / Multi-init Exploration ===")
    n_candidates = sorted(set([
        n_best, n_best - 100, n_best + 100,
        n_best + 200, n_best - 200,
        600, 900, 1000, 1200, 1500
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2000]
    # Remove duplicates and sort by proximity to current best
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_best))
    # Limit to at most 8 different n values
    n_candidates = n_candidates[:8]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Resample best to new length
        a_resampled = resample_sequence(best_seq, n_try)

        # Also try a few random inits at this length
        inits = [a_resampled]
        for m in range(3):
            inits.append(diverse_init(n_try, rng, mode=rng.integers(0, 7)))

        best_at_n = a_resampled
        best_obj_at_n = compute_obj_normalized(a_resampled)

        for a_init in inits:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Quick LogGD
            alloc_gd = min(remaining * 0.15, 40.0)
            a_opt = logspace_gradient_descent(a_init, time_budget=alloc_gd, T_max=40.0)
            obj_opt = compute_obj_normalized(a_opt)

            # Quick LP polish
            remaining = deadline - time.monotonic()
            if remaining > 15 and obj_opt < best_obj_at_n + 0.1:
                alloc_lp = min(remaining * 0.2, 30.0)
                a_lp = lp_polish(a_opt, time_budget=alloc_lp)
                obj_lp = compute_obj_normalized(a_lp)
                if obj_lp < obj_opt:
                    a_opt = a_lp
                    obj_opt = obj_lp

            if obj_opt < best_obj_at_n:
                best_obj_at_n = obj_opt
                best_at_n = a_opt

        if best_obj_at_n < best_score:
            best_score = best_obj_at_n
            best_seq = best_at_n.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: best={best_obj_at_n:.8f}")

    # === Phase 6: Final intensive LP polish ===
    print(f"\n=== Phase 6: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.35, 200.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 7: Combined polish (LP + GD alternating) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Combined Polish ===")
        alloc = min(remaining * 0.3, 100.0)
        a_combined = combined_polish(best_seq, time_budget=alloc, verbose=True)
        obj_combined = compute_obj_normalized(a_combined)
        if obj_combined < best_score:
            best_score = obj_combined
            best_seq = a_combined.copy()
            print(f"  ** Combined: obj={best_score:.8f} **")
        else:
            print(f"  Combined: obj={obj_combined:.8f}")

    # === Phase 8: Final peak chase ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 40.0)
        a_pc = peak_subgradient_descent(best_seq, time_budget=alloc)
        obj_pc = compute_obj_normalized(a_pc)
        if obj_pc < best_score:
            best_score = obj_pc
            best_seq = a_pc.copy()
            print(f"  ** PeakChase: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val < np.inf:
        # Use the evaluator's value (it clips negatives, so should match)
        best_score_val = val
        print(f"  Evaluator value: {best_score_val:.8f} vs computed: {best_score:.8f}")
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
