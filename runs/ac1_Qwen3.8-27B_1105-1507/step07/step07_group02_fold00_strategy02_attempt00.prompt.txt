def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def gradient_descent_optimize(a, time_budget=60.0, verbose=False):
    """
    Gradient-based optimization using subgradient of the peak convolution value.
    Uses the insight that for non-negative sequences, the max is at center (n-1),
    so we minimize sum(a[i]^2) subject to sum(a)=const, which is minimized by
    making the sequence as flat as possible.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient direction: reduce a[i] where it contributes most to peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        # Center the gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Adaptive step size
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.8
            if lr < 1e-6:
                # Try perturbation
                rng = np.random.default_rng(it * 31 + 7)
                a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                    else:
                        a = best_a.copy()
                lr = 0.05

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Smooth random via Gaussian kernel
        a = rng.uniform(0, 1, n)
        kernel_size = max(3, n // 50)
        kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size+1))**2 / (kernel_size/2)**2)
        a = np.convolve(a, kernel / kernel.sum(), mode='same')
        a = np.maximum(a, 0)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 3:
        # Bimodal
        centers = [n * rng.uniform(0.2, 0.4), n * rng.uniform(0.6, 0.8)]
        widths = [n * rng.uniform(0.05, 0.15)] * 2
        x = np.arange(n)
        a = sum(np.exp(-((x - c)**2) / (2*w**2)) for c, w in zip(centers, widths))
        a = a / np.sum(a)
    else:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x)**2 + 0.1
        a = a / np.sum(a)
    return a
