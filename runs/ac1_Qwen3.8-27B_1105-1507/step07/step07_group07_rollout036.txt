                if stale_count_to_restart(stagnant):
                    # Perturb and restart
                    rng_local = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                    a = np.maximum(a, 1e-30)
                    a = a / np.sum(a)
                    eta = 1.0 / n
                    stagnant = 0
                if stagnant > 200:
                    break
            else:
                if stagnant > 0:
                    stagnant = 0
                # Slowly increase eta if improving consistently
                eta = min(eta * 1.002, 5.0 / n)

            # Periodic verbose
            if verbose and int((time.monotonic() - start) * 2) % 10 == 0:
                pass  # Keep it quiet for speed

    return best_a


def stale_count_to_restart(stagnant):
    return stagnant > 50


def peak_attack_polish(a, time_budget, verbose=False):
    """
    Direct peak-attacking: identify top peaks and push mass away from them.
    Uses very small steps for careful polishing.
    """
    n = len(a)
    a = np.maximum(a, 1e-30).astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    corr_size = next_pow2(3 * n - 2)
    conv_size = next_pow2(2 * n - 1)

    # Use a low temperature to focus on top peaks
    tau = 0.3 / n

    while time.monotonic() - start < time_budget:
        fa = rfft(a, conv_size)
        conv = irfft(fa * fa, conv_size)[:2 * n - 1]

        c_max = float(np.max(conv))
        exp_c = np.exp((conv - c_max) / tau)
        sum_exp = float(np.sum(exp_c))
        w = exp_c / sum_exp

        fa_flip = rfft(a[::-1], corr_size)
        fw = rfft(w, corr_size)
        conv_aw = irfft(fa_flip * fw, corr_size)
        corr = conv_aw[n - 1: n - 1 + n]
        grad = 4.0 * n * corr

        # Very small step
        eta = 0.3 / n
        log_a = np.log(a) - eta * grad
        log_a = log_a - np.max(log_a)
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try even smaller step
            eta2 = 0.1 / n
            log_a = np.log(a) - eta2 * grad
            log_a = log_a - np.max(log_a)
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Stuck - perturb slightly
                rng_local = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng_local.normal(0, 0.01, n))
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    if new_n == 1:
        return np.array([float(a[0])])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def smooth_init(n, rng):
    """Generate a smooth initial sequence with good autocorrelation properties."""
    mode = rng.integers(0, 5)
    if mode == 0:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
    elif mode == 1:
        # Gaussian window
        x = np.arange(n) - n / 2
        a = np.exp(-x**2 / (2 * (n/3)**2))
    elif mode == 2:
        # Triangle with smoothing
        x = np.abs(np.arange(n) - n / 2) / (n / 2)
        a = 1.0 - x**0.5
    elif mode == 3:
        # Flat with smooth edges
        edge = n // 8
        a = np.ones(n)
        a[:edge] = 0.5 + 0.5 * np.linspace(0.3, 1, edge)
        a[-edge:] = 0.5 + 0.5 * np.linspace(1, 0.3, edge)
    else:
        # Random smooth
        a = rng.uniform(0.5, 1.5, n)
        # Apply boxcar filter
        ksize = max(3, n // 20)
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 1e-3)
    a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid of multiplicative gradient descent with adaptive temperature,
    peak-attacking polish, multi-n exploration, and diversified restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = smooth_init(800, rng)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"Starting optimization from n={n_parent}, obj={best_score:.10f}")

    # === Phase 1: Multi-n exploration with quick gradient ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    n_candidates = sorted(set([
        600, 700, 750, 800, 850, 900, 950, 1000, 1050, 1100, 1200
    ]))
    # Add parent n and neighbors
    n_candidates = sorted(set(n_candidates + [n_parent, n_parent - 50, n_parent + 50]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1400]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try) if best_seq is not None else smooth_init(n_try, rng)

        # Quick gradient optimization (5-15s)
        alloc = min(remaining * 0.12, 20.0)
        tau_quick = [10.0/n_try, 5.0/n_try, 2.0/n_try, 1.0/n_try]
        a_opt = multiplicative_grad_descent(a_try, time_budget=alloc, tau_list=tau_quick)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 2: Deep gradient optimization on best ===
    print(f"\n=== Phase 2: Deep Gradient (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        tau_full = [20.0/n_best, 10.0/n_best, 5.0/n_best, 3.0/n_best,
                    2.0/n_best, 1.0/n_best, 0.5/n_best, 0.3/n_best, 0.2/n_best]
        a_deep = multiplicative_grad_descent(best_seq, time_budget=min(remaining * 0.3, 200.0),
                                             tau_list=tau_full, verbose=True)
        obj = compute_obj(a_deep)
        if obj < best_score:
            best_score = obj
            best_seq = a_deep.copy()
            print(f"  After deep gradient: obj={best_score:.10f}")
        else:
            print(f"  Deep gradient: obj={obj:.10f} (no improve)")

    # === Phase 3: Peak-attacking polish ===
    print(f"\n=== Phase 3: Peak Attack (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_peak = peak_attack_polish(best_seq, time_budget=min(remaining * 0.2, 100.0))
        obj = compute_obj(a_peak)
        if obj < best_score:
            best_score = obj
            best_seq = a_peak.copy()
            print(f"  After peak attack: obj={best_score:.10f}")
        else:
            print(f"  Peak attack: obj={obj:.10f} (no improve)")

    # === Phase 4: Fine-tuning n around best ===
    print(f"\n=== Phase 4: Fine n-tuning ===")
    n_best = len(best_seq)
    n_fine = sorted(set([n_best - 20, n_best - 10, n_best, n_best + 10, n_best + 20,
                         n_best + 50, n_best - 50]))
    n_fine = [n for n in n_fine if 300 <= n <= 1600]

    for n_try in n_fine:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.08, 30.0)
        tau_fine = [5.0/n_try, 3.0/n_try, 2.0/n_try, 1.0/n_try, 0.5/n_try]
        a_opt = multiplicative_grad_descent(a_try, time_budget=alloc, tau_list=tau_fine)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 5: Diversified restarts ===
    print(f"\n=== Phase 5: Diversified Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    restart_budget = min(25.0, remaining * 0.06)
    n_restarts = max(3, int(remaining / (restart_budget + 10)))
    n_restarts = min(n_restarts, 8)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < restart_budget + 5:
            break

        if r == 0:
            # Small perturbation of best
            a_r = best_seq * np.exp(rng.normal(0, 0.02, n_best))
            a_r = np.maximum(a_r, 1e-30)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            # Different n
            n_r = n_best + int(rng.integers(-80, 80))
            n_r = max(400, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.02, len(a_r)))
            a_r = np.maximum(a_r, 1e-30)
            a_r = a_r / np.sum(a_r)
        elif r == 2:
            # Symmetric enforcement
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = np.maximum(a_r, 1e-30)
            a_r = a_r / np.sum(a_r)
        elif r == 3:
            # Anti-symmetric perturbation (push edges up, center down or vice versa)
            x = np.linspace(-1, 1, n_best)
            shape = 1.0 + 0.3 * x**2
            a_r = best_seq * shape
            a_r = np.maximum(a_r, 1e-30)
            a_r = a_r / np.sum(a_r)
        elif r == 4:
            # Fresh smooth init
            a_r = smooth_init(n_best, rng)
        else:
            # Mix of best and random smooth
            a_mix = smooth_init(n_best, rng)
            a_r = 0.7 * best_seq + 0.3 * a_mix
            a_r = np.maximum(a_r, 1e-30)
            a_r = a_r / np.sum(a_r)

        n_r = len(a_r)
        tau_r = [5.0/n_r, 3.0/n_r, 2.0/n_r, 1.0/n_r, 0.5/n_r]
        a_opt = multiplicative_grad_descent(a_r, time_budget=restart_budget, tau_list=tau_r)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_r
            print(f"  Restart {r} (n={n_r}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (n={n_r}): obj={obj:.10f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    if remaining > 30:
        # More gradient with finer temperature
        tau_final = [2.0/n_best, 1.0/n_best, 0.5/n_best, 0.3/n_best, 0.2/n_best, 0.1/n_best]
        alloc1 = min(remaining * 0.3, 80.0)
        a_f1 = multiplicative_grad_descent(best_seq, time_budget=alloc1, tau_list=tau_final)
        obj = compute_obj(a_f1)
        if obj < best_score:
            best_score = obj
            best_seq = a_f1.copy()
            print(f"  Final grad: obj={best_score:.10f}")

        # Peak attack
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.3, 50.0)
            a_f2 = peak_attack_polish(best_seq, time_budget=alloc2)
            obj = compute_obj(a_f2)
            if obj < best_score:
                best_score = obj
                best_seq = a_f2.copy()
                print(f"  Final peak: obj={best_score:.10f}")

        # One more gradient round
        remaining = deadline - time.monotonic()
        if remaining > 15:
            tau_very_fine = [0.5/n_best, 0.3/n_best, 0.2/n_best, 0.1/n_best]
            alloc3 = min(remaining * 0.3, 40.0)
            a_f3 = multiplicative_grad_descent(best_seq, time_budget=alloc3, tau_list=tau_very_fine)
            obj = compute_obj(a_f3)
            if obj < best_score:
                best_score = obj
                best_seq = a_f3.copy()
                print(f"  Final fine grad: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val < np.inf:
        best_score = min(best_score, val)
    else:
        # Should not happen, but fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
