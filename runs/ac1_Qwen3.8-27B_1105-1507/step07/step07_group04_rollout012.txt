def subgradient_optimize(a, time_budget=100.0, top_k=10, verbose=False):

    Subgradient optimization with Nesterov acceleration.

    a = np.maximum(a, 0).astype(np.float64).copy()

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.9
    step_size = 0.05
    x = a.copy()

        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        k_indices = top_k_indices(conv, top_k)
        
        # Subgradient: for each k in argmax, grad[j] += a[k-j] (when valid)
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                grad[j] += a[k - j]
        grad /= len(k_indices)
        
        # Center gradient (work on simplex)
        grad_centered = grad - np.mean(grad)
        
        # Momentum
        momentum = beta * momentum - step_size * grad_centered
        new_a = a + momentum
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            new_a = a.copy()
            s_new = 1.0
        new_a = new_a / s_new

        # Nesterov
        t_k = 2.0 / (it + 2.0)
        x = new_a + (1 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))

            x = x / sx

        a = new_a
        obj = compute_obj(a)

            best_a = a.copy()
        
        # Adaptive step size

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized step function search using full-constraint LP alternating with warm-starting,
Nesterov-accelerated subgradient, golden-section line search, and multi-length exploration
with structured initializations. Targets objective below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """FFT-based autocorrelation via convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        j_lo = max(0, 0)  # g index = k - j, k ranges over valid range
        # For row k, column g_idx: A[k, g_idx] = a[j] where g_idx = k - j
        # k ranges from 0 to 2n-2, and we need 0 <= k-j < n, so k ranges from j to j+n-1
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        g_lo = k_lo - j
        g_hi = k_hi - j
        A[k_lo:k_hi + 1, g_lo:g_hi + 1] = a[j]
    return A


def build_conv_matrix_fast(a):
    """Faster construction using vectorized indexing."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            g_indices = k - np.arange(j_lo, j_hi + 1)
            A[k, g_indices] = a[j_lo:j_hi + 1]
    return A


def solve_full_lp(a, warm_g=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_fast(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None

def golden_section_line_search(a, g_lp, n_evals=8):
    """Find optimal mixing parameter t in (0,1) via golden section search."""
    n = len(a)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    inv_phi = 1.0 / phi
    inv_phi2 = 1.0 / (phi * phi)

    lo, hi = 0.0, 1.0

    def obj_at_t(t):
        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)

    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)

    for _ in range(n_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = obj_at_t(d)

    best_t = (lo + hi) / 2.0
    best_obj = obj_at_t(best_t)

    # Also check t=0 (current) and t=1 (full LP solution)
    obj_0 = compute_obj(a)
    obj_1 = obj_at_t(1.0)
    if obj_1 < best_obj:
        best_t, best_obj = 1.0, obj_1
    if obj_0 < best_obj:
        best_t, best_obj = 0.0, obj_0

    return best_t, best_obj

                # Check if perturbation improved
                obj_pert = compute_obj(a)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a.copy()
                perturbation_count += 1
                stall = 0

    return best_a


def subgradient_nesterov(a, time_budget=100.0, top_k=10, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.85
    step_size = 0.08
    x = a.copy()

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        k_indices = np.argsort(conv)[::-1][:top_k]

        # Compute subgradient
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] += a[k - j_lo:k - j_hi + 1][::-1]
        grad /= len(k_indices)

        # Center on simplex
        grad_c = grad - np.mean(grad)

        # Momentum update
        momentum = beta * momentum - step_size * grad_c
        new_a = a + momentum
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            new_a = a.copy()
            s_new = 1.0
        new_a = new_a / s_new

        # Nesterov lookahead
        t_k = 2.0 / (it + 2.0)
        x = new_a + (1.0 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = new_a
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Decay step size
        if it % 300 == 299:
            step_size = max(step_size * 0.85, 0.003)

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation on [0,1]."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def make_symmetric(a):
    """Symmetrize a sequence."""
    return (a + a[::-1]) / 2.0


def cosine_window(n):
    """Generate a cosine window of length n."""
    x = np.linspace(0, np.pi, n)
    w = 0.5 * (1 + np.cos(x))
    return w / np.sum(w)


def triangle_window(n):
    """Generate a triangle window."""
    mid = n // 2
    if n % 2 == 0:
        up = np.linspace(1, 0, n // 2)
        w = np.concatenate([up[::-1], up])
    else:
        up = np.linspace(1, 0, (n - 1) // 2)
        w = np.concatenate([up[::-1], [1.0], up])
    return w / np.sum(w)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP alternating + Nesterov subgradient +
    multi-length exploration with structured initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    print(f"Starting: n={len(best_seq)}, score={best_score:.8f}, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: LP Alternating (primary optimization) ===
    print("\n=== Phase 1: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(400.0, remaining * 0.35)
        a1 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  LP Alt: obj={best_score:.8f} **")
        else:
            print(f"  LP Alt: obj={obj1:.8f}")

    # === Phase 2: Subgradient polish ===
    print("\n=== Phase 2: Subgradient ===")
    remaining = deadline - time.monotonic()
    for k_val in [5, 15, 30]:
        if remaining < 30:
            break
        alloc = min(50.0, remaining * 0.12)
        a2 = subgradient_nesterov(best_seq, time_budget=alloc, top_k=k_val)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  SG k={k_val}: obj={best_score:.8f} **")
        else:
            print(f"  SG k={k_val}: obj={obj2:.8f}")
        remaining = deadline - time.monotonic()

    # === Phase 3: Second LP round (refined) ===
    print("\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(300.0, remaining * 0.3)
        a3 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  LP R2: obj={best_score:.8f} **")
        else:
            print(f"  LP R2: obj={obj3:.8f}")

    # === Phase 4: Multi-length exploration ===
    print("\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_parent = len(best_seq)
    n_options = sorted(set([
        300, 400, 500, 600, 700, 800, 900,
        n_parent, n_parent + 50, n_parent + 100, n_parent + 200,
        1000, 1200, 1500,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000 and n != n_parent]

    # Sort by distance from parent n (try closest first)
    n_options.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_options[:8]:  # Limit to 8 trials
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc_lp = min(30.0, remaining * 0.35)
        alloc_sg = min(15.0, remaining * 0.2)

        a_opt = lp_alternating(a_try, time_budget=alloc_lp)
        a_opt = subgradient_nesterov(a_opt, time_budget=alloc_sg, top_k=10)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Structured initializations ===
    print("\n=== Phase 5: Structured Init ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    structures = [
        ("symmetric", make_symmetric(best_seq)),
        ("cosine", cosine_window(n_best)),
        ("triangle", triangle_window(n_best)),
    ]
    # Also try symmetric version of best
    for name, a_struct in structures:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_struct = np.maximum(a_struct, 0)
        s = float(np.sum(a_struct))
        if s < 1e-10:
            continue
        a_struct = a_struct / s
        alloc = min(40.0, remaining * 0.25)
        a_opt = lp_alternating(a_struct, time_budget=alloc)
        obj_s = compute_obj(a_opt)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_opt.copy()
            print(f"  {name} (n={n_best}): obj={best_score:.8f} **")
        else:
            print(f"  {name} (n={n_best}): obj={obj_s:.8f}")

    # === Phase 6: Final LP polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(100.0, remaining * 0.5)
        a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_f:.8f}")

    # === Phase 7: Random restarts with small perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Fine-tuning Restarts ===")
        n_best = len(best_seq)
        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            # Very small perturbation
            scale = 0.005 + 0.003 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
            alloc = min(40.0, remaining * 0.4)
            a_opt = lp_alternating(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Fine {r}: obj={best_score:.8f} **")
            else:
                print(f"  Fine {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
