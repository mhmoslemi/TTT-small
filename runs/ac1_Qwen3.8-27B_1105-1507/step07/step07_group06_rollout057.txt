    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def softmax(theta):
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)

def top_k_smooth_obj_and_grad(theta, k, tau):
    """Compute top-k smoothed objective and gradient w.r.t. theta."""
    n = len(theta)
    a = softmax(theta)
    conv = compute_conv(a)
    
    # Get top-k indices
    if k >= 2 * n - 1:
        top_k_idx = np.arange(2 * n - 1)
    else:
        top_k_idx = np.argpartition(conv, -k)[-k:]
    
    top_vals = conv[top_k_idx]
    m = np.max(top_vals)
    smooth_max_val = m + np.log(np.sum(np.exp((top_vals - m) * tau))) / tau
    obj_val = 2.0 * n * smooth_max_val
    
    # Soft weights
    w_top = tau * np.exp((top_vals - m) * tau)
    w_top = w_top / np.sum(w_top)
    
    # Full weight vector
    w = np.zeros(2 * n - 1)
    w[top_k_idx] = w_top
    
    # Compute c[i] = sum_{m=0}^{n-1} w[i+m] * a[m] via FFT
    # c = cross-correlation of w and a
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    A_flipped = a[::-1]
    c_full = irfft(rfft(w, size) * rfft(A_flipped, size), size)
    c = c_full[n-1:n-1+n]  # lags 0 to n-1
    
    mean_c = np.dot(a, c)
    grad_theta = 4.0 * n * a * (c - mean_c)
    
    return obj_val, grad_theta

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A
