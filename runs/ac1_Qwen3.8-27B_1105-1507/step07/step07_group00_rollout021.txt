```python
import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak_fast(a, M_init=None):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build A matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    # LP: min M subject to A@g <= M, sum(g) = 1, g >= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    
    return None, None

def solve_lp_max_sum(a, M_target):
    """
    Solve: max sum(g) s.t. conv(a,g)[k] <= M_target for all k, g >= 0.
    """
    n = len(a)
    a = a / np.sum(a)
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    A_ub = A
    b_ub = np.full(2 * n - 1, M_target)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
```

```python
def soft_max_subgradient(a, t=50.0):
    """Compute subgradient of soft_max of convolution."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    max_c = np.max(conv)
    exp_c = np.exp(t * (conv - max_c))
    weights = exp_c / exp_c.sum()
    
    # Gradient: sum_k weights[k] * d(conv[k])/da[j]
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] += weights[k] * a[j_start:j_end + 1][::-1]
    return g, float(np.max(conv))
```

```python
    n = len(a)
    a = a / a.sum()
    a = np.maximum(a, 1e-15)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    while time.monotonic() - start < time_budget:
        g, _ = soft_max_subgradient(a, t)
        # Mirror descent step (exponential family)
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = max(1e-5, lr * 0.999)
        else:
            lr = max(1e-5, lr * 0.95)
            
    return best_a
```

```python
"""
Enhanced hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast LP-alternating with HiGHS, smooth mirror descent (soft-max),
direct self-convolution subgradient, spectral perturbation, and multi-n exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS. Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n]), res.x[:n]
    return None, None, None


def solve_lp_max_sum(a, M_target):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M_target for all k, g >= 0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def subgradient_self_conv(a, k):
    """Subgradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def soft_max_conv_subgradient(a, t=30.0):
    """Compute soft-max weighted subgradient of the self-convolution peak."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    if max_c < 1e-15:
        return np.zeros(n), max_c
    exp_c = np.exp(t * (conv - max_c))
    weights = exp_c / exp_c.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-8:
            continue
        w = weights[k]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
    return g, max_c

def self_conv_subgradient_descent(a, time_budget=100.0, K=5, verbose=False):

    Direct subgradient descent on the self-convolution max, using top-K bins.
    Uses mirror descent (multiplicative) updates with adaptive step size.
    """

    a = np.maximum(a, 1e-15).astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.02
    min_lr = 1e-6
    stall = 0
    it = 0
    prev_conv = None
    prev_g = None
    prev_a = None

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Get top-K peak positions
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]

        # Softmax weights over top-K
        weights = np.exp((peak_vals - max_val) * 20.0 / max(max_val, 1e-10))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            g += w * subgradient_self_conv(a, int(k))

        # Barzilai-Borwein step size
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-15:
                bb = ss / abs(sy)
                lr = np.clip(bb, min_lr, 0.5)

        prev_a = a.copy()
        prev_g = g.copy()

        # Multiplicative (mirror) update
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                rng = np.random.default_rng(it * 101 + n)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(min_lr, lr * 0.5)
                stall = 0
                prev_a = None
                prev_g = None
            else:
                lr = max(min_lr, lr * 0.96)

        it += 1

    return best_a


def spectral_nudge(a, rng, strength=0.05):
    """
    Apply a small perturbation in the frequency domain to break symmetries.
    This modifies the phase of the FFT while preserving the magnitude approximately.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    # Add small random phase perturbation
    phase_noise = rng.normal(0, strength, fa.shape)
    fa_noisy = fa * np.exp(1j * phase_noise)
    a_new = irfft(fa_noisy, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def coordinate_refine(a, time_budget=30.0):
    """
    Paired coordinate descent: move mass between positions to reduce the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False

        # Find contributing pairs
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                contributors.append((contrib, i, j))
        contributors.sort(reverse=True)

        # Non-contributors
        contrib_set = set()
        for _, i, j in contributors:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib_list = [i for i in range(n) if i not in contrib_set]

        if not non_contrib_list:
            break

        for contrib, i, j in contributors[:30]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break
            for target in non_contrib_list[:8]:
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = frac * a[i]
                    if delta <= 0 or a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Inverted triangle
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # U-shaped
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 5:
        # Low-frequency random
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Tail-heavy
        x = np.linspace(0, 1, n)
        a = x**3 + (1-x)**3 + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Geometric (asymmetric)
        r = rng.uniform(0.7, 0.99)
        a = r ** np.arange(n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid of enhanced LP-alternating, smooth mirror descent,
    direct self-convolution subgradient, spectral nudging, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_target = 800
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Enhanced LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(remaining * 0.25, 350.0)

    if phase1_budget > 30:
        a_lp = lp_alternating_enhanced(best_seq, time_budget=phase1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP-Alt: obj={best_score:.8f} **")
        else:
            print(f"  LP-Alt: obj={obj_lp:.8f} (no improvement)")

    # Spectral nudge to break potential symmetry
    remaining = deadline - time.monotonic()
    if remaining > 100:
        a_nudged = spectral_nudge(best_seq, rng, strength=0.03)
        obj_nudged = compute_obj(a_nudged)
        if obj_nudged < best_score:
            best_score = obj_nudged
            best_seq = a_nudged.copy()
            print(f"  Spectral nudge: obj={best_score:.8f} **")

    # === Phase 2: Smooth Mirror Descent ===
    print(f"\n=== Phase 2: Smooth Mirror Descent ===")
    remaining = deadline - time.monotonic()
    phase2_budget = min(remaining * 0.15, 200.0)

    if phase2_budget > 30:
        a_smd = smooth_mirror_descent(best_seq, time_budget=phase2_budget, t_param=30.0, verbose=True)
        obj_smd = compute_obj(a_smd)
        if obj_smd < best_score:
            best_score = obj_smd
            best_seq = a_smd.copy()
            print(f"  SMD: obj={best_score:.8f} **")
        else:
            print(f"  SMD: obj={obj_smd:.8f} (no improvement)")

    # === Phase 3: LP Alternating round 2 ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    phase3_budget = min(remaining * 0.2, 250.0)

    if phase3_budget > 30:
        a_lp2 = lp_alternating_enhanced(best_seq, time_budget=phase3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP-Alt R2: obj={best_score:.8f} **")
        else:
            print(f"  LP-Alt R2: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: Direct Self-Conv Subgradient ===
    print(f"\n=== Phase 4: Self-Conv Subgradient ===")
    remaining = deadline - time.monotonic()
    phase4_budget = min(remaining * 0.1, 100.0)

    if phase4_budget > 20:
        a_sc = self_conv_subgradient_descent(best_seq, time_budget=phase4_budget, K=5, verbose=False)
        obj_sc = compute_obj(a_sc)
        if obj_sc < best_score:
            best_score = obj_sc
            best_seq = a_sc.copy()
            print(f"  SelfConv: obj={best_score:.8f} **")
        else:
            print(f"  SelfConv: obj={obj_sc:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 100),
        max(300, n_current - 50),
        min(1500, n_current + 50),
        min(1500, n_current + 100),
        min(1500, n_current + 200),
        600, 700, 800, 900, 1000, 1100
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1200]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_opt = lp_alternating_enhanced(a_try, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
        else:
            a_opt = a_try
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_restarts = 3

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r + 2) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        restart_budget = min(remaining * 0.2, 70.0)
        a_opt = lp_alternating_enhanced(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        md_alloc = min(remaining * 0.2, 60.0)
        a_md = smooth_mirror_descent(best_seq, time_budget=md_alloc, t_param=50.0)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final SMD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        lp_alloc = min(remaining * 0.25, 80.0)
        a_lp = lp_alternating_enhanced(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_alloc = min(remaining * 0.3, 30.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        sc_alloc = min(remaining * 0.3, 30.0)
        a_sc = self_conv_subgradient_descent(best_seq, time_budget=sc_alloc, K=3)
        obj_sc = compute_obj(a_sc)
        if obj_sc < best_score:
            best_score = obj_sc
            best_seq = a_sc.copy()
            print(f"  Final SC: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_last = lp_alternating_enhanced(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
