    b = irfft(fa * fa, N)[:2 * n - 1]

    b_max = float(np.max(b))
    b_shifted = b - b_max
    exp_vals = np.exp(beta * b_shifted)
    exp_sum = float(np.sum(exp_vals))

    if exp_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
        smooth_val = b_max
    else:
        w = exp_vals / exp_sum
        smooth_val = b_max + np.log(exp_sum) / beta

    # Gradient: grad[i] = 2 * sum_m w[i+m] * a[m]
    # Using FFT cross-correlation: corr[k] = sum_j w[j] * a[(j-k) mod N]
    # Grad[i] = 2 * corr[i] for i=0,...,n-1 (with proper zero-padding)
    N2 = 1
    while N2 < 3 * n - 1:
        N2 <<= 1
    W = rfft(w, N2)
    A = rfft(a, N2)
    corr = irfft(W * np.conj(A), N2)
    grad = 2.0 * corr[:n]

    return smooth_val, grad


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1} via clipping and normalization."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def fista_smooth_max(a, time_budget=60.0, beta_start=50.0, beta_end=50000.0,
                     n_beta_steps=15, iters_per_beta=2000, verbose=False):
    """
    FISTA-style optimization with smooth max and adaptive beta.
    Geometric beta schedule from smooth to sharp.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = project_simplex(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    for bs in range(n_beta_steps):
        if time.monotonic() - start > time_budget:
            break
        frac = bs / max(n_beta_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        # FISTA state
        x = best_a.copy()
        y = x.copy()
        k_fista = 1.0
        L = 2.0 * float(np.max(np.abs(x))) * 4.0  # rough Lipschitz estimate
        L = max(L, 1.0)
        stall = 0
        best_in_phase = float(np.max(compute_conv(y)))

        for it in range(iters_per_beta):
            if time.monotonic() - start > time_budget:
                break

            smooth_val, grad = compute_smooth_max_and_grad(y, beta, N)

            # Gradient step
            step = 1.0 / L
            z = y - step * grad
            z = project_simplex(z)

            # FISTA momentum update
            k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k_fista * k_fista))
            alpha = (1.0 - k_fista) / k_new
            y = z + alpha * (z - x)
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y > 1e-15:
                y = y / s_y
            x = z
            k_fista = k_new

            # Periodic objective check
            if it % 5 == 0:
                obj = compute_obj(x)
                if obj < best_obj:
                    best_obj = obj
                    best_a = x.copy()
                    stall = 0
                else:
                    stall += 1

            if stall > 100:
                # Reset momentum
                x = best_a.copy()
                y = x.copy()
                k_fista = 1.0
                stall = 0
                L = max(L * 0.9, 0.5)

            # Adaptive L
            if it % 50 == 0 and it > 0:
                L = min(L * 1.01, 10.0)

        if verbose and n_beta_steps > 1:
            elapsed = time.monotonic() - start
            print(f"      beta={beta:.1f}, obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a, time_limit=5.0)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 10:
                break

    return best_a


def peak_chase_fista(a, time_budget=30.0, verbose=False):
    """Focused optimization targeting the single peak using high beta."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    beta = 10000.0
    eta = 0.5
    momentum = 0.8
    v = np.zeros(n)

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))

        # Subgradient at the peak: grad[i] = 2 * a[k_star - i] (if in range)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Center to handle the sum constraint better
        grad = grad - np.mean(grad)

        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.002, 2.0)
            momentum = min(momentum + 0.0005, 0.95)
        else:
            eta *= 0.98
            v *= 0.7
            if eta < 1e-8:
                break

    return best_a


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Raised cosine (optimal-ish shape)
        x = np.arange(n) / n
        a = np.sin(np.pi * x) ** 2
        a = a / np.sum(a)
    elif mode == 6:
        # Sine power
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
        a = a / np.sum(a)
    else:
        # Gaussian centered
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.5) ** 2) + 0.05
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """

    """

    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"\nStarting optimization from n={n_best}, score={best_score:.8f}")

    # Phase 1: FISTA smooth max (main workhorse)
    print(f"\n=== Phase 1: FISTA Smooth Max (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.30, 300.0)
    a_fista = fista_smooth_max(best_seq, time_budget=alloc,
                               beta_start=50.0, beta_end=50000.0,
                               n_beta_steps=15, iters_per_beta=3000, verbose=True)

    else:
        print(f"  FISTA: obj={obj_fista:.8f}")

    # Phase 2: LP Alternating
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()

        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # Phase 3: FISTA again with higher beta
    print(f"\n=== Phase 3: FISTA High Beta ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.20, 180.0)
        a_fista2 = fista_smooth_max(best_seq, time_budget=alloc,
                                    beta_start=2000.0, beta_end=200000.0,
                                    n_beta_steps=8, iters_per_beta=2000, verbose=True)
        obj_fista2 = compute_obj(a_fista2)
        if obj_fista2 < best_score:
            best_score = obj_fista2
            best_seq = a_fista2.copy()
            print(f"  ** FISTA2: obj={best_score:.8f} **")
        else:
            print(f"  FISTA2: obj={obj_fista2:.8f}")

    # Phase 4: Peak Chase
    print(f"\n=== Phase 4: Peak Chase ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_pc = peak_chase_fista(best_seq, time_budget=min(remaining * 0.08, 40.0))
        obj_pc = compute_obj(a_pc)
        if obj_pc < best_score:
            best_score = obj_pc
            best_seq = a_pc.copy()
            print(f"  ** PeakChase: obj={best_score:.8f} **")
        else:
            print(f"  PeakChase: obj={obj_pc:.8f}")

    # Phase 5: Multi-n exploration

    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr + 100, n_curr + 200, n_curr + 500,
        n_curr - 100, n_curr - 200,
        1000, 1200, 1500, 2000, 2500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000 and n != n_curr]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

    return result
