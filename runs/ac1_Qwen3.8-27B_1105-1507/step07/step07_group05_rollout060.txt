"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating, multiplicative peak chasing, FFT-based energy GD,
and multi-length exploration with diverse initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Try scipy linprog first (fastest for this size)
    try:
        from scipy.optimize import linprog
        # Variables: g[0],...,g[n-1], M
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M
        
        # constraints: A@g - M <= 0 for each row
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        
        # sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, res.x[-1]
    except Exception:
        pass
    
    # Try cvxpy as fallback
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1, M >= 0]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, prob.value
            except Exception:
                continue
    except Exception:
        pass
    
    return None, np.inf


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP-based alternating optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    max_stall = 50
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, lp_val = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search for best mixing parameter
        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            a = best_a / float(np.sum(best_a))
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, lp_val={lp_val:.6f}")
        else:
            stall += 1
            if stall > max_stall:
                break
    
    return best_a


def multiplicative_peak_chase(a, time_budget=30.0, verbose=False):
    """Multiplicative peak chasing: a[i] *= exp(-eta * a[k*-i])."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))
        
        # Subgradient: grad[i] = a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        for i in range(i_min, i_max + 1):
            grad[i] = a[k_star - i]
        
        # Center the gradient (to maintain sum invariance approximately)
        grad = grad - np.mean(grad)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 1.0)
        else:
            eta *= 0.97
            if eta < 1e-6:
                break
    
    return best_a

"""

FFT-based energy gradient descent, subgradient methods, and multi-length

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build Toeplitz matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    result = np.empty((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        result[j:j + r, j] = a[:r]
        if j + r < 2 * n - 1:
            result[j + r:, j] = 0.0
    return result


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0, M>=0."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.column_stack([A, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={"presolve": True, "dual_feasibility_tolerance": 1e-9})
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[-1])
    except Exception:
        pass

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable(nonneg=True)
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.CVXOPT]:

                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # Phase 1: LP alternating (main workhorse)
    print("\n=== Phase 1: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.30, 300.0)
        a1 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        o1 = compute_obj(a1)
        if o1 < best_score:
            best_score, best_seq = o1, a1.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={o1:.8f}")

    # Phase 2: Multiplicative peak chase
    print("\n=== Phase 2: Multiplicative Peak Chase ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a2 = multiplicative_peak_chase(best_seq, time_budget=min(remaining * 0.10, 40.0))
        o2 = compute_obj(a2)
        if o2 < best_score:
            best_score, best_seq = o2, a2.copy()
            print(f"  ** MPC: obj={best_score:.8f} **")
        else:
            print(f"  MPC: obj={o2:.8f}")

    # Phase 3: FFT energy GD
    print("\n=== Phase 3: FFT Energy GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a3 = fft_energy_gd(best_seq, time_budget=min(remaining * 0.15, 80.0), verbose=True)
        o3 = compute_obj(a3)
        if o3 < best_score:
            best_score, best_seq = o3, a3.copy()
            print(f"  ** GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={o3:.8f}")

    # Phase 4: Subgradient top-k
    print("\n=== Phase 4: Subgradient Top-K ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a4 = subgradient_topk(best_seq, time_budget=min(remaining * 0.10, 40.0))
        o4 = compute_obj(a4)
        if o4 < best_score:
            best_score, best_seq = o4, a4.copy()
            print(f"  ** SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  SubGrad: obj={o4:.8f}")

    # Phase 5: Second LP round from improved point
    print("\n=== Phase 5: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a5 = lp_alternating(best_seq, time_budget=min(remaining * 0.20, 120.0), verbose=True)
        o5 = compute_obj(a5)
        if o5 < best_score:
            best_score, best_seq = o5, a5.copy()
            print(f"  ** LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={o5:.8f}")

    # Phase 6: Multi-n exploration
    print("\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best + 100, n_best + 200, n_best + 400, n_best + 600,
        max(400, n_best - 200), max(400, n_best - 400),
        1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500 and n != n_best]
    n_options = sorted(n_options)[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.20, 80.0)
        a_opt = combined_refine(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 7: Diverse random restarts
    print("\n=== Phase 7: Random Restarts ===")
    n_curr = len(best_seq)
    n_restarts = max(3, int((deadline - time.monotonic()) / 40.0))
    n_restarts = min(n_restarts, 10)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_r = diverse_init(n_curr, rng, mode=r % 8)
        alloc = min(remaining * 0.15, 35.0)
        a_opt = fft_energy_gd(a_r, time_budget=alloc)
        a_opt = multiplicative_peak_chase(a_opt, time_budget=min(remaining * 0.05, 10.0))
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r} (mode={r%8}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={r%8}): obj={obj_r:.8f}")

    # Phase 8: Final combined polish on best
    print("\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.40, 120.0)
        a_final = combined_refine(best_seq, time_budget=alloc, verbose=True)
        o_final = compute_obj(a_final)
        if o_final < best_score:
            best_score = o_final
            best_seq = a_final.copy()
            print(f"  ** Final: obj={best_score:.8f} **")

    # Phase 9: Last-second peak chase
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_pc = multiplicative_peak_chase(best_seq, time_budget=min(remaining * 0.5, 15.0))
        o_pc = compute_obj(a_pc)
        if o_pc < best_score:
            best_score = o_pc
            best_seq = a_pc.copy()
            print(f"  ** FinalPeak: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
