Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses soft-max smoothed gradient descent with FFT-based convolution,
multiple initialization strategies, and adaptive ε schedule.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

        return a

"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses soft-max smoothed gradient descent with FFT-based convolution,
multiple initialization strategies (parent, edge-weighted, perturbed),
adaptive learning rate and epsilon schedule, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def softmax_gd(a, time_budget=100.0, verbose=False):
    """
    Soft-max smoothed gradient descent to minimize max(conv(a,a))
    subject to sum(a) = 1, a >= 0.
    Returns the best sequence found.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_peak = float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr = 0.05
    eps_factor = 0.5
    min_eps = 0.003
    no_improve = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        conv_max = float(np.max(conv))

        # Soft-max weights
        eps = eps_factor * conv_max
        if eps < 1e-15:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            shifted = (conv - conv_max) / eps
            p = np.exp(shifted)
            p_sum = float(np.sum(p))
            if p_sum < 1e-30:
                p = np.zeros(len(conv))
                p[int(np.argmax(conv))] = 1.0
            else:
                p = p / p_sum

        # Gradient via cross-correlation
        grad = 2.0 * correlate(a, p, mode='valid')

        grad_c = grad - float(np.mean(grad))

        # Step
        a_new = a - lr * grad_c

            lr *= 0.5
            continue

            best_a = a_new.copy()
            a = a_new
            no_improve = 0

def peak_direct_reduce(a, time_budget=30.0, verbose=False):
    """
    Direct peak reduction: at each step, find the peak of conv and
    redistribute mass from peak-contributing indices to non-contributing ones.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_peak = float(np.max(compute_conv(a)))
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find indices j where a[k_star - j] is large (these contribute most to peak)
        # grad_sub[j] = a[k_star - j] if 0 <= k_star - j < n, else 0
        indices = np.arange(n)
        kir = k_star - indices  # k_star - j for j = 0, ..., n-1
        valid = (kir >= 0) & (kir < n)
        subgrad = np.zeros(n)
        subgrad[valid] = a[kir[valid]]

        # We want to decrease a[j] where subgrad[j] is large and increase where it's small
        # Use the "water-filling" idea: move mass from high-subgrad to low-subgrad
        # Step: a_new[j] = a[j] - lr * (subgrad[j] - mean(subgrad))
        lr = 0.01  # Small step
        a_new = a - lr * (subgrad - float(np.mean(subgrad)))
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        conv_new = compute_conv(a_new)
        peak_new = float(np.max(conv_new))

        if peak_new < best_peak - 1e-12:
            best_peak = peak_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            lr *= 0.5
            if lr < 1e-6:
                # Perturb
                rng = np.random.default_rng(it * 31 + 7)
                a = best_a * np.exp(rng.standard_normal(n) * 0.02)
                a = np.maximum(a, 0)
                s_val = float(np.sum(a))
                if s_val > 1e-10:
                    a = a / s_val
                    best_a = a.copy()
                    best_peak = float(np.max(compute_conv(a)))
                lr = 0.01

        it += 1

    return best_a


def edge_weighted(a, alpha):
    """Create edge-weighted version of a."""
    n = len(a)
    if n <= 1:
        return a.copy()
    x = np.linspace(-1, 1, n)
    weight = 1.0 + alpha * x**2
    a_new = a * weight
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses soft-max smoothed gradient descent with FFT-based convolution and cross-correlation
for gradient computation, L2 pre-optimization, edge-weighted perturbations,
multi-n exploration, and adaptive parameter scheduling.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def softmax_gd(a, time_budget=100.0, verbose=False, initial_eps=0.5, initial_lr=0.05):
    """
    Soft-max smoothed gradient descent to minimize max(conv(a,a))
    subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_peak = float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr = initial_lr
    eps_factor = initial_eps
    min_eps = 0.002
    no_improve = 0
    it = 0
    consecutive_fail = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        conv_max = float(np.max(conv))

        if conv_max < 1e-15:
            break

        # Soft-max weights (numerically stable)
        eps = eps_factor * conv_max
        if eps < 1e-14:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            shifted = (conv - conv_max) / eps
            p = np.exp(shifted)
            p_sum = float(np.sum(p))
            if p_sum < 1e-30:
                p = np.zeros(len(conv))
                p[int(np.argmax(conv))] = 1.0
            else:
                p = p / p_sum

        # Gradient via cross-correlation: grad[j] = 2 * sum_m a[m]*p[m+j]
        grad = 2.0 * correlate(a, p, mode='valid')

        # Center to preserve sum
        grad_c = grad - float(np.mean(grad))

        # Step with line search (try a few step sizes)
        best_step_obj = np.inf
        best_step_a = None
        for lr_try in [lr, lr * 0.5, lr * 2.0]:
            a_try = a - lr_try * grad_c
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = compute_conv(a_try)
            peak_try = float(np.max(conv_try))
            if peak_try < best_step_obj:
                best_step_obj = peak_try
                best_step_a = a_try

        if best_step_a is not None and best_step_obj < best_peak - 1e-14:
            best_peak = best_step_obj
            best_a = best_step_a.copy()
            a = best_step_a
            no_improve = 0
            consecutive_fail = 0
            lr = min(lr * 1.03, 0.5)
        else:
            # Accept anyway if it's not much worse (momentum-like)
            if best_step_a is not None and best_step_obj < best_peak * 1.01:
                a = best_step_a
            no_improve += 1
            lr *= 0.93
            consecutive_fail += 1

            if no_improve > 500 or consecutive_fail > 200:
                # Perturb and restart
                rng = np.random.default_rng(100000 + it * 13 + n)
                scale = 0.02 + 0.01 * min(no_improve // 500, 15)
                a = best_a * np.exp(rng.standard_normal(n) * scale)
                a = np.maximum(a, 0)
                s_val = float(np.sum(a))
                if s_val > 1e-10:
                    a = a / s_val
                    best_a = a.copy()
                    best_peak = float(np.max(compute_conv(a)))
                else:
                    a = np.ones(n) / n
                    best_a = a.copy()
                    best_peak = float(np.max(compute_conv(a)))
                no_improve = 0
                consecutive_fail = 0
                lr = initial_lr

        # Decay eps toward focusing on the true peak
        eps_factor = max(eps_factor * 0.9997, min_eps)
        it += 1

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      SM-GD it{it}: obj={2*n*best_peak:.8f}, "
                  f"lr={lr:.5f}, eps_f={eps_factor:.5f}, t={elapsed:.0f}s")

    return best_a


def l2_smooth_optimize(a, time_budget=60.0, verbose=False):
    """
    Minimize ||conv(a,a)||_2^2 (smooth objective) via gradient descent.
    This "flattens" the convolution, reducing the peak as a side effect.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_l2 = float(np.sum(compute_conv(a) ** 2))
    start = time.monotonic()

    # FFT size
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    lr = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - start < time_budget:
        A = np.fft.fft(a, size)
        # grad[j] = 2 * Re( size * ifft(|A|^2 * conj(A), size)[j] )
        C = np.abs(A) ** 2 * np.conj(A)
        grad_full = 2.0 * np.real(size * np.fft.ifft(C, size)[:n])

        # Center to preserve sum
        grad_c = grad_full - float(np.mean(grad_full))

        a_new = a - lr * grad_c
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        conv_new = compute_conv(a_new)
        l2_new = float(np.sum(conv_new ** 2))

        if l2_new < best_l2 - 1e-14:
            best_l2 = l2_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.02, 0.1)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 200:
                rng = np.random.default_rng(it * 41 + 3)
                a = best_a * np.exp(rng.standard_normal(n) * 0.02)
                a = np.maximum(a, 0)
                s_val = float(np.sum(a))
                if s_val > 1e-10:
                    a = a / s_val
                    best_a = a.copy()
                    best_l2 = float(np.sum(compute_conv(a) ** 2))
                no_improve = 0
                lr = 0.01

        it += 1
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      L2-opt it{it}: l2={best_l2:.10f}, "
                  f"peak_obj={2*n*float(np.max(compute_conv(best_a))):.8f}, t={elapsed:.0f}s")

    return best_a


def peak_direct_reduce(a, time_budget=30.0, verbose=False):
    """
    Direct peak reduction using subgradient at the argmax.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_peak = float(np.max(compute_conv(a)))
    start = time.monotonic()
    lr = 0.02
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient at the peak
        indices = np.arange(n)
        kir = k_star - indices
        valid = (kir >= 0) & (kir < n)
        subgrad = np.zeros(n)
        subgrad[valid] = a[kir[valid]]

        grad_c = subgrad - float(np.mean(subgrad))
        a_new = a - lr * grad_c
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        conv_new = compute_conv(a_new)
        peak_new = float(np.max(conv_new))

        if peak_new < best_peak - 1e-13:
            best_peak = peak_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.02, 0.2)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 100:
                # Try multiple peak indices (weighted subgradient)
                conv_sorted = np.sort(conv)[::-1]
                threshold = conv_sorted[min(10, len(conv_sorted) - 1)]
                peak_indices = np.where(conv >= threshold)[0]
                if len(peak_indices) > 1:
                    combined_grad = np.zeros(n)
                    for k_i in peak_indices[:5]:
                        kir_i = k_i - indices
                        v_i = (kir_i >= 0) & (kir_i < n)
                        combined_grad[v_i] += a[kir_i[v_i]]
                    combined_grad /= len(peak_indices[:5])
                    grad_c2 = combined_grad - float(np.mean(combined_grad))
                    a_try = a - lr * grad_c2
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st > 1e-10:
                        a_try = a_try / st
                        ct = compute_conv(a_try)
                        pt = float(np.max(ct))
                        if pt < best_peak:
                            best_peak = pt
                            best_a = a_try.copy()
                            a = a_try
                            no_improve = 0
                            continue

                # Perturb
                rng = np.random.default_rng(it * 17 + 11)
                a = best_a * np.exp(rng.standard_normal(n) * 0.03)
                a = np.maximum(a, 0)
                s_val = float(np.sum(a))
                if s_val > 1e-10:
                    a = a / s_val
                    best_a = a.copy()
                    best_peak = float(np.max(compute_conv(a)))
                no_improve = 0
                lr = 0.02

        it += 1

    return best_a


def edge_weighted(a, alpha, power=2):
    """Create edge-weighted version of a."""
    n = len(a)
    if n <= 1:
        return a.copy()
    x = np.linspace(-1, 1, n)
    weight = 1.0 + alpha * np.abs(x) ** power
    a_new = a * weight
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, np.maximum(a, 0))

    print(f"  Starting: n={n_curr}, score={best_score:.8f}")

    # === Phase 1: L2 pre-optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 45:
        print(f"\n=== Phase 1: L2 smoothing (budget={min(60,remaining*0.8):.0f}s) ===")
        a_p = best_seq.copy()
        s_p = float(np.sum(a_p))
        if s_p > 0:
            a_p = a_p / s_p
        alloc = min(60.0, remaining * 0.8)
        a_l2 = l2_smooth_optimize(a_p, time_budget=alloc, verbose=True)
        ev = evaluate_sequence([float(x) for x in a_l2])
        print(f"  L2 result: score={ev:.8f}")
        if ev < best_score:
            best_score = ev
            best_seq = a_l2.copy()
            print(f"  ** New best: {best_score:.8f} **")

    # === Phase 2: Main SM-GD from best ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: SM-GD main (budget={min(250,remaining*0.35):.0f}s) ===")
        a_m = best_seq.copy()
        s_m = float(np.sum(a_m))
        if s_m > 0:
            a_m = a_m / s_m
        alloc = min(250.0, remaining * 0.35)
        a_opt = softmax_gd(a_m, time_budget=alloc, verbose=True)
        ev = evaluate_sequence([float(x) for x in a_opt])
        print(f"  SM-GD: score={ev:.8f}")
        if ev < best_score:
            best_score = ev
            best_seq = a_opt.copy()
            print(f"  ** New best: {best_score:.8f} **")

    # === Phase 3: Edge-weighted variants ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: Edge-weighted variants ===")
        a_base = best_seq.copy()
        s_b = float(np.sum(a_base))
        if s_b > 0:
            a_base = a_base / s_b
        n_b = len(a_base)

        for alpha, power in [(0.3, 2), (0.5, 2), (1.0, 2), (0.5, 4), (1.0, 4), (2.0, 2)]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_e = edge_weighted(a_base, alpha, power)
            alloc = min(30.0, remaining * 0.15)
            a_opt = softmax_gd(a_e, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  alpha={alpha}, p={power}: ** score={best_score:.8f} **")
            else:
                print(f"  alpha={alpha}, p={power}: score={ev:.8f}")

    # === Phase 4: Peak direct reduction ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 4: Peak direct reduction ===")
        a_pd = best_seq.copy()
        s_pd = float(np.sum(a_pd))
        if s_pd > 0:
            a_pd = a_pd / s_pd
        alloc = min(40.0, remaining * 0.15)
        a_opt = peak_direct_reduce(a_pd, time_budget=alloc)
        ev = evaluate_sequence([float(x) for x in a_opt])
        if ev < best_score:
            best_score = ev
            best_seq = a_opt.copy()
            print(f"  Peak reduce: ** score={best_score:.8f} **")
        else:
            print(f"  Peak reduce: score={ev:.8f}")

    # === Phase 5: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Multi-n exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(300, n_best - 100), max(200, n_best - 50),
            n_best,
            n_best + 50, n_best + 100, n_best + 200,
            600, 800, 1000, 1200, 1500, 1800
        ]))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample(best_seq, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try
            alloc = min(30.0, remaining * 0.12)
            a_opt = softmax_gd(a_try, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  n={n_try}: ** score={best_score:.8f} **")
            else:
                print(f"  n={n_try}: score={ev:.8f}")

    # === Phase 6: Random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Random restarts ===")
        n_best = len(best_seq)
        per_restart = min(25.0, max(8.0, remaining * 0.04))
        max_restarts = min(30, int((remaining - 30) / per_restart))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < per_restart + 10:
                break

            mode = r % 6
            if mode == 0:
                a_r = best_seq * np.exp(rng.standard_normal(n_best) * 0.03)
            elif mode == 1:
                a_r = best_seq * np.exp(rng.standard_normal(n_best) * 0.1)
            elif mode == 2:
                a_e = edge_weighted(best_seq, rng.uniform(0.2, 1.5), rng.choice([2, 3, 4]))
                a_r = a_e * np.exp(rng.standard_normal(n_best) * 0.02)
            elif mode == 3:
                n_r = max(400, min(2000, n_best + int(rng.integers(-200, 200))))
                a_r = resample(best_seq, n_r)
                a_r = a_r * np.exp(rng.standard_normal(n_r) * 0.03)
            elif mode == 4:
                # Cosine-modulated
                n_r = n_best
                x = np.linspace(0, 2 * np.pi, n_r)
                mod = 1.0 + rng.uniform(0.05, 0.2) * np.cos(rng.uniform(1, 5) * x)
                a_r = best_seq * mod
                a_r = np.maximum(a_r, 0)
            else:
                # Smooth random
                n_r = n_best
                a_r = np.ones(n_r)
                freqs = rng.choice(np.arange(1, 10), size=3, replace=False)
                for f in freqs:
                    a_r += rng.uniform(0.05, 0.15) * np.sin(2 * np.pi * f * np.arange(n_r) / n_r + rng.uniform(0, 2 * np.pi))
                a_r = np.maximum(a_r, 0)

            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
                a_opt = softmax_gd(a_r, time_budget=per_restart)
                ev = evaluate_sequence([float(x) for x in a_opt])
                if ev < best_score:
                    best_score = ev
                    best_seq = a_opt.copy()
                    print(f"  Restart {r} (mode={mode}, n={len(a_opt)}): ** {best_score:.8f} **")
                else:
                    print(f"  Restart {r} (mode={mode}): {ev:.8f}")

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 7: Final polish (remaining={remaining:.0f}s) ===")
        a_final = best_seq.copy()
        s_f = float(np.sum(a_final))
        if s_f > 0:
            a_final = a_final / s_f

        # Long SM-GD with very low initial epsilon (fine-tuning)
        round_budget = min(remaining * 0.5, 200.0)
        if round_budget > 10:
            a_pol = softmax_gd(a_final, time_budget=round_budget,
                               initial_eps=0.05, initial_lr=0.02, verbose=True)
            ev = evaluate_sequence([float(x) for x in a_pol])
            if ev < best_score:
                best_score = ev
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Polish 1: ** score={best_score:.8f} **")
            else:
                print(f"  Polish 1: score={ev:.8f}")

        # Another round with even lower epsilon
        remaining = deadline - time.monotonic()
        if remaining > 15:
            round_budget = min(remaining * 0.5, 100.0)
            if round_budget > 10:
                a_pol = softmax_gd(a_final, time_budget=round_budget,
                                   initial_eps=0.02, initial_lr=0.01, verbose=True)
                ev = evaluate_sequence([float(x) for x in a_pol])
                if ev < best_score:
                    best_score = ev
                    best_seq = a_pol.copy()
                    print(f"  Polish 2: ** score={best_score:.8f} **")
                else:
                    print(f"  Polish 2: score={ev:.8f}")

        # Peak direct reduction as final touch
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_pd = best_seq.copy()
            s_pd = float(np.sum(a_pd))
            if s_pd > 0:
                a_pd = a_pd / s_pd
            alloc = min(20.0, remaining * 0.4)
            a_opt = peak_direct_reduce(a_pd, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  Peak reduce final: ** score={best_score:.8f} **")
            else:
                print(f"  Peak reduce final: score={ev:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
