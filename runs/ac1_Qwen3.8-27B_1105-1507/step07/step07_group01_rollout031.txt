```python
"""
Multiplicative Weights / Primal-Dual Mirror Descent optimizer for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based convolution for O(n log n) iterations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def multiplicative_weights(a, time_budget=60.0, eta=0.1, T_init=0.01, T_min=0.001, T_decay=0.9995):
    """
    Multiplicative weights / mirror descent for minimizing max(conv(a,a)).
    a should be normalized to sum 1.
    Uses FFT for convolution and direct correlate for gradient.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    T = T_init
    it = 0
    stall = 0
    best_stall_obj = best_obj
    
    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
            best_stall_obj = best_obj
        else:
            stall += 1
            if stall > 500:
                # Reset T to explore more
                T = min(T_init, T * 2)
                stall = 0
        
        # Smooth max weights (dual variable)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: A^T p = correlate(p, a, 'valid')
        grad = np.correlate(p, a, mode='valid')
        
        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -50, 50)  # numerical safety
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10 and np.isfinite(s_new):
            a = a_new / s_new
        else:
            # Fallback: small perturbation
            a = best_a * (1.0 + 0.01 * np.random.default_rng(it).normal(0, 1, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            break
        
        # Temperature schedule
        T = max(T_min, T * T_decay)
    
    return best_a


def lp_alternating_fast(a, time_budget=30.0):
    """
    Fast LP alternating using scipy linprog with the banded structure.
    For small n only.
    """
    try:
        from scipy.optimize import linprog
    except ImportError:
        return a
    
    n = len(a)
    if n > 500:  # Too slow for large n
        return a
    
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        # Build convolution matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a[:r]
        
        # Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        # Variables: [g_0, ..., g_{n-1}, M]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        g = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g))
        if s_g < 1e-10:
            break
        g = g / s_g
        
        # Line search for blend
        best_t_obj = np.inf
        best_t_val = 0.3
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a = ((1 - best_t_val) * best_a + best_t_val * g)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            best_a = a.copy()
        else:
            break
    
    return best_a


def mul_weights_with_dual(a, time_budget=60.0, eta=0.1, T_init=0.01, T_min=0.0005, T_decay=0.999):
    """
    Enhanced multiplicative weights with explicit dual tracking.
    Maintains both primal (x) and dual (y) variables.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Initialize dual uniformly
    y = np.ones(2 * n - 1) / (2 * n - 1)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    T = T_init
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Dual update: y_k <- y_k * exp(eta * conv_k / T) / Z
        # (normalize conv for numerical stability)
        conv_shifted = (conv - max_c) / T
        conv_shifted = np.clip(conv_shifted, -500, 500)
        y_new = y * np.exp(conv_shifted)
        s_y = float(np.sum(y_new))
        if s_y > 1e-10 and np.isfinite(s_y):
            y = y_new / s_y
        else:
            y = conv_shifted - np.min(conv_shifted)
            y = np.exp(y)
            y = y / float(np.sum(y))
        
        # Primal update: use y as weights
        grad = np.correlate(y, a, mode='valid')
        log_update = -eta * grad
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10 and np.isfinite(s_new):
            a = a_new / s_new
        else:
            break
        
        T = max(T_min, T * T_decay)
    
    return best_a


def multiplicative_weights_ftrl(a, time_budget=60.0, eta_init=1.0):
    """
    FTRL (Follow the Regularized Leader) form of multiplicative weights.
    Maintains running gradient sum for lazy updates.
    """

    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    grad_sum = np.zeros(n)
    eta = eta_init
    t = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Use softmax weights on conv
        T = 0.01
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        p = np.exp(shifted)
        p = p / float(np.sum(p))
        
        grad = np.correlate(p, a, mode='valid')
        grad_sum += grad
        
        # FTRL: x_i proportional to exp(-eta * grad_sum_i)
        eta_t = eta_init / np.sqrt(t)
        log_x = -eta_t * grad_sum
        log_x -= np.max(log_x)  # numerical stability
        a_new = np.exp(log_x)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10 and np.isfinite(s_new):
            a = a_new / s_new
        else:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
"""
Multiplicative Weights / Primal-Dual Mirror Descent optimizer for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based convolution for O(n log n) per iteration.
Key innovation: fast multiplicative updates exploiting the structure of the
convolution gradient, combined with adaptive temperature scheduling and
frequency-domain initialization for diverse starting points.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def multiplicative_weights(a, time_budget=60.0, eta=0.1, T_init=0.01, T_min=0.001, T_decay=0.9995):
    """
    Basic multiplicative weights / mirror descent for minimizing max(conv(a,a)).
    a should be normalized to sum 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()

    T = T_init
    stall = 0
    best_stall = best_obj

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
            best_stall = best_obj
        else:
            stall += 1
            if stall > 2000:
                T = min(T_init * 2, T * 3)
                stall = 0

        # Smooth max weights
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        p_total = float(np.sum(exp_vals))
        if p_total < 1e-300 or not np.isfinite(p_total):
            idx = int(np.argmax(conv))
            p = np.zeros(2 * n - 1)
            p[idx] = 1.0
        else:
            p = exp_vals / p_total

        # Gradient: A^T p
        grad = np.correlate(p, a, mode='valid')

        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -100, 100)
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))
        if s_new > 1e-300 and np.isfinite(s_new):
            a = a_new / s_new
        else:
            a = best_a.copy()
            break

        T = max(T_min, T * T_decay)

    return best_a


def mul_weights_dual(a, time_budget=60.0, eta=0.1, T_init=0.01, T_min=0.0005, T_decay=0.999):
    """
    Enhanced multiplicative weights with explicit dual variable tracking.
    The dual variable y is updated multiplicatively based on conv values.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Initialize dual uniformly
    y = np.ones(2 * n - 1) / (2 * n - 1)

    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()

    T = T_init

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        # Dual update: concentrate y where conv is large
        conv_shifted = (conv - max_c) / T
        conv_shifted = np.clip(conv_shifted, -500, 500)
        y_new = y * np.exp(conv_shifted)
        s_y = float(np.sum(y_new))
        if s_y > 1e-300 and np.isfinite(s_y):
            y = y_new / s_y
        else:
            y_shifted = conv_shifted - np.min(conv_shifted)
            y = np.exp(y_shifted)
            y = y / float(np.sum(y))

        # Primal update
        grad = np.correlate(y, a, mode='valid')
        log_update = -eta * grad
        log_update = np.clip(log_update, -100, 100)
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))
        if s_new > 1e-300 and np.isfinite(s_new):
            a = a_new / s_new
        else:
            a = best_a.copy()
            break

        T = max(T_min, T * T_decay)

    return best_a


def mul_weights_ftrl(a, time_budget=60.0, eta_init=1.0, T=0.01):
    """
    FTRL (Follow the Regularized Leader) form of multiplicative weights.
    Maintains running gradient sum for lazy probability updates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()

    grad_sum = np.zeros(n)
    t = 0

    while time.monotonic() - start < time_budget:
        t += 1
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        # Softmax weights
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        p = np.exp(shifted)
        p_total = float(np.sum(p))
        if p_total > 1e-300:
            p = p / p_total
        else:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(conv))] = 1.0

        grad = np.correlate(p, a, mode='valid')
        grad_sum += grad

        # FTRL: x_i proportional to exp(-eta_t * grad_sum_i)
        eta_t = eta_init / np.sqrt(t)
        log_x = -eta_t * grad_sum
        log_x -= np.max(log_x)
        a_new = np.exp(log_x)
        s_new = float(np.sum(a_new))
        if s_new > 1e-300 and np.isfinite(s_new):
            a = a_new / s_new
        else:
            break

    return best_a


def mul_weights_adaptive(a, time_budget=60.0, eta_init=0.3, eta_min=0.01):
    """
    Adaptive multiplicative weights with backtracking step size.
    Reduces eta when objective increases, increases when it decreases.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    prev_obj = best_obj
    start = time.monotonic()

    eta = eta_init
    T = 0.01
    T_min = 0.0005
    failed = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            # Increase eta slightly if making progress
            if failed == 0 and eta < eta_init:
                eta = min(eta_init, eta * 1.02)
            failed = 0
        elif obj > prev_obj * 1.01:
            failed += 1
            if failed > 5:
                eta = max(eta_min, eta * 0.7)
                T = min(0.05, T * 1.5)
                failed = 0
        prev_obj = obj

        # Smooth max weights
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        p_total = float(np.sum(exp_vals))
        if p_total > 1e-300 and np.isfinite(p_total):
            p = exp_vals / p_total
        else:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(conv))] = 1.0

        grad = np.correlate(p, a, mode='valid')

        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -100, 100)
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))
        if s_new > 1e-300 and np.isfinite(s_new):
            a = a_new / s_new
        else:
            a = best_a.copy()
            eta = max(eta_min, eta * 0.5)
            continue

        T = max(T_min, T * 0.9997)

    return best_a


def freq_domain_init(n, rng, mode=0):
    """
    Generate initial sequence using frequency domain construction.
    Design a target power spectrum, take its square root, and IFFT.
    """
    if mode == 0:
        # Flat spectrum (white noise) - gives uniform in time
        C = np.ones(n)
    elif mode == 1:
        # Band-limited
        C = np.ones(n)
        cutoff = max(1, n // 4)
        C[:cutoff] = 1.0
        C[-cutoff:] = 1.0
        C[cutoff:-cutoff] = 0.3
    elif mode == 2:
        # Raised cosine spectrum
        x = np.linspace(0, 1, n)
        C = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        C = np.maximum(C, 0.01)
    elif mode == 3:
        # Random smooth spectrum
        f = rfft(rng.uniform(0, 1, n))
        C = np.abs(f) ** 2
        C = np.maximum(C, 0.01)
    else:
        # Exponentially decaying
        C = np.exp(-np.linspace(0, 3, n))
        C = np.maximum(C, 0.01)

    # Take square root and IFFT
    A_mag = np.sqrt(C)
    # Make it a valid spectrum: symmetric for real output
    a = irfft(A_mag * np.exp(1j * rng.uniform(0, 2 * np.pi, len(A_mag))), n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def lp_alternating_fast(a, time_budget=30.0):
    """Fast LP alternating for smaller n values."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return a

    n = len(a)
    if n > 400:
        return a

    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()

    # Pre-build matrix structure
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A_mat
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})

        if res.status != 0:
            break

        g = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g))
        if s_g < 1e-10:
            break
        g = g / s_g

        best_t_obj = np.inf
        best_t_val = 0.3
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t

        if best_t_obj < best_obj - 1e-12:
            a_new = (1 - best_t_val) * a + best_t_val * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            a = a_new / s_new
            best_obj = best_t_obj
            best_a = a.copy()
            # Update matrix
            for j in range(n):
                r = min(n, 2 * n - 1 - j)
                A_mat[j:j + r, j] = a[:r]
        else:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """

    with multiple variants, frequency-domain initialization, and multi-n exploration.
```
