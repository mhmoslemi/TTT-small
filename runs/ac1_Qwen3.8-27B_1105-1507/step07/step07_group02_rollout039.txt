import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def subgradient_optimize(a, time_budget=60.0):
    """Subgradient descent on max(conv(a,a)) with sum normalization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1
    patience = 0
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max at k_star: grad[i] = a[k_star - i] for valid indices
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            grad[i] = a[k_star - i]
        
        # Center the gradient to preserve sum
        grad = grad - np.mean(grad)
        
        # Step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr = min(lr * 1.02, 1.0)
        else:
            patience += 1
            lr *= 0.95
            if patience > 100:
                # Restart with perturbation
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                a = best_a * (1 + 0.05 * rng.standard_normal(n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                patience = 0
                lr = 0.1
    
    return best_a

def project_simplex(v, target=1.0):

        return v

        if total_reduce < 1e-15:
            break

        # Scale factor
        alpha = 0.1  # fraction of mass to shift
        reduce_amt = alpha * total_reduce
        total_boost_space = (a[bot_idx]).sum()
        if total_boost_space < 1e-15:
            break

        boost_amt = reduce_amt

        a_new = a.copy()
        # Proportional reduction from top
        frac_reduce = contrib[top_idx] / total_reduce
        a_new[top_idx] -= frac_reduce * reduce_amt
        # Proportional boost to bottom
        frac_boost = a[bot_idx] / total_boost_space
        a_new[bot_idx] += frac_boost * boost_amt

        a_new = project_simplex(a_new, 1.0)
        obj_new = objective(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smoothing_step(a, kernel_width=3, alpha=0.5):
    """Apply a small smoothing (convolution with Gaussian kernel)."""
    n = len(a)
    k = 2 * kernel_width + 1
    x = np.arange(-kernel_width, kernel_width + 1)
    kernel = np.exp(-0.5 * (x / (kernel_width / 2.0)) ** 2)
    kernel /= kernel.sum()
    a_smooth = np.convolve(a, kernel, mode='same')
    a_new = (1 - alpha) * a + alpha * a_smooth
    return project_simplex(a_new, 1.0)
