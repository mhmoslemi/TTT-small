```python
"""
LP-based optimizer for step function coefficient minimization.
Uses iterated LP alternating with adaptive step sizes, multiple n values,
and random restarts to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_end = min(2 * n - 1, j + n)
        A[j:row_end, j] = a[:row_end - j]
    return A


def solve_lp_for_g(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Using scipy linprog with HiGHS
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize: M (i.e., [0, ..., 0, 1])
    # Subject to: A@g - M <= 0 for each k (i.e., A_ub @ x <= b_ub)
    #             sum(g) = 1
    #             g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: (2n-1) x (n+1) matrix
    # A_ub[k, j] = A[k,j] for j < n, A_ub[k, n] = -1
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    # Fallback: cvxpy
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable(nonneg=True)
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(M.value)
    except Exception:
        pass
    
    return None, None

    """

            a = a_new.copy()

        else:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: for each peak position, try reducing it by
    adjusting individual coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find indices that contribute to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        
        # Try reducing high contributors and moving mass to low contributors
        improved = False
        for _ in range(5):
            # Pick a random high contributor
            contributors = np.arange(j_start, j_end)
            if len(contributors) == 0:
                break
            # Weight by a[i] * a[k_star - i]
            weights = a[contributors] * a[k_star - contributors]
            total_w = np.sum(weights)
            if total_w < 1e-15:
                break
            probs = weights / total_w
            i = int(contributors[rng.choice(len(contributors), p=probs)])
            
            # Pick a target to increase (far from peak)
            # Choose indices that don't contribute to the peak
            non_contrib = np.setdiff1d(np.arange(n), contributors)
            if len(non_contrib) == 0:
                non_contrib = np.arange(n)
            j = int(non_contrib[rng.integers(0, len(non_contrib))])
            
            # Try moving a small amount
            for frac in [0.01, 0.05, 0.1]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    break
                a = a / s_new
                obj = compute_obj(a)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    a = np.maximum(a, 0)
                    a = a / float(np.sum(a))
                    break
        
        if not improved and it > 100:
            break
    
    if verbose:
        print(f"    CD done: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
    
    return best_a
```

```python
"""
LP-Alternating Optimizer with Multi-Scale Search for Step Function Coefficient Minimization.
Uses iterated LP (min M s.t. conv(a,g) <= M, sum(g)=1, g>=0) as the core engine,
with adaptive step management, multi-n exploration, and random restarts.
Targets 2*n*max(conv(a,a))/(sum(a))^2 minimization.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_end = min(2 * n - 1, j + n)
        A[j:row_end, j] = a[:row_end - j]
    return A


def solve_lp_for_g(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, M] -> n+1 variables
    # Minimize: c^T x where c = [0,...,0,1]
    # Subject to: A_ub @ x <= b_ub, A_eq @ x == b_eq, bounds
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    # Fallback to cvxpy
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable(nonneg=True)
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            prob.solve(solver=cp.PDLP)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(M.value)
    except Exception:
        pass

    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation with zero-padding handling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.10f}")

    n_parent = len(best_seq)
    a = best_seq.copy()

    # === Phase 1: Quick MWU warmup ===
    print(f"\n=== Phase 1: MWU Warmup (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(40.0, remaining * 0.03)
        a_mwu = mwu_targeted(a, alloc, top_k=10, verbose=False)
        obj_mwu = compute_obj(a_mwu)
        if obj_mwu < best_score:
            best_score = obj_mwu
            best_seq = a_mwu.copy()
            a = a_mwu.copy()
            print(f"  MWU: obj={best_score:.10f} **")
        else:
            print(f"  MWU: obj={obj_mwu:.10f}")

    # === Phase 2: Extended LP Alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP Alternating (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    # Give most of the budget to LP
    lp_budget = min(600.0, remaining * 0.4)
    
    # First round: aggressive LP
    a_opt = lp_alternate(a, lp_budget * 0.6, verbose=True, seed_offset=0)
    obj = compute_obj(a_opt)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        a = a_opt.copy()
        print(f"  LP round 1: obj={best_score:.10f} **")
    else:
        print(f"  LP round 1: obj={obj:.10f}")

    # Second round: continue from best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(lp_budget * 0.4, remaining * 0.2)
        a_opt = lp_alternate(a, alloc, verbose=True, seed_offset=10)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  LP round 2: obj={best_score:.10f} **")
        else:
            print(f"  LP round 2: obj={obj:.10f}")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 4000 and n != n_best]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(50.0, remaining * 0.06)
        a_opt = lp_alternate(a_try, alloc, seed_offset=n_try % 100)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 4: Random Restarts with LP ===
    print(f"\n=== Phase 4: Random Restarts ===")
    n_best = len(best_seq)
    n_restarts = 0
    while deadline - time.monotonic() > 60:
        remaining = deadline - time.monotonic()
        alloc = min(45.0, remaining * 0.08)

        # Different perturbation strategies
        strategy = n_restarts % 4
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.005 * min(n_restarts, 8)
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif strategy == 1:
            # Sparse perturbation (change a few coordinates)
            a_r = best_seq.copy()
            n_perturb = max(5, n_best // 10)
            idx = rng.choice(n_best, size=n_perturb, replace=False)
            a_r[idx] *= np.exp(rng.normal(0, 0.1, n_perturb))
        elif strategy == 2:
            # Smooth then perturb
            a_r = smooth_sequence(best_seq, sigma=2.0)
            a_r = a_r * np.exp(rng.normal(0, 0.03, n_best))
        else:
            # Mix with uniform
            mix = 0.05 + 0.02 * (n_restarts % 3)
            a_r = (1 - mix) * best_seq + mix * np.ones(n_best) / n_best

        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        a_opt = lp_alternate(a_r, alloc, seed_offset=1000 + n_restarts)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart {n_restarts} (strat={strategy}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {n_restarts} (strat={strategy}): obj={obj:.10f}")
        n_restarts += 1

    # === Phase 5: Final Polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Coordinate descent
    if remaining > 15:
        alloc = min(25.0, remaining * 0.3)
        a_cd = coordinate_refine(best_seq, alloc, verbose=True, seed_offset=500)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.10f} **")
        else:
            print(f"  CD: obj={obj_cd:.10f}")

    # One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(30.0, remaining * 0.5)
        a_final = lp_alternate(best_seq, alloc, verbose=True, seed_offset=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_final:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
```
