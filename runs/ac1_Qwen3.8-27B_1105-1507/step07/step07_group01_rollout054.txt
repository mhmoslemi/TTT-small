def build_conv_matrix(a):

    n = len(a)

    # So (A @ g)[k] = sum_i A[k,i] * g[i] = sum_i a[k-i] * g[i] = conv(a,g)[k]
    A = np.zeros((2*n-1, n))

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient of conv[k_star] w.r.t. a[i]:
        # conv[k_star] = sum_i a[i] * a[k_star - i]
        # d/d(a[i]) = a[k_star - i] if 0 <= k_star - i < n, else 0
        # But we also need d/d(a[i]) considering a appears on both sides
        # Actually conv[k] = sum_j a[j]*a[k-j], so d/d(a[i]) = a[k-i] + a[i+k-2*(k-i)]... 
        # No: conv[k] = sum_j a[j]*a[k-j]. d(conv[k])/d(a[i]) = a[k-i] (if valid) + a[i] * d(a[k-i])/d(a[i])...
        # Since a[j] and a[k-j] are both a's entries, d(conv[k])/d(a[i]) = a[k-i] + a[i] * (coefficient from the other side)
        # Wait: conv[k] = sum_j a[j] * a[k-j]. If i = j, term is a[i]*a[k-i]. If i = k-j, term is a[k-i]*a[i].
        # So d(conv[k])/d(a[i]) = a[k-i] + a[i] if i != k-i, or 2*a[k-i] if i = k-i (i.e., k is even and i = k/2).
        # But actually in the sum, each pair (j, k-j) appears once, so:
        # d(conv[k])/d(a[i]) = a[k-i] if 0 <= k-i < n (from the j=i term: a[i]*a[k-i])
        # Plus a[i] if 0 <= i < n AND i = k-j for some j, i.e., j = k-i, which is the same condition.
        # So actually d(conv[k])/d(a[i]) = a[k-i] + a[i] * (1 if the k-i != i case... )
        # Let me be careful: conv[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (where a[m]=0 if m<0 or m>=n)
        # d(conv[k])/d(a[i]) = a[k-i] (from j=i term) + [if k-i is in [0,n-1], the term where k-j=i means j=k-i, giving a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i]]
        # Wait no. Let me just index properly.
        # conv[k] = sum_j a[j] * a[k-j]
        # d/d(a[i]): for each j, d/d(a[i]) of a[j]*a[k-j] = (delta_{j,i})*a[k-j] + a[j]*(delta_{k-j,i})
        # = a[k-i] (if i in range) + a[i] (if k-i in range)
        # But if i = k-i (i.e., i = k/2), these are the same term counted twice? No:
        # The sum is over j. When j=i: contribution is a[i]*a[k-i], derivative = a[k-i].
        # When j=k-i: contribution is a[k-i]*a[i], derivative = a[k-i].
        # These are the SAME term in the sum (j=i and j=k-i give the same product a[i]*a[k-i]).
        # So if i = k-i, the term appears only once (j=i=k/2), and derivative = a[k/2] (not 2*a[k/2]).
        # If i != k-i, then j=i gives a[k-i] and j=k-i gives a[k-i], total = 2*a[k-i]... 
        # Wait no. When j=i: a[j]*a[k-j] = a[i]*a[k-i]. Derivative w.r.t. a[i] = a[k-i].
        # When j=k-i (and k-i != i): a[j]*a[k-j] = a[k-i]*a[i]. Derivative w.r.t. a[i] = a[k-i].
        # So total derivative = a[k-i] + a[k-i] = 2*a[k-i] (when i != k-i and both valid).
        # When i = k-i: only one term, derivative = a[i].
        # Hmm, but that doesn't seem right for the n=2 case. Let me verify.
        # n=2, a=[a0,a1]. conv[1] = a[0]*a[1] + a[1]*a[0] = 2*a0*a1.
        # d/d(a0) = 2*a1, d/d(a1) = 2*a0.
        # Using formula: for k=1, i=0: k-i=1, i!=k-i, so derivative = 2*a[1] = 2*a1. ✓
        # For i=1: k-i=0, i!=k-i, so derivative = 2*a[0] = 2*a0. ✓
        # For k=0, i=0: k-i=0, i==k-i, so derivative = a[0] = a0. 
        # conv[0] = a[0]*a[0] = a0^2. d/d(a0) = 2*a0. But formula gives a0. ✗!

        # I see the issue. Let me redo:
        # conv[k] = sum_j a[j]*a[k-j]
        # For k=0: conv[0] = sum_j a[j]*a[-j] = a[0]*a[0] (only j=0 valid, since a[-j]=0 for j>0)
        # d/d(a[0]) of a[0]*a[0] = 2*a[0]. 
        # But using the formula: d/d(a[i]) = a[k-i] + a[k-i] when i != k-i, a[k-i] when i = k-i.
        # For k=0, i=0: i = k-i = 0, so derivative = a[0]? That gives a0, but correct is 2*a0.

        # The issue is that when j=i and k-j=i (i.e., k=2i), the term a[j]*a[k-j] = a[i]*a[i] = a[i]^2,
        # and its derivative is 2*a[i], not a[i].
        # So the correct formula is:
        # d(conv[k])/d(a[i]) = a[k-i] if 0<=k-i<n and i!=k-i (i.e., i != k/2)
        #                      + a[k-i] if 0<=k-i<n (always adds from the j=i perspective)
        # Wait, I'm confusing myself. Let me just compute it directly:
        # d(conv[k])/d(a[i]) = sum_j [delta_{ji} * a[k-j] + a[j] * delta_{k-j,i}]
        # = a[k-i] (from first term, j=i) + a[k-i] (from second term, k-j=i means j=k-i)
        # = 2*a[k-i] if i != k-i (i.e., 2i != k)
        # = 2*a[i] if i == k-i (i.e., i = k/2, so the two contributions are the same and it's really just 2*a[i])
        # Actually when i = k-i: the first term gives a[k-i] = a[i], the second gives a[j] where j=k-i=i, so a[i]. Total = 2*a[i].
        # So in ALL cases: d(conv[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i < n, else 0.
        # Wait, that's only if both i and k-i are valid indices in [0, n-1].
        # If k-i < 0 or k-i >= n, then a[k-i] = 0, so that contribution vanishes.
        # But the second term: a[j] where j = k-i. If k-i is out of [0,n-1], this term is 0.
        # So: d(conv[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i < n, else 0.
        # Let me verify: n=2, k=0, i=0: 2*a[0] = 2*a0. conv[0]=a0^2, d/d(a0)=2a0. ✓
        # n=2, k=1, i=0: 2*a[1] = 2*a1. conv[1]=2*a0*a1, d/d(a0)=2*a1. ✓
        # n=3, k=0, i=0: 2*a[0]. conv[0]=a0^2, d/d(a0)=2a0. ✓
        # n=3, k=1, i=0: 2*a[1]. conv[1]=a0*a1+a1*a0=2*a0*a1, d/d(a0)=2*a1. ✓
        # n=3, k=2, i=0: 2*a[2]. conv[2]=a0*a2+a1*a1+a2*a0=2*a0*a2+a1^2, d/d(a0)=2*a2. ✓
        # n=3, k=2, i=1: 2*a[1]. d/d(a1)=2*a1. ✓ (from a1^2 term)

        # Great! So the subgradient is simply: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0.
        grad = np.zeros(n)
        valid_start = max(0, k_star - n + 1)
        valid_end = min(n, k_star + 1)
        for i in range(valid_start, valid_end):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad_centered = grad - np.mean(grad)
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        st = float(np.sum(a_new))
        if st < 1e-10:
            break
        a_new = a_new / st
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if lr < 0.03:
                lr *= 1.01
        else:
            stall += 1
            if stall > 20:
                lr *= 0.7
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(budget_s=1500):
    """
    Main search function. Combines fast LP alternating, smooth max gradient,
    peak reduction, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(12345)

    # Load parent if available
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent loaded (n={parent_n}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.8f}")

    a = best_seq.copy()

    # === Phase 1: Fast LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: Fast LP Alternating ===")
    remaining = deadline - time.monotonic()
    phase1_total = min(400.0, remaining * 0.30)

    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(phase1_total / (4 - round_idx), remaining * 0.4)
        use_active = (round_idx % 2 == 1)
        a_new = lp_alternating_fast(a, time_budget=alloc, use_active=use_active)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP Round {round_idx} (active={use_active}): obj={best_score:.8f}")
        else:
            print(f"  LP Round {round_idx} (active={use_active}): obj={obj_new:.8f} (no improvement)")

    # === Phase 2: Smooth Max Gradient Descent ===
    print(f"\n=== Phase 2: Smooth Max GD ===")
    for T_init in [0.0003, 0.0001, 0.00005]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(40.0, remaining * 0.15)
        a_sm = smooth_max_gd_fast(a, time_budget=alloc, T_init=T_init, T_min=0.00002)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            a = a_sm.copy()
            print(f"  SmoothMax (T={T_init}): obj={best_score:.8f}")
        else:
            print(f"  SmoothMax (T={T_init}): obj={obj_sm:.8f}")

    # === Phase 3: Peak Reduction ===
    print(f"\n=== Phase 3: Peak Reduction ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(30.0, remaining * 0.15)
        a_pr = peak_reduction(a, time_budget=alloc)
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            a = a_pr.copy()
            print(f"  PeakReduction: obj={best_score:.8f}")
        else:
            print(f"  PeakReduction: obj={obj_pr:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50),
        n_best,
        min(1400, n_best + 50), min(1400, n_best + 100), min(1400, n_best + 200),
        600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1400]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP
        alloc_lp = min(remaining * 0.12, 25.0)
        a_opt = lp_alternating_fast(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        # Peak reduction polish
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try <= best_score + 0.01:
            a_opt = peak_reduction(a_opt, time_budget=min(10.0, remaining * 0.1))
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Intensive Polish on Best ===
    print(f"\n=== Phase 5: Intensive Polish ===")
    a = best_seq.copy()
    for polish_round in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # LP
        alloc = min(remaining * 0.25, 35.0)
        a_lp = lp_alternating_fast(a, time_budget=alloc, use_active=(polish_round % 2 == 1))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  Polish LP {polish_round}: obj={best_score:.8f}")

        # Smooth max
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sm = smooth_max_gd_fast(a, time_budget=min(15.0, remaining * 0.2),
                                       T_init=0.0001, T_min=0.00002)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                a = a_sm.copy()
                print(f"  Polish SM {polish_round}: obj={best_score:.8f}")

        # Peak reduction
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_pr = peak_reduction(a, time_budget=min(10.0, remaining * 0.2))
            obj_pr = compute_obj(a_pr)
            if obj_pr < best_score:
                best_score = obj_pr
                best_seq = a_pr.copy()
                a = a_pr.copy()
                print(f"  Polish PR {polish_round}: obj={best_score:.8f}")

    # === Phase 6: Random Restarts (diversity) ===
    print(f"\n=== Phase 6: Random Restarts ===")
    n_best = len(best_seq)
    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        n_r = int(np.clip(n_best + rng.integers(-100, 101), 500, 1200))
        mode = r % 4
        if mode == 0:
            a_r = np.ones(n_r) / n_r
        elif mode == 1:
            x = np.linspace(0, 1, n_r)
            a_r = np.cos(np.pi * x) ** 2 + 0.1
            a_r = a_r / np.sum(a_r)
        elif mode == 2:
            a_r = rng.dirichlet(np.ones(n_r) * 0.5)
        else:
            center = n_r // 2
            width = max(1, n_r // 4)
            x = np.arange(n_r)
            a_r = np.exp(-((x - center) ** 2) / (2 * width ** 2))
            a_r = a_r / np.sum(a_r)

        alloc_r = min(remaining * 0.12, 20.0)
        a_opt = lp_alternating_fast(a_r, time_budget=alloc_r)
        obj_r = compute_obj(a_opt)

        remaining = deadline - time.monotonic()
        if remaining > 10 and obj_r < best_score + 0.02:
            a_opt = peak_reduction(a_opt, time_budget=min(8.0, remaining * 0.1))
            obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (n={n_r}, mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}, mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final LP + Peak polishing ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a = best_seq.copy()
        a_lp = lp_alternating_fast(a, time_budget=min(30.0, remaining * 0.4))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_pr = peak_reduction(a, time_budget=min(15.0, remaining * 0.4))
            obj_pr = compute_obj(a_pr)
            if obj_pr < best_score:
                best_score = obj_pr
                best_seq = a_pr.copy()
                print(f"  Final PR: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 8:
            a_sm = smooth_max_gd_fast(a_pr if obj_pr < obj_lp else a,
                                       time_budget=min(10.0, remaining * 0.5),
                                       T_init=0.00005, T_min=0.00001)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Final SM: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
