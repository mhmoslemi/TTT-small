import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # conv(a, e_j) has nonzeros at positions j, j+1, ..., j+n-1
        # (a * e_j)[k] = a[k-j] for j <= k <= j+n-1
        for i in range(n):
            A[j + i, j] = a[i]
    return A


def solve_lp_active_set(a, active_indices, M_init=None):
    """Solve restricted LP using only active constraints.
    min M s.t. A[active] @ g <= M, sum(g)=1, g>=0
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    # Use only active constraints
    A_act = A[active_indices, :]
    m_act = len(active_indices)
    
    # LP: min M
    # s.t. A_act @ g - M <= 0
    #      sum(g) = 1
    #      g >= 0, M >= 0
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((m_act, n + 1))
    A_ub[:, :n] = A_act
    A_ub[:, n] = -1.0  # A_act @ g - M <= 0
    
    b_ub = np.zeros(m_act)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(g) = 1
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        s = float(np.sum(g_val))

def fft_flat_padded(a, n_iters=500, time_budget=60.0, relaxation=0.5):

    if s < 1e-10:
        return a
    a = a / s
    
    # Pad to at least 2n-1 so circular autocorrelation = linear
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    a_padded = np.zeros(L)
    a_padded[:n] = a
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Forward FFT of padded signal
        A = fft(a_padded, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-12)
        phase = A / mag_safe
        
        # Target: flat magnitude = 1/sqrt(L) (so that autocorrelation peak = 1/L)
        target_mag = np.ones(L) / np.sqrt(L)
        
        # Blended target

        # Inverse FFT
        a_padded_new = ifft(A_new).real
        a_new = a_padded_new[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a_padded[:n] = a_new
                a_padded[n:] = 0
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    # Perturb and restart
                    rng = np.random.default_rng(hash((it, n)) % (2**31))
                    a_new2 = best_a * np.exp(rng.normal(0, 0.1, n))

                    s2 = float(np.sum(a_new2))
                    if s2 > 1e-10:

                        a_padded[:n] = a_new2
                        a_padded[n:] = 0
                        best_a = a_new2.copy()
                        best_obj = compute_obj(best_a)
                    stall = 0

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a diverse random initialization."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Smooth via random Fourier features with higher frequencies
        n_freqs = min(20, n // 10)
        freqs = rng.uniform(0.5, max(1, n / 10), n_freqs)
        phases = rng.uniform(0, 2 * np.pi, n_freqs)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # Multi-modal
        n_modes = rng.integers(2, 5)
        centers = rng.integers(0, n, n_modes)
        widths = rng.integers(n // 20, n // 5, n_modes)
        widths = np.maximum(widths, 1)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Random with smoothing
        a = rng.uniform(0, 1, n)
        kernel_size = max(3, n // 100)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        # Inverse of a smooth function (peaks at edges)
        x = np.arange(n) / n
        a = 1.0 / (0.1 + x * (1 - x) * 4)
        a = a / np.sum(a)
    return a

So the theoretical minimum of max_b is 1/(2n-1), giving objective = 2n/(2n-1) ≈ 1 for large n.

But can we achieve this? For non-negative sequences, the autocorrelation at lag 0 is `a[0]^2` and at lag 2n-2 is `a[n-1]^2`. For these to equal the "flat" value, we'd need `a[0] = a[n-1] = 1/sqrt(2n-1)`.

More generally, we'd need the autocorrelation to be perfectly flat, which is a very strong condition. It's related to the existence of "perfect" sequences, which are known not to exist for most lengths.

However, we can get close. The question is how close we can get computationally in 1500 seconds.

The current best is 1.506, which means `max_b = 1.506/(2*800) = 0.000941` (for n=800). The "flat" value would be `1/(2*800-1) = 0.000625`. So we're about 50% above the theoretical minimum.

To get to 1.5027, we need `max_b = 1.5027/(2*800) = 0.000939` (for n=800). That's a tiny improvement.

Or we could use a longer sequence. For n=1500, the target would be `max_b = 1.5027/3000 = 0.000501`. The flat value is `1/2999 = 0.000333`. So the ratio is 0.000501/0.
