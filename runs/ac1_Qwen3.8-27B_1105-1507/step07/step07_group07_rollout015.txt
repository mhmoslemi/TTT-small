def compute_conv(a):

    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def adam_optimize_peak(a_init, time_budget=60.0, top_k=3, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer targeting the top-k peaks of the convolution.
    Works with sum(a) = 1 normalization.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        # Compute convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj = 2.0 * n * max_conv
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Find top-k peaks
        k_idx = np.argsort(conv)[::-1][:top_k]
        
        # Compute gradient of the "smoothed max" at the peak locations
        # For each peak k, the gradient of conv[k] w.r.t. a[i] is a[k-i] (if valid)
        # We use a soft combination of the top peaks
        peak_vals = conv[k_idx]
        # Use temperature-based weighting
        temp = 0.001 + 0.01 * max_conv
        weights = np.exp(peak_vals / temp - np.max(peak_vals / temp))
        weights = weights / weights.sum()
        
        # Combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, k_idx):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w * a[j]
        
        # Also add a small regularization to push values away from zero
        grad -= 0.001 * a  # prevents all mass from concentrating
        
        # Adam update
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        # Occasionally check with evaluate_sequence
        if t % 1000 == 0:
            pass  # Don't call evaluate_sequence too often
    
    return best_a

def frequency_shaping(a_init, time_budget=30.0):

            break
        
        # Compute circular convolution from A^2
        B = A * A
        b = irfft(B, size)

        # Find the peak in the linear part

        peak = float(np.max(b_lin))

def multi_scale_search(a_init, n_values, time_budget_per=30.0):

def structured_init(n, rng):
    """Generate structured initial sequences that exploit asymmetry."""
    mode = rng.integers(0, 8)
    
    if mode == 0:
        # Exponential decay (asymmetric)
        a = np.exp(-np.arange(n) / (n * 0.3))
    elif mode == 1:
        # Reverse exponential decay
        a = np.exp(-np.arange(n)[::-1] / (n * 0.3))
    elif mode == 2:
        # Asymmetric Gaussian
        center = n * 0.3
        a = np.exp(-((np.arange(n) - center)**2) / (n * n * 0.05))
    elif mode == 3:
        # Cosine window (asymmetric)
        x = np.linspace(0, np.pi * 0.7, n)
        a = np.cos(x)**2 + 0.1
    elif mode == 4:
        # Sine power
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)**3 + 0.05
    elif mode == 5:
        # Two-scale structure
        a = np.ones(n)
        a[:n//2] = 1.5
        a[n//2:] = 0.5
        a += 0.1 * rng.normal(0, 1, n)
    elif mode == 6:
        # Power law
        a = (np.arange(n) + 1) ** (-0.5)
    else:
        # Random smooth
        a = rng.uniform(0.5, 1.5, n)
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

    return a_new, float(np.max(compute_conv(a_new)))

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        alloc = min(remaining * 0.25, 25.0)
        a_new, obj = adam_smoothed(a, time_budget=alloc, lr=0.002, temp_init=temp,
                                   temp_decay=0.999, temp_min=temp * 0.2)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
    return best_a, best_obj


def peak_shaving(a_init, time_budget=30.0):
    """Direct peak shaving via exponentiated gradient on the peak."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

        if time.monotonic() - start > time_budget:

        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        obj = 2.0 * n * peak_val

            best_a = a.copy()
