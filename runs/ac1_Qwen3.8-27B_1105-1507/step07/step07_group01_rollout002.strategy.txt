def compute_convolution(a):
    # Computes the full linear convolution of a with itself. Note: 
    #   a is a sequence of length n
    #   convolve(a, a) has 2*n-1 elements
    #
    n = len(a)
    size = n
    while size < 2 * n - 1:
        size *= 2
    # Using FFT but careful to only compute full convolution
    # Warning: Use natural normalization based on the problem, not outside scaling
    result = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        total = 0.0
        # Compute indices beyond k span
        # But only valid indices remain, use to避免 out-of-bound errors
        for i in range(0, n):
            j = k - i
            if 0 <= j < n:
                total += a[i] * a[j]
        result[k] = total
    return result
