    """

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.85
    step_size = 0.01
    x = a.copy()

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        k_indices = np.argsort(conv)[::-1][:top_k]

        # Subgradient: average of active constraint gradients
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            d_start = k + 1 - j_end
            d_end = k - j_start + 1
            grad[j_start:j_end] += 2.0 * a[d_start:d_end][::-1]
        grad /= len(k_indices)

        # Nesterov step
        momentum = beta * momentum - step_size * grad
        new_a = a + momentum
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            new_a = a.copy()
            s_new = 1.0
        new_a = new_a / s_new

        # Nesterov lookahead
        t_k = 2.0 / (it + 2.0)
        x = new_a + (1 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = new_a
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        if it % 200 == 199:
            step_size = max(step_size * 0.9, 0.001)

    return best_a


def generate_initialization(n, method='uniform', rng=None):
    """Generate diverse initial sequences."""
    if rng is None:
        rng = np.random.default_rng(42)

    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif method == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    elif method == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    elif method == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1)) + 0.08 * np.cos(4 * np.pi * np.arange(n) / (n - 1))
    elif method == 'triangular':
        a = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
    elif method == 'asymmetric_ramp':
        a = np.linspace(0.1, 1.0, n)
    elif method == 'asymmetric_decay':
        a = np.exp(-np.linspace(0, 3, n))
    elif method == 'bimodal':
        x = np.arange(n)
        a = np.exp(-((x - n * 0.25) ** 2) / (n * 0.1)) + np.exp(-((x - n * 0.75) ** 2) / (n * 0.1))
    elif method == 'random':
        a = rng.random(n) * 0.5 + 0.1
    elif method == 'random_sparse':
        a = np.zeros(n)
        num_nonzero = max(5, n // 10)
        indices = rng.choice(n, size=num_nonzero, replace=False)
        a[indices] = rng.random(num_nonzero) + 0.1
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_polish(a, time_budget=30.0):
    """
    Coordinate descent polish: try moving mass between pairs to reduce peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from peak contributors to non-contributors
        for _ in range(5):
            if time.monotonic() - start > time_budget:
                break
            # Find i that contributes most to conv[k_star]
            j_start = max(0, k_star - n + 1)
            j_end = min(n, k_star + 1)
            contributions = a[j_start:j_end] * a[k_star + 1 - j_end:k_star - j_start + 1][::-1]
            if len(contributions) == 0:
                break
            i_local = int(np.argmax(contributions))
            i = j_start + i_local

            # Find j to move mass to (prefer positions that don't increase other peaks)
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            for _ in range(20):
                j = int(rng.integers(0, n))
                if j == i:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if delta >= a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            # Small random perturbations
            rng = np.random.default_rng(int(time.monotonic() * 100000) % (2**31))
            for _ in range(10):
                idx1 = int(rng.integers(0, n))
                idx2 = int(rng.integers(0, n))
                if idx1 == idx2:
                    continue
                delta = 0.02 * min(a[idx1], a[idx2])
                if delta < 1e-10:
                    continue
                a[idx1] = max(0, a[idx1] - delta)
                a[idx2] += delta
                s = float(np.sum(a))
                if s > 1e-10:
                    a /= s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 50:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: Hybrid L-BFGS + LP + Subgradient + Coordinate Descent
    with diverse initializations and adaptive length sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    n_parent = len(best_seq) if best_seq is not None else 800

    # === Phase 1: Diverse Initialization + Quick Optimization (200s) ===
    print("\n=== Phase 1: Diverse Initialization ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(200.0, remaining * 0.15)

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # Optimize the parent first with L-BFGS
    alloc = min(phase1_budget * 0.4, 60.0)
    a_lbfgs = lbfgs_optimize(best_seq, time_budget=alloc, temp_init=0.003, verbose=True)
    obj = compute_obj(a_lbfgs)
    if obj < best_score:
        best_score = obj
        best_seq = a_lbfgs.copy()
        print(f"  L-BFGS parent: obj={best_score:.8f} **")

    # Try diverse initializations
    remaining = deadline - time.monotonic()
    n_init = len(best_seq)
    methods = ['cosine', 'hann', 'blackman', 'asymmetric_ramp', 'asymmetric_decay',
               'bimodal', 'triangular']
    per_init_budget = max(5.0, (remaining * 0.5) / len(methods))

    for method in methods:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        a_init = generate_initialization(n_init, method=method, rng=rng)
        # Quick L-BFGS
        a_opt = lbfgs_optimize(a_init, time_budget=min(per_init_budget, 30.0),
                               temp_init=0.01, temp_min=0.002)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Init '{method}': obj={best_score:.8f} **")
        else:
            print(f"  Init '{method}': obj={obj:.8f}")

    # === Phase 2: L-BFGS with Temperature Annealing (400s) ===
    print(f"\n=== Phase 2: L-BFGS Annealing (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    phase2_budget = min(350.0, remaining * 0.3)

    a = best_seq.copy()
    # Multiple rounds with different temperature schedules
    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(phase2_budget / (3 - round_idx), remaining * 0.5)
        temp_start = [0.005, 0.002, 0.001][round_idx]
        temp_end = [0.001, 0.0005, 0.0002][round_idx]
        a_new = lbfgs_optimize(a, time_budget=alloc, temp_init=temp_start,
                               temp_min=temp_end, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  L-BFGS round {round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS round {round_idx}: obj={obj:.8f}")

    # === Phase 3: LP Alternating (350s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    phase3_budget = min(350.0, remaining * 0.25)

    a = best_seq.copy()
    for lp_round in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(phase3_budget / (4 - lp_round), remaining * 0.5)
        a_new = lp_alternating(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP round {lp_round}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {lp_round}: obj={obj:.8f}")

    # === Phase 4: Subgradient + Coordinate Polish (150s) ===
    print(f"\n=== Phase 4: Subgradient + Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_sg = nesterov_subgradient(best_seq, time_budget=min(60.0, remaining * 0.2), top_k=15)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgradient: obj={best_score:.8f} **")
        else:
            print(f"  Subgradient: obj={obj_sg:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd = coordinate_polish(best_seq, time_budget=min(40.0, remaining * 0.25))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coordinate: obj={best_score:.8f} **")
        else:
            print(f"  Coordinate: obj={obj_cd:.8f}")

    # === Phase 5: Multi-n Exploration (200s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_base = len(best_seq)
    n_options = sorted(set([
        max(200, n_base - 200), max(200, n_base - 100),
        n_base,
        n_base + 100, n_base + 200, n_base + 400,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: L-BFGS + LP
        alloc_mwu = min(20.0, remaining * 0.2)
        alloc_lp = min(30.0, remaining * 0.3)

        a_opt = lbfgs_optimize(a_try, time_budget=alloc_mwu, temp_init=0.003, temp_min=0.001)
        a_opt = lp_alternating(a_opt, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final Polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Final L-BFGS with very low temperature
        alloc = min(remaining * 0.3, 60.0)
        a_final = lbfgs_optimize(best_seq, time_budget=alloc, temp_init=0.001, temp_min=0.0002)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Final LP
        alloc = min(remaining * 0.3, 40.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Coordinate polish
        alloc = min(remaining * 0.3, 30.0)
        a_cd = coordinate_polish(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # === Phase 7: Random Restarts (if time remains) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Perturb best solution
            scale = 0.03 + 0.02 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

            alloc = min(remaining * 0.3, 30.0)
            a_opt = lbfgs_optimize(a_r, time_budget=alloc * 0.5, temp_init=0.002, temp_min=0.0005)
            a_opt = lp_alternating(a_opt, time_budget=alloc * 0.5)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
