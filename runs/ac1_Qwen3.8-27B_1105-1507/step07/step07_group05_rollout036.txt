import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def compute_conv_fft(a, N=None):
    """Auto-convolution with precomputed FFT size."""
    n = len(a)
    if N is None:
        N = 1
        while N < 2 * n - 1:
            N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def gradient_conv_weighted(a, w, N=None):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_j a[j] * w[i+j]
    Uses FFT via cross-correlation trick.
    """
    n = len(a)
    if N is None:
        N = 1
        while N < 3 * n - 1:
            N <<= 1
    a_flip = a[::-1]
    faf = rfft(a_flip, N)
    fw = rfft(w, N)
    result = irfft(faf * fw, N)
    return 2.0 * result[n - 1:2 * n - 1]


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_max_and_weights(b, tau):
    """Compute softmax-smoothed max and its subgradient weights."""
    b_max = np.max(b)
    shifted = (b - b_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        return b_max, w
    w = exp_vals / s
    smooth_val = b_max + tau * np.log(s)
    return smooth_val, w


def fista_optimize(a, time_budget=120.0, verbose=False):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) with adaptive
    softmax temperature annealing for minimizing 2n*max(conv(a,a)) subject to
    a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # FFT size for gradient computation
    N_grad = 1
    while N_grad < 3 * n - 1:
        N_grad <<= 1
    N_conv = 1
    while N_conv < 2 * n - 1:
        N_conv <<= 1

    # Parameters
    tau = 0.05  # Initial temperature (softmax smoothing)
    tau_min = 0.001
    tau_max = 0.5
    L = 1.0  # Step size parameter (1/L is the step)
    L_init = 1.0
    t = 1.0  # FISTA momentum parameter
    a_prev = a.copy()

    # Compute initial gradient scale for L estimation
    b0 = compute_conv_fft(a, N_conv)
    b_max0 = np.max(b0)
    # Lipschitz constant estimate: the gradient of 2n*max_b has norm
    # bounded by ~ 2n * 2 * max(a) * (2n-1) roughly. Use adaptive.
    L = 4.0 * n * (2 * np.max(a) + 0.1 / n) / tau
    L = max(L, 1.0)

    stall = 0
    max_stall = 300
    improvement_count = 0

    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break

        # FISTA extrapolation
        if it > 0:
            y = a + ((t - 1.0) / (t + 1.0)) * (a - a_prev)
            y = np.maximum(y, 0.0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                y = a.copy()
            else:
                y = y / s_y
        else:
            y = a.copy()

        # Compute smoothed objective gradient
        b_y = compute_conv_fft(y, N_conv)
        _, w = softmax_max_and_weights(b_y, tau)
        grad = gradient_conv_weighted(y, w, N_grad)

        # Center gradient (invariance to constant shifts on simplex)
        grad = grad - np.mean(grad)

        # Adaptive L: ensure descent
        step = 1.0 / L
        a_new = y - step * grad
        a_new = simplex_project(a_new)

        # Backtracking: check if we made progress
        b_new = compute_conv_fft(a_new, N_conv)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            t = t_new
            stall = 0
            improvement_count += 1

            # Increase L slightly (take bigger steps)
            L = min(L * 1.01, 50.0 * n)

            # Anneal temperature
            if improvement_count % 50 == 0:
                tau = max(tau * 0.95, tau_min)
        else:
            a_prev = a
            # Backtrack: reduce step
            L = L * 1.5
            stall += 1

            if stall > max_stall:

                rng_local = np.random.default_rng(it + 12345)

                t = 1.0

def it_peak_target(a, time_budget=60.0, verbose=False):
    """
    Ibrahim-Turktas multi-peak targeting.
    Identifies dominant peaks in auto-convolution and applies
    targeted subgradient corrections.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_conv = 1
    while N_conv < 2 * n - 1:
        N_conv <<= 1
    N_grad = 1
    while N_grad < 3 * n - 1:
        N_grad <<= 1

    eta = 0.1  # Step size
    n_peaks = 5  # Number of peaks to target

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv_fft(a, N_conv)
        b_max = np.max(b)

        # Find top peaks
        if n_peaks >= len(b):
            peaks = np.argsort(b)[::-1]
        else:
            # Use partition for efficiency
            top_indices = np.argpartition(b, -n_peaks)[-n_peaks:]
            peaks = top_indices[np.argsort(b[top_indices])[::-1]]

        # Weight peaks by their values (exponential weighting)
        peak_vals = b[peaks]
        peak_weights = peak_vals / np.sum(peak_vals)

        # Build weighted function w for gradient
        w = np.zeros(2 * n - 1)
        for p, pw in zip(peaks, peak_weights):
            w[int(p)] = pw

        # Compute gradient
        grad = gradient_conv_weighted(a, w, N_grad)
        grad = grad - np.mean(grad)

        # Update
        a_new = a - eta * grad
        a_new = simplex_project(a_new)

        b_new = compute_conv_fft(a_new, N_conv)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            eta = min(eta * 1.02, 1.0)
            # Increase number of peaks over time
            if it % 1000 == 0 and n_peaks < 20:
                n_peaks += 1
        else:
            eta *= 0.9
            if eta < 1e-8:
                break

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"          IT it={it}: obj={best_obj:.8f}, eta={eta:.5f}, n_peaks={n_peaks}, time={elapsed:.1f}s")

    return best_a


def spectral_flatten(a, time_budget=60.0, verbose=False):
    """
    Flatten the power spectrum to reduce auto-convolution peak.
    Uses iterative spectral shaping: raise P to power alpha < 1 to
    compress dynamic range, then project back to time domain.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use full FFT size for spectral operations
    M = 1
    while M < 2 * n:
        M <<= 1

    alpha = 0.7  # Flattening exponent (1 = no change, 0 = completely flat)
    alpha_min = 0.1
    alpha_max = 0.99

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        # Compute spectrum
        A = fft(a, M)
        P = np.abs(A) ** 2

        # Flatten: compress dynamic range
        P_pos = np.maximum(P, 1e-30)
        P_flat = P_pos ** alpha

        # Preserve DC (sum constraint)
        P_flat[0] = P[0]
        # Preserve conjugate symmetry point if M is even
        if M % 2 == 0:
            P_flat[M // 2] = P[M // 2]

        # New spectrum with flattened amplitude, original phase
        amp_new = np.sqrt(P_flat)
        phase = np.angle(A)
        A_new = amp_new * np.exp(1j * phase)

        # Enforce Hermitian symmetry for real output
        # For real signal of length M: X[k] = conj(X[M-k])
        for k in range(1, M // 2 + 1):
            A_new[k] = 0.5 * (A_new[k] + np.conj(A_new[M - k]))
            A_new[M - k] = np.conj(A_new[k])
        A_new[0] = np.real(A_new[0])
        if M % 2 == 0:
            A_new[M // 2] = np.real(A_new[M // 2])

        # Inverse FFT and project
        a_new = np.real(ifft(A_new, M))[:n]
        a_new = simplex_project(a_new)

        b_new = compute_conv_fft(a_new)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            alpha = max(alpha * 0.995, alpha_min)
        else:
            alpha = min(alpha * 1.002, alpha_max)

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"          Spectral it={it}: obj={best_obj:.8f}, alpha={alpha:.4f}, time={elapsed:.1f}s")

    return best_a


def admm_optimize(a, time_budget=60.0, verbose=False):
    """
    ADMM for minimizing 2n*max(conv(a,a)) s.t. a in simplex.
    Alternates between smooth subproblem and simplex projection.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_conv = 1
    while N_conv < 2 * n - 1:
        N_conv <<= 1
    N_grad = 1
    while N_grad < 3 * n - 1:
        N_grad <<= 1

    rho = 10.0  # ADMM penalty
    tau_admm = 0.02  # Smoothing temperature
    z = a.copy()  # Constraint variable
    u = np.zeros(n)  # Dual variable

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # a-update: minimize 2n*smooth_max(conv(a,a)) + (rho/2)||a - z + u||^2
        # Use a few inner gradient steps
        a_inner = a.copy()
        for inner in range(3):
            b = compute_conv_fft(a_inner, N_conv)
            _, w = softmax_max_and_weights(b, tau_admm)
            grad = gradient_conv_weighted(a_inner, w, N_grad)
            grad = grad - np.mean(grad)
            # Quadratic term gradient
            grad_quad = rho * (a_inner - z + u)
            grad_total = 2.0 * n * grad + grad_quad
            # Lipschitz: ~ 2n * 4n/rho_smoothing + rho
            L_inner = (8.0 * n * n / tau_admm + rho)
            step = 1.0 / L_inner
            a_inner = a_inner - step * grad_total
            a_inner = np.maximum(a_inner, 0.0)
            s_inner = np.sum(a_inner)
            if s_inner > 1e-10:
                a_inner = a_inner / s_inner

        a = a_inner

        # z-update: project onto simplex
        z = simplex_project(a + u)

        # u-update
        u = u + a - z

        # Check objective
        b_check = compute_conv_fft(a, N_conv)
        obj_check = 2.0 * n * float(np.max(b_check))
        if obj_check < best_obj - 1e-14:
            best_obj = obj_check
            best_a = a.copy()

        # Adaptive rho
        if it % 500 == 0:
            rho *= 1.1
            tau_admm = max(tau_admm * 0.95, 0.001)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"          ADMM it={it}: obj={best_obj:.8f}, rho={rho:.2f}, tau={tau_admm:.5f}, time={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=60.0, verbose=False):
    """LP-based alternating refinement."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        if time.monotonic() - start > time_budget - 10:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = simplex_project(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = simplex_project(best_a)
            a = best_a.copy()
            if verbose:
                print(f"          LP it={it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            break

    return best_a


def random_init(n, rng, mode=None):
    """Generate diverse random initializations."""
    if mode is None:
        mode = rng.integers(0, 8)

    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Random
        a = rng.uniform(0, 1, n)
        a = simplex_project(a)
    elif mode == 2:
        # U-shaped (edge-concentrated)
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = simplex_project(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 5, 4 * n // 5]
        a = np.zeros(n)
        for c in centers:
            width = max(n // 8, 3)
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * width ** 2))
        a = simplex_project(a)
    elif mode == 4:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = simplex_project(a)
    elif mode == 5:
        # Optimal-ish: concentrated center with decay
        c = n // 2
        x = np.arange(n) - c
        a = np.exp(-0.5 * (x / (n * 0.15)) ** 2)
        a = simplex_project(a)
    elif mode == 6:
        # Periodic modulation
        x = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x / (n * 0.1))
        a = simplex_project(a)
    else:
        # Optimal Golay-like: alternating blocks
        block = max(n // 50, 2)
        a = np.zeros(n)
        for i in range(0, n, block * 2):
            a[i:i + block] = 1.0
            a[i + block:i + 2 * block] = 0.3
        a = simplex_project(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return simplex_project(a_new)


def combined_refine(a, time_budget=60.0, verbose=False):
    """
    Apply multiple optimization methods in sequence, keeping the best.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # FISTA (main workhorse)
    remaining = start + time_budget - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.45, time_budget * 0.45)
        a_gd = fista_optimize(a, time_budget=alloc, verbose=verbose)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_obj:
            best_obj = obj_gd
            best_a = a_gd.copy()
            a = a_gd.copy()
            if verbose:
                print(f"          Combined FISTA: {best_obj:.8f}")

    # IT peak targeting
    remaining = start + time_budget - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.25, time_budget * 0.25)
        a_it = it_peak_target(a, time_budget=alloc, verbose=verbose)
        obj_it = compute_obj(a_it)
        if obj_it < best_obj:
            best_obj = obj_it
            best_a = a_it.copy()
            a = a_it.copy()
            if verbose:
                print(f"          Combined IT: {best_obj:.8f}")

    # Spectral flattening
    remaining = start + time_budget - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.2, time_budget * 0.2)
        a_sp = spectral_flatten(a, time_budget=alloc, verbose=verbose)
        obj_sp = compute_obj(a_sp)
        if obj_sp < best_obj:
            best_obj = obj_sp
            best_a = a_sp.copy()
            a = a_sp.copy()
            if verbose:
                print(f"          Combined Spectral: {best_obj:.8f}")

    # FISTA again (from improved point)
    remaining = start + time_budget - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, time_budget * 0.2)
        a_gd2 = fista_optimize(a, time_budget=alloc, verbose=verbose)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_obj:
            best_obj = obj_gd2
            best_a = a_gd2.copy()
            if verbose:
                print(f"          Combined FISTA2: {best_obj:.8f}")

    # LP polish
    remaining = start + time_budget - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_lp = lp_refine(best_a, time_budget=alloc, verbose=verbose)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"          Combined LP: {best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA + IT + Spectral + LP + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization with n={n_best}, obj={best_score:.8f}")

    # === Phase 1: Intensive FISTA from parent ===
    print(f"\n=== Phase 1: FISTA (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 300.0)
    a_f1 = fista_optimize(best_seq, time_budget=alloc, verbose=True)
    obj_f1 = compute_obj(a_f1)
    if obj_f1 < best_score:
        best_score = obj_f1
        best_seq = a_f1.copy()
        print(f"  ** FISTA Phase 1: obj={best_score:.8f} **")
    else:
        print(f"  FISTA Phase 1: obj={obj_f1:.8f} (no improvement)")

    # === Phase 2: IT Peak Targeting ===
    print(f"\n=== Phase 2: IT Peak Targeting ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 150.0)
        a_it = it_peak_target(best_seq, time_budget=alloc, verbose=True)
        obj_it = compute_obj(a_it)
        if obj_it < best_score:
            best_score = obj_it
            best_seq = a_it.copy()
            print(f"  ** IT Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  IT Phase 2: obj={obj_it:.8f}")

    # === Phase 3: Spectral Flattening ===
    print(f"\n=== Phase 3: Spectral Flattening ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 100.0)
        a_sp = spectral_flatten(best_seq, time_budget=alloc, verbose=True)
        obj_sp = compute_obj(a_sp)
        if obj_sp < best_score:
            best_score = obj_sp
            best_seq = a_sp.copy()
            print(f"  ** Spectral Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Spectral Phase 3: obj={obj_sp:.8f}")

    # === Phase 4: FISTA Round 2 (from improved point, different tau) ===
    print(f"\n=== Phase 4: FISTA Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 150.0)
        a_f2 = fista_optimize(best_seq, time_budget=alloc, verbose=False)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  ** FISTA Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  FISTA Phase 4: obj={obj_f2:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best, n_best + 100, n_best + 200, n_best + 500,
        n_best - 100,
        1000, 1200, 1600, 2000, 2500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 3000]
    # Prioritize different n values
    n_options = [n for n in n_options if n != n_best][:5]
    n_options = sorted(n_options)

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try}...")
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.25, 120.0)
        a_opt = fista_optimize(a_try, time_budget=alloc * 0.6)
        obj_opt = compute_obj(a_opt)

        # IT refinement
        if deadline - time.monotonic() > 20 and obj_opt < best_score + 0.01:
            alloc2 = min(deadline - time.monotonic() - 10, 30.0)
            a_opt = it_peak_target(a_opt, time_budget=alloc2)
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f} (no improvement)")

    # === Phase 6: Random restarts ===
    print(f"\n=== Phase 6: Random Restarts ===")
    n_curr = len(best_seq)
    remaining = deadline - time.monotonic()
    n_restarts = max(2, int(remaining / 45.0))
    n_restarts = min(n_restarts, 10)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        mode = r % 8
        a_r = random_init(n_curr, rng, mode=mode)
        alloc = min(remaining * 0.3, 40.0)
        a_opt = fista_optimize(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.6, 180.0)
        a_final = combined_refine(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final polish: obj={best_score:.8f} **")
        else:
            print(f"  Final polish: obj={obj_final:.8f}")

    # === Phase 8: Last-chance FISTA ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.7, 60.0)
        a_last = fista_optimize(best_seq, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  ** Last FISTA: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
