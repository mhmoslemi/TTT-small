LP-alternating minimization of 2n·max(a⊛a)/(Σa)² with simulated annealing,
frequency-domain and shape-based restarts, and multi-length exploration.
"""

import numpy as np
import time

from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def _compute_conv(a):
    n = len(a)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = rfft(a, sz)
    return irfft(fa * fa, sz)[:2 * n - 1]


def _compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    an = a / s
    return 2.0 * n * float(np.max(_compute_conv(an)))


def _build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _solve_lp(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    an = a / s
    A = _build_A(an)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            prob = cp.Problem(cp.Minimize(M), [A @ g <= M, cp.sum(g) == 1])
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(np.array(g.value, dtype=float).ravel(), 0.0)
                sv = float(gv.sum())
                if sv > 1e-12:
                    return gv / sv, float(M.value)
        except Exception:
            continue
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.hstack([np.ones((1, n)), [[0.0]]])
        b_eq = np.array([1.0])
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, None)] * (n + 1), method='highs')
        if res.status == 0:
            gv = np.maximum(res.x[:n], 0.0)
            sv = float(gv.sum())
            if sv > 1e-12:

    except Exception:
        pass
