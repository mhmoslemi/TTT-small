```python
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient-based optimization
with log-sum-exp approximation of the max, L-BFGS-B, and multi-strategy search.
Combines FFT-based fast convolution with quasi-Newton optimization for efficiency.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def cross_correlation(a, b):
    """
    Compute cross-correlation: (a ⋆ b)[i] = sum_j a[j] * b[j+i]
    for i = 0, ..., len(a)-1, using FFT.
    b must have length >= len(a) + len(b) - 1 for the linear case,
    but we handle it by zero-padding.
    """
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    A = rfft(a, size)
    B = rfft(b, size)
    # ifft(conj(A) * B) gives circular cross-correlation
    # For zero-padded case, this equals linear cross-correlation
    corr = irfft(np.conj(A) * B, size)
    return corr[:n]


def smooth_obj_and_grad(a, t):
    """
    Compute smooth objective F(a) = 2n * LSE(t, conv(a,a)) / (sum(a))^2
    and its gradient, using FFT for efficiency.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = t * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    p = exp_shifted / sum_exp  # softmax weights over conv positions
    lse = max_c + np.log(sum_exp) / t
    obj = 2.0 * n * lse / (s * s)
    
    # d_lse[i] = (2/t) * sum_j a[j] * p[j+i]
    # The sum sum_j a[j] * p[j+i] is the cross-correlation (a ⋆ p)[i]

    grad = 2.0 * n * (d_lse * s - 2.0 * lse * a) / (s ** 3)
    return obj, grad

    """Optimize using L-BFGS-B with smooth objective."""
    n = len(a0)
    start = time.monotonic()

def subgradient_step(a, lr, t=100.0):
    """One step of projected subgradient/soft-gradient descent on the simplex."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient-based optimization
with log-sum-exp approximation, L-BFGS-B, LP alternating, multiplicative weights,
and multi-start exploration. Uses FFT for fast convolution and cross-correlation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def cross_correlation(a, b):
    """
    Compute (a cross_correlation b)[i] = sum_j a[j] * b[j+i] for i=0..len(a)-1.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    A = rfft(a, size)
    B = rfft(b, size)
    corr = irfft(np.conj(A) * B, size)
    return corr[:n]


def smooth_obj_and_grad(a, t):
    """
    Compute smooth objective F(a) = 2n * LSE(t, conv(a)) / (sum(a))^2
    and its gradient, using FFT for efficiency.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10, np.zeros(n)

    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = t * (conv - max_c)
    # Clip for numerical safety
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    if sum_exp < 1e-300:
        return 1e10, np.zeros(n)
    p = exp_shifted / sum_exp
    lse = max_c + np.log(sum_exp) / t
    obj = 2.0 * n * lse / (s * s)

    # d_lse[i] = (2/t) * sum_j a[j] * p[j+i]
    d_lse = cross_correlation(a, p) * (2.0 / t)

    # dF/da[i] = 2n * (d_lse[i] * s - 2 * lse * a[i]) / s^3
    grad = 2.0 * n * (d_lse * s - 2.0 * lse * a) / (s ** 3)
    return obj, grad


def exact_obj(a):
    """Exact objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def lbfgs_optimize(a0, t, time_budget=60.0, maxiter=300):
    """Optimize using L-BFGS-B with smooth log-sum-exp objective."""
    n = len(a0)
    start = time.monotonic()
    best_x = a0.copy()
    best_f = smooth_obj_and_grad(a0, t)[0]

    def fun_and_grad(x):
        if time.monotonic() - start > time_budget:
            return 1e10, np.zeros(n)
        x = np.maximum(x, 1e-15)
        obj, grad = smooth_obj_and_grad(x, t)
        if np.isnan(obj) or np.isinf(obj):
            return 1e10, np.zeros(n)
        return obj, np.maximum(grad, -1e6) - np.maximum(grad, 1e6) + grad  # clip grad

    # Simpler: just clip gradient magnitude
    def fun_and_grad_safe(x):
        if time.monotonic() - start > time_budget:
            return 1e10, np.zeros(n)
        x = np.maximum(x, 1e-15)
        obj, grad = smooth_obj_and_grad(x, t)
        if np.isnan(obj) or np.isinf(obj):
            return 1e10, np.zeros(n)
        grad = np.clip(grad, -1e4, 1e4)
        if np.any(np.isnan(grad)):
            grad = np.zeros(n)
        return obj, grad

    try:
        result = minimize(
            fun=fun_and_grad_safe,
            x0=a0.copy(),
            jac=True,
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': maxiter * 3,
                     'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
        )
        a_opt = np.maximum(result.x, 0)
        s = float(np.sum(a_opt))
        if s > 1e-10:
            a_opt = a_opt / s
        f_opt = smooth_obj_and_grad(a_opt, t)[0]
        if f_opt < best_f:
            best_x = a_opt
    except Exception:
        pass

    return best_x


def multiplicative_weights_step(a, eta, t=100.0):
    """One step of multiplicative weights on the simplex."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a_norm = a / s

    conv = compute_conv(a_norm)
    max_c = float(np.max(conv))
    shifted = t * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    if sum_exp < 1e-300:
        return a
    p = exp_shifted / sum_exp

    contrib = cross_correlation(a_norm, p)
    a_new = a_norm * np.exp(-eta * contrib)
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
    return a_new


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g (normalized to sum=1) or None.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s
    A = build_conv_matrix(a)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': timeout})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass

    return None


def lp_alternate_step(a, t=1.0):
    """One LP alternating step: solve for g, blend with a."""
    g_lp = solve_lp_minpeak(a)
    if g_lp is None:
        return a
    s_a = float(np.sum(a))
    if s_a < 1e-10:
        return g_lp
    a = a / s_a
    a_new = (1 - t) * a + t * g_lp
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def iterative_refine(a, time_budget=60.0, t_smooth=800):
    """Iterative refinement using multiplicative weights with annealing."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = exact_obj(a)
    start = time.monotonic()
    eta = 0.05
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        if it % 200 == 0 and it > 0:
            eta *= 0.85

        a = multiplicative_weights_step(a, eta, t=t_smooth)
        if it % 100 == 0:
            obj = exact_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 5:
                    a = best_a.copy()
                    eta *= 0.7
                    if eta < 1e-4:
                        break
                    stall = 0

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating, smooth gradient-based
    L-BFGS-B optimization (log-sum-exp surrogate), multiplicative weights,
    and multi-start exploration to minimize the peak auto-convolution objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000)
    except (NameError, Exception):
        parent = np.ones(800)

    n_parent = len(parent)
    s_parent = float(np.sum(parent))
    if s_parent < 0.01:
        parent = np.ones(800)
        n_parent = 800
        s_parent = 800.0

    best_seq = parent.copy()
    best_score = evaluate_sequence([float(x) for x in parent])
    print(f"Parent: n={n_parent}, score={best_score:.8f}")

    parent_norm = parent / s_parent
    t_base = n_parent

    # === Phase 1: LP Alternating ===
    print("\n=== Phase 1: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_lp = parent_norm.copy()
        best_lp_obj = exact_obj(a_lp)
        lp_start = time.monotonic()
        t_blend = 0.5
        for it in range(30):
            if time.monotonic() - deadline > 0 or time.monotonic() - lp_start > min(90, remaining * 0.3):
                break
            a_lp = lp_alternate_step(a_lp, t=t_blend)
            obj_lp = exact_obj(a_lp)
            if obj_lp < best_lp_obj:
                best_lp_obj = obj_lp
            else:
                t_blend *= 0.8
            if it % 5 == 0:
                print(f"  LP it{it}: obj={obj_lp:.8f}")

        score_lp = evaluate_sequence([float(x) for x in a_lp])
        print(f"  LP final: score={score_lp:.8f}")
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"  ** New best: {best_score:.8f} **")

    # === Phase 2: L-BFGS-B with various temperatures ===
    print("\n=== Phase 2: L-BFGS-B Optimization ===")
    a_start = best_seq / np.sum(best_seq)
    t_list = [t_base * 0.5, t_base, t_base * 2, t_base * 5, t_base * 10]

    for t in t_list:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(50.0, remaining * 0.2)
        a_opt = lbfgs_optimize(a_start, t, time_budget=alloc, maxiter=200)
        score = evaluate_sequence([float(x) for x in a_opt])
        print(f"  L-BFGS (t={t:.0f}): score={score:.8f}")
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            a_start = a_opt.copy()
            print(f"  ** New best: {best_score:.8f} **")
        else:
            # Keep the smoother result for chaining
            obj_smooth = smooth_obj_and_grad(a_opt, t)[0]
            obj_start = smooth_obj_and_grad(a_start, t)[0]
            if obj_smooth < obj_start:
                a_start = a_opt.copy()

    # === Phase 3: Iterative multiplicative weights ===
    print("\n=== Phase 3: Iterative Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_iter = best_seq / np.sum(best_seq)
        alloc = min(60.0, remaining * 0.2)
        a_opt = iterative_refine(a_iter, time_budget=alloc, t_smooth=t_base)
        score = evaluate_sequence([float(x) for x in a_opt])
        print(f"  Iterative: score={score:.8f}")
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  ** New best: {best_score:.8f} **")

    # === Phase 4: Chained L-BFGS from current best ===
    print("\n=== Phase 4: Chained L-BFGS ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cur = best_seq / np.sum(best_seq)
        for t in [t_base * 2, t_base * 5, t_base * 10, t_base * 20]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(40.0, remaining * 0.2)
            a_opt = lbfgs_optimize(a_cur, t, time_budget=alloc, maxiter=300)
            score = evaluate_sequence([float(x) for x in a_opt])
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                a_cur = a_opt.copy()
                print(f"  Chained L-BFGS (t={t:.0f}): score={best_score:.8f} **")
            else:
                print(f"  Chained L-BFGS (t={t:.0f}): score={score:.8f}")
                # Still chain if smooth objective improved
                if smooth_obj_and_grad(a_opt, t)[0] < smooth_obj_and_grad(a_cur, t)[0]:
                    a_cur = a_opt.copy()

    # === Phase 5: Length exploration ===
    print("\n=== Phase 5: Length Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 100), max(200, n_best - 200),
            n_best + 100, n_best + 200, n_best + 400,
            400, 500, 600, 700, 800, 900, 1000, 1200, 1400, 1600, 2000
        ]))
        n_options = [n for n in n_options if n != n_best and 100 <= n <= 3000]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            a_try = resample_sequence(best_seq, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try
            alloc = min(25.0, remaining * 0.1)
            a_opt = lbfgs_optimize(a_try, float(n_try), time_budget=alloc, maxiter=150)
            score = evaluate_sequence([float(x) for x in a_opt])
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: score={best_score:.8f} **")
            else:
                print(f"  n={n_try}: score={score:.8f}")

    # === Phase 6: Random perturbations + local opt ===
    print("\n=== Phase 6: Perturbation Exploration ===")
    rng = np.random.default_rng(seed)
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    for r in range(25):
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break

        if r % 5 == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.01, n_best))
        elif r % 5 == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
        elif r % 5 == 2:
            # Frequency perturbation
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1 + 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
            a_r = best_seq * mod
        elif r % 5 == 3:
            # Smooth noise
            noise = rng.normal(0, 0.02, n_best)
            kernel = np.ones(5) / 5
            noise = np.convolve(noise, kernel, mode='same')
            a_r = best_seq * np.exp(noise)
        else:
            # Different length
            n_r = max(200, min(2500, n_best + int(rng.integers(-200, 200))))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.02, len(a_r)))
            n_best_local = len(a_r)
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r
            n_use = len(a_r)
            alloc = min(20.0, remaining * 0.06)
            a_opt = lbfgs_optimize(a_r, float(n_use), time_budget=alloc, maxiter=150)
            score = evaluate_sequence([float(x) for x in a_opt])
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Perturb {r} (n={len(a_opt)}): score={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (n={len(a_opt)}): score={score:.8f}")

    # === Phase 7: Final intensive optimization ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final = best_seq / np.sum(best_seq)
        n_fin = len(a_final)

        # Multiple L-BFGS passes with increasing t
        for t in [n_fin, n_fin * 3, n_fin * 10, n_fin * 30, n_fin * 100]:
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            alloc = min(remaining * 0.25, 30.0)
            a_opt = lbfgs_optimize(a_final, t, time_budget=alloc, maxiter=200)
            score = evaluate_sequence([float(x) for x in a_opt])
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                a_final = a_opt.copy()
                print(f"  Final (t={t:.0f}): score={best_score:.8f} **")
            else:
                print(f"  Final (t={t:.0f}): score={score:.8f}")
                if smooth_obj_and_grad(a_opt, t)[0] < smooth_obj_and_grad(a_final, t)[0]:
                    a_final = a_opt.copy()

        # One more iterative refinement
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_iter = iterative_refine(a_final, time_budget=min(20, remaining * 0.4),
                                      t_smooth=n_fin * 5)
            score = evaluate_sequence([float(x) for x in a_iter])
            if score < best_score:
                best_score = score
                best_seq = a_iter.copy()
                print(f"  Final iterative: score={best_score:.8f} **")

    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf and final_val < best_score:
        best_score = final_val
    elif final_val >= np.inf:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
