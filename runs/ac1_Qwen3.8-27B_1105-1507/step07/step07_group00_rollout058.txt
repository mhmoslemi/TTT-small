"""
Hybrid optimization combining LP alternating (PDLP), smoothed gradient (log-sum-exp with Nesterov),
and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_max_conv(a):
    """Compute max(conv(a,a)) assuming a is normalized (sum=1)."""
    conv = compute_conv(a)
    return float(np.max(conv)), conv

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ p = conv(a, p)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_best_response(a, warm_start=None):

    constraints = [A @ p <= M, cp.sum(p) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_p = None
    best_M = np.inf

"""
Hybrid optimization combining LP alternating (PDLP/MOSEK), log-sum-exp smoothed gradient
with Nesterov acceleration, peak-targeted refinement, and multi-n exploration
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over prior: smoothed gradient with adaptive temperature + Nesterov acceleration,
faster LP via PDLP, and peak-structure-aware restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ p = conv(a, p)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_best_response(a, solver_pref=None):
    """
    Solve LP: min M s.t. conv(a,p)_k <= M for all k, sum(p)=1, p>=0.
    Returns (p_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    p = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ p <= M, cp.sum(p) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_pref is None:
        solvers = [cp.PDLP, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    else:
        solvers = [solver_pref] + [s for s in [cp.PDLP, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS] if s != solver_pref]

    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if p.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                p_val = np.maximum(p.value, 0)
                s = float(np.sum(p_val))
                if s > 1e-10:
                    return p_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_obj_and_grad(a, T):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    f(a) = (1/T) * log(sum_k exp(T * (a*a)_k))
    Returns (f_val, grad) where grad is w.r.t. a (assuming a is normalized).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    if max_c < 1e-15:
        return 0.0, np.zeros(n)
    # Numerically stable log-sum-exp
    exp_vals = np.exp(T * (conv - max_c))
    log_sum = max_c + np.log(exp_vals.sum()) / T
    f_val = float(log_sum)
    # Softmax weights
    w = exp_vals / exp_vals.sum()
    # Gradient: grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m] (w[k]=0 for k>=2n-1)
    # Build the Hankel-like matrix implicitly
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    mask = idx <= 2 * n - 2
    # grad[j] = 2 * dot(w[j:j+valid], a[:valid])
    # Vectorize: H[j,m] = w[j+m] if j+m <= 2n-2 else 0
    H_vals = np.where(mask, w[idx], 0.0)
    grad = 2.0 * (H_vals @ a)
    return f_val, grad


def nesterov_smoothed_optimize(a, time_budget=300.0, T_start=50.0, T_end=500.0, verbose=False):
    """
    Nesterov-accelerated gradient descent on the log-sum-exp smoothed objective.
    Projects onto the simplex after each step. Adaptive temperature annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov state
    y = a.copy()  # Nesterov lookahead point
    beta = 0.9  # momentum
    step = 1.0 / np.sqrt(n)
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** progress

        # Nesterov lookahead
        f_y, grad_y = smoothed_obj_and_grad(y, T)

        # Step
        y_new = y - step * grad_y
        y_new = project_simplex(y_new)

        # Update Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = y_new + ((1.0 - beta) / beta) * (y_new - y)
        a_new = project_simplex(a_new)

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            y = a_new.copy()
            stall = 0
            # Slightly increase step on success
            step = min(step * 1.001, 2.0 / np.sqrt(n))
        else:
            stall += 1
            if stall > 100:
                # Restart with perturbation
                rng = np.random.default_rng(it * 6151 + n)
                a_new = best_a * np.exp(rng.normal(0, 0.05, n))
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                y = a_new.copy()
                step = 0.5 / np.sqrt(n)
                stall = 0
            else:
                step *= 0.95

        it += 1
        if verbose and it % 200 == 0:
            print(f"    NSO it{it}: obj={best_obj:.8f}, T={T:.1f}, step={step:.6f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a

        y_new = nesterov_y - grad_step * grad_y
        y_new = project_simplex(y_new)
        nesterov_beta = (1.0 + np.sqrt(1.0 + 4.0 * nesterov_beta * nesterov_beta)) / 2.0
        a_new = y_new + ((1.0 - nesterov_beta) / nesterov_beta) * (y_new - nesterov_y)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            nesterov_y = a_new.copy()
            grad_step = min(grad_step * 1.002, 2.0 / np.sqrt(n))
            grad_stall = 0
        else:
            grad_stall += 1
            if grad_stall > 50:
                grad_step *= 0.9
                if grad_step < 0.01 / np.sqrt(n):
                    # Perturb
                    rng = np.random.default_rng(it * 4441 + n)
                    a_new = best_a * np.exp(rng.normal(0, 0.05, n))
                    a_new = np.maximum(a_new, 0)
                    s_new = float(np.sum(a_new))
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                    nesterov_y = a_new.copy()
                    best_a = a_new.copy()
                    best_obj = compute_obj(a_new)
                    grad_step = 0.5 / np.sqrt(n)
                    grad_stall = 0

        it += 1
        if verbose and it % 50 == 0:
            print(f"    COMB it{it}: obj={best_obj:.8f}, T={T:.0f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def peak_target_refine(a, time_budget=100.0, verbose=False):
    """
    Target the worst peak directly: find positions contributing most to the max peak
    and redistribute mass away from them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find contributing pairs
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        # Non-contributing positions (don't affect peak k_star)
        contributing_set = set()
        for _, i, j in contribs:
            contributing_set.add(i)
            contributing_set.add(j)
        non_contrib = [i for i in range(n) if i not in contributing_set]

        if not non_contrib:
            break

        improved = False
        for contrib_val, i, j in contribs[:15]:
            if contrib_val < 1e-10:
                break
            for target in non_contrib[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                    delta = frac * min(a[i], a[j]) if i != j else frac * a[i]
                    if a[i] - delta < 0 or (i != j and a[j] - delta < 0):
                        continue
                    a[i] -= delta
                    if i != j:
                        a[j] -= delta
                        a[target] += 2 * delta
                    else:
                        a[target] += delta
                    s_try = float(np.sum(a))
                    if s_try < 1e-10:
                        a[i] += delta
                        continue
                    a = a / s_try
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        if i != j:
                            a[j] += delta
                            a[target] -= 2 * delta
                        else:
                            a[target] -= delta
                        a = a / a.sum()
                        break

        if not improved:
            # Try reducing the peak by spreading mass more evenly among contributors
            rng = np.random.default_rng(it * 7 + n)
            for i, j in [(ci, cj) for _, ci, cj in contribs[:5]]:
                if i == j:
                    continue
                for target in non_contrib[:3]:
                    delta = 0.01 * a[i]
                    if a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    s_try = float(np.sum(a))
                    if s_try > 1e-10:
                        a = a / s_try
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[target] -= delta
                            a = a / a.sum()
            if not improved:
                break

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = x**3 + (1-x)**3 + 0.05
        a = a / a.sum()
    elif mode == 6:
        # Asymmetric: heavy on one side
        x = np.linspace(0, 1, n)
        a = np.exp(-x * 3) + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Two clusters
        a = np.zeros(n)
        a[:n//3] = 1.0
        a[2*n//3:] = 1.0
        a[n//2 - 5:n//2 + 5] = 0.5
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, Nesterov smoothed gradient,
    peak-targeted refinement, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_target = 800
        best_seq = diverse_init(n_target, rng, mode=4)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={best_seq.size}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"Starting optimization with n={n_current}, obj={best_score:.8f}")

    # Phase 1: LP Alternating (main workhorse)
    remaining = deadline - time.monotonic()
    phase1_budget = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 1: LP Alternating (budget={phase1_budget:.0f}s) ===")
    if phase1_budget > 30:
        a_lp = lp_alternating_v2(best_seq, time_budget=phase1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP alternating: obj={best_score:.8f} **")
        else:
            print(f"  LP alternating: obj={obj_lp:.8f} (no improvement)")

    # Phase 2: Nesterov Smoothed Gradient
    remaining = deadline - time.monotonic()
    phase2_budget = min(remaining * 0.2, 250.0)
    print(f"\n=== Phase 2: Nesterov Smoothed Gradient (budget={phase2_budget:.0f}s) ===")
    if phase2_budget > 30:
        a_ns = nesterov_smoothed_optimize(best_seq, time_budget=phase2_budget, T_start=30.0, T_end=300.0, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov smoothed: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov smoothed: obj={obj_ns:.8f}")

    # Phase 3: Combined LP + Gradient
    remaining = deadline - time.monotonic()
    phase3_budget = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 3: Combined LP+Gradient (budget={phase3_budget:.0f}s) ===")
    if phase3_budget > 30:
        a_comb = combined_optimize(best_seq, time_budget=phase3_budget, verbose=True)
        obj_comb = compute_obj(a_comb)
        if obj_comb < best_score:
            best_score = obj_comb
            best_seq = a_comb.copy()
            print(f"  Combined: obj={best_score:.8f} **")
        else:
            print(f"  Combined: obj={obj_comb:.8f}")

    # Phase 4: Peak-targeted refinement
    remaining = deadline - time.monotonic()
    phase4_budget = min(remaining * 0.1, 100.0)
    print(f"\n=== Phase 4: Peak Target Refinement (budget={phase4_budget:.0f}s) ===")
    if phase4_budget > 15:
        a_pt = peak_target_refine(best_seq, time_budget=phase4_budget, verbose=True)
        obj_pt = compute_obj(a_pt)
        if obj_pt < best_score:
            best_score = obj_pt
            best_seq = a_pt.copy()
            print(f"  Peak target: obj={best_score:.8f} **")
        else:
            print(f"  Peak target: obj={obj_pt:.8f}")

    # Phase 5: Multi-n exploration
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Multi-n Exploration (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 50, n_best - 100, n_best - 150,
        n_best + 50, n_best + 100, n_best + 150,
        600, 700, 800, 900, 1000, 1100
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1200]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: LP then gradient
        alloc = min(remaining * 0.25, 100.0)
        half = alloc / 2

        if half > 15:
            a_try = lp_alternating_v2(a_try, time_budget=half)

        remaining = deadline - time.monotonic()
        if remaining > 20:
            grad_alloc = min(remaining * 0.15, 60.0)
            if grad_alloc > 15:
                a_try = nesterov_smoothed_optimize(a_try, time_budget=grad_alloc, T_start=50.0, T_end=200.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 6: Diverse restarts
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Diverse Restarts (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)
    n_restarts = 5

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r * 3) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        restart_budget = min(remaining * 0.25, 80.0)
        if restart_budget > 20:
            a_r = combined_optimize(a_r, time_budget=restart_budget)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # Phase 7: Final intensive polish on best
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 7: Final Polish (remaining={remaining:.0f}s) ===")

    if remaining > 30:
        # LP polish
        lp_alloc = min(remaining * 0.3, 120.0)
        a_final = lp_alternating_v2(best_seq, time_budget=lp_alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Gradient polish with high T
        grad_alloc = min(remaining * 0.3, 80.0)
        a_final = nesterov_smoothed_optimize(best_seq, time_budget=grad_alloc, T_start=200.0, T_end=1000.0, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Gradient: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Peak target final
        pt_alloc = min(remaining * 0.3, 40.0)
        a_final = peak_target_refine(best_seq, time_budget=pt_alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Peak Target: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # One more combined pass
        comb_alloc = min(remaining * 0.5, 60.0)
        a_final = combined_optimize(best_seq, time_budget=comb_alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Combined: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
