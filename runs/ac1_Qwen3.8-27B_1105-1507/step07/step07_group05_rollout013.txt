def solve_lp_minpeak(a, solver=None):
    n = len(a)
    a = a / np.sum(a)
    
    # Build the constraint matrix A where A[k, i] = a[k-i] if 0 <= k-i < n else 0
    # Constraint: A @ g <= M
    # Variables: g[0..n-1], M
    # Objective: minimize M
    
    # Using scipy linprog:
    # c = [0, 0, ..., 0, 1] (n+1 variables, last is M)
    # A_ub @ x <= b_ub
    #   For each k: sum_i a[k-i]*g[i] - M <= 0
    # A_eq @ x = b_eq
    #   sum(g) = 1
    # bounds: g[i] >= 0, M free (or M >= 0)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = a[j]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # M >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None

"""
LP-augmented optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy HiGHS LP solver with alternating method, spectral initialization,
and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) [truncated]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / float(np.sum(a))
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = a[j]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None

def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search for best t
        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 15:
                # Perturb and continue
                rng = np.random.default_rng(it + len(a))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
            if stall > 40:
                break

    return best_a

def fft_gradient_descent(a, time_budget=120.0, mode='energy', verbose=False):
    """FFT-based gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    momentum = 0.8
    v = np.zeros(n)

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    stall = 0
    max_stall = 100

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Gradient of max(b) w.r.t. a
        # Using subgradient: if k* is argmax, grad[i] = a[k*-i] + a[i-k*] (with bounds)
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
        
        # Also add energy term for mode='energy'
        if mode == 'energy':
            # Gradient of sum(b^2) = 2 * conv(a_flip, b) [truncated]
            a_flip = a[::-1]
            faf = rfft(a_flip, N)
            fb = rfft(b, N)
            c_full = irfft(faf * fb, N)[:3 * n - 2]
            grad_energy = c_full[n - 1:2 * n - 1]
            grad = 0.5 * grad + 0.5 * grad_energy

        # Center gradient
        grad = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 3.0)
        else:
            stall += 1
            eta *= 0.95
            v *= 0.5
            if stall > max_stall:
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                stall = 0
                max_stall = min(max_stall + 50, 300)
            if eta < 1e-10:
                break

    return best_a

def multiplicative_weights(a, time_budget=60.0, verbose=False):
    """Multiplicative weights / mirror descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv(a)
        k_star = int(np.argmax(b))
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
        
        grad = grad - np.mean(grad)
        
        # Multiplicative update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * grad
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 1.0)
        else:
            eta *= 0.95
            if eta < 1e-8:
                break

    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, rng):
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 10) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.2
        a = a / np.sum(a)
    else:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.5
        a = a / np.sum(a)
    return a

    Solve LP with optional active set constraint.

    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]

                    active = set(np.where(conv_vals >= M - tol)[0].tolist())

    except Exception:
        pass

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_given(a, A_full, constraint_rows):
    """Solve LP: min M s.t. A[rows,:]@g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    m = len(constraint_rows)
    if m == 0:
        return None, None

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A_full[constraint_rows, :]
    A_ub[:, n] = -1.0

    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def fft_gradient_descent(a, time_budget=120.0, mode='peak', verbose=False):
    """FFT-based gradient descent with subgradient at peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.3
    momentum = 0.75
    v = np.zeros(n)

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    stall = 0
    max_stall = 150

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        max_b = float(np.max(b))

        if mode == 'peak':
            # Subgradient at the peak (and near-peak)
            k_star = int(np.argmax(b))
            # Use multiple nearby peaks for smoother gradient
            b_sorted = np.sort(b)[::-1]
            threshold = b_sorted[min(5, len(b_sorted) - 1)]
            active_peaks = np.where(b >= threshold * 0.99)[0]

            grad = np.zeros(n)
            for k in active_peaks:
                for i in range(n):
                    j1 = k - i
                    j2 = i - k
                    if 0 <= j1 < n:
                        grad[i] += a[j1]
                    if 0 <= j2 < n:
                        grad[i] += a[j2]
            grad /= max(len(active_peaks), 1)
        else:
            # Energy mode: gradient of sum(b^2)
            a_flip = a[::-1]
            faf = rfft(a_flip, N)
            fb = rfft(b, N)
            c_full = irfft(faf * fb, N)[:3 * n - 2]
            grad = c_full[n - 1:2 * n - 1]

        grad = grad - np.mean(grad)

        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.005, 2.0)
        else:
            stall += 1
            eta *= 0.96
            v *= 0.5
            if stall > max_stall:
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.3
                momentum = 0.75
                stall = 0
                max_stall = min(max_stall + 50, 400)
            if eta < 1e-10:
                break

    return best_a


def multiplicative_weights(a, time_budget=60.0, verbose=False):
    """Mirror descent / multiplicative updates on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.05
    stall = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv(a)
        max_b = float(np.max(b))
        k_star = int(np.argmax(b))

        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad = grad - np.mean(grad)

        # Multiplicative update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * grad
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, 0.5)
        else:
            stall += 1
            eta *= 0.97
            if stall > 200:
                rng = np.random.default_rng(it + 777)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
                eta = 0.05
            if eta < 1e-8 or stall > 500:
                break

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 10) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:

    elif mode == 5:

            print(f"  ** Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # Phase 7: Final LP polish
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 80.0)
        a_final = lp_alternating_active(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
