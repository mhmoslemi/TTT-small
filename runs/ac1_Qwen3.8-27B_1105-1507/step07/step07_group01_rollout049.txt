                    return g_val / s, float(M.value)
        except Exception:
            continue

    return _solve_lp_scipy(a)


def _solve_lp_scipy(a):
    """Fallback LP solver using scipy linprog (HiGHS)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp_reduced(a, active_indices, solver_priority=None):
    """Solve LP with only a subset of constraints (active set)."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)
    A = A_full[active_indices, :]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_priority is None:
        solver_priority = ['MOSEK', 'CLARABEL', 'ECOS']

    for solver in solver_priority:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_norm = g_val / s
                    M_val = float(M.value)
                    # Verify against ALL constraints
                    conv_check = A_full @ g_norm
                    true_M = float(np.max(conv_check))
                    return g_norm, true_M
        except Exception:
            continue
    return None, None


def convex_line_search(a, g, n_evals=8):
    """
    Find optimal t in [0,1] for a_new = (1-t)*a + t*g using convexity.
    f(t) = 2n * max_k [conv((1-t)*a + t*g)] is convex in t.
    Returns best t and resulting sequence.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        # conv((1-t)*a + t*g) = (1-t)^2 * conv_aa + 2*t*(1-t) * conv_ag + t^2 * conv_gg
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new


def lp_alternating_improved(a, time_budget=200.0, use_reduced_lp=True, verbose=False):
    """
    LP alternating with convex line search and active-set reduction.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    active_set = None

    max_iters = 500
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Get LP direction
        if use_reduced_lp and active_set is not None and len(active_set) < 2 * n - 1:
            g_lp, M_lp = solve_lp_reduced(a, active_set)
            if g_lp is None:
                g_lp, M_lp = solve_lp_full(a)
        else:
            g_lp, M_lp = solve_lp_full(a)

        if g_lp is None:
            break

        # Update active set based on current convolution
        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        # Active indices: where conv is within 2% of max
        threshold = max_conv * 0.98
        active_set = np.where(conv_cur >= threshold)[0]

        # Convex line search
        t_opt, a_new = convex_line_search(a, g_lp, n_evals=10)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"          LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M_lp={M_lp:.6f}, time={elapsed:.1f}s")
        else:
            stall_count += 1
            if stall_count > 20:
                # Try smaller step
                a_try = (1 - 0.05) * a + 0.05 * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall_count = 5
                    else:
                        # Perturb to escape
                        rng = np.random.default_rng(hash((it, n, stall_count)) % (2**32))
                        scale = 0.005
                        a_pert = best_a * np.exp(rng.normal(0, scale, n))
                        a_pert = np.maximum(a_pert, 0)
                        s_p = float(np.sum(a_pert))
                        if s_p > 1e-10:
                            a = a_pert / s_p
                        active_set = None
                        stall_count = 0
                        if stall_count > 50:
                            break

    return best_a


def coordinate_descent_polish(a, time_budget=100.0, delta=1e-4, verbose=False):
    """
    Coordinate descent: try tiny perturbations of individual coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv_cur = compute_conv(a)
    max_conv = np.max(conv_cur)

    pass_count = 0
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count + hash(n) % 10000)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Try increasing a[i]
            for delta_try in [delta, delta * 0.1, delta * 10]:
                a_try = a.copy()
                a_try[i] += delta_try
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            # Try decreasing a[i]
            for delta_try in [delta, delta * 0.1]:
                if a[i] < delta_try * 0.5:
                    continue
                a_try = a.copy()
                a_try[i] -= delta_try
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if verbose and pass_count % 5 == 0:
            print(f"          CD pass {pass_count}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")

        if not improved and pass_count > 3:
            # Reduce delta
            delta *= 0.5
            if delta < 1e-7:
                break

    return best_a


def smooth_max_gradient_descent(a, time_budget=60.0, T_init=0.0003, T_min=0.00002):
    """Gradient descent using smooth max with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.003
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient: grad[i] = sum_j weights[i+j] * a[j]
        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0 and lr < 0.01:
                lr = lr * 1.03
        else:
            stall += 1
            if stall > 40:
                lr *= 0.7
                if lr < 1e-6:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 30:
            T = max(T_min, T_init * (0.998 ** it))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_equalization(a, time_budget=30.0, verbose=False):
    """
    Try to equalize the top convolution peaks by targeted LP on subsets.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)

        # Find top peaks
        sort_idx = np.argsort(conv)[::-1]
        top_k = min(20, len(sort_idx))
        top_indices = sort_idx[:top_k]

        # Solve LP with only these constraints
        g_lp, M_lp = solve_lp_reduced(a, top_indices)
        if g_lp is None:
            g_lp, M_lp = solve_lp_full(a)
        if g_lp is None:
            break

        # Blend
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a
