```python
        b_eq = np.array([1.0])

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None
```

Given that LP solving for n=800 takes some time (building the 1599x800 matrix and solving), and we can probably do 20-50 LP solves in 1500 seconds, the LP alternation is a good primary method.

But the previous code already did this and got 1.506374. To improve, we need:
1. Better escape from local minima
2. More diverse exploration
3. Perhaps a different optimization landscape

Let me think about what's different about the problem. The LP alternation is a block coordinate descent: fix a, optimize g (the next iterate), then fix g, optimize a, etc. (In practice, it's alternating between "current a" and "LP solution g", with a convex combination.)

The key issue is that this can get stuck in local minima where the LP solution gives no improvement.

**New idea: Use the LP dual variables to construct a better update direction.**

The LP: min M s.t. A@g <= M*1, 1^T*g = 1, g >= 0

The dual (written in standard form):
Let y_k >= 0 be the dual variable for (A@g)_k <= M
Let lambda be the dual variable for 1^T*g = 1
Let z >= 0 be the dual variable for g >= 0

Dual: max -lambda
s.t. A^T*y - lambda*1 + z = 0 (from g >= 0)
     sum(y) - z_M = 0 (from M >= 0, but M is free... actually M >= 0 was a constraint in the primal I set up)

Hmm, let me redo. The primal is:
min M
s.t. A@g - M*1 <= 0 (2n-1 constraints, dual vars y >= 0)
     1^T*g = 1 (1 constraint, dual var lambda, free)
     g >= 0 (n constraints, dual vars... wait, in the standard form for the dual)

Actually, for the dual, the constraints g >= 0 give us:
A^T*y - lambda*1 <= 0 (the reduced cost condition for g)
And M >= 0 (if we add that constraint) gives: sum(y) >= 0 (which is automatic since y >= 0)

So the dual is:
max -lambda
s.t. A^T*y <= lambda*1
     y >= 0

The complementary slackness conditions tell us:
- If g_j > 0, then (A^T*y)_j = lambda (the j-th component of A^T*y equals lambda)
- If y_k > 0, then (A@g)_k = M (the k-th convolution value equals M)

So the dual variables y_k tell us which convolution positions are "tight" (at the peak). This is exactly the same information as the soft-argmax weights!

So the LP solution and its dual give us:
- g*: the optimal next iterate
- y*: the weights on the active constraints

We can use y* to compute a subgradient of the objective at the current point a:
subgrad = 2n * sum_k y*_k * ∇f_k(a) (normalized)

This can be used in a subgradient method with averaging.

1. **Phase 1: LP Alternation with diversity** (main phase)

2. **Phase 2: Subgradient with averaging**
   - Use the LP dual variables as subgradients

```python
"""
Refined LP-alternation + subgradient averaging + multi-n exploration.
Key improvements over previous:
1. Better use of LP dual variables for subgradient direction
2. More aggressive diversity with structured perturbations

4. Nesterov-style acceleration on the subgradient

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        r = min(n, 2*n - 1 - j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2*n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2*n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def softmax_from_theta(theta):
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)

def smooth_lbfgs(a_init, tau_start, tau_end, time_budget, max_iter=200):
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    n_stages = 5
    taus = np.geomspace(tau_start, tau_end, n_stages)
    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        stage_budget = min(remaining * 0.8, 80.0)
        
        def obj_func(th):
            a = softmax_from_theta(th)
            conv = compute_conv(a)
            conv_max = np.max(conv)
            return 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)
        
        def grad_func(th):
            a = softmax_from_theta(th)
            conv = compute_conv(a)
            conv_max = np.max(conv)
            w = tau * np.exp((conv - conv_max) * tau)
            w = w / np.sum(w)
            c = np.zeros(n)
            for j in range(n):
                end = min(2*n - 1, j + n)
                c[j] = np.dot(w[j:end], a[:end-j])
            mean_c = np.dot(c, a)
            return 4.0 * n * a * (c - mean_c)
        
        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-6, 'maxfun': 500})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass
    return best_a, best_obj

def lp_alternating(a, time_budget=60.0, verbose=False, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose and it % 10 == 0:
                print(f"    LP alt it{it}: obj={best_obj:.8f}")
        else:
            if it % 5 == 0:
                noise_scale = 0.03 + 0.02 * (it // 50)
                a = best_a * np.exp(rng.normal(0, min(noise_scale, 0.15), n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def coordinate_descent(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        # Try moving mass from peak contributors to non-peak positions
        peak_indices = np.argsort(conv)[::-1][:max(5, n//20)]
        for k_p in peak_indices[:10]:
            if time.monotonic() - start > time_budget:
                break
            for i in range(0, n, max(1, n // 100)):
                j_candidates = [j for j in range(n) if j != i and j < k_p and k_p - j < n]
                if not j_candidates:
                    continue
                j = j_candidates[0]
                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[i]
                    if delta > a[i] * 0.99:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
        if not improved:
            for _ in range(30):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * a[i] * rng.uniform(0.1, 1.0)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                break
    return best_a

def subgradient_averaging(a_init, time_budget=60.0, rng=None):
    """
    Subgradient method with Nesterov averaging for the non-smooth objective.
    Uses LP dual variables as subgradient weights.
    """
    if rng is None:
        rng = np.random.default_rng(99)
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov accelerated subgradient
    a_curr = a.copy()
    a_prev = a.copy()
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient using LP dual
        g_lp, M_lp = solve_lp_minpeak(a_curr)
        if g_lp is None:
            break
        
        # The subgradient of max(conv) at a_curr is:
        # sum_k y*_k * 2 * shift(a_curr, k) for active k
        # where y*_k are the LP dual variables
        # We approximate this using the LP solution
        # Direction: move towards g_lp
        direction = g_lp - a_curr
        
        # Project direction onto tangent space of simplex
        direction = direction - np.mean(direction)
        
        # Step size (Polyak-type)
        obj_curr = compute_obj(a_curr)
        if best_obj > 1.0:  # avoid division by tiny number
            step = (obj_curr - best_obj) / (np.sum(direction**2) + 1e-10)
            step = min(step, 1.0)
        else:
            step = 0.01
        
        a_new = a_curr + step * direction
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        # Nesterov update
        beta = 2.0 / (it + 2.0)
        a_next = (1 - beta) * a_curr + beta * a_prev  # momentum
        # Actually for subgradient, we use:
        # a_{k+1} = proj(a_curr - step * grad)
        # with averaging: a_avg = (1/(k+1)) * sum a_i
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_new
    
    return best_a
```

```python
def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)

            continue
    try:

        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

def softmax_from_theta(theta):
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)

    return np.maximum(a_new, 0)

            mid_end = 3 * len(fa) // 4
            mask[mid_start:mid_end] = 0.3
            fa = fa * mask
            a_current = np.maximum(irfft(fa, 2 * n_cur)[:n_cur], 0)
            s_c = float(np.sum(a_current))
            if s_c > 1e-10:
                a_current = a_current / s_c
            else:
                a_current = best_seq.copy()
        else:  # two_bump
            n_cur = best_n
            a_current = best_seq.copy()
            # Add a bump at a random position
            center = rng.integers(n_cur // 4, 3 * n_cur // 4)
            width = rng.uniform(0.05, 0.15) * n_cur
            bump = np.exp(-0.5 * ((np.arange(n_cur) - center) / width) ** 2)
            a_current = a_current * 0.8 + 0.2 * bump
            a_current = np.maximum(a_current, 0)
            a_current = a_current / np.sum(a_current)

        alloc = min(remaining * 0.22, 100.0)
        a_opt = lp_alternating(a_current, time_budget=alloc, verbose=(idx == 0), rng=rng)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            best_n = len(best_seq)
            print(f"  [{strategy}] (n={best_n}): obj={best_score:.8f} **")
        else:
            print(f"  [{strategy}] (n={n_cur}): obj={obj_opt:.8f}")

    # Phase 2: Smoothed L-BFGS-B from best
    print(f"\n=== Phase 2: Smoothed L-BFGS-B ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        inits = [
            best_seq.copy(),
            np.maximum(best_seq * np.exp(rng.normal(0, 0.04, best_n)), 0) / np.sum(best_seq),
            np.maximum(best_seq * np.linspace(0.85, 1.15, best_n), 0),
        ]
        for trial, a_init in enumerate(inits):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_init = a_init / np.sum(a_init)
            alloc = min(remaining * 0.3, 70.0)
            a_opt, obj_opt = smooth_lbfgs(a_init, tau_start=15.0, tau_end=600.0,
                                           time_budget=alloc, max_iter=200)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                best_n = len(best_seq)
                print(f"  Smooth {trial}: obj={best_score:.8f} **")
            else:
                print(f"  Smooth {trial}: obj={obj_opt:.8f}")

    # Phase 3: Long LP refinement from best
    print(f"\n=== Phase 3: Long LP Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.35, 120.0)
        a_opt = lp_alternating(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Long LP: obj={best_score:.8f} **")
        else:
            print(f"  Long LP: obj={obj_opt:.8f}")

    # Phase 4: Wider n exploration
    print(f"\n=== Phase 4: Wide n Exploration ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        max(500, best_n - 300), max(600, best_n - 200),
        max(700, best_n - 100), best_n,
        min(1000, best_n + 100), min(1100, best_n + 200),
        min(1300, best_n + 300), min(1500, best_n + 500),
    ]))
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        if n_try == best_n:
            continue
        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / np.sum(a_try)
        alloc = min(remaining * 0.2, 45.0)
        a_opt = lp_alternating(a_try, time_budget=alloc, verbose=False, rng=rng)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            best_n = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 5: Final polish
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        a_cd = coordinate_descent_targeted(best_seq, time_budget=min(remaining * 0.35, 50.0), rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD targeted: obj={best_score:.8f}")
        else:
            print(f"  CD targeted: obj={obj_cd:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_opt = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 50.0), verbose=False, rng=rng)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_opt:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_opt, obj_opt = smooth_lbfgs(best_seq, tau_start=80.0, tau_end=1200.0,
                                       time_budget=min(remaining * 0.5, 40.0), max_iter=150)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final Smooth (high tau): obj={best_score:.8f} **")
        else:
            print(f"  Final Smooth: obj={obj_opt:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_descent_targeted(best_seq, time_budget=min(remaining * 0.6, 25.0), rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
