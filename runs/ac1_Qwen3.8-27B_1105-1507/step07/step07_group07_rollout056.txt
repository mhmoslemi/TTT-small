Optimized solver for minimum-peak non-negative step function sequences.
Uses LP alternating as primary method, with L-BFGS-B smoothed gradient,
projected subgradient, and local search polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time
import math


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def self_convolve(a):
    """Linear self-convolution a*a via FFT."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = self_convolve(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_start = max(0, 0)  # k - j >= 0 => k >= j
        k_end = min(2 * n - 1, j + n)  # k - j < n => k < j + n
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a_norm, method='auto'):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a_norm)
    A = build_conv_matrix(a_norm)
    
    # Try scipy linprog with HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@ g - M <= 0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-12:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:

                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-12:

            except Exception:

    except ImportError:
        pass

                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                scale = min(0.1, 0.01 * (stall // 10))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
    
    return best_a


def smoothed_obj_and_grad(x, tau):
    """Smoothed objective and gradient for a = exp(x)."""
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    
    conv = self_convolve(a)
    
    c_scaled = conv / tau
    c_max = float(np.max(c_scaled))
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = float(np.sum(exp_c))
    w = exp_c / sum_exp
    
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    
    # Gradient computation
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """L-BFGS-B on smoothed objective with adaptive temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Temperature schedule
    taus = [20.0/n, 10.0/n, 5.0/n, 2.0/n, 1.0/n, 0.5/n, 0.2/n, 0.1/n, 0.05/n]
    
    x = np.log(a) - np.log(a).max()
    
    for idx, tau in enumerate(taus):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        iters = max(50, int(remaining / 0.1))  # rough estimate
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g
        
        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-10,
                                    'maxls': 50})
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={best_obj:.10f}")
        except Exception:
            pass
    
    return best_a


def subgradient_optimize(a_init, time_budget=30.0, verbose=False):
    """Projected subgradient method with restarts."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.5 / n
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = self_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max conv at a
        subgrad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                subgrad[i] = 2.0 * a[j]
        
        # Projected gradient step
        a_new = a - eta * subgrad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = max(eta * 0.9995, 0.05 / n)
        else:
            eta *= 0.95
            if eta < 0.005 / n:
                # Restart
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                eta = 0.5 / n
    
    return best_a


def local_search(a_init, time_budget=30.0, verbose=False):
    """Pairwise swap local search."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        
        # Transfer from i to j (and vice versa)
        amounts = [0.001, 0.005, 0.01, 0.02, 0.05]
        improved = False
        for amt in amounts:
            for direction in [1, -1]:
                a_try = a.copy()
                delta = direction * amt
                if a_try[i] + delta < 0 or a_try[j] - delta < 0:
                    continue
                a_try[i] += delta
                a_try[j] -= delta
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if improved:
                break
        
        if not improved and it % 1000 == 0:
            # Occasionally perturb
            a = best_a * (1 + 0.01 * rng.normal(0, 1, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Smooth (low-pass)
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)
    elif mode == 3:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(0.5 * x) ** 2 + 0.1
        a = a / np.sum(a)
    elif mode == 4:
        # Two Gaussians
        c1, c2 = n // 3, 2 * n // 3
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n * n / 10)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n * n / 10))
        a = a / np.sum(a)
    elif mode == 5:
        # Sine-modulated
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 6:
        # Power law
        a = np.arange(n) ** 0.5 + 1
        a = a / np.sum(a)
    else:
        # Exponential decay
        a = np.exp(-0.5 * np.abs(np.arange(n) - n/2) / (n/4))
        a = a / np.sum(a)
    return a
