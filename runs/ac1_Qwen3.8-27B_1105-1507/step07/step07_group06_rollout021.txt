"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_obj_and_peak(a):
    """Compute objective and location of peak."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, 0
    a_norm = a / s
    conv = compute_conv(a_norm)
    k_star = int(np.argmax(conv))
    max_b = float(conv[k_star])
    return 2.0 * n * max_b, k_star


def subgradient_of_peak(a, k_star):
    """Compute d(conv(a,a)[k_star])/d(a[i]) for normalized a (sum=1)."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] += a[j]
        j2 = k_star - i  # same as above for autocorrelation
        # Actually d/d a[i] of sum_j a[j]*a[k-j]:
        # term j=i gives a[k-i], term j=k-i gives a[i]
        # So total = a[k-i] + a[i] if i != k-i, or 2*a[i] if i = k-i
        # But since we sum over j, and both terms contribute:
        # d/da[i] sum_j a[j] a[k-j] = a[k-i] + a[k-i] = 2*a[k-i] if valid
        # Wait no. Let me redo:
        # f = sum_j a[j] * a[k-j]

        # = sum_j [delta(j,i) * a[k-j] + a[j] * delta(k-j, i)]
        # = a[k-i] + a[k-i] = 2*a[k-i] (if 0 <= k-i < n)
    grad2 = np.zeros(n)
    for i in range(n):

def subgradient_descent(a_init, time_budget, verbose=False):
    """Projected subgradient descent with adaptive step and restarts."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - start < time_budget:
        obj, k_star = compute_obj_and_peak(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            step = min(step * 1.02, 0.5)
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = np.maximum(a, 0)
            ss = float(np.sum(a))
            if ss > 1e-10:
                a = a / ss
            no_improve = 0
            step = 0.05
            if verbose:
                print(f"    Restart at obj={best_obj:.8f}")
            continue

        # Get subgradient
        grad = subgradient_of_peak(a, k_star)
        # Project onto simplex tangent: subtract mean
        grad_centered = grad - np.mean(grad)

        # Step
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        ss = float(np.sum(a_new))
        if ss > 1e-10:
            a_new = a_new / ss
            a = a_new
        else:
            step *= 0.5
            no_improve += 10

        if no_improve % 50 == 0 and no_improve > 0:
            step *= 0.95

    return best_a


def build_basis_functions(n, d=30):
    """Build a set of basis functions for low-dim parameterization."""
    x = np.linspace(0, 1, n)
    basis = []
    # Polynomials
    for k in range(8):
        basis.append(x ** k)
    # Exponentials (decay from right)
    for alpha in [0.5, 1, 2, 4, 8, 16]:
        basis.append(np.exp(-alpha * (1 - x)))
    # Exponentials (decay from left)
    for alpha in [0.5, 1, 2, 4, 8]:
        basis.append(np.exp(-alpha * x))
    # Gaussians
    for mu in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.1, 0.2, 0.3]:
            basis.append(np.exp(-((x - mu) ** 2) / (2 * sigma ** 2)))
    # Linear ramps
    for t in [0.3, 0.5, 0.7]:
        basis.append(np.clip((x - t) / 0.3, 0, 1))
        basis.append(np.clip((t - x) / 0.3, 0, 1))
    # Constant
    basis.append(np.ones(n))

    B = np.column_stack(basis[:d])
    return B


def low_dim_eval(coeffs, B, n):
    """Evaluate objective given low-dim coefficients."""
    a = B @ coeffs
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e6
    a = a / s
    return compute_obj(a)

def coordinate_descent_topk(a, time_budget, k_top=50, verbose=False):
    """Coordinate descent focusing on top-k contributors to the peak."""
    n = len(a)

        obj, k_star = compute_obj_and_peak(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Find top contributors to c[k_star]
        contribs = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs[i] = a[j]

        # Get top-k indices
        top_k = np.argsort(contribs)[-k_top:]

        # Try to reduce contributions from top-k
        improved = False
        for i in top_k:
            if time.monotonic() - start > time_budget:
                break
            # Find a target j that doesn't contribute to the peak
            j = k_star - i
            # Choose a random target that is NOT active
            for attempt in range(5):
                target = rng.integers(0, n)
                # Check if target is active (contributes to peak)
                t_active = (k_star - target)
                if 0 <= t_active < n:
                    # Try a different target
                    continue
                # Move mass from i to target
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[target] += delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s > 1e-10:
                            a = a / s
                        break
            if improved:
                break

            a = np.maximum(a, 0)

            obj_new = compute_obj(a)
            if obj_new < best_obj:

                best_a = a.copy()
            else:
                a = best_a.copy()

def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        n_cur = len(best_seq)
        B = build_basis_functions(n_cur, d=25)
        alloc = min(remaining * 0.25, 150.0)
        a_lm, obj_lm = low_dim_optimize(B, n_cur, time_budget=alloc, verbose=True)
        if obj_lm < best_score:
            best_score = obj_lm
            best_seq = a_lm.copy()
            n_parent = len(best_seq)
            print(f"  Low-dim: obj={best_score:.8f} **")
        else:
            print(f"  Low-dim: obj={obj_lm:.8f}")

        # Try with more basis functions
        remaining = deadline - time.monotonic()
        if remaining > 40:
            B2 = build_basis_functions(n_cur, d=40)
            alloc2 = min(remaining * 0.15, 80.0)
            a_lm2, obj_lm2 = low_dim_optimize(B2, n_cur, time_budget=alloc2, verbose=True)
            if obj_lm2 < best_score:
                best_score = obj_lm2
                best_seq = a_lm2.copy()
                n_parent = len(best_seq)
                print(f"  Low-dim (d=40): obj={best_score:.8f} **")
            else:
                print(f"  Low-dim (d=40): obj={obj_lm2:.8f}")

    # ============================================================
    # Phase 3: Smoothed L-BFGS-B refinement
    # ============================================================
    print(f"\n=== Phase 3: Smoothed L-BFGS-B ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.2, 100.0)
        a_smooth, obj_smooth = smooth_lbfgs_refine(best_seq, time_budget=alloc, verbose=True)
        if obj_smooth < best_score:
            best_score = obj_smooth
            best_seq = a_smooth.copy()
            n_parent = len(best_seq)
            print(f"  Smooth: obj={best_score:.8f} **")
        else:
            print(f"  Smooth: obj={obj_smooth:.8f}")

    # ============================================================
    # Phase 4: Multi-n exploration
    # ============================================================
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 100),
        max(400, n_cur - 50),
        n_cur,
        n_cur + 50,
        n_cur + 100,
        n_cur + 200,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick subgradient + low-dim
        alloc_sg = min(remaining * 0.15, 40.0)
        alloc_lm = min(remaining * 0.15, 40.0)

        if alloc_sg > 10:
            a_opt1 = subgradient_descent(a_try, time_budget=alloc_sg)
            obj1 = compute_obj(a_opt1)
        else:
            a_opt1, obj1 = a_try, compute_obj(a_try)

        B_try = build_basis_functions(n_try, d=20)
        if alloc_lm > 10:
            a_opt2, obj2 = low_dim_optimize(B_try, n_try, time_budget=alloc_lm)
        else:
            a_opt2, obj2 = a_try, compute_obj(a_try)

        # Pick the best
        if obj1 <= obj2:
            a_best_n, obj_best_n = a_opt1, obj1
        else:
            a_best_n, obj_best_n = a_opt2, obj2

        if obj_best_n < best_score:
            best_score = obj_best_n
            best_seq = a_best_n.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_best_n:.8f}")

    # ============================================================
    # Phase 5: Coordinate descent polish
    # ============================================================
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 60.0)
        a_cd = coordinate_descent_topk(best_seq, time_budget=alloc, k_top=30, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # ============================================================
    # Phase 6: Random restarts with diverse initializations
    # ============================================================
    print(f"\n=== Phase 6: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        init_type = r % 4
        if init_type == 0:
            # Geometric
            rate = rng.uniform(0.8, 1.2)
            a_r = np.exp(np.linspace(0, rate * np.log(n_best) * 0.3, n_best))
            a_r = a_r / np.sum(a_r)
        elif init_type == 1:
            # Exponential with noise
            base = rng.uniform(0.5, 2.0)
            a_r = np.exp(-base * np.linspace(0, 1, n_best) + rng.normal(0, 0.3, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            # Blend parent with random
            blend = rng.uniform(0.3, 0.7)
            a_rand = np.exp(rng.normal(0, 0.5, n_best))
            a_r = blend * best_seq + (1 - blend) * a_rand / np.sum(a_rand)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            # Two-bump
            c1 = rng.uniform(0.2, 0.4)
            c2 = rng.uniform(0.6, 0.8)
            sigma = rng.uniform(0.1, 0.25)
            x = np.linspace(0, 1, n_best)
            a_r = np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + \
                  np.exp(-((x - c2) ** 2) / (2 * sigma ** 2))
            a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.15, 40.0)
        a_opt = subgradient_descent(a_r, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (type={init_type}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_opt:.8f}")

    # ============================================================
    # Phase 7: Final intensive polish
    # ============================================================
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Another subgradient round
        alloc = min(remaining * 0.25, 50.0)
        a_f = subgradient_descent(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final subgrad: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Coordinate descent
        alloc = min(remaining * 0.2, 30.0)
        a_f = coordinate_descent_topk(best_seq, time_budget=alloc, k_top=20)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Smooth L-BFGS with high tau
        alloc = min(remaining * 0.3, 30.0)
        a_f, obj_f = smooth_lbfgs_refine(best_seq, time_budget=alloc)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final smooth: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
