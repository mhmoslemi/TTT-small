Optimization of non-negative sequences minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses all-active mirror descent with FFT-accelerated subgradients, LP alternating,
and parametric initialization.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def compute_subgrad_fft(a, w, N):
    """
    Compute grad[i] = 2 * sum_k w[k] * a[k-i] for all i.
    Using FFT cross-correlation.
    w: length 2n-1 (convolution weights)
    a: length n
    Returns grad of length n.
    """
    n = len(a)
    # Cross-correlation: (w ⋆ a)[i] = sum_j w[j+i] * a[j]
    # = ifft(conj(fft(w, N)) * fft(a, N))[i]
    W = fft(w, N)
    A = fft(a, N)
    corr = ifft(np.conj(W) * A).real

    # For zero-padded signals with N >= 2n-1+n-1 = 3n-2, this equals
    # sum_j w[j] * a[j-i] for the linear (non-wrapping) case.
    # But I want sum_k w[k] * a[k-i] = sum_j w[j] * a[j-i]
    # So corr[i] = sum_j w[j] * a[j-i] which is what I want!
    # Wait: a[j-i] is non-zero only for 0 <= j-i < n, i.e., i <= j <= i+n-1
    # So corr[i] = sum_{j=i}^{i+n-1} w[j] * a[j-i] (if i <= 2n-2 and j <= 2n-2)
    # This is exactly what I want.
    return 2.0 * corr[:n]

def linearized_lp(a, n_iters=100, time_budget=60.0):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv_aa = compute_conv(a)
        
        # LP: min M s.t. 2*conv(a,δ)_k + conv_aa_k <= M, sum(δ)=0, δ >= -a
        # Variables: δ_0,...,δ_{n-1}, M (n+1 variables)
        # 
        # conv(a,δ)_k = sum_j a[j] * δ[k-j] (linear in δ)
        # 
        # For constraint k: sum_j a[j] * δ[k-j] + conv_aa[k]/2 <= M/2
        # i.e., 2 * sum_j a[j] * δ[k-j] + conv_aa[k] <= M
        #
        # In matrix form: A[δ] <= M - conv_aa where A[δ] = 2*conv(a,δ)
        
        # Build constraint matrix (2n-1) x (n+1):
        # Row k: [2*coefficients of conv(a,δ)_k, -1] <= -conv_aa[k]
        # Wait: 2*conv(a,δ)_k + conv_aa[k] <= M
        # => 2*conv(a,δ)_k - M <= -conv_aa[k]
        
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M
        
        A_ub = np.zeros((2*n-1, n+1))
        b_ub = np.zeros(2*n-1)
        
        for k in range(2*n-1):
            # 2*conv(a,δ)_k - M <= -conv_aa[k]
            # conv(a,δ)_k = sum_j a[j] * δ[k-j] where 0<=j<n and 0<=k-j<n
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A_ub[k, idx] = 2.0 * a[j]
            A_ub[k, n] = -1.0
            b_ub[k] = -conv_aa[k]
        
        # Equality: sum(δ) = 0
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        # Bounds: δ_i >= -a_i, M >= 0
        bounds = [(-a[i], None) for i in range(n)] + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        delta = res.x[:n]
        a_new = a + delta
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # No improvement (due to clipping/renormalization), stop
            break
    
    return best_a
