"""
Spectral shaping + multi-scale optimization for low-autocorrelation non-negative sequences.
Uses iterative spectral flattening for initialization, multi-scale-up approach,
improved gradient methods, LP alternating, and local search to minimize
2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def spectral_shaping(a_init, n_iter=100, alpha=0.5, verbose=False):
    """
    Iteratively flatten the spectrum of a to reduce autocorrelation peak.
    """
    n = len(a_init)
    N = next_pow2(2 * n - 1)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        # Compute spectrum
        X = fft(a, N)
        mag = np.abs(X)
        phase = np.angle(X)
        
        # Power compression: reduce dynamic range of |X|^2
        mag_sq = mag ** 2
        mag_sq_max = np.max(mag_sq)
        if mag_sq_max < 1e-20:
            break
        # Compress: new_mag = mag / mag_max^alpha * (some scaling)
        # We want to reduce peaks in |X|^2
        new_mag_sq = mag_sq / (mag_sq_max ** alpha)
        # But we need to preserve the DC component roughly
        # Actually let's try: new_mag = mag / (mag_max)^alpha
        new_mag = new_mag_sq ** 0.5
        
        # Reconstruct
        a_new = np.real(ifft(new_mag * np.exp(1j * phase), N))[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))

            a = a_new

    return best_a

def kaiser_window(n, beta=5.0):
    from scipy.special import i0
    x = np.linspace(-1, 1, n)
    return i0(beta * np.sqrt(1 - x**2)) / i0(beta)

def subgradient_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Subgradient method with averaging for the peak objective.
    Uses the fact that at the optimal, multiple convolution values are at the max.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Polyak step size
    obj_bound = 1.0  # We know objective >= ~1 (theoretical lower bound)
    t = 1.0  # step size parameter
    
    avg_a = a.copy()
    count = 0
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = np.max(conv)
        F = 2.0 * n * M  # objective value (with sum=1)
        
        # Subgradient: for the active constraints (where conv[k] = M)
        # d/d a_i [conv[k]] = 2 * a[k-i] (for valid i)
        active = np.where(np.abs(conv - M) < 1e-10)[0]
        
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        step = (F - obj_bound) / (g_norm * g_norm + 1e-15)
        step = min(step, t / (it + 1))
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s
        
        # Sliding average
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        
        # Also check the average
        if count % 100 == 0:
            obj_avg = compute_obj(avg_a)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = avg_a.copy()
                a = avg_a.copy()
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, use_cvxpy=True):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    if use_cvxpy:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    
    # Fallback: scipy linprog
    try:
        A = build_conv_matrix(a_norm)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)

    if s < 1e-10:

def local_search(a, time_budget=30.0, verbose=False):
    """
    Coordinate-based local search. Try small perturbations of individual
    coordinates or pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < time_budget:
        # Try random coordinate perturbations
        improved = False
        for _ in range(n):
            i = rng.integers(0, n)
            # Try small change
            for delta in [0.01, -0.01, 0.005, -0.005]:
                a_try = a.copy()
                a_try[i] = max(0, a_try[i] + delta * best_obj / n)
                s = float(np.sum(a_try))
                if s < 1e-10:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        # Try pair swaps
        if not improved:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                for eps in [0.005, 0.01, 0.02]:
                    for sign in [1, -1]:
                        a_try = a.copy()
                        a_try[i] = max(0, a_try[i] + sign * eps * a[i])
                        a_try[j] = max(0, a_try[j] - sign * eps * a[j])
                        s = float(np.sum(a_try))
                        if s < 1e-10:
                            continue
                        a_try = a_try / s
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
        
        if not improved:
            # Reset with small perturbation
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_a

Spectral shaping + multi-scale + gradient + LP alternating + local search
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovations: iterative spectral flattening for initialization/polish,
multi-scale-up strategy, power smoothing gradient, and diverse window starts.

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 20 == 0:
                print(f"    spectral it={it}: obj={best_obj:.8f}")

        alpha = max(alpha * 0.997, 0.01)

    return best_a


def spectral_shaping_v2(a_init, n_iter=100, verbose=False):
    """Alternative spectral shaping: flatten |X|^2 directly via thresholding."""
    n = len(a_init)
    N = next_pow2(2 * n - 1)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(n_iter):
        X = fft(a, N)
        mag_sq = np.abs(X) ** 2
        phase = np.angle(X)

        # Threshold: clip the top of |X|^2
        target = np.mean(mag_sq)
        threshold = target * 3.0
        mag_sq_clipped = np.minimum(mag_sq, threshold)
        # Also boost the bottom slightly
        floor = target * 0.2
        mag_sq_clipped = np.maximum(mag_sq_clipped, floor)

        new_mag = np.sqrt(mag_sq_clipped)
        a_new = np.real(ifft(new_mag * np.exp(1j * phase), N))[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def cosine_window(n):
    x = np.linspace(0, 1, n)
    return 0.5 * (1 - np.cos(2 * np.pi * x))


def gaussian_window(n, sigma_frac=0.3):
    x = np.linspace(-1, 1, n)
    return np.exp(-x ** 2 / (2 * sigma_frac ** 2))


def kaiser_window(n, beta=5.0):
    from scipy.special import i0
    x = np.linspace(-1, 1, n)
    return i0(beta * np.sqrt(np.maximum(1 - x ** 2, 0))) / i0(beta)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    A = build_conv_matrix(a_norm)

    # Try cvxpy first
    try:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    # Fallback: scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternate_one_step(a):
    """One LP alternating step."""
    g_lp, M_lp = solve_lp_minpeak(a)
    if g_lp is None:
        return a, compute_obj(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in np.linspace(0.1, 1.0, 15):
        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def lp_alternate_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating optimization with adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        a_new, obj_new = lp_alternate_one_step(a)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2 ** 31))
                scale = 0.01 + 0.005 * min(stall // 10, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def smoothed_obj(x, tau):
    """Smoothed objective for a = exp(x)."""
    n = len(x)
    n_max = np.max(x)
    x = x - n_max
    a = np.exp(x)
    s = float(np.sum(a))

    conv = compute_conv(a)

    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)

    logsumexp_val = c_max + np.log(sum_exp)
    return logsumexp_val / (s * s)


def smoothed_grad(x, tau):
    """Gradient of smoothed objective w.r.t. x."""
    n = len(x)
    n_max = np.max(x)
    x = x - n_max
    a = np.exp(x)
    s = float(np.sum(a))

    conv = compute_conv(a)

    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)

    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]

    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """Gradient optimization with adaptive temperature schedule."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    taus = [10.0 / n, 5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n, 0.02 / n]

    x = np.log(a) - np.log(a).max()

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break

        res = minimize(smoothed_obj, x, method='L-BFGS-B',
                       jac=lambda xv: smoothed_grad(xv, tau),
                       args=(tau,),
                       options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-11,
                                'maxls': 50})

        if res.success or res.fun < smoothed_obj(x, tau):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.4f}: obj={best_obj:.8f}")

    return best_a


def gradient_optimize_v2(a_init, time_budget=60.0, verbose=False):
    """
    Gradient optimization using power smoothing (Huber-like).
    Minimizes sum((conv[i]-M)^p) + M for p>1 via outer loop on M.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use log-sum-exp with very fine temperature + direct gradient
    tau = 0.05 / n
    x = np.log(a) - np.log(a).max()

    for it in range(5):
        if time.monotonic() - start > time_budget:
            break

        res = minimize(smoothed_obj, x, method='L-BFGS-B',
                       jac=lambda xv: smoothed_grad(xv, tau),
                       args=(tau,),
                       options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 100})

        x = res.x - res.x.max()
        a_new = np.exp(x)
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        tau *= 0.5

    return best_a


def local_search(a, time_budget=30.0):
    """Coordinate and pair local search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        improved = False
        n_trials = min(n, 200)

        for _ in range(n_trials):
            i = rng.integers(0, n)
            base_scale = best_obj / n
            for delta_frac in [1e-3, -1e-3, 5e-4, -5e-4, 2e-3, -2e-3]:
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] + delta_frac * base_scale)
                s = float(np.sum(a_try))
                if s < 1e-10:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        # Pair moves
        if not improved:
            for _ in range(100):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                for eps in [1e-3, 2e-3, 5e-3]:
                    a_try = a.copy()
                    move = eps * min(a[i], a[j])
                    a_try[i] = max(0.0, a_try[i] - move)
                    a_try[j] = a_try[j] + move
                    s = float(np.sum(a_try))
                    if s < 1e-10:
                        continue
                    a_try = a_try / s
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            a = best_a * np.exp(rng.normal(0, 0.005, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    return best_a


def multi_scale_optimize(n_start, n_end, time_budget=60.0, verbose=False):
    """Start at small n, optimize, upscale, repeat."""
    start = time.monotonic()
    n = n_start

    # Initial: cosine window with noise
    rng = np.random.default_rng(123)
    a = cosine_window(n) + 0.1 * rng.uniform(0, 1, n)
    a = np.maximum(a, 0)
    a = a / np.sum(a)

    # Spectral shape at small n
    a = spectral_shaping(a, n_iter=60, alpha=0.4)

    if verbose:
        print(f"  Multi-scale n={n}: obj={compute_obj(a):.8f}")

    while n < n_end:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        new_n = min(n * 2, n_end)
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a = np.interp(x_new, x_old, a)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
        n = new_n

        # Spectral shape
        alloc = max(1.0, (time_budget - (time.monotonic() - start)) * 0.3)
        a = spectral_shaping(a, n_iter=40, alpha=0.3)

        if verbose:
            print(f"  Multi-scale n={n}: obj={compute_obj(a):.8f}")

    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):

    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_score = np.inf

    try:

        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))

                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)

    # === Phase 1: Spectral shaping on parent ===
    print(f"\n=== Phase 1: Spectral Shaping (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for alpha in [0.3, 0.5, 0.8]:

            if remaining < 10:
                break
            a_spec = spectral_shaping(best_seq, n_iter=100, alpha=alpha, verbose=True)

            if obj < best_score:
                best_score = obj
                best_seq = a_spec.copy()

            else:
                print(f"  Spectral (alpha={alpha}): obj={obj:.10f}")

        # Try v2
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_spec2 = spectral_shaping_v2(best_seq, n_iter=80)
            obj = compute_obj(a_spec2)
            if obj < best_score:
                best_score = obj
                best_seq = a_spec2.copy()

            else:
                print(f"  Spectral v2: obj={obj:.10f}")

    # === Phase 2: Gradient optimization ===
    print(f"\n=== Phase 2: Gradient Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_grad = gradient_optimize(best_seq, time_budget=min(remaining * 0.2, 100.0), verbose=True)

        if obj < best_score:

            print(f"  Gradient: obj={obj:.10f}")

    # === Phase 3: LP Alternating ===
    print(f"\n=== Phase 3: LP Alternating ===")

    if remaining > 40:
        a_lp = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.15, 80.0), verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:

            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.10f} **")
        else:
            print(f"  LP: obj={obj:.10f}")

    # === Phase 4: Multi-scale from scratch ===
    print(f"\n=== Phase 4: Multi-Scale from Scratch ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_target = min(n_parent + 100, 1200)
        a_ms = multi_scale_optimize(50, n_target, time_budget=min(remaining * 0.1, 40.0), verbose=True)
        # Polish it
        a_ms = gradient_optimize(a_ms, time_budget=min(remaining * 0.05, 20.0))
        obj = compute_obj(a_ms)
        if obj < best_score:
            best_score = obj
            best_seq = a_ms.copy()

            print(f"  Multi-scale: obj={obj:.10f}")

        a_f = gradient_optimize_v2(best_seq, time_budget=min(remaining * 0.3, 50.0), verbose=True)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final grad v2: obj={best_score:.10f} **")
        else:
            print(f"  Final grad v2: obj={obj:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.3, 40.0))
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj:.10f}")

    # Final spectral
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_s = spectral_shaping(best_seq, n_iter=60, alpha=0.15)
        obj = compute_obj(a_s)
        if obj < best_score:
            best_score = obj
            best_seq = a_s.copy()
            print(f"  Final spectral: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
