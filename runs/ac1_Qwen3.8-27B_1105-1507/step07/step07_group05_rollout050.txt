from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_best_response(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g (normalized) or None.
    Uses the dual for efficiency when possible.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s
    
    # Primal: min M s.t. A@g <= M*1, sum(g)=1, g>=0
    # Variables: [g_0, ..., g_{n-1}, M] (n+1 variables)
    # Constraints: A@g - M*1 <= 0 (2n-1 constraints), sum(g) = 1 (1 constraint)
    
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_conv_diff(a, b):
    """Compute conv(a, b) via FFT. a length n, b length m. Returns length n+m-1."""
    n = len(a)
    m = len(b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a  # Wait, this is wrong for the tail
    # Actually: (A @ g)[k] = sum_j A[k,j] * g[j] = sum_j a[k-j] * g[j] (valid)
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # For fixed j, A[k,j] = a[k-j] for k = j, j+1, ..., j+n-1
    # So column j has a[0], a[1], ..., a[n-1] starting at row j
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_best_response(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g (normalized to sum=1) or None.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s
    
    A = build_conv_matrix(a)
    
    # Primal: min M
    # s.t. A@g - M <= 0  (2n-1 constraints)
    #      sum(g) = 1    (1 constraint)
    #      g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None


def lp_best_response_iteration(a, time_budget=60.0, verbose=False):
    """
    Best-response iteration: alternate between LP for g given a, and LP for a given g.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iteration = 0
    stall = 0
    
    # We alternate: given current "fixed" sequence, solve LP for the other
    # But actually, the best response is: given a, find g that minimizes max(conv(a,g))
    # Then set a = g (or blend)
    # This is one-sided. Let's do:
    # 1. Given a, solve LP to get g*
    # 2. Set a = blend(a, g*)
    # 3. Repeat
    
    while time.monotonic() - start < time_budget:
        iteration += 1
        g_star = solve_lp_best_response(a)
        if g_star is None:
            break
        
        # Line search for best blend
        best_t_obj = np.inf
        best_t = 1.0
        for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_try = (1 - t) * a + t * g_star
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a = (1 - best_t) * a + best_t * g_star
            a = project_simplex(a)
            best_a = a.copy()
            stall = 0
            if verbose and iteration % 5 == 0:
                print(f"            LP-BR it{iteration}: obj={best_obj:.8f}, t={best_t:.1f}")
        else:
            stall += 1
            if stall > 30:
                break
    
    return best_a


def variance_gradient_descent(a, time_budget=60.0, verbose=False):
    """
    Minimize variance of conv(a,a) via gradient descent.
    This pushes the convolution toward being flat.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    
    eta = 0.5
    momentum = 0.5
    v = np.zeros(n)
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute C = conv(a,a)
        fa = rfft(a, N)
        C = irfft(fa * fa, N)[:2 * n - 1]
        
        # Compute gradient of sum_k (C[k] - mean(C))^2 w.r.t. a
        # = 2 * sum_k (C[k] - mean) * dC[k]/da
        # dC[k]/da[m] = a[k-m] + a[m]*delta (valid)
        # Dropping constant: grad[m] = 2 * sum_k (C[k] - mean) * a[k-m]
        # = 2 * conv(a, (C-mean)_flipped)[appropriate slice]
        
        d = C - np.mean(C)
        # grad[m] = 2 * sum_k d[k] * a[k-m] (valid k-m in [0,n-1])
        # = 2 * sum_j a[j] * d[j+m] (substituting k=j+m)
        # This is the cross-correlation of a and d
        
        # Compute via FFT:
        # h[m] = sum_j a[j]*d[j+m]
        # = ifft(fft(a) * conj(fft(d)))[n-1+m]... 
        # Actually, let's use the direct convolution approach
        # h[m] = sum_j a[j]*d[j+m] for m=0,...,n-1
        # Consider a of length n, d of length 2n-1
        # h[m] = (a * d_rev)[n-1+m] where d_rev is... 
        # Let me just use: h = ifft(fft(a, N) * fft(d, N))[n-1 : 2n-1] reversed?
        # 
        # conv(a, d)[k] = sum_i a[i]*d[k-i] for k=0,...,3n-3
        # h[m] = sum_i a[i]*d[i+m] = sum_i a[i]*d[(i+m)]
        # conv(a, d)[k] with k = i+m... but i varies
        # Let's define: h[m] = sum_i a[i]*d[i+m]
        # = sum_i a[i]*d[i+m]
        # If we compute conv(a, d)[k] = sum_i a[i]*d[k-i], then:
        # h[m] = sum_i a[i]*d[i+m] = conv(a, d)[2m]... no that's not right either
        # 
        # Let me just flip a: a_flip[i] = a[n-1-i]
        # conv(a_flip, d)[k] = sum_i a_flip[i]*d[k-i] = sum_i a[n-1-i]*d[k-i]
        # Let j = n-1-i: = sum_j a[j]*d[k-n+1+j]
        # For k = n-1+m: = sum_j a[j]*d[m+j] = h[m] ✓
        # So h[m] = conv(a_flip, d)[n-1+m] for m=0,...,n-1
        
        a_flip = a[::-1]
        fa_flip = rfft(a_flip, N)
        fd = rfft(d, N)
        cross = irfft(fa_flip * fd, N)
        h = cross[n-1:n-1+n]  # h[m] for m=0,...,n-1
        
        grad = 2.0 * h
        
        # Skip the constant component (it doesn't affect the projection)
        # Actually, the "a[m]*delta" term in dC[k]/da[m] gives a[m] * sum_k d[k]*delta(valid)
        # which is a[m] * (sum of d[k] for k in [m, m+n-1]) - this is NOT constant
        # But for large n and smooth d, it's approximately constant, so we ignore it.
        
        grad = grad - np.mean(grad)
        
        # Momentum + projection
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        fa_new = rfft(a_new, N)
        C_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        # Use both variance and max as objective
        var_new = np.var(C_new)
        max_new = np.max(C_new)
        obj_new = 2.0 * n * max_new  # Primary objective
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 5.0)
        else:
            eta *= 0.97
            v *= 0.5
            if eta < 1e-10:
                break
        
        if verbose and it % 200 == 0:
            print(f"            VarGD it{it}: obj={best_obj:.8f}, var={var_new:.2e}, max={max_new:.6f}")
    
    return best_a


def subgradient_max_descent(a, time_budget=30.0, verbose=False):
    """
    Subgradient descent on the max of the convolution, with softmax weighting.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    
    eta_avg = 0.1
    temp = 0.01  # softmax temperature (relative to max)
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        C = irfft(fa * fa, N)[:2 * n - 1]
        C_max = np.max(C)
        
        if C_max < 1e-15:
            break
        
        # Softmax weights
        b_shifted = C - C_max
        w = np.exp(b_shifted / (temp * C_max + 1e-15))
        w_sum = np.sum(w)
        w = w / w_sum
        
        # Gradient: grad[m] = sum_k w[k] * a[k-m] (valid)
        # = conv(a_flip, w)[n-1+m]... same as before
        a_flip = a[::-1]
        fa_flip = rfft(a_flip, N)
        fw = rfft(w, N)
        cross = irfft(fa_flip * fw, N)
        grad = cross[n-1:n-1+n]
        
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        eta = eta_avg / (gnorm * np.linalg.norm(a) + 1e-10)
        
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
        
        fa_new = rfft(a_new, N)
        C_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(C_new))
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta_avg = min(eta_avg * 1.02, 2.0)
            temp = max(temp * 0.999, 0.001)  # Decrease temperature (more peaked)
        else:
            eta_avg *= 0.97
            temp = min(temp * 1.01, 0.5)  # Increase temperature (smoother)
            if eta_avg < 1e-12:
                break
        
        if verbose and it % 500 == 0:
            print(f"            SubGrad it{it}: obj={best_obj:.8f}, temp={temp:.4f}")
    
    return best_a


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped (edge-concentrated)
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 5, 4 * n // 5]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    elif mode == 6:
        # Random Fourier: a = |fft(random)|^2, projected to non-negative
        freq = rng.standard_normal(n // 2 + 1) + 1j * rng.standard_normal(n // 2 + 1)
        a = np.zeros(n)
        # This is complex, let's just use magnitude
        a = np.abs(rfft(freq, n)) ** 2
        a = np.maximum(a, 0)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    else:
        # Triangular (peak in middle)
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_sequence(a, time_budget=60.0, verbose=False):
    """
    Combined optimization: LP best-response + variance GD + subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Phase 1: LP best-response (most powerful)
    remaining = start + time_budget - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_lp = lp_best_response_iteration(a, time_budget=alloc, verbose=verbose)
        obj = compute_obj(a_lp)
        if obj < best_obj:
            best_obj = obj
            best_a = a_lp.copy()
            a = a_lp.copy()
            if verbose:
                print(f"              LP: obj={best_obj:.8f}")
    
    # Phase 2: Variance GD
    remaining = start + time_budget - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_vg = variance_gradient_descent(a, time_budget=alloc, verbose=verbose)
        obj = compute_obj(a_vg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_vg.copy()
            a = a_vg.copy()
            if verbose:
