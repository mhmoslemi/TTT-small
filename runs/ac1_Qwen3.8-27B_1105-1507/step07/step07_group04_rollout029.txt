def compute_conv(a):
    """FFT-based convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)

        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

2. LP alternating with warm-starting and golden-section line search
3. Pair-move local search with exact 1D optimization
4. Multi-n exploration and structured initializations

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import cvxpy as cp


def compute_conv(a):
    """FFT-based autocorrelation (full convolution of a with itself)."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2 * n - 1]
    return result


def compute_obj(a):
    """Compute objective 2*n*max(conv)/(sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    return a / s


def compute_grad_lse(a, T, conv=None):
    """
    Compute gradient of 2n * T * log(sum exp(conv_k / T)) w.r.t. a.
    Uses FFT for efficiency. Returns gradient scaled by 4n.
    """
    n = len(a)
    if conv is None:
        conv = compute_conv(a)

    # Log-sum-exp probabilities
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    exp_s = np.exp(shifted)
    Z = float(np.sum(exp_s))
    p = exp_s / Z  # length 2n-1

    # grad[j] = 4n * sum_m a[m] * p[m+j] for m=0..n-1
    # This is a cross-correlation, computed via FFT:
    # c[j] = (a * p_rev)[2n-2-j] where p_rev = p[::-1]
    p_rev = p[::-1]  # length 2n-1

    size = 1
    while size < n + (2 * n - 1) - 1:
        size <<= 1

    a_pad = np.zeros(size)
    a_pad[:n] = a
    p_pad = np.zeros(size)
    p_pad[:2 * n - 1] = p_rev

    fa = rfft(a_pad, size)
    fp = rfft(p_pad, size)
    cross = irfft(fa * fp, size)

    # grad[j] = cross[2n-2-j] for j=0..n-1
    grad = cross[2 * n - 2 : n - 2 : -1]
    return 4.0 * n * grad

def build_conv_matrix_sparse(a):
    """Build the convolution matrix A (2n-1 x n) where A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[k_start - j : k_end - j]
    return A


def solve_lp_minpeak(a, timeout=30.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_sparse(a)

    # Try cvxpy solvers
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]
    start = time.monotonic()

    for solver in solvers:
        if time.monotonic() - start > timeout:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=300.0, verbose=False):
    """
    LP alternating: alternate between fixing a (solve LP for g) and updating a.
    Uses golden-section line search for the mixing parameter.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a, timeout=min(remaining * 0.5, 20.0))

        if g_lp is None:
            # Perturb and retry
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = normalize(a)
            stall += 1
            if stall > 20:
                break
            continue

        # Golden section line search for t in [0, 1]
        # f(t) = obj((1-t)*a + t*g_lp)
        phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
        a_lo, a_hi = 0.0, 1.0
        t1 = a_hi - phi * (a_hi - a_lo)
        t2 = a_lo + phi * (a_hi - a_lo)

        def eval_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        f1 = eval_t(t1)
        f2 = eval_t(t2)

        for _ in range(25):
            if time.monotonic() - start > time_budget - 2:
                break
            if f1 < f2:
                a_hi = t2
                t2 = t1
                f2 = f1
                t1 = a_hi - phi * (a_hi - a_lo)
                f1 = eval_t(t1)
            else:
                a_lo = t1
                t1 = t2
                f1 = f2
                t2 = a_lo + phi * (a_hi - a_lo)
                f2 = eval_t(t2)

        t_opt = (a_lo + a_hi) / 2.0
        a_try = (1.0 - t_opt) * a + t_opt * g_lp
        a_try = np.maximum(a_try, 0.0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={t_opt:.4f}, M_lp={M_lp:.6f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 15, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = normalize(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

    return best_a


def pair_move_search(a, time_budget=200.0, verbose=False):
    """
    Local search via pair moves: for pair (i,j), find optimal mass transfer delta.
    Uses the exact quadratic form of the convolution change.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    start = time.monotonic()

    rng = np.random.default_rng(12345)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        # Try a batch of random pairs
        batch_size = min(50, n * 2)
        improved = False

        for _ in range(batch_size):
            if time.monotonic() - start > time_budget:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-10 or a[j] < 1e-10:
                continue

            # Precompute linear and quadratic terms for this pair
            # new_conv[k] = conv[k] + 2*delta*(a[k-i] - a[k-j]) + delta^2*(d_{k,2i} - 2*d_{k,i+j} + d_{k,2j})
            # where a[k-i] means a at position k-i if 0<=k-i<n else 0

            L = np.zeros(2 * n - 1)
            # a[k-i]: valid when 0 <= k-i < n, i.e., i <= k < i+n
            k_lo = i
            k_hi = min(2 * n - 1, i + n)
            L[k_lo:k_hi] += a[:k_hi - k_lo]
            # -a[k-j]: valid when 0 <= k-j < n, i.e., j <= k < j+n
            k_lo2 = j
            k_hi2 = min(2 * n - 1, j + n)
            L[k_lo2:k_hi2] -= a[:k_hi2 - k_lo2]
            L *= 2.0

            Q = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                Q[2 * i] += 1.0
            if 2 * j < 2 * n - 1:
                Q[2 * j] += 1.0
            if i + j < 2 * n - 1:
                Q[i + j] -= 2.0

            # Golden section search for delta in [-a[i], a[j]]
            delta_lo = -a[i]
            delta_hi = a[j]

            if delta_hi - delta_lo < 1e-12:
                continue

            phi = (np.sqrt(5) - 1) / 2
            d1 = delta_hi - phi * (delta_hi - delta_lo)
            d2 = delta_lo + phi * (delta_hi - delta_lo)

            def eval_delta(d):
                new_max = float(np.max(conv + d * L + d * d * Q))
                return 2.0 * n * new_max

            f1 = eval_delta(d1)
            f2 = eval_delta(d2)

            for _ in range(20):
                if f1 < f2:
                    delta_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = delta_hi - phi * (delta_hi - delta_lo)
                    f1 = eval_delta(d1)
                else:
                    delta_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = delta_lo + phi * (delta_hi - delta_lo)
                    f2 = eval_delta(d2)

            delta_opt = (delta_lo + delta_hi) / 2.0
            obj_opt = eval_delta(delta_opt)

            if obj_opt < best_obj - 1e-12:
                a[i] += delta_opt
                a[j] -= delta_opt
                a = np.maximum(a, 0.0)
                a = normalize(a)
                conv = compute_conv(a)
                best_obj = compute_obj(a)
                best_a = a.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1

        if not improved:
            if no_improve > 200:
                # Perturb
                scale = 0.01 + 0.005 * min(no_improve // 200, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = normalize(a)
                conv = compute_conv(a)
                best_obj = compute_obj(a)
                best_a = a.copy()
                no_improve = 0
            if no_improve > 1000:
                break

        if verbose and (no_improve % 500 == 0 or improved):
            elapsed = time.monotonic() - start
            print(f"    PairMove: obj={best_obj:.8f}, no_improve={no_improve}, time={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 5)
    if choice == 0:
        # Uniform
        return np.ones(n)
    elif choice == 1:
        # Cosine bump
        x = np.linspace(0, 2 * np.pi, n)
        return np.maximum(1.0 + np.cos(x), 0.1)
    elif choice == 2:
        # Gaussian
        x = np.linspace(-2, 2, n)
        return np.exp(-x ** 2 / 4)
    elif choice == 3:
        # Two bumps
        x1 = (np.arange(n) - n * 0.25) / (n * 0.15)
        x2 = (np.arange(n) - n * 0.75) / (n * 0.15)
        return np.exp(-x1 ** 2) + np.exp(-x2 ** 2) + 0.01
    else:
        # Random smooth
        base = rng.uniform(0.5, 1.5, n)
        # Smooth with a window
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        smooth = np.convolve(base, kernel, mode='same')
        return np.maximum(smooth, 0.01)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines gradient descent, LP alternating,
    pair-move local search, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0  # Leave safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Gradient Descent with Log-Sum-Exp ===
    print(f"\n=== Phase 1: Gradient Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        gd_budget = min(250.0, remaining * 0.25)
        a_gd = gradient_descent_lse(best_seq, time_budget=gd_budget, verbose=True)
        obj_gd = compute_obj(a_gd)
        print(f"  GD result: obj={obj_gd:.8f}")
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        lp_budget = min(500.0, remaining * 0.4)
        a_lp = lp_alternating(best_seq, time_budget=lp_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        print(f"  LP result: obj={obj_lp:.8f}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 3: Pair-Move Local Search ===
    print(f"\n=== Phase 3: Pair-Move Local Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        pm_budget = min(300.0, remaining * 0.3)
        a_pm = pair_move_search(best_seq, time_budget=pm_budget, verbose=True)
        obj_pm = compute_obj(a_pm)
        print(f"  PairMove result: obj={obj_pm:.8f}")
        if obj_pm < best_score:
            best_score = obj_pm
            best_seq = a_pm.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 50, n_best - 20, n_best,
        n_best + 20, n_best + 50, n_best + 100, n_best + 200,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: GD + LP
        alloc_gd = min(20.0, remaining * 0.15)
        alloc_lp = min(30.0, remaining * 0.25)

        if remaining > 60:
            a_try = gradient_descent_lse(a_try, time_budget=alloc_gd)
            a_try = lp_alternating(a_try, time_budget=alloc_lp)
        else:
            a_try = lp_alternating(a_try, time_budget=min(alloc_lp, remaining * 0.5))

        obj_try = compute_obj(a_try)
        print(f"  n={n_try}: obj={obj_try:.8f}")
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 5: Fresh starts with diverse initializations ===
    print(f"\n=== Phase 5: Diverse Starts ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_try = len(best_seq)
        for init_i in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_init = structured_init(n_try, rng)
            a_init = normalize(a_init)

            # Blend with best
            blend = 0.3 + 0.2 * init_i
            a_blend = (1 - blend) * best_seq + blend * a_init
            a_blend = normalize(a_blend)

            alloc_gd = min(15.0, remaining * 0.2)
            alloc_lp = min(20.0, remaining * 0.3)

            a_opt = gradient_descent_lse(a_blend, time_budget=alloc_gd)
            a_opt = lp_alternating(a_opt, time_budget=alloc_lp)
            obj_opt = compute_obj(a_opt)
            print(f"  Init {init_i}: obj={obj_opt:.8f}")
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** Improved to {best_score:.8f}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Another round of LP
        alloc = min(remaining * 0.4, 100.0)
        a_fp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_fp = compute_obj(a_fp)
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final LP: {best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Pair move polish
        alloc = min(remaining * 0.4, 50.0)
        a_fp2 = pair_move_search(best_seq, time_budget=alloc)
        obj_fp2 = compute_obj(a_fp2)
        if obj_fp2 < best_score:
            best_score = obj_fp2
            best_seq = a_fp2.copy()
            print(f"  Final PM: {best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        # One more LP
        alloc = min(remaining * 0.5, 30.0)
        a_last = lp_alternating(best_seq, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
