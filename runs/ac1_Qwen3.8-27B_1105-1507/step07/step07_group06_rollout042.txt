def compute_conv(a):

def subgradient_polish(a, time_budget=60.0, eta_init=0.01):
    """
    Subgradient descent with exact max tracking.
    Very fast: each step is O(n) after the O(n log n) convolution.
    """

    a = np.maximum(a, 0).copy()

    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: g[j] = 2*a[k_star - j] if 0 <= k_star-j < n
        g = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            g[idx] = 2.0 * a[k_star - idx]
        
        # Project gradient to sum=0 hyperplane
        g_proj = g - np.mean(g)
        
        # Adaptive step size
        a_new = a - eta * g_proj
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.8
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            eta = min(eta * 1.05, 0.1)
        else:
            eta *= 0.95
        
        # Early stop if no improvement for a while
        if it > 100 and (it % 100 == 0):
            # Check if we're making progress
            if eta < 1e-6:
                break
    
    return best_a

            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try

            best_a = best_a_local.copy()
            a = best_a
        else:

            if it % 5 == 0 and it > 5:
                rng = np.random.default_rng(it * 104729 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_diverse_init(n, seed=42):
    """Generate diverse asymmetric initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # 1. Linear ramp
    a = np.linspace(0.01, 1.0, n)
    inits.append(a / np.sum(a))
    
    # 2. Exponential decay
    rate = rng.uniform(1.0, 3.0)
    a = np.exp(np.linspace(0, rate, n))
    inits.append(a / np.sum(a))
    
    # 3. Sinusoidal modulation of ramp
    freq = rng.uniform(1.0, 3.0)
    base = np.linspace(0.1, 1.0, n)
    mod = 0.3 * np.sin(2 * np.pi * freq * np.arange(n) / n)
    a = np.maximum(base + mod, 0.001)
    inits.append(a / np.sum(a))
    
    # 4. Power law
    power = rng.uniform(0.5, 2.0)
    a = np.power(np.linspace(0.01, 1.0, n), power)
    inits.append(a / np.sum(a))
    
    # 5. Two-step (high then low)
    split = n // rng.integers(3, 7)
    a = np.ones(n)
    a[:split] = rng.uniform(2.0, 5.0)
    inits.append(a / np.sum(a))
    
    # 6. Random smooth
    x = np.arange(n) / n
    a = np.exp(rng.normal(0, 1)) * np.exp(-2 * (x - rng.uniform(0.2, 0.8))**2)
    inits.append(a / np.sum(a))
    
    return inits

Key innovations over previous approach:

3. Fast subgradient polishing with exact max tracking (O(n) per step)

7. LP alternating with HiGHS solver
8. Multi-n exploration with diverse structured initializations

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize, linprog

# Global state for callback functions (avoids closures)
_tau_global = 1.0
_n_global = 1


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2, assuming a>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_smooth_obj_and_grad(theta, tau):
    """
    Compute smoothed objective and gradient w.r.t. theta (log-parameters).
    a = softmax(theta), obj = 2n * smooth_max(conv(a), tau)
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(theta)
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    a = exp_t / np.sum(exp_t)

    conv = compute_conv(a)
    conv_max = np.max(conv)
    smooth_val = conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau
    obj_val = 2.0 * n * smooth_val

    # Soft weights (softmax of tau * conv)
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)

    # Gradient via FFT cross-correlation: g[j] = 2 * sum_m w[j+m] * a[m]
    N = 1
    while N < (2 * n - 1) + n:
        N <<= 1
    W = fft(w, N)
    A = fft(a, N)
    c = ifft(W * np.conj(A)).real[:n]
    # d(obj)/d(a[j]) = 4n * c[j]
    # d(obj)/d(theta[j]) = 4n * a[j] * (c[j] - sum_i a[i]*c[i])
    mean_c = np.dot(a, c)
    grad_theta = 4.0 * n * a * (c - mean_c)

    return obj_val, grad_theta


def _smooth_obj_cb(th):
    """Top-level callback for minimize (objective)."""
    val, _ = compute_smooth_obj_and_grad(th, _tau_global)
    return val

    _, g = compute_smooth_obj_and_grad(th, _tau_global)

   Uses FFT-accelerated gradients.

    best_obj = compute_obj(a_init)

        np.geomspace(tau_start, 50.0, 3),
        np.geomspace(50.0, 300.0, 4),
        np.geomspace(300.0, tau_end, 3),
    ])

    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        _tau_global = tau
        stage_budget = min(remaining * 0.85, 50.0)

        # Estimate max iterations from budget
        max_iter = min(300, int(stage_budget / 0.02) + 10)

        try:

                             method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-9,
                                      'maxfun': max_iter * 3, 'ftol': 1e-14})
            if result.x is not None:
                theta_shift = result.x - np.max(result.x)
                exp_t = np.exp(theta_shift)
                a_new = exp_t / np.sum(exp_t)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def subgradient_polish(a, time_budget=60.0, eta_init=0.01):
    """
    Subgradient descent with exact max tracking.
    Each step: O(n log n) for convolution + O(n) for update.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    no_improve_count = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: g[j] = 2*a[k_star - j] if 0 <= k_star-j < n
        g = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            g[idx] = 2.0 * a[k_star - idx]

        # Project to sum=0
        g_proj = g - np.mean(g)

        a_new = a - eta * g_proj
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.8
            no_improve_count += 1
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            eta = min(eta * 1.03, 0.05)
            no_improve_count = 0
        else:
            eta *= 0.97
            no_improve_count += 1

        if no_improve_count > 200:
            # Perturb to escape
            rng = np.random.default_rng(it * 31 + n)
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            no_improve_count = 0
            eta = eta_init * 0.5

    return best_a


def multiplicative_weights(a, time_budget=60.0):
    """
    Mirror descent / multiplicative weights on simplex.
    Natural gradient for entropy-regularized simplex constraint.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.01
    no_improve = 0

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        g = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            g[idx] = 2.0 * a[k_star - idx]

        g = g - np.mean(g)

        log_a_new = np.log(a) - eta * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            eta = min(eta * 1.01, 0.3)
            no_improve = 0
        else:
            eta *= 0.98
            no_improve += 1

        if no_improve > 300:
            break

    return best_a


def peak_targeted_cd(a, time_budget=60.0):
    """
    Coordinate descent targeting top peak lags.
    Moves mass from high contributors to low contributors of the peak.
    """
    global rng_cd
    rng_cd = np.random.default_rng(12345)
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_peaks = np.argsort(conv)[-3:][::-1]

        improved = False
        for k_star in top_peaks:
            k_star = int(k_star)
            j_min = max(0, k_star - n + 1)
            j_max = min(n - 1, k_star)
            if j_min > j_max:
                continue

            # Compute contributions of each coordinate to this peak
            idx = np.arange(j_min, j_max + 1)
            contributions = a[idx] * a[k_star - idx]
            avg_contrib = np.mean(contributions)

            for _ in range(30):
                if time.monotonic() - start > time_budget:
                    break
                # Pick high contributor to reduce
                high_threshold = avg_contrib * 1.5
                high_candidates = idx[contributions > high_threshold]
                if len(high_candidates) == 0:
                    high_candidates = idx[contributions > avg_contrib]
                if len(high_candidates) == 0:
                    break
                j_src = high_candidates[rng_cd.integers(len(high_candidates))]

                # Pick low contributor to increase
                low_threshold = avg_contrib * 0.5
                low_candidates = idx[contributions < low_threshold]
                if len(low_candidates) == 0:
                    low_candidates = idx[contributions < avg_contrib]
                if len(low_candidates) == 0:
                    # Move to a position outside the peak range
                    outside = np.setdiff1d(np.arange(n), idx)
                    if len(outside) == 0:
                        break
                    j_dst = outside[rng_cd.integers(len(outside))]
                else:
                    j_dst = low_candidates[rng_cd.integers(len(low_candidates))]

                if j_src == j_dst:
                    continue

                delta = 0.005 * a[j_src] * rng_cd.uniform(0.1, 1.0)
                if delta >= a[j_src]:
                    continue

                a[j_src] -= delta
                a[j_dst] += delta
                a = np.maximum(a, 0)
                s_try = float(np.sum(a))
                if s_try < 1e-10:
                    continue
                a = a / s_try

                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[j_src] += delta
                    a[j_dst] -= delta
                    a = a / np.sum(a)

        if not improved and it > 50:
            break

    return best_a


def lp_alternating_v2(a, time_budget=120.0):
    """
    LP alternating: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Uses HiGHS solver.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(30):
        if time.monotonic() - start > time_budget:
            break

        # Build convolution matrix A (2n-1 x n): A[k,j] = a[k-j] if valid
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_min = j
            k_max = min(2 * n - 2, j + n - 1)
            A[k_min:k_max + 1, j] = a[:k_max - k_min + 1]

        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                continue
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g < 1e-10:
                continue
            g_val = g_val / s_g
        except Exception:
            continue

        # Line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            if it % 5 == 0 and it > 5:
                rng_lp = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng_lp.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coarse_shape_search(n_target, time_budget=60.0):
    """
    Search for good shape at coarse resolution, then upsample.
    Parameterizes sequence as smooth function with few degrees of freedom.
    """
    best_a = None
    best_obj = np.inf
    start = time.monotonic()

    # Try different "shape families" at coarse resolution
    coarse_n = max(20, n_target // 40)
    x = np.linspace(0, 1, coarse_n)

    shapes = []
    # Linear ramp (various slopes)
    for slope in [0.5, 1.0, 2.0, 3.0, 5.0]:
        s = np.linspace(0.01, slope, coarse_n)
        shapes.append(s / np.sum(s))
    # Exponential
    for rate in [1.0, 2.0, 4.0, 6.0]:
        s = np.exp(np.linspace(0, rate, coarse_n))
        shapes.append(s / np.sum(s))
    # Power law
    for p in [0.5, 1.5, 2.5, 4.0]:
        s = np.power(np.linspace(0.01, 1.0, coarse_n), p)
        shapes.append(s / np.sum(s))
    # Inverted (decreasing)
    for slope in [2.0, 4.0]:
        s = np.linspace(slope, 0.01, coarse_n)
        shapes.append(s / np.sum(s))
    # Two-step
    for split_frac in [0.3, 0.5, 0.7]:
        split = int(coarse_n * split_frac)
        s = np.ones(coarse_n)
        s[:split] = 3.0
        shapes.append(s / np.sum(s))
    # Gaussian-like (asymmetric)
    for center in [0.3, 0.5, 0.7]:
        s = np.exp(-10 * (x - center) ** 2) + 0.01
        shapes.append(s / np.sum(s))

    # Evaluate each shape at full resolution (quick: just resample and compute obj)
    for shape in shapes:
        if time.monotonic() - start > time_budget * 0.3:
            break
        a_full = resample_sequence(shape, n_target)
        a_full = a_full / np.sum(a_full)
        obj = compute_obj(a_full)
        if obj < best_obj:
            best_obj = obj
            best_a = a_full.copy()

    if best_a is None:
        best_a = resample_sequence(shapes[0], n_target)
        best_a = best_a / np.sum(best_a)
        best_obj = compute_obj(best_a)

    # Refine the best shape
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.7, 30.0)
        a_opt, obj_opt = smooth_lbfgs_adaptive(best_a, time_budget=alloc,
                                                tau_start=10.0, tau_end=500.0)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a, best_obj


def generate_diverse_init(n, seed=42):
    """Generate diverse asymmetric initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # 1. Linear ramp
    a = np.linspace(0.01, 1.0, n)
    inits.append(a / np.sum(a))

    # 2. Steep exponential
    a = np.exp(np.linspace(0, 4.0, n))
    inits.append(a / np.sum(a))

    # 3. Sinusoidal modulation of ramp
    base = np.linspace(0.1, 1.0, n)
    mod = 0.4 * np.sin(2 * np.pi * 2.0 * np.arange(n) / n)
    a = np.maximum(base + mod, 0.001)
    inits.append(a / np.sum(a))

    # 4. Power law (steep)
    a = np.power(np.linspace(0.01, 1.0, n), 3.0)
    inits.append(a / np.sum(a))

    # 5. Two-step
    split = n // 4
    a = np.ones(n)
    a[:split] = 4.0
    inits.append(a / np.sum(a))

    # 6. Inverted ramp
    a = np.linspace(5.0, 0.01, n)
    inits.append(a / np.sum(a))

    # 7. Asymmetric Gaussian
    x = np.arange(n) / n
    a = np.exp(-15 * (x - 0.4) ** 2) + 0.01
    inits.append(a / np.sum(a))

    return inits

            best_seq = a_cd.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
