from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog

def compute_conv_fft(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns g."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # LP: min M
    # s.t. A @ g - M <= 0  (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0, M free (but M >= 0 implicitly)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv_fft(a):
    """Compute full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def spectral_kick(a, n_iter=30):
    """
    Gerchberg-Saxton style: iterate between enforcing non-negativity in time domain
    and enforcing flat magnitude in frequency domain.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Zero-pad length for DFT
    N = 1
    while N < 4 * n:
        N <<= 1

    for it in range(n_iter):
        # Forward DFT of zero-padded signal
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A_freq = fft(a_padded)

        # Target: flat magnitude (CAZAC-like)
        target_mag = np.full(N, np.sqrt(np.sum(np.abs(A_freq) ** 2) / N))

        # Preserve phase, replace magnitude
        phase = np.angle(A_freq)
        A_new = target_mag * np.exp(1j * phase)

        # Inverse DFT
        a_new_full = np.real(ifft(A_new))
        a_new = np.maximum(a_new_full[:n], 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Try direct and blended
        obj_direct = compute_obj(a_new)
        if obj_direct < best_obj:
            best_obj = obj_direct
            best_a = a_new.copy()
            a = a_new
        else:
            improved = False
            for t in [0.2, 0.4, 0.6, 0.8]:
                a_blend = (1 - t) * a + t * a_new
                a_blend = np.maximum(a_blend, 0)
                sb = float(np.sum(a_blend))
                if sb < 1e-10:
                    continue
                a_blend = a_blend / sb

                if obj_blend < best_obj:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend
                    improved = True
                    break
            if not improved and it > n_iter // 2:
                break

    return best_a


def mirror_descent(a, time_budget):
    """Mirror descent (multiplicative weights) targeting the peak bin."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.05

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fft(a)
        k_star = int(np.argmax(conv))

        # Subgradient of conv[k_star] w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        # Project gradient to be orthogonal to a (for sum constraint)
        grad_proj = grad - np.dot(grad, a) / max(np.dot(a, a), 1e-20) * a

        # Multiplicative update
        a_new = a * np.exp(-eta * grad_proj)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            if it % 50 == 0 and eta < 0.15:
                eta *= 1.03
        else:
            eta *= 0.97
            if eta < 0.001:
                break
            a = best_a.copy()

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smooth_max_gradient_descent(a, time_budget, T_init=0.001, T_min=0.00005):
    """Gradient descent with smooth max approximation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    lr = 0.01
    momentum = 0.8
    v = np.zeros(n)
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fft(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)

        # Gradient: grad[i] = sum_k weights[k] * a[k-i] (correlation)
        grad = np.correlate(weights, a, mode='valid')

        # Center for simplex
        grad_centered = grad - np.dot(grad, a) / max(np.dot(a, a), 1e-20) * a

        # Momentum update
        v = momentum * v - lr * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0 and lr < 0.05:
                lr *= 1.02
        else:
            stall += 1
            if stall > 50:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 0.0005:
                    break

        # Temperature annealing
        T = max(T_min, T_init * (0.995 ** it))

    return best_a

        alloc = min(remaining * 0.1, 50.0)
        a_md = mirror_descent(a, time_budget=alloc)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            a = a_md
            print(f"  Mirror Descent: obj={best_score:.8f}")
        else:
            print(f"  Mirror Descent: obj={obj_md:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 150), max(300, n_best - 100), max(300, n_best - 50),
        max(300, n_best - 25), n_best,
        n_best + 25, n_best + 50, n_best + 100, n_best + 150, n_best + 200,
        500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)

        # LP optimization
        alloc_lp = min(remaining * 0.12, 50.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        # Spectral polish
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.005:
            a_opt = spectral_kick(a_opt, n_iter=20)
            obj_try = compute_obj(a_opt)

        # Smooth max polish
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.005:
            a_opt = smooth_max_gradient_descent(a_opt, time_budget=min(remaining * 0.08, 20.0))
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    a = best_seq.copy()
    for round_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # LP round
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_alternating(a, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp
            print(f"  Polish LP {round_idx}: obj={best_score:.8f}")

        # Spectral kick
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sk = spectral_kick(a, n_iter=20)
            obj_sk = compute_obj(a_sk)
            if obj_sk < best_score:
                best_score = obj_sk
                best_seq = a_sk.copy()
                a = a_sk
                print(f"  Polish Spectral {round_idx}: obj={best_score:.8f}")

        # Mirror descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_md = mirror_descent(a, time_budget=min(remaining * 0.1, 20.0))
            obj_md = compute_obj(a_md)
            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                a = a_md
                print(f"  Polish MD {round_idx}: obj={best_score:.8f}")

    # === Phase 6: Random Restarts ===
    print(f"\n=== Phase 6: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    rng = np.random.default_rng(seed)

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Various initializations
        n_r = n_best + int(rng.integers(-50, 51))
        n_r = max(400, min(1400, n_r))

        mode = r % 5
        if mode == 0:
            a_r = np.ones(n_r) / n_r
        elif mode == 1:
            x = np.arange(n_r)
            a_r = np.cos(np.pi * x / (n_r - 1)) ** 2 + 0.1
            a_r = a_r / np.sum(a_r)
        elif mode == 2:
            center = n_r // 2
            width = max(1, n_r // 4)
            x = np.arange(n_r)
            a_r = np.exp(-((x - center) ** 2) / (2 * width ** 2))
            a_r = a_r / np.sum(a_r)
        elif mode == 3:
            a_r = rng.dirichlet(np.ones(n_r) * 0.5)
        else:
            x = np.linspace(0, 1, n_r)
            a_r = np.sin(np.pi * x) ** 2 + 0.2
            a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.1, 25.0)
        a_opt = lp_alternating(a_r, time_budget=alloc * 0.7)
        obj_opt = compute_obj(a_opt)

        remaining = deadline - time.monotonic()
        if remaining > 10 and obj_opt < best_score + 0.01:
            a_opt = spectral_kick(a_opt, n_iter=10)
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r} (n={n_r}, mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}, mode={mode}): obj={obj_opt:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
