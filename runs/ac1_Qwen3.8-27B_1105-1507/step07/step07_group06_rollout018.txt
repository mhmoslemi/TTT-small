```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        lo = max(0, 0)  # k-j >= 0 => k >= j
        hi = min(2*n-1, j+n)  # k-j < n => k < j+n
        A[j:hi, j] = a[:hi-j]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # LP: min M
    # vars: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s > 1e-10:
            g = g / s
        return g, float(res.x[n])
    return None, None
```

"""
Optimized LP-alternating with adaptive step size, fast HiGHS solver,
smoothed gradient refinement, multi-n exploration, and structured restarts.
Key improvements over previous: much faster LP (scipy HiGHS vs cvxpy),
adaptive backtracking line search, periodic escape mechanisms, and
better time budgeting for maximum LP iterations.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - j]
    return A


def solve_lp_fast(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternating_intensive(a, time_budget, max_iter=5000):
    """Intensive LP alternating with adaptive step size and periodic escapes."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve_count = 0

    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            # Try with perturbed a
            a = best_a * (1 + 0.01 * rng.normal(0, 1, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            continue

        # Adaptive step size: try multiple values, pick the best
        best_t_obj = best_obj
        best_a_local = a.copy()
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Escape mechanism: after some stagnation, perturb
            if no_improve_count % 50 == 0:
                noise_level = min(0.1, 0.02 + 0.01 * (no_improve_count // 100))
                a = best_a * np.exp(rng.normal(0, noise_level, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

        # Progress reporting
        if it % 500 == 0:
            print(f"    LP-iter {it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def softmax_from_theta(theta):
    """Numerically stable softmax."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def smoothed_obj_and_grad(theta, tau, n):
    """
    Compute smoothed objective and gradient w.r.t. log-parameters theta.
    a = softmax(theta), obj = 2n * smooth_max(conv(a), tau)
    """
    a = softmax_from_theta(theta)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)

    # Soft-argmax weights for the active peaks
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)

    # Gradient computation: c_j = sum_m w[j+m] * a[m]
    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < 2 * n - 1 and end > j:
            c[j] = np.dot(w[j:end], a[:end - j])

    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta


def smooth_lbfgs(a_init, tau_start, tau_end, time_budget, max_iter=150):
    """Smoothed L-BFGS-B with tau annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    n_stages = 4
    taus = np.geomspace(tau_start, tau_end, n_stages)

    for tau in taus:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        stage_budget = min(remaining * 0.6, 60.0)
        maxfun = max(50, int(stage_budget * 20))

        def obj_func(th):
            val, _ = smoothed_obj_and_grad(th, tau, n)
            return val

        def grad_func(th):
            _, g = smoothed_obj_and_grad(th, tau, n)
            return g

        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-7, 'maxfun': maxfun})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def projected_gradient_simplex(a, tau, steps=200, time_budget=30.0):
    """Projected gradient descent on the simplex with smoothed objective."""
    n = len(a)
    a = np.maximum(a, 1e-10).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.2

    for it in range(steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        conv_max = np.max(conv)
        w = tau * np.exp((conv - conv_max) * tau)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            break
        w = w / w_sum

        c = np.zeros(n)
        for j in range(n):
            end = min(2 * n - 1, j + n)
            if j < 2 * n - 1 and end > j:
                c[j] = np.dot(w[j:end], a[:end - j])

        mean_c = np.dot(c, a)
        grad_a = 4.0 * n * (c - mean_c)
        grad_projected = grad_a - np.mean(grad_a)

        a_new = a - step_size * grad_projected
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                step_size = min(step_size * 1.05, 2.0)
            else:
                step_size *= 0.85
        else:
            step_size *= 0.5

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_initializations(n, rng):
    """Generate diverse structured initializations."""
    inits = []

    # Exponential decay
    for rate in [0.5, 1.0, 2.0, 3.0]:
        a = np.exp(-rate * np.arange(n) / n)
        inits.append(a / np.sum(a))

    # Power law decay
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 / (np.arange(n) + 1) ** alpha
        inits.append(a / np.sum(a))

    # Linear decay
    a = np.linspace(1.0, 0.1, n)
    inits.append(a / np.sum(a))
    a = np.linspace(0.1, 1.0, n)
    inits.append(a / np.sum(a))

    # Bimodal (two peaks)
    centers = [(0.2, 0.3), (0.3, 0.7), (0.1, 0.5)]
    for c1, c2 in centers:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - c1) ** 2) + np.exp(-20 * (x - c2) ** 2)
        inits.append(a / np.sum(a))

    # Sine-modulated
    x = np.linspace(0, 1, n)
    a = (1 + 0.5 * np.sin(2 * np.pi * x)) * np.exp(-x)
    inits.append(np.maximum(a, 0) / np.sum(np.maximum(a, 0)))

    return inits


def coordinate_descent_peak_shave(a, time_budget=30.0):
    """Coordinate descent targeting peak reduction."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Move mass from peak contributors to non-peak positions
        for i in range(0, n, max(1, n // 100)):
            if time.monotonic() - start > time_budget:
                break

            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 1e-8:
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        # Also try moving mass to random positions
        if not improved:
            for _ in range(30):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-8:
                    continue
                delta = 0.01 * a[i] * rng.uniform(0.5, 2.0)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:
                    break
                a = a / s
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a = best_a.copy()
            if not improved and it > 20:
                break

    return best_a
