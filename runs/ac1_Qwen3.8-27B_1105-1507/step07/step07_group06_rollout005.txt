import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def softmax_from_theta(theta):
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)

def smoothed_obj_and_grad(theta, tau, n):
    a = softmax_from_theta(theta)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    obj_val = 2.0 * n * (conv_max + np.log(np.sum(np.exp((conv - conv_max) * tau))) / tau)
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)
    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        c[j] = np.dot(w[j:end], a[:end - j])
    mean_c = np.dot(c, a)
    grad_theta = 4.0 * n * a * (c - mean_c)
    return obj_val, grad_theta

def smooth_lbfgs(a_init, tau_list, time_budget, max_iter=200):
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    for tau in tau_list:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        stage_budget = min(remaining * 0.7, 60.0)
        
        def obj_func(th):
            val, _ = smoothed_obj_and_grad(th, tau, n)
            return val
        
        def grad_func(th):
            _, g = smoothed_obj_and_grad(th, tau, n)
            return g
        
        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-7, 'maxfun': 300})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass
    return best_a, best_obj

def multiplicative_subgradient(a, n_iters=200, time_budget=30.0):
    """Multiplicative weight updates with subgradient of true max + momentum."""
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.5
    momentum = np.zeros(n)
    beta = 0.7  # momentum coefficient
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        conv_max = conv[k_star]
        
        # Subgradient of conv[k_star] w.r.t. a
        # d conv[k*]/d a[m] = a[k*-m] (for valid m) + a[m] if 2m = k*
        g = np.zeros(n)
        for m in range(n):
            km = k_star - m
            if 0 <= km < n:
                if 2 * m == k_star:
                    g[m] = a[m]
                else:
                    g[m] = 2 * a[km]
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * g
        momentum = momentum - np.mean(momentum)  # center to preserve sum
        
        # Multiplicative update
        a_new = a * np.exp(-eta * momentum)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            eta = min(eta * 1.05, 5.0)
        else:
            eta *= 0.9
            if eta < 0.01:
                break
    
    return best_a, best_obj

def peak_painting(a, time_budget=20.0):
    """Directly reduce the peak by modifying top-contributing entries."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        conv_max = conv[k_star]
        
        # Compute contributions
        contribs = np.zeros(n)
        for j in range(n):
            km = k_star - j
            if 0 <= km < n:
                if 2 * j == k_star:
                    contribs[j] = a[j] * a[j]
                else:
                    contribs[j] = 2 * a[j] * a[km]
        
        # Find top contributor
        j_star = int(np.argmax(contribs))
        if contribs[j_star] < 1e-15:
            break
        
        # Try to move mass from j_star to a position that doesn't contribute to conv[k_star]
        # A position j' contributes to conv[k_star] if a[k_star - j'] > 0 (i.e., k_star - j' is in [0, n-1])
        # So we want j' such that k_star - j' is out of range, OR a[k_star - j'] is very small
        candidates = []
        for j_prime in range(n):
            km_prime = k_star - j_prime
            if not (0 <= km_prime < n):
                candidates.append((j_prime, 1.0))  # no contribution
            else:
                candidates.append((j_prime, a[km_prime]))
        
        # Sort by "safety" (lower a[km_prime] is better)
        candidates.sort(key=lambda x: x[1])
        
        improved = False
        for j_prime, _ in candidates[:10]:
            if j_prime == j_star:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = frac * a[j_star]
                if delta < 1e-15:
                    continue
                a_save = a.copy()
                a[j_star] -= delta
                a[j_prime] += delta
                a = np.maximum(a, 0)
                s_try = float(np.sum(a))
                if s_try < 1e-10:
                    a = a_save.copy()
                    continue
                a = a / s_try
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = a_save.copy()
            if improved:
                break
        
        if not improved:
            # Try random perturbations
            rng = np.random.default_rng(it * 104729 + n)
            for _ in range(20):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.05 * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                break
    
    return best_a

        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_cur = len(best_seq)
    
    # Phase 1: Parametric search for diverse initializations
    print(f"\n=== Phase 1: Parametric Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 90.0)
        a_param, obj_param = parametric_search(n_cur, time_budget=alloc)
        print(f"  Best parametric: obj={obj_param:.8f}")
        if obj_param < best_score:
            best_score = obj_param
            best_seq = a_param.copy()
            n_cur = len(best_seq)
            print(f"  ** Improved to {best_score:.8f}")
    
    # Phase 2: Smoothed L-BFGS-B (main optimization)
    print(f"\n=== Phase 2: Smoothed L-BFGS-B ===")
    for trial in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if trial == 0:
            a_init = best_seq.copy()
        elif trial == 1:
            a_param2, obj_param2 = parametric_search(n_cur, time_budget=min(30, remaining*0.1))
            a_init = a_param2.copy() if obj_param2 < best_score else best_seq * np.exp(rng.normal(0, 0.05, n_cur))
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / np.sum(a_init)
        elif trial == 2:
            # Sigmoid-like init
            t = np.linspace(0, 1, n_cur)
            a_init = 1.0 / (1.0 + np.exp(-20 * (t - 0.4)))
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / np.sum(a_init)
        else:
            # From parametric search
            a_param3, _ = parametric_search(n_cur, time_budget=min(20, remaining*0.05))
            a_init = a_param3.copy()
        
        # Higher tau values for sharper optimization
        alloc = min(remaining * 0.25, 100.0)
        taus = [20.0, 100.0, 500.0, 1000.0, 2000.0]
        a_opt, obj_opt = smooth_lbfgs(a_init, taus, alloc, max_iter=150)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  Trial {trial}: obj={best_score:.8f} **")
        else:
            print(f"  Trial {trial}: obj={obj_opt:.8f}")
    
    # Phase 3: Multiplicative subgradient (fast polish)
    print(f"\n=== Phase 3: Multiplicative Subgradient ===")
    remaining = deadline - time.monotonic()

    remaining = deadline - time.monotonic()

        alloc = min(remaining * 0.1, 30.0)
        a_pp = peak_painting(best_seq, time_budget=alloc)
        obj_pp = compute_obj(a_pp)
        if obj_pp < best_score:
            best_score = obj_pp
            best_seq = a_pp.copy()
            print(f"  Peak paint: obj={best_score:.8f} **")
        else:
            print(f"  Peak paint: obj={obj_pp:.8f}")
    
    # Phase 5: Basin hopping
    print(f"\n=== Phase 5: Basin Hopping ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.3, 180.0)
        a_bh = basin_hop(best_seq, n_hops=5, time_budget=alloc)
        obj_bh = compute_obj(a_bh)
        if obj_bh < best_score:
            best_score = obj_bh
            best_seq = a_bh.copy()
            print(f"  Basin hop: obj={best_score:.8f} **")
        else:
            print(f"  Basin hop: obj={obj_bh:.8f}")
    
    # Phase 6: LP alternating
    print(f"\n=== Phase 6: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.25, 90.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")
    
    # Phase 7: Final intensive polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more smoothed optimization with very high tau
        alloc = min(remaining * 0.3, 40.0)
        a_opt, obj_opt = smooth_lbfgs(best_seq, [500.0, 1000.0, 2000.0, 5000.0], alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final smooth: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Multiplicative polish
        alloc = min(remaining * 0.3, 20.0)
        a_mul, obj_mul = multiplicative_subgradient(best_seq, n_iters=200, time_budget=alloc)
        if obj_mul < best_score:
            best_score = obj_mul
            best_seq = a_mul.copy()
            print(f"  Final multi: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Peak painting polish
        alloc = min(remaining * 0.4, 20.0)
        a_pp = peak_painting(best_seq, time_budget=alloc)
        obj_pp = compute_obj(a_pp)
        if obj_pp < best_score:
            best_score = obj_pp
            best_seq = a_pp.copy()
            print(f"  Final PP: obj={best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
