"""
Hybrid optimizer combining exact coordinate descent, simulated annealing,
exponentiated gradient, and multi-n exploration to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key improvement: exact 1D coordinate minimization with incremental convolution
updates, plus aggressive simulated annealing to escape local minima.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_after_change(a, i, x):
    """Compute convolution of a with a[i] replaced by x, efficiently."""
    n = len(a)
    old_v = a[i]
    if abs(x - old_v) < 1e-15:
        return compute_conv(a)
    # c_new[k] = c_old[k] + 2*(x-old_v)*a'[k-i] + (x^2 - old_v^2)*delta(k,2i)
    # where a' = a with a[i]=0
    a[i] = 0.0
    c_base = compute_conv(a)
    a[i] = old_v
    # d[k] = a[k-i] if 0<=k-i<n and k-i != i, else 0
    d = np.zeros(2 * n - 1)
    lo = max(0, i - (n - 1))
    hi = min(2 * n - 2, i + (n - 1))
    for k in range(lo, hi + 1):
        j = k - i
        if 0 <= j < n and j != i:
            d[k] = a[j]
    delta = x - old_v
    c_new = c_base + 2.0 * delta * d + (x * x - old_v * old_v) * np.zeros(2 * n - 1)
    if 2 * i <= 2 * n - 2:
        c_new[2 * i] += x * x - old_v * old_v
    return c_new


def f_of_x(c_base, d, k2i, x):
    """Evaluate f(x) = max_k (c_base[k] + 2*x*d[k] + x^2*delta(k,k2i))."""
    vals = c_base + 2.0 * x * d
    if 0 <= k2i < len(vals):
        vals[k2i] += x * x
    return np.max(vals)


def solve_1d_coord(a, i, max_iter=50):
    """
    Solve 1D subproblem: find x >= 0 minimizing max_k (a_new*a_new)[k]
    where a_new = a with a[i] replaced by x.
    """
    n = len(a)
    v = a[i]
    old_v = v
    a[i] = 0.0
    c_base = compute_conv(a)
    a[i] = v

    d = np.zeros(2 * n - 1)
    lo = max(0, i - (n - 1))
    hi = min(2 * n - 2, i + (n - 1))
    for k in range(lo, hi + 1):
        j = k - i
        if 0 <= j < n and j != i:
            d[k] = a[j]
    k2i = 2 * i
    if k2i >= 2 * n - 1:
        k2i = -1

    # f(x) = max(c_base + 2*x*d, c_base[k2i] + x^2) for x >= 0
    # Use golden section search on [0, upper]
    f0 = f_of_x(c_base, d, k2i, 0.0)
    f_v = f_of_x(c_base, d, k2i, v)

    # Determine search bracket
    if f0 <= f_v:
        # Optimum is in [0, v] or at 0
        lo_b, hi_b = 0.0, v
    else:
        # Optimum might be > v
        # f(x) >= c_base[k2i] + x^2, so if f_v < c_base[k2i] + (2v)^2, try up to 2v
        hi_try = max(2.0 * v, 0.1)
        f_hi = f_of_x(c_base, d, k2i, hi_try)
        if f_hi < f_v:
            lo_b, hi_b = v, hi_try
        else:
            lo_b, hi_b = 0.0, hi_try

    if hi_b - lo_b < 1e-12:
        best_x = (lo_b + hi_b) / 2
    else:
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_g = lo_b
        b_g = hi_b
        c_pt = b_g - gr * (b_g - a_g)
        d_pt = a_g + gr * (b_g - a_g)
        fc = f_of_x(c_base, d, k2i, c_pt)
        fd = f_of_x(c_base, d, k2i, d_pt)
        for _ in range(max_iter):
            if b_g - a_g < 1e-12 * max(1.0, abs(a_g)):
                break
            if fc < fd:
                b_g = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = b_g - gr * (b_g - a_g)
                fc = f_of_x(c_base, d, k2i, c_pt)
            else:
                a_g = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = a_g + gr * (b_g - a_g)
                fd = f_of_x(c_base, d, k2i, d_pt)
        best_x = (a_g + b_g) / 2.0

    best_x = max(0.0, best_x)
    return best_x


def coordinate_descent(a, passes=20, time_budget=30.0, verbose=False):
    """Run coordinate descent with exact 1D minimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    order = np.arange(n)

    for p in range(passes):
        if time.monotonic() - start > time_budget:
            break
        np.random.shuffle(order)
        improved = 0
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            new_x = solve_1d_coord(a, i, max_iter=30)
            if new_x >= 0 and abs(new_x - a[i]) > 1e-14:
                a[i] = new_x
                improved += 1

        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        if verbose and p % 5 == 0:
            print(f"    CD pass {p}: obj={obj:.10f}, improved={improved}")
        if improved < n * 0.01:
            break

    return best_a


def exp_gradient_fast(a, n_iter=5000, time_budget=30.0, verbose=False):
    """Fast exponentiated gradient with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.5 / np.sqrt(n)

    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient: d(conv[k_star])/d(a[i]) = 2*a[k_star-i] (if valid)
        grad = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = max(eta * 0.9995, 0.05 / np.sqrt(n))
        else:
            eta *= 0.95
            if eta < 0.01 / np.sqrt(n):
                # Restart with perturbation
                rng = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                eta = 0.5 / np.sqrt(n)

    return best_a


def simulated_annealing(a, time_budget=60.0, verbose=False):
    """Simulated annealing with coordinate descent restarts."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    T = 0.1
    T_min = 0.001
    cool_rate = 0.999

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        # Generate perturbation
        scale = T * np.sqrt(n)
        perturbation = rng.normal(0, scale, n)
        a_try = a + perturbation
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a = best_a.copy()
            T *= cool_rate
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        obj_cur = compute_obj(a)

        # Acceptance criterion
        if obj_try < obj_cur or rng.random() < np.exp(-(obj_try - obj_cur) / max(T, 1e-10)):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                if verbose and it % 100 == 0:
                    print(f"    SA it {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            if it % 50 == 0:
                # Periodic polish
                a_polished = coordinate_descent(a, passes=3, time_budget=2.0)
                obj_pol = compute_obj(a_polished)
                if obj_pol < best_obj:
                    best_obj = obj_pol
                    best_a = a_polished.copy()
                    a = a_polished

        T = max(T * cool_rate, T_min)

    # Final polish
    final = coordinate_descent(best_a, passes=5, time_budget=max(5.0, time_budget * 0.1))
    obj_final = compute_obj(final)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = final

    return best_a


def gradient_polish(a, time_budget=30.0, verbose=False):
    """L-BFGS-B gradient polish on smoothed objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    x = np.log(np.maximum(a, 1e-10)) - np.log(np.maximum(a, 1e-10)).max()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in [5.0 / n, 2.0 / n, 0.5 / n, 0.1 / n]:
        if time.monotonic() - start > time_budget:
            break

        def fun(xv):
            xv = xv - xv.max()
            av = np.exp(xv)
            av = av / np.sum(av)
            conv = compute_conv(av)
            cs = conv / tau
            cs_max = cs.max()
            return float(cs_max + np.log(np.sum(np.exp(cs - cs_max))))

        def fgrad(xv):
            xv = xv - xv.max()
            av = np.exp(xv)
            av = av / np.sum(av)
            conv = compute_conv(av)
            cs = conv / tau
            cs_max = cs.max()
            w = np.exp(cs - cs_max)
            w = w / w.sum()
            # Cross-correlation
            size = next_pow2(2 * n - 1)
            a_flip = av[::-1]
            fa_flip = rfft(a_flip, size)
            fw = rfft(w, size)
            conv_w_af = irfft(fw * fa_flip, size)
            corr = conv_w_af[n - 1:n - 1 + n]
            s_val = np.sum(av)
            F = fun(xv) / 1.0  # already normalized by 1/s^2 through the exp normalization
            grad = 2.0 * corr / tau - 2.0 * F * av / s_val
            return grad

        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-10})
            if res.x is not None:
                x_opt = res.x - res.x.max()
                a_new = np.exp(x_opt)
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = x_opt
                    if verbose:
                        print(f"    Grad polish tau={tau:.4f}: obj={best_obj:.10f}")
        except Exception:
            pass

    return best_a


def symmetric_coordinate_descent(a, passes=15, time_budget=30.0):
    """Coordinate descent on palindromic sequences."""
    n = len(a)
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 0)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    half = (n + 1) // 2

    for p in range(passes):
        if time.monotonic() - start > time_budget:
            break
        for i in range(half):
            if time.monotonic() - start > time_budget:
                break
            # Optimize a[i] and a[n-1-i] simultaneously (they're equal)
            # Try a few values
            v = a[i]
            best_x = v
            best_f = compute_obj(a)
            for factor in [0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0, 0.0]:
                a_try = a.copy()
                xv = v * factor
                a_try[i] = xv
                a_try[n - 1 - i] = xv
                a_try = np.maximum(a_try, 0)
                s_t = np.sum(a_try)
                if s_t < 1e-10:
                    continue
                a_try = a_try / s_t
                f_t = compute_obj(a_try)
                if f_t < best_f:
                    best_f = f_t
                    best_x = xv
            a[i] = best_x
            a[n - 1 - i] = best_x

        a = np.maximum(a, 0)
        a = a / np.sum(a)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        obj = compute_obj(a_cd)

            best_score = obj
            best_seq = a_cd.copy()
            print(f"  After CD: obj={best_score:.10f}")
        else:
            print(f"  CD: obj={obj:.10f} (no improvement)")

    # === Phase 2: Exponentiated gradient ===
    print(f"\n=== Phase 2: Exponentiated Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 40.0)
        a_eg = exp_gradient_fast(best_seq, n_iter=20000, time_budget=alloc)
        obj = compute_obj(a_eg)
        if obj < best_score:
            best_score = obj
            best_seq = a_eg.copy()
            print(f"  After EG: obj={best_score:.10f}")
        else:
            print(f"  EG: obj={obj:.10f}")

    # === Phase 3: Multi-n exploration with coordinate descent ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_candidates = sorted(set([
        max(400, n_parent - 200),
        max(400, n_parent - 100),
        n_parent,
        min(2000, n_parent + 100),
        min(2000, n_parent + 200),
        min(2000, n_parent + 400),
        600, 800, 1000, 1200, 1500, 1800
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Coordinate descent
        alloc_cd = min(remaining * 0.12, 30.0)
        a_opt = coordinate_descent(a_try, passes=15, time_budget=alloc_cd)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 4: Simulated annealing on best ===
    print(f"\n=== Phase 4: Simulated Annealing (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 120.0)
        print(f"  Budget: {alloc:.0f}s")
        a_sa = simulated_annealing(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_sa)
        if obj < best_score:
            best_score = obj
            best_seq = a_sa.copy()
            print(f"  After SA: obj={best_score:.10f}")
        else:
            print(f"  SA: obj={obj:.10f}")

    # === Phase 5: Symmetric variant ===
    print(f"\n=== Phase 5: Symmetric Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 30.0)
        a_sym = symmetric_coordinate_descent(best_seq, passes=10, time_budget=alloc)
        obj = compute_obj(a_sym)
        if obj < best_score:
            best_score = obj
            best_seq = a_sym.copy()
            print(f"  Symmetric: obj={best_score:.10f} **")
        else:
            print(f"  Symmetric: obj={obj:.10f}")

    # === Phase 6: Coordinate descent on best n (more passes) ===
    print(f"\n=== Phase 6: Deep Coordinate Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.2, 80.0)
        a_cd2 = coordinate_descent(best_seq, passes=50, time_budget=alloc, verbose=True)
        obj = compute_obj(a_cd2)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd2.copy()
            print(f"  Deep CD: obj={best_score:.10f}")
        else:
            print(f"  Deep CD: obj={obj:.10f}")

    # === Phase 7: Gradient polish ===
    print(f"\n=== Phase 7: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.15, 40.0)
        a_gp = gradient_polish(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_gp)
        if obj < best_score:
            best_score = obj
            best_seq = a_gp.copy()
            print(f"  Grad polish: obj={best_score:.10f}")
        else:
            print(f"  Grad polish: obj={obj:.10f}")

    # === Phase 8: Final SA + CD on slightly different n ===
    print(f"\n=== Phase 8: Final Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_best = len(best_seq)
        # Try one more n value close to best
        n_extra = n_best + 50
        if n_extra != n_best and n_extra <= 2000:
            a_extra = resample_sequence(best_seq, n_extra)
            a_extra = a_extra * np.exp(rng.normal(0, 0.05, n_extra))
            a_extra = np.maximum(a_extra, 0)
            a_extra = a_extra / np.sum(a_extra)
            alloc2 = min(remaining * 0.2, 30.0)
            a_extra = coordinate_descent(a_extra, passes=10, time_budget=alloc2)
            obj = compute_obj(a_extra)
            if obj < best_score:
                best_score = obj
                best_seq = a_extra.copy()
                print(f"  Extra n={n_extra}: obj={best_score:.10f} **")
            else:
                print(f"  Extra n={n_extra}: obj={obj:.10f}")

    # === Phase 9: Final coordinate descent polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        a_final = coordinate_descent(best_seq, passes=20, time_budget=alloc)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
        print(f"  Final: n={len(best_seq)}, obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.array(result))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
