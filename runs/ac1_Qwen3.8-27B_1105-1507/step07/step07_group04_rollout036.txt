import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def softmax_conv_weights(conv, tau):
    """Compute softmax weights over convolution values."""
    # Numerical stability
    conv_shifted = conv - np.max(conv)
    exp_vals = np.exp(tau * conv_shifted)
    weights = exp_vals / np.sum(exp_vals)
    return weights

def compute_gradient(a, weights):
    """
    Compute gradient of LSE(tau * conv(a)) w.r.t. a.
    grad[i] = 2 * sum_k weights[k] * a[k-i]
    This is 2 * (weights cross-correlate a)[i]
    """
    n = len(a)
    # weights has length 2n-1
    # grad[i] = 2 * sum_{k=max(0,i)}^{min(2n-2, i+n-1)} weights[k] * a[k-i]
    # This is the cross-correlation of weights and a
    # Equivalent to: convolve(reversed(weights), a) at position i + (2n-2)
    
    # Use FFT for efficiency
    size = 1
    while size < len(weights) + n - 1:
        size <<= 1
    
    # cross-correlation: (w ⊛ a)[i] = sum_k w[k] * a[k-i]
    # = conv(flipped_w, a)[i + len(w) - 1]
    flipped_w = weights[::-1]
    size2 = 1
    while size2 < len(flipped_w) + n - 1:
        size2 <<= 1
    fw = rfft(flipped_w, size2)
    fa = rfft(a, size2)
    corr = irfft(fw * fa, size2)
    # corr[i + len(w) - 1] = (w ⊛ a)[i]
    # len(w) = 2n-1, so offset = 2n-2
    offset = len(weights) - 1
    grad = 2.0 * corr[offset:offset+n]
    return grad

    """
    Exponentiated gradient (mirror descent) for minimizing LSE(tau * conv(a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_objective = compute_objective(a)
    
    eta = 0.1  # step size
    lam = 0.9  # momentum
    
    momentum = np.zeros(n)
    
    for it in range(n_iter):
        conv = compute_conv(a)
        weights = softmax_conv_weights(conv, tau)
        grad = compute_gradient(a, weights)
        
        # Center gradient (project to sum=0 hyperplane for the constraint)
        grad_centered = grad - np.mean(grad)
        
        # Momentum
        momentum = lam * momentum + grad_centered
        
        # Exponentiated gradient update
        a_new = a * np.exp(-eta * momentum)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Check objective
        obj = compute_objective(a_new)
        if obj < best_objective:
            best_objective = obj
            best_a = a_new.copy()
        
        # Adaptive step size
        if obj < compute_objective(a):
            eta = min(eta * 1.05, 2.0)
            a = a_new
        else:

            # Don't update a if it got worse
            # But occasionally accept for exploration
            if it % 20 == 0:

    return best_a

def projected_gradient_descent(a, tau, n_iter=500, verbose=False):

    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_objective = compute_objective(a)
    
    eta = 0.05
    x = a.copy()  # Fista lookahead
    t = 1.0
    
    for it in range(n_iter):
        conv = compute_conv(x)
        weights = softmax_conv_weights(conv, tau)
        grad = compute_gradient(x, weights)
        grad_centered = grad - np.mean(grad)
        
        # Gradient step
        y = x - eta * grad_centered
        
        # Project onto simplex
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy
        
        # Fista update
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        x = y + (t-1)/t_new * (y - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx
        
        a = y
        t = t_new
        
        obj = compute_objective(a)
        if obj < best_objective:
            best_objective = obj
            best_a = a.copy()
        
        # Decay step size
        if it % 100 == 99:
            eta = max(eta * 0.8, 0.001)
    
    return best_a

"""
Continuation-based optimization with exponentiated gradient, LP alternating,
and peak-shaving heuristics. Optimizes 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_max_conv_normalized(a):
    """Return max of conv(a_norm, a_norm) where a_norm has sum 1."""
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(np.max(conv))


def softmax_topk_weights(conv, tau, topk=None):
    """Compute softmax weights over top-K convolution values."""
    if topk is None or topk >= len(conv):
        topk = len(conv)
    
    # Get top-k indices
    if topk < len(conv):
        top_idx = np.argpartition(conv, -topk)[-topk:]
    else:
        top_idx = np.arange(len(conv))
    
    # Compute weights only over top-k
    vals = conv[top_idx]
    vals_shifted = vals - np.max(vals)
    exp_vals = np.exp(tau * vals_shifted)
    w = np.zeros(len(conv))
    w[top_idx] = exp_vals / np.sum(exp_vals)
    
    return w


def compute_grad_from_weights(a, weights):
    """
    Compute gradient: grad[i] = 2 * sum_k weights[k] * a[k-i].
    Uses FFT cross-correlation.
    """
    n = len(a)
    lw = len(weights)
    
    # cross-correlation: (w ⊛ a)[i] = sum_k w[k] * a[k-i]
    # = linear_convolution(flipped_w, a)[i + lw - 1]
    flipped_w = weights[::-1]
    
    size = 1
    while size < lw + n - 1:
        size <<= 1
    
    fw = rfft(flipped_w, size)
    fa = rfft(a, size)
    corr = irfft(fw * fa, size)
    
    offset = lw - 1
    grad = 2.0 * corr[offset:offset + n]
    return grad

    Continuation exponentiated gradient (mirror descent) with LSE smoothing.

    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    for tau in tau_values:
        eta = 0.05
        for it in range(max_iter_per_tau):
            if it % 50 == 0:
                conv = compute_conv(a)
                max_c = float(np.max(conv))
                current_obj = 2.0 * n * max_c
                if current_obj < best_obj:
                    best_obj = current_obj
                    best_a = a.copy()

            conv = compute_conv(a)
            weights = softmax_topk_weights(conv, tau, topk=min(50, len(conv)))
            grad = compute_grad_from_weights(a, weights)
            
            # Center gradient
            grad_c = grad - np.mean(grad)
            
            # Exponentiated gradient update
            a_new = a * np.exp(-eta * grad_c)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            # Check improvement
            conv_new = compute_conv(a_new)
            max_c_new = float(np.max(conv_new))
            
            if max_c_new < max_c:
                a = a_new
                eta = min(eta * 1.02, 0.5)
            else:
                eta = max(eta * 0.85, 0.005)
                if it % 100 == 99:
                    a = a_new  # occasional exploration
    
    # Final check
    final_obj = compute_objective(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    
    return best_a


def fista_optimize(a, tau_values, max_iter_per_tau=500, verbose=False):
    """
    Continuation Fista (accelerated projected gradient) with LSE smoothing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    for tau in tau_values:
        eta = 0.02
        x = a.copy()
        t = 1.0
        
        for it in range(max_iter_per_tau):
            conv = compute_conv(x)
            if it % 20 == 0 and it > 0:
                obj_check = 2.0 * n * float(np.max(compute_conv(a)))
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()
            
            weights = softmax_topk_weights(conv, tau, topk=min(50, len(conv)))
            grad = compute_grad_from_weights(x, weights)
            grad_c = grad - np.mean(grad)
            
            y = x - eta * grad_c
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy > 1e-10:
                y = y / sy
            
            t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
            x = y + ((t - 1) / t_new) * (y - a)
            x = np.maximum(x, 0)
            sx = float(np.sum(x))
            if sx > 1e-10:
                x = x / sx
            
            a = y
            t = t_new
            eta = max(eta * 0.999, 0.001)
    
    final_obj = compute_objective(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL, cp.CVXOPT]
    if solver:
        solvers = [solver] + [s for s in solvers if s != solver]
    
    for sol in solvers:
        try:
            prob.solve(solver=sol)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternating(a, max_rounds=50, verbose=False):
    """
    LP alternating: alternate between solving LP for g given a, 
    and updating a toward g.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    t = 0.5
    stall = 0
    
    for round_idx in range(max_rounds):
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Try several mixing parameters
        best_t_obj = best_obj
        best_t_val = t
        best_a_local = best_a.copy()
        
        candidate_ts = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for t_try in candidate_ts:
            a_try = (1 - t_try) * a + t_try * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_objective(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            t = best_t_val
            stall = 0
            if verbose:
                print(f"  LP iter {round_idx}: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            stall += 1
            if stall > 10:
                # Perturb and try again
                rng = np.random.default_rng(round_idx * 7919)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
                t = 0.5
            elif stall > 20:
                break
    
    return best_a


def peak_shaving(a, n_iter=100, verbose=False):
    """
    Direct peak shaving: identify peak positions and reduce contributing coefficients.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    conv = compute_conv(a)
    
    for it in range(n_iter):
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        
        # Find which pairs (i, max_idx - i) contribute to this peak
        # Contribution of a[i] is a[max_idx - i]
        # We want to reduce the product a[i] * a[max_idx - i]
        
        i_max = 0
        max_contrib = 0
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                if contrib > max_contrib:
                    max_contrib = contrib
                    i_max = i
        
        # Reduce a[i_max] and increase a[some other position]
        # Choose position to increase: one that doesn't contribute to any peak
        # Simple heuristic: increase the position with smallest value
        j_max = max_idx - i_max
        
        # Find a "safe" position to increase
        # Positions that don't appear in the top peaks are safe
        top_peaks = np.argsort(conv)[::-1][:5]
        dangerous = set()
        for p in top_peaks:
            p = int(p)
            for i in range(n):
                j = p - i
                if 0 <= j < n:
                    dangerous.add(i)
                    dangerous.add(j)
        
        safe_positions = [i for i in range(n) if i not in dangerous and i != i_max]
        if safe_positions:
            target = min(safe_positions, key=lambda i: a[i])
        else:
            target = (i_max + n // 2) % n
        
        # Move some mass
        delta = 0.01 * a[i_max]
        if delta > a[i_max] * 0.5:
            delta = a[i_max] * 0.1
        
        a[i_max] -= delta
        a[target] += delta
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
        
        conv = compute_conv(a)
        obj = 2.0 * n * float(np.max(conv))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
