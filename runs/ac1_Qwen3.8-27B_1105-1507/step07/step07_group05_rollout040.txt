from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))

- For large n, the optimal approaches a "flat" autocorrelation

"""
Optimizes 2n*max(autoconv(a))/(sum(a))^2 using:
1. Smooth peak minimization (log-sum-exp gradient)
2. LP alternating optimization  
3. Frequency-domain flattening
4. Multi-n exploration

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2*n - 1]

def compute_obj(a):
    """Compute 2n*max(autoconv)/(sum)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = compute_conv(a / s)
    return 2.0 * n * float(np.max(b))

def next_pow2(m):
    N = 1
    while N < m:
        N <<= 1
    return N

def smooth_peak_gd(a, time_budget=120.0, tau_init=30.0, verbose=False):

        return a

    best_obj = compute_obj(a)

    N3 = next_pow2(3*n - 2)

    eta = 0.5
    momentum = 0.5

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp_dual(a):
    """
    Solve the dual of the LP: max sum(y) s.t. A^T y <= 1, y >= 0
    where A is the conv matrix for fixed a.
    Returns the primal solution g.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    try:
        from scipy.optimize import linprog
        # Dual: max sum(y_k) s.t. for each j: sum_k A[k,j] y_k <= 1, y_k >= 0
        # Primal: min M s.t. A @ g <= M, sum(g) = 1, g >= 0
        # We solve the primal directly with linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2*n-1)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

        best_t = 0.3

            s_try = float(np.sum(a_try))

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:

                best_t = t

            best_obj = best_t_obj
            a_new = np.maximum((1-best_t)*a + best_t*g_lp, 0)
            a_new = a_new / np.sum(a_new)
            best_a = a_new.copy()

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(autoconv(a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = compute_conv(a / s)
    return 2.0 * n * float(np.max(b))


def next_pow2(m):
    N = 1
    while N < m:
        N <<= 1
    return N

        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v[:] = 0.0
            continue
        a_new = a_new / s_new

        # Compute objective
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.003, 5.0)
            tau = min(tau * 1.0008, 300.0)
            momentum = min(momentum + 0.0005, 0.85)

            # Polyak averaging (exponential)
            avg_a = 0.95 * avg_a + 0.05 * a_new
            avg_count += 1
            fa_avg = rfft(avg_a / np.sum(avg_a), N)
            b_avg = irfft(fa_avg * fa_avg, N)[:2 * n - 1]
            avg_obj = 2.0 * n * float(np.max(b_avg))
            if avg_obj < avg_best_obj:
                avg_best_obj = avg_obj
                avg_best_a = avg_a.copy()
        else:
            stall += 1
            eta *= 0.97
            v *= 0.6
            if stall > 300:
                # Restart with perturbation
                rng_local = np.random.default_rng((it * 7919 + n) & 0xFFFFFFFF)
                noise = rng_local.normal(0, 0.03, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                v[:] = 0.0
                eta = 1.0
                tau = tau_init
                momentum = 0.6
                stall = 0
            if eta < 1e-10:
                break

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      SP-GD it{it}: best={best_obj:.8f}, avg={avg_best_obj:.8f}, "
                  f"tau={tau:.1f}, eta={eta:.3f}, t={elapsed:.0f}s")

    # Check if Polyak average is better
    if avg_best_obj < best_obj:
        best_obj = avg_best_obj
        best_a = avg_best_a / np.sum(avg_best_a)

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization with step-size adaptation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = np.maximum((1 - t) * a + t * g_lp, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            a_new = np.maximum((1 - best_t) * a + best_t * g_lp, 0)
            a_new = a_new / float(np.sum(a_new))
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall > 15:
                break

    return best_a
