def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Convolution a*b via FFT."""
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf

def smoothed_obj_and_grad(x, tau):
    """
    Compute smoothed objective and gradient for a = exp(x).
    Objective: logsumexp(conv(a,a)/tau) / (sum(a))^2
    Gradient computed via FFT.
    """
    n = len(x)
    # Normalize x for numerical stability (objective is scale-invariant)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    
    # Convolution
    conv = compute_conv(a)
    
    # Softmax weights
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    
    # Objective value: logsumexp(c/tau) / s^2
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    
    # Gradient: dF/dx_i = 2*corr[i]/(tau*s^2) - 2*F*a_i/s
    # where corr[i] = sum_k w[k]*a[k-i] = cross-correlation
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, use_cvxpy=True):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    if use_cvxpy:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    
    # Fallback: scipy linprog
    try:
        A = build_conv_matrix(a_norm)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def resample_sequence(a, new_n):
    """Resample to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Smooth random (low-pass filtered)
        a = rng.uniform(0, 1, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)
    elif mode == 3:
        # Triangular
        a = 1.0 - np.abs(np.arange(n) - n / 2) / (n / 2)
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 4:
        # Two peaks
        c1, c2 = n // 3, 2 * n // 3
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n * n / 20)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n * n / 20))
        a = a / np.sum(a)
    else:
        # Flat with noise
        a = np.ones(n) + 0.3 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a
