from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    n = len(a)
    a = a / np.sum(a)

    # conv(a,g)[k] - M <= 0 for each k
    # A has shape (2n-1, n+1)
    
    A = np.zeros((2 * n - 1, n + 1))

        for i in range(max(0, k - n + 1), min(k, n - 1) + 1):

            A[k, j] = a[i]
        A[k, n] = -1.0

    # f(t) = max_k [(1-t)^2 * conv_a[k] + 2t(1-t) * cross[k] + t^2 * conv_g[k]]
    # Each term is quadratic in t: A_k - 2A_k*t + A_k*t^2 + 2B_k*t - 2B_k*t^2 + C_k*t^2
    #   = A_k + (2B_k - 2A_k)*t + (A_k - 2B_k + C_k)*t^2
    
    def f(t):
        vals = (1-t)**2 * conv_a + 2*t*(1-t) * cross + t**2 * conv_g
        return np.max(vals)
    
    # Golden section search on [0, 1]
    phi = (np.sqrt(5) - 1) / 2  # ~0.618
    a, b = 0.0, 1.0
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = f(c)
    fd = f(d)
    
    for _ in range(50):  # 50 iterations gives precision ~ 0.618^50 ~ 1e-11
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = f(d)
    
    t_opt = (a + b) / 2
    return t_opt

"""
Optimized solver for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP alternating with golden-section line search,
Nesterov momentum, multi-n exploration, and smooth-max polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross_conv(a, b):
    """Cross convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def golden_t(conv_a, cross, conv_g, a, g):
    """
    Find optimal t in [0,1] for blend (1-t)*a + t*g.
    Uses golden section search on the max of the blended convolution.
    """
    def f(t):
        vals = (1.0 - t) ** 2 * conv_a + 2.0 * t * (1.0 - t) * cross + t * t * conv_g
        return float(np.max(vals))
    
    # Golden section search
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(60):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = f(d)
    
    t_opt = (lo + hi) / 2.0
    return t_opt, f(t_opt)


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-15:
        return None, None
    a_norm = a / a_sum
    
    # Build LP: min M s.t. A @ [g; M] <= 0, sum(g) = 1, g >= 0
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n + 1))
    
    for k in range(num_constraints):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            A_ub[k, j] = a_norm[i]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(num_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternating_fast(a, time_budget=60.0, verbose=False):
    """
    Fast LP alternating with golden-section line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    conv_a = compute_conv(a)
    
    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

            stall += 10
            if stall > 30:
                break
            continue

def smooth_max_gradient_descent(a, time_budget=30.0, T_init=0.001, T_min=0.00005):
    """
    Gradient descent with smooth max (log-sum-exp) approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = T_init
    lr = 0.01
    v = np.zeros(n)
    momentum = 0.9
    stall = 0
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        
        # Gradient: grad[i] = sum_k weights[k] * a[k-i] (for valid k-i)
        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)
        
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break
        
        if it > 20:
            T = max(T_min, T_init * (0.998 ** it))
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized solver for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using fast LP
alternating with golden-section line search, smooth-max gradient polishing,
multi-n exploration, and diverse restarts. Key innovation: exact 1D
optimization over the blend parameter via golden section search on the
convex function f(t)=max_k[(1-t)^2*A_k+2t(1-t)*B_k+t^2*C_k].
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross_conv(a, b):
    """Cross convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    total = float(np.sum(v))
    if total <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def golden_section_t(conv_a, cross, conv_g):
    """
    Find optimal t in [0,1] minimizing max_k of blended convolution:
    f(t) = max_k [(1-t)^2 * conv_a[k] + 2t(1-t) * cross[k] + t^2 * conv_g[k]]
    This function is convex in t, so golden section search is exact.
    """
    def f_t(t):
        one_minus_t = 1.0 - t
        vals = one_minus_t * one_minus_t * conv_a + 2.0 * t * one_minus_t * cross + t * t * conv_g
        return float(np.max(vals))

    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = f_t(c)
    fd = f_t(d)

    for _ in range(70):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = f_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = f_t(d)

    t_opt = (lo + hi) * 0.5
    return t_opt, f_t(t_opt)


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-15:
        return None, None
    a_norm = a / a_sum

    num_con = 2 * n - 1
    A_ub = np.zeros((num_con, n + 1))

    for k in range(num_con):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            A_ub[k, j] = a_norm[i]
        A_ub[k, n] = -1.0

    b_ub = np.zeros(num_con)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    bounds = [(0.0, None)] * n + [(-np.inf, None)]

    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def smooth_max_gradient_descent(a, time_budget=30.0, T_init=0.001, T_min=0.00003):
    """
    Gradient descent using log-sum-exp smooth approximation of max.
    Uses momentum and adaptive learning rate.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.008
    v = np.zeros(n)
    momentum = 0.9
    stall = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400.0, 400.0)
        exp_vals = np.exp(shifted)
        total_exp = float(np.sum(exp_vals))
        if total_exp < 1e-300:
            break
        weights = exp_vals / total_exp

        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - float(np.mean(grad))

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 60:
                lr *= 0.75
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-7:
                    break

        if it > 30:
            T = max(T_min, T_init * (0.9985 ** it))

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def random_init(n, rng, mode=0):
    """Generate a diverse random non-negative sequence normalized to sum 1."""
    if mode == 0:
        a = rng.dirichlet(np.ones(n))
    elif mode == 1:
        center = n // 2
        width = max(3, n // 5)
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
        a = a / np.sum(a)
    elif mode == 2:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        a = a / np.sum(a)
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 4:
        # Two-peak
        c1 = n // 3
        c2 = 2 * n // 3
        w = max(3, n // 10)
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - c1) ** 2) / (2.0 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2.0 * w ** 2))
        a = a / np.sum(a)
    else:
        a = np.ones(n) / n
    return a

                                           T_init=0.0005, T_min=0.00002)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Final Smooth: obj={best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 8:
        a_cp = chase_peak_descent(a_final, time_budget=min(remaining * 0.1, 12.0))
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  Final Chase: obj={best_score:.10f}")

    # Final validation with the actual evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
