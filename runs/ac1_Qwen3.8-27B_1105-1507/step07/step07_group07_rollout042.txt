Key improvements: momentum subgradient as main driver, frequent LP interleaving,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time


def next_pow2(m):

    return 2.0 * len(a) * float(np.max(conv))


def subgrad_conv(a, k):
    """Subgradient of conv(a,a)[k] w.r.t. a (vectorized)."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    if lo <= hi:

        grad[idx1] += a[k - idx1]
    m_lo = max(0, k - hi)
    m_hi = min(n - 1, k - lo)
    if m_lo <= m_hi:
        idx2 = np.arange(m_lo, m_hi + 1)
        grad[idx2] += a[k - idx2]
    return grad


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = float(np.sum(w))
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def build_conv_matrix_for_lp(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[0:k_end - k_start + 1]
    return A


def solve_lp(a_fixed, time_limit=30.0):
    """Solve LP: min M s.t. conv(a_fixed, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a_fixed)
    a_norm = a_fixed / np.sum(a_fixed)
    A = build_conv_matrix_for_lp(a_norm)

    # Try cvxpy solvers
    for solver_name in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver_name, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_obj_and_grad(x, tau):
    """Smoothed objective and gradient for a = exp(x) with sum normalization."""
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)

    # Gradient via cross-correlation
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0):
    """L-BFGS-B on smoothed objective."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    x = np.log(a) - np.log(a).max()

    taus = [5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        xv = x.copy()

        def fun(xv_in):
            fv, _ = smoothed_obj_and_grad(xv_in, tau)
            return fv

        def fgrad(xv_in):
            _, g = smoothed_obj_and_grad(xv_in, tau)
            return g

        try:
            res = minimize(fun, xv, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-11,
                                    'maxls': 50})
            if res.fun < fun(xv):
                x_new = res.x - res.x.max()
                a_new = np.exp(x_new)
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = x_new
        except Exception:

    return best_a


def momentum_subgrad_optimize(a_init, time_budget=60.0, verbose=False):
    """Momentum subgradient method with projection to simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.6
    eta = 0.5
    eta_min = 0.01 / n
    t = 0

    while time.monotonic() - start < time_budget:
        conv, k_star, max_val = (compute_conv(a), int(np.argmax(compute_conv(a))), 0)
        # Recompute properly
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = float(conv[k_star])
        obj = 2.0 * n * max_val
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        g = subgrad_conv(a, k_star)
        # Normalize gradient for stable step
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Momentum update
        if t == 0:
            v = np.zeros(n)
        v = momentum * v - eta * g / g_norm

        # Try the update
        a_new = project_simplex(a + v)
        obj_new = compute_obj(a_new)

        if obj_new < obj:
            a = a_new
            # Gradually decrease momentum
            momentum = max(0.8 * momentum + 0.2 * 0.9, 0.9)
        else:
            # Backtrack
            eta *= 0.7
            if eta < eta_min:
                # Perturb and reset
                rng = np.random.default_rng(int(time.monotonic() * 100000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = project_simplex(a)
                eta = 0.5
                v = np.zeros(n)
                momentum = 0.6
        t += 1

        # Adaptively decrease eta
        eta = max(eta * 0.9995, eta_min)

    return best_a


def lp_polish(a_init, time_budget=30.0, n_steps=5):

    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

        if time.monotonic() - start > time_budget:
            break

        if g_lp is None:
            break
        best_step = best_obj
        best_a_step = a.copy()
        for t in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_step:
                best_step = obj_try
                best_a_step = a_try.copy()
        if best_step < best_obj:
            best_obj = best_step
            best_a = best_a_step.copy()
            a = best_a_step
        else:
            break

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: momentum subgradient + frequent LP + gradient polish + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    best_seq = None
    best_score = np.inf
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"Starting optimization from n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Quick momentum subgradient (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Phase 1: Momentum Subgradient ===")
        t_budget = min(remaining * 0.2, 200.0)
        a1 = momentum_subgrad_optimize(best_seq, time_budget=t_budget)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
        print(f"  After phase 1: obj={best_score:.10f}")

    # === Phase 2: LP polish (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 2: LP Polish ===")
        t_budget = min(remaining * 0.12, 150.0)
        a2 = lp_polish(best_seq, time_budget=t_budget, n_steps=10)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
        print(f"  After phase 2: obj={best_score:.10f}")

    # === Phase 3: Gradient optimization (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 3: Gradient (L-BFGS-B) ===")
        t_budget = min(remaining * 0.15, 200.0)
        a3 = gradient_optimize(best_seq, time_budget=t_budget)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
        print(f"  After phase 3: obj={best_score:.10f}")

    # === Phase 4: Main alternation loop (500s) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Phase 4: Subgradient + LP Alternation ===")
        n_best = len(best_seq)
        a_cur = best_seq.copy()
        loop_start = time.monotonic()
        loop_budget = min(remaining * 0.35, 500.0)
        iteration = 0
        no_improve_count = 0

        while time.monotonic() - loop_start < loop_budget:
            # Subgradient step
            sub_budget = min(15.0, loop_budget - (time.monotonic() - loop_start))
            if sub_budget > 2:
                a_sub = momentum_subgrad_optimize(a_cur, time_budget=sub_budget)
                obj_sub = compute_obj(a_sub)
                if obj_sub < best_score:
                    best_score = obj_sub
                    best_seq = a_sub.copy()
                    a_cur = a_sub
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            # LP step (every 2-3 iterations)
            if iteration % 2 == 1:
                lp_budget = min(10.0, loop_budget - (time.monotonic() - loop_start))
                if lp_budget > 3:
                    a_lp = lp_polish(a_cur, time_budget=lp_budget, n_steps=2)
                    obj_lp = compute_obj(a_lp)
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        a_cur = a_lp
                        no_improve_count = 0
                    else:
                        no_improve_count += 1

            # Perturbation if stuck
            if no_improve_count > 4:
                scale = min(0.02 + 0.01 * no_improve_count, 0.15)
                a_cur = best_seq * np.exp(rng.normal(0, scale, n_best))
                a_cur = project_simplex(a_cur)
                no_improve_count = 0

            iteration += 1
            if iteration % 10 == 0:
                elapsed = time.monotonic() - loop_start
                print(f"    iter {iteration}, t={elapsed:.0f}s, best={best_score:.10f}")

        print(f"  After phase 4: obj={best_score:.10f}")

    # === Phase 5: Multi-n exploration (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 5: Multi-n Exploration ===")
        n_best = len(best_seq)
        n_candidates = sorted(set([
            max(300, n_best - 200), max(300, n_best - 100),
            n_best,
            min(1500, n_best + 100), min(1500, n_best + 200),
            500, 700, 900, 1000, 1100, 1200, 1300, 1500
        ]))
        n_candidates = [n for n in n_candidates if 300 <= n <= 1600]

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
                a_try = a_try * np.exp(rng.normal(0, 0.02, n_try))
                a_try = project_simplex(a_try)

            # Quick optimize: subgrad + short gradient
            alloc_sub = min(15.0, remaining * 0.06)
            alloc_grad = min(10.0, remaining * 0.04)
            a_opt = momentum_subgrad_optimize(a_try, time_budget=alloc_sub)
            a_opt = gradient_optimize(a_opt, time_budget=alloc_grad)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 6: Final intensive polish (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 6: Final Polish ===")
        # Gradient
        alloc = min(remaining * 0.3, 60.0)
        a_f = gradient_optimize(best_seq, time_budget=alloc)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final grad: obj={best_score:.10f}")

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc_lp = min(remaining * 0.3, 50.0)
            a_lp = lp_polish(best_seq, time_budget=alloc_lp, n_steps=8)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.10f}")

        # Subgradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc_sub = min(remaining * 0.3, 40.0)
            a_sub = momentum_subgrad_optimize(best_seq, time_budget=alloc_sub)
            obj = compute_obj(a_sub)
            if obj < best_score:
                best_score = obj
                best_seq = a_sub.copy()
                print(f"  Final subgrad: obj={best_score:.10f}")

        # Symmetric refinement
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sym = (best_seq + best_seq[::-1]) / 2.0
            a_sym = project_simplex(a_sym)
            obj_sym = compute_obj(a_sym)
            if obj_sym < best_score:
                best_score = obj_sym
                best_seq = a_sym.copy()
                print(f"  Symmetric: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
