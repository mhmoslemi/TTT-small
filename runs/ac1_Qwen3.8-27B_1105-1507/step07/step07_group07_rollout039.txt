"""
Hybrid optimizer for low-autocorrelation non-negative sequences.
Uses: (1) fast multiplicative gradient with top-m peak weighting,
(2) LP-based alternating with golden section search,
(3) spectral shaping initialization,
(4) multi-n exploration with gradient refinement.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=10.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Try cvxpy solvers
    for solver in [cp.CPX, cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            A = build_conv_matrix(a_norm)
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver, timeout=timeout)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        A = build_conv_matrix(a_norm)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': timeout})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def fast_gradient_step(a, top_m=10, eta=0.01):
    """
    One step of multiplicative gradient descent on the peak objective.
    Uses top-m peaks with exponential weighting.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = conv.max()
    
    # Find top-m peaks
    if top_m >= len(conv):
        top_indices = np.arange(len(conv))
        weights = np.ones(len(conv)) / len(conv)
    else:
        # Use exponential weighting based on peak values
        threshold = max_val * 0.95
        peak_indices = np.where(conv >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        elif len(peak_indices) > top_m:
            # Take top_m
            peak_vals = conv[peak_indices]
            top_idx_in_peak = np.argsort(peak_vals)[-top_m:]
            peak_indices = peak_indices[top_idx_in_peak]
        
        # Weight by how close to max
        peak_vals = conv[peak_indices]
        weights = np.exp((peak_vals - max_val) / (max_val * 0.1 + 1e-10))
        weights = weights / weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(n):
        w_sum = 0.0
        for k_idx, w in zip(peak_indices, weights):
            j = int(k_idx) - i
            if 0 <= j < n:
                w_sum += w * a[j]
        grad[i] = 2.0 * w_sum
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new, conv

def lp_alternate_with_golden(a, time_budget=30.0, verbose=False):

    best_obj = compute_obj(a_norm)
    start = time.monotonic()
    
    # Golden ratio

    stall = 0
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a_norm, timeout=min(5.0, time_budget - (time.monotonic() - start)))
        if g_lp is None:
            break
        
        # Golden section search for optimal t
        lo, hi = 0.0, 1.0
        c1 = lo + gr * (hi - lo)
        c2 = hi - gr * (hi - lo)
        
        a_c1 = (1 - c1) * a_norm + c1 * g_lp
        a_c1 = np.maximum(a_c1, 0)
        a_c1 = a_c1 / a_c1.sum()

                a_c2 = a_c1
                obj_c2 = obj_c1
                c1 = lo + gr * (hi - lo)

def metabolic_gradient(a_init, time_budget=60.0, verbose=False):

    the reduction effort proportionally.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive temperature for peak selection
    tau = 0.1 / n
    eta = 0.5 / n
    
    while time.monotonic() - start < time_budget:

        # Soft peak selection

        total = exp_conv.sum()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

def spectral_init(n, rng):
    """
    Initialize using spectral methods.
    Design a sequence whose autocorrelation has desired properties.
    """
    mode = rng.integers(0, 5)
    
    if mode == 0:
        # Raised cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif mode == 1:
        # Tukey window (alpha=0.5)
        alpha = 0.5
        a = np.ones(n)
        m = int(alpha * n / 2)
        for i in range(m):
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * n) - 1)))
            a[n - 1 - i] = a[i]
    elif mode == 2:
        # Gaussian window
        center = (n - 1) / 2
        sigma = n / 4
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * sigma ** 2))
    elif mode == 3:
        # Optimal spectral: flat power spectrum (inverse of flat autocorrelation)
        # Start with flat power spectrum, take IFFT, take absolute value
        size = next_pow2(2 * n - 1)
        # Random phase with flat magnitude
        phases = rng.uniform(0, 2 * np.pi, size // 2 + 1)
        spectra = np.exp(1j * phases)
        # Make it Hermitian for real output
        if size % 2 == 0:
            full_spectrum = np.zeros(size, dtype=complex)
            full_spectrum[:size // 2 + 1] = spectra
            full_spectrum[size // 2 + 1:] = np.conj(spectra[1:size // 2][::-1])
        else:
            full_spectrum = np.zeros(size, dtype=complex)
            full_spectrum[:size // 2 + 1] = spectra
            full_spectrum[size // 2 + 1:] = np.conj(spectra[1:(size - 1) // 2 + 1][::-1])
        a_circ = np.real(ifft(full_spectrum))[:n]
        a = np.abs(a_circ)
    else:
        # Bump function
        x = np.linspace(0, 1, n)
        a = (x * (1 - x) ** 2)
    
    a = np.maximum(a, 0.01)
    a = a + 0.05 * rng.uniform(0, 1, n)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_polish(a, time_budget=20.0, verbose=False):
    """
    Coordinate descent: try adjusting one element at a time.
    For each element, find the best value via ternary search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    rng = np.random.default_rng(42)
    order = rng.permutation(n)
    
    idx = 0
    no_improve = 0
    while time.monotonic() - start < time_budget:
        i = order[idx % n]
        idx += 1
        
        # Save original
        a_orig = a[i]
        s_orig = a.sum()
        
        # The objective as a function of a[i] (with others fixed):
        # conv(a,a)[k] changes for k = i (add a[i]^2) and k = i+j for all j != i
        # This is complex, so just try a few values
        
        # Try: increase, decrease, zero
        candidates = [0.0, a_orig * 0.5, a_orig * 0.8, a_orig * 1.2, a_orig * 1.5]
        candidates = [c for c in candidates if c >= 0]
        
        best_val = a_orig
        best_local = compute_obj(a)
        
        for val in candidates:
            a[i] = val
            s = a.sum()
            if s < 0.01:
                continue
            obj_try = compute_obj(a)
            if obj_try < best_local:
                best_local = obj_try
                best_val = val
        
        a[i] = best_val
        a = a / a.sum()
        
        if best_local < best_obj - 1e-12:
            best_obj = best_local
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > n * 2:
                break
    
    return a

"""
Hybrid optimizer for low-autocorrelation non-negative sequences.
Uses: (1) metabolic gradient (soft peak weighting with FFT-based gradient, multiplicative updates),
(2) LP-based alternating with golden section search,
(3) fast top-m peak multiplicative gradient,
(4) spectral shaping initialization,
(5) multi-n exploration, (6) coordinate descent polish.

import numpy as np
from scipy.fft import rfft, irfft, ifft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=8.0):
    n = len(a)
    a_norm = a / np.sum(a)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.SCS]:
        try:
            A = build_conv_matrix(a_norm)
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver, timeout=timeout)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        A = build_conv_matrix(a_norm)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': timeout})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def metabolic_gradient(a_init, time_budget=60.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = 0.15 / n
    eta = 0.8 / n
    min_eta = 0.005 / n
    max_eta = 2.0 / n
    min_tau = 0.005 / n
    stall = 0
    max_stall = 8000

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = conv.max()

        exp_conv = np.exp((conv - max_conv) / tau)
        total = exp_conv.sum()
        if total < 1e-30:
            break
        w = exp_conv / total

        size = next_pow2(3 * n - 2)
        a_flip = a[::-1]
        fa_flip = rfft(a_flip, size)
        fw = rfft(w, size)
        corr_full = irfft(fw * fa_flip, size)
        c = corr_full[n - 1:n - 1 + n]

        max_c = c.max()
        if max_c < 1e-30:
            break

        a_new = a * np.exp(-eta * c / tau * (tau * n / 2.0 + 1e-10))
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            tau = max(tau * 0.9995, min_tau)
            eta = min(eta * 1.001, max_eta)
        else:
            stall += 1
            eta *= 0.95
            tau = min(tau * 1.005, 0.5 / n)
            if eta < min_eta:
                if stall > max_stall:
                    if verbose:
                        print(f"    Metabolic converged at it={it}, obj={best_obj:.10f}")
                    break
                rng = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                eta = 0.5 / n
                tau = 0.1 / n
                stall = 0

        it += 1
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Metabolic it={it}, obj={best_obj:.10f}, tau={tau:.2e}, eta={eta:.2e}, t={elapsed:.1f}s")

    return best_a


def fast_mg(a_init, time_budget=60.0, top_m=10, verbose=False):

    min_eta = 0.01 / n
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()

        threshold = max_val * 0.97
        peak_indices = np.where(conv >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        elif len(peak_indices) > top_m:
            peak_vals = conv[peak_indices]
            top_idx = np.argsort(peak_vals)[-top_m:]
            peak_indices = peak_indices[top_idx]

        peak_vals = conv[peak_indices]
        w = np.exp((peak_vals - max_val) / (max_val * 0.05 + 1e-10))
        w = w / w.sum()

        size = next_pow2(3 * n - 2)
        a_flip = a[::-1]
        fa_flip = rfft(a_flip, size)
        fw = rfft(w, size)
        corr_full = irfft(fw * fa_flip, size)
        grad = corr_full[n - 1:n - 1 + n]

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0

                    break
                rng = np.random.default_rng(it + 99)
                a = best_a * np.exp(rng.normal(0, 0.02, n))

                a = a / a.sum()
                eta = 1.0 / n
                stall = 0

        it += 1
        if verbose and it % 10000 == 0:

def lp_alternate_golden(a, time_budget=30.0, verbose=False):

        a_c1 = np.maximum(a_c1, 0)

                c1 = lo + gr * (hi - lo)

def coordinate_polish(a, time_budget=20.0):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_obj = compute_obj(a)
    start = time.monotonic()

    rng = np.random.default_rng(7)
    order = rng.permutation(n)
    idx = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        i = order[idx % n]
        idx += 1
        a_orig = a[i]

        for factor in [0.3, 0.6, 0.8, 1.2, 1.5, 2.0, 0.0]:
            a[i] = a_orig * factor
            s = a.sum()
            if s < 0.01:
                a[i] = a_orig
                continue
            obj_try = compute_obj(a)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                no_improve = 0
                break
        else:
            a[i] = a_orig
            no_improve += 1
            if no_improve > n:
                break

        a = a / a.sum()

    return a


def spectral_init(n, rng):
    mode = rng.integers(0, 6)

    if mode == 0:
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif mode == 1:
        alpha = 0.5
        a = np.ones(n)
        m = max(1, int(alpha * n / 2))
        for i in range(m):
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * n) - 1)))
            a[n - 1 - i] = a[i]
    elif mode == 2:
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * sigma ** 2))
    elif mode == 3:
        size = next_pow2(2 * n - 1)
        phases = rng.uniform(0, 2 * np.pi, size // 2 + 1)
        full_spectrum = np.zeros(size, dtype=complex)
        full_spectrum[:size // 2 + 1] = np.exp(1j * phases)
        if size % 2 == 0:
            full_spectrum[size // 2 + 1:] = np.conj(full_spectrum[1:size // 2][::-1])
        else:
            full_spectrum[size // 2 + 1:] = np.conj(full_spectrum[1:(size - 1) // 2 + 1][::-1])
        a = np.abs(np.real(ifft(full_spectrum)))[:n]
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = (x * (1 - x) ** 1.5) + 0.05
    else:
        a = np.ones(n)
        for _ in range(3):
            center = rng.integers(n // 4, 3 * n // 4)
            width = rng.integers(n // 8, n // 3)
            a += np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))

    a = np.maximum(a, 0.01)
    a += 0.05 * rng.uniform(0, 1, n)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
        elif r == 1:
            a_r = spectral_init(n_best, rng)
        elif r == 2:
            n_r = max(400, min(1400, n_best + int(rng.integers(-200, 200))))
            a_r = spectral_init(n_r, rng)
        elif r == 3:
            a_r = best_seq + 0.02 * rng.normal(0, 1, n_best)
            a_r = (a_r + a_r[::-1]) / 2.0
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
        elif r == 4:
            n_r = max(400, min(1400, n_best + int(rng.integers(-100, 100))))
            a_r = resample_sequence(best_seq, n_r)
            a_r *= np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
        else:
            a_r = rng.uniform(0.1, 1.0, n_best)
            kernel = np.ones(5) / 5.0
            a_r = np.convolve(a_r, kernel, mode='same')
            a_r = a_r / a_r.sum()

        alloc = min(35.0, remaining * 0.1)
        a_opt = metabolic_gradient(a_r, time_budget=alloc)
        obj = compute_obj(a_opt)

        remaining2 = deadline - time.monotonic()
        if remaining2 > 25 and obj < best_score + 0.005:
            a_opt2 = lp_alternate_golden(a_opt, time_budget=min(20.0, remaining2 * 0.3))
            obj2 = compute_obj(a_opt2)
            if obj2 < obj:
                obj = obj2
                a_opt = a_opt2

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 25:
        a_f = metabolic_gradient(best_seq, time_budget=min(remaining * 0.3, 50.0), verbose=True)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final metabolic: obj={best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp = lp_alternate_golden(best_seq, time_budget=min(remaining * 0.3, 35.0))
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_mg = fast_mg(best_seq, time_budget=min(remaining * 0.3, 25.0), top_m=30)
        obj = compute_obj(a_mg)
        if obj < best_score:
            best_score = obj
            best_seq = a_mg.copy()
            print(f"  Final MG: obj={best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd = coordinate_polish(best_seq, time_budget=min(remaining * 0.4, 15.0))
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  Final coord: obj={best_score:.10f}")

    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
