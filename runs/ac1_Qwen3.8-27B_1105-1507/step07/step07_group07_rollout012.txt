"""
Optimized search for low-autocorrelation non-negative sequences.
Key improvements: better non-symmetric initializations, Adam optimizer,
multi-n exploration, and refined temperature annealing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_obj_and_grad(x, tau):
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    # Gradient
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad

def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    taus = [20.0/n, 10.0/n, 5.0/n, 2.0/n, 1.0/n, 0.5/n, 0.2/n, 0.1/n, 0.05/n]
    x = np.log(a) - np.log(a).max()
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        time_per_tau = (time_budget - (time.monotonic() - start)) / (len(taus) - taus.index(tau) + 1)
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g
        
        maxiter = max(50, int(time_per_tau * 200))  # Estimate iterations
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxls': 100})
        
        x_opt = res.x - res.x.max()
        a_new = np.exp(x_opt)
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            x = x_opt
            if verbose:
                print(f"    tau={tau:.5f}: obj={best_obj:.8f} (improved)")
    
    return best_a

and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(x, tau):
    """
    Compute smoothed objective and gradient for a = exp(x) / sum(exp(x)).
    Uses logsumexp softening of the max convolution.
    """
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    
    conv = compute_conv(a)
    
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    
    # Gradient: dF/dx_i
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Optimize using L-BFGS-B on smoothed objective with adaptive multi-scale temperature.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Fine-grained temperature schedule
    num_taus = min(14, max(7, int(time_budget / 8)))
    tau_max = 30.0 / n
    tau_min = 0.03 / n
    ratios = np.linspace(np.log(tau_max), np.log(tau_min), num_taus)
    taus = np.exp(ratios)
    
    x = np.log(a) - np.log(a).max()
    
    for ti, tau in enumerate(taus):
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 3:
            break
        
        time_left_taus = time_budget - time.monotonic()
        time_per_tau = time_left_taus / (num_taus - ti)
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g
        
        # Estimate reasonable maxiter based on time
        # Each iteration takes roughly 2 FFTs of size ~4n
        approx_iter_time = max(0.005, 4 * n / 1e7)
        maxiter = max(30, int(time_per_tau * 0.8 / approx_iter_time))
        maxiter = min(maxiter, 2000)
        
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13,
                                'maxls': 200})
        
        x_opt = res.x - res.x.max()
        a_new = np.exp(x_opt)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            x = x_opt
            if verbose:
                print(f"    grad tau={tau:.6f} (iter={ti+1}/{num_taus}): obj={best_obj:.10f}")
        else:
            if verbose:
                print(f"    grad tau={tau:.6f} (iter={ti+1}/{num_taus}): obj={obj_new:.10f} (no improve)")
    
    return best_a


def poly_init(n, degree, rng):
    """Generate a polynomial-parameterized non-negative sequence."""
    t = np.linspace(0, 1, n)
    
    if degree == 1:
        # Linear ramp with random slope direction
        if rng.random() < 0.5:
            a = t + rng.uniform(0.1, 0.5)
        else:
            a = (1 - t) + rng.uniform(0.1, 0.5)
    elif degree == 2:
        # Quadratic: random vertex position
        c = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.3, 1.0)
        a = np.exp(-((t - c) / width) ** 2)
        # Add asymmetry
        a = a * (1 + rng.uniform(0, 0.5) * (t - 0.5))
        a = np.maximum(a, 0.01)
    elif degree == 3:
        # Cubic with random coefficients
        c0 = rng.uniform(0.5, 2.0)
        c1 = rng.uniform(-1.5, 1.5)
        c2 = rng.uniform(-1.0, 1.0)
        c3 = rng.uniform(-0.5, 0.5)
        a = c0 + c1 * t + c2 * t * t + c3 * t * t * t
        a = np.maximum(a, 0.01)
    elif degree == 4:
        # Quartic
        coeffs = rng.uniform(0.5, 2.0, 5)
        coeffs[1:] = coeffs[1:] * rng.uniform(-1, 1, 4)
        a = np.zeros(n)
        for i in range(5):
            a += coeffs[i] * t ** i
        a = np.maximum(a, 0.01)
    else:
        # Higher degree via Bernstein basis for guaranteed non-negativity
        k = degree + 1
        control = rng.uniform(0.1, 1.0, k)
        # Evaluate using de Casteljau
        for _ in range(degree):
            control = (1 - t / degree) * control[:-1] + (t / degree) * control[1:]
            degree -= 1
            if degree == 0:
                break
        a = control if isinstance(control, np.ndarray) and control.shape == (n,) else np.full(n, control)
        a = np.maximum(a, 0.01)
    
    return a / np.sum(a)


def chirp_init(n, rng):
    """Rectified chirp sequence."""
    t = np.linspace(0, 1, n)
    freq = rng.uniform(2, 8)
    phase = rng.uniform(0, 2 * np.pi)
    z = np.exp(2j * np.pi * (freq * t ** 2 / 2 + phase))
    a = np.real(z) + rng.uniform(0.5, 1.5)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def dual_peak_init(n, rng):
    """Two Gaussian peaks with random positions and widths."""
    t = np.linspace(0, 1, n)
    c1 = rng.uniform(0.1, 0.4)
    c2 = rng.uniform(0.6, 0.9)
    w1 = rng.uniform(0.05, 0.15)
    w2 = rng.uniform(0.05, 0.15)
    a = np.exp(-((t - c1) / w1) ** 2) + np.exp(-((t - c2) / w2) ** 2)
    # Add small background
    a += 0.05 * rng.uniform(0, 1)
    return a / np.sum(a)


def random_smooth_init(n, rng):
    """Random sequence smoothed by low-pass filtering."""
    a = rng.uniform(0, 1, n)
    # Apply boxcar smoothing
    kernel_size = rng.integers(3, 11) | 1  # odd
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def subgradient_step(a, steps=100, eta_init=0.1, verbose=False):
    """
    Direct subgradient descent on the max-convolution objective.
    For each step, find the peak lag and push probability mass away.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = eta_init
    
    for step in range(steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max_k conv[k] w.r.t. a[i]:
        # d conv[k]/d a[i] = a[k-i] if 0 <= k-i < n, else 0
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
        
        # Also account for sum normalization: subtract mean
        grad = grad - np.sum(grad) * a / np.sum(a)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s = float(np.sum(a_new))
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = max(eta * 0.995, 0.001 / n)
        else:
            eta *= 0.8
            if eta < 0.005 / n:
                break
    
    return best_a
