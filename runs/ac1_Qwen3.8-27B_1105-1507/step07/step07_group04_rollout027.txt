s.t. sum(u) = 1
     A^T u + v*1 >= 0

s.t. sum(u) = 1

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for r, c, v in zip(rows, cols, vals):
        A[r, c] = v
    return A.tocsr()

def solve_lp_given_a(a):

"""
High-performance optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP alternating with HiGHS solver, adaptive step sizes,
structured perturbations, and Nesterov-accelerated subgradient polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 2 * n - 1
    size = 1 << (size - 1).bit_length()
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        # a is "centered" at index i in the convolution
        # conv[k] gets contribution a[k-i] * g[i] for valid k-i
        # k-i ranges from 0 to n-1, so k ranges from i to i+n-1
        # A[k, i] = a[k-i] for k in [i, i+n-1]
        k_start = i
        k_end = min(m, i + n)
        A[k_start:k_end, i] = a[:k_end - k_start]
    return A

def solve_lp(a, verbose=False):
    """Solve min M s.t. A(a)@g <= M, sum(g)=1, g>=0. Returns (g, M, u)."""
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            g = g / gs
            M = float(res.x[n])
            # Get dual variables for A_ub constraints
            u = None
            if res.ineqlin is not None:
                u = np.maximum(res.ineqlin[:m] if len(res.ineqlin) >= m else np.zeros(m), 0)
            return g, M, u
    return None, None, None
