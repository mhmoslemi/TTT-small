        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_obj_and_grad(x, tau):
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    # Gradient via cross-correlation
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad

def gradient_optimize_adaptive(a_init, time_budget=60.0, verbose=False):
    """Adaptive temperature L-BFGS with restarts."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = np.log(a) - np.log(a).max()
    
    # Adaptive temperature schedule
    tau = 5.0 / n
    min_tau = 0.01 / n
    cool_rate = 0.95
    stall_count = 0
    max_stall = 10
    it_count = 0
    
    while time.monotonic() - start < time_budget:
        it_count += 1
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g
        
        obj_before = fun(x)
        
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxls': 30})
        
        x_new = res.x - res.x.max()
        a_new = np.exp(x_new)
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            x = x_new
            stall_count = 0
            # Cool down
            tau = max(tau * cool_rate, min_tau)
            if verbose:
                print(f"    iter {it_count}: tau={tau:.6f}, obj={best_obj:.10f}")
        else:
            stall_count += 1
            if stall_count >= max_stall:
                # Restart temperature
                tau = max(5.0 / n, tau * 10)
                # Perturb
                rng = np.random.default_rng(it_count * 7919)
                x = x + rng.normal(0, 0.05, n)
                stall_count = 0
            else:
                # Try smaller step
                tau = max(tau * 0.8, min_tau)
    
    return best_a

def subgradient_polish(a, time_budget=30.0, verbose=False):
    """Subgradient method with averaging for the unsmoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy() / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Polyaku-Shapiro step size
    step = 1.0
    avg_a = a.copy() * step
    total_weight = step
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_b = conv[k_star]
        s = float(np.sum(a))
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        grad = 2.0 * n * grad / (s * s) - 4.0 * n * max_b / (s ** 3)
        
        # Step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                # Decrease step
                step *= 0.99
            else:
                step *= 0.9
                if step < 1e-4:
                    break
        else:
            step *= 0.9
    
    return best_a

def pairs_exchange_search(a, time_budget=20.0, verbose=False):
    """Local search: exchange amounts between pairs to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).copy() / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    epsilon = 0.01
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(best_a)
        k_star = int(np.argmax(conv))
        
        # Try exchanges involving the active lag
        improved = False
        n_tries = min(50, max(10, n // 10))
        
        for _ in range(n_tries):
            i = rng.integers(0, n)
            j = k_star - i
            if j == i or j < 0 or j >= n:
                continue
            
            # Transfer from larger to smaller
            if best_a[i] >= best_a[j]:
                donor = i
                receiver = j
            else:
                donor = j
                receiver = i
            
            transfer = min(epsilon * best_a[donor], best_a[donor] * 0.1)
            a_try = best_a.copy()
            a_try[donor] -= transfer
            a_try[receiver] += transfer
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
        
        if not improved:
            # Try random pairs
            for _ in range(n_tries):
                i, j = rng.integers(0, n, 2)
                if i == j:
                    continue
                if best_a[i] >= best_a[j]:
                    donor, receiver = i, j
                else:
                    donor, receiver = j, i
                transfer = min(epsilon * best_a[donor], best_a[donor] * 0.05)
                a_try = best_a.copy()
                a_try[donor] -= transfer
                a_try[receiver] += transfer
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))

                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True

            # Decrease epsilon
            epsilon *= 0.9
            if epsilon < 1e-6:

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def sinusoidal_init(n, n_harmonics, rng):
    """Generate a sequence as a sum of sinusoids."""
    base = np.ones(n) / n
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0, 0.3) / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        base += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    base = np.maximum(base, 0.01)
    base = base / np.sum(base)
    return base

                    # dF/da[i] = 0 => a[i] = S_others - C/(2*a[j])... 
                    # Let me redo: f(x) = (2*x*a[j] + C) / (x + S_others)^2

                    # Set = 0: a[j] = (2*x*a[j]+C)/(x+S_others)
                    # a[j]*(x+S_others) = 2*x*a[j] + C
                    # a[j]*x + a[j]*S_others = 2*x*a[j] + C
                    # a[j]*S_others - C = x*a[j]
                    # x = S_others - C/a[j]
                    x_opt = S_others - C / a[j]
                    if x_opt > 0:
                        candidates.append(x_opt)
            
            # Also try a perturbed value
            candidates.append(a[i] * 1.1)
            candidates.append(a[i] * 0.9)
            
            best_val = best_obj  # Will be updated if we find improvement
            best_candidate = a[i]
            
            for x_try in candidates:
                x_try = max(0.0, x_try)
                a[i] = x_try
                # Recompute b (affected only)
                # For efficiency, just recompute full conv (n=800 is fast)
                b_new = compute_conv(a)
                obj_new = 2.0 * n * float(np.max(b_new)) / (float(np.sum(a))**2)
                if obj_new < best_val:
                    best_val = obj_new
                    best_candidate = x_try
                    k_star = int(np.argmax(b_new))
            
            a[i] = best_candidate
            if best_candidate != a[i]:  # This won't happen since we just set it
                improved = True
        
        # Check overall improvement
        obj_check = compute_obj(a)
        if obj_check < best_obj:
            best_obj = obj_check
            best_a = a.copy()
            improved = True
    
    return best_a

old: b[k] includes a[i]^2

For i ≠ k-i:
old: b[k] includes 2*a[i]*a[k-i]
new: b[k] includes 2*a_new[i]*a[k-i]
