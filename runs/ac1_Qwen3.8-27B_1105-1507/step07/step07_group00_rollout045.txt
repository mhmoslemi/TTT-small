"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS with LSE-smoothed objective in log-space, improved LP alternating,
Adam optimizer, and multi-n exploration. Key innovations: temperature annealing,
log-parameterization for unconstrained optimization, and dual-guided perturbation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
from scipy.optimize import linprog, minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_conv_and_grad(a, tau):
    """
    Compute LSE-smoothed convolution peak and gradient.
    Returns (lse_value, gradient) where gradient is df/da for f = 2n*tau*log(sum(exp(conv/tau))).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    exp_c = np.exp((conv - max_c) / tau)
    total = np.sum(exp_c)
    lse_val = 2.0 * n * (max_c + tau * np.log(total))
    weights = exp_c / total
    # Gradient: g[i] = 2 * sum_j weights[i+j] * a[j]
    g = 2.0 * correlate(weights, a, mode='valid')[:n]
    return lse_val, g


def log_space_obj_grad(x, tau):
    """
    Objective and gradient in log-space.
    x is the log-parameter: a = softmax(x).
    Returns (obj, grad_x).
    """
    # Convert to probability vector
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)

    # Compute LSE smoothed objective and gradient w.r.t. a
    obj, g_a = lse_conv_and_grad(a, tau)

    # Convert gradient to log-space: df/dx[i] = a[i] * (g_a[i] - dot(a, g_a))
    mean_g = float(np.dot(a, g_a))
    g_x = a * (g_a - mean_g)

    return obj, g_x


def lbfgs_lse_optimize(a_init, time_budget=300.0, verbose=False):
    """
    L-BFGS optimization in log-space with LSE smoothing and temperature annealing.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Temperature schedule: start high, anneal down
    tau_schedule = [1.0 / n * t for t in [50, 30, 20, 15, 10, 8, 6, 5, 4, 3, 2, 1.5, 1]]

    # Initial log-space
    x = np.log(np.maximum(a, 1e-15))

    for tau in tau_schedule:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Time per tau level
        time_per_tau = min(remaining * 0.6, 80.0)

        try:
            res = minimize(
                log_space_obj_grad,
                x,
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 500, 'maxfun': 5000, 'ftol': 1e-12, 'gtol': 1e-10,
                         'maxls': 50}
            )
            x = res.x
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error at tau={tau:.4f}: {e}")
            continue

        # Convert back
        x_shifted = x - np.max(x)
        exp_x = np.exp(x_shifted)
        a_new = exp_x / np.sum(exp_x)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        if verbose:
            elapsed = time.monotonic() - start
            print(f"      LSE L-BFGS tau={tau:.4f}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

def adam_logspace_optimize(a_init, time_budget=200.0, verbose=False):
    """
    Adam optimizer in log-space with LSE smoothing.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = np.log(np.maximum(a, 1e-15))
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0

    tau_schedule = [1.0 / n * tt for tt in [30, 20, 15, 10, 8, 6, 5, 4, 3, 2, 1.5, 1, 0.8, 0.6, 0.4]]
    lr_schedule = [0.05, 0.04, 0.03, 0.02, 0.02, 0.015, 0.01, 0.008, 0.006, 0.004, 0.003, 0.002, 0.001, 0.0008, 0.0005]

    stall = 0
    for idx, (tau, lr) in enumerate(zip(tau_schedule, lr_schedule)):
        tau_remaining = time_budget - (time.monotonic() - start)
        if tau_remaining < 5:
            break

        # Run Adam for a fixed number of steps or until time runs out
        steps_per_tau = 100
        for _ in range(steps_per_tau):
            if time.monotonic() - start > time_budget - 5:
                break
            t += 1
            obj, g_x = log_space_obj_grad(x, tau)

            m = beta1 * m + (1 - beta1) * g_x
            v = beta2 * v + (1 - beta2) * g_x ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Evaluate
            x_shifted = x - np.max(x)
            exp_x = np.exp(x_shifted)
            a_try = exp_x / np.sum(exp_x)
            obj_try = compute_obj(a_try)

            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                stall = 0
            else:
                stall += 1
                if stall > 200:
                    # Perturb
                    rng = np.random.default_rng(t * 31 + n)
                    x = x + rng.normal(0, 0.1, n)
                    stall = 0

        if verbose:
            elapsed = time.monotonic() - start
            print(f"      Adam log tau={tau:.4f}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        C[k_start:k_end, j] = a[:(k_end - k_start)]
    return C


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    C = build_conv_matrix(a)

    # Use linprog with highs
    try:
        # Variables: g[0..n-1], M
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = C
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    # Try cvxpy
    try:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [C @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    return None, None

    iter_count = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget - 5:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Backtracking line search
        best_t_obj = best_obj
        best_t_a = best_a.copy()
        t = last_good_t
        for _ in range(20):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                t *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-12:
                best_t_obj = obj_try
                best_t_a = a_try.copy()
                last_good_t = t
                break
            t *= 0.7

        iter_count += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_a
            stall = 0
            if verbose and iter_count % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"      LP imp it{iter_count}: obj={best_obj:.8f}, t={last_good_t:.3f}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 20:
                # Adaptive perturbation based on dual
                rng = np.random.default_rng(iter_count * 104729 + n)
                # Perturb more where the sequence is "confident"
                scale = 0.02 + 0.03 * min(stall / 50, 1.0)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                # Also try a "flattening" perturbation
                a_flat = a * (1.0 + 0.1 * np.sin(2 * np.pi * rng.uniform(1, 5, 1)[0] * np.arange(n) / n + rng.uniform(0, 2*np.pi, 1)[0]))
                a_flat = np.maximum(a_flat, 0)
                s_flat = float(np.sum(a_flat))
                if s_flat > 1e-10:
                    a_flat = a_flat / s_flat
                    if compute_obj(a_flat) < compute_obj(a):
                        a = a_flat
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                last_good_t = rng.uniform(0.2, 0.8)

    return best_a


def chase_peak_refine(a_init, time_budget=60.0):
    """
    Chase-the-peak refinement with adaptive step size.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05 / np.sqrt(n)
    prev_a = a.copy()
    prev_g = None

    while time.monotonic() - start < time_budget - 5:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient for peak at k_star
        g = np.zeros(n)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            j_start = k_star - i_end
            j_end = k_star - i_start
            g[i_start:i_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        # Barzilai-Borwein
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-15:
                bb_step = ss / abs(sy)
                step_size = min(max(1e-6, step_size * 0.7), min(bb_step, 1.0))
        prev_a = a.copy()
        prev_g = g.copy()

        # Try multiple step sizes
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1]:
            a_new = a - step_size * mult * g
            # Project to simplex
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if not improved:
            step_size *= 0.7
            if step_size < 1e-8:
                break

    return best_a


def coordinate_refine(a_init, time_budget=30.0):
    """
    Coordinate descent: move mass from peak-contributing to non-contributing positions.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    while time.monotonic() - start < time_budget - 5:
        k_star = int(np.argmax(conv))
        improved = False

        # Get contributors to peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib = 2.0 * a[i] * a[j]
                contributors.append((contrib, i, j))
        contributors.sort(reverse=True)

        # Non-contributors
        contrib_indices = set()
        for _, i, j in contributors:
            contrib_indices.add(i)
            contrib_indices.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_indices]

        if not non_contrib:
            break

        for contrib, i, j in contributors[:30]:
            if time.monotonic() - start > time_budget - 5:
                break
            if contrib < 1e-10:
                break
            for target in non_contrib[:10]:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = frac * a[i]
                    if delta <= 0 or a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Periodic pattern
        period = max(2, n // 10)
        a = np.zeros(n)
        for i in range(n):
            a[i] = 1.0 + 0.3 * np.cos(2 * np.pi * i / period)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining L-BFGS, LP alternating, Adam, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: L-BFGS with LSE annealing ===
    print(f"\n=== Phase 1: L-BFGS LSE (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        p1_budget = min(remaining * 0.25, 350.0)
        a_lbfgs = lbfgs_lse_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_lbfgs = compute_obj(a_lbfgs)
        if obj_lbfgs < best_score:
            best_score = obj_lbfgs
            best_seq = a_lbfgs.copy()
            n_current = len(best_seq)
            print(f"  L-BFGS LSE: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS LSE: obj={obj_lbfgs:.8f} (no improvement)")

    # === Phase 2: LP Alternating (improved) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p2_budget = min(remaining * 0.3, 400.0)
        # Run multiple rounds
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / (4 - rnd) * 0.8, remaining * 0.4)
            if alloc < 20:
                break
            a_lp = lp_alternating_improved(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                n_current = len(best_seq)
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Adam in log-space ===
    print(f"\n=== Phase 3: Adam Log-Space ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.15, 150.0)
        a_adam = adam_logspace_optimize(best_seq, time_budget=p3_budget, verbose=True)
        obj_adam = compute_obj(a_adam)
        if obj_adam < best_score:
            best_score = obj_adam
            best_seq = a_adam.copy()
            n_current = len(best_seq)
            print(f"  Adam: obj={best_score:.8f} **")
        else:
            print(f"  Adam: obj={obj_adam:.8f} (no improvement)")

    # === Phase 4: Chase peak + coordinate refinement ===
    print(f"\n=== Phase 4: Chase Peak + Coord ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_cp = chase_peak_refine(best_seq, time_budget=min(remaining * 0.15, 60.0))
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  Chase peak: obj={best_score:.8f} **")
        else:
            print(f"  Chase peak: obj={obj_cp:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_cd = coordinate_refine(best_seq, time_budget=min(remaining * 0.1, 30.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord: obj={best_score:.8f} **")
        else:
            print(f"  Coord: obj={obj_cd:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and n != n_best]
    n_options += [n_best]  # Include current

    for n_try in n_options[:10]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick L-BFGS
        lbfgs_alloc = min(remaining * 0.15, 60.0)
        if lbfgs_alloc > 15:
            a_opt = lbfgs_lse_optimize(a_try, time_budget=lbfgs_alloc)
            obj_opt = compute_obj(a_opt)
            if obj_opt < compute_obj(a_try):
                a_try = a_opt

        # Quick LP
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.1, 40.0)
        if lp_alloc > 15:
            a_lp_n = lp_alternating_improved(a_try, time_budget=lp_alloc)
            obj_lp_n = compute_obj(a_lp_n)
            if obj_lp_n < compute_obj(a_try):
                a_try = a_lp_n

        obj_final = compute_obj(a_try)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_final:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_restarts = 5

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r % 8
        a_r = diverse_init(len(best_seq), rng, mode=mode)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Optimize
        restart_budget = min(remaining * 0.15, 60.0)
        a_opt_r = lp_alternating_improved(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        a_p1 = lbfgs_lse_optimize(best_seq, time_budget=min(remaining * 0.25, 60.0))
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_p2 = lp_alternating_improved(best_seq, time_budget=min(remaining * 0.3, 80.0), verbose=True)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_p3 = chase_peak_refine(best_seq, time_budget=min(remaining * 0.3, 40.0))
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_score:
            best_score = obj_p3
            best_seq = a_p3.copy()
            print(f"  Final CP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_p4 = coordinate_refine(best_seq, time_budget=min(remaining * 0.3, 25.0))
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_score:
            best_score = obj_p4
            best_seq = a_p4.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
