import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating (accelerated), energy-based multiplicative descent,
Gaussian smoothing perturbations, multi-n exploration, and coordinate
refinement. Key improvements over previous: faster LP via pre-built
Toeplitz matrix, adaptive step scheduling, energy descent hybrid,
and structured perturbations for escaping local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build Toeplitz matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Try cvxpy first (usually fastest for this size)
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in ['CLARABEL', 'MOSEK', 'SCS', 'ECOS']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    total = np.sum(v)
    if total <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lp_alternating(a, time_budget=60.0, verbose=False):
    """
    LP alternating with adaptive step size and stall detection.
    Uses geometric step sizes and resets on stall.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_current = 0.3
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        g, M = solve_lp_direction(a)
        if g is None:
            break

        # Try geometric step sizes around current best
        candidates = sorted(set([
            max(0.01, t_current * 0.5),
            max(0.01, t_current * 0.75),
            t_current,
            min(1.0, t_current * 1.25),
            min(1.0, t_current * 1.5),
            min(1.0, t_current * 2.0),
        ]))

        best_local_obj = best_obj
        best_local_a = a
        best_t = t_current

        for t in candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-12:
                best_local_obj = obj_try
                best_local_a = a_try
                best_t = t

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_local_a
            t_current = best_t
            stall = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, time={elapsed:.1f}s")
        else:
            stall += 1
            t_current = max(0.05, t_current * 0.85)
            if stall > 40:
                break

    return best_a


def energy_multiplicative_descent(a, time_budget=40.0, lr_init=0.5):
    """
    Minimize sum(conv(a,a)^2) using multiplicative (exponentiated gradient) updates.
    This flattens the autocorrelation spectrum, indirectly reducing the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = np.max(conv)
        if max_conv < 1e-15:
            break

        # Gradient of 0.5*sum(conv^2) w.r.t. a[i]
        # = sum_k conv[k] * a[k-i] for valid (k-i)
        # This is the cross-correlation of conv with a
        grad = np.correlate(conv, a, mode='valid')
        max_grad = np.max(grad)
        if max_grad < 1e-15:
            break

        # Multiplicative update: a_new[i] = a[i] / (1 + lr * grad[i]/max_grad)
        damping = lr * grad / max_grad
        a_new = a / (1.0 + damping)
        a_new = np.maximum(a_new, 1e-12)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new
            a = a_new
            lr = max(lr * 0.9995, 0.01)
        else:
            lr = max(lr * 0.9, 0.01)
            a = best_a

    return best_a


def smooth_max_gradient_descent(a, time_budget=30.0, T_init=0.001, T_min=0.0001):
    """Smooth max (log-sum-exp) gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.008
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -200, 0)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient: grad[i] = sum_k weights[k] * a[k-i]
        grad = np.correlate(weights, a, mode='valid')
        grad = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break

        T = max(T_min, T * 0.998)

    return best_a


def gaussian_smooth(a, sigma):
    """Apply Gaussian smoothing."""
    n = len(a)
    radius = max(1, int(3 * sigma))
    x = np.arange(-radius, radius + 1)
    kernel = np.exp(-x ** 2 / (2 * sigma ** 2))
    kernel = kernel / np.sum(kernel)
    padded = np.pad(a, radius, mode='edge')
    smoothed = np.convolve(padded, kernel, mode='valid')[:n]
    return np.maximum(smoothed, 0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, energy descent,
    smooth-max gradient, Gaussian smoothing perturbations,
    multi-n exploration, and structured random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Intensive LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating ===")
    for round_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.12, 90.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=(round_idx == 0))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP Round {round_idx}: obj={best_score:.8f}")
        else:
            print(f"  LP Round {round_idx}: obj={obj_lp:.8f} (stalled)")
            break

    # === Phase 2: Energy Multiplicative Descent ===
    print(f"\n=== Phase 2: Energy Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.08, 60.0)
        a_ed = energy_multiplicative_descent(best_seq, time_budget=alloc, lr_init=0.5)
        obj_ed = compute_obj(a_ed)
        if obj_ed < best_score:
            best_score = obj_ed
            best_seq = a_ed.copy()
            print(f"  Energy Descent: obj={best_score:.8f}")
        else:
            print(f"  Energy Descent: obj={obj_ed:.8f} (no improvement)")

        # Follow up with LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.05, 30.0)
            a_lp2 = lp_alternating(best_seq, time_budget=alloc)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP after Energy: obj={best_score:.8f}")

    # === Phase 3: Smooth Max Gradient ===
    print(f"\n=== Phase 3: Smooth Max Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.06, 45.0)
        a_sm = smooth_max_gradient_descent(best_seq, time_budget=alloc, T_init=0.001, T_min=0.0001)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Smooth Max: obj={best_score:.8f}")
        else:
            print(f"  Smooth Max: obj={obj_sm:.8f} (no improvement)")

    # === Phase 4: Gaussian Smoothing + Re-optimize ===
    print(f"\n=== Phase 4: Gaussian Smoothing Perturbations ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for sigma in [1.0, 2.0, 3.0, 5.0, 8.0]:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            a_smooth = gaussian_smooth(best_seq, sigma)
            ss = float(np.sum(a_smooth))
            if ss < 1e-10:
                continue
            a_smooth = a_smooth / ss
            # Optimize the smoothed version
            alloc = min(remaining * 0.06, 30.0)
            a_opt = lp_alternating(a_smooth, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Gaussian sigma={sigma}: obj={best_score:.8f} **")
            else:
                print(f"  Gaussian sigma={sigma}: obj={obj_opt:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100), max(300, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200, n_best + 300,
        500, 600, 700, 750, 800, 900, 1000, 1100, 1200, 1300, 1400, 1500, 1600, 1800
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]
    # Sort by proximity to current best
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try

        # LP optimize
        alloc_lp = min(remaining * 0.08, 40.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp)
        obj_opt = compute_obj(a_opt)

        # Energy polish if promising
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_opt < best_score + 0.005:
            alloc_ed = min(remaining * 0.04, 20.0)
            a_opt2 = energy_multiplicative_descent(a_opt, time_budget=alloc_ed)
            obj_opt2 = compute_obj(a_opt2)
            if obj_opt2 < obj_opt:
                a_opt = a_opt2
                obj_opt = obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Structured Random Restarts ===
    print(f"\n=== Phase 6: Structured Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        n_r = int(np.clip(n_best + rng.integers(-150, 151), 400, 1800))
        mode = r % 6

        if mode == 0:
            # Uniform
            a_r = np.ones(n_r) / n_r
        elif mode == 1:
            # Sine window
            x = np.linspace(0, np.pi, n_r)
            a_r = np.sin(x) ** 2
            a_r = a_r / np.sum(a_r)
        elif mode == 2:
            # Gaussian bump
            x = np.arange(n_r)
            center = n_r / 2
            a_r = np.exp(-((x - center) ** 2) / (2 * (n_r / 5) ** 2))
            a_r = a_r / np.sum(a_r)
        elif mode == 3:
            # Raised cosine
            x = np.linspace(0, 1, n_r)
            a_r = 0.5 * (1 + np.cos(2 * np.pi * x))
            a_r = a_r / np.sum(a_r)
        elif mode == 4:
            # Truncated power
            x = np.linspace(0, 1, n_r)
            a_r = (1 - x) ** 0.5 + 0.1
            a_r = a_r / np.sum(a_r)
        else:
            # Dirichlet
            a_r = rng.dirichlet(np.ones(n_r) * 0.3)

        # Optimize
        alloc = min(remaining * 0.1, 25.0)
        a_opt = lp_alternating(a_r, time_budget=alloc * 0.6)

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = energy_multiplicative_descent(a_opt, time_budget=min(alloc * 0.3, 15.0))

        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP rounds
        for _ in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.2, 35.0)
            a_lp = lp_alternating(best_seq, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Polish LP: obj={best_score:.8f}")
            else:
                break

        # Energy
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_ed = energy_multiplicative_descent(best_seq, time_budget=min(remaining * 0.2, 20.0), lr_init=0.3)
            obj_ed = compute_obj(a_ed)
            if obj_ed < best_score:
                best_score = obj_ed
                best_seq = a_ed.copy()
                print(f"  Polish Energy: obj={best_score:.8f}")

        # Smooth max with low T
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sm = smooth_max_gradient_descent(best_seq, time_budget=min(remaining * 0.15, 15.0),
                                               T_init=0.0005, T_min=0.00005)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Polish SM: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
