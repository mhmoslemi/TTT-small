```python
Frank-Wolfe + LP Alternating hybrid optimizer for minimizing
2*n*max(autocorrelation)/(sum)^2 over non-negative sequences.
Exploits convexity of the problem and uses FW for fast convergence
followed by LP for final polish.

from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)

def compute_subgrad(a, k):
```

```python
def frank_wolfe_ls(a, time_budget=300.0, verbose=False):
    """Frank-Wolfe with exact line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        obj_val = 2.0 * n * float(conv[k_star])
        
        if obj_val < best_obj - 1e-12:
            best_obj = obj_val
            best_a = a.copy()
        
        grad = compute_subgrad(a, k_star)
        i_min = int(np.argmin(grad))
        
        e = np.zeros(n)
        e[i_min] = 1.0
        d = e - a  # direction
        
        # Line search: minimize f(a + t*d) for t in [0, 1]
        # Use golden section search
        gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
        
        def f_of_t(t):
            a_t = a + t * d
            a_t = np.maximum(a_t, 0)
            s_t = float(np.sum(a_t))
            if s_t < 1e-10:
                return np.inf
            a_t = a_t / s_t
            conv_t = compute_conv(a_t)
            return 2.0 * n * float(np.max(conv_t))
        
        a_gs = 0.0
        b_gs = 1.0
        c_gs = b_gs - gr * (b_gs - a_gs)
        d_gs = a_gs + gr * (b_gs - a_gs)
        fc = f_of_t(c_gs)
        fd = f_of_t(d_gs)
        
        for _ in range(30):  # 30 iterations of golden section gives ~1e-6 precision
            if fc < fd:
                b_gs = d_gs
                d_gs = c_gs
                fd = fc
                c_gs = b_gs - gr * (b_gs - a_gs)
                fc = f_of_t(c_gs)
            else:
                a_gs = c_gs
                c_gs = d_gs
                fc = fd
                d_gs = a_gs + gr * (b_gs - a_gs)
                fd = f_of_t(d_gs)
        
        t_opt = (a_gs + b_gs) / 2
        a = a + t_opt * d
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        
        it += 1
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW-LS iter {it}: obj={obj_val:.8f}, t_opt={t_opt:.4f}, time={elapsed:.1f}s")
    
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a
```

```python
"""
Frank-Wolfe + LP Alternating hybrid optimizer for minimizing
2*n*max(autocorrelation)/(sum)^2 over non-negative sequences.
Exploits convexity: FW for fast global convergence, LP for local polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgrad(a, k):
    """Compute subgradient of a^T C_k a at point a. O(n)."""
    n = len(a)
    if k == 0:
        return 2.0 * a.copy()
    grad = np.zeros(n)
    grad[:n - k] += 2.0 * a[k:]
    grad[k:] += 2.0 * a[:n - k]
    return grad


def compute_batch_subgrad(a, k_indices, weights):
    """Compute weighted sum of subgradients for multiple peaks."""
    n = len(a)
    grad = np.zeros(n)
    for idx in range(len(k_indices)):
        k = int(k_indices[idx])
        w = weights[idx]
        if k == 0:
            grad += w * 2.0 * a
        else:
            grad[:n - k] += w * 2.0 * a[k:]
            grad[k:] += w * 2.0 * a[:n - k]
    return grad


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def frank_wolfe_ls(a, time_budget=300.0, top_k=1, verbose=False):
    """
    Frank-Wolfe with line search and optional batch subgradient.
    The problem is convex, so FW converges to global optimum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        obj_val = 2.0 * n * max_b

        if obj_val < best_obj - 1e-13:
            best_obj = obj_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Get top-k peaks for batch subgradient
        if top_k >= 2 * n - 1:
            k_indices = np.arange(2 * n - 1)
        else:
            k_indices = np.argsort(conv)[::-1][:top_k]

        # Softmax weights for batch subgradient
        peak_vals = conv[k_indices]
        temp = 0.01 * max_b + 1e-10
        weights = np.exp((peak_vals - peak_vals[0]) / temp)
        weights = weights / np.sum(weights)

        # Compute batch subgradient
        grad = compute_batch_subgrad(a, k_indices, weights)

        # Find simplex vertex minimizing grad^T s
        i_min = int(np.argmin(grad))

        # Direction
        e = np.zeros(n)
        e[i_min] = 1.0
        d = e - a

        # Check if direction gives descent
        if d[i_min] <= 1e-12 and np.max(-d) <= 1e-12:
            # Already at a vertex
            break

        # Line search using golden section (or simplified bisection)
        # For speed, use a few bisection steps
        lo = 0.0
        hi = 1.0

        def f_of_t(t):
            a_t = a + t * d
            a_t = np.maximum(a_t, 0)
            s_t = float(np.sum(a_t))
            if s_t < 1e-10:
                return np.inf
            a_t = a_t / s_t
            conv_t = compute_conv(a_t)
            return 2.0 * n * float(np.max(conv_t))

        # Use 20 bisection steps (enough precision)
        for _ in range(20):
            mid = 0.5 * (lo + hi)
            f_lo = f_of_t(lo)
            f_mid = f_of_t(mid)
            f_hi = f_of_t(hi)
            # For convex function, if f_mid < f_lo, search [mid, hi]
            # Otherwise search [lo, mid]
            if f_mid < f_lo and f_mid < f_hi:
                # mid is better than both endpoints
                # Search around mid
                if f_mid <= f_lo:
                    hi = mid
                else:
                    lo = mid
            elif f_mid < f_lo:
                hi = mid
            elif f_mid < f_hi:
                lo = mid
            else:
                # mid is worse than both, pick the better endpoint
                if f_lo < f_hi:
                    hi = mid
                else:
                    lo = mid

        t_opt = 0.5 * (lo + hi)

        # Update
        a = a + t_opt * d
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        it += 1

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            obj_now = compute_obj(a)
            print(f"    FW-LS iter {it}: obj={obj_now:.8f}, t_opt={t_opt:.4f}, time={elapsed:.1f}s")

    # Final evaluation
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a

    """

        best_a = a.copy()

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation, preserving symmetry."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    # Enforce symmetry
    a_new = 0.5 * (a_new + a_new[::-1])
    return np.maximum(a_new, 0)


def symmetrize(a):
    """Enforce palindromic symmetry."""
    return 0.5 * (a + a[::-1])

    print(f"\n=== Phase 2: FW with line search ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        fwls_budget = min(200.0, remaining * 0.2)
        a_fwl = frank_wolfe_ls(a, time_budget=fwls_budget, top_k=3, verbose=True)
        obj_fwl = compute_obj(a_fwl)
        if obj_fwl < best_score:
            best_score = obj_fwl
            best_seq = a_fwl.copy()
            a = a_fwl.copy()
            print(f"  FW-LS: obj={best_score:.8f} **")
        else:
            print(f"  FW-LS: obj={obj_fwl:.8f}")

    # === Phase 3: LP Alternating (300s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(300.0, remaining * 0.3)

    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(lp_budget / (3 - round_idx), remaining * 0.4)
        a_lp = lp_alternating(a, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  LP Round {round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP Round {round_idx}: obj={obj_lp:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    n_options = sorted(set([
        max(100, n_parent - 200),
        max(100, n_parent - 100),
        n_parent,
        n_parent + 100,
        n_parent + 200,
        min(2000, n_parent + 400),
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]

    # Remove n_parent (already optimized)
    n_options = [n for n in n_options if n != n_parent]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = symmetrize(a_try)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try

        # Quick FW + LP
        alloc_fw = min(40.0, remaining * 0.25)
        alloc_lp = min(40.0, remaining * 0.3)

        a_opt = frank_wolfe_fast(a_try, time_budget=alloc_fw, top_k=3)
        a_opt = lp_alternating(a_opt, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final LP polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 6: Random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Different perturbation scales
            scale = 0.02 + 0.03 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

            alloc = min(remaining * 0.3, 25.0)
            a_opt = frank_wolfe_fast(a_r, time_budget=alloc * 0.5, top_k=3)
            a_opt = lp_alternating(a_opt, time_budget=alloc * 0.5)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: One more FW pass on best ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 60.0)
        a_last = frank_wolfe_fast(best_seq, time_budget=alloc, top_k=3)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final FW: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
