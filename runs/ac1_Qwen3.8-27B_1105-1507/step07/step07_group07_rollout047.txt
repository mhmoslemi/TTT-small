"""
Hybrid convex-structured optimizer for low-autocorrelation non-negative sequences.
Uses FISTA with adaptive logsumexp smoothing, fast active-set LP polish via HiGHS,
sparsity exploitation, and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    size = next_pow2(3 * n)
    fa = rfft(a, size)
    fw = rfft(w, size)
    cross = irfft(np.conj(fa) * fw, size)
    g = cross[:n]

    grad = (2.0 / tau) * g
    return grad, conv


def fista_smoothed(a_init, time_budget=60.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    taus = [30.0/n, 15.0/n, 8.0/n, 4.0/n, 2.0/n, 1.0/n, 0.5/n, 0.25/n, 0.125/n, 0.0625/n, 0.03/n, 0.015/n]

    for tau_idx, tau in enumerate(taus):
        if time.monotonic() - start > time_budget:
            break

        y = best_a.copy()
        t_k = 1.0
        a_curr = best_a.copy()
        step = tau / (6.0 * n)

        max_iter = max(200, int(30.0 / (time_budget / max(len(taus), 1)) * (n / 400)))
        max_iter = min(max_iter, 2000)

        for it in range(max_iter):
            if time.monotonic() - start > time_budget:
                break

            grad, conv = smoothed_grad(y, tau)

            a_new = project_simplex(a_new, 1.0)

def solve_lp_active(a, epsilon=0.0, time_budget=5.0):

    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    max_conv = np.max(conv)

    if epsilon == 0.0:
        active = list(range(2 * n - 1))
    else:
        active = [k for k in range(2 * n - 1) if conv[k] >= max_conv * (1.0 - epsilon)]
        if len(active) < 5:
            top_k = np.argsort(conv)[::-1][:max(10, len(active))]
            active = list(top_k)

    A = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        lo = max(0, k - (n - 1))
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                A[idx, i] = a_norm[j]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])

        pass

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.12, 35.0)
        a_opt = fista_smoothed(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # Phase 6: Final FISTA + LP on best
    print(f"\n=== Phase 6: Final Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_f2 = fista_smoothed(best_seq, time_budget=min(remaining * 0.25, 80.0), verbose=True)
        obj = compute_obj(a_f2)
        if obj < best_score:
            best_score = obj
            best_seq = a_f2.copy()
            print(f"  FISTA2: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_lp2 = lp_alternate_fast(best_seq, time_budget=min(remaining * 0.3, 60.0))
            obj = compute_obj(a_lp2)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp2.copy()
                print(f"  LP2: obj={best_score:.10f}")

    # Phase 7: Random restarts
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    n_restarts = max(2, min(8, int(remaining / 30)))
    restart_budget = min(25.0, remaining / max(n_restarts, 1) * 0.8)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < restart_budget + 5:
            break

        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            n_r = n_best + int(rng.integers(-150, 150))
            n_r = max(400, min(1600, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 2:
            a_r = diverse_init(n_best, r % 6, rng)
        elif r == 3:
            n_r = int(n_best * rng.uniform(0.8, 1.3))
            n_r = max(350, min(1800, n_r))
            a_r = diverse_init(n_r, int(rng.integers(0, 6)), rng)
        else:
            a_r = best_seq + 0.05 * rng.normal(0, 1, n_best)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        a_opt = fista_smoothed(a_r, time_budget=restart_budget)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (n={len(a_r)}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (n={len(a_r)}): obj={obj:.10f}")

    # Phase 8: Final fine-grained polish
    print(f"\n=== Phase 8: Final Fine Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fin = fista_smoothed(best_seq, time_budget=min(remaining * 0.4, 40.0))
        obj = compute_obj(a_fin)
        if obj < best_score:
            best_score = obj
            best_seq = a_fin.copy()
            print(f"  Final FISTA: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_fin2 = lp_alternate_fast(best_seq, time_budget=min(remaining * 0.5, 30.0))
            obj = compute_obj(a_fin2)
            if obj < best_score:
                best_score = obj
                best_seq = a_fin2.copy()
                print(f"  Final LP: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
