import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, project
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    v_sum = np.sum(v)
    if v_sum <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_value_and_weights(conv, T):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    return smooth_val, weights


def grad_smooth_max(weights, a):
    """Gradient of smooth max w.r.t. a using correlation."""
    return np.correlate(weights, a, mode='valid')


def solve_lp(a, M_target, reg=0.0):
    """
    Solve LP: maximize sum(g) - reg*sum(g^2_approx) subject to conv(a,g) <= M_target, g >= 0.
    Returns g normalized to sum 1, or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_target, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(-cp.sum(g)), constraints)  # maximize sum(g) = 1, so this is trivial
    # Actually, since sum(g) = 1 is a constraint, we should minimize something else
    # Let's minimize the max violation or use a different formulation

    constraints2 = [A @ g <= M_var, cp.sum(g) == 1, g >= 0]
    prob = cp.Problem(cp.Minimize(M_var), constraints2)
    
    solvers = ['CLARABEL', 'SCS', 'MOSEK', 'ECOS']
    for solver in solvers:
        try:
            prob.solve(solver=solver)

def solve_lp_reg(a, M_target, reg_weight=0.01):

"""
Enhanced multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def simplex_project(v):
    n = len(v)
    v_sum = np.sum(v)
    if v_sum <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_weights(conv, T):
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    return weights


def grad_smooth_max(weights, a):
    return np.correlate(weights, a, mode='valid')


def solve_lp_minpeak(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g <= M_var, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = ['CLARABEL', 'SCS', 'MOSEK']
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))

    return None


def solve_qp_smooth(a, reg_weight=0.01):
    """Solve QP: minimize reg_weight * smoothness(g) s.t. conv(a,g) <= M_current, sum(g)=1, g>=0."""

    M_current = float(np.max(conv))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    smoothness = cp.sum_squares(g[1:] - g[:-1])
    constraints = [A @ g <= M_current, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(reg_weight * smoothness), constraints)
    solvers = ['CLARABEL', 'SCS', 'MOSEK']
    for solver in solvers:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

def peak_shave(a, time_budget=30.0):
    """Directly reduce the top peaks using subgradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    
    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Get top 3 peaks
        top_k = 3
        top_indices = np.argsort(conv)[-top_k:]
        
        # Subgradient: sum over top peaks
        grad = np.zeros(n)
        for k_idx in top_indices:
            k = int(k_idx)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        
        grad_centered = grad - np.mean(grad)
        a_new = simplex_project(a - step * grad_centered)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if step < 0.1:
                step = min(step * 1.02, 0.1)
        else:
            step *= 0.95
            a = best_a.copy()
            if step < 1e-5:
                break
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
