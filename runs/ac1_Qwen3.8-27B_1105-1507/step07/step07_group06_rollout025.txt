Key improvements over previous: (1) Anderson acceleration on LP fixed-point

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_diverse_init(n, rng, parent=None):
    """Generate a diverse set of initial sequences."""
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n) / n)
    
    # 2. Linear ramps (various)
    for slope in [0.5, 1.0, 2.0, 5.0, 10.0]:
        x = np.linspace(0, 1, n)
        a = 1 + slope * x
        a = np.maximum(a, 0)
        a = a / a.sum()
        inits.append(a)
    
    # 3. Exponential (various rates)
    for rate in [1.0, 3.0, 5.0, 10.0, 20.0]:
        x = np.linspace(0, 1, n)
        a = np.exp(rate * x)
        a = a / a.sum()
        inits.append(a)
    
    # 4. Cosine modulated
    for freq in [1.0, 2.0, 5.0, 10.0]:
        for phase in [0, np.pi/2, np.pi]:
            x = np.linspace(0, 1, n)
            a = 1 + 0.5 * np.cos(2 * np.pi * freq * x + phase)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            inits.append(a)
    
    # 5. Binary sparse (various densities)
    for density in [0.1, 0.3, 0.5, 0.7]:
        for _ in range(3):
            a = (rng.random(n) < density).astype(float)
            if a.sum() > 0:
                a = a / a.sum()
                inits.append(a)
    
    # 6. Random smooth
    for _ in range(5):
        a = np.abs(rng.normal(1.0, 0.5, n))
        # Smooth it
        kernel_size = max(3, n // 50)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
        inits.append(a)
    
    # 7. Triangle / tent shapes
    for center_frac in [0.25, 0.5, 0.75]:
        center = int(center_frac * n)
        a = 1.0 - np.abs(np.arange(n) - center) / n
        a = np.maximum(a, 0)
        a = a / a.sum()
        inits.append(a)
    
    # 8. If parent available, perturbed versions
    if parent is not None and len(parent) == n:
        for scale in [0.02, 0.05, 0.1, 0.2]:
            a = parent * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                inits.append(a)
    
    # 9. Geometric progression
    for ratio in [1.001, 1.005, 1.01, 1.05]:
        a = ratio ** np.arange(n)
        a = a / a.sum()
        inits.append(a)
    
    # 10. Sigmoid-like
    for sharp in [5, 10, 20, 50]:
        x = np.linspace(0, 1, n)
        a = 1 / (1 + np.exp(-sharp * (x - 0.5)))
        a = a / a.sum()
        inits.append(a)
    
    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines diverse initialization, Anderson-accelerated LP alternating,
    peak shaving, smoothed gradient, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_parent = len(parent)
                print(f"Parent (n={n_parent}): obj={best_score:.8f}")
    except Exception:
        parent = None
    
    if best_seq is None:
        parent = None
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")
    
    # === Phase 1: Diverse initialization screening ===
    print(f"\n=== Phase 1: Diverse Initialization Screening ===")
    remaining = deadline - time.monotonic()
    n_screen = n_parent
    
    # Generate diverse inits
    diverse_inits = generate_diverse_init(n_screen, rng, parent)
    print(f"  Generated {len(diverse_inits)} diverse initializations")
    
    # Quick screen: evaluate each with a few gradient steps
    screen_results = []
    for idx, a_init in enumerate(diverse_inits):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if a_init is None or len(a_init) != n_screen:
            continue
        
        a_init = np.maximum(a_init, 1e-10)
        a_init = a_init / a_init.sum()
        
        # Quick smoothed optimizer (short budget)
        alloc = min(remaining * 0.05, 15.0)
        a_opt, obj_opt = smooth_lbfgs(a_init, tau_start=10.0, tau_end=200.0,
                                       time_budget=alloc, max_iter=50)
        screen_results.append((obj_opt, a_opt, idx))
    
    # Sort by objective
    screen_results.sort(key=lambda x: x[0])
    print(f"  Screened {len(screen_results)} inits, best: obj={screen_results[0][0]:.8f}")
    
    # Keep top candidates
    top_candidates = []
    for obj_i, a_i, idx_i in screen_results[:8]:
        if obj_i < best_score:
            best_score = obj_i
            best_seq = a_i.copy()
        top_candidates.append(a_i.copy())
    
    # === Phase 2: Intensive LP alternating on top candidates ===
    print(f"\n=== Phase 2: Intensive LP Alternating (Anderson) ===")
    remaining = deadline - time.monotonic()
    
    # Allocate time to top candidates
    n_lp_trials = min(len(top_candidates), max(2, int(remaining / 80)))
    for idx in range(n_lp_trials):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        a_start = top_candidates[idx]
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_alternating_anderson(a_start, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            top_candidates[0] = a_lp.copy()  # Put best at front
            print(f"  LP trial {idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP trial {idx}: obj={obj_lp:.8f}")
    
    # === Phase 3: Peak shaving on best ===
    print(f"\n=== Phase 3: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 40.0)
        a_ps = peak_shaving(best_seq, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Peak shaving: obj={best_score:.8f} **")
        else:
            print(f"  Peak shaving: obj={obj_ps:.8f}")
    
    # === Phase 4: More LP alternating on improved best ===
    print(f"\n=== Phase 4: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.3, 70.0)
        a_lp2 = lp_alternating_anderson(best_seq, time_budget=alloc, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP round 2: obj={best_score:.8f} **")
        else:
            print(f"  LP round 2: obj={obj_lp2:.8f}")
    
    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(300, n_cur - 200), max(300, n_cur - 100),
        n_cur - 50, n_cur, n_cur + 50,
        n_cur + 100, n_cur + 200,
        max(300, n_cur + 400),
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if n_try == n_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        # Quick LP alternating
        alloc = min(remaining * 0.15, 35.0)
        a_opt = lp_alternating_anderson(a_try, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 6: Smoothed gradient refinement ===
    print(f"\n=== Phase 6: Smoothed Gradient Refinement ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    
    for tau_s in [30.0, 100.0, 300.0, 800.0]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.15, 30.0)
        a_opt, obj_opt = smooth_lbfgs(best_seq, tau_start=tau_s, tau_end=tau_s * 5,
                                       time_budget=alloc, max_iter=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Smooth tau_s={tau_s}: obj={best_score:.8f} **")
        else:
            print(f"  Smooth tau_s={tau_s}: obj={obj_opt:.8f}")
    
    # === Phase 7: Final LP + peak shaving polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        alloc = min(remaining * 0.35, 50.0)
        a_lp3 = lp_alternating_anderson(best_seq, time_budget=alloc)
        obj_lp3 = compute_obj(a_lp3)
        if obj_lp3 < best_score:
            best_score = obj_lp3
            best_seq = a_lp3.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 30.0)
        a_ps2 = peak_shaving(best_seq, time_budget=alloc)
        obj_ps2 = compute_obj(a_ps2)
        if obj_ps2 < best_score:
            best_score = obj_ps2
            best_seq = a_ps2.copy()
            print(f"  Final peak shave: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 25.0)
        a_pg, obj_pg = projected_gradient_simplex(best_seq, tau=500.0,
                                                   steps=200, time_budget=alloc)
        if obj_pg < best_score:
            best_score = obj_pg
            best_seq = a_pg.copy()
            print(f"  Final PG: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
