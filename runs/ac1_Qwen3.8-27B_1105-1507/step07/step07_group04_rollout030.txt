import time

def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def grad_conv_at(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] = a[j]
    return grad

def grad_max_conv(a, top_k=5):
    """Approximate gradient of max_k (a*a)[k] using top-k positions."""
    conv = compute_conv(a)
    indices = np.argsort(conv)[::-1][:top_k]
    # Weight by how close to max
    weights = np.exp((conv[indices] - conv[indices[0]]) / 0.01)
    weights /= weights.sum()
    grad = np.zeros(len(a))
    for w, k in zip(weights, indices):
        grad += w * grad_conv_at(a, int(k))
    return grad, conv

def projected_gradient_descent(a, time_budget=100, lr=0.01, top_k=5):
    """Projected gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

This is a convex QP! (Each a^T Q_k a is a quadratic form, and we're minimizing the max of quadratic functions, which is convex if each Q_k is positive semi-definite.)

Wait, is Q_k positive semi-definite? Q_k[i,j] = 1 if i+j = k or i+j = k (with the symmetric contribution). Actually, (a*a)[k] = sum_i a[i]*a[k-i] = a^T Q_k a where Q_k is a permutation matrix (or half-permutation). Permutation matrices are not PSD in general.

So it's not convex. Back to the drawing board.

Let me just go with a practical approach that's different enough from the previous code to potentially find something better.

**Final algorithm plan:**

```
Phase 0: Load parent, compute initial objective
Phase 1: Parametric family sweep (find best parametric init)
Phase 2: Main optimization loop:
  - Alternate between:
    a. LP-based update (solve the LP for given a, blend)
    b. Gradient step (subgradient of max conv)
    c. Fourier perturbation (random phase in freq domain)
  - Adaptive learning rate
  - Track best solution
Phase 3: Length sweep (try different n, resample best)
Phase 4: Polish (coordinate descent, small perturbations)
```

One more thought: the previous code uses `cvxpy` for the LP. This can be slow. An alternative is to use `scipy.optimize.linprog` directly, which might be faster for this specific structure. Or even better, implement a custom simplex for this specific LP structure.

Actually, for the LP:
min M
s.t. conv(a, g)[k] <= M for k = 0, ..., 2n-2
     sum(g) = 1
     g >= 0

The constraint matrix has a banded structure (each column has at most n nonzeros). For n = 800, the matrix is 1599 x 800, which is manageable for linprog.

Let me also consider: instead of solving the full LP every iteration, we can use a **warm start** approach where we only solve the LP on a subset of constraints (the active ones).

"""
Improved optimizer using:
1. Parametric family initialization
2. LP alternating with warm-start and active-set reduction
3. Fourier-domain random perturbation
4. Subgradient with momentum
5. Multi-length exploration
6. Final coordinate descent polish

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def obj_normalized(a):
    """Objective assuming sum(a) = 1: 2*n*max(conv(a,a))"""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def normalize(a):
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    return a / s

def build_conv_matrix(a):
    """A such that A @ g = conv(a, g). Shape: (2n-1, n)"""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k - j] if 0 <= k-j < n, else 0
        # k ranges from j to j+n-1
        for k in range(j, min(2 * n - 1, j + n)):
            A[k, j] = a[k - j]
    return A

def solve_lp_given_a(a, active_only=False, top_m=None):
    """
    Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    Returns (g, M) or (None, None)
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix
    A_mat = build_conv_matrix(a)  # (2n-1, n)
    
    if active_only and top_m is not None:
        conv_aa = compute_conv(a)
        top_indices = np.argsort(conv_aa)[::-1][:top_m]
        A_mat = A_mat[top_indices, :]
    
    # LP: min M
    # vars: g[0..n-1], M
    # A_ub @ [g, M] <= b_ub
    #   A_mat @ g - M <= 0  =>  A_mat @ g + (-1)*M <= 0
    # A_eq @ [g, M] = b_eq
    #   sum(g) = 1
    # bounds: g >= 0, M free
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.hstack([A_mat, -np.ones((A_mat.shape[0], 1))])
    b_ub = np.zeros(A_mat.shape[0])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs, float(res.x[n])
    return None, None

def fourier_perturb(a, sigma=0.1):
    """Add random phase in frequency domain, project back to simplex."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    # Add random phase
    rng = np.random.default_rng(int(time.monotonic() * 1000000) % (2**31))
    phases = rng.uniform(-sigma, sigma, size)
    fa_new = fa * np.exp(1j * phases)
    a_new = np.real(ifft(fa_new, size)[:n])
    return normalize(a_new)

def subgradient_step(a, eta, top_k=5):
    """One subgradient step for max_k (a*a)[k]."""
    n = len(a)
    conv = compute_conv(a)
    k_indices = np.argsort(conv)[::-1][:top_k]
    
    # Weighted average of subgradients
    max_val = conv[k_indices[0]]
    weights = np.exp((conv[k_indices] - max_val) / (max_val * 0.1 + 1e-10))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, k_indices):
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += w * a[j]
    
    # Center (project to sum=0)
    grad -= grad.mean()
    
    a_new = a - eta * grad
    return normalize(a_new)

def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating with adaptive step and active set."""
    n = len(a)
    a = normalize(a.copy())
    if a is None:
        return a
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        
        # Adaptive: use active set if enough time, full LP if not
        if remaining > 5:
            g_lp, M_lp = solve_lp_given_a(a, active_only=False)
        else:
            g_lp, M_lp = solve_lp_given_a(a, active_only=True, top_m=50)
        
        if g_lp is None:
            # Try perturbed
            a_pert = fourier_perturb(a, 0.05)
            if a_pert is not None:
                g_lp, M_lp = solve_lp_given_a(a_pert)
            if g_lp is None:
                break
        
        # Line search
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()
        
        ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        if 0.01 < last_t < 0.99:
            ts = sorted(set(ts + [last_t * 0.5, last_t * 1.5]))
            ts = [t for t in ts if 0.01 <= t <= 1.0]
        
        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = normalize(a_try)
            if a_try is None:
                continue
            obj_try = obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            last_t = best_t_val
            stall = 0
            if verbose and int(elapsed) % 20 == 0:
                print(f"    LP: obj={best_obj:.8f}, t={best_t_val:.3f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 10:
                a = fourier_perturb(best_a, 0.03 + 0.01 * min(stall // 10, 5))
                if a is None:
                    a = best_a * np.exp(np.random.default_rng(stall).normal(0, 0.02, n))
                    a = normalize(a.copy())
                best_a = a.copy()
                best_obj = obj_normalized(a)
                stall = 0
                last_t = 0.3
    
    return best_a

def parametric_sweep(n, time_budget=30.0):
    """Sweep parametric families to find good initialization."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf
    
    x = np.linspace(0, 1, n)
    
    families = []
    
    # Linear ramps
    for alpha in np.linspace(-0.5, 0.5, 11):
        a = 1.0 + alpha * x
        if np.min(a) < 0:
            continue
        a = a / np.sum(a)
        families.append(a)
    
    # Cosine
    for beta in np.linspace(0, 0.5, 5):
        a = 1.0 + beta * np.cos(2 * np.pi * x)
        if np.min(a) < 0:
            continue
        a = a / np.sum(a)
        families.append(a)
    
    # Sinusoidal
    for eps in np.linspace(-0.3, 0.3, 7):
        a = 1.0 + eps * np.sin(2 * np.pi * x)
        if np.min(a) < 0:
            continue
        a = a / np.sum(a)
        families.append(a)
    
    # Exponential
    for gamma in np.linspace(-1, 1, 7):
        a = np.exp(gamma * x)
        a = a / np.sum(a)
        families.append(a)
    
    # Parabolic
    for delta in np.linspace(-0.3, 0.3, 5):
        a = 1.0 + delta * (2 * x - 1) ** 2
        if np.min(a) < 0:
            continue
        a = a / np.sum(a)
        families.append(a)
    
    # Half-sine
    for phase in np.linspace(0, 2*np.pi, 7):
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x + phase)
        if np.min(a) < 0:
            continue
        a = a / np.sum(a)
        families.append(a)
    
    for a in families:
        if time.monotonic() - start > time_budget:
            break
        obj = obj_normalized(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent polish."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        k_star = int(np.argmax(conv))
        improved = False
        
        # Try moving mass from high to low
        for i in range(0, n, 3):
            if time.monotonic() - start > time_budget:
                break
            j = (k_star - i) % n
            if j == i or j < 0 or j >= n:
                continue
            for frac in [0.02, 0.05, 0.1]:
                delta = frac * a[i]
                if delta < 1e-15:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    break
                a /= s
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    a /= s
                    break
        
        if not improved:
            # Random pair swap
            rng = np.random.default_rng(it * 7919 + 1)
            for _ in range(5):
                i1, i2 = rng.integers(0, n, 2)
                if i1 == i2:
                    continue
                delta = 0.03 * max(a[i1], a[i2])
                a[i1] = max(0, a[i1] - delta)
                a[i2] += delta
                a = normalize(a.copy())
                if a is None:
                    continue
                new_obj = obj_normalized(a)
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
            if not improved and it > 100:
                break
    
    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)
