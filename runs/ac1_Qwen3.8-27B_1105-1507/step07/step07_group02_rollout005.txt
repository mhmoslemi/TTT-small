    n = len(a)

and spectral shaping. Combines multiple optimization strategies with multi-n exploration.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Linear autocorrelation a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_internal(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 internally."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    c = compute_conv(a_norm)
    max_c = float(np.max(c))
    return 2.0 * n * max_c


def compute_pressure(a, w):
    """
    Compute pressure p[i] = sum_k w[k] * a[k-i] for valid indices.
    w has length 2n-1.
    Uses FFT for efficiency.
    """
    n = len(a)
    w_rev = w[::-1]
    size = 1
    while size < n + len(w_rev) - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv = irfft(fa * fw, size)
    # p[i] = conv[2n-2-i] for i=0..n-1
    p = conv[n-1:2*n-1][::-1]
    return p

            stall = 0
            # Anneal: decrease tau, increase eta slightly
            tau = max(tau * 0.995, 0.005)
            eta = min(eta * 1.01, 2.0)

def logsumexp_gradient_optimize(a, time_budget=60.0, verbose=False):
    """
    Gradient descent with log-sum-exp smoothing of the max autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj_internal(a)
    start = time.monotonic()
    
    sigma = 0.1  # smoothing parameter
    lr = 0.05
    stall = 0
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        # Adapt sigma
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        sigma = 0.1 * (0.01 / 0.1) ** progress
        
        c = compute_conv(a)
        c_max = float(np.max(c))
        
        # Log-sum-exp
        c_shifted = (c - c_max) / sigma
        exp_vals = np.exp(np.minimum(c_shifted, 0))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            break
        
        w = exp_vals / sum_exp  # softmax weights
        
        # Gradient (same as pressure computation)
        grad = compute_pressure(a, w)
        grad_centered = grad - np.mean(grad)
        
        # Gradient step
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj_internal(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.02, 0.5)
        else:
            a = best_a.copy()
            lr *= 0.8
            stall += 1
            if stall > 100:
                rng = np.random.default_rng((it * 7 + n) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s_p = float(np.sum(a))
                if s_p > 1e-10:
                    a = a / s_p
                    obj_p = compute_obj_internal(a)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a.copy()
                stall = 0
                lr = 0.05

def spectral_shape_optimize(a, time_budget=30.0, verbose=False):
    """
    Spectral shaping: try to flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

            break
        
        # Compute FFT
        fa = rfft(a)
        # Power spectrum
        power = np.abs(fa) ** 2
        phase = np.angle(fa)
        
        # Try to flatten: compress dynamic range
        for alpha in [0.5, 0.7, 0.8, 0.9]:
            power_flat = power ** alpha
            # Reconstruct
            mag = np.sqrt(power_flat)
            fa_new = mag * np.exp(1j * phase)
            a_new = irfft(fa_new, n)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                break
        
        # Also try: additive spectral shaping
        noise_level = 0.01 * np.max(power)
        power_noisy = power + noise_level
        mag = np.sqrt(power_noisy)
        fa_new = mag * np.exp(1j * phase)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:

            obj_new = compute_obj_internal(a_new)

                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0).astype(np.float64)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=30.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_work = a / float(np.sum(a))
    A = build_conv_matrix(a_work)
    
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def alternating_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_internal(a)
    start = time.monotonic()
    stall = 0
    
    max_iters = 100
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        best_t_obj = np.inf
        best_a_local = best_a.copy()
        
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_internal(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    
    return best_a
