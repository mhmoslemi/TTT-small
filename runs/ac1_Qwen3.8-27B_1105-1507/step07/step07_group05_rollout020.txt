def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def logsumexp_gradient(a, b, T):
    """
    Compute gradient of soft_max(b, T) = (1/T)*log(sum(exp(T*b_k)))
    with respect to a. Returns gradient vector of length n.
    The soft max is a smooth approximation of max(b).
    """
    n = len(a)
    T_safe = max(T, 1e-6)
    b_shifted = T_safe * (b - np.max(b))
    weights = np.exp(b_shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        return np.zeros(n)
    w = weights / w_sum  # softmax weights over b

    # Gradient: sum_k w[k] * d(b[k])/d(a[i]) = sum_k w[k] * a[k-i] (if valid)
    # This is a cross-correlation of w with a, computed via FFT
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    a_flip = a[::-1]
    faf = rfft(a_flip, N)
    fw = rfft(w, N)
    c_full = irfft(faf * fw, N)[:3 * n - 2]
    grad = c_full[n - 1:2 * n - 1] / T_safe
    return grad


def energy_gradient(a, b):
    """Compute gradient of sum(b_k^2) with respect to a. This is 2 * conv(a_flip, b) centered."""
    n = len(a)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    a_flip = a[::-1]
    faf = rfft(a_flip, N)
    fb = rfft(b, N)
    c_full = irfft(faf * fb, N)[:3 * n - 2]
    grad = c_full[n - 1:2 * n - 1] * 2.0
    return grad


def spectral_shaping(a, alpha=0.5, iters=5):
    """
    Spectral-domain optimization: modify the magnitude spectrum to flatten
    the auto-convolution. Iteratively reduce spectral power at frequencies
    that contribute most to the peak.
    """
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(iters):
        fa = rfft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)
        b = irfft(fa * fa, N)[:2 * n - 1]
        b_max = np.max(b)
        b_mean = np.mean(b)

        # Compute how much each frequency contributes to the peak
        # The peak is at position k_star
        k_star = int(np.argmax(b))
        # Contribution of freq m to b[k_star] is |A[m]|^2 * exp(2*pi*i*m*k_star/N)/N
        # To reduce b[k_star], reduce |A[m]|^2 for all m (they all contribute with phase)
        # Simple approach: reduce |A| where it's largest
        # Better: compute gradient in spectral domain

        # Gradient approach: d(b[k_star])/d(|A[m]|) = 2*|A[m]|*cos(2*pi*m*k_star/N + 2*phase[m])
        # Simplified: just reduce |A| proportionally to its contribution to peak
        freqs = np.arange(N // 2 + 1)
        # Phase of contribution to peak
        phases = 2 * np.pi * freqs * k_star / N
        # Real contribution to b[k_star]
        contrib = mag ** 2 * np.cos(phases + 2 * phase)
        # Normalize
        contrib = np.maximum(contrib, 0)
        c_sum = np.sum(contrib)
        if c_sum > 1e-20:
            contrib = contrib / c_sum

        # Reduce magnitude where contribution is high
        new_mag = mag * (1.0 - alpha * contrib * (N // 2))
        new_mag = np.maximum(new_mag, 1e-10)

        new_fa = new_mag * np.exp(1j * phase)
        a_new = irfft(new_fa, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a = a_new / s_new

    return a


def adaptive_gd(a, time_budget=60.0, mode='adaptive'):
    """
    Gradient descent with adaptive temperature for log-sum-exp gradient.
    Mode 'adaptive': T increases over time, focusing more on peak.
    Mode 'energy': T stays low (energy minimization).
    Mode 'peak': T starts high (peak focused).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    eta = 0.5
    momentum = 0.5
    v = np.zeros(n)

    if mode == 'energy':
        T_start = 1.0 / np.max(compute_conv(a) + 1e-10) * 0.5
        T_end = T_start
    elif mode == 'peak':
        T_start = 1.0 / np.max(compute_conv(a) + 1e-10) * 5.0
        T_end = T_start * 10
    else:  # adaptive
        T_start = 1.0 / np.max(compute_conv(a) + 1e-10) * 1.0
        T_end = T_start * 20

    stall = 0
    max_stall = 300
    elapsed = 0

    for it in range(200000):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        # Progress: 0 to 1
        progress = min(elapsed / time_budget, 1.0)
        T = T_start * (T_end / T_start) ** progress

        # Compute b
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Compute gradient
        if mode == 'energy':
            grad = energy_gradient(a, b)
        else:
            grad = logsumexp_gradient(a, b, T)

        # Center gradient (since we normalize sum to 1)
        grad = grad - np.mean(grad) * a / max(float(np.sum(a)), 1e-10)
        grad = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        # Evaluate
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.003, 3.0)
            momentum = min(momentum + 0.0005, 0.9)
        else:
            stall += 1
            eta *= 0.98
            v *= 0.7
            if stall > max_stall:
                # Reset with perturbation
                rng_local = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
                noise = rng_local.normal(0, 0.03 * np.max(best_a), n)
                a = np.maximum(best_a + noise, 0)
                s_r = float(np.sum(a))
                if s_r > 1e-10:
                    a = a / s_r
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.5
                stall = 0
                max_stall = min(max_stall + 100, 800)
            if eta < 1e-9:
                break

    return best_a


def lp_polish(a, time_budget=60.0):
    """
    LP-based polishing: alternate between LP solve and blending.
    More aggressive than previous: uses active-set detection to warm start.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    prev_g = None

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Find best blending parameter
        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            prev_g = g_lp
        else:
            stall += 1
            if stall > 15:
                break

    return best_a


def mass_redistribute(a, n_moves=1000, time_budget=30.0):
    """
    Local search: randomly move small amounts of mass between positions.
    Accept moves that reduce the objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    move_size = 0.01 * np.max(a)

    for it in range(n_moves):
        if time.monotonic() - start > time_budget:
            break

        # Pick source and destination
        i_src = rng.integers(0, n)
        i_dst = rng.integers(0, n)
        if i_src == i_dst:
            continue
        if a[i_src] < move_size:
            continue

        # Try different move fractions
        for frac in [0.5, 1.0, 0.3, 0.7]:
            delta = a[i_src] * frac
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new

            fa_new = rfft(a_new, N)
            b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
            obj_new = 2.0 * n * float(np.max(b_new))

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                break

        # Occasionally try larger structural changes
        if it % 100 == 50:
            # Block flip
            block_len = min(n // 4, 50)
            start_idx = rng.integers(0, max(n - block_len, 1))
            a_new = a.copy()
            a_new[start_idx:start_idx + block_len] = a[start_idx:start_idx + block_len][::-1]
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                fa_new = rfft(a_new, N)
                b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
                obj_new = 2.0 * n * float(np.max(b_new))
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new

    return best_a


def asymmetrize(a, strength=0.3):
    """
    Push the sequence toward asymmetry to reduce the center convolution peak.
    Multiply by a linear ramp to create asymmetry.
    """
    n = len(a)
    x = np.linspace(0, 1, n)
    # Asymmetric weight: more mass at one end
    w = 1.0 + strength * (2 * x - 1)
    w = np.maximum(w, 0.01)
    a_new = a * w
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init_asymmetric(n, rng):
    """Generate asymmetric random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        # Exponential decay
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-x / (n * 0.3))
    elif mode == 1:
        # Exponential rise
        x = np.arange(n, dtype=np.float64)
        a = np.exp(x / (n * 0.3))
    elif mode == 2:
        # Asymmetric Gaussian
        center = n * 0.3
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - center) ** 2) / (n * 0.15))
    elif mode == 3:
        # Asymmetric Gaussian (other side)
        center = n * 0.7
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - center) ** 2) / (n * 0.15))
    elif mode == 4:
        # Power law (heavy at start)
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**1.5
    elif mode == 5:
        # Power law (heavy at end)
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / (n + 1 - x)**1.5
    elif mode == 6:
        # Sine-based asymmetric
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 * np.cos(x / 2)
        a = np.maximum(a, 0)
    else:
        # Random with asymmetry
        a = rng.uniform(0, 1, n)
        ramp = np.linspace(0.5, 1.5, n)
        a = a * ramp

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def combined_optimize(a, time_budget=120.0):
    """
    Combined optimization: adaptive GD + spectral shaping + LP + mass redistribution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def _remaining():
        return max(0, start + time_budget - time.monotonic())

    # Step 1: Energy GD (smooth, fast convergence)
    rem = _remaining()
    if rem > 15:
        a_gd = adaptive_gd(a, time_budget=min(rem * 0.3, 40.0), mode='energy')
        obj = compute_obj(a_gd)
        if obj < best_obj:
            best_obj = obj
            best_a = a_gd.copy()
            a = a_gd.copy()
            print(f"    [Combined] Energy GD: {best_obj:.8f}")

    # Step 2: Spectral shaping
    rem = _remaining()
    if rem > 10:
        a_spec = spectral_shaping(a, alpha=0.3, iters=8)
        obj = compute_obj(a_spec)
        if obj < best_obj:
            best_obj = obj
            best_a = a_spec.copy()
            a = a_spec.copy()
            print(f"    [Combined] Spectral: {best_obj:.8f}")

    # Step 3: Adaptive GD (peak-focused)
    rem = _remaining()
    if rem > 15:
        a_gd2 = adaptive_gd(a, time_budget=min(rem * 0.3, 35.0), mode='adaptive')
        obj = compute_obj(a_gd2)
        if obj < best_obj:
            best_obj = obj
            best_a = a_gd2.copy()
            a = a_gd2.copy()
            print(f"    [Combined] Adaptive GD: {best_obj:.8f}")

    # Step 4: Mass redistribution
    rem = _remaining()
    if rem > 10:
        a_mr = mass_redistribute(a, n_moves=800, time_budget=min(rem * 0.2, 20.0))
        obj = compute_obj(a_mr)
        if obj < best_obj:
            best_obj = obj
            best_a = a_mr.copy()
            a = a_mr.copy()
            print(f"    [Combined] Mass Redist: {best_obj:.8f}")

    # Step 5: LP Polish
    rem = _remaining()
    if rem > 20:
        a_lp = lp_polish(a, time_budget=min(rem * 0.35, 40.0))
        obj = compute_obj(a_lp)
        if obj < best_obj:
            best_obj = obj
            best_a = a_lp.copy()
            a = a_lp.copy()
            print(f"    [Combined] LP: {best_obj:.8f}")

    # Step 6: Final adaptive GD
    rem = _remaining()
    if rem > 10:
        a_final = adaptive_gd(a, time_budget=min(rem * 0.5, 25.0), mode='peak')
        obj = compute_obj(a_final)
        if obj < best_obj:
            best_obj = obj
            best_a = a_final.copy()
            print(f"    [Combined] Final GD: {best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multiple optimization strategies:
    - Adaptive log-sum-exp gradient descent
    - Spectral-domain mass redistribution
    - LP polishing with active set
    - Multi-n exploration with asymmetric initializations
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Intensive adaptive GD from parent ===
    print(f"\n=== Phase 1: Adaptive GD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.2, 200.0)
    a_gd = adaptive_gd(best_seq, time_budget=alloc, mode='adaptive')
    obj = compute_obj(a_gd)
    if obj < best_score:
        best_score = obj
        best_seq = a_gd.copy()
        print(f"  ** Adaptive GD: obj={best_score:.8f} **")
    else:
        print(f"  Adaptive GD: obj={obj:.8f}")

    # === Phase 2: Spectral shaping ===
    print(f"\n=== Phase 2: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_spec = spectral_shaping(best_seq, alpha=0.4, iters=15)
        obj = compute_obj(a_spec)
        if obj < best_score:
            best_score = obj
            best_seq = a_spec.copy()
            print(f"  ** Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Spectral: obj={obj:.8f}")

    # === Phase 3: LP Polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 100.0)
        a_lp = lp_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj:.8f}")

    # === Phase 4: Mass redistribution ===
    print(f"\n=== Phase 4: Mass Redistribution ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.1, 30.0)
        a_mr = mass_redistribute(best_seq, n_moves=2000, time_budget=alloc)
        obj = compute_obj(a_mr)
        if obj < best_score:
            best_score = obj
            best_seq = a_mr.copy()
            print(f"  ** Mass Redist: obj={best_score:.8f} **")
        else:
            print(f"  Mass Redist: obj={obj:.8f}")

    # === Phase 5: Second GD round (peak mode) ===
    print(f"\n=== Phase 5: Peak GD Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 80.0)
        a_gd2 = adaptive_gd(best_seq, time_budget=alloc, mode='peak')
        obj = compute_obj(a_gd2)
        if obj < best_score:
            best_score = obj
            best_seq = a_gd2.copy()
            print(f"  ** Peak GD2: obj={best_score:.8f} **")
        else:
            print(f"  Peak GD2: obj={obj:.8f}")

    # === Phase 6: Asymmetric push + re-optimize ===
    print(f"\n=== Phase 6: Asymmetric Push ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for strength in [0.2, 0.3, 0.5]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_asm = asymmetrize(best_seq, strength=strength)
            # Brief optimization after asymmetrization
            a_opt = adaptive_gd(a_asm, time_budget=min(remaining * 0.15, 25.0), mode='adaptive')
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** Asym({strength}): obj={best_score:.8f} **")
            else:
                print(f"  Asym({strength}): obj={obj:.8f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr + 50, n_curr + 100, n_curr + 200, n_curr + 400,
        n_curr - 50, n_curr - 100,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and n != n_curr]
    n_options = n_options[:8]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        # Resample and optimize
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 50.0)
        a_opt = adaptive_gd(a_try, time_budget=alloc, mode='adaptive')
        obj_gd = compute_obj(a_opt)

        # Also try LP on the GD result
        if deadline - time.monotonic() > 20:
            a_lp = lp_polish(a_opt, time_budget=min(deadline - time.monotonic() - 1, 15.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < obj_gd:
                a_opt = a_lp
                obj_gd = obj_lp

        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_gd:.8f}")

    # === Phase 8: Random asymmetric restarts ===
    print(f"\n=== Phase 8: Random Asymmetric Restarts ===")
    n_curr = len(best_seq)
    n_restarts = max(2, int((deadline - time.monotonic()) / 50.0))
    n_restarts = min(n_restarts, 8)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Try a couple of n values for restarts
        if r % 2 == 0:
            n_r = n_curr
        else:
            n_r = n_curr + 100

        a_r = random_init_asymmetric(n_r, rng)
        alloc = min(remaining * 0.2, 30.0)
        a_opt = adaptive_gd(a_r, time_budget=alloc, mode='adaptive')
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r} (n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}): obj={obj_r:.8f}")

    # === Phase 9: Final combined polish ===
    print(f"\n=== Phase 9: Final Combined Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.6, 120.0)
        a_final = combined_optimize(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final Combined: obj={best_score:.8f} **")
        else:
            print(f"  Final Combined: obj={obj_final:.8f}")

    # === Phase 10: Ultimate LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_ult = lp_polish(best_seq, time_budget=min(remaining * 0.5, 40.0))
        obj_ult = compute_obj(a_ult)
        if obj_ult < best_score:
            best_score = obj_ult
            best_seq = a_ult.copy()
            print(f"  ** Ultimate LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
