import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def conv_fft(a):
    """Compute full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)

    cross = cross_full[n - 1:2 * n - 1]
    scale = 2.0 * sum_bp ** (1.0 / p - 1.0)
    grad = scale * cross
    return obj, grad


def adam_gd(a, time_budget=60.0, p_start=6.0, p_end=128.0, lr_init=0.5, verbose=False):
    """Adam gradient descent with adaptive smoothing parameter."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-11:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = lr_init
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - start) / time_budget)
        p = p_start * (p_end / p_start) ** (elapsed_frac ** 0.7)

        _, grad = smoothed_obj_grad(a, p)
        grad = grad - np.mean(grad)

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-11:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.005, 5.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.97
            if stall > 200:
                # Perturb and restart
                rng_local = np.random.default_rng(it + 12345)
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-11:
                    a = a / s
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init
                stall = 0
                max_stall = 200
        if lr < 1e-7:
            break

        if verbose and it % 1000 == 0:
            print(f"    Adam it{it}: p={p:.1f}, obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def subgradient_chase(a, time_budget=60.0, verbose=False):
    """Subgradient method targeting the peak of the convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-11:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    k = 0
    avg_obj = 0.0

    while time.monotonic() - start < time_budget:
        b = conv_fft(a)
        k_star = int(np.argmax(b))

        # Subgradient of max(b) at k_star: grad[i] = 2*a[k_star - i] (valid range)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        # Also the "other side" contribution
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]  # Wait, this double counts
        # Actually grad[i] = 2*a[k_star - i] already accounts for both sides
        # Let me redo: ∂b[k]/∂a[i] = a[k-i] + a[i-k] but since b[k] = sum a[j]*a[k-j],
        # ∂b[k]/∂a[i] = a[k-i] + (contribution from a[i] appearing as a[k-j] where j=k-i)
        # = a[k-i] + a[k-i] = 2*a[k-i] (when i != k-i) or 2*a[i] (when i = k-i)
        # So grad[i] = 2*a[k_star - i] (with a[m]=0 for invalid m)
        # This is what I had. Let me just use it.

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 >= 0 and j >= 0 and j < n:
                grad[i] = 2.0 * a[j]

        # Average the subgradient (for Nesterov-type averaging)
        g_avg = (k * g_avg + grad) / (k + 1) if k > 0 else grad.copy()
        k += 1

        # Step size: O(1/sqrt(t))
        t = k
        step_size = 1.0 / (np.sqrt(t) * np.linalg.norm(g_avg) + 1e-10)
        step_size = min(step_size, 0.5)

        a_new = a - step_size * g_avg
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-11:
            continue
        a_new = a_new / s_new

        # Also try the averaged iterate
        a_avg = (k * best_a + a_new) / (k + 1) if k > 0 else a_new.copy()
        a_avg = np.maximum(a_avg, 0)
        s_avg = float(np.sum(a_avg))
        if s_avg > 1e-11:
            a_avg = a_avg / s_avg

        obj_new = compute_obj(a_new)
        obj_avg = compute_obj(a_avg)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()

        if verbose and k % 1000 == 0:
            print(f"    SubGrad it{k}: obj={best_obj:.8f}, k*={k_star}")

    return best_a


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g/sum(g) or None."""
    n = len(a)
    a = a / np.sum(a)

    # Try HiGHS via scipy
    try:
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    # Try cvxpy
    try:
        import cvxpy as cp
        A = build_conv_matrix(a)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1, M_var >= 0]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass

    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-11:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t = 0.3
        best_t_obj = np.inf
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-11:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 3 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall > 10:
                break

    return best_a


def trust_region_qp(a, M_target, n_iter=50):
    """
    Trust region: minimize ||g - a||^2 s.t. conv(a, g) <= M_target, sum(g)=1, g>=0.
    This is a QP with linear constraints.
    """
    n = len(a)
    a = a / np.sum(a)

    try:
        import cvxpy as cp
        A = build_conv_matrix(a)
        g_var = cp.Variable(n, nonneg=True)
        objective = cp.Minimize(cp.sum_squares(g_var - a))
        constraints = [A @ g_var <= M_target, cp.sum(g_var) == 1]
        prob = cp.Problem(objective, constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass

    # Fallback: use LP solution scaled to be close to a
    g_lp = solve_lp_minpeak(a)
    if g_lp is not None:
        # Blend with a for trust region effect
        return 0.5 * a + 0.5 * g_lp

    return None


def multi_temp_chase(a, time_budget=60.0, verbose=False):
    """
    Chase the peak with multiple temperature levels for the soft-max.
    Alternates between different p values to escape local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-11:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    p_levels = [4, 8, 16, 32, 64, 128]
    idx = 0

    while time.monotonic() - start < time_budget:
        p = p_levels[idx % len(p_levels)]
        # Short burst of Adam at this p
        sub_budget = min(10.0, time_budget - (time.monotonic() - start))
        if sub_budget < 2:
            break
        a_new = adam_gd(a, time_budget=sub_budget, p_start=p, p_end=p * 2, lr_init=0.3)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose:
                print(f"    MultiTemp p={p}: obj={best_obj:.8f}")
        idx += 1

    return best_a


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = int(rng.integers(0, 8))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped (edge-concentrated)
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 8 + 0.3
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            width = max(n // 8, 2)
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * width ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.3
        a = a / np.sum(a)
    elif mode == 5:
        # Power law decay from center
        center = n / 2.0
        x = np.abs(np.arange(n) - center) + 1
        a = 1.0 / x ** 0.8
        a = a / np.sum(a)
    elif mode == 6:
        # Random with smooth noise
        a = np.ones(n) / n + 0.1 * rng.normal(0, 1, n)
        a = np.maximum(a, 0)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    else:
        # raised cosine
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.5 * np.cos(x) + 0.1
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-11:
        a_new = a_new / s
    return a_new
