"""
Optimized step function search: LP-alternating with momentum, diverse initializations,
active-set exploitation, length sweep, and subgradient polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a, prefer_solvers=None):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, active_indices) or (None, None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None, None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = prefer_solvers if prefer_solvers else [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    # Find active constraints
                    conv_val = A @ g_val
                    active = np.where(conv_val >= M_val - 1e-6)[0]
                    return g_val, M_val, active
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                conv_val = A @ g_val
                active = np.where(conv_val >= M_val - 1e-6)[0]
                return g_val, M_val, active
    except Exception:
        pass

    return None, None, None


def lp_alternating_momentum(a, time_budget=300.0, verbose=False, momentum_beta=0.5):
    """
    LP alternating with momentum and adaptive step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    momentum = np.zeros(n)
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        g_lp, M_lp, active_idx = solve_lp_for_response(a)
        if g_lp is None:
            break

        # Adaptive step candidates
        step_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0 < last_t < 1.0:
            step_candidates = sorted(set(step_candidates + [last_t * 0.7, last_t * 1.3]))
            step_candidates = [t for t in step_candidates if 0.05 <= t <= 1.5]

        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in step_candidates:
            # Apply momentum to the direction
            direction = g_lp - a
            momentum = momentum_beta * momentum + (1 - momentum_beta) * direction
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                if elapsed > 5:
                    print(f"    LP-mom: obj={best_obj:.8f}, t={best_t_val:.3f}, "
                          f"stall={stall}, time={elapsed:.0f}s")
        else:
            stall += 1
            momentum *= 0.5
            if stall % 10 == 0:
                # Perturbation to escape
                scale = 0.02 + 0.01 * min(stall // 10, 10)
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
                last_t = 0.5

    return best_a


def subgradient_polish(a, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method targeting active peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()  # Nesterov lookahead
    y = a.copy()  # actual point
    beta = 0.7
    lr = 0.05
    momentum = np.zeros(n)

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution at lookahead point
        conv = compute_conv(x)
        max_val = float(np.max(conv))
        # Get active peaks (within 10% of max)
        threshold = max_val * 0.9
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        if len(active) > 30:
            active = np.argsort(conv)[::-1][:30]

        # Compute subgradient: average of gradients of active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            # d(a*a)[k]/da[i] = a[k-i] if 0 <= k-i < n
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                # grad[i] += a[k-i] for i in [i_min, i_max]
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += a[k - indices]
        grad /= len(active)

        # Center gradient (project to hyperplane sum=0)
        grad_centered = grad - np.mean(grad)

        # Momentum step
        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        # Nesterov update
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        # Adaptive learning rate
        if it % 200 == 199:
            lr = max(lr * 0.9, 0.001)

    return best_a


def coordinate_adjust(a, time_budget=60.0):
    """
    Pairwise coordinate adjustment: move mass between pairs to reduce peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for iteration in range(500):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        # Identify positions contributing most to the peak
        # (a*a)[k_star] = sum_j a[j]*a[k_star-j]
        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]

        # Sort by contribution (highest first)
        sorted_idx = np.argsort(contributions)[::-1]

        # Try moving mass from high-contributors to low-contributors
        for frac in [0.005, 0.01, 0.02, 0.05]:
            done = False
            for i in sorted_idx[:20]:
                if done or time.monotonic() - start > time_budget:
                    break
                # Find a position j that contributes least and is "far" from k_star-i
                target = (k_star - int(i) + n // 2) % n
                if target == i:
                    target = (i + n // 2) % n
                delta = frac * a[i]
                if delta <= 0 or delta > a[i]:
                    continue
                a[i] -= delta
                a[target] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    done = True
                else:
                    a[i] += delta
                    a[target] -= delta

            if done:
                improved = True

        if not improved:
            # Random pair swap
            rng = np.random.default_rng(int(time.monotonic() * 1e4) % (2**31))
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * min(a[i], a[j])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and iteration > 100:
            break

    return best_a


def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.1
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bilinear':
        # Linear ramp (asymmetric)
        a = np.linspace(1, 2, n)
        return a / np.sum(a)
    elif kind == 'plateau':
        # High in middle 60%, low at edges
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a / np.sum(a)
    elif kind == 'periodic':
        # Two peaks
        a = np.ones(n) * 0.5
        a[n//4] = 2.0
        a[3*n//4] = 2.0
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: diverse initializations + LP-momentum + subgradient + coordinate polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Diverse Initializations (80s) ===
    print(f"\n=== Phase 1: Diverse Initializations (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    init_budget = min(80.0, remaining * 0.06)

    init_kinds = ['cosine', 'sine_taper', 'gaussian', 'random', 'plateau', 'bilinear', 'periodic']
    per_init = max(5.0, init_budget / len(init_kinds))

    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_init = generate_initialization(kind, n_parent, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    # Also try a few random inits
    for _ in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_init = generate_initialization('random', n_parent, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Random init: obj={obj_init:.8f} **")

    a = best_seq.copy()
    print(f"  Best after Phase 1: obj={best_score:.8f}")

    # === Phase 2: LP Alternating with Momentum (600s) ===
    print(f"\n=== Phase 2: LP Alternating with Momentum ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(600.0, remaining * 0.45)

    # Multiple rounds with different momentum
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        beta_vals = [0.3, 0.5, 0.7, 0.9]
        beta = beta_vals[round_idx % len(beta_vals)]
        alloc = min(lp_budget / (4 - round_idx), remaining * 0.6)
        print(f"  LP Round {round_idx} (beta={beta}): ")
        a_new = lp_alternating_momentum(a, time_budget=alloc, verbose=True,
                                         momentum_beta=beta)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            # If no improvement, perturb
            if round_idx > 0:
                scale = 0.03
                a = best_seq * np.exp(rng.normal(0, scale, n_parent))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    # === Phase 3: Subgradient Polish (200s) ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        sg_budget = min(200.0, remaining * 0.2)
        a_sg = subgradient_polish(a, time_budget=sg_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 4: Coordinate Adjustment (100s) ===
    print(f"\n=== Phase 4: Coordinate Adjustment ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        cd_budget = min(100.0, remaining * 0.12)
        a_cd = coordinate_adjust(a, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  Coord: obj={best_score:.8f} **")
        else:
            print(f"  Coord: obj={obj_cd:.8f}")

    # === Phase 5: Length Sweep (250s) ===
    print(f"\n=== Phase 5: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try nearby lengths
    n_options = sorted(set([
        max(100, n_best - 100),
        max(100, n_best - 50),
        max(100, n_best - 25),
        n_best,
        n_best + 25,
        n_best + 50,
        n_best + 100,
        n_best + 200,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: LP + subgradient
        alloc_lp = min(30.0, remaining * 0.4)
        alloc_sg = min(20.0, remaining * 0.2)

        a_opt = lp_alternating_momentum(a_try, time_budget=alloc_lp,
                                         momentum_beta=0.5)
        a_opt = subgradient_polish(a_opt, time_budget=alloc_sg)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final LP rounds from best (remaining time) ===
    print(f"\n=== Phase 6: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    a_final = best_seq.copy()

    # Multiple short LP rounds
    while remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_new = lp_alternating_momentum(a_final, time_budget=alloc,
                                         momentum_beta=0.5)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            # Perturb and try again
            scale = 0.015
            a_final = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)
            print(f"  Final LP: obj={obj:.8f} (no improvement, perturbing)")

        remaining = deadline - time.monotonic()

    # === Phase 7: Very final coordinate polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_adjust(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
