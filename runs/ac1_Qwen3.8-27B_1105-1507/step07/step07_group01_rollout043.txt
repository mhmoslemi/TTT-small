        if max_c < best_val:
            best_val = max_c
            best_g = g.copy()
        
        # Polyak step: lr = (upper_bound - lower_bound) / ||sg||^2
        # Subgradient: weighted average of active constraint rows
        # Active: constraints where c_k is close to max
        threshold = max_c - 0.1 * max_c  # 10% tolerance
        active = c >= threshold
        n_active = np.sum(active)
        if n_active == 0:
            active[np.argmax(c)] = True
            n_active = 1
        
        # Subgradient: average of active rows of A
        # A_{k,j} = a_{k-j} if 0 <= k-j < n
        # sg[j] = (1/n_active) * sum_{k active} a_{k-j} (valid)
        # This is the correlation of (active/n_active) with a
        y = active.astype(float) / n_active
        sg = np.correlate(y, a, mode='valid')
        sg_norm_sq = np.sum(sg * sg)
        
        if sg_norm_sq < 1e-15:
            break
        
        # Polyak step
        gap = best_val - (1.0 / (2*n - 1))  # lower bound for normalized
        if gap < 1e-10:
            break
        polyak_lr = gap / sg_norm_sq
        step = min(lr, polyak_lr)
        
        g_new = simplex_project(g - step * sg)
        
        # Check if improvement
        fg_new = rfft(g_new, fa_size)
        c_new = irfft(fa * fg_new, fa_size)[:2*n-1]
        if np.max(c_new) < best_val:
            g = g_new
        
        # Decay lr
        if it % 500 == 0:
            lr = max(0.01, lr * 0.95)
    
    return best_g

def coordinate_descent(a, time_budget=10.0):
    """
    Coordinate descent: try adjusting pairs of elements to reduce max convolution.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    
    rng = np.random.default_rng(42)
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Pick two random coordinates
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        # Try to move mass from j to i (or vice versa)
        # a_i' = a_i + delta, a_j' = a_j - delta
        # Objective as function of delta: 2*n*max(conv(a'), 1)
        # conv(a')_k = conv(a)_k + delta*(a_{k-i} - a_{k-j}) (approximately, for valid indices)
        
        # Binary search for optimal delta
        # The max is a piecewise linear function of delta
        # We want to find delta that minimizes max(conv + delta * d)
        # where d_k = a_{k-i} - a_{k-j} (with appropriate indexing)
        
        conv = compute_conv(a)
        d = np.zeros(2*n-1)
        # Contribution of changing a_i by +delta and a_j by -delta
        # conv_new[k] = conv[k] + delta * (a[k-i] - a[k-j]) for valid k
        # a[k-i] is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
        # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
        lo_i = i
        hi_i = min(2*n-2, i + n - 1)
        lo_j = j  
        hi_j = min(2*n-2, j + n - 1)
        
        for k in range(lo_i, hi_i + 1):
            d[k] += a[k - i]
        for k in range(lo_j, hi_j + 1):
            d[k] -= a[k - j]
        
        # Now find delta that minimizes max(conv + delta * d)
        # subject to a_i + delta >= 0 and a_j - delta >= 0
        # i.e., -a_i <= delta <= a_j
        
        delta_lo = -a[i]
        delta_hi = a[j]
        
        if delta_lo >= delta_hi:
            continue
        
        # The function f(delta) = max_k (conv[k] + delta * d[k]) is convex in delta
        # Find minimum by ternary search or by checking where subgradient changes sign
        # Use binary search on the subgradient
        
        def f_at(delta):
            vals = conv + delta * d
            return np.max(vals)
        
        # Ternary search (fast enough for 1D)
        lo, hi = delta_lo, delta_hi
        for _ in range(50):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            if f_at(m1) < f_at(m2):
                hi = m2
            else:
                lo = m1
        
        delta_opt = (lo + hi) / 2
        
        # Apply
        a_try = a.copy()
        a_try[i] += delta_opt
        a_try[j] -= delta_opt
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try
            a = a_try
    
    return best_a

def enforce_symmetry(a):

def lp_alternating_fast(a, time_budget=30.0, use_subgrad=False):
    """LP alternating with fast solver."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        improved = False

            a_try = (1 - t) * a + t * g_new

            if st < 1e-10:

            a_try = a_try / st
            obj_try = compute_obj(a_try)

                break

            for t in [0.05, 0.1, 0.15]:
                a_try = (1 - t) * a + t * g_new
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st

                if obj_try < best_obj - 1e-12:

def smooth_max_gradient_descent(a, time_budget=40.0, T_init=0.0003, T_min=0.00001):
    """Gradient descent with smooth max and Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = T_init
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    
    fa_size = 1
    while fa_size < 2 * n:
        fa_size <<= 1
    
    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute smooth max and gradient
        fa = rfft(a, fa_size)
        conv = irfft(fa * fa, fa_size)[:2*n-1]
        
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        
        # Gradient: grad[i] = sum_j weights[i+j] * a[j]
        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)
        
        # Nesterov
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break
        
        if it > 100:
            T = max(T_min, T_init * (0.998 ** it))
        
        # Adaptive lr
        if stall == 0 and it % 200 == 0 and lr < 0.05:
            lr *= 1.02
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: fast LP (HiGHS) alternating, analytic coordinate descent,
smooth-max gradient with Nesterov momentum, symmetry exploitation,
and aggressive multi-n exploration with adaptive time allocation.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear auto-convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_pair(a, b):
    """Compute conv(a, b) via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        A[j:j + end, j] = a[:end]
    return A


def solve_lp_direct(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0 using scipy HiGHS."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def subgradient_minpeak(a, time_budget=3.0):
    """
    Projected subgradient method for min max_k conv(a,g)_k s.t. g in simplex.
    Fast alternative to LP when LP is too slow.
    """
    n = len(a)
    a = a / np.sum(a)
    fa_size = 1
    while fa_size < 2 * n:
        fa_size <<= 1
    fa = rfft(a, fa_size)

    g = np.ones(n) / n
    best_g = g.copy()
    best_val = np.max(compute_conv_pair(a, g))
    lb = 1.0 / (2 * n - 1)

    start = time.monotonic()
    lr = 0.2

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        fg = rfft(g, fa_size)
        c = irfft(fa * fg, fa_size)[:2 * n - 1]
        max_c = np.max(c)

        if max_c < best_val:
            best_val = max_c
            best_g = g.copy()

        # Subgradient: average active constraint rows
        threshold = max_c - 0.2 * max_c
        active = (c >= threshold).astype(float)
        n_act = np.sum(active)
        if n_act < 1:
            active[np.argmax(c)] = 1.0
            n_act = 1
        active /= n_act

        sg = np.correlate(active, a, mode='valid')
        sg_norm_sq = np.sum(sg * sg)
        if sg_norm_sq < 1e-15:
            break

        gap = best_val - lb
        if gap < 1e-12:
            break
        polyak_lr = gap / sg_norm_sq
        step = min(lr, polyak_lr)

        g_new = simplex_project(g - step * sg)

        # Check improvement
        fg_new = rfft(g_new, fa_size)
        c_new = irfft(fa * fg_new, fa_size)[:2 * n - 1]
        if np.max(c_new) < best_val:
            g = g_new

        if it % 1000 == 0:
            lr = max(0.02, lr * 0.95)

    return best_g


def coordinate_descent_analytic(a, time_budget=10.0):
    """
    Coordinate descent using analytic direction for 2-element moves.
    For pair (i,j), direction d[k] = 2*a[k-i] - 2*a[k-j] (valid indices).
    Solves 1D convex problem: min_delta max_k (conv[k] + delta*d[k]).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(777)

    conv = compute_conv(a)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        # Direction: d[k] = 2*a[k-i]*[0<=k-i<n] - 2*a[k-j]*[0<=k-j<n]
        d = np.zeros(2 * n - 1)
        lo_i = i
        hi_i = min(2 * n - 2, i + n - 1)
        for k in range(lo_i, hi_i + 1):
            d[k] += 2.0 * a[k - i]
        lo_j = j
        hi_j = min(2 * n - 2, j + n - 1)
        for k in range(lo_j, hi_j + 1):
            d[k] -= 2.0 * a[k - j]

        # Find delta that minimizes max_k (conv[k] + delta * d[k])
        # subject to: delta >= -a[i], delta <= a[j]
        # Use binary search on subgradient
        delta_lo = -a[i]
        delta_hi = a[j]

        if delta_lo >= delta_hi:
            continue

        # Check if already at boundary
        def obj_at(delta):
            return float(np.max(conv + delta * d))

        # Subgradient at a point is d[k*] where k* = argmax
        def subgrad_at(delta):
            vals = conv + delta * d
            k_star = int(np.argmax(vals))
            return d[k_star]

        # Binary search
        sg_lo = subgrad_at(delta_lo)
        sg_hi = subgrad_at(delta_hi)

        if sg_lo >= 0:
            delta_opt = delta_lo
        elif sg_hi <= 0:
            delta_opt = delta_hi
        else:
            lo, hi = delta_lo, delta_hi
            for _ in range(40):
                mid = (lo + hi) / 2
                sg_mid = subgrad_at(mid)
                if sg_mid > 0:
                    hi = mid
                else:
                    lo = mid
            delta_opt = (lo + hi) / 2

        if abs(delta_opt) < 1e-12:
            continue

        # Apply and verify
        a[i] += delta_opt
        a[j] -= delta_opt
        a[i] = max(0.0, a[i])
        a[j] = max(0.0, a[j])

        conv = compute_conv(a)
        obj = 2.0 * n * float(np.max(conv))

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            # Revert
            a[i] -= delta_opt
            a[j] += delta_opt
            a[i] = max(0.0, a[i])
            a[j] = max(0.0, a[j])
            conv = compute_conv(a)
            no_improve += 1
            if no_improve > 200:
                # Try random perturbation
                idx = rng.integers(n)
                if a[idx] > 1e-6:
                    a[idx] *= (1.0 + rng.normal(0, 0.1))
                    a[idx] = max(0.0, a[idx])
                no_improve = 0

        # Renormalize periodically
        if (no_improve + int((time.monotonic() - start) * 100)) % 50 == 0:
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
                conv = compute_conv(a)

    return best_a


def smooth_max_gd(a, time_budget=30.0, T_init=0.0003, T_min=0.00001):
    """
    Smooth max gradient descent with Nesterov momentum and adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    fa_size = 1
    while fa_size < 2 * n:
        fa_size <<= 1

    for it in range(30000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, fa_size)
        conv = irfft(fa * fa, fa_size)[:2 * n - 1]

        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-300:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights = exp_vals / total

        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-7:
                    break

        if it > 200:
            T = max(T_min, T_init * (0.997 ** it))
        if stall == 0 and it % 300 == 0 and lr < 0.04:
            lr *= 1.03

    return best_a


def lp_alternating(a, time_budget=30.0):
    """LP alternating optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP or use subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        g_new, M_lp = solve_lp_direct(a)
        if g_new is None:
            g_new = subgradient_minpeak(a, time_budget=min(2.0, remaining * 0.2))

        if g_new is None:
            break

        # Line search
        improved = False
        ts = [0.4, 0.6, 0.3, 0.5, 0.2, 0.7, 0.1, 0.8]
        for t in ts:
            a_try = (1.0 - t) * a + t * g_new
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Perturb
                rng = np.random.default_rng(it * 31 + 7)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def enforce_symmetry(a):
    """Make sequence palindromic."""
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = np.sum(a_sym)
    if s > 1e-10:
        a_sym = a_sym / s
    return a_sym


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + coordinate descent + smooth max GD +
    symmetry + multi-n exploration with adaptive time allocation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_score = np.inf
    best_seq = None
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_seq = parent.copy()
                best_score = compute_obj(parent)
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    # === Phase 1: Intensive LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating ===")
    for round_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(50.0, remaining * 0.12)
        a_new = lp_alternating(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  LP R{round_idx}: obj={best_score:.8f}")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")

    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.1)
        a_cd = coordinate_descent_analytic(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f}")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 3: Smooth Max GD ===
    print(f"\n=== Phase 3: Smooth Max GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(40.0, remaining * 0.08)
        a_sm = smooth_max_gd(best_seq, time_budget=alloc, T_init=0.0003, T_min=0.00001)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  SM: obj={best_score:.8f}")
        else:
            print(f"  SM: obj={obj_sm:.8f}")

    # === Phase 4: Symmetry ===
    print(f"\n=== Phase 4: Symmetry ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sym = enforce_symmetry(best_seq)
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  Sym: obj={best_score:.8f}")
        else:
            print(f"  Sym: obj={obj_sym:.8f} (worse)")
        # If close, try optimizing symmetric version
        if obj_sym < best_score + 0.005 and remaining > 40:
            alloc = min(30.0, remaining * 0.06)
            a_sym_opt = lp_alternating(a_sym, time_budget=alloc)
            obj_so = compute_obj(a_sym_opt)
            if obj_so < best_score:
                best_score = obj_so
                best_seq = a_sym_opt.copy()
                print(f"  Sym Opt: obj={best_score:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 150), max(300, n_best - 75),
        max(300, n_best - 30), n_best,
        n_best + 30, n_best + 75, n_best + 150, n_best + 300,
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # LP optimize
        alloc_lp = min(remaining * 0.08, 20.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp)

        # Coordinate descent polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc_cd = min(remaining * 0.04, 10.0)
            a_opt2 = coordinate_descent_analytic(a_opt, time_budget=alloc_cd)
            if compute_obj(a_opt2) < compute_obj(a_opt):
                a_opt = a_opt2

        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    for polish in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # LP
        alloc = min(remaining * 0.15, 30.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

        # CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.1, 15.0)
            a_cd = coordinate_descent_analytic(best_seq, time_budget=alloc)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()

        # SM
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.08, 12.0)
            a_sm = smooth_max_gd(best_seq, time_budget=alloc,
                                  T_init=0.0001, T_min=0.000005)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()

        print(f"  P{polish}: obj={best_score:.8f}, n={len(best_seq)}")

    # === Phase 7: Random Restarts (if time remains) ===
    print(f"\n=== Phase 7: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Different initialization strategies
        n_r = int(min(1500, max(400, n_best + rng.integers(-100, 101))))
        if r % 3 == 0:
            a_r = np.ones(n_r) / n_r
        elif r % 3 == 1:
            x = np.linspace(0, 1, n_r)
            a_r = np.exp(-4 * (x - 0.5) ** 2) + 0.1
            a_r = a_r / np.sum(a_r)
        else:
            a_r = rng.dirichlet(np.ones(n_r) * 0.5)

        alloc = min(remaining * 0.08, 20.0)
        a_opt = lp_alternating(a_r, time_budget=alloc * 0.6)

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = coordinate_descent_analytic(a_opt, time_budget=min(alloc * 0.4, remaining * 0.05))

        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}): obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
