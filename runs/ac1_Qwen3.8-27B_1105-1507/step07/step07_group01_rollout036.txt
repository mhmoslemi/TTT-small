"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses power-smooth max gradient descent, LP alternating, peak-flattening,
and coordinate perturbation search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(rho[cond])
    theta = (cssv[idx - 1] - 1.0) / idx
    return np.maximum(v - theta, 0)


def power_smooth_max(conv, p):
    """Compute power-based smooth max and softmax-like weights."""
    max_c = float(np.max(conv))
    if max_c <= 0:
        return 0.0, np.zeros_like(conv)
    shifted = conv / max_c
    shifted = np.clip(shifted, 0.0, 1.0)
    weights = shifted ** p
    total = float(np.sum(weights))
    if total < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return max_c, w
    weights = weights / total
    smooth_val = max_c * (total) ** (1.0 / p)
    return smooth_val, weights


def compute_gradient(weights, a):
    """
    Compute gradient of smooth_max(conv(a,a)) w.r.t. a.
    grad[i] = sum_k weights[k] * a[k-i] for valid indices.
    """
    n = len(a)
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for i in range(n):
        j_max = min(n, n_full - i)
        if j_max > 0:
            grad[i] = float(np.dot(weights[i:i + j_max], a[:j_max]))
    return grad

    Gradient targeting the top-k peaks specifically.
    """
    n = len(a)
    n_full = 2 * n - 1
    top_indices = np.argsort(conv)[-k_top:]
    w_top = np.zeros(n_full)
    for idx in top_indices:
        w_top[idx] = 1.0 / k_top
    return compute_gradient(w_top, a)


def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0
        g_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        # Try several step sizes
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                stall_count = 0
                break

        if not improved:
            stall_count += 1
            if stall_count > 50:
                # Perturb
                rng = np.random.default_rng(it + n)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0

    return best_a


def solve_lp_direction(a):
    """Solve LP to find direction g minimizing peak of conv(a,g)."""
    n = len(a)
    try:
        import cvxpy as cp
        a = a / np.sum(a)
        # Build constraint matrix: (A @ g)[k] = conv(a, g)[k]
        # conv(a, g)[k] = sum_i a[i] * g[k-i] for valid indices
        # As a matrix: A[k, j] = a[k-j] if 0<=k-j<n, else 0
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            # A[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
            for k in range(j, min(j + n, 2 * n - 1)):
                A[k, j] = a[k - j]

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in ['CLARABEL', 'SCS', 'ECOS', 'MOSEK']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except ImportError:
        pass

    # Fallback: use scipy linprog
    try:
        from scipy.optimize import linprog
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            for k in range(j, min(j + n, 2 * n - 1)):
                A[k, j] = a[k - j]

        # min M s.t. A@g - M*1 <= 0, sum(g) = 1, g >= 0
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def gradient_descent_power(a, time_budget=60.0, p_init=8, p_max=64):
    """
    Gradient descent using power-based smooth max with adaptive exponent.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    p = p_init
    stall = 0

    for it in range(10000):

            break

        conv = compute_conv(a)
        smooth_val, weights = power_smooth_max(conv, p)
        grad = compute_gradient(weights, a)

        # Center gradient (for simplex constraint)
        grad = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Adaptively increase p to sharpen the max
            if it % 100 == 0:
                p = min(p * 1.1, p_max)
                lr = min(lr * 1.02, 0.05)
        else:
            stall += 1
            if stall > 20:
                lr *= 0.8
                p = max(4, p * 0.9)
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating, power-smooth max gradient descent, peak flattening,
subgradient with Polyak step, coordinate perturbation, and multi-n exploration.
Key improvements: power-based smooth max (more flexible than log-sum-exp),
top-k peak targeting, Polyak step-size subgradient, and iterative polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(rho[cond])
    theta = (cssv[idx - 1] - 1.0) / idx
    return np.maximum(v - theta, 0)


def power_smooth_max(conv, p):
    """Compute power-based smooth max and weights for gradient."""
    max_c = float(np.max(conv))
    if max_c <= 0:
        return 0.0, np.zeros_like(conv)
    shifted = conv / max_c
    shifted = np.clip(shifted, 0.0, 1.0)
    weights = shifted ** p
    total = float(np.sum(weights))
    if total < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return max_c, w
    weights = weights / total
    smooth_val = max_c * (total) ** (1.0 / p)
    return smooth_val, weights


def compute_gradient(weights, a):
    """
    Gradient of sum_k weights[k]*conv_k(a) w.r.t. a.
    grad[i] = sum_k weights[k] * a[k-i] for valid (k-i) in [0, n-1].
    """
    n = len(a)
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for i in range(n):
        j_max = min(n, n_full - i)
        if j_max > 0:
            grad[i] = float(np.dot(weights[i:i + j_max], a[:j_max]))
    return grad


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            A[k, j] = a[k - j]
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in ['CLARABEL', 'SCS', 'ECOS', 'MOSEK', 'PDLP']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except ImportError:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization with adaptive step sizes."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    stall_count = 0

        if g_lp is None:
            break

        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st

            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                stall_count = 0
                break

        if not improved:
            stall_count += 1
            if stall_count > 40:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))

def gradient_descent_power(a, time_budget=60.0, p_init=8, p_max=128):
    """Gradient descent with power-based smooth max and Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.008
    momentum = 0.9
    v = np.zeros(n)
    p = p_init
    stall = 0

    for it in range(15000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        _, weights = power_smooth_max(conv, p)
        grad = compute_gradient(weights, a)
        grad = grad - np.mean(grad)

        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 200 == 0:
                p = min(p * 1.05, p_max)
                lr = min(lr * 1.01, 0.03)
        else:
            stall += 1
            if stall > 25:
                lr *= 0.8
                p = max(4, p * 0.95)
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break

    return best_a


def subgradient_polyak(a, time_budget=40.0):
    """
    Subgradient method with Polyak step size.
    At each step, find k*=argmax(conv), take subgradient step of conv_{k*}.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    f_star_approx = best_obj  # running best
    avg_a = a.copy()
    count = 0

    for it in range(8000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = float(conv[k_star])

        # Subgradient of conv_{k*} w.r.t. a
        subgrad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                subgrad[i] = a[j]

        # Polyak step size
        current_obj = 2.0 * n * max_val  # objective value (sum=1)
        gap = current_obj - f_star_approx
        norm_sq = float(np.dot(subgrad, subgrad))
        if norm_sq < 1e-20:
            eta = 0.001
        else:
            eta = max(0.001, min(0.1, gap / norm_sq)) if gap > 0 else 0.001

        # Center for simplex
        subgrad_centered = subgrad - np.mean(subgrad)
        a_new = simplex_project(a - eta * subgrad_centered)

        # Averaging
        avg_a = (count * avg_a + a_new) / (count + 1)
        count += 1

        obj_new = compute_obj(a_new)
        obj_avg = compute_obj(avg_a)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            f_star_approx = best_obj
        elif obj_avg < best_obj:
            best_obj = obj_avg
            best_a = avg_a.copy()
            a = a_new
            f_star_approx = best_obj
        else:
            a = a_new

    # Return best of averaged and point
    obj_final = compute_obj(avg_a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = avg_a.copy()

    return best_a


def peak_flatten(a, time_budget=40.0, k_top=5):
    """
    Gradient descent targeting the top-k convolution peaks simultaneously.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.005
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    for it in range(8000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_indices = np.argsort(conv)[-k_top:]
        w_top = np.zeros(len(conv))
        for idx in top_indices:
            w_top[idx] = 1.0 / k_top

        grad = compute_gradient(w_top, a)
        grad = grad - np.mean(grad)

        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.8
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-6:
                    break

    return best_a

            no_improve = 0
            scale = max(0.002, scale * 0.998)
        else:
            no_improve += 1
            if no_improve > 300:
                scale = max(0.001, scale * 0.85)
                if scale < 0.001:
                    break
                no_improve = 0

    return best_a


def structured_perturbation(a, time_budget=25.0, seed=99):
    """
    Try structural moves: reverse segments, swap halves, add sinusoidal perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        move = rng.integers(0, 4)
        a_try = a.copy()

        if move == 0:
            # Reverse a random segment
            lo = rng.integers(0, n - 10)
            hi = lo + rng.integers(10, min(50, n - lo))
            a_try[lo:hi] = a_try[lo:hi][::-1]
        elif move == 1:
            # Swap two segments
            seg_len = rng.integers(5, min(30, n // 4))
            i1 = rng.integers(0, n - seg_len)
            i2 = rng.integers(0, n - seg_len)
            if abs(i1 - i2) > seg_len:
                a_try[i1:i1+seg_len], a_try[i2:i2+seg_len] = a_try[i2:i2+seg_len].copy(), a_try[i1:i1+seg_len].copy()
        elif move == 2:
            # Add sinusoidal perturbation
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.01, 0.05)
            x = np.arange(n) / n
            a_try += amp * np.sin(2 * np.pi * freq * x + phase)
        else:
            # Taper ends
            taper = rng.uniform(0.1, 0.5)
            n_taper = int(n * taper)
            fade = np.linspace(0, 1, n_taper)
            a_try[:n_taper] *= fade
            a_try[-n_taper:] *= fade[::-1]

        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(budget_s=1500):
    """
    Main search function combining multiple optimization strategies:
    LP alternating, power-smooth gradient descent, subgradient Polyak,
    peak flattening, coordinate/structured perturbation, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_seq = parent.copy()
                best_score = compute_obj(parent)
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Init uniform (n={n_init}): obj={best_score:.8f}")

    a = best_seq.copy()

    # === Phase 1: LP Alternating ===
    print("\n=== Phase 1: LP Alternating ===")
    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.18, 90.0)
        a_new = lp_alternating(a, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP round {r}: obj={best_score:.8f}")
        else:
            print(f"  LP round {r}: obj={obj:.8f} (no improvement)")

    # === Phase 2: Power-Smooth Gradient Descent ===
    print("\n=== Phase 2: Power-Smooth Gradient Descent ===")
    for p_init in [8, 16, 32, 64]:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(remaining * 0.12, 70.0)
        a_new = gradient_descent_power(a, time_budget=alloc, p_init=p_init, p_max=256)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  GradDesc (p={p_init}): obj={best_score:.8f}")
        else:
            print(f"  GradDesc (p={p_init}): obj={obj:.8f}")

    # === Phase 3: Subgradient with Polyak ===
    print("\n=== Phase 3: Subgradient Polyak ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 50.0)
        a_new = subgradient_polyak(a, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  SubGradPolyak: obj={best_score:.8f}")
        else:
            print(f"  SubGradPolyak: obj={obj:.8f}")

    # === Phase 4: Peak Flattening ===
    print("\n=== Phase 4: Peak Flattening ===")
    for kt in [3, 5, 8]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.08, 35.0)
        a_new = peak_flatten(a, time_budget=alloc, k_top=kt)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  PeakFlatten (k={kt}): obj={best_score:.8f}")
        else:
            print(f"  PeakFlatten (k={kt}): obj={obj:.8f}")

    # === Phase 5: Coordinate Perturbation ===
    print("\n=== Phase 5: Coordinate Perturbation ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 30.0)
        a_new = coordinate_perturbation(a, time_budget=alloc, seed=42)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  CoordPerturb: obj={best_score:.8f}")
        else:
            print(f"  CoordPerturb: obj={obj:.8f}")

    # === Phase 6: Structured Perturbation ===
    print("\n=== Phase 6: Structured Perturbation ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.06, 25.0)
        a_new = structured_perturbation(a, time_budget=alloc, seed=77)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  StructPerturb: obj={best_score:.8f}")
        else:
            print(f"  StructPerturb: obj={obj:.8f}")

    # === Phase 7: Multi-n Exploration ===
    print("\n=== Phase 7: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100),
        max(300, n_best - 50), n_best,
        n_best + 50, n_best + 100, n_best + 200,
        400, 500, 600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_options = [x for x in n_options if 300 <= x <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
            s_try = float(np.sum(a_try))
            a_try = a_try / s_try if s_try > 1e-10 else np.ones(n_try) / n_try

        # Quick LP
        alloc_lp = min(remaining * 0.1, 20.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        # Gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_try < best_score + 0.005:
            a_opt = gradient_descent_power(a_opt, time_budget=min(remaining * 0.08, 20.0), p_init=16)
            obj_try = compute_obj(a_opt)

        # Coord perturb
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.003:
            a_opt = coordinate_perturbation(a_opt, time_budget=min(remaining * 0.06, 15.0), seed=n_try)
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print("\n=== Phase 8: Final Polish ===")
    a = best_seq.copy()
    for polish_round in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break

        # LP
        alloc = min(remaining * 0.2, 35.0)
        a_lp = lp_alternating(a, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  P{polish_round} LP: obj={best_score:.8f}")

        # Gradient
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_gd = gradient_descent_power(a, time_budget=min(remaining * 0.15, 25.0), p_init=32, p_max=128)
            obj_gd = compute_obj(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                a = a_gd.copy()
                print(f"  P{polish_round} GD: obj={best_score:.8f}")

        # Peak flatten
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_pf = peak_flatten(a, time_budget=min(remaining * 0.12, 20.0), k_top=4)
            obj_pf = compute_obj(a_pf)
            if obj_pf < best_score:
                best_score = obj_pf
                best_seq = a_pf.copy()
                a = a_pf.copy()
                print(f"  P{polish_round} PF: obj={best_score:.8f}")

        # Coord perturb
        remaining = deadline - time.monotonic()
        if remaining > 8:
            a_cp = coordinate_perturbation(a, time_budget=min(remaining * 0.1, 15.0), seed=100+polish_round)
            obj_cp = compute_obj(a_cp)
            if obj_cp < best_score:
                best_score = obj_cp
                best_seq = a_cp.copy()
                a = a_cp.copy()
                print(f"  P{polish_round} CP: obj={best_score:.8f}")

        # Subgradient
        remaining = deadline - time.monotonic()
        if remaining > 8:
            a_sg = subgradient_polyak(a, time_budget=min(remaining * 0.1, 15.0))
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                a = a_sg.copy()
                print(f"  P{polish_round} SG: obj={best_score:.8f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
