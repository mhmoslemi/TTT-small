Hybrid optimizer: log-sum-exp smooth NLP (L-BFGS) + LP alternating + multi-n search.
Innovations: temperature-annealed LSE objective for smooth peak minimization,
warm-start LP with HiGHS, diverse initializations, and adaptive n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def _fft_size(n):
    """Next power of 2 >= n."""
    N = 1
    while N < n:
        N <<= 1
    return N


def _compute_conv(a):
    """Full linear convolution a*a."""
    n = len(a)
    N = _fft_size(2 * n - 1)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def _compute_obj(a):
    """2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    b = _compute_conv(a_n)
    return 2.0 * n * float(np.max(b))


def _lse_obj_grad(z, w):
    """
    Log-sum-exp objective and gradient w.r.t. z.
    a = softmax(z), obj = (1/w)*log(sum(exp(w*conv(a,a))))
    """
    n = len(z)
    # Softmax
    z_shift = z - np.max(z)
    exp_z = np.exp(z_shift)
    a = exp_z / np.sum(exp_z)
    
    # b = conv(a, a)
    N = _fft_size(2 * n - 1)
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]
    
    # LSE
    wb = w * b
    wb_max = np.max(wb)
    exp_wb = np.exp(wb - wb_max)
    sum_exp = np.sum(exp_wb)
    obj = (wb_max + np.log(sum_exp)) / w
    
    # p = softmax(w*b)
    p = exp_wb / sum_exp
    
    # grad_a = 2 * conv(p, a)[:n]
    N2 = _fft_size((2 * n - 1) + n - 1)
    fp = rfft(p, N2)
    fa2 = rfft(a, N2)
    pa_full = irfft(fp * fa2, N2)
    grad_a = 2.0 * pa_full[:n]
    
    # Project to tangent space (sum(a)=1)
    grad_a = grad_a - np.mean(grad_a)
    
    # Chain rule for softmax parameterization
    ga_dot_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - ga_dot_a)
    
    return obj, grad_z

    a = a / np.sum(a)
    m = 2 * n - 1
    
    # Build A matrix: A[k,j] = a[k-j] if 0<=k-j<=n-1, else 0
    A = np.zeros((m, n))
    for j in range(n):
        k_lo = j
        k_hi = min(m - 1, j + n - 1)
        for k in range(k_lo, k_hi + 1):
            A[k, j] = a[k - j]
    
    # LP: min [0,...,0,1]^T [g; M]
    # s.t. [A, -I] [g; M] <= 0
    #      [1,...,1, 0] [g; M] = 1
    #      g >= 0, M >= 0
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception as e:
        print(f"    LP error: {e}")
    
    return None


def _lp_alternating(a, time_budget, max_iters=30, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        remaining = start + time_budget - time.monotonic()
        if remaining < 10:
            break
        
        # Solve LP: given a, find g minimizing max conv(a,g)
        g_lp = _solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Find best blend
        best_t_obj = np.inf
        best_t = 0
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = _compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            a_new = (1 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP alt it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall > 15:
                break
    
    return best_a


def _subgradient_descent(a, time_budget, verbose=False):
    """
    Subgradient descent on the exact max objective with adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    N = _fft_size(2 * n - 1)
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))
        
        # Subgradient at the peak: d(conv[k_star])/da[i] = 2*a[k_star-i] if 0<=k_star-i<=n-1
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        grad = grad - np.mean(grad)
        
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.95
            if eta < 1e-10:
                break
    
    return best_a


def _random_init(n, rng):
    """Diverse random initializations."""
    mode = rng.integers(0, 8)
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # U-shaped
        t = x / max(n - 1, 1)
        a = (t - 0.5) ** 2 * 4 + 0.05
    elif mode == 3:
        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        sigma = max(n // 15, 2)
        a = np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2)) + 0.01
    elif mode == 4:
        # Cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.3
    elif mode == 5:
        # Power law (tail-heavy)
        a = 1.0 / (x + 1) ** 0.3
    elif mode == 6:
        # Sine-modulated
        a = 0.5 + 0.4 * np.sin(4 * np.pi * x / n) ** 2 + 0.1
    else:
        # Random smooth
        a = np.exp(rng.normal(0, 0.5, n))
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Hybrid optimizer: log-sum-exp smooth NLP (L-BFGS) + LP alternating + multi-n search.
Innovations: temperature-annealed LSE for smooth peak minimization with exact gradients,
warm-start LP with HiGHS, diverse spectral initializations, and adaptive n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def _fft_size(n):
    """Next power of 2 >= n."""
    N = 1
    while N < n:
        N <<= 1
    return N


def _compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = _fft_size(2 * n - 1)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def _compute_obj(a):
    """2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    b = _compute_conv(a_n)
    return 2.0 * n * float(np.max(b))


def _build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    k_grid = np.arange(m)[:, None]
    j_grid = np.arange(n)[None, :]
    diff = k_grid - j_grid
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((m, n))
    A[mask] = a[diff[mask]]
    return A


def _lse_obj_grad(z, w):
    """
    Log-sum-exp objective and gradient w.r.t. z.
    a = softmax(z), obj = (1/w)*log(sum(exp(w*conv(a,a))))
    Returns (obj, grad_z).
    """
    n = len(z)
    z_shift = z - np.max(z)
    exp_z = np.exp(z_shift)
    a = exp_z / np.sum(exp_z)

    N = _fft_size(2 * n - 1)
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]

    wb = w * b
    wb_max = np.max(wb)
    exp_wb = np.exp(wb - wb_max)
    sum_exp = np.sum(exp_wb)
    obj = (wb_max + np.log(sum_exp)) / w

    p = exp_wb / sum_exp

    N2 = _fft_size((2 * n - 1) + n - 1)
    fp = rfft(p, N2)
    fa2 = rfft(a, N2)
    pa_full = irfft(fp * pa_full if False else fp * fa2, N2)
    grad_a = 2.0 * pa_full[:n]

    grad_a = grad_a - np.mean(grad_a)

    ga_dot_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - ga_dot_a)

    return obj, grad_z


def _lse_optimize(a_init, w_schedule, time_budget, verbose=False):
    """
    Optimize using log-sum-exp NLP with L-BFGS and temperature annealing.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / np.sum(a)
    z = np.log(a)

    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()

    for w in w_schedule:
        remaining = start + time_budget - time.monotonic()
        if remaining < 5:
            break

        try:
            result = minimize(
                _lse_obj_grad,
                z,
                args=(w,),
                method='L-BFGS-B',
                jac=True,
                options={
                    'maxiter': 800,
                    'ftol': 1e-14,
                    'gtol': 1e-9,
                    'maxls': 50,
                }
            )

            z_new = result.x
            a_new = np.exp(z_new - np.max(z_new))
            a_new = a_new / np.sum(a_new)
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s > 0.01:
                a_new = a_new / s
                obj_new = _compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    z = z_new.copy()
                    if verbose:
                        print(f"      LSE w={w}: obj={obj_new:.8f} (iters={result.nit})")
        except Exception as e:
            if verbose:
                print(f"      LSE w={w} error: {e}")

        if verbose:
            print(f"      LSE w={w} done: best={best_obj:.8f}")

    return best_a


def _solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    m = 2 * n - 1

    A = _build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def _lp_alternating(a, time_budget, max_iters=30, verbose=False):
    """LP-based alternating optimization with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        remaining = start + time_budget - time.monotonic()
        if remaining < 10:
            break

        g_lp = _solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Adaptive blending: try multiple t values
        best_t_obj = np.inf
        best_t = 0.0
        for t in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = _compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            a_new = (1 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: no improvement (stall={stall})")
            if stall > 15:
                break

    return best_a


def _joint_lse_lp(a, time_budget, verbose=False):
    """
    Joint optimization: alternate LSE NLP and LP in a tight loop.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()

    w_schedule = [50, 200, 1000, 5000]
    w_idx = 0
    cycle = 0

    while True:
        remaining = start + time_budget - time.monotonic()
        if remaining < 20:
            break

        # LSE step
        time_per_lse = min(remaining * 0.4, 30.0)
        if time_per_lse > 5 and w_idx < len(w_schedule):
            w = w_schedule[w_idx]
            a_lse = _lse_optimize(a, [w], time_budget=time_per_lse)
            obj_lse = _compute_obj(a_lse)
            if obj_lse < best_obj:
                best_obj = obj_lse
                best_a = a_lse.copy()
                a = a_lse.copy()
                if verbose:
                    print(f"      Joint LSE w={w}: {best_obj:.8f}")
            w_idx += 1

        # LP step
        remaining = start + time_budget - time.monotonic()
        time_per_lp = min(remaining * 0.6, 30.0)
        if time_per_lp > 10:
            a_lp = _lp_alternating(a, time_budget=time_per_lp, max_iters=5)
            obj_lp = _compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a = a_lp.copy()
                if verbose:
                    print(f"      Joint LP: {best_obj:.8f}")

        cycle += 1
        if cycle > 20:
            break

    return best_a


def _subgradient_descent(a, time_budget, verbose=False):
    """
    Subgradient descent on the exact max objective with adaptive step size
    and multi-peak weighting.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()
    eta = 0.05
    N = _fft_size(2 * n - 1)
    momentum = 0.5
    v = np.zeros(n)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Use top-3 peaks for multi-peak subgradient
        top_k = min(3, len(b))
        peak_indices = np.argsort(b)[-top_k:]
        peak_weights = b[peak_indices]
        peak_weights = peak_weights / np.sum(peak_weights)

        grad = np.zeros(n)
        for pw, k_star in zip(peak_weights, peak_indices):
            for i in range(n):
                j = int(k_star) - i
                if 0 <= j < n:
                    grad[i] += pw * 2.0 * a[j]

        grad = grad - np.mean(grad)
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 0.3)
            momentum = min(momentum + 0.002, 0.9)
        else:
            eta *= 0.97
            v *= 0.5
            if eta < 1e-10:
                break

    return best_a


def _random_init(n, rng):
    """Diverse random initializations."""
    mode = int(rng.integers(0, 10))
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        t = x / max(n - 1, 1)
        a = (t - 0.5) ** 2 * 4 + 0.05
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        sigma = max(n // 15, 2)
        a = np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2)) + 0.01
    elif mode == 4:
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.3
    elif mode == 5:
        a = 1.0 / (x + 1) ** 0.3
    elif mode == 6:
        a = 0.5 + 0.4 * np.sin(4 * np.pi * x / n) ** 2 + 0.1
    elif mode == 7:
        a = np.exp(rng.normal(0, 0.5, n))
    elif mode == 8:
        # Fejer-like kernel
        a = np.sinc(x / max(n, 1)) ** 2 + 0.05
    else:
        # Triangular modulated
        t = x / max(n - 1, 1)
        a = (1 - np.abs(2 * t - 1)) * 0.5 + 0.5 + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def _polish(a, time_budget):
    """Final polishing: quick LP + LSE."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()

    # LP
    remaining = start + time_budget - time.monotonic()
    if remaining > 15:
        a_lp = _lp_alternating(a, time_budget=min(remaining * 0.5, 40.0), max_iters=15)
        obj_lp = _compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp.copy()

    # LSE
    remaining = start + time_budget - time.monotonic()
    if remaining > 15:
        a_lse = _lse_optimize(a, [1000, 5000], time_budget=min(remaining * 0.5, 30.0))
        obj_lse = _compute_obj(a_lse)
        if obj_lse < best_obj:
            best_obj = obj_lse
            best_a = a_lse.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LSE NLP + LP alternating + joint optimization + multi-n + restarts.
    Uses temperature-annealed log-sum-exp for smooth peak minimization with exact
    FFT-based gradients, HiGHS LP for structural improvements, and diverse
    initializations to avoid local minima.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = _compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = _compute_obj(best_seq)
        print(f"Default (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: LSE NLP with temperature annealing ===
    print(f"\n=== Phase 1: LSE NLP (n={n_best}) ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.18, 180.0)
    w_schedule = [10, 50, 200, 1000]
    a_lse = _lse_optimize(best_seq, w_schedule, time_budget=alloc, verbose=True)
    obj_lse = _compute_obj(a_lse)
    if obj_lse < best_score:
        best_score = obj_lse
        best_seq = a_lse.copy()
        print(f"  ** LSE: {best_score:.8f} **")
    else:
        print(f"  LSE: {obj_lse:.8f}")
    print(f"  [Phase 1 took {time.monotonic()-t0:.1f}s]")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.18, 150.0)
        a_lp = _lp_alternating(best_seq, time_budget=alloc, max_iters=25, verbose=True)
        obj_lp = _compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: {best_score:.8f} **")
        else:
            print(f"  LP: {obj_lp:.8f}")
    print(f"  [Phase 2 took {time.monotonic()-t0:.1f}s]")

    # === Phase 3: Joint LSE+LP ===
    print(f"\n=== Phase 3: Joint LSE+LP ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 120.0)
        a_joint = _joint_lse_lp(best_seq, time_budget=alloc, verbose=True)
        obj_joint = _compute_obj(a_joint)
        if obj_joint < best_score:
            best_score = obj_joint
            best_seq = a_joint.copy()
            print(f"  ** Joint: {best_score:.8f} **")
        else:
            print(f"  Joint: {obj_joint:.8f}")
    print(f"  [Phase 3 took {time.monotonic()-t0:.1f}s]")

    # === Phase 4: Multi-peak Subgradient ===
    print(f"\n=== Phase 4: Multi-peak Subgradient ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.08, 60.0)
        a_sg = _subgradient_descent(best_seq, time_budget=alloc)
        obj_sg = _compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** SubGrad: {best_score:.8f} **")
        else:
            print(f"  SubGrad: {obj_sg:.8f}")
    print(f"  [Phase 4 took {time.monotonic()-t0:.1f}s]")

    # === Phase 5: LSE NLP Round 2 (high temperature) ===
    print(f"\n=== Phase 5: LSE NLP Round 2 ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.12, 80.0)
        w_schedule2 = [500, 2000, 10000]
        a_lse2 = _lse_optimize(best_seq, w_schedule2, time_budget=alloc, verbose=True)
        obj_lse2 = _compute_obj(a_lse2)
        if obj_lse2 < best_score:
            best_score = obj_lse2
            best_seq = a_lse2.copy()
            print(f"  ** LSE2: {best_score:.8f} **")
        else:
            print(f"  LSE2: {obj_lse2:.8f}")
    print(f"  [Phase 5 took {time.monotonic()-t0:.1f}s]")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    t0 = time.monotonic()
    n_best = len(best_seq)
    n_options = sorted(set([
        max(300, n_best - 300),
        max(300, n_best - 100),
        n_best + 100,
        n_best + 300,
        n_best + 500,
        1000, 1200, 1500, 2000, 2500,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 3000 and n != n_best]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = _resample(best_seq, n_try)
        alloc_total = min(remaining * 0.3, 80.0)

        # LSE
        a_opt = _lse_optimize(a_try, [50, 500], time_budget=alloc_total * 0.6)
        # LP
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_opt = _lp_alternating(a_opt, time_budget=min(remaining2 * 0.2, alloc_total * 0.4), max_iters=8)

        obj_try = _compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    print(f"  [Phase 6 took {time.monotonic()-t0:.1f}s]")

    # === Phase 7: Random restarts ===
    print(f"\n=== Phase 7: Random Restarts ===")
    t0 = time.monotonic()
    n_curr = len(best_seq)
    n_restarts = max(2, int((deadline - time.monotonic()) / 60.0))
    n_restarts = min(n_restarts, 6)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_r = _random_init(n_curr, rng)
        alloc = min(remaining * 0.25, 50.0)
        a_opt = _lse_optimize(a_r, [50, 500], time_budget=alloc * 0.7)
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_opt = _lp_alternating(a_opt, time_budget=min(remaining2 * 0.15, alloc * 0.3), max_iters=8)
        obj_r = _compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r}: {best_score:.8f} **")
        else:
            print(f"  Restart {r}: {obj_r:.8f}")

    print(f"  [Phase 7 took {time.monotonic()-t0:.1f}s]")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.6, 120.0)
        a_final = _polish(best_seq, time_budget=alloc)
        obj_final = _compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Polish: {best_score:.8f} **")
        else:
            print(f"  Polish: {obj_final:.8f}")

    print(f"  [Phase 8 took {time.monotonic()-t0:.1f}s]")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
