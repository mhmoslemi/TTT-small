def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):

    size = next_pow2(2*n - 1)
    fa = rfft(a, size)

`min_a max_k (a*a)[k]` s.t. `sum(a)=1, a>=0`

This is a min-max problem. We can reformulate as:
`min M` s.t. `(a*a)[k] <= M for all k, sum(a)=1, a>=0`

This is a non-convex problem because `(a*a)[k]` is bilinear in `a`.

However, we can use a **SOCP (Second-Order Cone) relaxation** or **SDP relaxation**:

For each `k`, `(a*a)[k] = sum_i a[i]*a[k-i] <= M`.

If we introduce variables `x_{i,j} = a[i]*a[j]`, then we have a matrix `X = aa^T` which is PSD and rank 1. The constraint becomes `sum_i X[i, k-i] <= M` (with appropriate indexing).

The SDP relaxation drops the rank-1 constraint:
`min M` s.t. `sum_i X[i, k-i] <= M for all k, 1^T X_ii... `

Actually, the constraint `sum(a) = 1` is `(1^T a)^2 = 1`, which in terms of `X` is `1^T X 1 = 1` (since `1^T X 1 = (1^T a)(a^T 1) = (sum a)^2 = 1`). And `X_ii = a[i]^2` so `X` has non-zero diagonal.

The SDP is:
`min M`
`s.t. sum_{i: valid} X[i, k-i] <= M for k=0,...,2n-2`
`     1^T X 1 = 1`
`     X >= 0` (PSD)
`     X_ii >= 0` (implied by PSD)

This is a semidefinite program with `O(n^2)` variables and `O(n)` constraints. For `n=1000`, that's a million variables - too large for most solvers.

"""
Improved optimizer for low-autocorrelation non-negative sequences.
Uses L2-informed pre-optimization with triangular target, Nesterov-accelerated
subgradient for peak minimization, warm-started LP alternating, spectral
initializations, and multi-n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time


def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def triangular_envelope(n):
    """Triangular envelope for convolution of two length-n sequences."""
    k = np.arange(2 * n - 1)
    return np.minimum(k + 1, np.minimum(n, 2 * n - 1 - k)).astype(np.float64)


def l2_informed_obj_and_grad(x, a=None):
    """
    L2 objective with triangular target.
    obj = sum_k ((a*a)[k] / env[k] - mean_ratio)^2
    Returns (obj, grad) where grad is w.r.t. x (log-parameterization).
    """
    n = len(x)
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    s = float(np.sum(a))
    a = a / s

    conv = compute_conv(a)
    env = triangular_envelope(n)
    ratio = conv / env
    mean_r = float(np.mean(ratio))
    e = ratio - mean_r
    obj = float(np.sum(e * e))

    # grad_a[i] = 4 * sum_j e[i+j]/env[i+j] * a[j] (cross-correlation)
    # grad_x[i] = grad_a[i] * a[i] (chain rule through exp)
    weighted_e = e / env
    size = next_pow2(3 * n)
    we_rev = np.zeros(size)
    we_rev[:2 * n - 1] = weighted_e[::-1]
    a_pad = np.zeros(size)
    a_pad[:n] = a
    result = irfft(rfft(we_rev, size) * rfft(a_pad, size), size)
    grad_a = 4.0 * result[n - 1:2 * n - 1][::-1]
    grad_x = grad_a * a

    return obj, grad_x


def l2_optimize(a_init, time_budget=60.0, verbose=False):
    """Optimize L2 informed objective using L-BFGS-B with log parameterization."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy()
    a = a / np.sum(a)
    x = np.log(a) - np.log(a).max()

    best_a = a.copy()
    best_l2 = l2_informed_obj_and_grad(x, a)[0]

    start = time.monotonic()

    def fun(xv):
        xv = xv - np.max(xv)
        val, _ = l2_informed_obj_and_grad(xv)
        return val

    def fgrad(xv):
        xv = xv - np.max(xv)
        _, g = l2_informed_obj_and_grad(xv)
        return g

    res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                   options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 100})

    if res.fun < best_l2:
        x_opt = res.x - res.x.max()
        a_new = np.exp(x_opt)
        a_new = a_new / np.sum(a_new)
        if time.monotonic() - start < time_budget:
            return a_new
    return best_a


def subgradient_peak(a_init, time_budget=60.0, verbose=False, seed=0):
    """
    Nesterov-accelerated projected subgradient method for minimizing max(conv(a,a)).
    Uses adaptive step size and periodic restarts.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    y = a.copy()
    t_k = 1.0
    step = 0.5 / np.sqrt(n)

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient of max(conv(a,a)) at y
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * y[j]

        # Projected subgradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-15:
            y = a_new.copy()
        else:
            y = y / sy
        t_k = t_new
        a = a_new

        # Adaptive step size
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"    subgrad it={it}: obj={best_obj:.8f}")
        else:
            step *= 0.98
            if step < 1e-8:
                # Restart with perturbation
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                y = a.copy()
                step = 0.3 / np.sqrt(n)
                t_k = 1.0

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None
    a_norm = a / np.sum(a)

    # Try cvxpy first
    try:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    # Fallback: scipy linprog
    try:
        A = build_conv_matrix(a_norm)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternate_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating optimization with adaptive blending and restarts."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        a_norm = a / np.sum(a)
        g_lp, M_lp = solve_lp_minpeak(a_norm)
        if g_lp is None:
            break

        a_new = None
        obj_new = best_obj

        for t in [0.5, 0.7, 0.9, 1.0, 0.3, 0.8, 0.6, 0.4, 0.95]:
            a_try = (1.0 - t) * a_norm + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < obj_new:
                obj_new = obj_try
                a_new = a_try.copy()

        if a_new is not None and obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 8:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.02 + 0.02 * min(stall // 8, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

    return best_a


def smoothed_peak_optimize(a_init, time_budget=60.0, verbose=False):
    """Optimize using L-BFGS-B on smoothed (log-sum-exp) peak objective."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = np.log(a) - np.log(a).max()

    taus = [5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break

        def fun(xv):
            xv = xv - np.max(xv)
            av = np.exp(xv)
            sv = float(np.sum(av))
            av = av / sv
            conv = compute_conv(av)
            c_scaled = conv / tau
            c_max = np.max(c_scaled)
            F = (c_max + np.log(np.sum(np.exp(c_scaled - c_max))))
            return F

        def fgrad(xv):
            xv = xv - np.max(xv)
            av = np.exp(xv)
            sv = float(np.sum(av))
            av = av / sv
            conv = compute_conv(av)
            c_scaled = conv / tau
            c_max = np.max(c_scaled)
            exp_c = np.exp(c_scaled - c_max)
            sum_exp = np.sum(exp_c)
            w = exp_c / sum_exp

            size = next_pow2(2 * n - 1)
            a_flip = av[::-1]
            fa_flip = rfft(a_flip, size)
            fw = rfft(w, size)
            conv_w_af = irfft(fw * fa_flip, size)
            corr = conv_w_af[n - 1:n - 1 + n]

            return 2.0 * corr / (tau * sv * sv) - 2.0 * fun(xv) * av / sv

        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 150, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})

        if res.x is not None:
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = x_opt
                    if verbose:
                        print(f"    smooth tau={tau:.4f}: obj={best_obj:.8f}")

    return best_a


def spectral_init(n, rng, mode=0):
    """Generate spectral-based random initializations."""
    x = np.arange(n)
    if mode == 0:
        # Cosine basis combination
        a = np.ones(n)
        for k in range(1, min(6, n // 2)):
            a += rng.uniform(-0.3, 0.3) * np.cos(2 * np.pi * k * x / n)
        a = np.maximum(a, 0.01)
    elif mode == 1:
        # Gaussian windows at random positions
        a = np.ones(n) * 0.1
        for _ in range(rng.integers(2, 5)):
            c = rng.integers(0, n)
            w = rng.uniform(n // 10, n // 3)
            a += np.exp(-((x - c) ** 2) / (2 * w * w))
    elif mode == 2:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        a += 0.1 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
    elif mode == 3:
        # Two-lobe (anti-symmetric shape)
        a = 1.0 - 0.8 * np.cos(2 * np.pi * x / n)
        a += 0.05 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
    else:
        # Power-law decay from center
        c = n / 2.0
        a = 1.0 / (1.0 + 0.5 * np.abs(x - c) / (n // 4 + 1))
        a += 0.05 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
    return a / np.sum(a)

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)

    if remaining > 60:
        a_opt, obj = full_pipeline(best_seq, time_budget=min(remaining * 0.2, 150.0), verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt
            print(f"  After pipeline: obj={best_score:.10f}")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        n_best - 100, n_best - 50, n_best,
        n_best + 50, n_best + 100, n_best + 200, n_best + 300,
        600, 700, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.12, 60.0)
        a_opt, obj = full_pipeline(a_try, time_budget=alloc)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 3: Diversified initializations ===
    print(f"\n=== Phase 3: Diversified Initializations ===")
    n_best = len(best_seq)

    init_list = []
    # Spectral inits
    for mode in range(5):
        init_list.append(spectral_init(n_best, rng, mode))
    # Random inits
    for _ in range(3):
        init_list.append(random_init(n_best, rng))
    # Perturbed best
    init_list.append(np.maximum(best_seq * np.exp(rng.normal(0, 0.05, n_best)), 0) /
                     np.sum(np.maximum(best_seq * np.exp(rng.normal(0, 0.05, n_best)), 0)))
    # Symmetric
    a_sym = (best_seq + best_seq[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    init_list.append(a_sym / np.sum(a_sym))

    for idx, a_init in enumerate(init_list):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        alloc = min(remaining * 0.08, 40.0)
        a_opt, obj = full_pipeline(a_init, time_budget=alloc)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt
            n_best = len(best_seq)
            print(f"  Init {idx}: obj={best_score:.10f} **")
        else:
            print(f"  Init {idx}: obj={obj:.10f}")

    # === Phase 4: Intensify on best ===
    print(f"\n=== Phase 4: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # More aggressive LP alternating
        alloc = min(remaining * 0.25, 100.0)
        a_lp = lp_alternate_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp
            print(f"  LP intensify: obj={best_score:.10f}")

        # Subgradient with fresh start
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(remaining * 0.2, 60.0)
            a_sg = subgradient_peak(best_seq, time_budget=alloc, verbose=True, seed=123)
            obj = compute_obj(a_sg)
            if obj < best_score:
                best_score = obj
                best_seq = a_sg
                print(f"  Subgrad intensify: obj={best_score:.10f}")

        # Smoothed peak
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.2, 50.0)
            a_sm = smoothed_peak_optimize(best_seq, time_budget=alloc, verbose=True)
            obj = compute_obj(a_sm)
            if obj < best_score:
                best_score = obj
                best_seq = a_sm
                print(f"  Smooth intensify: obj={best_score:.10f}")

    # === Phase 5: Try different n around best ===
    print(f"\n=== Phase 5: Fine n Tuning ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for dn in [-20, -10, 10, 20, 30, -30]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            n_try = n_best + dn
            if n_try < 400 or n_try > 2500:
                continue
            a_try = resample_sequence(best_seq, n_try)
            alloc = min(remaining * 0.15, 30.0)
            a_opt, obj = full_pipeline(a_try, time_budget=alloc)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt
                n_best = n_try
                print(f"  n={n_try}: obj={best_score:.10f} **")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Symmetric refinement
        alloc = min(remaining * 0.2, 25.0)
        a_sym = symmetric_refine(best_seq, time_budget=alloc)
        obj = compute_obj(a_sym)
        if obj < best_score:
            best_score = obj
            best_seq = a_sym
            print(f"  Symmetric: obj={best_score:.10f}")

        # Final smoothed
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 30.0)
            a_fin = smoothed_peak_optimize(best_seq, time_budget=alloc, verbose=True)
            obj = compute_obj(a_fin)
            if obj < best_score:
                best_score = obj
                best_seq = a_fin
                print(f"  Final smooth: obj={best_score:.10f}")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 25.0)
            a_lp = lp_alternate_optimize(best_seq, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp
                print(f"  Final LP: obj={best_score:.10f}")

    # === Validation ===
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
