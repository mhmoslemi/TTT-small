from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(fixed):
    """Build matrix A (2n-1 x n) such that A @ x = conv(fixed, x)."""
    n = len(fixed)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = fixed[:r]
    return A


def solve_lp_minpeak(fixed):
    """Solve LP: min M s.t. conv(fixed,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(fixed)
    fixed_norm = fixed / np.sum(fixed)
    A = build_conv_matrix(fixed_norm)

    # Try cvxpy first
    if HAS_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solvers = [cp.MOSEK, cp.CLAARABEL, cp.ECOS, cp.SCS, cp.PDLP]
        for solver in solvers:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def softmax_gradient_descent(a, n_iters=2000, time_budget=60.0, t_start=50, t_end=200):
    """
    FFT-based softmax gradient descent with adaptive temperature.
    Minimizes 2*n*max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 3 * n:  # Need length for a * w' convolution
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive temperature
        t = t_start + (t_end - t_start) * (it / max(1, n_iters))
        t = max(10, min(500, t))

        conv = compute_conv(a)
        
        # Softmax weights
        t_conv = t * conv
        t_conv -= np.max(t_conv)
        weights = np.exp(t_conv)
        weights_sum = np.sum(weights)
        if weights_sum < 1e-300:
            weights = np.zeros(2 * n - 1)
            weights[n - 1] = 1.0
        else:
            weights /= weights_sum

        # Compute gradient: grad[i] = sum_j a[j] * w[i+j]
        # Using FFT: grad[i] = (a * w')[2n-2-i] where w' = reverse(w)
        w_rev = weights[::-1]
        fa = fft(a, L)
        fw = fft(w_rev, L)
        conv_aw = ifft(fa * fw).real[:2 * n - 1 + n - 1]  # length 3n-2
        # Actually, a has length n, w_rev has length 2n-1, so conv has length 3n-2
        conv_aw = ifft(fa * fw).real[:3 * n - 2]
        
        # grad[i] = conv_aw[2n-2-i]
        grad = np.zeros(n)
        for i in range(n):
            idx = 2 * n - 2 - i
            if idx < len(conv_aw):
                grad[i] = conv_aw[idx]

        # Projected gradient descent (with momentum)
        if it == 0:
            momentum = np.zeros(n)
            beta = 0.7
            
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean

        momentum = beta * momentum + (1 - beta) * grad_centered
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Decrease lr slightly
            lr *= 0.999
        else:
            lr *= 0.95
            if lr < 0.001:
                # Perturb and restart
                rng = np.random.default_rng(it + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                a = a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = 0.1
                momentum = np.zeros(n)

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random non-negative sequences."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Smooth random via random Fourier features
        n_freqs = min(16, n // 10)
        freqs = rng.uniform(0, 5, n_freqs)
        phases = rng.uniform(0, 2 * np.pi, n_freqs)
        coeffs = rng.uniform(0.1, 0.5, n_freqs)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p, c in zip(freqs, phases, coeffs):
            a += c * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal / multimodal
        n_modes = rng.integers(2, 5)
        centers = rng.permutation(n)[:n_modes].astype(float) * (n - 1) / (n - 1)
        w = n // 8
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window variants
        x = np.linspace(0, np.pi, n)
        p = rng.uniform(1, 4)
        a = np.cos(x) ** p + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Geometric / exponential decay
        rate = rng.uniform(0.01, 0.1)
        x = np.arange(n)
        a = np.exp(-rate * x) + np.exp(-rate * (n - 1 - x))
        a = a / np.sum(a)
    elif mode == 6:
        # Sparse-ish with random gaps
        a = rng.uniform(0, 1, n)
        a[rng.choice(n, size=n // 4, replace=False)] = 0
        a += 0.01
        a = a / np.sum(a)
    else:
        # Alternating high/low
        a = np.ones(n)
        a[rng.integers(0, 2, n)] *= rng.uniform(1.5, 3.0)
        a = a / np.sum(a)
    return a


def coordinate_descent_refined(a, time_budget=30.0):
    """Refined coordinate descent targeting the peak."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    conv = compute_conv(a)
    no_improve = 0

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from positions that heavily contribute to the peak
        # to positions that don't
        step = max(1, n // 300)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break

                    new_conv = compute_conv(a)

                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        if new_obj < best_obj:
                            # Still better than original, keep the direction
                            pass
                        break

        if not improved:
            # Small random perturbations
            rng = np.random.default_rng(it * 13)
            for _ in range(20):

def annealed_lp(a, time_budget=120.0, t_init=0.1, t_min=0.001):
    """LP alternating with simulated annealing for escaping local minima."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    n_steps = 0

    while time.monotonic() - start < time_budget:
        n_steps += 1
        T = t_init * (t_min / t_init) ** (n_steps / max(1, time_budget // 2))
        T = max(T, t_min)

        g_lp, _ = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Damped update
        t_step = np.random.default_rng(n_steps).uniform(0.2, 0.8)
        a_new = (1 - t_step) * a + t_step * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        # Accept or reject
        delta = obj_new - current_obj
        if delta < 0 or np.random.default_rng(n_steps + 7) < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        else:
            # Occasionally perturb to explore
            if n_steps % 20 == 0:
                rng = np.random.default_rng(n_steps * 3)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                current_obj = compute_obj(a)

    return best_a
