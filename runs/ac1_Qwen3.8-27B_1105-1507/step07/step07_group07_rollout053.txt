from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build sparse convolution matrix
    # A @ g = conv(a, g)
    # A has 2n-1 rows and n columns
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, use the structure
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            rows.append(k)
            cols.append(j)
            vals.append(a_norm[k - j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: minimize M
    # Variables: g[0..n-1], M
    # Constraints: A @ g - M <= 0 (2n-1 constraints)
    #              sum(g) = 1
    #              g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub[:, :n] = A.toarray()
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def next_pow2(m):

def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_minpeak(a):

    Returns (g_normalized, M_opt) or (None, None)."""

    a_norm = a / np.sum(a)

    # Build convolution matrix A (2n-1 x n) where A @ g = conv(a, g)
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0

    vals = np.concatenate([a_norm[:min(n, 2 * n - 1 - j)] for j in range(n)])

    b_eq = np.array([1.0])

def smoothed_gradient_step(a, tau):
    """One smoothed gradient step (exponentiated gradient with soft-max weights)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a_norm = a / s

    conv = compute_conv(a_norm)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
    # This is a cross-correlation of w with a
    size = next_pow2(2 * n - 1)
    a_flip = a_norm[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)

    eta = 2.0 / (tau * n)
    a_new = a_norm * np.exp(-eta * corr)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / np.sum(a_new)
    return a_new

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            t1 = valid_ts[best_idx - 1]
            t2 = valid_ts[best_idx]
            t3 = valid_ts[best_idx + 1]
            o1 = obj_values[best_idx - 1]
            o2 = obj_values[best_idx]
            o3 = obj_values[best_idx + 1]
            denom = (o1 - o3) * (t1 - t2) - (o1 - o2) * (t3 - t2)
            if abs(denom) > 1e-15:
                t_star = t2 - 0.5 * ((o1 - o3) * (t1 - t2) ** 2 - (o1 - o2) * (t3 - t2) ** 2) / denom
                if 0.0 <= t_star <= 1.0:
                    a_try = (1.0 - t_star) * a_norm + t_star * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()

    return best_a, best_obj


def lp_alternate_optimize(a, time_budget=60.0):
    """LP alternating optimization with adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    perturb_scale = 0.02
    iteration = 0

    while time.monotonic() - start < time_budget:
        iteration += 1
        a_new, obj_new = lp_alternate_step(a)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            perturb_scale = max(0.005, perturb_scale * 0.92)
        else:
            stall += 1
            if stall >= 4:
                # Perturb to escape local minimum
                rng = np.random.default_rng(iteration * 7919 + int(time.monotonic() * 1000))
                a = best_a * np.exp(rng.normal(0, perturb_scale, n))
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                stall = 0
                perturb_scale = min(0.15, perturb_scale * 1.5)

    return best_a


def smoothed_gradient_step(a, tau):
    """One smoothed gradient step using soft-max weights over convolution peaks."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a_norm = a / s

    conv = compute_conv(a_norm)
    c_scaled = conv / tau
    c_max = float(np.max(c_scaled))
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = float(np.sum(exp_c))
    w = exp_c / sum_exp

    # Compute gradient via cross-correlation: grad[i] = sum_k w[k] * a_norm[k-i]
    size = next_pow2(2 * n - 1)
    a_flip = a_norm[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]

    # Exponentiated gradient update
    eta = 2.0 / (tau * n)
    a_new = a_norm * np.exp(-eta * corr)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / np.sum(a_new)
    return a_new


def gradient_optimize(a_init, time_budget=60.0):
    """Smoothed gradient optimization with adaptive temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    tau = max_conv * 3.0
    min_tau = max_conv * 0.05

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        a_new = smoothed_gradient_step(a, tau)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            tau = max(min_tau, tau * 0.97)
        else:
            tau = min(max_conv * 15.0, tau * 1.3)
            if tau > max_conv * 15.0:
                rng = np.random.default_rng(it * 31)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-30)
                a = a / np.sum(a)
                tau = max_conv * 3.0

    return best_a


def peak_shaving(a, time_budget=30.0):
    """Local search: adjust individual elements to reduce the peak convolution value."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    conv = compute_conv(a)
    peak_idx = int(np.argmax(conv))

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        # Pick a random element
        i = int(rng.integers(0, n))
        j = peak_idx - i

        if 0 <= j < n and a[i] > 1e-10:
            # Try reducing a[i] and redistributing
            for delta_frac in [0.005, 0.01, 0.02, 0.05]:
                delta = a[i] * delta_frac
                if delta < 1e-12:
                    continue
                a_try = a.copy()
                a_try[i] = a_try[i] - delta
                if a_try[i] < 0:
                    a_try[i] = 0
                    delta = a[i]
                # Redistribute to maintain sum=1
                others_weight = a_try.copy()
                others_weight[i] = 0.0
                sum_others = float(np.sum(others_weight))
                if sum_others > 1e-10:
                    a_try += delta * others_weight / sum_others
                a_try = np.maximum(a_try, 0)
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    conv = compute_conv(a)
                    peak_idx = int(np.argmax(conv))
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with parabolic line search, adaptive perturbation,
    peak-shaving local search, smoothed gradient, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, obj={best_score:.10f}")

    # === Phase 1: Quick gradient to improve starting point ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 1: Gradient (remaining={remaining:.0f}s) ===")
        a_grad = gradient_optimize(best_seq, time_budget=min(remaining * 0.08, 40.0))
        obj = compute_obj(a_grad)
        if obj < best_score:
            best_score = obj
            best_seq = a_grad.copy()
            print(f"  After gradient: obj={best_score:.10f}")
        else:
            print(f"  Gradient: obj={obj:.10f} (no improvement)")

    # === Phase 2: LP Alternating (primary method) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: LP Alternating (remaining={remaining:.0f}s) ===")
        lp_budget = min(remaining * 0.35, 250.0)
        a_lp = lp_alternate_optimize(best_seq, time_budget=lp_budget)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  After LP: obj={best_score:.10f}")
        else:
            print(f"  LP: obj={obj:.10f} (no improvement)")

    # === Phase 3: Peak shaving ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 3: Peak Shaving (remaining={remaining:.0f}s) ===")
        ps_budget = min(remaining * 0.1, 40.0)
        a_ps = peak_shaving(best_seq, time_budget=ps_budget)
        obj = compute_obj(a_ps)
        if obj < best_score:
            best_score = obj
            best_seq = a_ps.copy()
            print(f"  After PS: obj={best_score:.10f}")
        else:
            print(f"  PS: obj={obj:.10f} (no improvement)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_candidates = sorted(set([
        max(400, n_parent - 150), max(400, n_parent - 75),
        n_parent,
        min(1600, n_parent + 75), min(1600, n_parent + 150), min(1600, n_parent + 300),
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1300, 1500
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1600]
    n_candidates.remove(n_parent) if n_parent in n_candidates else None
    n_candidates.insert(0, n_parent)  # Try parent n first (already optimized)

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if n_try == n_parent and len(best_seq) == n_try:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
            a_try = a_try * np.exp(rng.normal(0, 0.015, n_try))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)

        # Gradient then LP
        alloc_g = min(remaining * 0.06, 15.0)
        alloc_lp = min(remaining * 0.12, 35.0)

        a_opt = gradient_optimize(a_try, time_budget=alloc_g)
        obj_g = compute_obj(a_opt)

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_opt = lp_alternate_optimize(a_opt, time_budget=alloc_lp)
            obj_lp = compute_obj(a_opt)
            if obj_lp < obj_g:
                obj_g = obj_lp

        if obj_g < best_score:
            best_score = obj_g
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_g:.10f}")

    # === Phase 5: Final intensification ===
    print(f"\n=== Phase 5: Intensify (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc_lp = min(remaining * 0.3, 100.0)
        a_lp2 = lp_alternate_optimize(best_seq, time_budget=alloc_lp)
        obj = compute_obj(a_lp2)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()
            print(f"  LP final: obj={best_score:.10f}")
        else:
            print(f"  LP final: obj={obj:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_ps2 = peak_shaving(best_seq, time_budget=min(remaining * 0.3, 30.0))
            obj = compute_obj(a_ps2)
            if obj < best_score:
                best_score = obj
                best_seq = a_ps2.copy()
                print(f"  PS final: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 800
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
