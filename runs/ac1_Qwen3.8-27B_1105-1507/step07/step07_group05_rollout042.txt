"""
Combined optimization: spectral flattening (GS-like), multi-peak subgradient,
energy GD, and LP polishing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def fft_convolve(a, b, N=None):
    """Linear convolution via FFT."""
    if N is None:
        N = 1
        while N < len(a) + len(b) - 1:
            N <<= 1
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:len(a) + len(b) - 1]


def gs_spectral_flatten(a, time_budget=60.0, alpha_range=(0.9, 0.3), verbose=False):
    """
    Gerchberg-Saxton-like spectral flattening.
    Iteratively flattens the power spectrum while maintaining non-negativity and support.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Forward FFT
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2

        # Flatten: P_flat = P ** alpha
        # Use adaptive alpha based on iteration
        progress = min(it / 1000.0, 1.0)
        alpha = alpha_range[0] + (alpha_range[1] - alpha_range[0]) * progress
        alpha = max(alpha, 0.1)

        P_flat = P ** alpha

        # Preserve DC component (sum constraint)
        P_flat[0] = P[0]

        # Ensure non-negative
        P_flat = np.maximum(P_flat, 0)

        # Inverse: a_candidate = Re(IFFT(sqrt(P_flat) * exp(j*angle(A))))
        phase = np.angle(A)
        magnitude = np.sqrt(P_flat)
        A_new = magnitude * np.exp(1j * phase)
        a_candidate = np.fft.irfft(A_new, N)[:n]

        # Enforce non-negativity
        a_candidate = np.maximum(a_candidate, 0)

        # Renormalize
        s_new = float(np.sum(a_candidate))
        if s_new < 1e-10:
            break
        a_candidate = a_candidate / s_new

        # Check objective
        obj = compute_obj(a_candidate)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_candidate.copy()
            a = a_candidate.copy()

        if verbose and it % 500 == 0:

def multi_peak_subgradient(a, time_budget=60.0, K=5, verbose=False):
    """
    Subgradient method targeting the top K peaks simultaneously.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    momentum = 0.5
    v = np.zeros(n)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        # Compute autocorrelation
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Find top K peaks
        perm = np.argsort(b)[::-1]
        top_k = perm[:K]

        # Compute combined gradient: sum of gradients at top K peaks
        grad = np.zeros(n)
        for k in top_k:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]

        # Center gradient
        grad = grad - np.sum(grad) * a / s  # project to sum=0 hyperplane

        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 1.0)
            momentum = min(momentum + 0.002, 0.9)
        else:
            eta *= 0.98
            v *= 0.8
            if eta < 1e-8:
                break

    return best_a


def fft_gradient_descent(a, time_budget=120.0, mode='energy', verbose=False):
    """FFT-based gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    stall = 0
    max_stall = 300

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        if mode == 'energy':
            # Gradient of sum(b^2) = 2 * conv(a_flip, b)
            a_flip = a[::-1]
            faf = rfft(a_flip, N)
            fb = rfft(b, N)
            c_full = irfft(faf * fb, N)[:3 * n - 2]
            grad = c_full[n - 1:2 * n - 1]
        elif mode == 'peak':
            # Softmax-weighted peak gradient
            temp = 1.0 / (2 * n * float(np.max(b)))  # adaptive temperature
            b_shifted = b / (temp * n)
            w = np.exp(b_shifted - np.max(b_shifted))
            w_sum = np.sum(w)
            if w_sum > 1e-30:
                w = w / w_sum
            a_flip = a[::-1]
            faf = rfft(a_flip, N)
            fw = rfft(w, N)
            c_full = irfft(faf * fw, N)[:3 * n - 2]
            grad = c_full[n - 1:2 * n - 1]
        else:  # 'multi'
            # Top-K weighted gradient
            K = 5
            perm = np.argsort(b)[::-1]
            top_k = perm[:K]
            w = np.zeros(2 * n - 1)
            for k in top_k:
                w[k] = 1.0 / K
            a_flip = a[::-1]
            faf = rfft(a_flip, N)
            fw = rfft(w, N)
            c_full = irfft(faf * fw, N)[:3 * n - 2]
            grad = c_full[n - 1:2 * n - 1]

        # Project gradient to sum=0
        grad = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.003, 3.0)
            momentum = min(momentum + 0.001, 0.95)
        else:
            stall += 1
            eta *= 0.97
            v *= 0.5
            if stall > max_stall:
                # Reset with perturbation
                rng_local = np.random.default_rng(it * 7 + 13)
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.7
                stall = 0
                max_stall = min(max_stall + 200, 1000)
            if eta < 1e-9:
                break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 2000:
        # Too large for LP, skip
        return None
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    if n > 2000:
        return a
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 10) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Tail-heavy
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    elif mode == 6:
        # Quadratic residue-like (Barker-inspired)
        a = np.ones(n)
        for i in range(1, n):
            if (i * i) % 4 == 1:
                a[i] = 0.5
        a = a / np.sum(a)
    else:
        # Gaussian window
        x = np.arange(n) / max(n - 1, 1)
        a = np.exp(-4 * (x - 0.5) ** 2) + 0.05
        a = a / np.sum(a)
    return a

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def gs_spectral_spread(a, time_budget=60.0, verbose=False):
    """
    Spectral spreading: iteratively flatten the power spectrum to reduce
    the peak of the linear autocorrelation.
    
    Key insight: A peaky power spectrum (energy concentrated at DC) leads to
    a peaky autocorrelation. By spreading spectral energy (P^alpha, alpha<1),
    we flatten the autocorrelation and reduce its peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Progressive alpha annealing
    alphas = [0.95, 0.9, 0.85, 0.8, 0.75, 0.7, 0.65, 0.6, 0.55, 0.5, 0.45, 0.4, 0.35, 0.3, 0.25, 0.2]
    t_blend = 0.3  # blending factor for stability

    for alpha in alphas:
        if time.monotonic() - start > time_budget * 0.7:
            break
        a_curr = best_a.copy()
        n_inner = 200
        for it in range(n_inner):
            if time.monotonic() - start > time_budget * 0.7:
                break
            A = np.fft.rfft(a_curr, N)
            P = np.abs(A) ** 2

            # Spread: P_new[k] = P[k]^alpha for k>0, keep DC fixed
            P_new = np.empty_like(P)
            P_new[0] = P[0]
            P_new[1:] = np.power(P[1:], alpha)
            P_new = np.maximum(P_new, 0)

            # Reconstruct with current phase
            phase = np.angle(A)
            mag = np.sqrt(P_new)
            A_new = mag * np.exp(1j * phase)
            a_candidate = np.fft.irfft(A_new, N)[:n]

            # Project: non-neg + normalize
            a_candidate = np.maximum(a_candidate, 0)
            s_new = float(np.sum(a_candidate))
            if s_new < 1e-10:
                break
            a_candidate = a_candidate / s_new

            # Blend for stability
            a_try = (1 - t_blend) * a_curr + t_blend * a_candidate
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                break
            a_try = a_try / s_try

            obj_try = compute_obj(a_try)
            if obj_try < compute_obj(a_curr) - 1e-14:
                a_curr = a_try
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    if verbose:
                        print(f"    GS alpha={alpha:.2f} it{it}: obj={best_obj:.8f}")

        if time.monotonic() - start > time_budget:
            break
        a_curr = best_a.copy()
        for it in range(100):
            if time.monotonic() - start > time_budget:
                break
            A = np.fft.rfft(a_curr, N)
            P = np.abs(A) ** 2
            P_new = np.empty_like(P)
            P_new[0] = P[0]
            P_new[1:] = np.power(P[1:], alpha)
            P_new = np.maximum(P_new, 0)
            phase = np.angle(A)
            mag = np.sqrt(P_new)
            A_new = mag * np.exp(1j * phase)
            a_candidate = np.fft.irfft(A_new, N)[:n]
            a_candidate = np.maximum(a_candidate, 0)
            s_new = float(np.sum(a_candidate))

            obj_try = compute_obj(a_candidate)

                    best_a = a_curr.copy()
                    if verbose:
                        print(f"    GS2 alpha={alpha:.2f} it{it}: obj={best_obj:.8f}")

def exponentiated_gradient(a, time_budget=120.0, mode='peak', verbose=False):
    """
    Multiplicative weights (exponentiated gradient) descent.
    Naturally preserves non-negativity: a_new[i] = a[i] * exp(-eta * grad[i]).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.01
    stall = 0
    max_stall = 500

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Compute gradient
        a_flip = a[::-1]
        faf = rfft(a_flip, N)

        if mode == 'peak':
            # Adaptive temperature softmax
            peak_val = float(np.max(b))
            temp = peak_val * 0.1
            if temp < 1e-15:
                temp = 1e-15
            b_shifted = (b - peak_val) / temp
            w = np.exp(b_shifted)
            w_sum = float(np.sum(w))
            if w_sum > 1e-30:
                w = w / w_sum
        elif mode == 'energy':
            w = b
        elif mode == 'multi':
            K = 5
            perm = np.argsort(b)[::-1]
            w = np.zeros(2 * n - 1)
            w[perm[:K]] = 1.0 / K
        else:
            w = b

        fb_w = rfft(w, N)
        c_full = irfft(faf * fb_w, N)[:3 * n - 2]
        grad = c_full[n - 1:2 * n - 1]

        # Center gradient (for simplicity; the normalization handles the sum)
        grad = grad - np.mean(grad)

        # Exponentiated gradient update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * grad
        # Numerical stability
        log_a_new = log_a_new - np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.005, 0.5)
        else:
            stall += 1
            eta *= 0.98
            if stall > max_stall:
                # Perturb and reset
                rng_local = np.random.default_rng(it * 31 + 7)
                a = best_a * np.exp(rng_local.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                eta = 0.01
                stall = 0
                max_stall = min(max_stall + 300, 2000)
            if eta < 1e-10:
                break

        if verbose and it % 2000 == 0:
            print(f"    EG({mode}) it{it}: obj={best_obj:.8f}, eta={eta:.6f}")

    return best_a


def fft_gradient_descent(a, time_budget=120.0, mode='peak', verbose=False):
    """FFT-based gradient descent with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.3
    momentum = 0.75
    v = np.zeros(n)
    stall = 0
    max_stall = 300

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        a_flip = a[::-1]
        faf = rfft(a_flip, N)

        if mode == 'energy':
            fb = rfft(b, N)
            c_full = irfft(faf * fb, N)[:3 * n - 2]
            grad = c_full[n - 1:2 * n - 1]
        elif mode == 'peak':
            peak_val = float(np.max(b))
            temp = max(peak_val * 0.05, 1e-15)
            b_sh = (b - peak_val) / temp
            w = np.exp(b_sh)
            w_sum = float(np.sum(w))
            if w_sum > 1e-30:
                w = w / w_sum
            else:
                w = np.zeros(2 * n - 1)
                w[np.argmax(b)] = 1.0
            fw = rfft(w, N)
            c_full = irfft(faf * fw, N)[:3 * n - 2]
            grad = c_full[n - 1:2 * n - 1]
        elif mode == 'multi':
            K = 3
            perm = np.argsort(b)[::-1]
            w = np.zeros(2 * n - 1)
            w[perm[:K]] = 1.0 / K
            fw = rfft(w, N)
            c_full = irfft(faf * fw, N)[:3 * n - 2]
            grad = c_full[n - 1:2 * n - 1]
        else:
            grad = b  # fallback

        grad = grad - np.mean(grad)

        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v = np.zeros(n)
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.003, 2.0)
            momentum = min(momentum + 0.0005, 0.95)
        else:
            stall += 1
            eta *= 0.97
            v *= 0.5
            if stall > max_stall:
                rng_local = np.random.default_rng(it * 17 + 3)
                a = best_a * np.exp(rng_local.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                s_val = float(np.sum(a))
                if s_val > 1e-10:
                    a = a / s_val
                v = np.zeros(n)
                eta = 0.3
                momentum = 0.75
                stall = 0
                max_stall = min(max_stall + 200, 1500)
            if eta < 1e-9:
                break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 1500:
        return None
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    if n > 1500:

    a = np.maximum(a, 0).astype(np.float64)

    for it in range(80):

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try

            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):

    mode = int(rng.integers(0, 8))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = float(np.sum(a))
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 10) ** 2, 1))
        a = a / np.sum(a)
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    elif mode == 6:
        a = np.ones(n)
        for i in range(1, n):
            if (i * i) % 4 == 1:
                a[i] = 0.3
        a = a / np.sum(a)
    else:
        x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = np.exp(-3 * (x - 0.5) ** 2) + 0.08
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: spectral spreading + exponentiated gradient + FFT GD + LP + multi-n.
    Novel: GS spectral spreading to flatten autocorrelation, and multiplicative weights update.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # Phase 1: GS Spectral Spreading (new technique)
    print(f"\n=== Phase 1: GS Spectral Spreading (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.12, 100.0)
    a_gs = gs_spectral_spread(best_seq, time_budget=alloc, verbose=True)
    obj_gs = compute_obj(a_gs)
    if obj_gs < best_score:
        best_score = obj_gs
        best_seq = a_gs.copy()
        print(f"  ** GS: obj={best_score:.8f} **")
    else:
        print(f"  GS: obj={obj_gs:.8f}")

    # Phase 2: Exponentiated Gradient (peak mode) - new
    print(f"\n=== Phase 2: Exponentiated Gradient (peak) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 150.0)
        a_eg = exponentiated_gradient(best_seq, time_budget=alloc, mode='peak', verbose=True)
        obj_eg = compute_obj(a_eg)
        if obj_eg < best_score:
            best_score = obj_eg
            best_seq = a_eg.copy()
            print(f"  ** EG: obj={best_score:.8f} **")
        else:
            print(f"  EG: obj={obj_eg:.8f}")

    # Phase 3: FFT GD (peak mode)
    print(f"\n=== Phase 3: FFT GD Peak ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 150.0)
        a_gd = fft_gradient_descent(best_seq, time_budget=alloc, mode='peak', verbose=False)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  ** GD Peak: obj={best_score:.8f} **")
        else:
            print(f"  GD Peak: obj={obj_gd:.8f}")

    # Phase 4: Exponentiated Gradient (multi-peak)
    print(f"\n=== Phase 4: EG Multi-Peak ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.1, 100.0)
        a_eg2 = exponentiated_gradient(best_seq, time_budget=alloc, mode='multi')
        obj_eg2 = compute_obj(a_eg2)
        if obj_eg2 < best_score:
            best_score = obj_eg2
            best_seq = a_eg2.copy()
            print(f"  ** EG Multi: obj={best_score:.8f} **")
        else:
            print(f"  EG Multi: obj={obj_eg2:.8f}")

    # Phase 5: GS Spectral Spreading Round 2
    print(f"\n=== Phase 5: GS Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.08, 60.0)
        a_gs2 = gs_spectral_spread(best_seq, time_budget=alloc)
        obj_gs2 = compute_obj(a_gs2)
        if obj_gs2 < best_score:
            best_score = obj_gs2
            best_seq = a_gs2.copy()
            print(f"  ** GS2: obj={best_score:.8f} **")
        else:
            print(f"  GS2: obj={obj_gs2:.8f}")

    # Phase 6: FFT GD (energy mode) for smooth refinement
    print(f"\n=== Phase 6: FFT GD Energy ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.1, 80.0)
        a_gd2 = fft_gradient_descent(best_seq, time_budget=alloc, mode='energy')
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            print(f"  ** GD Energy: obj={best_score:.8f} **")
        else:
            print(f"  GD Energy: obj={obj_gd2:.8f}")

    # Phase 7: LP (if n is reasonable)
    print(f"\n=== Phase 7: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 40 and n_best <= 1200:
        alloc = min(remaining * 0.08, 60.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # Phase 8: Multi-n exploration
    print(f"\n=== Phase 8: Multi-n Exploration ===")
    n_current = len(best_seq)
    n_options = sorted(set([
        n_current + 100, n_current + 200, n_current + 300,
        n_current - 100,
        900, 1000, 1100, 1200, 1500, 1800, 2000
    ]))
    n_options = [n for n in n_options if 500 <= n <= 2500 and n != n_current]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        a_try = resample_sequence(best_seq, n_try)
        # GS + EG + GD pipeline
        a_try = gs_spectral_spread(a_try, time_budget=min(remaining * 0.08, 25.0))
        a_try = exponentiated_gradient(a_try, time_budget=min(remaining * 0.1, 30.0), mode='peak')
        a_try = fft_gradient_descent(a_try, time_budget=min(remaining * 0.1, 30.0), mode='peak')
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 9: Random restarts
    print(f"\n=== Phase 9: Random Restarts ===")
    n_curr = len(best_seq)
    remaining = deadline - time.monotonic()
    n_restarts = max(2, int(remaining / 50.0))
    n_restarts = min(n_restarts, 6)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        a_r = random_init(n_curr, rng)
        a_r = gs_spectral_spread(a_r, time_budget=min(remaining * 0.15, 20.0))
        a_r = fft_gradient_descent(a_r, time_budget=min(remaining * 0.2, 30.0), mode='peak')
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # Phase 10: Final polish (alternate methods)
    print(f"\n=== Phase 10: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        # EG peak
        alloc = min(remaining * 0.15, 60.0)
        a_f = exponentiated_gradient(best_seq, time_budget=alloc, mode='peak')
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  ** Final EG: obj={best_score:.8f} **")

        # GD peak
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(remaining * 0.15, 50.0)
            a_f = fft_gradient_descent(best_seq, time_budget=alloc, mode='peak')
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  ** Final GD: obj={best_score:.8f} **")

        # GS
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.1, 30.0)
            a_f = gs_spectral_spread(best_seq, time_budget=alloc)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  ** Final GS: obj={best_score:.8f} **")

        # LP if small enough
        remaining = deadline - time.monotonic()
        if remaining > 20 and len(best_seq) <= 1000:
            alloc = min(remaining * 0.15, 25.0)
            a_f = lp_alternating(best_seq, time_budget=alloc)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  ** Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
