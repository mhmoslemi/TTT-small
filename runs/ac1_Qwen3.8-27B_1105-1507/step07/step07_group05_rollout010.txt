def compute_conv_fft(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build convolution matrix using Toeplitz structure (faster for large n)."""
    n = len(a)
    out_len = 2 * n - 1
    A = np.zeros((out_len, n))
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # For each column j: A[k, j] = a[k-j] for k in [j, j+n-1] (clamped to [0, 2n-2])
    for j in range(n):

        k_end = min(j + n, out_len)

        # Gapped: high at regularly spaced positions
        gap = int(n // rng.integers(3, 8))
        a = np.zeros(n)
        for i in range(0, n, gap):
            a[i] = 1.0
        # Smooth
        kernel = np.ones(3) / 3
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
    else:
        # Random log-normal
        a = np.exp(rng.normal(0, rng.uniform(0.3, 1.0), n))
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def quick_gd(a, time_budget=20.0):
    """Quick energy-mode GD for initialization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    
    eta = 0.5
    momentum = 0.8
    v = np.zeros(n)
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        a_flip = a[::-1]
        faf = rfft(a_flip, N)
        fb = rfft(b, N)
        c_full = irfft(faf * fb, N)[:3 * n - 2]
        grad = c_full[n - 1:2 * n - 1]
        grad = grad - np.mean(grad)
        
        v = momentum * v - eta * grad
        a_new = np.maximum(a + v, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 2.0)
        else:
            eta *= 0.97
            v *= 0.5
            if eta < 1e-6:
                break
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimization combining annealed Nesterov GD,
    LP alternating with backtracking, peak chase, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # --- Load parent ---
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: Quick multi-n scan (120s) ===
    print(f"\n=== Phase 1: Multi-n Scan ===")
    phase1_deadline = start_time + 120.0
    n_options = sorted(set([
        600, 700, n_best, 900, 1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000]
    
    for n_try in n_options:
        if time.monotonic() > phase1_deadline:
            break
        if n_try == n_best and best_seq is not None:
            continue
        
        # Try multiple initializations
        best_local = np.inf
        best_local_seq = None
        
        for init_mode in range(3):
            if time.monotonic() > phase1_deadline:
                break
            if init_mode == 0:
                a_init = np.ones(n_try) / n_try
            elif init_mode == 1:
                a_init = structured_init(n_try, rng)
            else:
                a_init = fourier_init(n_try, rng)
            
            a_opt = quick_gd(a_init, time_budget=15.0)
            obj = compute_obj(a_opt)
            if obj < best_local:
                best_local = obj
                best_local_seq = a_opt.copy()
        
        if best_local_seq is not None:
            if best_local < best_score:
                best_score = best_local
                best_seq = best_local_seq.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={best_local:.8f}")
    
    # === Phase 2: Annealed Nesterov GD (350s) ===
    print(f"\n=== Phase 2: Annealed Nesterov GD (n={len(best_seq)}) ===")
    phase2_deadline = min(start_time + 470.0, deadline)
    alloc = min(phase2_deadline - time.monotonic(), 350.0)
    if alloc > 30:
        a_annealed = annealed_nesterov_gd(best_seq, time_budget=alloc, T_init=0.08, T_min=0.003, verbose=True)
        obj = compute_obj(a_annealed)
        if obj < best_score:
            best_score = obj
            best_seq = a_annealed.copy()
            print(f"  ** Annealed GD: obj={best_score:.8f} **")
        else:
            print(f"  Annealed GD: obj={obj:.8f}")
    
    # === Phase 3: LP Alternating (300s) ===
    print(f"\n=== Phase 3: LP Alternating (n={len(best_seq)}) ===")
    phase3_deadline = min(start_time + 770.0, deadline)
    alloc = min(phase3_deadline - time.monotonic(), 300.0)
    if alloc > 60:
        a_lp = lp_alternating_v2(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj:.8f}")
    
    # === Phase 4: Second Annealed GD round (200s) ===
    print(f"\n=== Phase 4: Annealed GD Round 2 ===")
    phase4_deadline = min(start_time + 970.0, deadline)
    alloc = min(phase4_deadline - time.monotonic(), 200.0)
    if alloc > 30:
        a_annealed2 = annealed_nesterov_gd(best_seq, time_budget=alloc, T_init=0.04, T_min=0.002, verbose=True)
        obj = compute_obj(a_annealed2)
        if obj < best_score:
            best_score = obj
            best_seq = a_annealed2.copy()
            print(f"  ** Annealed GD2: obj={best_score:.8f} **")
        else:
            print(f"  Annealed GD2: obj={obj:.8f}")
    
    # === Phase 5: Peak Chase + Fine LP (200s) ===
    print(f"\n=== Phase 5: Peak Chase + Fine LP ===")
    phase5_deadline = min(start_time + 1170.0, deadline)
    alloc = min(phase5_deadline - time.monotonic(), 200.0)
    if alloc > 20:
        a_pc = peak_chase_v2(best_seq, time_budget=alloc * 0.4)
        obj = compute_obj(a_pc)
        if obj < best_score:
            best_score = obj
            best_seq = a_pc.copy()
            print(f"  ** Peak Chase: obj={best_score:.8f} **")
    
    alloc = min(phase5_deadline - time.monotonic(), 80.0)
    if alloc > 30:
        a_lp2 = lp_alternating_v2(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp2)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()
            print(f"  ** Fine LP: obj={best_score:.8f} **")
    
    # === Phase 6: Multi-n transfer optimization (200s) ===
    print(f"\n=== Phase 6: Multi-n Transfer ===")
    n_curr = len(best_seq)
    n_transfers = sorted(set([
        n_curr - 100, n_curr + 100,
        n_curr - 200, n_curr + 200,
        n_curr + 400
    ]))
    n_transfers = [n for n in n_transfers if 400 <= n <= 2500 and n != n_curr]
    n_transfers = n_transfers[:5]
    
    for n_try in n_transfers:
        if time.monotonic() > deadline:
            break
        a_resampled = resample_sequence(best_seq, n_try)
        
        # Quick optimization at new n
        alloc1 = 30.0
        a_gd = annealed_nesterov_gd(a_resampled, time_budget=min(alloc1, max(10, (deadline - time.monotonic()) * 0.3)), T_init=0.05, T_min=0.005)
        obj_gd = compute_obj(a_gd)
        
        alloc2 = 30.0
        if obj_gd < best_score:
            a_lp = lp_alternating_v2(a_gd, time_budget=min(alloc2, max(10, (deadline - time.monotonic()) * 0.3)))
            obj_lp = compute_obj(a_lp)
            if obj_lp < obj_gd:
                a_gd = a_lp
                obj_gd = obj_lp
        
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            n_curr = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_gd:.8f}")
    
    # === Phase 7: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # LP with small steps
        alloc = min(remaining * 0.3, 120.0)
        a_final_lp = lp_alternating_v2(best_seq, time_budget=alloc, verbose=False)
        obj = compute_obj(a_final_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_final_lp.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Annealed GD final
        alloc = min(remaining * 0.4, 100.0)
        a_final_gd = annealed_nesterov_gd(best_seq, time_budget=alloc, T_init=0.02, T_min=0.001)
        obj = compute_obj(a_final_gd)
        if obj < best_score:
            best_score = obj
            best_seq = a_final_gd.copy()
            print(f"  ** Final GD: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Peak chase
        a_final_pc = peak_chase_v2(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj = compute_obj(a_final_pc)
        if obj < best_score:
            best_score = obj
            best_seq = a_final_pc.copy()
            print(f"  ** Final Peak: obj={best_score:.8f} **")
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
