def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def solve_lp_minpeak(a, n):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0"""
    a = a / np.sum(a)
    # Build constraint matrix A (2n-1 x n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    # Try cvxpy solvers
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback to linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

def smoothed_grad_step(a, tau, n):
    """One step of smoothed gradient descent on the simplex."""
    a = a / np.sum(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    w = tau * np.exp((conv - conv_max) * tau)
    w = w / np.sum(w)
    
    # Compute c_j = sum_m w[j+m] * a[m] for j = 0..n-1
    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < 2 * n - 1:
            c[j] = np.dot(w[j:end], a[:end - j])
    
    mean_c = np.dot(c, a)
    grad = 4.0 * n * (c - mean_c)
    # Project to tangent space (subtract mean)
    grad = grad - np.mean(grad)
    return grad

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, n):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def smoothed_grad_direction(a, tau, n):
    """Compute smoothed gradient direction (on simplex tangent space)."""
    a = a / np.sum(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    w = tau * np.exp((conv - conv_max) * tau)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum

    c = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if j < 2 * n - 1:
            c[j] = np.dot(w[j:end], a[:end - j])

    mean_c = np.dot(c, a)
    grad = 4.0 * n * (c - mean_c)
    grad = grad - np.mean(grad)
    return grad

        s_new = np.sum(a_new)

def coordinate_descent_peak(a, time_budget, max_iter=300):

                        # Target: move to an index that doesn't contribute to this peak
                        # or to the opposite end
                        target = j  # move from i to j (symmetric move for this peak)
                        # Actually moving from i to j changes c[k_star] by:
                        # -delta*a[j] + a[i]*delta (net change depends on values)
                        # Better: move from i to an index far from the peak
                        safe_target = (i + n // 2) % n
                        if safe_target == i:
                            safe_target = (i + 1) % n

                        # Try both targets
                        for tgt in [j, safe_target]:
                            if tgt == i:
                                continue
                            a[i] -= delta
                            a[tgt] += delta
                            obj_new = compute_obj(a)
                            if obj_new < best_obj - 1e-14:
                                best_obj = obj_new
                                best_a = a.copy()
                                improved = True
                            else:
                                a[i] += delta
                                a[tgt] -= delta
                            if improved:
                                break
                        if improved:
                            break
                if improved:
                    break
            if improved:
                a = best_a.copy()
                continue

        if not improved:
            # Random perturbations
            rng = np.random.default_rng(it * 104729 + n)
            any_improved = False
            for _ in range(40):
                if time.monotonic() - start > time_budget:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-10:
                    continue
                delta = a[i] * rng.uniform(0.005, 0.05)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a.copy()
                        any_improved = True
                    else:
                        a[i] += delta
                        a[j] -= delta
            if not any_improved:
                # Larger perturbation to escape
                scale = 0.03 + 0.01 * (it % 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                if it > 50:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_diverse_inits(n, rng, best_seq=None):
    """Generate diverse initializations for sequence of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential ramps (various rates and directions)
    for rate in [0.8, 1.5, 2.5, 4.0]:
        a = np.exp(np.linspace(0, rate, n))
        inits.append(a / np.sum(a))
        a = np.exp(np.linspace(rate, 0, n))
        inits.append(a / np.sum(a))

    # Bimodal (mass at edges)
    for center_width in [0.1, 0.2, 0.3]:
        a = np.ones(n)
        mask = np.abs(np.arange(n) - n // 2) > center_width * n // 2
        a[~mask] = 0.1
        inits.append(a / np.sum(a))

    # Triangular
    a = np.abs(np.linspace(-1, 1, n))
    inits.append(a / np.sum(a))
    a = 1.0 - np.abs(np.linspace(-1, 1, n))
    inits.append(a / np.sum(a))

    # Sine-based
    for freq in [1.0, 2.0, 3.0, 5.0]:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        inits.append(a / np.sum(a))

    # Power law
    for alpha in [0.5, 1.0, 2.0]:
        a = np.power(np.arange(1, n + 1), -alpha)
        inits.append(a / np.sum(a))

    # Resampled parent
    if best_seq is not None and len(best_seq) > 0:
        inits.append(resample_sequence(best_seq, n))

    # Smooth random
    a = np.exp(rng.normal(0, 0.5, n))
    # Smooth it
    kernel_size = max(3, n // 50)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    inits.append(a / np.sum(a))

    return inits


def coarse_to_fine(a_coarse, n_fine, time_budget, rng):
    """Upsample coarse solution to fine resolution and refine."""
    a_fine = resample_sequence(a_coarse, n_fine)
    # Add small random fine structure
    a_fine = a_fine * np.exp(rng.normal(0, 0.03, n_fine))
    a_fine = np.maximum(a_fine, 0)
    s = np.sum(a_fine)
    if s > 1e-10:
        a_fine = a_fine / s
    return lp_alternating(a_fine, time_budget=time_budget, max_iter=100)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale hybrid optimizer:
    1. Coarse optimization at small n
    2. Upsample and refine at larger n
    3. LP alternating with structured perturbations
    4. Smoothed gradient + coordinate descent polishing
    5. Multi-n exploration and final refinement
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Quick diverse screening at multiple n ===
    print(f"\n=== Phase 1: Diverse Screening ===")
    n_screen = [600, 800, 1000, 1200, 1500]
    candidates = []

    for n_try in n_screen:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        inits = generate_diverse_inits(n_try, rng, best_seq)

        for a_init in inits:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_init = np.maximum(a_init, 0)
            s = np.sum(a_init)
            if s < 1e-10:
                continue
            a_init = a_init / s
            obj_init = compute_obj(a_init)

            # Quick refinement (short LP alternating)
            alloc = min(remaining * 0.08, 12.0)
            a_ref = lp_alternating(a_init, time_budget=alloc, max_iter=30)
            obj_ref = compute_obj(a_ref)

            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  n={n_try}: obj={best_score:.8f} **")
            candidates.append((obj_ref, a_ref.copy()))

    # Sort and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidate: obj={candidates[0][0]:.8f} (n={len(candidates[0][1])})")

    # === Phase 2: Coarse-to-fine refinement ===
    print(f"\n=== Phase 2: Coarse-to-Fine ===")
    # Use the best coarse solution to refine at larger n
    best_coarse = min(candidates, key=lambda x: x[0])[1]
    n_coarse = len(best_coarse)

    target_ns = sorted(set([n for n in [n_coarse + 200, n_coarse + 400, 1200, 1500, 1800, 2000]
                            if n > n_coarse and n <= 2200]))

    for n_target in target_ns:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining * 0.15, 60.0)
        a_fine = coarse_to_fine(best_coarse, n_target, alloc, rng)
        obj_fine = compute_obj(a_fine)
        if obj_fine < best_score:
            best_score = obj_fine
            best_seq = a_fine.copy()
            print(f"  Coarse-to-fine n={n_target}: obj={best_score:.8f} **")
        else:
            print(f"  Coarse-to-fine n={n_target}: obj={obj_fine:.8f}")

    # === Phase 3: Intensive optimization on top candidates ===
    print(f"\n=== Phase 3: Intensive Optimization ===")

    # Build working set: best + a few diverse candidates
    working_set = [best_seq.copy()]
    for obj_c, a_c in candidates[:5]:
        if len(a_c) != len(best_seq):
            a_c = resample_sequence(a_c, len(best_seq))
        if np.linalg.norm(a_c - best_seq) / np.linalg.norm(best_seq) > 0.1:
            working_set.append(a_c.copy())

    for wi, a_work in enumerate(working_set[:4]):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        n_w = len(a_work)
        alloc_total = min(remaining * 0.25, 150.0)

        # 3a: LP alternating (primary)
        alloc_lp = alloc_total * 0.45
        a_lp = lp_alternating(a_work, time_budget=alloc_lp, max_iter=200)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Work {wi} LP (n={n_w}): obj={best_score:.8f} **")
        else:
            print(f"  Work {wi} LP (n={n_w}): obj={obj_lp:.8f}")
        a_work = best_seq.copy() if obj_lp < best_score else a_lp

        # 3b: Smoothed gradient
        remaining = deadline - time.monotonic()
        if remaining < 30:
            continue
        alloc_sg = min(remaining * 0.2, 45.0)
        a_sg = smooth_gradient_descent(a_work, time_budget=alloc_sg,
                                        tau_start=30.0, tau_end=800.0, max_iter=150)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Work {wi} SG (n={n_w}): obj={best_score:.8f} **")
        else:
            print(f"  Work {wi} SG (n={n_w}): obj={obj_sg:.8f}")
        a_work = best_seq.copy()

        # 3c: Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining < 20:
            continue
        alloc_cd = min(remaining * 0.15, 35.0)
        a_cd = coordinate_descent_peak(a_work, time_budget=alloc_cd, max_iter=200)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Work {wi} CD (n={n_w}): obj={best_score:.8f} **")
        else:
            print(f"  Work {wi} CD (n={n_w}): obj={obj_cd:.8f}")

    # === Phase 4: Deep focus on best ===
    print(f"\n=== Phase 4: Deep Focus ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    if remaining > 120:
        # Extended LP alternating
        alloc = min(remaining * 0.35, 150.0)
        a_ext = lp_alternating(best_seq, time_budget=alloc, max_iter=400, verbose=True)
        obj_ext = compute_obj(a_ext)
        if obj_ext < best_score:
            best_score = obj_ext
            best_seq = a_ext.copy()
            print(f"  Extended LP: obj={best_score:.8f} **")
        else:
            print(f"  Extended LP: obj={obj_ext:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 60:
        # High-tau smoothed gradient
        alloc = min(remaining * 0.25, 80.0)
        a_sg2 = smooth_gradient_descent(best_seq, time_budget=alloc,
                                         tau_start=80.0, tau_end=2000.0, max_iter=250)
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  High-tau SG: obj={best_score:.8f} **")
        else:
            print(f"  High-tau SG: obj={obj_sg2:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Extended coordinate descent
        alloc = min(remaining * 0.3, 50.0)
        a_cd2 = coordinate_descent_peak(best_seq, time_budget=alloc, max_iter=400)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Extended CD: obj={best_score:.8f} **")
        else:
            print(f"  Extended CD: obj={obj_cd2:.8f}")

    # === Phase 5: N-adjustment exploration ===
    print(f"\n=== Phase 5: N-Adjustment ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_adj = sorted(set([n_cur + d for d in [-100, -50, 50, 100, 200, 300]
                        if 500 <= n_cur + d <= 2500]))

    for n_try in n_adj:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        a_try = resample_sequence(best_seq, n_try)
        # Add small perturbation
        a_try = a_try * np.exp(rng.normal(0, 0.02, n_try))
        a_try = np.maximum(a_try, 0)
        a_try = a_try / np.sum(a_try)

        alloc = min(remaining * 0.15, 35.0)
        a_opt = lp_alternating(a_try, time_budget=alloc, max_iter=80)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Alternating polish: LP -> CD -> LP -> SG
    for polish_round in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # LP
        alloc = min(remaining * 0.25, 40.0)
        a_p = lp_alternating(best_seq, time_budget=alloc, max_iter=100)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()

        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # CD
        alloc = min(remaining * 0.2, 25.0)
        a_p = coordinate_descent_peak(best_seq, time_budget=alloc, max_iter=150)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()

        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # SG
        alloc = min(remaining * 0.2, 25.0)
        a_p = smooth_gradient_descent(best_seq, time_budget=alloc,
                                       tau_start=100.0, tau_end=2000.0, max_iter=100)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Polish {polish_round}: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
