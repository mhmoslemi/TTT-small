```python
def lp_gradient(a, p, time_budget):
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_val = np.sum(compute_conv(a)**p)
    start = time.monotonic()
    
    lr = 1.0 / (p * n)  # initial learning rate
    momentum = np.zeros(n)
    beta = 0.9  # momentum parameter
    
    while time.monotonic() - start < time_budget:
        c = compute_conv(a)
        # Weight for each lag
        w = p * c**(p-1)  # length 2n-1
        # Gradient: grad[i] = sum_k w[k] * a[k-i] (valid k)
        # This is cross-correlation of w with a
        grad = cross_correlate(w, a)  # length n
        # Center the gradient to preserve sum
        grad = grad - np.mean(grad)
        
        # Momentum update
        momentum = beta * momentum + (1-beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s
        
        # Check improvement
        c_new = compute_conv(a_new)
        val_new = np.sum(c_new**p)
        
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.1, 1.0)
        else:
            lr *= 0.7
            a = best_a.copy()

def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

    conv = compute_conv(a)  # length 2n-1, all non-negative since a >= 0
    if p <= 1:
        w = np.ones(2 * n - 1)
    else:
        w = p * np.power(np.maximum(conv, 0), p - 1)
    
    # grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # Via FFT: conv(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1 + 1:  # 3n-1
        size <<= 1
    W = rfft(w, size)
    A = rfft(a_rev, size)
    result = irfft(W * A, size)
    grad = result[n - 1:2 * n - 1].copy()
    # Center to preserve sum
    grad -= np.mean(grad)
    return grad


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_nesterov(a, p, time_budget, lr_init=0.1, beta=0.9, verbose=False):
    """
    Nesterov-accelerated projected gradient on Lp norm of auto-convolution.
    a is assumed to be on the simplex (sum=1, a>=0).
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_val = float(np.sum(compute_conv(a) ** p))
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    stall = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        # Compute gradient
        grad = compute_grad_lp(a, p)
        
        # Nesterov update
        v = beta * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = compute_conv(a_new)
        val_new = float(np.sum(conv_new ** p))
        
        if val_new < best_val * (1 - 1e-12):
            best_val = val_new
            best_a = a_new.copy()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
            a = a_new
            lr = min(lr * 1.02, 1.0)
            stall = 0
        else:
            stall += 1
            a = best_a.copy()
            if stall % 10 == 0:
                lr *= 0.8
                v *= 0.5
            if stall > 200:
                # Random restart from best
                rng = np.random.default_rng((it * 7919 + n) % (2**32))
                a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
                a_pert = project_simplex(a_pert)
                val_pert = float(np.sum(compute_conv(a_pert) ** p))
                if val_pert < best_val:
                    best_val = val_pert
                    best_a = a_pert.copy()
                a = best_a.copy()
                lr = lr_init
                v = np.zeros(n)
                stall = 0
                if verbose:
                    print(f"    L{p} it{it}: restart, val={best_val:.10f}")
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    L{p} it{it}: val={best_val:.10f}, obj={best_obj:.8f}, lr={lr:.4f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_cross(a, time_budget=60.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g (on simplex) or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None:
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs
            except Exception:
                continue
    except Exception:
        pass
    
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    
    return None


def lp_alternate(a, time_budget=60.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        g = solve_lp_cross(a, time_budget=min(30.0, (time_budget - (time.monotonic() - start)) * 0.5))
        if g is None:
            break
        
        # Line search on t
        best_t = 0.5
        best_t_obj = np.inf
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_simplex((1 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            a = project_simplex((1 - best_t) * a + best_t * g)
            best_obj = compute_obj(a)
            best_a = a.copy()
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP alt it{it}: obj={best_obj:.8f}, t={best_t:.1f}, time={elapsed:.1f}s")
        else:
            # Stuck, try perturbation
            rng = np.random.default_rng((it * 31 + n) % (2**32))
            a_pert = best_a * np.exp(rng.normal(0, 0.03, n))
            a = project_simplex(a_pert)
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence on the simplex."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
    elif mode == 2:
        # Smooth random
        a = rng.uniform(0, 1, n)
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=max(1, n/100))
    elif mode == 3:
        # Cosine-like
        x = np.linspace(0, np.pi, n)
        a = np.cos(x)**2 + 0.1
    elif mode == 4:
        # Low-frequency
        x = np.arange(n)
        freqs = rng.uniform(0.1, 2.0, 3)
        phases = rng.uniform(0, 2*np.pi, 3)
        a = 1.0 + 0.3 * sum(np.sin(2*np.pi*f*x/n + ph) for f, ph in zip(freqs, phases))
        a = np.maximum(a, 0.01)
    else:
        # Exponential decay from center
        center = n // 2
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - center) / (n * 0.3))**2)
        a = a + 0.1
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n

    Multi-stage optimization of a sequence on the simplex.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if verbose:
        print(f"  Initial: n={n}, obj={best_obj:.8f}")
    
    # Stage 1: Lp with homotopy (main workhorse)
    p_values = [4, 8, 16, 32, 64]
    per_p_time = time_budget * 0.5 / len(p_values)
    
    for p in p_values:
        remaining = time_budget * 0.5 - (time.monotonic() - start)
        if remaining < 5:
            break
        a_curr, obj_curr = lp_nesterov(best_a, p, time_budget=min(per_p_time, remaining), verbose=verbose)
        if obj_curr < best_obj:
            best_obj = obj_curr
            best_a = a_curr.copy()
            if verbose:
                print(f"  After L{p}: obj={best_obj:.8f}")

            best_a = a_lp.copy()
