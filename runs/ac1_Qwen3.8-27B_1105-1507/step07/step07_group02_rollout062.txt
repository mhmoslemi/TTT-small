"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses mirror descent (exponential gradient), LP alternating optimization,
trigonometric perturbation, and multi-scale search. 
Different approach from parent: primary optimization via mirror descent
with Nesterov acceleration, supplemented by LP and frequency-domain methods.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a, timeout=30.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_work = a / s
    A = build_conv_matrix(a_work)
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
        for solver in solvers:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs, float(M.value)
            except Exception:
                continue
    except Exception:
        pass
    
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def mirror_descent_optimize(a, time_budget=60.0, verbose=False):
    """
    Mirror descent (exponential gradient) for minimizing max auto-convolution peak.
    Uses subgradient of the peak value with adaptive step size and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.1  # Initial step size
    momentum = 0.9  # Nesterov-like momentum
    v = np.zeros(n)  # Momentum accumulator
    stall = 0
    best_stall_reward = best_obj
    no_improve = 0
    
    prev_a = a.copy()
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Find all positions where conv is close to max (for averaging subgradient)
        max_val = np.max(conv)
        threshold = max_val * 0.999
        active = np.where(conv >= threshold)[0]
        
        # Subgradient: average over active positions
        g = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += a[j]
        g /= len(active)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * g
        v_centered = v - np.mean(v)
        
        # Multiplicative update
        a_raw = a * np.exp(-eta * v_centered)
        a_raw = np.maximum(a_raw, 1e-30)
        s_raw = float(np.sum(a_raw))
        a_new = a_raw / s_raw
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Increase step size slightly
            eta = min(eta * 1.02, 0.5)
            if verbose and it % 100 == 0:
                print(f"        MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}")
        else:
            no_improve += 1
            if no_improve > 50:
                # Restart with perturbation
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                best_obj = compute_obj(a)
                best_a = a.copy()
                eta = 0.05
                v = np.zeros(n)
                no_improve = 0
            else:
                eta *= 0.95
        
    return best_a


def subgradient_optimize(a, time_budget=60.0, verbose=False):
    """
    Subgradient method with projection onto simplex.
    Different from mirror descent - uses additive updates with projection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Polyak step size
    obj_lower = 1.0  # Theoretical lower bound
    t = 1.0
    no_improve = 0
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = np.max(conv)
        threshold = max_val * 0.999
        active = np.where(conv >= threshold)[0]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += a[j]
        g /= len(active)
        
        # Center gradient (to preserve sum)
        g_centered = g - np.mean(g)
        
        # Polyak step size
        step = (best_obj - obj_lower) / (np.dot(g_centered, g_centered) + 1e-10)
        step = min(step, 0.5)
        
        a_new = a - step * g_centered
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"        SG it{it}: obj={best_obj:.8f}")
        else:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                no_improve = 0
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def subgradient_peak(a, conv):
    """Compute subgradient of max(conv) w.r.t. a, averaged over near-max positions."""
    n = len(a)
    max_val = np.max(conv)
    threshold = max_val * 0.995
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    g = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            g[i_lo:i_hi + 1] += a[k - i_lo:k - i_hi - 1:-1]
    g /= len(active)
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a, timeout=30.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_work = a / s
    A = build_conv_matrix(a_work)

    try:
        import cvxpy as cp
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
        for solver in solvers:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs, float(M_var.value)
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    return None, None


def mirror_descent_optimize(a, time_budget=60.0, verbose=False):
    """Mirror descent with momentum for minimizing peak auto-convolution."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.08
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        g = subgradient_peak(a, conv)
        g_centered = g - np.mean(g)

        v = momentum * v + (1.0 - momentum) * g_centered
        v_centered = v - np.mean(v)

        a_raw = a * np.exp(-eta * v_centered)
        a_raw = np.maximum(a_raw, 1e-30)
        s_raw = float(np.sum(a_raw))
        a_new = a_raw / s_raw

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(eta * 1.01, 0.4)
            if verbose and it % 200 == 0:
                print(f"        MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}")
        else:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng((it * 7919 + n) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.06, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                eta = 0.05
                v = np.zeros(n)
                no_improve = 0
            elif no_improve > 30:
                eta *= 0.95

    return best_a


def subgradient_optimize(a, time_budget=60.0, verbose=False):
    """Subgradient method with projection onto simplex and Polyak steps."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    momentum = 0.7
    v = np.zeros(n)

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        g = subgradient_peak(a, conv)
        g_centered = g - np.mean(g)

        v = momentum * v + (1.0 - momentum) * g_centered
        v_centered = v - np.mean(v)

        grad_norm_sq = np.dot(v_centered, v_centered) + 1e-10
        step = (best_obj - 1.0) / grad_norm_sq
        step = min(step, 0.3)

        a_new = a - step * v_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"        SG it{it}: obj={best_obj:.8f}")
        else:
            no_improve += 1
            if no_improve > 150:
                rng = np.random.default_rng((it * 31 + n) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                no_improve = 0
            elif no_improve > 50:
                momentum = max(0.3, momentum * 0.98)

    return best_a


def lp_alternating_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating optimization with adaptive mixing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    last_t = 1.0

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t_val = 1.0
        best_a_local = best_a.copy()

        t_candidates = [0.3, 0.5, 0.7, 0.85, 1.0]
        if 0 < last_t < 1:
            t_candidates = sorted(set(t_candidates + [
                max(0.1, last_t * 0.5), max(0.1, last_t * 1.3)
            ]))

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"        LP it{it}: obj={best_obj:.8f}, t={last_t:.2f}")
        else:
            no_improve += 1
            if no_improve > 10:
                rng = np.random.default_rng((it * 7919 + n) % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                no_improve = 0

    return best_a


def trigonometric_optimize(a, time_budget=60.0, verbose=False):
    """
    Optimize via trigonometric perturbation amplitudes.
    Uses coordinate descent on harmonic parameters.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    x = np.arange(n)

    for M in [2, 3, 5, 8]:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        M = min(M, n // 10)
        if M < 1:
            continue

        # Parameters: amplitude and phase for each harmonic
        params = {}
        for m in range(1, M + 1):
            params[m] = [0.0, 0.0]  # [amplitude, phase]

        current_a = a.copy()
        current_obj = compute_obj(current_a)
        improved = True
        max_coord_iters = 50

        for coord_it in range(max_coord_iters):
            if time.monotonic() - start > time_budget:
                break
            best_delta = 1e-10
            best_param = None
            best_val = None

            for m in range(1, M + 1):
                for pidx in [0, 1]:  # 0=amplitude, 1=phase
                    for delta in [0.05, 0.1, 0.2, 0.3, -0.05, -0.1, -0.2, -0.3]:
                        params[m][pidx] += delta
                        trial_a = current_a.copy()
                        for mm in range(1, M + 1):
                            amp = params[mm][0]
                            phase = params[mm][1]
                            if abs(amp) > 1e-8:
                                trial_a = trial_a * (1.0 + amp * np.cos(
                                    2.0 * np.pi * mm * x / n + phase))
                        trial_a = np.maximum(trial_a, 0)
                        s_t = float(np.sum(trial_a))
                        if s_t < 1e-10:
                            params[m][pidx] -= delta
                            continue
                        trial_a = trial_a / s_t
                        obj_t = compute_obj(trial_a)
                        if obj_t < best_obj - 1e-10:
                            best_delta = delta
                            best_param = (m, pidx)
                            best_val = obj_t
                        params[m][pidx] -= delta

            if best_param is not None:
                m, pidx = best_param
                params[m][pidx] += best_delta
                current_a = best_a.copy()
                for mm in range(1, M + 1):
                    amp = params[mm][0]
                    phase = params[mm][1]
                    if abs(amp) > 1e-8:
                        current_a = current_a * (1.0 + amp * np.cos(
                            2.0 * np.pi * mm * x / n + phase))
                current_a = np.maximum(current_a, 0)
                s_c = float(np.sum(current_a))
                if s_c > 1e-10:
                    current_a = current_a / s_c
                    best_obj = best_val
                    best_a = current_a.copy()
                    a = current_a
            else:
                break

    if verbose:
        print(f"        Trig: obj={best_obj:.8f}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
