```python
import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))

def smooth_gradient(a, beta):
    """
    Compute smoothed gradient of max_k conv(a,a)_k.
    Returns gradient vector of length n.
    """
    n = len(a)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    
    # Compute b = conv(a, a), length 2n-1
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]
    
    # Compute weights w = exp(beta * b), normalized
    b_shifted = b - np.max(b)
    w = np.exp(beta * b_shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        # Fallback: uniform weight (like energy gradient)
        w = np.ones(2 * n - 1)
        w_sum = 2 * n - 1
    w = w / w_sum
    
    # Compute g[i] = sum_j a[j] * w[i+j] for i=0,...,n-1
    # This is conv(flip(a), w) at positions [n-1, ..., 2n-2]
    a_flip = a[::-1]
    faf = rfft(a_flip, N)
    fw = rfft(w, N)
    c = irfft(faf * fw, N)[:3 * n - 2]  # conv of two length-n and length-(2n-1) arrays
    g = c[n - 1:2 * n - 1]
    
    return g, b
```

```python
"""
Multi-strategy optimization for minimizing 2n*max(autoconv(a,a))/(sum(a))^2.
Innovations over previous: mirror descent (multiplicative weights) for natural simplex
handling, adaptive smoothed gradient with FFT, U-shaped initialization based on
theoretical insight about edge concentration, peak-reduction heuristic, and
aggressive multi-n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def smooth_gradient(a, beta):
    """
    Compute smoothed gradient of max_k conv(a,a)_k using softmax weighting.
    Returns (gradient, convolution, max_b).
    Uses FFT for O(N log N) computation.
    """
    n = len(a)
    N = 1
    while N < 3 * n - 2:
        N <<= 1

    # Compute b = conv(a, a), length 2n-1
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]

    # Compute normalized weights w = softmax(beta * b)
    b_shifted = b - np.max(b)
    w = np.exp(beta * b_shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum

    # g[i] = sum_j a[j] * w[i+j] for i=0,...,n-1
    # = conv(flip(a), w)[n-1 : 2n-1]
    a_flip = a[::-1]
    faf = rfft(a_flip, N)
    fw = rfft(w, N)
    c = irfft(faf * fw, N)[:3 * n - 2]
    g = c[n - 1:2 * n - 1]

    return g, b, float(np.max(b))

    """
    Mirror descent (multiplicative weights) on the probability simplex.
    Uses smoothed subgradient with adaptive temperature.
    """

        return a
    a = a / s

    best_obj = compute_obj(a)

    beta = beta_init
    stall = 0
    max_stall = 150
    rng = np.random.default_rng(12345)

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        g, b, max_b = smooth_gradient(a, beta)

        # Center gradient for numerical stability
        g = g - np.mean(g)

        # Mirror descent: a_i <- a_i * exp(-eta * g_i) / Z
        log_a = np.log(np.maximum(a, 1e-300))
        log_new = log_a - eta * g
        log_new -= np.max(log_new)
        new_unnorm = np.exp(log_new)
        Z = np.sum(new_unnorm)
        if Z < 1e-300:

                a = np.maximum(a, 0)

def peak_reduction(a, n_steps=20, time_budget=30.0):
    """
    Directly reduce the peak by decreasing coordinates that contribute most to it.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv(a)
        k_star = int(np.argmax(b))

        # Contribution of each a_i to the peak at k_star
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Reduce contributing coordinates proportionally
        reduction = 0.02 * contrib / (np.max(contrib) + 1e-300)
        a_new = a - reduction
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def u_shaped_init(n, alpha=2.0):
    """U-shaped initialization: concentrate mass at edges."""
    x = np.linspace(0, 1, n)
    a = (x - 0.5) ** 2 * 4 * alpha + 0.1
    a = a / np.sum(a)
    return a


def bimodal_init(n, width_frac=0.1):
    """Bimodal: two Gaussian bumps at quarter and three-quarter positions."""
    x = np.arange(n, dtype=np.float64)
    centers = [n * 0.2, n * 0.8]
    width = max(n * width_frac, 1.0)
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-((x - c) ** 2) / (2 * width ** 2))
    a += 0.01
    a = a / np.sum(a)
    return a


def edge_concentrated_init(n, decay=0.5):
    """Edge-concentrated: exponential decay from edges toward center."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(x - center) / center
    a = np.exp(-decay * dist) + 0.05
    a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_sequence(a, time_budget=60.0, verbose=False):
    """
    Full pipeline optimization for a single sequence:
    Mirror Descent -> Additive GD -> Peak Reduction -> LP Polish
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Step 1: Mirror Descent
    remaining = start + time_budget - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.35, 30.0)
        a_md = mirrored_descent(a, time_budget=alloc, beta_init=5.0)
        obj_md = compute_obj(a_md)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()
            a = a_md.copy()
            if verbose:
                print(f"          MD: {best_obj:.8f}")

    # Step 2: Additive GD (different dynamics, may find different basin)
    remaining = start + time_budget - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 20.0)
        a_gd = additive_gd(a, time_budget=alloc, beta_init=5.0)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_obj:
            best_obj = obj_gd
            best_a = a_gd.copy()
            a = a_gd.copy()
            if verbose:
                print(f"          AGD: {best_obj:.8f}")

    # Step 3: Peak Reduction
    remaining = start + time_budget - time.monotonic()
    if remaining > 10:
        a_pr = peak_reduction(a, n_steps=30, time_budget=min(remaining * 0.15, 15.0))
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_obj:
            best_obj = obj_pr
            best_a = a_pr.copy()
            a = a_pr.copy()
            if verbose:
                print(f"          PeakRed: {best_obj:.8f}")

    # Step 4: LP Polish (expensive but precise)
    remaining = start + time_budget - time.monotonic()
    if remaining > 20 and n <= 2000:
        alloc = min(remaining * 0.3, 30.0)
        a_lp = lp_alternating(a, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp.copy()
            if verbose:
                print(f"          LP: {best_obj:.8f}")

    # Step 5: Final short MD round
    remaining = start + time_budget - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 15.0)
        a_final = mirrored_descent(a, time_budget=alloc, beta_init=10.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            if verbose:
                print(f"          FinalMD: {best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-n exploration with diverse initializations,
    mirror descent, additive GD, peak reduction, and LP polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")
        parent = None

    if best_seq is None:
        best_seq = u_shaped_init(800)
        best_score = compute_obj(best_seq)
        print(f"Default U-shape init (n=800): obj={best_score:.8f}")
        parent = best_seq.copy()

    n_parent = len(best_seq)

    # Define n values to explore
    n_values = sorted(set([
        n_parent,
        max(400, n_parent - 200),
        n_parent + 200,
        n_parent + 400,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    n_values = [n for n in n_values if 400 <= n <= 2500]
    # Sort by time complexity (smaller first for quick wins)
    n_values.sort()

    print(f"\nExploring n values: {n_values}")

    for n_try in n_values:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == len(best_seq) and best_seq is not None:
            # Already optimized this n in parent - skip or do lighter
            alloc_per_init = min(remaining * 0.15, 40.0)
        else:
            alloc_per_init = min(remaining * 0.12, 50.0)

        if remaining < alloc_per_init * 2:
            alloc_per_init = remaining * 0.4

        print(f"\n--- n={n_try} (alloc {alloc_per_init:.0f}s per init) ---")

        # Generate initializations for this n
        inits = []
        init_names = []

        # Resample parent
        if parent is not None:
            inits.append(resample_sequence(parent, n_try) / np.sum(resample_sequence(parent, n_try)))
            init_names.append("parent_resample")

        # U-shaped variants
        inits.append(u_shaped_init(n_try, alpha=2.0))
        init_names.append("u_shaped_2.0")
        inits.append(u_shaped_init(n_try, alpha=4.0))
        init_names.append("u_shaped_4.0")

        # Bimodal
        inits.append(bimodal_init(n_try, width_frac=0.08))
        init_names.append("bimodal_0.08")
        inits.append(bimodal_init(n_try, width_frac=0.15))
        init_names.append("bimodal_0.15")

        # Edge concentrated
        inits.append(edge_concentrated_init(n_try, decay=1.0))
        init_names.append("edge_1.0")

        # Random
        a_r = rng.uniform(0.1, 1.0, n_try)
        a_r = a_r / np.sum(a_r)
        inits.append(a_r)
        init_names.append("random")

        # Limit number of inits based on time
        max_inits = max(2, min(len(inits), int(remaining / max(alloc_per_init, 10))))
        inits = inits[:max_inits]
        init_names = init_names[:max_inits]

        for idx, (a_init, name) in enumerate(zip(inits, init_names)):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            alloc = min(alloc_per_init, remaining * 0.4)
            if alloc < 10:
                break

            print(f"  [{name}] n={n_try}", end="")

            a_opt = optimize_sequence(a_init, time_budget=alloc, verbose=False)
            obj_opt = compute_obj(a_opt)
            print(f" -> obj={obj_opt:.8f}", end="")

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(" *** NEW BEST ***")
            else:
                print()

    # Final intensive polish on the best sequence
    print(f"\n=== Final Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 120.0)
        a_final = optimize_sequence(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish: {best_score:.8f} ***")
        else:
            print(f"  Final polish: {obj_final:.8f}")

    # One more LP pass if time allows
    remaining = deadline - time.monotonic()
    if remaining > 40 and len(best_seq) <= 2000:
        a_lp = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 60.0), verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: {best_score:.8f} ***")

    # Peak reduction final
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_pr = peak_reduction(best_seq, n_steps=30, time_budget=min(remaining * 0.3, 20.0))
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  Final PeakRed: {best_score:.8f} ***")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
