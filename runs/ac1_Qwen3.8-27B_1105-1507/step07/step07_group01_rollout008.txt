    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = np.sum(v)
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_direction(a, top_k=1):
    """
    Solve LP to find direction g that minimizes the max (or top-k sum) of conv(a,g).
    Returns g (normalized, sum=1) or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # LP: min M s.t. A@g - M <= 0, sum(g)=1, g>=0
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def solve_lp_topk_direction(a, k):
    """
    Solve LP: min sum of top-k peaks of conv(a,g) approximated by
    min t s.t. sum_{all} max(0, conv_k - t) <= 0 (too complex)
    Instead: min M s.t. sum of all conv(a,g)_k / k_eff <= M
    Approximation: minimize the average of top-k by using a threshold variable.
    
    Simpler approach: minimize M s.t. for each k, conv_k <= M + slack_k,
    sum(slack_k) <= k * M_extra ... 
    
    Actually, let's just minimize the sum of all peaks (L1 norm of conv):
    min sum_k conv(a,g)_k s.t. sum(g)=1, g>=0
    This is: min sum_k sum_i sum_j [i+j=k] a[i]*g[j] = min sum_j g[j] * (sum_i a[i] for valid i)
    = min sum_j g[j] * c_j where c_j = sum_{i: 0<=i<n, 0<=i+j<2n-1} a[i]
    Since all a[i] >= 0, c_j >= 0, so we just put all mass on the j with smallest c_j.
    That's too degenerate. Let's use the smooth max instead.
    """
    # Just use the standard LP direction for now
    return solve_lp_direction(a)


def coord_descent_pass(a, max_iters=1):
    """
    One full pass of coordinate descent. For each coordinate j, find optimal value
    via golden section search on [0, upper_bound].
    Works in normalized space (sum=1).
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 0.01:
        return a
    a = a / s

    # Precompute convolution
    conv = compute_conv(a)

    for j in range(n):
        old_val = a[j]
        # Upper bound for x: can't exceed 1 (since sum=1 and all non-negative)
        # But to be safe, use a wider range
        x_hi = max(2.0 * old_val + 1.0 / n, 1.0 / (n * 0.5))
        x_hi = min(x_hi, 1.0)  # Can't exceed 1 since sum=1
        x_lo = 0.0

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        x1 = x_lo + (1.0 - gr) * (x_hi - x_lo)
        x2 = x_lo + gr * (x_hi - x_lo)

        def f(x):
            """Objective with a[j] = x, others scaled to maintain sum=1."""
            if x < -1e-15:
                return np.inf
            x = max(x, 0.0)
            new_sum = 1.0 - old_val + x
            if new_sum < 1e-10:
                return np.inf
            # New conv: conv[k] changes only where a[j] appears
            # conv[k] = sum_i a[i]*a[k-i]
            # Change: for k where 0<=k-j<n and k!=2j: delta = 2*(x/ new_sum - old_val/new_sum) * a[k-j]/new_sum
            # Actually, let's just compute the new conv properly but efficiently
            # new_a[i] = a[i]/new_sum for i!=j, new_a[j] = x/new_sum
            # new_conv[k] = conv[k]/new_sum^2 + correction
            # The correction comes from the fact that a[j] changed
            # Old contribution of a[j] to conv[k]:
            #   if 0<=k-j<n and k!=2j: 2*old_val*a[k-j]
            #   if k==2j: old_val^2
            # New contribution:
            #   if 0<=k-j<n and k!=2j: 2*x*a[k-j]
            #   if k==2j: x^2
            # All divided by new_sum^2, but also other terms change by factor 1/new_sum^2 vs old 1 (since old sum was 1)
            # So: new_conv[k] = (conv[k] - old_contrib[k] + new_contrib[k]) / new_sum^2

            # For efficiency, compute max directly:
            # new_conv[k] = (conv[k] - old_c[k] + new_c[k]) / new_sum^2
            # where old_c[k] and new_c[k] are the contributions involving a[j]

            # Max of new_conv = (max over k of [conv[k] - old_c[k] + new_c[k]]) / new_sum^2
            # old_c[k] = 2*old_val*a[k-j] if (0<=k-j<n and k!=2j), old_val^2 if k==2j, 0 else
            # new_c[k] = 2*x*a[k-j] if (0<=k-j<n and k!=2j), x^2 if k==2j, 0 else

            # So: conv[k] - old_c[k] + new_c[k]
            # For k where 0<=k-j<n and k!=2j: conv[k] + 2*(x-old_val)*a[k-j]
            # For k==2j: conv[2j] + x^2 - old_val^2
            # Else: conv[k]

            delta = x - old_val
            # Build the effective convolution values
            # We need max over all k of:
            #   base[k] + linear_part[k] where
            #   base[k] = conv[k] - old_c[k]
            #   linear_part[k] = new_c[k]
            
            # For efficiency, precompute base values outside the inner loop
            # Actually let's just compute the max directly
            max_val = 0.0
            # k ranges from 0 to 2n-2
            # For k where 0<=k-j<n (i.e., j<=k<=j+n-1) and k != 2j:
            #   val = conv[k] + 2*delta*a[k-j]
            # For k = 2j (if 2j < 2n-1):
            #   val = conv[2j] + x*x - old_val*old_val
            # For other k:
            #   val = conv[k]
            
            # The max over "other k" is just the max of conv[k] for k not in [j, j+n-1] or k==2j
            # Let's compute this as: max(conv[k] for k not affected)
            
            # For speed, just compute the affected part and compare with global max
            gmax = np.max(conv)  # This is an upper bound on the "unaffected" part
            
            # Affected k values: k in [max(0,j), min(2n-2, j+n-1)]
            k_start = max(0, j)
            k_end = min(2 * n - 2, j + n - 1)
            
            for k in range(k_start, k_end + 1):
                if k == 2 * j:
                    val = conv[k] + x * x - old_val * old_val
                else:
                    # 0 <= k-j < n is guaranteed since k in [j, j+n-1]
                    val = conv[k] + 2.0 * delta * a[k - j]
                if val > max_val:
                    max_val = val
            
            if gmax > max_val:
                max_val = gmax
            
            obj = 2.0 * n * max_val / (new_sum * new_sum)
            return obj

        # Golden section
        f1 = f(x1)
        f2 = f(x2)
        for _ in range(40):
            if f1 < f2:
                x_hi = x2
                x2 = x1
                f2 = f1
                x1 = x_lo + (1.0 - gr) * (x_hi - x_lo)
                f1 = f(x1)
            else:
                x_lo = x1
                x1 = x2
                f1 = f2
                x2 = x_lo + gr * (x_hi - x_lo)
                f2 = f(x2)

        best_x = (x_lo + x_hi) / 2.0
        f_best = f(best_x)
        f_current = f(old_val)

        if f_best < f_current - 1e-14:
            a[j] = best_x
            s = np.sum(a)
            if s > 0.01:
                a = a / s
                conv = compute_conv(a)

    return a


def coord_descent_fast(a, n_passes=1):
    """
    Faster coordinate descent using vectorized max computation.
    For each coordinate j, we only need to check the max over affected k values.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 0.01:
        return a
    a = a / s
    conv = compute_conv(a)
    conv_max_global = np.max(conv)

    for _pass in range(n_passes):
        improved = False
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-12:
                continue

            # The affected range of k: [max(0, j), min(2n-2, j+n-1)]
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)

            # Precompute the "base" values: conv[k] minus old contribution
            # and the coefficients for the new contribution
            # For k in affected range:
            #   if k == 2j: new_conv[k] = conv[k] - old^2 + x^2
            #   else: new_conv[k] = conv[k] + 2*(x - old)*a[k-j]
            
            # We want to find x in [0, x_max] minimizing 2n * max(new_conv) / (1-old+x)^2
            # Let t = x (the new value of a[j])
            # new_sum = 1 - old + t
            # max_new_conv = max(
            #   max_{k affected, k!=2j} (conv[k] + 2*(t-old)*a[k-j]),
            #   conv[2j] + t^2 - old^2  (if 2j in affected range)
            #   max_{k not affected} conv[k]
            # )
            
            # max not affected:
            # conv values at k not in [k_lo, k_hi]
            if k_lo > 0:
                unaffected_left = np.max(conv[:k_lo])
            else:
                unaffected_left = 0.0
            if k_hi < 2 * n - 2:
                unaffected_right = np.max(conv[k_hi + 1:])
            else:
                unaffected_right = 0.0
            max_unaffected = max(unaffected_left, unaffected_right, 0.0)

            # Affected part (excluding k=2j):
            # For k in [k_lo, k_hi], k != 2j:
            #   val[k] = conv[k] + 2*(t-old)*a[k-j]
            # This is a linear function of t with slope 2*a[k-j]
            # The max over these is the upper envelope of lines

            # For k=2j (if in range):
            #   val = conv[2j] + t^2 - old^2

            # We'll do a small grid search since golden section is slow in Python
            # Grid: 0, then logarithmically spaced, then fine near current best
            
            # Upper bound for t: since sum=1, t <= 1
            # But practically, t shouldn't be too different from old_val
            t_max = min(1.0, max(4.0 * old_val, 2.0 / n))
            
            # Quick grid: 30 points
            t_grid = np.linspace(0.0, t_max, 30)
            best_t = old_val
            best_f = compute_obj_local(j, old_val, a, conv, max_unaffected, n)
            
            for t in t_grid:
                f_val = compute_obj_local(j, t, a, conv, max_unaffected, n)
                if f_val < best_f:
                    best_f = f_val
                    best_t = t
            
            # Refine with 20 points around best_t
            if best_t > 0:
                margin = t_max / 30.0
                t_fine = np.linspace(max(0, best_t - 2*margin), min(t_max, best_t + 2*margin), 20)
                for t in t_fine:
                    f_val = compute_obj_local(j, t, a, conv, max_unaffected, n)
                    if f_val < best_f:
                        best_f = f_val
                        best_t = t

            if best_f < best_f - 1e-14:  # This is always false; check differently
                pass
            if abs(best_t - old_val) > 1e-8:
                a[j] = best_t
                s = np.sum(a)
                if s > 0.01:
                    a = a / s
                    conv = compute_conv(a)
                improved = True

    return a


def compute_obj_local(j, t, a, conv, max_unaffected, n):
    """Compute objective when a[j] is set to t (others unchanged, then renormalize)."""
    old_val = a[j]
    if t < 0:
        t = 0.0
    new_sum = 1.0 - old_val + t
    if new_sum < 1e-10:
        return np.inf

    # Max of new convolution
    k_lo = max(0, j)
    k_hi = min(2 * n - 2, j + n - 1)
    
    max_val = max_unaffected
    
    # Affected k values
    delta = t - old_val
    a_slice = a  # We need a[k-j] for k in [k_lo, k_hi]
    # a[k-j] where k-j ranges from max(0, k_lo-j) to min(n-1, k_hi-j)
    idx_lo = max(0, k_lo - j)
    idx_hi = min(n - 1, k_hi - j)
    
    # Vectorized computation
    ks = np.arange(k_lo, k_hi + 1)
    idxs = ks - j  # = a[k-j] index
    linear_vals = conv[k_lo:k_hi + 1] + 2.0 * delta * a[idxs]
    
    # Handle k = 2j case
    if k_lo <= 2 * j <= k_hi:
        pos = 2 * j - k_lo
        linear_vals[pos] = conv[2 * j] + t * t - old_val * old_val
    
    max_affected = np.max(linear_vals)
    max_val = max(max_val, max_affected)
    
    return 2.0 * n * max_val / (new_sum * new_sum)


def smooth_max_step(a, T, lr, momentum_v, momentum=0.9):
    """One step of gradient descent using smooth max (log-sum-exp)."""
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    
    # Gradient: grad[i] = sum_j weights[i+j] * a[j] (correlation)
    grad = np.correlate(weights, a, mode='valid')
    grad_centered = grad - np.mean(grad)
    
    momentum_v = momentum * momentum_v - lr * grad_centered
    a_new = simplex_project(a + momentum_v)
    return a_new, momentum_v


def lp_alternating_cd(a, time_budget, use_cd=True, cd_passes=2):
    """
    Main iteration: alternate between LP direction and coordinate descent.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.4
    T = 0.001
    lr = 0.01
    momentum_v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        # Step 1: LP direction
        g_lp = solve_lp_direction(a)
        if g_lp is not None:
            # Line search for blending
            best_local = np.inf
            best_a_local = best_a.copy()
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = np.maximum((1.0 - t) * a + t * g_lp, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_a_local = a_try.copy()

            if best_local < best_obj - 1e-12:
                best_obj = best_local
                best_a = best_a_local.copy()
                a = best_a.copy()
                stall = 0
            else:
                stall += 1

        # Step 2: Coordinate descent
        if use_cd and time.monotonic() - start < time_budget - 1.0:
            a_cd = coord_descent_fast(a, n_passes=cd_passes)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_obj - 1e-12:
                best_obj = obj_cd
                best_a = a_cd.copy()
                a = a_cd.copy()
                stall = 0

        # Step 3: Smooth max gradient (every 3 iterations)
        if stall % 3 == 2 and time.monotonic() - start < time_budget - 0.5:
            a_sm, momentum_v = smooth_max_step(a, T, lr, momentum_v)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_obj - 1e-12:
                best_obj = obj_sm
                best_a = a_sm.copy()
                a = a_sm.copy()
            else:
                lr *= 0.8

        # Stall handling
        if stall > 20:
            # Perturb
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            scale = 0.02 + 0.01 * min(stall // 20, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0
            lr = 0.01
            momentum_v = np.zeros(n)

        # Temperature anneal
        T = max(0.00005, T * 0.995)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif mode == 3:
        # Bimodal
        center = n // 2
        width = max(1, n // 6)
        x = np.arange(n)
        a = np.exp(-((x - center * 0.3) ** 2) / (2 * width ** 2))
        a += np.exp(-((x - center * 1.4) ** 2) / (2 * width ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.5
        a = a / np.sum(a)
    elif mode == 5:
        # Triangle
        x = np.arange(n) / max(n - 1, 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    elif mode == 6:
        # High edges, low center (to avoid center peak)
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (2 * x - 1) ** 2
        a = a / np.sum(a)
    elif mode == 7:
        # Two blocks
        a = np.zeros(n)
        a[:n//3] = 1.0
        a[2*n//3:] = 1.0
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization with LP+CD alternating, multi-n exploration,
    diverse initializations, and final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={best_score:.8f}")
    except Exception:
        parent = np.ones(800) / 800
        best_seq = parent.copy()
        best_score = compute_obj(best_seq)
        print(f"Default init (n={len(best_seq)}): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Intensive optimization on parent n ===
    print(f"\n=== Phase 1: LP+CD on n={n_parent} ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(200.0, remaining * 0.3)
        a_opt = lp_alternating_cd(best_seq, time_budget=alloc, use_cd=True, cd_passes=2)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 1: obj={best_score:.8f}")
        else:
            print(f"  Phase 1: obj={obj:.8f} (no improvement)")

    # === Phase 2: More CD passes with different parameters ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(100.0, remaining * 0.2)
        a_opt2 = lp_alternating_cd(best_seq, time_budget=alloc, use_cd=True, cd_passes=5)
        obj2 = compute_obj(a_opt2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a_opt2.copy()
            print(f"  Phase 2 (CD-heavy): obj={best_score:.8f}")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        500, 600, 700, 800, 900, 1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Optimize
        alloc = min(remaining * 0.12, 50.0)
        a_opt_n = lp_alternating_cd(a_try, time_budget=alloc, use_cd=True, cd_passes=2)
        obj_n = compute_obj(a_opt_n)

        if obj_n < best_score:
            best_score = obj_n
            best_seq = a_opt_n.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_n:.8f}")
            n_best = len(best_seq)

    # === Phase 4: Diverse random initializations ===
    print(f"\n=== Phase 4: Diverse Random Restarts ===")
    n_best = len(best_seq)

    for r in range(20):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        init_mode = r % 8
        n_r = int(n_best + rng.integers(-50, 51))
        n_r = max(300, min(1500, n_r))
        a_r = random_init(n_r, rng, mode=init_mode)

        alloc_r = min(remaining * 0.08, 40.0)
        a_opt_r = lp_alternating_cd(a_r, time_budget=alloc_r, use_cd=True, cd_passes=2)
        obj_r = compute_obj(a_opt_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt_r.copy()
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            if r % 5 == 0:
                print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # Multiple rounds of LP+CD
    for polish_round in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.2, 60.0)
        a_p = lp_alternating_cd(best_seq, time_budget=alloc, use_cd=True, cd_passes=3)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Polish {polish_round}: obj={best_score:.8f}")
        else:
            print(f"  Polish {polish_round}: obj={obj_p:.8f} (stalled)")
            break

    # Final coordinate descent only (very fine)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd_final = coord_descent_fast(best_seq, n_passes=3)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD: obj={best_score:.8f}")

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(200) / 200

    # Clip to valid range
    best_seq = np.clip(best_seq, 0, 1000.0)
    s_final = np.sum(best_seq)
    if s_final < 0.01:
        best_seq = np.ones(100) / 100

    result = [float(x) for x in best_seq]
    
    # Validate with actual evaluator
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
