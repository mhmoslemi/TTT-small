Improved optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with adaptive step size, MWU warmup, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For efficiency, use the structure: A is a banded Toeplitz-like matrix
    # Actually it's more of a "convolution matrix"
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        # But k max is 2n-2, so k in [j, min(j+n-1, 2n-2)]
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    # Try solvers in order of preference
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    return None, None

def mwu_optimize(a, time_budget=60.0, top_k=10, eta_init=0.05):
    """
    Multiplicative Weight Update optimization.

    a = np.maximum(a, 0).astype(np.float64).copy()

    if s < 1e-10:

    a = a / s

    best_obj = compute_obj(a)

    no_improve = 0

        conv = compute_conv(a)
        max_b = float(np.max(conv))

            k_indices = np.arange(len(conv))
        else:
            k_indices = np.argsort(conv)[::-1][:top_k]
        
        # Compute subgradient: grad[j] = sum_{k in K} a[k-j] (with bounds)
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            # conv[k] = sum_j a_j * a[k-j] where 0<=k-j<n
            # d conv[k] / d a_j = a[k-j] for j in [max(0,k-n+1), min(n-1, k)]
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            # a[k-j] for j in [j_start, j_end): indices k-j_start to k-(j_end-1)
            # = a[k-j_start : k-j_end+1+1] = a[k-j_start : k-j_end+2]...

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def compute_subgrad(a, k_indices):
    """Compute subgradient of max conv at positions k_indices w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in k_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_end > j_start:
            # grad[j] += a[k - j] for j in [j_start, j_end)
            idx = k - np.arange(j_start, j_end)
            grad[j_start:j_end] += a[idx]
    return grad / len(k_indices)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)<=M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    return None, None

    while time.monotonic() - start < time_budget:
        # Step 1: LP with current a
        g1, M1 = solve_lp(a)
        if g1 is None:
            break

        # Step 2: LP with g1 to get refined direction
        g2, M2 = solve_lp(g1)
        if g2 is not None:
            # Use g2 as direction (more accurate)
            direction = g2
        else:
            direction = g1

        # Adaptive line search
        best_t = 0.3
        best_obj_t = best_obj
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        # Add neighborhood of current good t
        if 0 < t_current < 1:
            for dt in [0.5, 1.0, 2.0]:
                t_new = t_current * dt
                if 0.01 < t_new <= 1.0:
                    t_candidates.append(round(t_new, 3))
        t_candidates = sorted(set(t for t in t_candidates if 0.01 <= t <= 1.0))

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try /= st
            obj_t = compute_obj(a_try)
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t

        if best_obj_t < best_obj - 1e-14:
            a_try = (1.0 - best_t) * a + best_t * direction
            a_try = np.maximum(a_try, 0)
            a_try /= np.sum(a_try)
            best_obj = best_obj_t
            best_a = a_try.copy()
            a = a_try
            t_current = best_t
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall % 8 == 0:
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                scale = min(0.03 + 0.005 * stall, 0.15)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a /= a.sum()
                best_obj = compute_obj(a)
                best_a = a.copy()
            if stall > 40:
                break

    return best_a


def spectral_reinit(a, rng):
    """Create a new initialization by modifying the Fourier magnitude."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    mag = np.abs(fa)
    # Flatten peaks in magnitude spectrum
    mag_thresh = np.percentile(mag, 80)
    mag[np.where(mag > mag_thresh)[0]] = mag_thresh + 0.1 * (mag - mag_thresh)[np.where(mag > mag_thresh)[0]]
    phase = np.angle(fa)
    fa_new = mag * np.exp(1j * phase)
    # Add small random perturbation to phase
    fa_new *= np.exp(1j * rng.normal(0, 0.1, size // 2 + 1))
    a_new = irfft(fa_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n == 1 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new /= s
    return a_new


def coordinate_polish(a, time_budget=20.0):
    """Quick coordinate descent: swap mass between pairs to reduce max conv."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a /= s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for _ in range(500):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        # Find the pair (i, j) that most contributes to conv[k_star]
        contributions = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        for j in range(j_start, j_end):
            contributions[j] = a[j] * a[k_star - j]
        # Try moving mass from high contributor to low contributor
        i_src = int(np.argmax(contributions))
        # Move to a position that doesn't contribute to k_star
        candidates = [j for j in range(n) if j < j_start or j >= j_end]
        if not candidates:
            candidates = [(i_src + n // 2) % n]
        i_dst = candidates[rng.integers(0, len(candidates))]
        for frac in [0.02, 0.05, 0.1]:
            delta = frac * a[i_src]
            if delta >= a[i_src]:
                continue
            a[i_src] -= delta
            a[i_dst] += delta
            a = np.maximum(a, 0)
            a /= a.sum()
            obj = compute_obj(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                break
            else:
                a[i_src] += delta
                a[i_dst] -= delta
                a /= a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses adaptive LP alternating with two-step direction,
    MWU with threshold-based peaks, spectral reinitialization, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if len(parent) > 0 and parent.sum() > 0.01:
            parent = parent / parent.sum()
            obj = compute_obj(parent)
            best_score = obj
            best_seq = parent.copy()
            print(f"Parent (n={len(parent)}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default (n=800): obj={best_score:.8f}")

    a = best_seq.copy()
    n = len(a)

    # === Phase 1: MWU warmup with threshold peaks ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 1: MWU Warmup ===")
        budget = min(50.0, remaining * 0.08)
        for thresh in [0.95, 0.9, 0.85]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_new = mwu_threshold(a, time_budget=budget / 3.0, threshold_pct=thresh, eta=0.03)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  MWU thresh={thresh}: obj={best_score:.8f} **")

    # === Phase 2: Main LP alternating with two-step direction ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 2: LP Alternating (Two-Step) ===")
        # Run multiple rounds with decreasing perturbation
        for round_idx in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(120.0, remaining * 0.2)
            a_new = lp_alternating_improved(a, time_budget=alloc, verbose=True)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  LP Round {round_idx}: obj={best_score:.8f} **")
            else:
                print(f"  LP Round {round_idx}: obj={obj:.8f} (stalled)")
                # Perturb to escape
                scale = 0.02 + 0.01 * round_idx
                a = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                a = np.maximum(a, 0)
                a /= a.sum()

    # === Phase 3: Spectral reinitialization + reoptimize ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 3: Spectral Reinit ===")
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_spec = spectral_reinit(best_seq, rng)
            # Quick optimize
            a_opt = mwu_threshold(a_spec, time_budget=min(15.0, remaining * 0.1), threshold_pct=0.9, eta=0.03)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_opt = lp_alternating_improved(a_opt, time_budget=min(40.0, remaining * 0.25), verbose=False)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  Spectral {attempt}: obj={best_score:.8f} **")
            else:
                print(f"  Spectral {attempt}: obj={obj:.8f}")

    # === Phase 4: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 4: Multi-n Exploration ===")
        n_parent = len(best_seq)
        n_options = sorted(set([
            max(200, n_parent - 200), max(200, n_parent - 100),
            n_parent,
            n_parent + 100, n_parent + 200, n_parent + 400,
        ]))
        n_options = [x for x in n_options if 200 <= x <= 2500]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)

            # MWU then LP
            a_opt = mwu_threshold(a_try, time_budget=min(20.0, remaining * 0.15), threshold_pct=0.9, eta=0.03)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_opt = lp_alternating_improved(a_opt, time_budget=min(50.0, remaining * 0.3), verbose=False)

            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 5: Extended LP on best ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 5: Extended LP Polish ===")
        alloc = min(remaining * 0.4, 150.0)
        a_new = lp_alternating_improved(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Extended LP: obj={best_score:.8f} **")

    # === Phase 6: Coordinate polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 6: Coordinate Polish ===")
        a_pol = coordinate_polish(a, time_budget=min(20.0, remaining * 0.3))
        obj = compute_obj(a_pol)
        if obj < best_score:
            best_score = obj
            best_seq = a_pol.copy()
            print(f"  CD: obj={best_score:.8f} **")

    # === Phase 7: Final LP + coordinate ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 7: Final ===")
        alloc = min(remaining * 0.4, 60.0)
        a_new = lp_alternating_improved(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_pol = coordinate_polish(best_seq, time_budget=min(10.0, remaining * 0.3))
            obj = compute_obj(a_pol)
            if obj < best_score:
                best_score = obj
                best_seq = a_pol.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
