def compute_conv_fftw(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fftw(a_norm)
    return 2.0 * n * np.max(conv)

def solve_lp(a, a_norm):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    # Build constraint matrix A where (A@g)[k] = conv(a_norm, g)[k]
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    # LP: min M
    # Variables: g[0..n-1], M
    # Constraints: A@g - M <= 0 (2n-1 constraints)
    #              sum(g) = 1
    #              g >= 0, M >= 0 (M >= 0 is implicit since A@g >= 0)
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, M free (but will be >= 0)
    # Actually M should be >= 0 since all convex combinations of non-negative things
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})

        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None

        g = solve_lp(a, a_norm)
        if g is None:

        best_t_val = t
        for t_try in [t * 0.5, t * 0.75, t, t * 1.25, t * 1.5, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with adaptive golden-section line search,
log-domain subgradient with Nesterov momentum, FFT flat spectrum,
and multi-n exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))

def build_lp_matrices(n, a_norm):
    """Build LP constraint matrices for the convolution LP."""
    # A is (2n-1) x n where (A@g)[k] = conv(a_norm, g)[k]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A

def solve_lp(a_norm, A=None):
    """Solve the LP and return optimal g."""
    n = len(a_norm)
    if A is None:
        A = build_lp_matrices(n, a_norm)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None

def golden_section_line_search(a, g, best_obj, a0=None):
    """Golden section search for optimal mixing parameter t."""
    if a0 is None:
        a0 = a
    
    def obj_at_t(t):
        a_try = (1 - t) * a0 + t * g
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)
    
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    lo, hi = 0.01, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(40):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)
    
    t_best = (lo + hi) / 2
    obj_best = obj_at_t(t_best)
    
    # Also check endpoints
    for t_check in [0.01, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0]:
        obj_check = obj_at_t(t_check)
        if obj_check < obj_best:
            obj_best = obj_check
            t_best = t_check
    
    return t_best, obj_best

def lp_alternating(a, time_budget=120.0, verbose=False):
    """LP alternating with golden section line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        a_norm = a / np.sum(a)
        A = build_lp_matrices(n, a_norm)
        g = solve_lp(a_norm, A)
        if g is None:
            break
        
        t_best, obj_best = golden_section_line_search(a, g, best_obj, a0=a)
        
        if obj_best < best_obj - 1e-13:
            a_new = (1 - t_best) * a + t_best * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = obj_best
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and stall % 5 == 0:
                print(f"    LP: obj={best_obj:.8f}, t={t_best:.4f}")
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(hash((n, stall)) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03 + 0.01 * (stall // 15), n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
    
    return best_a

def log_domain_subgradient(a, time_budget=60.0, n_iters=5000):
    """Subgradient method in log-domain with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    s = np.sum(a)
    a = a / s
    x = np.log(a)  # log-domain parameter
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = x.copy()
    v = np.zeros(n)
    beta = 0.9
    m_peaks = 3
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        a_y = np.exp(y - np.max(y))  # shift for numerical stability
        a_y = a_y / np.sum(a_y)
        s_y = np.sum(a_y)
        
        conv_y = compute_conv(a_y)
        top_idx = np.argsort(conv_y)[-m_peaks:]
        top_val = conv_y[top_idx]
        max_val = top_val[-1]
        
        temp = max(0.001, (max_val - top_val[0]) * 0.3)
        weights = np.exp((top_val - max_val) / temp)
        weights /= np.sum(weights)
        
        # Subgradient in log-domain
        # d(C_k/S^2)/dx_j = 2*a_j*(a_{k-j}*S - C_k)/S^3 (when valid)
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            Ck = conv_y[ki]
            for j in range(n):
                kj = ki - j
                if 0 <= kj < n:
                    grad[j] += w * 2 * a_y[j] * (a_y[kj] * s_y - Ck) / (s_y ** 3)
        
        # Nesterov step
        alpha = 2.0 / ((it + 10) * (np.linalg.norm(grad) + 1e-10))
        alpha = min(alpha, 0.1)
        
        v = beta * v + (1 - beta) * grad
        y_new = y - alpha * v
        
        a_new = np.exp(y_new - np.max(y_new))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        y = y_new
    
    return best_a

def fft_flat_spectrum(a, time_budget=30.0, relaxation=0.5, n_iters=300):
    """FFT-based flat spectrum iteration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = np.fft.fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe
        
        A_flat = phase
        A_new = (1 - relaxation) * A + relaxation * A_flat
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    rng = np.random.default_rng(it * 7919)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def random_init(n, rng):
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        freqs = rng.uniform(0, 5, 10)
        phases = rng.uniform(0, 2 * np.pi, 10)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # U-shape
        x = np.arange(n)
        center = n / 2
        a = np.ones(n) + 0.5 * ((x - center) / center) ** 2
        a = a / np.sum(a)
    elif mode == 4:
        # Two peaks
        centers = [n // 4, 3 * n // 4]
        w = n // 15
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    else:
        # Cosine
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        a = a / np.sum(a)
    return a
