Hybrid optimizer using Lp-norm smoothing, Nesterov accelerated subgradient,
spectral initialization, and multi-n exploration to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """

    f = (sum(c^p))^(1/p) / (sum(a))^2
    """
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    
    conv = compute_conv(a)
    
    # Lp norm
    cp_p = np.power(conv, p)  # conv >= 0
    sum_cp = np.sum(cp_p)
    lp_val = sum_cp ** (1.0 / p)
    
    F = lp_val / (s * s)
    
    # Gradient
    # dF/dx_i = [sum_k c_k^(p-1) * 2*a_i*a_{k-i} / sum_cp^((p-1)/p)] / s^2
    #          - 2*F * a_i / s
    
    c_p_minus_1 = np.power(conv, p - 1)  # c_k^(p-1)
    factor = 1.0 / np.power(sum_cp, (p - 1.0) / p)
    
    # Cross-correlation: sum_k c_k^(p-1) * a_{k-i} for each i
    # This is conv(c_p_minus_1, a_flip)[n-1 : n-1+n]
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fc = rfft(c_p_minus_1, size)
    faf = rfft(a_flip, size)
    corr = irfft(fc * faf, size)
    corr = corr[n - 1:n - 1 + n]
    
    grad = (2.0 * a * corr * factor) / (s * s) - 2.0 * F * a / s
    return F, grad

def lpsmooth_optimize(a_init, p_schedule, time_budget, verbose=False):
    """Optimize using Lp-norm smoothed objective with L-BFGS-B."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    x = np.log(a)
    x = x - x.max()
    
    for p in p_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        def fun(xv):
            xv = xv - xv.max()
            val, _ = lp_obj_and_grad(xv, p)
            return val
        
        def fgrad(xv):
            xv = xv - xv.max()
            _, g = lp_obj_and_grad(xv, p)
            return g
        
        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10,
                                   'maxls': 50, 'maxfun': 2000})
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    Lp p={p}: obj={best_obj:.8f}")
        except Exception as e:
            if verbose:
                print(f"    Lp p={p} failed: {e}")
    
    return best_a

def nesterov_subgradient(a_init, time_budget, verbose=False):
    """Nesterov accelerated projected subgradient for max(conv(a,a)) with sum(a)=1, a>=0."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    a_prev = a.copy()
    a_curr = a.copy()
    t = 1
    eta = 1.0  # initial step size
    
    while time.monotonic() - start < time_budget:
        # Nesterov momentum
        theta = (t - 1) / (t + 2.0)
        a_y = a_curr + theta * (a_curr - a_prev)
        a_y = np.maximum(a_y, 0)
        sy = float(np.sum(a_y))
        if sy > 1e-10:
            a_y = a_y / sy
        
        # Compute subgradient
        conv = compute_conv(a_y)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max(c) = c[k_star]: dc[k_star]/da[i] = 2*a[k_star-i] (valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a_y[j]
        
        # Adaptive step size
        L = 2.0 * float(np.linalg.norm(a_y, 2))
        eta_t = eta / (1.0 + L * t * 0.01)
        
        a_new = a_y - eta_t * grad
        # Project: a >= 0, sum(a) = 1
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and t % 100 == 0:
                print(f"    Nesterov t={t}: obj={best_obj:.8f}")
        
        a_prev = a_curr
        a_curr = a_new
        t += 1
        eta = max(eta * 0.999, 0.01)
    
    return best_a

def spectral_init(n, mode=0):
    """Generate spectral initializations."""
    if mode == 0:
        # Raised cosine
        x = np.arange(n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    elif mode == 1:
        # Hamming window
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x / n)
    elif mode == 2:
        # Flat-top window (normalized)
        a = np.ones(n)
        a[:n//10] *= 0.1  # taper edges
        a[-n//10:] *= 0.1
    elif mode == 3:
        # Sinc-like (truncated)
        x = np.arange(n) - n // 2
        a = np.sinc(x / (n * 0.3))
        a = np.abs(a)
    elif mode == 4:
        # Two-lobe
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x / n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a

def gs_spectral_refine(a, n_iters=200, time_budget=30.0):
    """
    Gerchberg-Saxton style: alternate between spectral modification and non-negativity projection.
    Goal: reduce the peak of conv(a,a) by modifying the spectrum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    L = next_pow2(2 * n - 1)
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = rfft(a, L)
        # A[0] is DC = sum(a) = 1
        # We want to reduce the "sidelobes" of |A|^2 (which is the spectrum of conv)
        # Strategy: apply a soft threshold to |A|
        abs_A = np.abs(A)
        # Keep DC, attenuate rest
        if it < 50:
            # Gentle attenuation
            mask = abs_A.copy()
            mask[1:] = mask[1:] * 0.95
        elif it < 150:
            mask = abs_A.copy()
            mask[1:] = mask[1:] * 0.9
        else:
            mask = abs_A.copy()
            mask[1:] = mask[1:] * 0.85
        
        # Reconstruct with modified magnitude
        phase = np.angle(A)
        A_new = mask * np.exp(1j * phase)
        a_candidate = irfft(A_new, L)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_candidate, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
