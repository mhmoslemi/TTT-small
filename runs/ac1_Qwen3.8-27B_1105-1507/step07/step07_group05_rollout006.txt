import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2*n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    p = a / s
    b = compute_conv(p)
    return 2.0 * n * float(np.max(b))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a, Active_set=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0"""
    n = len(a)
    a_norm = a / np.sum(a)
    
    if Active_set is not None:
        # Use only active constraints
        A = np.zeros((len(Active_set), n))
        for idx, k in enumerate(Active_set):
            for j in range(n):
                if 0 <= k-j < n:
                    A[idx, j] = a_norm[k-j]
    else:
        A = build_conv_matrix(a_norm)
    
    m = A.shape[0]
    # Variables: g_0, ..., g_{n-1}, M
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None

    n = len(a)

        return a
    a = a / s

    stall = 0

        # Find active set: constraints where conv(a,a) is near the max

        threshold = max_b * 0.95

        g_lp = solve_lp(a, active)

        # Line search for step size

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    p = a / s
    b = compute_conv(p)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = 0
        end = min(n, 2 * n - 1 - j)
        A[j:j + end, j] = a[:end]
    return A


def build_partial_conv_matrix(a, active_indices):
    """Build partial conv matrix for active constraints only."""
    n = len(a)
    A = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        for j in range(n):
            row = k - j
            if 0 <= row < n:
                A[idx, j] = a[row]
    return A


def solve_lp_given_a(a, active_set=None, max_constraints=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s

    if active_set is None:
        if max_constraints is not None and max_constraints < 2 * n - 1:
            b = compute_conv(a_norm)
            idx = np.argsort(b)[::-1][:max_constraints]
            A = build_partial_conv_matrix(a_norm, list(idx))
        else:
            A = build_conv_matrix(a_norm)
    else:
        A = build_partial_conv_matrix(a_norm, active_set)

    m = A.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            sg = float(np.sum(g))

def multiplicative_weights(a, time_budget=60.0, verbose=False):
    """
    Multiplicative weights / mirror descent update.
    a_j <- a_j * exp(-eta * a_{k*-j}) / Z
    This naturally maintains non-negativity and can escape some local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.5
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    # Track average subgradient for smoothing
    avg_grad = np.zeros(n)
    alpha_smooth = 0.1

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))

        # Subgradient: g_j = a[k_star - j]
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = a[idx]

        # Smooth the gradient
        avg_grad = (1 - alpha_smooth) * avg_grad + alpha_smooth * grad

        # Multiplicative update
        log_update = -eta * avg_grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -5, 5)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 3.0)
            alpha_smooth = min(alpha_smooth + 0.005, 0.5)
        else:
            eta *= 0.97
            alpha_smooth = max(alpha_smooth - 0.002, 0.05)
            if eta < 0.01:
                break

    return best_a


def subgradient_averaging(a, time_budget=60.0, verbose=False):
    """
    Subgradient descent with polyak step size and gradient averaging.
    Uses multiple peak locations for a more stable descent direction.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    # Track best objective for Polyak step size
    obj_oracle = best_obj
    t = 1.0
    avg_grad = np.zeros(n)
    beta = 0.3

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Use top m peaks for averaged subgradient
        m_peaks = min(5, 2 * n - 1)
        peak_indices = np.argsort(b)[::-1][:m_peaks]
        peak_weights = b[peak_indices] - np.min(b[peak_indices]) + 1e-10
        peak_weights = peak_weights / np.sum(peak_weights)

        # Compute averaged subgradient
        grad = np.zeros(n)
        for w, k in zip(peak_weights, peak_indices):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += w * a[idx]

        # Smooth
        avg_grad = (1 - beta) * avg_grad + beta * grad
        # Center to maintain sum constraint
        avg_grad = avg_grad - np.mean(avg_grad)

        # Polyak-like step size
        t += 1
        step = 0.5 / (t ** 0.4)

        a_new = a - step * avg_grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            step *= 0.5
            continue
        a_new = a_new / s_new

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            obj_oracle = min(obj_oracle, obj_new)
            # Increase step if making good progress
            step = min(step * 1.01, 0.2)
        else:
            step *= 0.98
            if step < 1e-6:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        d = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
        a = 1.0 / d ** 0.3
        a = a / np.sum(a)
    elif mode == 6:
        # Sine-modulated
        x = np.arange(n)
        a = 0.5 + 0.3 * np.sin(2 * np.pi * x / n * 3) + 0.2 * np.cos(2 * np.pi * x / n * 7)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    else:
        # Smooth random (low-pass filtered noise)
        a = rng.uniform(0, 1, n)
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: adaptive LP-alternating + multiplicative weights + subgradient
    + multi-n exploration + diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0  # 25s safety buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01 and np.isfinite(parent_arr).all():
            parent = parent_arr / s
            best_score = compute_obj(parent)
            best_seq = parent.copy()
            print(f"Parent loaded (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting optimization. Target < 1.5027")

    # === Phase 1: LP Alternating from parent (most important) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 1: LP Alternating (n={n_parent}), budget={min(remaining*0.35, 350):.0f}s ===")
    alloc = min(remaining * 0.35, 350.0)
    a_lp = lp_alternating_fast(best_seq, time_budget=alloc, verbose=True)
    obj = compute_obj(a_lp)
    if obj < best_score:
        best_score = obj
        best_seq = a_lp
        print(f"  ** LP1: obj={best_score:.8f} **")
    else:
        print(f"  LP1: obj={obj:.8f} (no improvement)")

    # === Phase 2: Multiplicative Weights ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: Multiplicative Weights, budget={min(remaining*0.15, 100):.0f}s ===")
        alloc = min(remaining * 0.15, 100.0)
        a_mw = multiplicative_weights(best_seq, time_budget=alloc)
        obj = compute_obj(a_mw)
        if obj < best_score:
            best_score = obj
            best_seq = a_mw
            print(f"  ** MW: obj={best_score:.8f} **")
        else:
            print(f"  MW: obj={obj:.8f}")

    # === Phase 3: LP Round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: LP Round 2, budget={min(remaining*0.25, 250):.0f}s ===")
        alloc = min(remaining * 0.25, 250.0)
        a_lp2 = lp_alternating_fast(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp2)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2
            print(f"  ** LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj:.8f}")

    # === Phase 4: Subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 4: Subgradient, budget={min(remaining*0.1, 60):.0f}s ===")
        alloc = min(remaining * 0.1, 60.0)
        a_sg = subgradient_averaging(best_seq, time_budget=alloc)
        obj = compute_obj(a_sg)
        if obj < best_score:
            best_score = obj
            best_seq = a_sg
            print(f"  ** SG: obj={best_score:.8f} **")
        else:
            print(f"  SG: obj={obj:.8f}")

    # === Phase 5: LP Round 3 (polish) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: LP Polish, budget={min(remaining*0.2, 150):.0f}s ===")
        alloc = min(remaining * 0.2, 150.0)
        a_lp3 = lp_alternating_fast(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp3)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp3
            print(f"  ** LP3: obj={best_score:.8f} **")
        else:
            print(f"  LP3: obj={obj:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr + 100, n_curr + 200, n_curr + 400,
        1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 600 <= n <= 2500 and abs(n - n_curr) > 50]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Quick LP optimization on new n
        alloc = min(remaining * 0.2, 80.0)
        a_opt = lp_alternating_fast(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    n_final = len(best_seq)
    n_restarts = max(2, int(max(0, deadline - time.monotonic()) / 50.0))
    n_restarts = min(n_restarts, 8)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_r = diverse_init(n_final, rng)
        alloc = min(remaining * 0.2, 50.0)
        a_opt = lp_alternating_fast(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt
            print(f"  ** Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final LP Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.6, 200.0)
        a_final = lp_alternating_fast(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final
            print(f"  ** Final LP: obj={best_score:.8f} **")

        # One more MW pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_mw2 = multiplicative_weights(best_seq, time_budget=min(remaining * 0.3, 30.0))
            obj = compute_obj(a_mw2)
            if obj < best_score:
                best_score = obj
                best_seq = a_mw2
                print(f"  ** Final MW: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
