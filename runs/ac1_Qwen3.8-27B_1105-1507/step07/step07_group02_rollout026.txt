```python
def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (soft-max approximation of max)
2. LP alternating optimization
3. Multi-start with structured initialization
4. Multi-n exploration

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_gradient_step(a, beta):
    """Compute one step of smoothed gradient descent."""
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights
    shifted = beta * (conv - np.max(conv))
    p = np.exp(shifted)
    p = p / p.sum()
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j]
    # = 2 * conv(p, a[::-1])[n-1+j]
    size = 1
    while size < 2 * n:
        size <<= 1
    P = rfft(p, size)
    Af = rfft(a[::-1], size)
    conv_pa = irfft(P * Af, size)
    xcorr = conv_pa[n - 1: n - 1 + n]
    grad = 2.0 * xcorr
    return grad, conv, p

def smooth_gradient_descent(a, time_budget=60.0, beta_start=10.0, beta_end=1000.0, lr_start=0.1, verbose=False):
    """Smoothed gradient descent with annealing beta."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_start
    it = 0
    total_iters = 5000
    
    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal beta
        progress = min(it / (total_iters * 0.8), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad, conv, p = smooth_gradient_step(a, beta)
        
        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)
        
        # Take step
        a_new = a - lr * grad_centered
        a_new = proj_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.9
            if lr < 1e-6:
                lr = lr_start * 0.5
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))

def structured_init(n, rng):
    """Generate structured initial sequences."""
    mode = rng.integers(0, 6)
    x = np.arange(n)
    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    elif mode == 2:
        # Gaussian window
        sigma = n * rng.uniform(0.15, 0.35)
        center = n / 2
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif mode == 3:
        # Two Gaussians
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.15)
        a = np.exp(-((x - c1)**2) / (2*w**2)) + np.exp(-((x - c2)**2) / (2*w**2))
    elif mode == 4:
        # Cosine power window
        p = rng.uniform(0.5, 3.0)
        a = np.cos(np.pi * x / n) ** (2 * p) + 1e-6
    else:
        # Triangular
        a = 1 - np.abs(2 * x / n - 1)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (soft-max approximation of max with annealing)
2. LP alternating optimization (fix one variable, solve LP for the other)
3. Multi-start with structured initialization (cosine, Gaussian, bimodal windows)
4. Multi-n exploration and final polishing
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient_step(a, beta):
    """Compute gradient of the softened max-convolution objective."""
    n = len(a)
    conv = compute_conv(a)
    shifted = beta * (conv - np.max(conv))
    p = np.exp(shifted)
    p = p / p.sum()
    # grad[j] = 2 * sum_k p[k] * a[k-j] = 2 * conv(p, a[::-1])[n-1+j]
    size = 1
    while size < 2 * n:
        size <<= 1
    P = rfft(p, size)
    Af = rfft(a[::-1], size)
    conv_pa = irfft(P * Af, size)
    xcorr = conv_pa[n - 1: n - 1 + n]
    grad = 2.0 * xcorr
    return grad, conv, p

                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                    else:
                        a = best_a.copy()
                stall = 0

    return best_a


def structured_init(n, rng):
    """Generate diverse structured initial sequences."""
    mode = rng.integers(0, 7)
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    elif mode == 2:
        sigma = n * rng.uniform(0.15, 0.35)
        a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    elif mode == 3:
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.15)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif mode == 4:
        p = rng.uniform(0.5, 3.0)
        a = np.cos(np.pi * x / n) ** (2 * p) + 1e-8
    elif mode == 5:
        a = 1 - np.abs(2 * x / n - 1)
    else:
        # Fejér-like: average of cosines
        n_harm = rng.integers(2, 6)
        a = np.ones(n)
        for k in range(1, n_harm):
            a += np.cos(2 * np.pi * k * x / n)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines smoothed gradient descent, LP alternating,
    multi-start exploration, and multi-n search to minimize the objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Baselines
    print("=== Phase 0: Baseline ===")
    for n_base in [600, 800, 1000, 1200]:
        if time.monotonic() > deadline:
            break
        score = evaluate_sequence([1.0] * n_base)
        if score < best_score:
            best_score = score
            best_seq = np.ones(n_base)
        print(f"  Constant n={n_base}: score={score:.8f}")

    # Try parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                score = evaluate_sequence([float(x) for x in parent_arr])
                if score < np.inf:
                    print(f"  Parent (n={parent_arr.size}): score={score:.8f}")
                    if score < best_score:
                        best_score = score
                        best_seq = parent_arr.copy()
    except Exception as e:
        print(f"  Parent not available: {e}")

    print(f"  Best so far: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: Smooth gradient descent on multiple n values and initializations
    print("\n=== Phase 1: Smooth gradient descent multi-start ===")
    n_best = len(best_seq)
    n_options = sorted(set([n_best, max(300, n_best - 200), n_best + 200, 600, 1000, 1400]))

    candidates = {}
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        n_starts = 4
        alloc_per_start = min(25.0, remaining / (len(n_options) * n_starts))

        for s_i in range(n_starts):
            if time.monotonic() > deadline:
                break
            if s_i == 0 and n_try == n_best and best_seq is not None and len(best_seq) == n_try:
                a_init = best_seq / np.sum(best_seq)
            else:
                a_init = structured_init(n_try, rng)

            a_opt = smooth_gradient_descent(a_init, time_budget=alloc_per_start,
                                            beta_start=5.0, beta_end=800.0)
            ev_opt = evaluate_sequence([float(x) for x in a_opt])

            if ev_opt < best_score:
                best_score = ev_opt
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}, init={s_i}: score={best_score:.8f} **")
            else:
                print(f"  n={n_try}, init={s_i}: score={ev_opt:.8f}")

            obj_init = compute_obj(a_opt)
            if n_try not in candidates or obj_init < candidates[n_try][0]:
                candidates[n_try] = (obj_init, a_opt.copy())

    # Phase 2: LP alternating on the best candidate
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: LP alternating (remaining={remaining:.0f}s) ===")
        a_curr = best_seq.copy()
        s = float(np.sum(a_curr))
        a_curr = a_curr / s

        alloc = min(remaining * 0.35, 280.0)
        a_lp = alternating_optimize(a_curr, time_budget=alloc, verbose=True)
        ev_lp = evaluate_sequence([float(x) for x in a_lp])
        if ev_lp < best_score:
            best_score = ev_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: score={best_score:.8f} **")
        else:
            print(f"  LP: score={ev_lp:.8f}")
        if ev_lp < candidates.get(len(best_seq), (np.inf, None))[0]:
            candidates[len(best_seq)] = (compute_obj(a_lp), a_lp.copy())

    # Phase 3: High-beta smooth GD refinement
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: High-beta smooth GD (remaining={remaining:.0f}s) ===")
        a_curr = best_seq.copy()
        s = float(np.sum(a_curr))
        a_curr = a_curr / s

        alloc = min(remaining * 0.25, 100.0)
        a_sg = smooth_gradient_descent(a_curr, time_budget=alloc,
                                       beta_start=30.0, beta_end=5000.0, lr_start=0.03)
        ev_sg = evaluate_sequence([float(x) for x in a_sg])
        if ev_sg < best_score:
            best_score = ev_sg
            best_seq = a_sg.copy()
            print(f"  ** High-beta GD: score={best_score:.8f} **")
        else:
            print(f"  High-beta GD: score={ev_sg:.8f}")

    # Phase 4: LP on other promising candidates
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: LP on other candidates (remaining={remaining:.0f}s) ===")
        for n_try in sorted(candidates.keys(), key=lambda k: candidates[k][0]):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            if n_try == len(best_seq) and abs(candidates[n_try][0] - compute_obj(best_seq)) < 1e-8:
                continue
            alloc = min(50.0, remaining * 0.25)
            a_init = candidates[n_try][1]
            a_lp2 = alternating_optimize(a_init, time_budget=alloc)
            ev_lp2 = evaluate_sequence([float(x) for x in a_lp2])
            if ev_lp2 < best_score:
                best_score = ev_lp2
                best_seq = a_lp2.copy()
                print(f"  ** LP n={n_try}: score={best_score:.8f} **")
            else:
                print(f"  LP n={n_try}: score={ev_lp2:.8f}")

    # Phase 5: Intensive final polish - alternate LP and smooth GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 5: Final polish (remaining={remaining:.0f}s) ===")
        a_curr = best_seq.copy()
        s = float(np.sum(a_curr))
        a_curr = a_curr / s

        for round_idx in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.18, 50.0)
            if round_idx % 2 == 0:
                a_pol = alternating_optimize(a_curr, time_budget=alloc)
            else:
                a_pol = smooth_gradient_descent(a_curr, time_budget=alloc,
                                               beta_start=100.0, beta_end=10000.0,
                                               lr_start=0.02)

            ev_pol = evaluate_sequence([float(x) for x in a_pol])
            if ev_pol < best_score:
                best_score = ev_pol
                best_seq = a_pol.copy()
                a_curr = a_pol.copy()
                print(f"  Polish {round_idx}: score={best_score:.8f} **")
            else:
                print(f"  Polish {round_idx}: score={ev_pol:.8f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
