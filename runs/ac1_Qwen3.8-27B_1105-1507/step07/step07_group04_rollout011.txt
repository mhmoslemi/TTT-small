import numpy as np

def compute_conv(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_topk(a, k_threshold):
    """Build sparse-ish conv matrix, only keeping top-k peak rows."""
    n = len(a)
    conv = compute_conv(a)
    # Get indices of top peaks
    num_constraints = min(len(conv), k_threshold)
    peak_indices = np.argsort(conv)[::-1][:num_constraints]
    
    A_rows = []
    b_vals = []
    for k in peak_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        row = np.zeros(n)
        for j in range(j_start, j_end):
            row[j] = a[k - j]
        A_rows.append(row)
        b_vals.append(1.0)
    
    A = np.array(A_rows)
    return A, peak_indices


def solve_lp_fast(a, k_threshold=300, solver=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for top-k peaks, sum(g)=1, g>=0.
    Uses only top-k peak constraints for speed.

    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers
    
    for s in solvers:

                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    # Fallback: simple heuristic - find g that minimizes max convolution
    # Use iterative proportional fitting
    g = np.ones(n) / n
    for _ in range(50):

        g = np.maximum(g, 1e-10)

def compute_conv_full(a, b):
    """Compute convolution of a and b using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def parametric_gaussian(n, center=0.5, width=0.3):
    """Gaussian-shaped sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return a


def parametric_raised_cosine(n, power=1.0, width=1.0):
    """Raised cosine shaped sequence."""
    x = np.linspace(0, 1, n)
    a = np.maximum(0.0, np.cos(np.pi * (x - 0.5) / width)) ** power
    return a


def parametric_polynomial(n, p=2.0):
    """Polynomial shaped: (1 - (2x-1)^2)^p"""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1
    a = np.maximum(0.0, 1 - t**2) ** p
    return a


def parametric_bell(n, alpha=1.0):
    """Generalized bell curve."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1
    a = 1.0 / (1.0 + (t / (0.5 * alpha)) ** (2 * alpha + 1))
    return a


def smooth_random_sequence(n, rng, bandwidth=5):
    """Generate a smooth random non-negative sequence."""
    # Generate in frequency domain with low-pass filter
    freq = np.arange(n)
    mask = np.exp(-freq / bandwidth)
    coeffs = (rng.normal(0, 1, n) + 0.1 * n) * mask
    a = irfft(np.r_[coeffs[0], coeffs[1:(n//2)+1], np.conj(coeffs[1:(n//2)+1][::-1])] if n % 2 == 0 else np.r_[coeffs[0], coeffs[1:n//2], coeffs[n//2], np.conj(coeffs[1:n//2][::-1])], n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return np.maximum(a_new, 0)

def dual_ascent_step(a, n_iter=100, eta=0.01):
    """
    Dual ascent method: maintain dual variables (weights on peaks) and
    update a to minimize the weighted sum of peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Dual variables: weights on each peak
    weights = np.ones(2 * n - 1) / (2 * n - 1)

    for it in range(n_iter):
        if time.monotonic() - start > 60.0:
            break

        conv = compute_conv(a)
        
        # Update dual weights (multiplicative weights)
        max_conv = float(np.max(conv))
        weights = weights * np.exp(eta * conv / (max_conv + 1e-10))
        weights /= np.sum(weights)
        
        # Update a: minimize weighted sum of peaks
        # Gradient of weighted sum w.r.t. a[i] = sum_k weights[k] * a[k-i] (for valid k-i)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = weights[k]
            if wk < 1e-10:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                grad[j] += wk * a[k - j]
        
        # Subgradient step with projection
        a = a - eta * grad
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a

    candidates.append(('poly_p1', parametric_polynomial(n_base, 1.0)))
    candidates.append(('poly_p05', parametric_polynomial(n_base, 0.5)))
    candidates.append(('bell_1', parametric_bell(n_base, 1.0)))
    candidates.append(('bell_2', parametric_bell(n_base, 2.0)))
    candidates.append(('bell_05', parametric_bell(n_base, 0.5)))

    # Smooth random
    for r in range(5):
        candidates.append((f'smooth_rand_{r}', smooth_random_sequence(n_base, rng, bandwidth=3 + 2*r)))

    # Perturbed parent
    if parent is not None:
        for scale in [0.05, 0.1, 0.2]:
            a_pert = parent * np.exp(rng.normal(0, scale, n_base))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            candidates.append((f'parent_pert_{scale}', a_pert))

    # Evaluate all candidates
    print(f"\n=== Evaluating {len(candidates)} starting points ===")
    for name, a in candidates:
        obj = compute_obj(a)
        print(f"  {name} (n={len(a)}): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()

    print(f"\nBest initial: {best_score:.8f}")

    # === Phase 1: LP Alternating on best candidates ===
    print(f"\n=== Phase 1: LP Alternating on Top Candidates ===")
    remaining = deadline - time.monotonic()
    
    # Sort candidates by initial objective
    cand_objs = [(compute_obj(a), name, a) for name, a in candidates]
    cand_objs.sort()
    
    # Take top 8 candidates
    top_cands = cand_objs[:8]
    alloc_per = min(remaining * 0.3 / len(top_cands), 60.0)
    
    for idx, (obj_init, name, a) in enumerate(top_cands):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(alloc_per, remaining * 0.5)
        print(f"  [{idx+1}/{len(top_cands)}] {name} (init={obj_init:.6f}), budget={alloc:.0f}s")
        
        a_opt = lp_alternating_enhanced(a, time_budget=alloc, k_threshold=300, verbose=False)
        obj_opt = compute_obj(a_opt)
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"    ** Improved: {best_score:.8f}")
        else:
            print(f"    Result: {obj_opt:.8f}")

    # === Phase 2: Deep LP alternating on best ===
    print(f"\n=== Phase 2: Deep LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.35, 300.0)
        print(f"  Budget: {alloc:.0f}s, k_threshold=400")
        a_deep = lp_alternating_enhanced(best_seq, time_budget=alloc, k_threshold=400, verbose=True)
        obj_deep = compute_obj(a_deep)
        if obj_deep < best_score:
            best_score = obj_deep
            best_seq = a_deep.copy()
            print(f"  ** Deep LP: {best_score:.8f}")
        else:
            print(f"  Deep LP: {obj_deep:.8f}")

    # === Phase 3: Dual ascent on best ===
    print(f"\n=== Phase 3: Dual Ascent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_dual = dual_ascent_step(best_seq, n_iter=500, eta=0.05)
        obj_dual = compute_obj(a_dual)
        if obj_dual < best_score:
            best_score = obj_dual
            best_seq = a_dual.copy()
            print(f"  ** Dual: {best_score:.8f}")
        # Also try from a slightly perturbed version
        a_pert = best_seq * np.exp(rng.normal(0, 0.03, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / np.sum(a_pert)
        a_dual2 = dual_ascent_step(a_pert, n_iter=300, eta=0.05)
        obj_dual2 = compute_obj(a_dual2)
        if obj_dual2 < best_score:
            best_score = obj_dual2
            best_seq = a_dual2.copy()
            print(f"  ** Dual (perturbed): {best_score:.8f}")
        else:
            print(f"  Dual: min({obj_dual:.8f}, {obj_dual2:.8f})")

    # === Phase 4: More LP passes with different k_threshold ===
    print(f"\n=== Phase 4: LP with different K ===")
    remaining = deadline - time.monotonic()
    for k_th in [200, 500, 1000]:
        if remaining < 20:
            break
        alloc = min(remaining * 0.15, 60.0)
        a_k = lp_alternating_enhanced(best_seq, time_budget=alloc, k_threshold=k_th, verbose=False)
        obj_k = compute_obj(a_k)
        if obj_k < best_score:
            best_score = obj_k
            best_seq = a_k.copy()
            print(f"  ** K={k_th}: {best_score:.8f}")
        else:
            print(f"  K={k_th}: {obj_k:.8f}")
        remaining = deadline - time.monotonic()

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([n_current - 10, n_current - 5, n_current, n_current + 5, n_current + 10]))
    n_options = [n for n in n_options if 400 <= n <= 2000]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_current:
            continue  # Already optimized
        
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 45.0)
        a_opt = lp_alternating_enhanced(a_try, time_budget=alloc, k_threshold=300, verbose=False)
        obj_try = compute_obj(a_opt)
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: {best_score:.8f}")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # LP polish
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_lp = lp_alternating_enhanced(best_seq, time_budget=alloc, k_threshold=500, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP polish: {best_score:.8f}")
        else:
            print(f"  LP polish: {obj_lp:.8f}")
        remaining = deadline - time.monotonic()

    # Coordinate descent
    if remaining > 15:
        alloc = min(remaining * 0.2, 30.0)
        a_cd = coordinate_descent_polish(best_seq, time_budget=alloc, verbose=False)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD polish: {best_score:.8f}")
        else:
            print(f"  CD polish: {obj_cd:.8f}")
        remaining = deadline - time.monotonic()

    # Another LP pass after CD
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_final = lp_alternating_enhanced(best_seq, time_budget=alloc, k_threshold=400, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP: {best_score:.8f}")
        else:
            print(f"  Final LP: {obj_final:.8f}")

    # === Phase 7: Last-chance random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 7: Last-chance Restarts ===")
        n_best = len(best_seq)
        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # Different perturbation strategies
            if r == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.02, n_best))
            elif r == 1:
                # Block perturbation
                a_r = best_seq.copy()
                block = n_best // 4
                start_b = r * block
                a_r[start_b:start_b+block] *= np.exp(rng.normal(0, 0.1, block))
            elif r == 2:
                # Add a smooth bump
                a_r = best_seq.copy()
                bump = parametric_gaussian(n_best, rng.uniform(0.2, 0.8), 0.1)
                a_r = a_r + 0.05 * bump
            elif r == 3:
                # Taper the edges
                a_r = best_seq.copy()
                taper = np.linspace(0.5, 1.0, n_best // 4)
                a_r[:n_best//4] *= taper
                a_r[-n_best//4:] *= taper[::-1]
            else:
                a_r = best_seq * (1 + 0.05 * np.sin(2*np.pi*rng.uniform(1,5)*np.linspace(0,1,n_best)))
            
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
            
            alloc = min(remaining * 0.3, 20.0)
            a_opt = lp_alternating_enhanced(a_r, time_budget=alloc, k_threshold=300, verbose=False)
            obj_r = compute_obj(a_opt)
            
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  ** Restart {r}: {best_score:.8f}")
            else:
                print(f"  Restart {r}: {obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
