"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed objective with projected gradient (simplex constraint)
2. Adaptive temperature schedule
3. LP-free alternating with scipy linprog
4. Multi-scale n exploration
5. Momenta and backtracking line search
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective assuming sum(a) >= 0.01. Scale-invariant."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max(conv, T):
    """Log-sum-exp approximation of max(conv) with temperature T."""
    m = np.max(conv)
    if m < 1e-15:
        return 0.0
    shifted = T * (conv - m)
    # Clip for numerical stability
    shifted = np.maximum(shifted, -50)
    return m + np.log(np.sum(np.exp(shifted))) / T


def smoothed_grad(conv, T):
    """Gradient of (1/T)*log(sum(exp(T*conv_k))) with respect to conv_k.
    Returns softmax weights."""
    m = np.max(conv)
    if m < 1e-15:
        return np.ones(len(conv)) / len(conv)
    shifted = T * (conv - m)
    shifted = np.maximum(shifted, -50)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    return weights


def subgradient_smoothed(a, weights):
    """Compute subgradient of max(conv(a,a)) given soft weights over conv bins.
    grad[j] = sum_k weights[k] * 2*a[k-j] (for valid k-j in [0,n))
    """
    n = len(a)
    conv_len = 2 * n - 1
    # For each j in [0, n), grad[j] = 2 * sum_{k: 0<=k-j<n} weights[k] * a[k-j]
    # = 2 * sum_{m=0}^{n-1} weights[m+j] * a[m]  (where m+j < 2n-1)
    # This is a cross-correlation: (weights ⊛ a)[j] = sum_m a[m] * weights[m+j]
    # Compute via FFT
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(weights, size)
    # cross_corr[j] = sum_m a[m] * w[m+j] = ifft(conj(F_a) * F_w)[j]
    cross = irfft(np.conj(fa) * fw, size)
    grad = 2.0 * cross[:n]
    return grad


def smooth_gradient_optimize(a, time_budget=200.0, T_start=5.0, T_end=80.0,
                             verbose=True, momentum_initial=0.3):
    """Projected gradient with log-sum-exp smoothing, momentum, and backtracking."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)  # momentum buffer
    momentum = momentum_initial
    step_size = 1.0 / np.sqrt(n)
    stall = 0
    patience = 100
    it = 0

    while time.monotonic() - start < time_budget:
        # Adaptive temperature: start low (smooth), increase (sharper)
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** progress

        conv = compute_conv(a)
        weights = smoothed_grad(conv, T)
        g = subgradient_smoothed(a, weights)

        # Momentum
        v = momentum * v + g

        # Backtracking line search
        alpha = step_size
        improved = False
        for _ in range(20):
            a_try = a - alpha * v
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                stall = 0
                break
            alpha *= 0.5

        if not improved:
            stall += 1
            step_size *= 0.8
            if stall > patience:
                # Reset with perturbation
                rng = np.random.default_rng(it * 7919 + n)
                scale = min(0.4, 0.05 + 0.03 * (stall // patience))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = momentum_initial
                step_size = 1.0 / np.sqrt(n)
                stall = 0

        # Adapt momentum
        if improved:
            momentum = min(0.9, momentum + 0.002)
            step_size = min(2.0 / np.sqrt(n), step_size * 1.01)
        else:
            momentum = max(0.1, momentum - 0.005)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG it{it}: obj={best_obj:.8f}, T={T:.1f}, alpha={alpha:.6f}, elapsed={elapsed:.1f}s")

        it += 1

    return best_a


def build_conv_matrix_sparse(a):
    """Build the convolution matrix A (2n-1 x n) where A@g = conv(a,g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=30):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog for speed.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    if n > 2000:
        return None, None  # Too large for linprog
    a = a / np.sum(a)
    A = build_conv_matrix_sparse(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def lp_alternating_fast(a, time_budget=300.0, verbose=False):
    """LP alternating with scipy linprog and adaptive mixing."""
    n = len(a)
    if n > 2500:

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))

        return a

    best_obj = compute_obj(a)

    stall = 0
    last_t = 0.5
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        lp_time = min(20.0, remaining * 0.3)
        if lp_time < 2:
            break

        g_lp, M_lp = solve_lp_minpeak(a, time_limit=lp_time)
        if g_lp is None:

        # Line search over t

        best_t_val = 0.5

        if best_t_obj < best_obj - 1e-13:

            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val

                a = np.maximum(a, 1e-15)
                a = a / a.sum()

def chase_peak_refine(a, time_budget=60.0, verbose=False):
    """Chase-the-peak with Barzilai-Borwein step size on simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_a = a.copy()
    prev_g = None
    step_size = 0.02 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        g = np.zeros(n)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            j_start = k_star - i_end
            j_end = k_star - i_start
            g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]

        # BB step
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-15:
                bb = ss / abs(sy)
                step_size = min(max(step_size * 0.5, 1e-8), bb, 1.0)

        prev_a = a.copy()
        prev_g = g.copy()

        a_new = a - step_size * g
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            step_size = min(2.0 / np.sqrt(n), step_size * 1.05)
        else:
            step_size *= 0.7

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    CP it{it}: obj={best_obj:.8f}, step={step_size:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def coordinate_descent_refine(a, time_budget=30.0):
    """Move mass from peak-contributing positions to reduce max conv."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))

        # Find contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Non-contributors
        contrib_set = set()
        for _, i, j in contributors:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [x for x in range(n) if x not in contrib_set]

        if not non_contrib:
            non_contrib = [x for x in range(n) if a[x] < np.mean(a)]

        improved = False
        for contrib_val, i, j in contributors[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < 1e-12:
                break
            for target in non_contrib[:8]:
                for frac in [0.05, 0.1, 0.15, 0.25]:
                    delta = frac * a[i]
                    if a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
            if improved:
                conv = compute_conv(a)

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def make_initialization(n, mode, rng):
    """Generate diverse initialization."""
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        # Hanning window
        x = np.arange(n)
        a = 0.5 * (1 - np.cos(2 * np.pi * x / max(n - 1, 1)))
        a = np.maximum(a, 1e-6)
    elif mode == 3:
        # Inverted triangle
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = 1.0 - np.abs(x - center) / center
        a = np.maximum(a, 0.01)
    elif mode == 4:
        # Sine
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 1e-6)
    elif mode == 5:
        # Raised cosine (blackman-like)
        x = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        width = n // 10
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * width ** 2))
        a += 0.01
    elif mode == 7:
        # Smooth random (low freq)
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amplitudes = rng.uniform(0.2, 0.8, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amplitudes):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
    elif mode == 8:
        # Flat bottom, peaked top
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * x) ** 2
        a = np.maximum(a, 0.01)
    elif mode == 9:
        # Two-scale structure
        a = np.ones(n)
        block = max(1, n // 20)
        for i in range(0, n, block * 2):
            a[i:i+block] *= 0.5
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-15)
    return a / a.sum()

    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_budget = min(remaining * 0.05, 30.0)
        a_cd = coordinate_descent_refine(best_seq, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coordinate: obj={best_score:.8f} **")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 100), n_current,
        min(2000, n_current + 100), min(2000, n_current + 200),
        600, 800, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000]

    n_done = set()
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try in n_done:
            continue
        n_done.add(n_try)

        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick smooth gradient
        sg_alloc = min(remaining * 0.12, 60.0)
        if sg_alloc > 15:
            a_opt = smooth_gradient_optimize(a_try, time_budget=sg_alloc,
                                             T_start=5.0, T_end=50.0)
            obj_opt = compute_obj(a_opt)
            if obj_opt < compute_obj(a_try):
                a_try = a_opt

        # Quick LP (only if n is small enough)
        remaining = deadline - time.monotonic()
        if remaining > 30 and len(a_try) <= 1500:
            lp_alloc = min(remaining * 0.1, 45.0)
            if lp_alloc > 10:
                a_lp2 = lp_alternating_fast(a_try, time_budget=lp_alloc)
                obj_lp2 = compute_obj(a_lp2)
                if obj_lp2 < compute_obj(a_try):
                    a_try = a_lp2

        obj_final = compute_obj(a_try)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_final:.8f}")

    # === Phase 5: Second round of optimization on best ===
    print(f"\n=== Phase 5: Second Round Optimization ===")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        sg2_budget = min(remaining * 0.2, 150.0)
        a_sg2 = smooth_gradient_optimize(best_seq, time_budget=sg2_budget,
                                         T_start=10.0, T_end=200.0,
                                         verbose=True, momentum_initial=0.5)
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  SG round 2: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 40 and len(best_seq) <= 2000:
        lp2_budget = min(remaining * 0.25, 150.0)
        a_lp3 = lp_alternating_fast(best_seq, time_budget=lp2_budget, verbose=True)
        obj_lp3 = compute_obj(a_lp3)
        if obj_lp3 < best_score:
            best_score = obj_lp3
            best_seq = a_lp3.copy()
            print(f"  LP round 2: obj={best_score:.8f} **")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3 + 1) % 10
        a_r = make_initialization(n_best, mode, rng)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 1e-15)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.15, 80.0)
        a_opt = smooth_gradient_optimize(a_r, time_budget=restart_budget,
                                         T_start=5.0, T_end=80.0)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        cp2 = chase_peak_refine(best_seq, time_budget=min(remaining * 0.2, 50.0))
        obj_cp2 = compute_obj(cp2)
        if obj_cp2 < best_score:
            best_score = obj_cp2
            best_seq = cp2.copy()
            print(f"  Final chase peak: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd2 = coordinate_descent_refine(best_seq, time_budget=min(remaining * 0.2, 30.0))
        obj_cd2 = compute_obj(cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = cd2.copy()
            print(f"  Final coordinate: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 1500:
        lp_final = lp_alternating_fast(best_seq, time_budget=min(remaining * 0.4, 80.0), verbose=True)
        obj_final2 = compute_obj(lp_final)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        sg_final = smooth_gradient_optimize(best_seq, time_budget=min(remaining * 0.5, 40.0),
                                            T_start=50.0, T_end=500.0, momentum_initial=0.6)
        obj_sgf = compute_obj(sg_final)
        if obj_sgf < best_score:
            best_score = obj_sgf
            best_seq = sg_final.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    result = [float(max(0.0, min(999.0, x))) for x in best_seq]

    # Validate with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
