Uses Adam gradient on smoothed objective, fast LP alternating (HiGHS),

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_ab(a, b):
    n, m = len(a), len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0 using HiGHS."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    # Variables: g_0, ..., g_{n-1}, M
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Inequality: A @ g - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        s = float(np.sum(g_val))

    return None, None

    while p < m:
        p <<= 1

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_grad_simplex(a, tau):
    """
    Compute gradient of logsumexp(conv(a)/tau) w.r.t. a, assuming sum(a)=1.
    Uses FFT for efficiency.
    Returns (objective_value, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    
    # Gradient: dF/da[j] = (2/tau) * sum_k w[k] * a[k-j] (with valid indices)
    # This is (2/tau) * cross_correlation(w, a) evaluated at the right positions
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    
    grad = (2.0 / tau) * corr
    return logsumexp_val, grad

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 8:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                scale = 0.02 + 0.02 * min(stall // 8, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
    
    return best_a


def coordinate_descent_polish(a, time_budget=30.0):
    """Coordinate descent: optimize each coordinate one at a time."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            old_val = a[i]
            # Try a few candidate values for coordinate i
            candidates = [0.0, old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5,
                         old_val + 0.01 / n, old_val - 0.01 / n, 2.0 / n]
            best_local = old_val
            best_local_obj = best_obj
            for cv in candidates:
                cv = max(0.0, cv)
                a[i] = cv
                s = float(np.sum(a))
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s
                obj = compute_obj(a_norm)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local = cv
            a[i] = best_local
            if best_local != old_val:
                improved = True
                a = a / np.sum(a)
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()
        
        if not improved:
            # Try perturbation
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            idx = rng.integers(0, n, size=min(n // 10, 100))
            a[idx] *= np.exp(rng.normal(0, 0.1, len(idx)))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_a


def spectral_init(n, mode=0, rng=None):
    """Generate spectral-based initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    if mode == 0:
        # Flat spectrum (except DC) - gives "white noise" like sequence
        A = np.zeros(n, dtype=complex)
        A[0] = 1.0
        for k in range(1, n):
            A[k] = np.exp(1j * rng.uniform(0, 2 * np.pi)) / np.sqrt(n)
        a = irfft(A * n)[:n]
        a = np.maximum(a, 0)
    elif mode == 1:
        # Raised cosine spectrum
        A = np.zeros(n, dtype=complex)
        A[0] = 1.0
        freqs = np.arange(n) / n
        for k in range(1, n):
            envelope = 0.5 * (1 + np.cos(np.pi * freqs[k] * 0.5))
            A[k] = envelope / np.sqrt(n) * np.exp(1j * rng.uniform(0, 2 * np.pi))
        a = irfft(A * n)[:n]
        a = np.maximum(a, 0)
    elif mode == 2:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a += 0.01 * rng.uniform(0, 1, n)
    elif mode == 3:
        # Power law
        beta = rng.uniform(0.5, 2.0)
        a = (np.arange(n) + 1) ** (-beta)
        a += 0.01
    elif mode == 4:
        # Bimodal
        a = np.exp(-4 * ((np.arange(n) / n - 0.25) ** 2)) + np.exp(-4 * ((np.arange(n) / n - 0.75) ** 2))
        a += 0.01 * rng.uniform(0, 1, n)
    elif mode == 5:
        # Inverse cosine (dip in middle)
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.5 * np.cos(x)
        a += 0.01 * rng.uniform(0, 1, n)
    else:
        a = rng.uniform(0.1, 1.0, n)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def lgbfgs_optimize(a_init, time_budget=60.0, verbose=False):
    """L-BFGS-B on log-parameterized smoothed objective."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = np.log(a)
    x = x - x.max()
    
    taus = [5.0 / n, 2.0 / n, 0.5 / n, 0.1 / n, 0.05 / n]
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        def fun(xv):
            xv = xv - xv.max()
            aa = np.exp(xv)
            aa = aa / np.sum(aa)
            conv = compute_conv(aa)
            cs = conv / tau
            cm = np.max(cs)
            return cm + np.log(np.sum(np.exp(cs - cm)))
        
        def fgrad(xv):
            xv = xv - xv.max()
            aa = np.exp(xv)
            aa = aa / np.sum(aa)
            conv = compute_conv(aa)
            cs = conv / tau
            cm = np.max(cs)
            ec = np.exp(cs - cm)
            se = np.sum(ec)
            w = ec / se
            size = next_pow2(2 * n - 1)
            af = aa[::-1]
            faf = rfft(af, size)
            fw = rfft(w, size)
            cwaf = irfft(fw * faf, size)
            corr = cwaf[n - 1:n - 1 + n]
            grad_a = (2.0 / tau) * corr
            grad_x = aa * grad_a - aa * np.mean(aa * grad_a)
            return grad_x
        
        from scipy.optimize import minimize as sp_min
        res = sp_min(fun, x, method='L-BFGS-B', jac=fgrad,
                     options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50})
        
        if res.fun < fun(x):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    L-BFGS tau={tau:.4f}: obj={best_obj:.8f}")
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase hybrid optimization combining Adam gradient,
    L-BFGS-B, LP alternating, coordinate descent, and spectral initialization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Diverse initialization and selection ===
    print(f"\n=== Phase 1: Diverse Initialization ===")
    remaining = deadline - time.monotonic()
    
    candidates = [best_seq.copy()]
    
    # Spectral initializations at parent length
    for mode in range(6):
        a_try = spectral_init(n_parent, mode=mode, rng=rng)
        candidates.append(a_try)
    
    # Random initializations
    for _ in range(4):
        a_try = rng.uniform(0.05, 1.0, n_parent)
        a_try = a_try / np.sum(a_try)
        candidates.append(a_try)
    
    # Evaluate all candidates
    scored_candidates = []
    for c in candidates:
        obj_c = compute_obj(c)
        scored_candidates.append((obj_c, c))
        if obj_c < best_score:
            best_score = obj_c
            best_seq = c.copy()
    
    scored_candidates.sort(key=lambda x: x[0])
    print(f"  Best init: obj={best_score:.10f}")
    for i, (sc, _) in enumerate(scored_candidates[:5]):
        print(f"    Candidate {i}: obj={sc:.10f}")
    
    # === Phase 2: Adam optimization on top candidates ===
    print(f"\n=== Phase 2: Adam Optimization ===")
    remaining = deadline - time.monotonic()
    top_k = min(3, len(scored_candidates))
    adam_budget_per = min(remaining * 0.15, 80.0)
    
    for idx in range(top_k):
        remaining = deadline - time.monotonic()
        if remaining < adam_budget_per:
            break
        a_init = scored_candidates[idx][1].copy()
        a_opt = adam_optimize_simplex(a_init, time_budget=adam_budget_per, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Adam on cand {idx}: obj={best_score:.10f} **")
        else:
            print(f"  Adam on cand {idx}: obj={obj:.10f}")
    
    # === Phase 3: L-BFGS-B optimization ===
    print(f"\n=== Phase 3: L-BFGS-B Optimization ===")
    remaining = deadline - time.monotonic()
    lgbfgs_budget = min(remaining * 0.15, 100.0)
    a_lb = lgbfgs_optimize(best_seq, time_budget=lgbfgs_budget, verbose=True)
    obj = compute_obj(a_lb)
    if obj < best_score:
        best_score = obj
        best_seq = a_lb.copy()
        print(f"  After L-BFGS-B: obj={best_score:.10f} **")
    
    # === Phase 4: LP Alternating ===
    print(f"\n=== Phase 4: LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(remaining * 0.2, 150.0)
    a_lp = lp_alternate_optimize(best_seq, time_budget=lp_budget, verbose=True)
    obj = compute_obj(a_lp)
    if obj < best_score:
        best_score = obj
        best_seq = a_lp.copy()
        print(f"  After LP: obj={best_score:.10f} **")
    
    # === Phase 5: Adam + LP alternation ===
    print(f"\n=== Phase 5: Adam + LP Alternation ===")
    remaining = deadline - time.monotonic()
    cycles = max(2, int(remaining / 60))
    cycles = min(cycles, 5)
    
    for cyc in range(cycles):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        half_budget = min(remaining * 0.4, 50.0)
        
        # Adam step
        a_ad = adam_optimize_simplex(best_seq, time_budget=half_budget)
        obj = compute_obj(a_ad)
        if obj < best_score:
            best_score = obj
            best_seq = a_ad.copy()
            print(f"  Cycle {cyc} Adam: obj={best_score:.10f}")
        
        # LP step
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        lp_b = min(remaining * 0.3, 40.0)
        a_lp2 = lp_alternate_optimize(best_seq, time_budget=lp_b)
        obj = compute_obj(a_lp2)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()
            print(f"  Cycle {cyc} LP: obj={best_score:.10f}")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    n_candidates = sorted(set([
        n_best - 100, n_best - 50, n_best,
        n_best + 50, n_best + 100, n_best + 200,
        600, 700, 900, 1000, 1200, 1400
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1600]
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try == n_best:
            continue
        
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 35.0)
        
        # Quick Adam
        a_opt = adam_optimize_simplex(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")
    
    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coordinate Descent Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        cd_budget = min(remaining * 0.2, 60.0)
        a_cd = coordinate_descent_polish(best_seq, time_budget=cd_budget)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  After CD: obj={best_score:.10f} **")
    
    # === Phase 8: Final intensive optimization ===
    print(f"\n=== Phase 8: Final Intensive Optimization ===")
    remaining = deadline - time.monotonic()
    
    # Fine Adam
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_fin = adam_optimize_simplex(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_fin)
        if obj < best_score:
            best_score = obj
            best_seq = a_fin.copy()
            print(f"  Final Adam: obj={best_score:.10f}")
    
    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_lp3 = lp_alternate_optimize(best_seq, time_budget=alloc)
        obj = compute_obj(a_lp3)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp3.copy()
            print(f"  Final LP: obj={best_score:.10f}")
    
    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_cd2 = coordinate_descent_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_cd2)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd2.copy()
            print(f"  Final CD: obj={best_score:.10f}")
    
    # === Phase 9: Random restarts ===
    print(f"\n=== Phase 9: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    n_restarts = max(2, int(remaining / 40))
    n_restarts = min(n_restarts, 5)
    
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        
        if r == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
        elif r == 1:
            n_r = max(500, min(1500, n_best + int(rng.integers(-100, 100))))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
        elif r == 2:
            a_r = spectral_init(n_best, mode=int(rng.integers(0, 6)), rng=rng)
            a_r = 0.5 * a_r + 0.5 * best_seq
        else:
            a_r = best_seq + 0.01 * rng.normal(0, 1, n_best)
        
        a_r = np.maximum(a_r, 0)
        a_r = a_r / np.sum(a_r)
        
        alloc = min(remaining * 0.2, 30.0)
        a_opt = adam_optimize_simplex(a_r, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r}: obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj:.10f}")
    
    # === Final polish ===
    print(f"\n=== Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 20.0)
        a_final = adam_optimize_simplex(best_seq, time_budget=alloc)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_cd_final = coordinate_descent_polish(best_seq, time_budget=min(remaining * 0.5, 10.0))
            obj = compute_obj(a_cd_final)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd_final.copy()
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
