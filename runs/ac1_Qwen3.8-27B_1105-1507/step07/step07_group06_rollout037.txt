    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)

def smooth_max_and_weights(conv, tau):
    """Returns smoothed max and soft-argmax weights."""
    m = np.max(conv)
    scaled = (conv - m) * tau
    # Numerical stability
    max_scaled = np.max(scaled)  # should be 0
    exp_scaled = np.exp(scaled - max_scaled)
    log_sum = max_scaled + np.log(np.sum(exp_scaled))
    smooth_val = m + log_sum / tau
    w = tau * exp_scaled / np.sum(exp_scaled)
    return smooth_val, w

def obj_and_grad_theta(theta, tau, n):
    """Objective and gradient w.r.t. log-softmax parameters."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    a = exp_t / np.sum(exp_t)
    
    conv = compute_conv(a)
    smooth_val, w = smooth_max_and_weights(conv, tau)
    obj = 2.0 * n * smooth_val
    
    # c[j] = sum_m w[j+m] * a[m]
    c = np.correlate(w, a, 'valid')
    
    # dF/dtheta[j] = a[j] * (G[j] - sum_i G[i]*a[i]) where G = 4n*c
    G = 4.0 * n * c
    mean_G = np.dot(G, a)
    grad = a * (G - mean_G)
    
    return obj, grad

"""
Multi-strategy optimization for minimizing auto-correlation peak of non-negative sequences.
Key improvements over previous approach:
1. Fast gradient via np.correlate for the soft-argmax weighting
2. Diverse initializations including spectral-shaping and structured starts
3. Aggressive tau-annealed L-BFGS-B with warm-starting across scales
4. Subgradient descent on the exact (non-smoothed) objective for final refinement
5. Simulated-anneal-style random restarts to escape local minima
6. Peak-targeting coordinate descent focusing on top-k convolution entries
7. LP alternating for structured improvement when time permits
"""
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear auto-convolution via FFT. Length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum a)^2. Scale-invariant."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return float(2.0 * n * np.max(conv))


def softmax_fn(theta):
    """Numerically stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def smooth_max_and_weights(conv, tau):
    """Compute smoothed max and soft-argmax weights."""
    m = np.max(conv)
    scaled = (conv - m) * tau
    exp_s = np.exp(scaled)
    s = np.sum(exp_s)
    smooth_val = m + np.log(s) / tau
    w = tau * exp_s / s
    return smooth_val, w


def obj_grad_log(theta, tau, n):
    """Smoothed objective and gradient w.r.t. log-softmax params."""
    a = softmax_fn(theta)
    conv = compute_conv(a)
    smooth_val, w = smooth_max_and_weights(conv, tau)
    obj = 2.0 * n * smooth_val
    # c[j] = sum_m w[j+m] * a[m]  (cross-correlation)
    c = np.correlate(w, a, 'valid')
    G = 4.0 * n * c
    mean_G = np.dot(G, a)
    grad = a * (G - mean_G)
    return obj, grad


def lssb_optimize(a_init, time_budget, tau_start=5.0, tau_end=600.0, n_stages=6):
    """
    Smoothed L-BFGS-B with geometric tau annealing and warm-starting.
    More iterations at low tau, fewer at high tau.
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-12)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    # Geometric schedule: more stages at low tau
    taus = np.geomspace(tau_start, tau_end, n_stages)
    # Allocate more iterations to lower tau
    max_iters = [300, 250, 200, 150, 100, 80][:n_stages]

    for idx, tau in enumerate(taus):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        def _obj(th):
            v, _ = obj_grad_log(th, tau, n)
            return v

        def _grad(th):
            _, g = obj_grad_log(th, tau, n)
            return g

        mi = max_iters[idx] if idx < len(max_iters) else 80
        try:
            res = minimize(_obj, theta, jac=_grad, method='L-BFGS-B',
                           options={'maxiter': mi, 'gtol': 1e-8, 'maxfun': mi * 3,
                                    'ftol': 1e-10})
            if res.x is not None:
                theta = res.x.copy()
                a_new = softmax_fn(theta)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
        except Exception:
            pass

        # Check time and continue
        if time.monotonic() - start > time_budget * 0.95:
            break

    return best_a, best_obj


def subgradient_descent(a, time_budget=30):
    """
    Subgradient descent on the exact objective (max of linear functions).
    Uses top-k peaks for a better subgradient estimate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    top_k = 5

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Identify top-k peaks
        peak_indices = np.argsort(conv)[-top_k:]
        # Average subgradient from top-k peaks
        # d conv[k] / d a[j] = 2*a[k-j] if 0 <= k-j < n
        subgrad = np.zeros(n)
        for k in peak_indices:
            offset = k  # k-j ranges; valid when 0<=k-j<n => j in [max(0,k-n+1), min(k,n-1)]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                subgrad[j_start:j_end + 1] += 2.0 * a[k - j_start:k - j_start + 1][::-1] if k < n else 2.0 * a[:k - j_start + 1]

        # Actually let me just compute it properly
        subgrad = np.zeros(n)
        for k in peak_indices:
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    subgrad[j] += 2.0 * a[kj]

        # Normalize subgradient
        sg_norm = np.linalg.norm(subgrad)
        if sg_norm < 1e-15:
            break
        subgrad = subgrad / sg_norm

        # Project onto tangent space (sum = 1 constraint)
        subgrad_proj = subgrad - np.mean(subgrad)

        # Step
        a_new = a - step_size * subgrad_proj
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            step_size *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            step_size = min(step_size * 1.05, 0.5)
        else:
            step_size *= 0.8

        if step_size < 1e-5:
            break

    return best_a


def build_conv_matrix(a):
    """Matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, timeout=10):
    """LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        from time import monotonic
        t0 = monotonic()
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if monotonic() - t0 > timeout:
            return None, None
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_refine(a, time_budget=30):
    """LP alternating: alternate between fixing a and solving for g."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Line search
        best_local = best_obj
        best_local_a = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local:
                best_local = obj_try
                best_local_a = a_try.copy()

        if best_local < best_obj - 1e-13:
            best_obj = best_local
            best_a = best_local_a.copy()
            a = best_a
        else:
            if it % 5 == 0:
                rng = np.random.default_rng(it * 7 + n)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
            else:
                break

    return best_a


def spectral_init(n, rng):
    """Initialize with a sequence whose power spectrum is approximately flat."""
    N = 2 * (2 * n - 1)
    # Random phases with approximately flat magnitude
    freqs = np.fft.rfftfreq(N)
    # Avoid DC to reduce the mean
    mask = np.abs(freqs) > 1.0 / N
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    mags = rng.uniform(0.3, 1.0, N // 2 + 1)
    # Suppress low frequencies
    low_cutoff = max(2, N // 20)
    mags[:low_cutoff] *= 0.1
    spectrum = mags * np.exp(1j * phases)
    # Conjugate symmetry
    full_spec = np.zeros(N, dtype=complex)
    half = N // 2 + 1
    full_spec[:half] = spectrum
    full_spec[half:] = np.conj(spectrum[1:half][::-1])
    sig = irfft(full_spec, N)
    a = np.abs(sig[:n])
    a = a + 0.01  # ensure positive
    a = a / np.sum(a)
    return a


def random_init(n, rng, style='uniform'):
    """Various random initializations."""
    if style == 'uniform':
        a = rng.uniform(0.5, 1.0, n)
    elif style == 'exponential':
        rate = rng.uniform(0.5, 2.0)
        a = np.exp(np.linspace(0, rate, n) * (1 if rng.random() > 0.5 else -1))
    elif style == 'ramp':
        slope = rng.uniform(1.0, 3.0)
        a = np.linspace(1.0, slope, n)
    elif style == 'spiky':
        a = np.ones(n) * 0.1
        n_spikes = rng.integers(n // 4, n // 2)
        positions = rng.integers(0, n, n_spikes)
        a[positions] = rng.uniform(0.5, 2.0, n_spikes)
    elif style == 'binary':
        a = rng.binomial(1, 0.5, n).astype(float)
        if a.sum() < 0.01:
            a = np.ones(n) * 0.5
    elif style == 'gauss':
        a = np.abs(rng.normal(1.0, 0.5, n))
    else:
        a = rng.uniform(0.1, 1.0, n)
    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    return a


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-8)


def peak_coord_descent(a, time_budget=30):
    """Coordinate descent targeting the top convolution peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42 + n)

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Find contributors to the peak
        for i in range(0, n, max(1, n // 300)):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 1e-10:
                for frac in [0.005, 0.01, 0.02, 0.03]:
                    delta = frac * a[i]
                    if delta >= a[i]:
                        continue
                    # Move to a "quiet" position
                    quiet = rng.integers(0, n)
                    a[i] -= delta
                    a[quiet] += delta
                    a = np.maximum(a, 0)
                    s2 = np.sum(a)
                    if s2 < 1e-10:
                        a[i] += delta
                        a[quiet] -= delta
                        continue
                    a = a / s2
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                    else:
                        a = best_a.copy()
                        break
                if improved:
                    a = best_a.copy()
                    break

        if not improved:
            # Try random perturbations
            for _ in range(20):
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 == i2:
                    continue
                delta = 0.005 * a[i1] * rng.uniform(0.5, 2.0)
                a[i1] -= delta
                a[i2] += delta
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 < 1e-10:
                    continue
                a = a / s2
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()

            if not improved and it > 100:
                break
            elif not improved:
                # Larger restart
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2

    return best_a

            best_score = obj_sg

            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # Phase 3: LP refinement
    print("\n=== Phase 3: LP Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_a) <= 1000:
        alloc = min(remaining * 0.2, 60.0)
        a_lp = lp_refine(best_a, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_a = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # Phase 4: Peak-targeting coordinate descent
    print("\n=== Phase 4: Peak-Targeting CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 50.0)
        a_cd = peak_coord_descent(best_a, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_a = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # Phase 5: Another round of smoothed optimization from current best
    print("\n=== Phase 5: Refinement with High Tau ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 60.0)

        def _obj5(th):
            v, _ = obj_grad_log(th, 200.0, len(best_a))
            return v

        def _grad5(th):
            _, g = obj_grad_log(th, 200.0, len(best_a))
            return g

        theta = np.log(np.maximum(best_a, 1e-15))
        try:
            res = minimize(_obj5, theta, jac=_grad5, method='L-BFGS-B',
                          options={'maxiter': 300, 'gtol': 1e-9, 'maxfun': 1000,
                                  'ftol': 1e-11})
            if res.x is not None:
                a_new = softmax_fn(res.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_score:
                        best_score = obj_new
                        best_a = a_new.copy()
                        print(f"  High-tau refine: obj={best_score:.8f} **")
                    else:
                        print(f"  High-tau refine: obj={obj_new:.8f}")
        except Exception:
            print("  High-tau refine: failed")

    # Phase 6: Random restarts (simulated anneal style)
    print("\n=== Phase 6: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_restarts = min(5, int(remaining // 40))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Create perturbed version
        sigma = 0.05 + 0.05 * r
        a_r = best_a * np.exp(rng.normal(0, sigma, n_best))
        a_r = np.maximum(a_r, 0)
        a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.2, 40.0)
        a_opt, obj_opt = lssb_optimize(a_r, time_budget=alloc, tau_start=5.0, tau_end=400.0, n_stages=4)
        if obj_opt < best_score:
            best_score = obj_opt
            best_a = a_opt.copy()
            print(f"  Restart {r} (sigma={sigma:.2f}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (sigma={sigma:.2f}): obj={obj_opt:.8f}")

    # Phase 7: Final polish
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 15:
        a_final = subgradient_descent(best_a, time_budget=min(remaining * 0.3, 20.0))
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_a = a_final.copy()
            print(f"  Final subgrad: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final = peak_coord_descent(best_a, time_budget=min(remaining * 0.3, 20.0))
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_a = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f}")

    # Final validation and return
    if best_a is None:
        best_a = np.ones(100)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    s_result = sum(result)
    if s_result < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.ones(100))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
