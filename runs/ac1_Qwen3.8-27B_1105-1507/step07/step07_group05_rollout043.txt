**Adaptive Temperature Softmax:**

The key idea is to use `w_k = softmax(b_k / T)` where `T` is a temperature parameter. As `T → 0`, this focuses more on the maximum. As `T → ∞`, it becomes uniform (equivalent to energy minimization).

We can anneal `T` over time: start with a moderate `T` and decrease it.

**Step Size:**

For the subgradient method, the step size is crucial. Since we're minimizing a convex function over the simplex, we can use:
- Fixed step size with decay: `eta_t = c / sqrt(t)`
- Backtracking line search
- Barzilai-Borwein step size

I'll use a backtracking approach: try a step, and if the objective increases, halve the step size.

**Simplex Projection:**

The projection of a point `x` onto the simplex `{y >= 0, sum(y) = 1}` can be computed in O(n log n):
1. Compute `mu = (x - tau)_{+}` where `tau` is chosen so that `sum(y) = 1`.
2. Find `tau` by sorting: `tau = (sum(sorted_x) - 1) / n` where we find the right `k` such that... 

Actually, the standard algorithm is:
1. Sort `x` in descending order: `x_{(1)} >= x_{(2)} >= ... >= x_{(n)}`
2. Find the largest `k` such that `x_{(k)} - (sum_{i=1}^k x_{(i)} - 1)/k > 0`
3. Set `tau = (sum_{i=1}^k x_{(i)} - 1)/k`
4. Set `y_i = max(x_i - tau, 0)`

This is O(n log n) due to sorting.

For large `n` (like 800), this is fast enough.

def simplex_project(x):
    """Project x onto the probability simplex {y >= 0, sum(y) = 1}."""
    n = len(x)
    # Sort in descending order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - cssv/rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.where(cond)[0][-1] + 1
    theta = (cssv[k-1] - 1.0) / k
    return np.maximum(x - theta, 0.0)

Subgradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: FFT-based weighted subgradient with adaptive softmax temperature,

import numpy as np

from scipy.fft import rfft, irfft


def compute_conv_fft(a):
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(b) where b=conv(a,a), assuming sum(a)=1."""
    n = len(a)
    b = compute_conv_fft(a)
    return 2.0 * n * float(np.max(b))


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def compute_weighted_gradient(a, w, N_fft):
    """
    Compute g_j = sum_k w[k] * (a[k-j] + a[j]) * [0 <= k-j < n]
    using FFT.
    a: length n, w: length 2n-1
    """
    n = len(a)
    
    # Term 1: cross-correlation
    w_rev = w[::-1]
    fw_rev = rfft(w_rev, N_fft)
    fa = rfft(a, N_fft)
    conv_full = irfft(fw_rev * fa, N_fft)[:3*n-2]
    term1 = conv_full[n-1:2*n-1][::-1]
    
    # Term 2: sliding window sum
    w_cum = np.concatenate([[0.0], np.cumsum(w)])
    j_idx = np.arange(n)
    L = np.minimum(n, 2*n-1 - j_idx)
    term2 = a * (w_cum[j_idx + L] - w_cum[j_idx])
    
    return term1 + term2


def softmax_weights(b, temperature):
    """Compute softmax weights for convolution peaks."""
    b_max = np.max(b)
    exp_b = np.exp((b - b_max) / temperature)
    return exp_b / np.sum(exp_b)

def simplex_project(x):
    """Project onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]

    rho = np.arange(1, n + 1)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve: min M s.t. conv(a, g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns optimal g (normalized) or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""

from scipy.fft import rfft, irfft


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(b) where b=conv(a,a), assuming sum(a)=1, a>=0."""
    n = len(a)
    b = compute_conv_fft(a)
    return 2.0 * n * float(np.max(b))

    """Compute full objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def simplex_project(x):
    """Project x onto the probability simplex {y >= 0, sum(y) = 1} in O(n log n)."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - cssv / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.where(cond)[0][-1] + 1
    theta = (cssv[k - 1] - 1.0) / k
    return np.maximum(x - theta, 0.0)

    fw_rev = rfft(w_rev, N_fft)
    fa = rfft(a, N_fft)
    conv_full = irfft(fw_rev * fa, N_fft)[:3 * n - 2]
    term = conv_full[n - 1:2 * n - 1][::-1]
    return 2.0 * term


def softmax_weights(b, temperature):
    """Compute softmax weights."""
    b_max = np.max(b)
    exp_b = np.exp((b - b_max) / temperature)
    s = np.sum(exp_b)
    if s < 1e-30:
        return np.full(len(b), 1.0 / len(b))
    return exp_b / s


def subgradient_optimize(a, time_budget=100.0, verbose=False):
    """
    Adaptive-temperature FFT subgradient descent on the simplex.
    Input: a normalized (sum=1, a>=0).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)

    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1

    start = time.monotonic()
    total_iter_target = 50000
    elapsed_start = 0.0

    # Cosine annealing parameters
    eta_max = 2.0 / np.sqrt(n)
    eta_min = 0.01 / np.sqrt(n)
    temp_init = 0.5
    temp_min = 0.005

    stall = 0
    max_stall = 150
    iteration = 0

    while time.monotonic() - start < time_budget:
        # Annealing schedule
        progress = min(iteration / max(total_iter_target, 1), 1.0)
        eta = eta_min + 0.5 * (eta_max - eta_min) * (1.0 + np.cos(np.pi * progress))
        temp = temp_min + (temp_init - temp_min) * (1.0 + np.cos(np.pi * progress))

        # Compute convolution
        b = compute_conv_fft(a)
        b_max = np.max(b)

        # Adaptive: if b is nearly flat, reduce temperature
        b_std = np.std(b)
        if b_std < 1e-8 * b_max:
            temp = max(temp * 0.5, temp_min)

        # Compute weights
        w = softmax_weights(b, temp)

        # Compute gradient via FFT
        g = compute_subgradient_fft(a, w, N_fft)

        # Also add a direct peak component (hardest peak gets extra weight)
        k_star = int(np.argmax(b))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        peak_grad = np.zeros(n)
        for j in range(lo, hi + 1):
            peak_grad[j] = 2.0 * a[k_star - j]
        g = 0.7 * g + 0.3 * peak_grad

        # Center gradient for simplex projection
        g = g - np.mean(g)

        # Try step with backtracking
        success = False
        for frac in [1.0, 0.5, 0.25, 0.1]:
            step = eta * frac
            a_new = simplex_project(a - step * g)
            obj_new = compute_obj_normalized(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                success = True
                break

        if success:
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Diversification: spectral perturbation
                rng_local = np.random.default_rng(iteration + n)
                fa = rfft(a, N_fft)
                # Perturb magnitude
                mag = np.abs(fa)
                phase = fa / np.maximum(mag, 1e-30)
                mag_pert = mag * np.exp(rng_local.normal(0, 0.05, len(mag)))
                fa_new = mag_pert * phase
                a_new = irfft(fa_new, N_fft)[:n]
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj_normalized(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                a = best_a.copy()
                stall = 0
                max_stall = min(max_stall + 100, 500)
                eta = eta_max

        iteration += 1
        if verbose and iteration % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"          SGD it{iteration}: obj={best_obj:.8f}, eta={eta:.4e}, temp={temp:.4f}, t={elapsed:.1f}s")

        if eta < eta_min * 0.5:
            break

    return best_a


def energy_gradient_descent(a, time_budget=60.0, verbose=False):
    """Minimize sum(b^2) - smooths the entire convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)

    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1

    start = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv_fft(a)

        # Energy gradient: g_j = 2 * sum_k b[k] * 2*a[k-j]*[0<=k-j<n]
        # = 4 * cross-correlation(b, a)
        b_rev = b[::-1]
        fb_rev = rfft(b_rev, N_fft)
        fa = rfft(a, N_fft)
        conv_full = irfft(fb_rev * fa, N_fft)[:3 * n - 2]
        g = 4.0 * conv_full[n - 1:2 * n - 1][::-1]

        g = g - np.mean(g)

        v = momentum * v - eta * g
        a_new = simplex_project(a + v)
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.002, 5.0 / np.sqrt(n))
        else:
            stall += 1
            eta *= 0.95
            v *= 0.5
            if stall > 200:
                a = best_a.copy()
                v = np.zeros(n)
                eta = 1.0 / np.sqrt(n)
                momentum = 0.7
                stall = 0
            if eta < 1e-8:
                break

        if verbose and it % 5000 == 0:
            print(f"          EnergyGD it{it}: obj={best_obj:.8f}, eta={eta:.4e}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, verbose=False):
    """
    Solve: min M s.t. conv(a, g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns optimal g (normalized) or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Try cvxpy first (more robust for larger problems)
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def lp_block_coordinate(a, time_budget=60.0, verbose=False):
    """
    LP-based block coordinate descent.
    Alternates: fix a, optimize g via LP; then a = (1-t)*a + t*g.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0

    # Adaptive mixing parameter
    t_base = 0.5

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp(a, verbose=verbose)
        if g_lp is None:
            break

        # Line search for mixing parameter
        best_t_obj = np.inf
        best_t = 0.3
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 15:
                break

    return best_a


def peak_chase(a, time_budget=20.0):
    """Direct subgradient chase on the maximum peak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    eta = 0.5 / np.sqrt(n)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv_fft(a)
        k_star = int(np.argmax(b))

        # Subgradient
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            g[j] = 2.0 * a[k_star - j]

        g = g - np.mean(g)

        a_new = simplex_project(a - eta * g)
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 2.0 / np.sqrt(n))
        else:
            eta *= 0.93
            if eta < 1e-10:
                break

    return best_a


def spectral_diversify(a, rng, strength=0.1):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    N_fft = 1
    while N_fft >= n:
        N_fft <<= 1

    fa = rfft(a, N_fft)
    mag = np.abs(fa)
    phase = fa / np.maximum(mag, 1e-30)
    mag_pert = mag * np.exp(rng.normal(0, strength, len(mag)))
    fa_new = mag_pert * phase
    a_new = irfft(fa_new, N_fft)[:n]
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def random_init(n, rng):
    """Generate a random initial sequence."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n // 8) ** 2) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n // 8) ** 2)
        a = a / np.sum(a)
    elif mode == 4:
        # Triangle (peaked in middle)
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0 - np.abs(x - 0.5) * 2 + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x ** 0.7
        a = a / np.sum(a)
    elif mode == 6:
        # Sine-modulated
        x = np.arange(n) / n
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x) + 0.1
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        # Gaussian mixture
        centers = rng.uniform(0.1, 0.9, 2) * n
        sigmas = rng.uniform(n // 20, n // 5, 2)
        a = np.zeros(n)
        for c, s in zip(centers, sigmas):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * s ** 2))
        a = a / np.sum(a)
    return a

    try:

        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n = len(best_seq)
    print(f"Starting optimization with n={n}, obj={best_score:.8f}")

    print(f"\n=== Phase 1: Energy GD (n={n}) ===")
    remaining = deadline - time.monotonic()

        alloc = min(remaining * 0.15, 120.0)
        a_eng = energy_gradient_descent(best_seq, time_budget=alloc, verbose=True)
        obj_eng = compute_obj_normalized(a_eng)

            best_score = obj_eng
            best_seq = a_eng.copy()

        a_sgd = subgradient_optimize(best_seq, time_budget=alloc, verbose=True)

            best_seq = a_sgd.copy()

        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
