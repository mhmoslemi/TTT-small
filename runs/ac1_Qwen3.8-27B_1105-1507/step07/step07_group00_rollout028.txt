"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with adaptive temperature

4. Coordinate descent polishing

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, w):
    """Compute gradient of log-sum-exp of conv(a,a). g_i = 2*sum_j a_j * w_{j+i}."""
    n = len(a)
    idx = np.arange(n)
    M = w[idx[:, None] + idx[None, :]]
    return 2.0 * (M @ a)


def lse_optimize(a, time_budget=300.0, T_init=20.0, T_max=5000.0, verbose=False):
    """
    Log-sum-exp smoothed gradient descent on the simplex.
    Adaptive temperature and learning rate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 2.0 / n
    min_lr = 1e-7 / n
    stall = 0
    patience = 300

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = np.max(conv)

        # Compute softmax weights
        shifted = T * (conv - max_conv)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Gradient
        g = lse_gradient(a, w)

        # Projected gradient step
        a_new = a - lr * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Gradually increase temperature for sharper focus
            T = min(T_max, T * 1.02)
            lr = max(min_lr, lr * 0.9995)
        else:
            stall += 1
            if stall > patience:
                # Escape local minimum
                rng = np.random.default_rng(it * 7919 + n)
                scale = min(0.3, 0.03 + 0.02 * (stall // patience))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(min_lr, lr * 0.5)
                T = max(T_init, T * 0.3)
                stall = 0
            else:
                lr = max(min_lr, lr * 0.97)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    LSE it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.2e}, elapsed={elapsed:.1f}s")

        it += 1

    return best_a


def subgradient_topK(a, K=5):
    """Compute subgradient using top-K peaks with softmax weighting."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n)
    temp = 5.0 * max_val
    weights = np.exp((peak_vals - max_val) / temp)
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]
    return g


def multi_peak_subgradient(a, time_budget=200.0, verbose=False):
    """
    Multi-peak subgradient with Barzilai-Borwein step size and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    step_size = 1.0 / n
    momentum = 0.7
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    stall = 0
    patience = 300
    K = 5

    it = 0
    while time.monotonic() - start < time_budget:
        g = subgradient_topK(a, K=K)

        # BB step size
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-15:
                bb = ss / abs(sy)
                step_size = min(max(1e-6/n, step_size * 0.5 + bb * 0.5), 5.0/n)

        v = momentum * v + g
        a_new = a - step_size * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            prev_a = a.copy()
            a = a_new.copy()
            stall = 0
        else:
            # Backtrack
            a_new2 = a - step_size * 0.3 * v
            a_new2 = project_simplex(a_new2)
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_obj - 1e-13:
                best_obj = obj_new2
                best_a = a_new2.copy()
                prev_a = a.copy()
                a = a_new2.copy()
                stall = 0
            else:
                step_size *= 0.7
                stall += 1
                if stall > patience:
                    rng = np.random.default_rng(it * 31 + n)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    v = np.zeros(n)
                    step_size = 1.0 / n
                    momentum = 0.5
                    stall = 0

        prev_g = g.copy()
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    MPS it{it}: obj={best_obj:.8f}, step={step_size:.2e}, elapsed={elapsed:.1f}s")
        it += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def coordinate_refine(a, time_budget=50.0):
    """
    Paired coordinate descent: move mass between positions to reduce peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

    if s < 1e-10:

    start = time.monotonic()
    conv = compute_conv(a)

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False

        # Find top contributing pairs to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                contributors.append((contrib, i, j))

        contributors.sort(reverse=True)

        # Non-contributors
        contrib_set = set()
        for _, i, j in contributors:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [x for x in range(n) if x not in contrib_set]

        if not non_contrib:
            # Try reducing within contributors
            non_contrib = [contributors[-1][1]]

        for contrib, i, j in contributors[:30]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break
            for target in non_contrib[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                    delta = frac * a[i]
                    if a[i] - delta < -1e-15:
                        continue
                    old_i = a[i]
                    old_t = a[target]
                    a[i] = old_i - delta
                    a[target] = old_t + delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] = old_i
                        a[target] = old_t
                        break

        if not improved:
            # Try symmetric move: reduce both i and j, add to target
            for contrib, i, j in contributors[:10]:
                if time.monotonic() - start > time_budget:
                    break
                for target in non_contrib[:5]:
                    delta = 0.05 * min(a[i], a[j])
                    if delta < 1e-12:
                        continue
                    old_i, old_j, old_t = a[i], a[j], a[target]

                        best_a = a.copy()

                        break
                    else:

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # U-shaped
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.1))
        a = a / a.sum()
    elif mode == 3:
        # Triangular inverted
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x) * 0.5
        a = a / a.sum()
    elif mode == 4:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = np.cos(4 * np.pi * x)**2 + 0.3
        a = a / a.sum()
    elif mode == 5:
        # Smooth random
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Tail-heavy
        x = np.linspace(0, 1, n)
        a = x**3 + (1-x)**3 + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Periodic modulation
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            best_score = obj_mp
            best_seq = a_mp.copy()
            print(f"  MPS: obj={best_score:.8f} **")
        else:
            print(f"  MPS: obj={obj_mp:.8f} (no improvement)")

    # === Phase 3: LP Alternating (main workhorse) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: LP Alternating ===")
        # Multiple rounds with increasing intensity
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(remaining * 0.3, 200.0)
            a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Coordinate Refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Coordinate Refinement ===")
        cd_budget = min(remaining * 0.1, 60.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Multi-n Exploration ===")
        n_current = len(best_seq)

        n_options = sorted(set([
            n_current - 100, n_current - 50,
            n_current,
            n_current + 50, n_current + 100, n_current + 200, n_current + 400,
            500, 700, 1000, 1200, 1500
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_current]

        for n_try in n_options[:6]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break

            a_try = resample_sequence(best_seq, n_try)
            alloc = min(remaining * 0.15, 80.0)
            a_opt = optimize_at_n(a_try, n_total_budget=alloc)
            obj_opt = compute_obj(a_opt)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Diverse Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Diverse Restarts ===")
        n_best = len(best_seq)
        n_restarts = min(6, int(remaining // 80) + 1)

        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break

            mode = (r * 3) % 8
            a_r = diverse_init(n_best, rng, mode=mode)

            # Blend with best
            blend = 0.4
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()

            restart_budget = min(remaining * 0.15, 80.0)
            a_opt = optimize_at_n(a_r, n_total_budget=restart_budget)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # LSE with high temperature
        lse_alloc = min(remaining * 0.25, 100.0)
        a_lse = lse_optimize(best_seq, time_budget=lse_alloc, T_init=50.0, T_max=5000.0, verbose=True)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  Final LSE: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP polish
        lp_alloc = min(remaining * 0.3, 150.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Multi-peak
        mp_alloc = min(remaining * 0.2, 60.0)
        a_mp = multi_peak_subgradient(best_seq, time_budget=mp_alloc)
        obj_mp = compute_obj(a_mp)
        if obj_mp < best_score:
            best_score = obj_mp
            best_seq = a_mp.copy()
            print(f"  Final MPS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Coordinate descent
        cd_alloc = min(remaining * 0.3, 40.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more LP round
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 60.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
