def compute_conv(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def softmax_gradient_descent(a, time_budget=120.0, verbose=False):

    best_obj = compute_obj(a)
    start = time.monotonic()

    # Temperature for softmax (lower = sharper peak focus)
    temp = 0.1
    eta = 0.1  # step size for multiplicative updates
    N = 1
    while N < 3 * n - 1:
        N <<= 1

    max_stall = 100

        if time.monotonic() - start > time_budget:
            break

            w = w / w_sum

            best_obj = obj_new

            a = a_new

def multiplicative_peak_chase(a, time_budget=30.0):

        N <<= 1

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        b = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(b))

        # Subgradient at peak: grad[i] = a[k_star - i] if 0 <= k_star-i < n, else 0
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            grad[i] = a[k_star - i]

        # Center
        grad = grad - np.mean(grad)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.5)
        else:
            eta *= 0.97
            if eta < 1e-8:
                break

    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
