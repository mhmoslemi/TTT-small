def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)

    n = len(a)

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated log-sum-exp gradient descent, LP alternating,
and multi-scale exploration to find near-optimal non-negative sequences.
Key novelty: FFT-based cross-correlation trick for O(n log n) gradient computation
of the smooth max approximation, combined with adaptive temperature scheduling.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_smooth_max(conv, beta):
    """Compute LSE smooth approximation of max and softmax weights."""
    shifted = beta * conv - np.max(beta * conv)
    weights = np.exp(shifted)
    total = weights.sum()
    weights /= total
    lse_val = (np.max(beta * conv) + np.log(total)) / beta
    return lse_val, weights


def lse_gradient_fft(a, beta):
    """
    Compute gradient of LSE(beta * conv(a,a)) using FFT cross-correlation trick.
    grad[i] = 2 * sum_j weights[i+j] * a[j]
    Computed via: 2 * conv(weights, a_flip)[n-1 : 2n-1]
    """
    n = len(a)
    conv = compute_conv(a)
    _, weights = lse_smooth_max(conv, beta)

    a_flip = a[::-1].copy()

    size = 1
    while size < (2 * n - 1) + n - 1 + 1:
        size <<= 1

    fw = rfft(weights, size)
    fa = rfft(a_flip, size)
    corr_full = irfft(fw * fa, size)

    grad = 2.0 * corr_full[n - 1:n - 1 + n]
    return grad, conv


def subgradient_topk(a, K=5):
    """Compute subgradient using top-K peaks."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n), conv

    # Softmax weighting over top peaks
    temp = max_val / (K * 2.0)
    weights = np.exp((peak_vals - max_val) / temp)
    weights /= weights.sum()

    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            grad[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

    # Scale to match LSE gradient magnitude
    grad *= 2.0
    return grad, conv


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient_descent(a, time_budget=100.0, beta_start=10.0, beta_end=5000.0,
                            n_iter_max=50000, verbose=False):
    """
    Smooth gradient descent using LSE approximation with adaptive beta.
    Uses Nesterov momentum and backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    momentum = 0.9
    v = np.zeros(n)

    # Adaptive beta schedule
    beta = beta_start
    total_iters = max(n_iter_max, int(time_budget / 0.01))
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    SGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, elapsed={elapsed:.1f}s")

        # Anneal beta
        progress = min(1.0, it / (total_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = min(beta, beta_end)

        # Compute gradient
        grad, conv = lse_gradient_fft(a, beta)

        # Nesterov look-ahead
        a_look = a + momentum * v
        grad_look, _ = lse_gradient_fft(a_look, beta)

        # Update
        v = momentum * v - grad_look
        a_new = a + v

        # Project to simplex
        a_new = project_simplex(a_new)

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Reset momentum, take smaller step
                v = np.zeros(n)
                momentum = max(0.5, momentum * 0.9)
                a = best_a.copy()
                best_obj = compute_obj(a)
                stall = 0
            elif stall > 50:
                # Try with reduced gradient
                a_try = a - 0.5 * grad / (np.linalg.norm(grad) + 1e-10)
                a_try = project_simplex(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    stall = 0

        it += 1

    return best_a

def build_conv_matrix_sparse(a):
    """Build A matrix (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_max_sum(a, M_target):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target for all k, g >= 0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_sparse(a)

    # Use linprog (HiGHS)
    try:
        c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        A_ub = A
        b_ub = np.full(2 * n - 1, M_target)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    # Fallback: try cvxpy with ECOs
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M_target]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def solve_lp_min_peak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_sparse(a)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    # Fallback: bisection with max-sum LP
    conv_a = compute_conv(a)
    M_lo = float(np.max(conv_a)) * 0.5
    M_hi = float(np.max(conv_a)) * 1.5

    for _ in range(20):
        M_mid = (M_lo + M_hi) / 2
        g = solve_lp_max_sum(a, M_mid)
        if g is not None and np.sum(g) > 0.99:
            M_hi = M_mid
        else:
            M_lo = M_mid

    g = solve_lp_max_sum(a, M_hi)
    if g is not None:
        return g, M_hi

    return None, None

def mirror_descent(a, time_budget=100.0, topK=3, verbose=False):

    """

    best_a = a.copy()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)

        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - start

        g, conv = subgradient_topk(a, K=topK)

        v = momentum * v + g
        # Multiplicative update (mirror descent)
        log_a = np.log(a)
        log_a_new = log_a - lr * v
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = max(1e-6, lr * 0.998)

        else:
            stall += 1

                v = np.zeros(n)

            else:

def coordinate_refine(a, time_budget=30.0):

    if s < 1e-10:
        return a

        improved = False

            break

        for contrib, i, j in contributors[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break
            for target in non_contrib[:5]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if delta < 1e-12:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try symmetric coordinate moves
            for _ in range(10):
                if time.monotonic() - start > time_budget:
                    break
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                for delta in [0.001, 0.003, 0.005]:
                    if a[i] < delta:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] += delta
                        a[j] -= delta
                    if improved:
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Inverted triangular
        xn = x / (n - 1)
        a = 1.0 - 2.0 * np.abs(xn - 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # U-shaped (high at edges)
        xn = x / (n - 1)
        a = 1.0 / (np.sqrt(0.01 + (xn - 0.05) * (xn - 0.95)) + 0.1)
        a = a / a.sum()
    elif mode == 5:
        # Smooth random
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amplitudes = rng.uniform(0.1, 0.4, size=4)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amplitudes):
            a += amp * np.sin(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Quadratic (high at edges)
        xn = x / (n - 1)
        a = (xn - 0.5) ** 2 + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Sine wave
        xn = x / (n - 1)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * xn)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid of smooth gradient descent, LP alternating,
    mirror descent, and coordinate refinement with multi-scale exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(n_target, rng, mode=6)  # Edge-heavy
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Smooth Gradient Descent (new approach) ===
    print(f"\n=== Phase 1: Smooth Gradient Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    sgd_budget = min(remaining * 0.15, 150.0)
    if sgd_budget > 20:
        a_sgd = smooth_gradient_descent(best_seq, time_budget=sgd_budget,
                                        beta_start=5.0, beta_end=3000.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  SGD: obj={best_score:.8f} **")
        else:
            print(f"  SGD: obj={obj_sgd:.8f} (no improvement)")

    # === Phase 2: Mirror Descent ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        md_budget = min(remaining * 0.12, 120.0)
        a_md = mirror_descent(best_seq, time_budget=md_budget, topK=3, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 3: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(remaining * 0.35, 400.0)

    if lp_budget > 30:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            alloc = min(lp_budget / (4 - rnd), remaining * 0.4)
            if alloc < 20:
                break
            a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")
                if rnd > 1:
                    break

    # === Phase 4: Chase Peak Refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        cp_budget = min(remaining * 0.08, 60.0)
        a_cp = topk_subgradient_descent(best_seq, time_budget=cp_budget, K=5, verbose=True)
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  TSGD: obj={best_score:.8f} **")
        else:
            print(f"  TSGD: obj={obj_cp:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100, n_current - 50,
        n_current + 50, n_current + 100, n_current + 200,
        500, 600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and n != n_current]
    n_options = n_options[:6]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: smooth GD + LP
        half_budget = min(remaining * 0.15, 80.0)
        third = half_budget / 3

        if third > 10:
            a_try = smooth_gradient_descent(a_try, time_budget=third,
                                            beta_start=5.0, beta_end=2000.0)

        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.1, 50.0)
        if lp_alloc > 15:
            a_try = lp_alternating(a_try, time_budget=lp_alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3 + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)

        # Blend with best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.15, 70.0)
        # Use LP as primary
        a_opt = lp_alternating(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Smooth GD polish with high beta
    if remaining > 20:
        md_alloc = min(remaining * 0.15, 50.0)
        a_md = smooth_gradient_descent(best_seq, time_budget=md_alloc,
                                       beta_start=50.0, beta_end=10000.0)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        lp_alloc = min(remaining * 0.2, 80.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Coordinate refinement
    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_alloc = min(remaining * 0.15, 30.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.3, 40.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
