def smoothed_obj_and_grad_x(x, tau):
    """
    Smoothed objective and gradient for log-parametrization.
    a = exp(x - max(x)), objective = 2*n*max(conv(a,a)) (a is normalized to sum 1).
    
    Since the objective is scale-invariant, we work with a = exp(x)/sum(exp(x)).
    """
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    a_norm = a / s  # sum to 1
    
    conv = compute_conv(a_norm)
    
    # Log-sum-exp smoothing
    c = conv / tau
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp  # softmax weights
    
    logsumexp_val = c_max + np.log(sum_exp)
    F = 2.0 * n * tau * logsumexp_val  # This is 2*n*T*logsumexp(conv/T) ≈ 2*n*max(conv)
    
    # Gradient w.r.t. a_norm:
    # F = 2*n*T*logsumexp(conv(a_norm)/T) where a_norm = a/s, s=sum(a)
    # But a_norm is already sum-1, so conv depends on a_norm components.
    # dF/da_norm[i] = 2*n*sum_k w[k] * d(conv[k])/da_norm[i]
    # = 2*n*sum_k w[k] * 2*a_norm[k-i] * 1{0<=k-i<n}
    # = 4*n * (corr_w)[i]
    # where corr_w[i] = sum_k w[k]*a_norm[k-i]*1{0<=k-i<n}
    # = sum_k w[k]*a_norm[k-i] (with valid indices)
    # This is the cross-correlation of w with a_norm at lag i.
    
    # Compute corr_w[i] = sum_k w[k]*a_norm[k-i] for 0<=k-i<n
    # = sum_j a_norm[j]*w[j+i] for 0<=j+i<2n-1
    # This is (a_norm * w_reversed)[?] ... let me use FFT.
    
    # corr_w[i] = sum_j a_norm[j] * w[j+i]
    # This is the cross-correlation of a_norm with w.
    # Using FFT: ifft(conj(FFT(a_norm)) * FFT(w)) gives the circular correlation.
    # For linear, we zero-pad.
    
    size = next_pow2(2 * n - 1)
    fa = rfft(a_norm, size)
    fw = rfft(w, size)
    # Linear cross-correlation: ifft(conj(fa) * fw) gives sum_j a[j]*w[j+k] for each k
    # Actually, ifft(conj(fft(a)) * fft(b)) = circular correlation of a and b
    # For linear, zero-pad to size >= n + len(w) - 1
    # w has length 2n-1, a has length n, so we need size >= n + 2n-1-1 = 3n-2
    # Let's use a bigger size.
    
    # Actually, let me just compute it directly using the convolution relationship.
    # corr_w[i] = sum_j a_norm[j] * w[j+i]
    # Let's reverse w: w_rev[k] = w[len(w)-1-k]
    # Then corr_w[i] = sum_j a_norm[j] * w[j+i] = conv(a_norm, w_rev)[?]
    # conv(a_norm, w_rev)[i + (2n-2)] = sum_j a_norm[j] * w_rev[i + 2n-2 - j]
    # = sum_j a_norm[j] * w[(2n-1-1) - (i+2n-2-j)]
    # = sum_j a_norm[j] * w[j-i]
    # Hmm, that gives w[j-i] not w[j+i].
    
    # Let me just use: corr[i] = sum_k w[k] * a_norm[k-i] for valid k-i
    # = sum_k w[k] * a_norm[(k-i) mod ...] no, linear.
    # = sum_{k=max(i,0)}^{min(2n-2, i+n-1)} w[k] * a_norm[k-i]
    
    # This is the cross-correlation. Let me compute it via FFT properly.
    # corr[i] = sum_k w[k]*a[k-i] = (w * a_flipped)[2n-2-i] ... 
    # Actually, let's just use np.correlate or compute via the convolution we already have.
    
    # Simpler: since we need sum_k w[k]*a_norm[k-i] for i=0..n-1
    # and w has length 2n-1, a has length n
    # This is a "sliding dot product" which can be done via FFT.
    
    # Let me just do it with a loop for correctness (n is at most ~2000):
    # Actually for n=800, a loop of 800 iterations each doing O(n) work is O(n^2) = 640K ops, fine.
    
    # Better: use the fact that it's a convolution with a reversed sequence.
    # sum_k w[k]*a[k-i] = conv(w, a_flipped)[(2n-2)-i+(n-1)] ... 
    
    # You know what, let me just compute it directly:
    # For each i in 0..n-1: grad_a[i] = sum_{k=i}^{i+n-1} w[k] * a_norm[k-i]
    # (since k-i must be in 0..n-1, so k is in i..i+n-1, and k must be in 0..2n-2)
    
    # This is O(n^2) which is fine for n up to a few thousand.
    # But let's use FFT for efficiency.
    
    # Actually, the cleanest way: 
    # grad_a[i] = sum_k w[k]*a[k-i]*1{0<=k-i<n}
    # Let b = a (length n), and we want to compute for each i: sum_k w[k]*b[k-i]
    # where the sum is over k such that 0<=k<=2n-2 and 0<=k-i<=n-1.
    # This equals: (w * b)[?] using appropriate slicing.
    # w * b (convolution) has length (2n-1) + n - 1 = 3n-2.
    # (w*b)[m] = sum_k w[k]*b[m-k]
    # We want sum_k w[k]*b[k-i] = sum_j w[j+i]*b[j] (substituting j=k-i)
    # = (b * w)[i] if we define (b*w)[i] = sum_j b[j]*w[i-j]... 
    
    # I'm overcomplicating this. Let me just use the direct computation.
    # For n=800, it's fast enough.
    
    # Actually, the fastest way with numpy:
    # corr[i] = sum_{j=0}^{n-1} a[j] * w[j+i]
    # This is the cross-correlation of a and w at lag i.
    # np.correlate(w, a, 'full')[n-1:n-1+n] would give this... let me check.
    # np.correlate(w, a, 'full')[k] = sum_j w[j+k-(n-1)] * a[j] ... this is confusing.
    
    # Let me just use the FFT approach that I know works:
    # To compute c[i] = sum_j a[j]*w[j+i] for i=0..n-1:
    # Zero-pad a to length L >= n + (2n-1) - 1 = 3n-2
    # c = ifft(conj(fft(a, L)) * fft(w, L))[:n]
    # Wait, I need to be careful about conventions.
    
    # scipy.fft: ifft(fft(a)*fft(b)) = circular convolution of a and b (zero-padded)
    # correlate(a, b) = convolve(a, flip(b))
    # So: sum_j a[j]*w[j+i] = convolve(w, flip(a))[i + n - 1] ... 
    
    # OK let me just do it the simple way for correctness:
    corr = np.zeros(n)
    for i in range(n):
        # corr[i] = sum_{j=0}^{n-1} a_norm[j] * w[j+i]
        # j+i ranges from i to i+n-1, which must be < 2n-1 (length of w)
        # Since i < n and j < n, j+i < 2n, so j+i <= 2n-1, which is valid index for w (length 2n-1)
        corr[i] = np.dot(a_norm, w[i:i+n])
    
    grad_a = 4.0 * n * corr  # dF/da_norm[i] (for the normalized a)
    
    # Now chain rule: a_norm = a/s, so da_norm[i]/da[i] = 1/s - a[i]/s^2
    # dx/d(da[i]) = 1/a[i] (since a = exp(x), da/dx = a, so dx/da = 1/a)
    # dF/dx[i] = dF/da_norm[i] * da_norm[i]/da[i] * da[i]/dx[i]
    # = dF/da_norm[i] * (1/s - a[i]/s^2) * a[i]
    # = dF/da_norm[i] * (a[i]/s - a[i]^2/s^2)
    # = dF/da_norm[i] * a_norm[i] * (1 - a_norm[i])
    
    # Wait, but we also need to account for the fact that a_norm depends on ALL a_j through s.
    # F depends on a_norm = a/s.
    # dF/dx[i] = sum_j (dF/da_norm[j]) * (da_norm[j]/dx[i])
    # da_norm[j]/da[j] = 1/s - a[j]/s^2 = (s - a[j])/s^2 = (1-a_norm[j])/s
    # da_norm[j]/dx[i] = 0 for j != i (since a[j] doesn't depend on x[i])
    # da_norm[i]/dx[i] = (1-a_norm[i])/s * a[i] = a_norm[i]*(1-a_norm[i])/s * s/a[i] * a[i]
    # Wait: a[i] = exp(x[i]), da[i]/dx[i] = a[i]
    # da_norm[i]/dx[i] = (1-a_norm[i])/s * da[i]/dx[i] = (1-a_norm[i])/s * a[i] = a_norm[i]*(1-a_norm[i])
    
    # Hmm wait: da_norm[j]/dx[i] = 0 for j ≠ i? No!
    # a_norm[j] = a[j]/s where s = sum(a[k])
    # da_norm[j]/dx[i] = (da[j]/dx[i]*s - a[j]*ds/dx[i])/s^2
    # = (δ_{ji}*a[i]*s - a[j]*a[i])/s^2
    # = a[i]*(δ_{ji}*s - a[j])/s^2
    # = a[i]*s*(δ_{ji} - a_norm[j])/s^2
    # = a_norm[i]*(δ_{ji} - a_norm[j])
    
    # So dF/dx[i] = sum_j dF/da_norm[j] * a_norm[i]*(δ_{ji} - a_norm[j])
    # = a_norm[i] * (dF/da_norm[i] - sum_j dF/da_norm[j]*a_norm[j])
    
    # Let C = sum_j dF/da_norm[j]*a_norm[j]
    # Then dF/dx[i] = a_norm[i]*(dF/da_norm[i] - C)
    
    C = np.dot(grad_a, a_norm)
    grad_x = a_norm * (grad_a - C)
    
    return F, grad_x


def gradient_optimize_v2(a_init, time_budget=60.0, verbose=False, seed=42):
    """
    Improved gradient optimization with:
    - Longer temperature schedule
    - Adam-like adaptive learning (via scipy's L-BFGS-B with more iterations)
    - Better convergence criteria
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Extended temperature schedule
    base_tau = 1.0 / n
    log_temps = np.linspace(np.log(20.0 * base_tau), np.log(0.05 * base_tau), 15)
    
    x = np.log(a) - np.log(a).max()
    
    for ti, tau in enumerate(log_temps):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad_x(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad_x(xv, tau)
            return g
        
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 100})
        
        if res.x is not None:
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.6f} (step {ti}): obj={best_obj:.10f}")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a  # A[k, j] = a[k-j] if 0 <= k-j < n
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    a is assumed normalized (sum=1).
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    
    # Try cvxpy first
    try:
        A = build_conv_matrix(a)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass
    
    # Fallback: scipy linprog
    try:
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_step(a, t=0.5):
    """One LP alternating step with blending factor t."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_lp, M_lp = solve_lp_minpeak(a_norm)
    if g_lp is None:
        return a, compute_obj(a)
    
    a_new = (1.0 - t) * a_norm + t * g_lp
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a, compute_obj(a)
    a_new = a_new / s
    return a_new, compute_obj(a_new)


def lp_optimize_v2(a_init, time_budget=60.0, verbose=False):
    """LP alternating with adaptive blending and anti-stalling."""
    n = len(a_init)
    a = a_init / np.sum(a_init)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    # Adaptive blending
    t = 0.5
    
    while time.monotonic() - start < time_budget:
        a_new, obj_new = lp_step(a, t)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            t = min(t * 1.1, 0.9)  # Increase blending when making progress
        else:
            stall += 1
            if stall % 5 == 0:
                # Try different blend factors
                for t_try in [0.1, 0.2, 0.3, 0.7, 0.8, 0.95]:
                    a_try, obj_try = lp_step(best_a, t_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        t = t_try
                        stall = 0
                        break
                else:
                    # Perturb
                    rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                    scale = 0.005 + 0.005 * min(stall // 20, 3)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 1e-10)
                    a = a / np.sum(a)
                    stall = 0
                    t = 0.3
    
    return best_a


def peak_deflection(a, n_peaks=5, n_samples=50, eta=0.01):
    """
    Targeted peak deflection: for each of the top peaks,
    apply small random perturbations that reduce that peak.
    """
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    conv = compute_conv(a)
    top_indices = np.argsort(conv)[-n_peaks:]
    
    for peak_idx in top_indices:
        for _ in range(n_samples):
            # Random perturbation biased to reduce the peak at peak_idx
            # d(conv[peak_idx])/da[i] = 2*a[peak_idx-i] (if valid)
            # To reduce conv[peak_idx], decrease a[i] where a[peak_idx-i] is large
            perturbation = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    perturbation[i] = -a[j]
                else:
                    perturbation[i] = 0
            
            # Add some random exploration
            rng = np.random.default_rng(peak_idx * 1000 + _ )
            perturbation += 0.1 * rng.normal(0, 1, n)
            
            # Apply perturbation
            a_try = a * np.exp(eta * perturbation)
            a_try = np.maximum(a_try, 1e-30)
            a_try = a_try / np.sum(a_try)
            
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
    
    return best_a


def spectral_init(n, mode=0, seed=42):
    """
    Create initial sequences using spectral shaping.
    The idea: shape the power spectrum to be flat, then take IFFT and rectify.
    """
    rng = np.random.default_rng(seed)
    
    if mode == 0:
        # Flat spectrum (white noise in frequency → sinc-like in time)
        A = np.ones(n)
        a = np.real(ifft(A))
    elif mode == 1:
        # Raised cosine spectrum
        f = np.fft.fftfreq(n)
        A = np.cos(np.pi * f * n / 2) ** 2
        A = np.maximum(A, 0.1)
        a = np.real(ifft(A * np.ones(n)))  # Actually need to construct properly
        # Better: random phase with shaped magnitude
        phase = rng.uniform(0, 2*np.pi, n)
        A = A * np.exp(1j * phase)
        a = np.real(ifft(A))
    elif mode == 2:
        # Water-filling: minimize peak of |IFFT(A)|^2
        # Start with flat, iterate
        A = np.ones(n) * np.exp(1j * rng.uniform(0, 2*np.pi, n))
        for _ in range(100):
            a = np.real(ifft(A))
            a_abs = np.abs(a)
            a_max = np.max(a_abs)
            if a_max < 1e-6:
                break
            # Water-filling: reduce elements where |a| is large
            A = A * np.exp(-0.5 * np.fft.fft(a * np.sign(a) * (a_abs > 0.9*a_max)))
            A = A / np.abs(A) * np.abs(A)  # Keep magnitude, just phase... 
            # Actually this is getting complicated. Let's just do iterative clipping.
            a_clip = np.clip(a, -3, 3)
            A = np.fft.fft(a_clip)
        a = np.real(ifft(A))
    elif mode == 3:
        # Gaussian window (in time domain)
        x = np.linspace(-1, 1, n)
        a = np.exp(-4 * x**2)
    elif mode == 4:
        # Two-peak with phase
        c1, c2 = n//3, 2*n//3
        sigma1 = n/8
        sigma2 = n/8
        a = np.exp(-(np.arange(n) - c1)**2 / (2*sigma1**2)) + \
            0.8 * np.exp(-(np.arange(n) - c2)**2 / (2*sigma2**2))
    elif mode == 5:
        # Random with low-pass
        a = rng.uniform(0, 1, n)
        # Low-pass filter
        fa = fft(a)
        cutoff = n // 4
        fa[cutoff:] = 0
        fa[:cutoff-1][::-1] = 0  # Keep symmetric
        a = np.real(ifft(fa))
    else:
        a = rng.uniform(0, 1, n)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
