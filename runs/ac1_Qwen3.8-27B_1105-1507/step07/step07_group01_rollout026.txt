```python
"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary methods: LP alternating with quadratic minimax step selection,
Adam on softmax parameterization with adaptive temperature,
pairwise swap coordinate descent, and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_and_weights(conv, T):
    """Compute smooth max value and softmax weights for conv."""
    max_c = np.max(conv)
    shifted = np.clip((conv - max_c) / T, -500, 0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    value = max_c + T * np.log(total)
    return value, weights


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, M_est).
    Uses scipy linprog with HiGHS.
    """
    from scipy.optimize import linprog
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    # LP: min M s.t. A@g - M <= 0 (for all k), sum(g)=1, g>=0, M free
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def quadratic_minimax_step(a, g, c, d, e, n_t=50):
    """
    Find optimal t in [0,1] minimizing max_k [(1-t)^2*c[k] + 2t(1-t)*d[k] + t^2*e[k]].
    Uses dense grid + golden section refinement.
    """
    ts = np.linspace(0, 1, n_t)
    best_t = 0
    best_val = np.inf
    for t in ts:
        val = np.max((1-t)**2 * c + 2*t*(1-t)*d + t**2 * e)
        if val < best_val:
            best_val = val
            best_t = t
    # Refine around best_t with golden section
    if 0 < best_t < 1:
        lo = max(0, best_t - 1.0/n_t)
        hi = min(1, best_t + 1.0/n_t)
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(30):
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            v1 = np.max((1-t1)**2 * c + 2*t1*(1-t1)*d + t1**2 * e)
            v2 = np.max((1-t2)**2 * c + 2*t2*(1-t2)*d + t2**2 * e)
            if v1 < v2:
                hi = t2
            else:
                lo = t1
        t_mid = (lo + hi) / 2
        v_mid = np.max((1-t_mid)**2 * c + 2*t_mid*(1-t_mid)*d + t_mid**2 * e)
        if v_mid < best_val:
            best_t = t_mid
            best_val = v_mid
    return best_t, best_val


def lp_alternating(a, time_budget=100.0, verbose=False):
    """
    LP alternating optimization with quadratic minimax step selection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_t = 0.3

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        c = compute_conv(a)
        d = compute_conv(a * g_lp)  # conv(a, g) via: c[i+j] = sum a[k]*g[i+j-k]
        # Actually need conv(a, g) not (a*g) convolved with itself
        # conv(a, g)[k] = sum_j a[j]*g[k-j]
        # Let me compute this properly
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a, size)
        fg = rfft(g_lp, size)
        d = irfft(fa * fg, size)[:2 * n - 1]
        e = compute_conv(g_lp)

        t_opt, val_opt = quadratic_minimax_step(a, g_lp, c, d, e, n_t=60)

        # Also try smaller t values (the quadratic might not capture everything)
        candidates = [t_opt] + [max(0.01, last_t * 0.5), min(1.0, last_t * 1.3), last_t, 0.1, 0.2, 0.3, 0.5]
        best_t_local = t_opt
        best_obj_local = np.inf

        for t in candidates:
            if t <= 0 or t > 1:
                continue
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_local:
                best_obj_local = obj_try
                best_t_local = t

        if best_obj_local < best_obj - 1e-10:
            best_obj = best_obj_local
            best_a = ((1.0 - best_t_local) * a + best_t_local * g_lp)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a
            last_t = best_t_local
            stall_count = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_local:.4f}")
        else:
            stall_count += 1
            if stall_count > 40:
                rng = np.random.default_rng(it * 7919 + n)
                scale = 0.02 + 0.02 * min(stall_count // 40, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                last_t = 0.3

    return best_a


def adam_softmax_optimize(a_init, n, time_budget=200.0, T_init=0.001, T_min=0.00005, lr_init=0.3):
    """
    Adam optimizer on softmax parameterization of the simplex.
    Minimizes smooth max of conv(a,a).
    """
    a_init = np.maximum(a_init, 1e-30)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    theta = theta - theta[0]  # gauge fixing

    theta_best = theta.copy()
    a_best = a_init.copy()
    best_obj = compute_obj(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_step = 0
    lr = lr_init

    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        T = max(T_min, T_init * (0.9985 ** t_step))

        # Compute a from theta
        max_t = np.max(theta)
        a = np.exp(theta - max_t)
        a = a / np.sum(a)

        conv = compute_conv(a)
        _, weights = smooth_max_and_weights(conv, T)

        # Gradient of smooth_max(conv(a,a)) w.r.t. a
        grad_a = 2.0 * np.correlate(weights, a, mode='valid')

        # Gradient w.r.t. theta: grad_theta_j = a_j * (grad_a_j - dot(grad_a, a))
        dot_ga_a = np.dot(grad_a, a)
        grad_theta = a * (grad_a - dot_ga_a)
        grad_theta[0] = 0.0  # gauge

        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)

        theta_new = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        theta_new[0] = 0.0

        # Convert back
        max_t_new = np.max(theta_new)
        a_new = np.exp(theta_new - max_t_new)
        a_new = a_new / np.sum(a_new)

        true_obj = compute_obj(a_new)

        if true_obj < best_obj - 1e-12:
            best_obj = true_obj
            a_best = a_new.copy()
            theta_best = theta_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                lr *= 0.75
                if lr < 1e-5:
                    break
                theta = theta_best.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t_step = 0
                stall = 0

        theta = theta_new

    return a_best, best_obj


def pairwise_swaps(a, time_budget=100.0, n_swaps_per_sec=5000):
    """
    Coordinate descent via random pairwise swaps with exact 1D minimization.
    For each swap (i,j,eps): minimize max_k (conv[k] + 2*eps*(a[k-i]-a[k-j]))
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    swaps_done = 0

    # Precompute indices for valid k-i and k-j
    # For a given i, d_k = a[k-i] if 0<=k-i<n else 0, for k=0..2n-2
    # This is just a shifted version of a

    while time.monotonic() - start < time_budget:
        batch_size = max(100, int(n_swaps_per_sec * 0.01))  # do ~10ms worth of swaps
        for _ in range(batch_size):
            if time.monotonic() - start > time_budget:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Compute d_k = a[k-i] - a[k-j] for valid indices
            # d[k] = a[k-i] if 0<=k-i<n else 0, minus a[k-j] if 0<=k-j<n else 0
            # For k in [0, 2n-2]:
            #   a[k-i] is valid when 0 <= k-i <= n-1, i.e., i <= k <= i+n-1
            #   a[k-j] is valid when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1

            # Efficient computation: d is the difference of two shifted copies of a
            d = np.zeros(2 * n - 1)
            # a[k-i] for k=i..i+n-1: d[i:i+n] += a
            d[i:i + n] += a
            # a[k-j] for k=j..j+n-1: d[j:j+n] -= a
            d[j:j + n] -= a

            # Now minimize max_k (conv[k] + 2*eps*d[k]) for eps in [-a[i], a[j]]
            conv = compute_conv(a)
            # f(eps) = max_k (conv[k] + 2*eps*d[k])
            # This is convex in eps. Find minimum via ternary search.
            lo = -a[i]
            hi = a[j]
            if lo >= hi:
                continue

            # Quick check: evaluate at a few points
            # The minimum of max of lines is at a "median-like" point
            # Use ternary search
            for _ in range(25):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                f1 = np.max(conv + 2 * m1 * d)
                f2 = np.max(conv + 2 * m2 * d)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            eps_opt = (lo + hi) / 2
            f_opt = np.max(conv + 2 * eps_opt * d)

            if f_opt < best_obj / (2 * n) - 1e-12:
                a[i] += eps_opt
                a[j] -= eps_opt
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:
                    a = best_a.copy()
                    continue
                a = a / s
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    stall = 0
                else:
                    a = best_a.copy()
                    stall += 1
            else:
                stall += 1

            swaps_done += 1

        if stall > 200:
            # Perturb
            scale = 0.03
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = compute_obj(a)
            best_a = a.copy()
            stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate a random non-negative sequence."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Smooth via random power spectrum
        freq = np.arange(n)
        power = np.exp(-freq / max(1, n // 5)) + rng.uniform(0, 0.5, n)
        a = irfft(power * rfft(rng.normal(0, 1, n), n * 2), n * 2)[:n]
        a = np.abs(a) + 0.01
        a = a / np.sum(a)
    elif mode == 3:
        # Tilted (inspired by n=2 optimal)
        x = np.arange(n) / n
        a = (1 - x) ** 2 + 0.1 * x  # linear tilt
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2 + 0.2
        a = a / np.sum(a)
    elif mode == 5:
        # Two bumps
        c1, c2 = n // 3, 2 * n // 3
        w = max(1, n // 8)
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def propose_candidate(budget_s=1500):
    """
    Main search function. Combines LP alternating, Adam-softmax, pairwise swaps,
    and multi-n exploration to minimize the convolution peak objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(42)

    # Load parent
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.8f}")

    # === Phase 1: LP Alternating on parent n ===
    print(f"\n=== Phase 1: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        phase1_budget = min(350.0, remaining * 0.35)
        print(f"  Budget: {phase1_budget:.0f}s")
        a = best_seq.copy()
        for round_idx in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(phase1_budget / (3 - round_idx), remaining * 0.4)
            a_new = lp_alternating(a, time_budget=alloc, verbose=True)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  LP Round {round_idx}: obj={best_score:.8f}")
            else:
                print(f"  LP Round {round_idx}: obj={obj:.8f} (no improvement)")
                a = best_seq.copy()

    # === Phase 2: Adam on softmax ===
    print(f"\n=== Phase 2: Adam Softmax ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(200.0, remaining * 0.25)
        print(f"  Budget: {alloc:.0f}s")
        a_adam, obj_adam = adam_softmax_optimize(best_seq, len(best_seq), time_budget=alloc,
                                                  T_init=0.002, T_min=0.0001, lr_init=0.5)
        if obj_adam < best_score:
            best_score = obj_adam
            best_seq = a_adam.copy()
            print(f"  Adam: obj={best_score:.8f} **")
        else:
            print(f"  Adam: obj={obj_adam:.8f} (no improvement)")

        # Try with different params
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc2 = min(100.0, remaining * 0.15)
            a_adam2, obj_adam2 = adam_softmax_optimize(best_seq, len(best_seq), time_budget=alloc2,
                                                        T_init=0.005, T_min=0.00005, lr_init=1.0)
            if obj_adam2 < best_score:
                best_score = obj_adam2
                best_seq = a_adam2.copy()
                print(f"  Adam (high T): obj={best_score:.8f} **")

    # === Phase 3: Pairwise Swaps ===
    print(f"\n=== Phase 3: Pairwise Swaps ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.2)
        print(f"  Budget: {alloc:.0f}s")
        a_swaps = pairwise_swaps(best_seq, time_budget=alloc)
        obj_swaps = compute_obj(a_swaps)
        if obj_swaps < best_score:
            best_score = obj_swaps
            best_seq = a_swaps.copy()
            print(f"  Swaps: obj={best_score:.8f} **")
        else:
            print(f"  Swaps: obj={obj_swaps:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best, n_best + 50, n_best + 100,
        n_best + 200, n_best + 300,
        600, 700, 800, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1800]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP
        alloc_lp = min(remaining * 0.15, 25.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        # Adam polish if close
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.02:
            alloc_adam = min(remaining * 0.1, 15.0)
            a_opt2, obj_try2 = adam_softmax_optimize(a_opt, n_try, time_budget=alloc_adam,
                                                      T_init=0.001, T_min=0.00005, lr_init=0.3)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        # Swaps polish
        remaining = deadline - time.monotonic()
        if remaining > 10 and obj_try < best_score + 0.02:
            alloc_sw = min(remaining * 0.08, 10.0)
            a_opt3 = pairwise_swaps(a_opt, time_budget=alloc_sw)
            obj_try3 = compute_obj(a_opt3)
            if obj_try3 < obj_try:
                a_opt = a_opt3
                obj_try = obj_try3

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Random Restarts ===
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        init_mode = r % 6
        n_r = n_best + rng.integers(-100, 151)
        n_r = max(500, min(1500, n_r))
        a_r = random_init(n_r, rng, mode=init_mode)

        alloc_r = min(remaining * 0.1, 20.0)
        a_opt = lp_alternating(a_r, time_budget=alloc_r * 0.6)

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt, obj_r = adam_softmax_optimize(a_opt, n_r, time_budget=min(alloc_r * 0.4, remaining * 0.3),
                                                  T_init=0.002, T_min=0.0001, lr_init=0.5)
        else:
            obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()
        n_final = len(a_final)

        # LP rounds
        for polish_round in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.2, 30.0)
            a_lp = lp_alternating(a_final, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_final = a_lp.copy()
                print(f"  Polish LP {polish_round}: obj={best_score:.8f} **")

        # Adam final
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sm, obj_sm = adam_softmax_optimize(a_final, n_final, time_budget=min(remaining * 0.15, 20.0),
                                                  T_init=0.0005, T_min=0.00002, lr_init=0.2)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Final Adam: obj={best_score:.8f} **")

        # Swaps final
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sw = pairwise_swaps(best_seq, time_budget=min(remaining * 0.15, 15.0))
            obj_sw = compute_obj(a_sw)
            if obj_sw < best_score:
                best_score = obj_sw
                best_seq = a_sw.copy()
                print(f"  Final Swaps: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
