```python
"""
Optimized step function search using ramp initialization and smoothed gradient descent.
Key insight: a linear ramp a[i] ∝ (i+1) gives objective ≈ 4/3, far better than uniform (2.0).
Further refinement via L-BFGS-B on smoothed log-sum-exp objective in log-space.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def smoothed_obj_and_grad(x, tau):
    """
    x: log-space parameters, a = softmax(x)
    Returns (objective_value, gradient_wrt_x)
    """
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    a = a / s  # normalize
    
    conv = compute_conv(a)
    
    # Log-sum-exp of conv/tau
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    logsumexp_val = c_max + np.log(sum_exp)
    
    # Objective: 2n/tau * logsumexp(conv/tau)
    F = 2.0 * n / tau * logsumexp_val
    
    # Weights
    w = exp_c / sum_exp
    
    # Gradient of conv[k] w.r.t. a[i] = a[k-i] (for valid i)
    # Cross-correlation of w with a
    size = next_pow2(2 * n - 1)
    # We need corr[i] = sum_k w[k] * a[k-i] for k-i in [0,n-1]
    # This is the convolution of w and reversed a, evaluated at specific positions
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    # corr[i] = conv_w_af[i + n - 1]
    corr = conv_w_af[n - 1:2 * n - 1]
    
    # dF/da[i] = (2n/tau) * corr[i]
    # dF/dx[j] = a[j] * (dF/da[j] - sum_i a[i]*dF/da[i])
    # sum_i a[i]*dF/da[i] = (2n/tau) * sum_i a[i]*corr[i]
    # But sum_i a[i]*corr[i] = sum_i a[i]*sum_k w[k]*a[k-i] = sum_k w[k]*sum_i a[i]*a[k-i] = sum_k w[k]*conv[k]
    sum_ac = float(np.dot(w, conv))
    dF_da = (2.0 * n / tau) * corr
    dF_dx = a * (dF_da - (2.0 * n / tau) * sum_ac)
    
    return F, dF_dx
```

```python
"""
Optimized step function search using ramp initialization and smoothed gradient descent.
Key insight: a linear ramp a[i] ∝ (i+1) gives objective ≈ 4/3, far better than uniform (2.0).
Further refinement via L-BFGS-B on smoothed log-sum-exp objective in log-space with

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(x, tau):
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    a = a / s

    conv = compute_conv(a)

    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    logsumexp_val = c_max + np.log(sum_exp)

    F = 2.0 * n / tau * logsumexp_val

    w = exp_c / sum_exp

    # = conv(w, flip(a))[n-1+i]

                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = x_opt
                    if verbose:
                        print(f"    tau={tau:.8f}: obj={best_obj:.10f}")
        except Exception:
            pass

    return best_a


def subgradient_optimize(a_init, time_budget=30.0, verbose=False):
    """Direct subgradient on the max convolution value."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.5 / n
    patience = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient of conv[k_star] w.r.t. a[i] = a[k_star - i] for valid i
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        # Project: subtract mean to maintain sum=1
        grad = grad - np.mean(grad)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            eta = min(eta * 1.02, 2.0 / n)
        else:
            patience += 1
            eta *= 0.95
            if patience > 200:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                eta = 0.5 / n
                patience = 0

    return best_a


def make_ramp(n, power=1.0):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    return a / np.sum(a)


def make_asymmetric_ramp(n, skew=1.0):
    """Ramp with adjustable skew."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = t ** (1.0 / (1.0 + skew))
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def make_bimodal(n, ratio=0.3):
    """Two peaks at different positions."""
    i = np.arange(n, dtype=np.float64)
    c1 = int(n * ratio)
    c2 = int(n * (1 - ratio))
    width = n / 8
    a = np.exp(-((i - c1) ** 2) / (2 * width ** 2)) + \
        0.7 * np.exp(-((i - c2) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def main_search(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 10 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    # === Phase 1: Try diverse parametric initializations ===
    print("\n=== Phase 1: Parametric Exploration ===")

    n_values = [500, 600, 700, 800, 900, 1000, 1200]
    init_fns = []
    for p in [0.5, 0.7, 1.0, 1.2, 1.5, 2.0, 3.0]:
        init_fns.append((p, make_ramp))
    for sk in [0.5, 1.0, 2.0, 3.0]:
        init_fns.append((sk, make_asymmetric_ramp))

    for n in n_values:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        for init_id, (param, fn) in enumerate(init_fns):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if fn == make_ramp:
                a = make_ramp(n, power=param)
            else:
                a = make_asymmetric_ramp(n, skew=param)

            obj_init = compute_obj(a)

            # Quick gradient optimization
            alloc = min(10.0, remaining * 0.08)
            a_opt = gradient_optimize(a, time_budget=alloc)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** n={n}, type={fn.__name__}, p={param}: obj={best_score:.10f}")

    # === Phase 2: Deep optimization on best ===
    print(f"\n=== Phase 2: Deep Optimization (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_deep = gradient_optimize(best_seq, time_budget=min(remaining * 0.25, 120.0), verbose=True)
        obj = compute_obj(a_deep)
        if obj < best_score:
            best_score = obj
            best_seq = a_deep.copy()
            print(f"  Deep grad: obj={best_score:.10f}")

        # Subgradient polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_sg = subgradient_optimize(best_seq, time_budget=min(remaining * 0.15, 45.0))
            obj = compute_obj(a_sg)
            if obj < best_score:
                best_score = obj
                best_seq = a_sg.copy()
                print(f"  Subgrad: obj={best_score:.10f}")

    # === Phase 3: N-Exploration around best ===
    print(f"\n=== Phase 3: N-Exploration ===")
    n_best = len(best_seq)
    n_tries = sorted(set([n_best - 100, n_best - 50, n_best - 20, n_best,
                          n_best + 20, n_best + 50, n_best + 100, n_best + 200]))
    n_tries = [n for n in n_tries if 300 <= n <= 2000]

    for n_try in n_tries:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(15.0, remaining * 0.12)
        a_opt = gradient_optimize(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f}")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 4: Multi-start perturbations ===
    print(f"\n=== Phase 4: Multi-Start Perturbations ===")
    n_final = len(best_seq)

    for trial in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if trial == 0:
            a_p = best_seq * np.exp(rng.normal(0, 0.1, n_final))
        elif trial == 1:
            shift = int(rng.integers(-30, 30))
            a_p = np.roll(best_seq, shift)
            a_p = np.maximum(a_p, 1e-10)
        elif trial == 2:
            a_fresh = make_ramp(n_final, power=rng.uniform(0.5, 3.0))
            t = rng.uniform(0.3, 0.7)
            a_p = t * best_seq + (1 - t) * a_fresh
        elif trial == 3:
            a_p = best_seq * (1.0 + 0.3 * rng.normal(0, 1, n_final))
        elif trial == 4:
            # Blend with bimodal
            a_bi = make_bimodal(n_final, ratio=rng.uniform(0.2, 0.4))
            a_p = 0.6 * best_seq + 0.4 * a_bi
        elif trial == 5:
            # Local modification around peak region
            center = n_final // 2
            width = max(10, n_final // 5)
            mask = np.exp(-((np.arange(n_final) - center) ** 2) / (2 * width ** 2))
            a_p = best_seq * (1.0 + 0.2 * rng.normal(0, 1, n_final) * mask)
        elif trial == 6:
            # Inverse ramp blend
            a_inv = make_ramp(n_final, power=1.0)[::-1]
            t = rng.uniform(0.2, 0.5)
            a_p = (1 - t) * best_seq + t * a_inv
        else:
            a_p = best_seq * np.exp(rng.normal(0, 0.05, n_final))

        a_p = np.maximum(a_p, 1e-10)
        a_p = a_p / np.sum(a_p)

        alloc = min(12.0, remaining * 0.08)
        a_opt = gradient_optimize(a_p, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Trial {trial}: obj={best_score:.10f} **")
        else:
            print(f"  Trial {trial}: obj={obj:.10f}")

    # === Phase 5: Final deep polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_f = gradient_optimize(best_seq, time_budget=min(remaining * 0.4, 60.0), verbose=True)
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final grad: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sg = subgradient_optimize(best_seq, time_budget=min(remaining * 0.3, 30.0))
            obj = compute_obj(a_sg)
            if obj < best_score:
                best_score = obj
                best_seq = a_sg.copy()
                print(f"  Final subgrad: obj={best_score:.10f}")

    # Validate
    if best_seq is None:
        best_seq = make_ramp(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result


def propose_candidate():
    """Main entry point: search for best non-negative step function sequence."""
    return main_search(seed=42, budget_s=1500)
```
