def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))

        return np.inf
    a_norm = a / s

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_g(a, A=None):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M, dual_vars)."""
    n = len(a)
    if A is None:
        A = build_conv_matrix(a)
    
    # LP: min M
    # s.t. A@g - M <= 0  (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            g_val = g_val / gs
            M_val = float(res.x[n])
            # Get dual variables (lambda for inequality constraints)
            duals = res.ineqlin.marginals if hasattr(res, 'ineqlin') and res.ineqlin is not None else None
            return g_val, M_val, duals
    return None, None, None


def golden_section_search(a, g, a_norm_fn, f_obj_fn, max_iter=30):
    """Golden section search for t in (0,1) minimizing f((1-t)*a + t*g)."""
    gr = (math.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    
    lo = 0.001
    hi = 0.999
    
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    
    f_c = f_obj_fn((1 - c) * a + c * g)
    f_d = f_obj_fn((1 - d) * a + d * g)
    
    for _ in range(max_iter):
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - gr * (hi - lo)
            f_c = f_obj_fn((1 - c) * a + c * g)
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + gr * (hi - lo)
            f_d = f_obj_fn((1 - d) * a + d * g)
    
    t_best = (lo + hi) / 2.0
    f_best = min(f_c, f_d)
    return t_best, f_best

        lse = float(np.max(conv)) + T * np.log(np.sum(np.exp((conv - np.max(conv)) / T)))
        
        # Gradient: d/d(a_i) [lse] = sum_k p[k] * d(conv[k])/d(a_i) = 2 * sum_k p[k] * a[k-i]
        # Compute h[i] = sum_k p[k] * a[k-i]
        h = np.zeros(n)
        for i in range(n):
            k_min = max(0, i)
            k_max = min(2 * n - 1, i + n - 1)
            for k in range(k_min, k_max + 1):
                j = k - i
                if 0 <= j < n:
                    h[i] += p[k] * a[j]
        
        grad = 2.0 * n * 2.0 * h  # d(2*n*lse)/d(a_i) = 2*n * 2*sum_k p[k]*a[k-i]
        
        # Project gradient to sum=0 hyperplane
        grad_centered = grad - np.mean(grad)
        
        # Gradient step with adaptive learning rate
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.1)
            T = max(T * 0.98, 0.0001)
        else:
            no_improve += 1
            lr = max(lr * 0.9, 0.0001)
            if no_improve > 100:
                T = min(T * 1.1, 0.01)
                if no_improve > 300:
                    rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    no_improve = 0
                    lr = 0.01
                    T = T_init
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    GradLSE: obj={best_obj:.10f}, T={T:.5f}, lr={lr:.5f}, t={elapsed:.0f}s")
    
    return best_a


def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent polish: adjust individual coordinates to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        
        # Try adjusting each coordinate
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            # Try reducing a[i] and redistributing
            for frac in [0.05, 0.1, 0.2]:
                delta = frac * a[i]
                if delta < 1e-10:
                    continue
                a_old = a[i].copy()
                a[i] -= delta
                # Distribute to positions that don't contribute to peak
                # Find positions where a[k_star - j] is small
                j_target = (k_star - i + n) % n
                if j_target == i:
                    j_target = (i + 1) % n
                a[j_target] += delta
                a_tmp = np.maximum(a, 0)
                s_tmp = float(np.sum(a_tmp))
                if s_tmp < 1e-10:
                    a[i] = a_old
                    a[j_target] -= delta
                    continue
                a_tmp = a_tmp / s_tmp
                obj_tmp = compute_obj(a_tmp)
                if obj_tmp < best_obj - 1e-14:
                    best_obj = obj_tmp
                    best_a = a_tmp.copy()
                    a = a_tmp.copy()
                    improved = True
                    break
                else:
                    a[i] = a_old
                    a[j_target] -= delta
        
        if not improved and it > 100:
            # Random perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            for _ in range(5):
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                if i1 == i2:
                    continue
                frac = float(rng.uniform(0.01, 0.1))
                delta = frac * a[i1]
                a[i1] -= delta
                a[i2] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP Alternating (golden section) + MWU + Gradient LSE + Coordinate polish.
    Multi-length exploration with diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")
    
    a = best_seq.copy()
    n_base = len(a)
    
    # === Phase 1: Quick MWU to establish baseline (150s) ===
    print(f"\n=== Phase 1: MWU (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    mwu_budget = min(150.0, remaining * 0.15)
    a_mwu = mwu_optimize_improved(a, time_budget=mwu_budget, top_k=15, verbose=True)
    obj_mwu = compute_obj(a_mwu)
    if obj_mwu < best_score:
        best_score = obj_mwu
        best_seq = a_mwu.copy()
        a = a_mwu.copy()
        print(f"  MWU: obj={best_score:.10f} **")
    else:
        print(f"  MWU: obj={obj_mwu:.10f}")
    
    # === Phase 2: LP Alternating with line search (500s) ===
    print(f"\n=== Phase 2: LP Alternating (improved) ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(500.0, remaining * 0.4)
    
    # Multiple rounds with decreasing time
    for round_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = lp_budget / (5 - round_idx) * 0.8
        alloc = min(alloc, remaining * 0.4)
        a_lp = lp_alternating_improved(a, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  LP Round {round_idx}: obj={best_score:.10f} **")
        else:
            print(f"  LP Round {round_idx}: obj={obj_lp:.10f}")
    
    # === Phase 3: Gradient LogSumExp (150s) ===
    print(f"\n=== Phase 3: Gradient LSE ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        lse_budget = min(150.0, remaining * 0.15)
        a_lse = gradient_logsumexp(a, time_budget=lse_budget, T_init=0.002, verbose=True)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            a = a_lse.copy()
            print(f"  Gradient LSE: obj={best_score:.10f} **")
        else:
            print(f"  Gradient LSE: obj={obj_lse:.10f}")
    
    # === Phase 4: LP + MWU alternating cycle (300s) ===
    print(f"\n=== Phase 4: LP+MWU Cycle ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        cycle_budget = min(300.0, remaining * 0.3)
        for cycle in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            # LP step
            alloc_lp = min(cycle_budget / 8.0, remaining * 0.3)
            a_c = lp_alternating_improved(a, time_budget=alloc_lp, verbose=False)
            obj_c = compute_obj(a_c)
            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                a = a_c.copy()
            
            # MWU step
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc_mwu = min(cycle_budget / 8.0, remaining * 0.2)
            a_c = mwu_optimize_improved(a, time_budget=alloc_mwu, top_k=10, verbose=False)
            obj_c = compute_obj(a_c)
            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                a = a_c.copy()
                print(f"  Cycle {cycle}: obj={best_score:.10f} **")
    
    # === Phase 5: Coordinate polish (60s) ===
    print(f"\n=== Phase 5: Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_cd = coordinate_polish(a, time_budget=min(60.0, remaining * 0.2))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coordinate: obj={best_score:.10f} **")
        else:
            print(f"  Coordinate: obj={obj_cd:.10f}")
    
    # === Phase 6: Length sweep (200s) ===
    print(f"\n=== Phase 6: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_base - 200), max(200, n_base - 100),
        n_base - 50, n_base, n_base + 50,
        n_base + 100, n_base + 200, n_base + 400,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        # Quick optimization for this length
        alloc = min(remaining * 0.15, 25.0)
        a_opt = mwu_optimize_improved(a_try, time_budget=alloc * 0.4, top_k=10, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = lp_alternating_improved(a_opt, time_budget=min(alloc * 0.6, remaining * 0.3), verbose=False)
        
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")
    
    # === Phase 7: Final intensive LP (remaining time) ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = lp_alternating_improved(a, time_budget=min(remaining * 0.5, 100.0), verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_final:.10f}")
    
    # === Phase 8: Random restarts with best structure ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            scale = 0.02 + 0.01 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
            
            alloc = min(remaining * 0.25, 20.0)
            a_opt = lp_alternating_improved(a_r, time_budget=alloc, verbose=False)
            obj_r = compute_obj(a_opt)
            
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.10f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.10f}")
    
    # === Final: One more LP polish with remaining time ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_last = lp_alternating_improved(best_seq, time_budget=min(remaining * 0.7, 30.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.10f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
