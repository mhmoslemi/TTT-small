"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothed optimization, spectral shaping, LP alternating,
peak-shaving, and multi-n exploration with diverse initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def logsumexp_optimize(a, n_iters=3000, time_budget=120.0, tau_init=5.0, tau_max=200.0):
    """
    Log-sum-exp smoothed optimization using softmax parameterization.
    Minimize (1/tau)*log(sum(exp(tau*conv(a,a)))) with adaptive tau.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)

    # Log parameterization: a_i = exp(x_i) / sum(exp(x_j))
    x = np.log(a)
    x -= np.max(x)  # numerical stability

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam optimizer
    m_x = np.zeros(n)
    v_x = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    size = next_fast_len(2 * n - 1)

    stall = 0
    tau = tau_init
    tau_increase = tau_max / max(1, n_iters)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Update tau (annealing schedule)
        tau = min(tau_init + tau_increase * it, tau_max)

        # Compute conv and gradient
        a_curr = np.exp(x - np.max(x))
        s_curr = np.sum(a_curr)
        a_curr = a_curr / s_curr

        fa = rfft(a_curr, size)
        conv = irfft(fa * fa, size)[:2 * n - 1]

        # Log-sum-exp weights
        tau_conv = tau * conv
        tau_conv_shifted = tau_conv - np.max(tau_conv)
        exp_w = np.exp(tau_conv_shifted)
        w = exp_w / np.sum(exp_w)

        # Gradient: grad_i = 2 * sum_k w_k * a_{k-i} (cross-correlation)
        # Use FFT for efficiency
        fw = fft(w, size)
        fa_full = fft(a_curr, size)
        # Cross-correlation: ifft(conj(fw) * fa_full)
        grad_a = np.zeros(n)
        corr = ifft(np.conj(fw) * fa_full).real
        # The cross-correlation (w * a)_i = sum_k w[k]*a[k-i]
        # The full result has length 2n-1 in the appropriate range
        # corr[k] for k=0..2n-2 gives sum_i w[i] * a[i + k - (n-1)]... 
        # Let me just compute it directly for the first n entries
        # Actually, let's use a different approach:
        # grad_i = sum_k w_k * (a_{k-i} + a_{k-i})[corrected] ≈ 2 * (w cross a)_i
        
        # Direct computation using the fact that w has length 2n-1 and a has length n
        # (w cross a)_i = sum_k w[k] * a[k - i] for valid k, i
        # This is a convolution: flip a, convolve with w, take appropriate slice
        a_flip = a_curr[::-1]
        wa_conv = np.convolve(w, a_flip, mode='full')
        # wa_conv[k] = sum_j w[j] * a_flip[k-j] = sum_j w[j] * a[n-1-(k-j)]
        # = sum_j w[j] * a[n-1-k+j]
        # We want sum_k w[k] * a[k-i] = sum_k w[k] * a_flip[n-1-k+i]
        # Let j = n-1-k+i, k = n-1-i+j
        # = sum_j w[n-1-i+j] * a_flip[j]
        # This is (w shifted) convolved with a_flip at position n-1
        # Actually, let me just use the FFT approach properly.
        
        # The gradient of sum_k w_k * conv(a,a)_k w.r.t. a_i is:
        # 2 * sum_k w_k * a_{k-i} (for i where 0 <= k-i < n)
        # = 2 * (w ⊛ a)(i) where ⊛ is cross-correlation
        
        # Using FFT: if W = DFT(w_padded), A = DFT(a_padded)
        # then cross_corr = IDFT(conj(W) * A)
        # But w has length 2n-1 and a has length n, so we need padding to 3n-2
        
        size2 = next_fast_len(3 * n)
        W2 = fft(w, size2)
        A2 = fft(a_curr, size2)
        cross = ifft(np.conj(W2) * A2).real
        # cross[i] = sum_k w[k] * a[(k - i) mod size2]
        # For i = 0, ..., n-1: cross[i] ≈ sum_k w[k] * a[k-i] (the part we want)
        # But we need to be careful about wrap-around. Since size2 >= 3n > 2n-1+n-1,
        # there's no wrap-around for the range we care about.
        
        grad_a = 2.0 * cross[:n]
        
        # Subtract mean for the sum constraint (since we're using softmax, 
        # the mean subtraction handles it)
        grad_a = grad_a - np.mean(grad_a)

        # Softmax gradient: grad_x = (I - a*a^T) @ grad_a = grad_a - a*(a @ grad_a)
        grad_x = grad_a - a_curr * np.dot(a_curr, grad_a)

        # Adam update
        m_x = beta1 * m_x + (1 - beta1) * grad_x
        v_x = beta2 * v_x + (1 - beta2) * grad_x ** 2
        m_hat = m_x / (1 - beta1 ** (it + 1))
        v_hat = v_x / (1 - beta2 ** (it + 1))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= np.max(x)  # numerical stability

        # Compute new a and objective
        a_new = np.exp(x)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Revert x
            x = np.log(np.maximum(best_a, 1e-15))
            x -= np.max(x)
            # Reduce learning rate
            if stall > 50:
                lr *= 0.8
                if it % 100 == 0:
                    print(f"    LSE it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, stall={stall}")

        if it % 200 == 0:
            print(f"    LSE it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}")

    return best_a


def spectral_shaping(a, n_iters=2000, time_budget=60.0):
    """
    Spectral shaping: iteratively modify the spectrum of a to flatten conv(a,a).
    Uses a blended approach between flat spectrum and current spectrum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    size = next_fast_len(2 * n - 1)

    stall = 0
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = fft(a, size)
        mag = np.abs(fa)
        phase = fa / np.maximum(mag, 1e-15)

        # Target: flatten the power spectrum |FFT(a)|^2
        # Current power spectrum:
        power = mag ** 2
        avg_power = np.mean(power)
        
        # Target: constant power spectrum (flat)
        # Blend: move toward flat
        target_mag = np.sqrt(avg_power)
        
        # Use a power-law blending: new_mag = mag^alpha * target_mag^(1-alpha)
        # alpha=1 means no change, alpha=0 means fully flat
        alpha = 0.7 + 0.2 * (it / max(1, n_iters))  # start aggressive, become conservative
        
        # Actually, let's use a different approach:
        # Reduce peaks in the power spectrum, boost valleys
        # This is like a "water-filling" in the frequency domain
        excess = np.maximum(power - avg_power, 0)
        deficit = np.maximum(avg_power - power, 0)
        
        # New target: reduce peaks, keep total energy
        reduction = 0.3 * excess  # reduce peaks by 30%
        new_power = power - reduction
        # Redistribute to valleys proportionally
        total_reduction = np.sum(reduction)
        if np.sum(deficit) > 1e-10:
            new_power += deficit / np.sum(deficit) * total_reduction
        new_power = np.maximum(new_power, 1e-10)
        
        new_mag = np.sqrt(new_power)
        
        # Reconstruct
        fa_new = new_mag * phase
        a_new = ifft(fa_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Perturb
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

        if it % 500 == 0:
            print(f"    SS it{it}: obj={best_obj:.8f}, stall={stall}")

    return best_a


def peak_shaving(a, time_budget=60.0):
    """
    Peak shaving: identify peaks in conv(a,a) and move mass to reduce them.
    Uses a water-filling approach on the influence vectors.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = np.max(conv)
        
        if max_val <= 1e-15:
            break

        # Get top peaks (positions where conv is close to max)
        threshold = max_val * 0.95
        peak_threshold = max_val * 0.99
        top_peaks = np.where(conv >= peak_threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.array([np.argmax(conv)])

        # Compute the "influence" at each position: how much does each a_i 
        # contribute to the top peaks?
        influence = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    influence[i] += a[j]

        # Normalized influence
        max_inf = np.max(influence)
        if max_inf < 1e-15:
            break
        influence_norm = influence / max_inf

        # Water-filling: move mass from high-influence to low-influence positions
        # Fraction of mass to move
        move_frac = 0.1

        # Source: top influence positions
        # Target: bottom influence positions
        sorted_idx = np.argsort(influence)
        n_move = max(1, n // 10)
        
        sources = sorted_idx[-n_move:]  # highest influence
        targets = sorted_idx[:n_move]    # lowest influence

        moved = 0.0
        for i_src in sources:
            delta = move_frac * a[i_src]
            if delta < 1e-15:
                continue
            # Distribute to targets proportionally to their "need"
            for i_tgt in targets:
                if i_tgt == i_src:
                    continue
                share = delta / len(targets)
                a[i_src] -= share
                a[i_tgt] += share
                moved += share

        if moved < 1e-15:
            break

        a = np.maximum(a, 0)
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            break
        a = a / s_new

        obj_new = compute_obj(a)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
        else:
            a = best_a.copy()
            # Try smaller move fraction
            move_frac *= 0.7
            if move_frac < 0.01:
                # Perturb and restart
                rng = np.random.default_rng(rnd + n)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                move_frac = 0.1

        if rnd % 200 == 0:
            print(f"    PS it{rnd}: obj={best_obj:.8f}")

    return best_a

def mirror_descent(a, n_iters=3000, time_budget=60.0):
    """Mirror descent with multi-peak subgradient."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.1
    m_peaks = 5

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[-m_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        temp = max(0.001, (max_val - top_val[0]) * 0.5)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * a[j]

        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.97
        else:
            alpha *= 0.5

        if it % 500 == 499:
            alpha = min(alpha * 1.2, 0.5)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        freqs = rng.uniform(0.5, 5.0, 10)
        phases = rng.uniform(0, 2 * np.pi, 10)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        # Random with oscillation
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * rng.uniform(3, 15) * x) + \
            0.3 * np.sin(2 * np.pi * rng.uniform(5, 20) * x)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a

    if remaining > 10:
        a2 = mirror_descent(a, n_iters=2000, time_budget=min(20, remaining * 0.5))
        results.append((compute_obj(a2), a2))

    remaining = time_budget - (time.monotonic() - t1)
    if remaining > 10:
        a3 = spectral_shaping(a, n_iters=1000, time_budget=min(20, remaining * 0.5))
        results.append((compute_obj(a3), a3))

    remaining = time_budget - (time.monotonic() - t1)
    if remaining > 10:
        a4 = peak_shaving(a, time_budget=min(15, remaining * 0.7))
        results.append((compute_obj(a4), a4))

    if not results:
        return a
    results.sort(key=lambda x: x[0])
    return results[0][1]


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining log-sum-exp optimization, spectral shaping,
    peak shaving, LP alternating, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: Quick Spectral Preparation
    print(f"\n=== Phase 1: Spectral Preparation (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_fft = fft_flat_spectrum(best_seq, n_iters=300, time_budget=min(30, remaining * 0.15), relaxation=0.5)
        obj = compute_obj(a_fft)
        print(f"  FFT flat: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fft.copy()

        a_ss = spectral_shaping(best_seq, n_iters=1000, time_budget=min(30, remaining * 0.15))
        obj = compute_obj(a_ss)
        print(f"  Spectral shaping: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_ss.copy()

    # Phase 2: Log-Sum-Exp Optimization (primary new method)
    print(f"\n=== Phase 2: Log-Sum-Exp Optimization (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 150.0)
        a_lse = logsumexp_optimize(best_seq, n_iters=5000, time_budget=alloc,
                                   tau_init=10.0, tau_max=300.0)
        obj = compute_obj(a_lse)
        print(f"  LSE (tau 10-300): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lse.copy()

        # Second LSE with different parameters
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.1, 60.0)
            a_lse2 = logsumexp_optimize(best_seq, n_iters=3000, time_budget=alloc,
                                        tau_init=50.0, tau_max=500.0)
            obj = compute_obj(a_lse2)
            print(f"  LSE (tau 50-500): obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_lse2.copy()

    # Phase 3: Peak Shaving
    print(f"\n=== Phase 3: Peak Shaving (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_ps = peak_shaving(best_seq, time_budget=alloc)
        obj = compute_obj(a_ps)
        print(f"  Peak shaving: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_ps.copy()

    # Phase 4: LP Alternating
    print(f"\n=== Phase 4: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    for rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.15, 80.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=(rnd == 0))
        obj = compute_obj(a_lp)
        print(f"  LP round {rnd}: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # Phase 5: Mirror Descent Refinement
    print(f"\n=== Phase 5: Mirror Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_md = mirror_descent(best_seq, n_iters=5000, time_budget=alloc)
        obj = compute_obj(a_md)
        print(f"  Mirror descent: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # Phase 6: Multi-n Exploration
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(400, n_best - 200), max(400, n_best - 100),
        n_best + 50, n_best + 100, n_best + 200,
        600, 700, 800, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if abs(n_try - n_best) < 20:
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 60.0)
        a_opt = diverse_optimize(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try

    # Phase 7: Random Restars with Diverse Optimization
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    for r in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        init_type = r % 5
        if init_type == 0:
            a_r = random_init(n_best, rng)
        elif init_type == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.1, n_best))
            a_r = np.maximum(a_r, 0)
            if np.sum(a_r) > 1e-10:
                a_r = a_r / np.sum(a_r)
            else:
                a_r = np.ones(n_best) / n_best
        elif init_type == 2:
            n_r = int(n_best + rng.integers(-200, 200))
            n_r = max(400, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 0)
            if np.sum(a_r) > 1e-10:
                a_r = a_r / np.sum(a_r)
            else:
                a_r = np.ones(n_r) / n_r
        elif init_type == 3:
            x = np.linspace(0, np.pi, n_best)
            a_r = np.cos(x) ** 2 + 0.5
            a_r = a_r / np.sum(a_r)
        else:
            # Anti-symmetric perturbation
            a_r = best_seq.copy()
            mid = n_best // 2
            a_r[:mid] *= 1.2 + 0.2 * rng.random()
            a_r[mid:] *= 0.8 - 0.2 * rng.random()
            a_r = np.maximum(a_r, 0)
            if np.sum(a_r) > 1e-10:
                a_r = a_r / np.sum(a_r)
            else:
                a_r = np.ones(n_best) / n_best

        alloc = min(remaining * 0.12, 50.0)
        a_opt = diverse_optimize(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # Phase 8: Final Intensive Polish
    print(f"\n=== Phase 8: Final Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()

    # LSE with high tau
    if remaining > 40:
        alloc = min(remaining * 0.25, 80.0)
        a_lse = logsumexp_optimize(best_seq, n_iters=5000, time_budget=alloc,
                                   tau_init=100.0, tau_max=1000.0)
        obj = compute_obj(a_lse)
        print(f"  Final LSE: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lse.copy()

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 60.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # Peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 30.0)
        a_ps = peak_shaving(best_seq, time_budget=alloc)
        obj = compute_obj(a_ps)
        print(f"  Final PS: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_ps.copy()

    # Mirror descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 30.0)
        a_md = mirror_descent(best_seq, n_iters=3000, time_budget=alloc)
        obj = compute_obj(a_md)
        print(f"  Final MD: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        a_final = lp_alternating(best_seq, time_budget=alloc)
        obj = compute_obj(a_final)
        print(f"  Final LP2: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
