"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Strategy: Log-sum-exp smoothed gradient descent (fast, smooth local optimization)
combined with LP alternating (exact local steps) and multi-n exploration.
Key insight: optimal sequences are asymmetric, with convolution having a flat-top structure.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_smooth_gradient(a, t):
    """
    Compute the gradient of the LSE-smoothed max of conv(a,a).
    Returns (gradient, max_conv, p) where p is the softmax weight.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    # Softmax (stable)
    conv_t = conv / t
    conv_t -= np.max(conv_t)
    p = np.exp(conv_t)
    p_sum = np.sum(p)
    p /= p_sum

    # gradient[j] = (2/t) * sum_m a[m] * p[m+j]
    # Compute via: R[j] = conv(a_flipped, p)[n-1+j]
    a_flipped = a[::-1]
    N = 1
    while N < n + len(p) - 1:
        N <<= 1
    fa = rfft(a_flipped, N)
    fp = rfft(p, N)
    conv_result = irfft(fa * fp, N)
    R = conv_result[n - 1:n - 1 + n]

    grad = (2.0 / t) * R
    return grad, max_conv, p


def lse_gradient_descent(a, time_budget, t_init=None, t_decay=0.997, lr_init=0.3, verbose=False):
    """
    Log-sum-exp smoothed gradient descent to minimize max(conv(a,a))
    subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if t_init is None:
        conv = compute_conv(a)
        t_init = float(np.max(conv) - np.median(conv)) * 2.0 + 1e-10

    best_a = a.copy()
    best_obj = compute_obj(a)
    t = t_init
    lr = lr_init
    start = time.monotonic()
    no_improve = 0

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        grad, max_conv, p = lse_smooth_gradient(a, t)

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            if lr < 1e-6:
                break
            continue
        a_new = a_new / s_new

    # No: conv(a,g)[k] = sum_j a[j]*g[k-j] (standard convolution)

        # Variables: g[0..n-1], M
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M

        # Constraints: conv(a_norm, g)[k] <= M for k=0..2n-2
        # conv(a_norm, g)[k] = sum_j a_norm[j]*g[k-j] (where 0<=k-j<n)

        A_ub = np.zeros((2 * n - 1, n + 1))

        b_ub = np.zeros(2 * n - 1)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.status == 0:

    except Exception:

        g = cp.Variable(n, nonneg=True)

        # conv(a_norm, g) as a cp expression

        constraints = [A_mat @ g <= M, cp.sum(g) == 1]

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate diverse random non-negative initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        # Exponential decay (asymmetric)
        a = np.exp(-np.arange(n) * rng.uniform(0.01, 0.1))
    elif mode == 3:
        # Reverse exponential decay
        a = np.exp(-np.arange(n-1, -1, -1) * rng.uniform(0.01, 0.1))
    elif mode == 4:
        # Two-sided exponential (asymmetric centers)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w1 = n * rng.uniform(0.05, 0.2)
        w2 = n * rng.uniform(0.05, 0.2)
        x = np.arange(n)
        a = np.exp(-np.abs(x - c1) / w1) + np.exp(-np.abs(x - c2) / w2)
    elif mode == 5:
        # Polynomial window
        x = np.linspace(0, 1, n)
        p = rng.integers(2, 6)
        q = rng.integers(2, 6)
        a = x**p * (1-x)**q + 0.01
    elif mode == 6:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.sin(np.pi * x * rng.uniform(1, 3)) + 0.01
    else:
        # Random smooth (Gaussian process-like)
        a = rng.uniform(0, 1, n)
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=n * 0.05)
        a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n

Strategy: Log-sum-exp smoothed gradient descent (fast, smooth local optimization)

Key insight: optimal sequences are asymmetric, with convolution having a flat-top structure.

import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_smooth_gradient(a, t):
    """
    Compute gradient of LSE-smoothed max of conv(a,a).
    Returns (gradient, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    conv_t = conv / t
    conv_t -= np.max(conv_t)
    p = np.exp(conv_t)
    p /= np.sum(p)

    # grad[j] = (2/t) * sum_m a[m] * p[m+j]
    # R[j] = conv(a_flipped, p)[n-1+j]
    a_flipped = a[::-1]
    N = 1
    while N < n + len(p) - 1:
        N <<= 1
    fa = rfft(a_flipped, N)
    fp = rfft(p, N)
    conv_result = irfft(fa * fp, N)
    R = conv_result[n - 1:n - 1 + n]

    grad = (2.0 / t) * R
    return grad, max_conv

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
