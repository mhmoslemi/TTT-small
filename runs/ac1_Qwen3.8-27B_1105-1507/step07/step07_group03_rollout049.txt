import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # Column j: A[k, j] = a[k-j] if 0 <= k-j < n
        # k ranges from j to j+n-1
        A[j:j+n, j] = a
    return A

def solve_lp(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0"""
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M
    c = np.zeros(n+1)
    c[-1] = 1.0
    # Constraints: A@g - M <= 0 (i.e., A@g <= M)
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g/s, float(res.x[-1])
    return None, None

def lp_alternation(a, time_budget):
    """Main LP alternating optimizer."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))

        return a
    a = a / s

    start = time.monotonic()

        if time.monotonic() - start > time_budget:
            break
        
        g, M = solve_lp(a)

def soft_max_adam(a, time_budget, T_init=0.1):
    """Adam optimization on softened objective."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.1
    T = T_init
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient of soft-max objective
        conv = compute_conv(a)
        conv_shifted = conv - np.max(conv)
        weights = np.exp(conv_shifted / T)
        weights /= weights.sum()
        
        # Gradient: grad[m] = (2/T) * sum_{j=0}^{n-1} weights[j+m] * a[j]
        # Compute via correlation
        L = 1
        while L < 3*n:
            L <<= 1
        W = fft(weights, L)
        A = fft(a, L)
        corr = ifft(np.conj(W) * A).real[:n]
        grad = (2.0 / T) * corr
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Adam update
        t = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        
        # Anneal T and lr
        if it % 100 == 99:
            T *= 0.95
            lr *= 0.97
        
        if T < 0.01:
            T = 0.01
        if lr < 0.001:
            lr = 0.001
    
    return best_a
