import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    # Fast convolution via FFT
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    # Build matrix A (2n-1 x n) such that A @ g = conv(a, g)
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, a_normalized=True):
    # Solve: min M s.t. A @ g <= M, sum(g) = 1, g >= 0
    # where A = build_conv_matrix(a)
    n = len(a)
    if not a_normalized:
        a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    # Try solvers in order of preference
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except:
            continue
    return None, None

def lp_alternating(a, time_budget, verbose=False):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

        g_new, M_val = solve_lp(a)
        if g_new is None:
            break
        
        # Line search for step size
        best_t = t
        best_try_obj = best_obj
        best_try_a = best_a.copy()
        
        for try_t in [t * 0.5, t, t * 1.5, 0.3, 0.5, 0.7, 0.9]:
            if try_t <= 0 or try_t > 1:

def compute_conv(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_gid(a):
    """Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a.copy()
    a /= float(np.sum(a))
    A = conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0.0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternating(a, budget, max_iters=1000, verbose=False):
    """LP alternating optimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a /= s

    best = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    t = 0.5
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > budget:
            break

        g_new, M_val = solve_lp_gid(a)
        if g_new is None:
            break

        # Adaptive line search
        candidates = [max(0.01, t * 0.5), max(0.01, t), min(1.0, t * 1.5), 0.3, 0.5, 0.7, 0.9]
        best_t = t
        best_aux = best_obj
        best_aux_a = best.copy()

        for try_t in candidates:
            try_t = float(np.clip(try_t, 0.01, 1.0))
            a_try = np.maximum((1.0 - try_t) * a + try_t * g_new, 0.0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try /= st
            o = compute_obj(a_try)
            if o < best_aux:
                best_aux = o
                best_t = try_t
                best_aux_a = a_try.copy()

        if best_aux < best_obj - 1e-12:
            best_obj = best_aux
            best = best_aux_a
            a = best
            stall = 0
            t = best_t
            if verbose and it % 20 == 0:
                print(f"    LP iter {it}: obj={best_obj:.8f}, t={t:.3f}, {time.monotonic()-t0:.0f}s")
        else:
            stall += 1
            t = max(0.05, t * 0.8)
            if stall > 30:
                break

    return best, best_obj


def lse_gradient(y, w):
    """Compute gradient of LSE: grad[j] = 2 * sum_m y[m] * w[j+m]."""
    n = len(y)
    # M[j, m] = w[j + m]
    idx = np.arange(n)
    M = w[idx[:, None] + idx[None, :]]
    return 2.0 * (M @ y)


def fista_lse(a, budget, T_start=3.0, T_end=80.0, verbose=False):
    """FISTA with log-sum-exp smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a /= s

    best = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()

    x = a.copy()
    y = a.copy()
    t_k = 1.0

    # Initial Lipschitz estimate
    conv0 = compute_conv(x)
    L = max(2.0 * float(np.max(conv0)), 0.1)

    it = 0
    while time.monotonic() - t0 < budget:
        # Adaptive temperature (continuation method)
        frac = min((time.monotonic() - t0) / budget, 1.0)
        T = T_start + (T_end - T_start) * frac ** 1.5

        # Compute smoothed gradient at y
        conv_y = compute_conv(y)
        cv = T * (conv_y - np.max(conv_y))
        w = np.exp(cv)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            w = np.zeros(len(conv_y))
            w[int(np.argmax(conv_y))] = 1.0
        else:
            w /= w_sum

        # Gradient
        grad = lse_gradient(y, w)
        # Project to sum=0 (tangent to simplex)
        grad_c = grad - float(np.mean(grad))

        # Adaptive step size
        L_new = max(1.1 * 2.0 * float(np.max(conv_y)), 0.05)
        step = 1.0 / L_new
        L = L_new

        # FISTA update
        x_new = np.maximum(x - step * grad_c, 0.0)
        sx = float(np.sum(x_new))
        if sx < 1e-10:
            x_new = x.copy()
            sx = 1.0
        x_new /= sx

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y /= sy
        else:
            y = x_new.copy()

        t_k = t_new
        x = x_new
        it += 1

        if it % 50 == 0:
            obj = compute_obj(x)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best = x.copy()
                if verbose:
                    print(f"    FISTA iter {it}: obj={best_obj:.8f}, T={T:.1f}, {time.monotonic()-t0:.0f}s")

    # Final check
    obj = compute_obj(x)
    if obj < best_obj:
        best_obj = obj
        best = x.copy()

    return best, best_obj


def gradient_perturb(a, scale, rng, alpha=0.5):
    """Perturb along negative subgradient mixed with random direction."""
    n = len(a)
    a = (a / float(np.sum(a))).copy()
    conv = compute_conv(a)

    # Get top-k peaks for a better gradient
    k_top = np.argsort(conv)[::-1][:5]
    grad = np.zeros(n)
    for k in k_top:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        d_start = k + 1 - j_end
        d_end = k - j_start + 1
        grad[j_start:j_end] += a[d_start:d_end][::-1]
    grad /= float(len(k_top))
    grad = grad - float(np.mean(grad))
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-10:
        grad /= gnorm

    rand_dir = rng.normal(0, 1, n)
    rand_dir = rand_dir - float(np.mean(rand_dir))
    rnorm = np.linalg.norm(rand_dir)
    if rnorm > 1e-10:
        rand_dir /= rnorm

    direction = alpha * (-grad) + (1.0 - alpha) * rand_dir
    a_pert = a + scale * direction
    a_pert = np.maximum(a_pert, 0.0)
    sp = float(np.sum(a_pert))
    if sp > 1e-10:
        a_pert /= sp
    else:
        a_pert = a.copy()
    return a_pert


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def coord_descent(a, budget, verbose=False):
    """Coordinate descent: try pairwise mass transfers."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a /= s

    best = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    for it in range(10000):
        if time.monotonic() - t0 > budget:
            break

        improved = False
        # Try structured moves based on current peaks
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Try moving mass away from peak contributors
        n_tries = 3
        for _ in range(n_tries):
            # Pick i from peak contributors, j from non-contributors
            j_start = max(0, k_star - n + 1)
            j_end = min(n, k_star + 1)
            i = int(rng.integers(j_start, j_end)) if j_end > j_start else int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue

            for frac in [0.005, 0.01, 0.02, 0.05]:
                d = frac * a[i]
                if d >= a[i] * 0.9:
                    continue
                a[i] -= d
                a[j] += d
                a = np.maximum(a, 0.0)
                a /= float(np.sum(a))
                o = compute_obj(a)
                if o < best_obj - 1e-13:
                    best_obj = o
                    best = a.copy()
                    improved = True
                    break
                else:
                    a[i] += d
                    a[j] -= d
                    a /= float(np.sum(a))
                    break

        if not improved:
            # Random perturbation to escape
            scale = 0.02 + 0.01 * (it % 10)
            a = best * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0.0)
            a /= float(np.sum(a))
            if it % 200 == 199:
                best = a.copy()
                best_obj = compute_obj(a)
                if verbose:
                    print(f"    CD iter {it}: obj={best_obj:.8f}, {time.monotonic()-t0:.0f}s")

    return best, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: LP alternating + FISTA LSE + gradient perturbations
    + multi-n exploration + coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800)
        best_score = 2.0
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting optimization from n={n_parent}, score={best_score:.8f}")

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating ===")
    a = best_seq.copy()

    # Run LP alternating with multiple rounds and perturbations
    n_rounds = 4
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(remaining / (n_rounds - rnd), 350.0)
        print(f"  Round {rnd}: budget={alloc:.0f}s, current obj={best_score:.8f}")

        a_new, obj_new = lp_alternating(a, alloc, verbose=True)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** Round {rnd} improved: obj={best_score:.8f} **")
        else:
            print(f"  Round {rnd}: obj={obj_new:.8f} (no improvement)")

        # Gradient-guided perturbation for next round
        if rnd < n_rounds - 1:
            scale = 0.15 + 0.1 * rnd
            a = gradient_perturb(best_seq, scale, rng, alpha=0.6)

    # === Phase 2: FISTA with LSE smoothing ===
    print(f"\n=== Phase 2: FISTA LSE ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        fista_budget = min(250.0, remaining * 0.35)
        a_fista = fista_lse(best_seq, fista_budget, T_start=5.0, T_end=60.0, verbose=True)
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  ** FISTA improved: obj={best_score:.8f} **")
        else:
            print(f"  FISTA: obj={obj_fista:.8f}")

        # Run FISTA from a perturbed start
        remaining = deadline - time.monotonic()
        if remaining > 100:
            a_pert = gradient_perturb(best_seq, 0.1, rng, alpha=0.3)
            fista_budget2 = min(150.0, remaining * 0.3)
            a_fista2 = fista_lse(a_pert, fista_budget2, T_start=3.0, T_end=40.0)
            obj_fista2 = compute_obj(a_fista2)
            if obj_fista2 < best_score:
                best_score = obj_fista2
                best_seq = a_fista2.copy()
                print(f"  ** FISTA2 improved: obj={best_score:.8f} **")
            else:
                print(f"  FISTA2: obj={obj_fista2:.8f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_base = len(best_seq)
    n_options = []
    for dn in [-200, -100, -50, 0, 50, 100, 200, 300]:
        n_try = n_base + dn
        if 200 <= n_try <= 2500:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        if n_try == n_base:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)

        # Quick LP + FISTA
        lp_alloc = min(60.0, remaining * 0.3)
        fista_alloc = min(40.0, remaining * 0.2)

        a_opt, _ = lp_alternating(a_try, lp_alloc)
        a_opt, _ = fista_lse(a_opt, fista_alloc, T_start=5.0, T_end=50.0)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Random restarts with structured perturbations ===
    print(f"\n=== Phase 4: Structured Restarts ===")
    n_best = len(best_seq)
    n_restarts = 8

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        # Different perturbation strategies
        if r % 3 == 0:
            # Gaussian perturbation
            scale = 0.05 + 0.03 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif r % 3 == 1:
            # Gradient-guided perturbation
            a_r = gradient_perturb(best_seq, 0.2 + 0.05 * r, rng, alpha=0.7)
        else:
            # Focused perturbation: zero out some coordinates
            a_r = best_seq.copy()
            n_zero = n_best // 10
            zero_idx = rng.choice(n_best, n_zero, replace=False)
            a_r[zero_idx] = 0.0
            a_r = np.maximum(a_r, 0)
            s = float(np.sum(a_r))
            if s > 1e-10:
                a_r /= s
            else:
                a_r = best_seq.copy()

        # Quick optimization: LP + short FISTA
        lp_alloc = min(40.0, remaining * 0.25)
        fista_alloc = min(30.0, remaining * 0.15)

        a_opt, _ = lp_alternating(a_r, lp_alloc)
        a_opt, _ = fista_lse(a_opt, fista_alloc, T_start=5.0, T_end=40.0)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        # Final LP round
        alloc = min(remaining * 0.35, 120.0)
        a_lp, _ = lp_alternating(best_seq, alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Coordinate descent
        alloc = min(remaining * 0.35, 80.0)
        a_cd, _ = coord_descent(best_seq, alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** Coord Descent: obj={best_score:.8f} **")
        else:
            print(f"  Coord Descent: obj={obj_cd:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        # One more short LP
        alloc = min(remaining * 0.5, 50.0)
        a_final, _ = lp_alternating(best_seq, alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP2: obj={best_score:.8f} **")
        else:
            print(f"  Final LP2: obj={obj_final:.8f}")

    # Final validation and cleanup
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isinf(val):
        result = [1.0] * n_best
        best_score = 2.0
    else:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
