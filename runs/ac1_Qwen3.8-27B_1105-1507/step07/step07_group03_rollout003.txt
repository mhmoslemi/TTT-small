def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    LP-Nest it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M:.6f}")
        else:
            stall += 1
            if stall > 10:
                # Perturb to escape
                rng = np.random.default_rng((it * 31 + n) % (2**32))
                scale = 0.02 + 0.01 * min(stall // 10, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

        # Nesterov momentum update
        y = a + beta * (a - best_a)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = a.copy()

    return best_a


def logsumexp_mirror_descent(a, n_iters=3000, time_budget=120.0, beta_start=5.0, beta_end=50.0):
    """Mirror descent with log-sum-exp smoothed objective for dense gradients."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    alpha = 0.1
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Anneal beta
        frac = it / max(1, n_iters)
        beta = beta_start + (beta_end - beta_start) * frac

        conv = compute_conv(a)

        # Log-sum-exp weights (stable computation)
        conv_shifted = conv - np.max(conv)
        exp_vals = np.exp(beta * conv_shifted)
        weights = exp_vals / np.sum(exp_vals)

        # Gradient: grad_i = sum_k weights_k * a_{k-i} (convolution of weights with a)
        # This is the cross-correlation of weights with a
        L = 1
        while L < 2 * n - 1:
            L <<= 1
        W = rfft(weights, L)
        A = rfft(a, L)
        grad = irfft(W * A, L)[:n]

        # Ensure gradient is non-negative (it should be since weights and a are non-negative)
        grad = np.maximum(grad, 0)

        # Mirror descent step
        a_new = a * np.exp(-alpha * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                alpha *= 0.95
                if stall > 50:
                    # Perturb
                    rng = np.random.default_rng(it)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
                    alpha = 0.1
        else:
            alpha *= 0.5

        # Periodically boost alpha
        if it % 500 == 499:
            alpha = min(alpha * 1.2, 0.5)

    return best_a


def freq_domain_shaping(a, n_iters=200, time_budget=60.0):
    """Frequency domain flat-spectrum shaping with soft non-negativity."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    target_relax = 0.6

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe

        # Target: flat magnitude (normalize to have same total energy)
        mean_mag = np.mean(mag)
        target = mean_mag

        # Blend: soft flatness
        A_new_mag = (1 - target_relax) * mag + target_relax * target
        A_new = A_new_mag * phase

        a_new = ifft(A_new).real[:n]

        # Soft non-negativity (softplus instead of hard clip)
        theta = 1.0 / (10 + it * 0.5)  # decreasing softness
        a_new = theta * np.log1p(np.exp(a_new / theta))

        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    # Try harder relaxation
                    target_relax = min(0.9, target_relax + 0.1)
                    rng = np.random.default_rng(it)
                    a = best_a
                    phase_noise = np.exp(1j * rng.normal(0, 0.1, L))
                    A = fft(a, L)
                    A = A * phase_noise
                    a = np.maximum(ifft(A).real[:n], 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    stall = 0
                    target_relax = 0.6

    return best_a


def parametric_cosine_optimize(a_init, n_basis=30, time_budget=60.0):
    """Optimize using a cosine basis parameterization with softmax."""
    n = len(a_init)
    x = np.arange(n)

    # Build cosine basis
    basis = np.ones((1, n))
    for k in range(1, n_basis):
        basis = np.vstack([basis, np.cos(2 * np.pi * k * x / n)])
        if k % 2 == 0:
            basis = np.vstack([basis, np.sin(2 * np.pi * k * x / n)])

    n_params = basis.shape[0]

    # Initialize parameters from a_init via least squares
    coeffs, _, _, _ = np.linalg.lstsq(basis.T, a_init, rcond=None)

    def objective(params):
        a = basis @ params
        # Softmax for non-negativity
        a_max = np.max(a)
        a_exp = np.exp(a - a_max)
        a_norm = a_exp / np.sum(a_exp)
        conv = compute_conv(a_norm)
        return 2.0 * n * float(np.max(conv))

    # Use L-BFGS-B
    best_obj = objective(coeffs)
    best_params = coeffs.copy()
    start = time.monotonic()

    def obj_and_grad(params):
        a = basis @ params
        a_max = np.max(a)
        a_exp = np.exp(a - a_max)
        a_norm = a_exp / np.sum(a_exp)
        obj = 2.0 * n * float(np.max(compute_conv(a_norm)))
        return obj

    result = minimize(obj_and_grad, coeffs, method='L-BFGS-B',
                      options={'maxiter': 200, 'time_limit': time_budget})

    if result.fun < best_obj:
        best_obj = result.fun
        best_params = result.x

    # Reconstruct
    a_opt = basis @ best_params
    a_max = np.max(a_opt)
    a_opt = np.exp(a_opt - a_max)
    a_opt = a_opt / np.sum(a_opt)
    a_opt = np.maximum(a_opt, 0)

    return a_opt, best_obj


def simulated_annealing_lp(a, time_budget=60.0):
    """Simulated annealing on LP alternating iterations."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 0.1  # Initial temperature
    T_min = 0.001
    cooling = 0.995

    while time.monotonic() - start < time_budget:
        if T < T_min:
            T = 0.05  # Reheat occasionally

        # Get LP direction
        conv = compute_conv(a)
        M = float(np.max(conv))
        g_lp = solve_lp_minpeak(a, M)
        if g_lp is None:
            T *= 0.9
            continue

        # Propose move: a_new = (1-t)*a + t*g + noise
        t = 0.3
        noise_scale = T * 0.05
        rng = np.random.default_rng()
        a_proposed = (1 - t) * a + t * g_lp + rng.normal(0, noise_scale, n)
        a_proposed = np.maximum(a_proposed, 0)
        s_prop = float(np.sum(a_proposed))
        if s_prop < 1e-10:
            T *= 0.9
            continue
        a_proposed = a_proposed / s_prop

        obj_proposed = compute_obj(a_proposed)
        delta = obj_proposed - best_obj

        # Metropolis criterion
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_proposed
            if obj_proposed < best_obj:
                best_obj = obj_proposed
                best_a = a_proposed.copy()
        T *= cooling

    return best_a


def multi_peak_subgradient(a, n_iters=5000, time_budget=90.0):
    """Subgradient method with multiple peak tracking and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m_peaks = 10
    momentum = np.zeros(n)
    beta_m = 0.9
    avg_grad_norm_sq = 0.0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top m peaks
        top_idx = np.argsort(conv)[-m_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        # Temperature for softmax weighting of peaks
        temp = max(0.001, 0.1 * (max_val - top_val[0] + 1e-15))
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        # Compute subgradient (convolution of weighted peaks with a)
        L = 1
        while L < 2 * n - 1:
            L <<= 1
        w_ext = np.zeros(L)
        for w, k in zip(weights, top_idx):
            w_ext[int(k)] = w
        W = rfft(w_ext, L)
        A = rfft(a, L)
        grad = irfft(W * A, L)[:n]
        grad = np.maximum(grad, 0)

        # Polyak step size
        grad_norm_sq = float(np.sum(grad ** 2))
        avg_grad_norm_sq = 0.9 * avg_grad_norm_sq + 0.1 * grad_norm_sq
        gap = best_obj - 1.0  # Lower bound is ~1 for this problem
        gap = max(gap, 0.01)
        alpha = gap / max(avg_grad_norm_sq, 1e-10)
        alpha = min(alpha, 0.5)

        # Nesterov momentum
        momentum = beta_m * momentum + (1 - beta_m) * grad
        a_new = a - alpha * momentum
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        freqs = rng.uniform(0, 5, 10)
        phases = rng.uniform(0, 2 * np.pi, 10)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        c1, c2 = n // 3, 2 * n // 3
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # U-shaped
        x = np.arange(n) / n
        a = (x - 0.5) ** 2 + 0.1
        a = a / np.sum(a)
    else:
        # Random smooth
        a = rng.uniform(0, 1, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a

        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
