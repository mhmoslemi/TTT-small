import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def objective(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    b = a / s
    c = compute_conv(b)
    return 2.0 * n * float(np.max(c))

def simplex_project(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    # Standard simplex projection
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_topk(b, k_indices):

    n = len(b)
    grad = np.zeros(n)
    K = len(k_indices)

"""

Combines: accelerated projected subgradient on simplex, MWU, LP alternating (cvxpy/HiGHS),
coordinate descent polish, multi-length sweep, and random restarts.
Key innovation: fast O(n log n) subgradient iterations with Nesterov acceleration,

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(b,b)) where b = a/sum(a)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    b = a / s
    conv = compute_conv(b)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} using sort-based algorithm."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.zeros(n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_for_k(b, k):
    """Compute 2*S_k*b where (S_k*b)_i = b_{k-i} if valid."""
    n = len(b)
    grad = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        grad[i_start:i_end + 1] = 2.0 * b[k - i_end:k - i_start + 1][::-1]
    return grad


def subgradient_topk(b, k_indices):
    """Average subgradient over top-K peak positions."""
    n = len(b)
    grad = np.zeros(n)
    K = len(k_indices)
    for k in k_indices:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            grad[i_start:i_end + 1] += b[k - i_end:k - i_start + 1][::-1]
    grad *= 2.0 / K
    return grad

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_start = max(0, j)  # k-j >= 0 => k >= j
        k_end = min(2 * n - 2, j + n - 1)  # k-j < n => k < j+n
        if k_start <= k_end:
            A[k_start:k_end + 1, j] = a[k_start - j:k_end - j + 1]
    return A


def solve_lp_minpeak(a, warm_start=None):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.HIGHS, cp.PDLP, cp.SCS, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy HiGHS
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))

                a[low_idx] += delta
                new_obj = 2.0 * n * float(np.max(compute_conv(a)))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[high_idx] += delta
                    a[low_idx] -= delta
                    break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_init(n, rng):
    """Initialize using magnitude of random Fourier transform."""
    phases = rng.uniform(0, 2 * np.pi, n)
    complex_seq = np.exp(1j * phases)
    # IFFT gives a time-domain signal
    real_signal = np.abs(irfft(rfft(complex_seq, n), n))
    # Actually let's just use random with some structure
    # Use a "smoothed" random
    raw = rng.random(n)
    # Apply a moving average to smooth
    kernel_size = max(3, n // 100)
    kernel = np.ones(kernel_size) / kernel_size
    smoothed = np.convolve(raw, kernel, mode='same')
    return np.maximum(smoothed, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Hybrid of accelerated subgradient, MWU, LP alternating,
    coordinate polish, multi-length sweep, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(600) / 600
        best_score = compute_obj(best_seq)
        print(f"Default init (n=600): obj={best_score:.8f}")

    n_parent = len(best_seq)
    a = best_seq.copy()

    # === Phase 1: Fast subgradient to get a quick improvement (120s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 1: Accelerated Subgradient ===")
    sg_budget = min(120.0, remaining * 0.1)
    a_sg = accelerated_subgradient(a, time_budget=sg_budget, top_k=10, verbose=True)
    obj_sg = compute_obj(a_sg)
    if obj_sg < best_score:
        best_score = obj_sg
        best_seq = a_sg.copy()
        a = a_sg.copy()
        print(f"  SubGrad: obj={best_score:.8f} **")
    else:
        print(f"  SubGrad: obj={obj_sg:.8f}")

    # === Phase 2: MWU optimization (200s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 2: MWU Optimization ===")
    mwu_budget = min(200.0, remaining * 0.18)
    for k_val in [10, 20]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(mwu_budget / 2.0, remaining * 0.45)
        a_mwu = mwu_optimize(a, time_budget=alloc, top_k=k_val, verbose=True)
        obj_mwu = compute_obj(a_mwu)
        if obj_mwu < best_score:
            best_score = obj_mwu
            best_seq = a_mwu.copy()
            a = a_mwu.copy()
            print(f"  MWU K={k_val}: obj={best_score:.8f} **")
        else:
            print(f"  MWU K={k_val}: obj={obj_mwu:.8f}")

    # === Phase 3: LP Alternating (main workhorse, 500s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 3: LP Alternating ===")
    lp_budget = min(500.0, remaining * 0.4)

    for round_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(lp_budget / (5 - round_idx), remaining * 0.45)
        a_lp = lp_alternating(a, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  LP Round {round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP Round {round_idx}: obj={obj_lp:.8f}")

    # === Phase 4: Another round of subgradient (100s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: SubGradient Round 2 ===")
    if remaining > 60:
        sg_budget2 = min(100.0, remaining * 0.12)
        a_sg2 = accelerated_subgradient(best_seq, time_budget=sg_budget2, top_k=20, verbose=True)
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            a = a_sg2.copy()
            print(f"  SubGrad2: obj={best_score:.8f} **")
        else:
            print(f"  SubGrad2: obj={obj_sg2:.8f}")

    # === Phase 5: Multi-length exploration (200s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    n_options = sorted(set([
        max(200, n_parent - 100), max(300, n_parent - 50),
        n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        int(n_parent * 1.3),
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc_mwu = min(20.0, remaining * 0.15)
        alloc_lp = min(30.0, remaining * 0.2)

        a_opt = mwu_optimize(a_try, time_budget=alloc_mwu, top_k=10)
        a_opt = lp_alternating(a_opt, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final LP polish (remaining * 0.35) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Final LP Polish ===")
    if remaining > 30:
        alloc = min(remaining * 0.35, 120.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 7: Coordinate polish (remaining * 0.2) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 7: Coordinate Polish ===")
    if remaining > 20:
        alloc = min(remaining * 0.2, 40.0)
        a_cd = coordinate_polish(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord: obj={best_score:.8f} **")
        else:
            print(f"  Coord: obj={obj_cd:.8f}")

    # === Phase 8: Random restarts with diverse perturbations ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 8: Random Restarts ===")
    n_best = len(best_seq)
    max_restarts = 12
    for r in range(max_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Different perturbation strategies
        if r % 3 == 0:
            # Gaussian multiplicative
            scale = 0.02 + 0.01 * (r // 3)
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif r % 3 == 1:
            # Sparse perturbation
            a_r = best_seq.copy()
            n_perturb = max(10, n_best // 10)
            idx = rng.choice(n_best, n_perturb, replace=False)
            a_r[idx] *= np.exp(rng.normal(0, 0.1, n_perturb))
        else:
            # Blend with spectral init
            a_spec = spectral_init(n_best, rng)
            a_spec = a_spec / np.sum(a_spec)
            a_r = 0.7 * best_seq + 0.3 * a_spec

        a_r = np.maximum(a_r, 0)
        a_r = a_r / np.sum(a_r)

        # Quick optimization
        alloc = min(remaining * 0.3, 30.0)
        a_opt = mwu_optimize(a_r, time_budget=alloc * 0.4, top_k=10)
        a_opt = lp_alternating(a_opt, time_budget=alloc * 0.6)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Final: one more LP pass on the best ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Final LP Pass ===")
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.7, 60.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
