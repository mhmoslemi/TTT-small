```python
import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def gradient_energy(a):
    """Compute gradient of sum_k (a*a)[k]^2 with respect to a."""
    n = len(a)
    L = 1
    while L < 3*n:
        L <<= 1
    A = fft(a, L)
    conv_full = ifft(A * A).real[:2*n-1]  # full linear convolution
    # Cross-correlation of a and conv
    D = fft(conv_full, L)  # conv_full zero-padded to L
    grad = 4.0 * ifft(np.conj(A) * D).real[:n]
    return grad

def energy_gradient_descent(a, n_iters=5000, time_budget=60.0):
    """Minimize sum_k conv[k]^2 using mirror descent (exponential weights)."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1
    
    L = 1
    while L < 3*n:
        L <<= 1
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, L)
        conv_full = ifft(A * A).real[:2*n-1]
        D = fft(conv_full, L)
        grad = 4.0 * ifft(np.conj(A) * D).real[:n]
        
        # Mirror descent step
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-15)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
            # Adaptive learning rate
            if it % 100 == 0 and it > 0:
                lr *= 0.95
        else:
            lr *= 0.5
    
    return best_a
```

def find_best_t(a, g, n_grid=200):
    """Find t in [0,1] that minimizes obj((1-t)*a + t*g)."""
    n = len(a)
    # Precompute conv_a, conv_g, and cross = a*g
    L = 1
    while L < 2*n-1:
        L <<= 1
    A = fft(a, L)
    G = fft(g, L)
    conv_a = ifft(A*A).real[:2*n-1]
    conv_g = ifft(G*G).real[:2*n-1]
    cross = ifft(A*G).real[:2*n-1]
    
    # obj(t) = 2n * max_k (conv_a[k] + t*2*cross[k] + t^2*conv_g[k] - 2t*conv_a[k] + t^2*conv_a[k])
    # Wait, let me redo: ((1-t)*a + t*g) * ((1-t)*a + t*g)
    # = (1-t)^2 * a*a + 2t(1-t)*a*g + t^2 * g*g
    # So at lag k: (1-t)^2 * conv_a[k] + 2t(1-t) * cross[k] + t^2 * conv_g[k]
    
    # This is a quadratic in t for each k:
    # q_k(t) = (1-2t+t^2)*conv_a[k] + (2t-2t^2)*cross[k] + t^2*conv_g[k]
    #        = conv_a[k] + t*(-2*conv_a[k] + 2*cross[k]) + t^2*(conv_a[k] - 2*cross[k] + conv_g[k])
    #        = conv_a[k] + t*2*(cross[k]-conv_a[k]) + t^2*(conv_a[k] - 2*cross[k] + conv_g[k])
    
    # Note: conv_a[k] - 2*cross[k] + conv_g[k] = (a-g)*(a-g) at lag k >= 0
    # So the quadratic coefficient is always >= 0 (it's a sum of squares)

    # Grid search first
    ts = np.linspace(0, 1, n_grid)
    objs = np.zeros(n_grid)
    for i, t in enumerate(ts):
        conv_t = (1-t)**2 * conv_a + 2*t*(1-t) * cross + t**2 * conv_g
        objs[i] = 2*n*np.max(conv_t)
    
    best_i = np.argmin(objs)
    best_t = ts[best_i]
    best_obj = objs[best_i]
    
    # Refine with golden section around the best point
    if 0 < best_i < n_grid-1:
        lo = ts[best_i-1]
        hi = ts[best_i+1]
        # Golden section search
        gr = (np.sqrt(5)-1)/2
        for _ in range(50):
            t1 = hi - gr*(hi-lo)
            t2 = lo + gr*(hi-lo)
            conv_t1 = (1-t1)**2 * conv_a + 2*t1*(1-t1) * cross + t1**2 * conv_g
            conv_t2 = (1-t2)**2 * conv_a + 2*t2*(1-t2) * cross + t2**2 * conv_g
            obj1 = 2*n*np.max(conv_t1)
            obj2 = 2*n*np.max(conv_t2)
            if obj1 < obj2:
                hi = t2
            else:
                lo = t1
            if hi - lo < 1e-8:
                break
        best_t = (lo+hi)/2
        conv_best = (1-best_t)**2 * conv_a + 2*best_t*(1-best_t) * cross + best_t**2 * conv_g
        best_obj = 2*n*np.max(conv_best)
    
    return best_t, best_obj

"""
Optimized step function search combining:
1. Energy gradient descent (minimizes sum of squared convolution values via FFT)
2. LP alternating with golden-section line search
3. Multi-n exploration with adaptive optimization
4. Coordinate descent polish
The key new idea is the energy gradient descent which minimizes the L2 norm
of the autocorrelation (equivalently, its variance since sum is fixed),
providing a fast O(n log n) pre-conditioning step before the LP refinement.
"""

import numpy as np
from scipy.fft import fft, ifft
import time
from scipy.optimize import linprog


def next_pow2(x):
    """Return next power of 2 >= x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def energy_gradient(a, L=None):
    """Compute gradient of sum_k (a*a)[k]^2 with respect to a.
    Uses FFT for O(n log n) computation."""
    n = len(a)
    if L is None:
        L = next_pow2(3 * n)
    A = fft(a, L)
    conv_full = ifft(A * A).real[:2 * n - 1]
    D = fft(conv_full, L)
    grad = 4.0 * ifft(np.conj(A) * D).real[:n]
    return grad


def energy_gradient_descent(a, n_iters=10000, time_budget=60.0):
    """Minimize sum_k conv[k]^2 using mirror descent (exponential weights).
    This flattens the autocorrelation spectrum."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05
    L = next_pow2(3 * n)
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        grad = energy_gradient(a, L)

        # Mirror descent step
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-15)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0

            else:

                    lr *= 0.9

                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    lr = 0.05
                    stall = 0

        # Anneal learning rate
        if it % 1000 == 999:
            lr *= 0.85

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. a@g <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # Use scipy linprog with HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    # Fallback: cvxpy
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    return None, None


def find_best_t_golden(a, g, n_grid=100):
    """Find optimal t in [0,1] for mixing a and g using golden section search.
    Uses the fact that conv((1-t)a+tg) is a quadratic in t for each lag."""
    n = len(a)
    L = next_pow2(2 * n - 1)
    A = fft(a, L)
    G = fft(g, L)
    conv_a = ifft(A * A).real[:2 * n - 1]
    conv_g = ifft(G * G).real[:2 * n - 1]
    cross = ifft(A * G).real[:2 * n - 1]

    def obj_at_t(t):
        conv_t = (1 - t) ** 2 * conv_a + 2 * t * (1 - t) * cross + t ** 2 * conv_g
        return 2.0 * n * float(np.max(conv_t))

    # Grid search to find rough minimum
    ts = np.linspace(0, 1, n_grid)
    objs = np.array([obj_at_t(t) for t in ts])
    best_i = int(np.argmin(objs))
    best_t = ts[best_i]
    best_obj = objs[best_i]

    # Golden section refinement
    if 0 < best_i < n_grid - 1:
        lo = ts[max(0, best_i - 2)]
        hi = ts[min(n_grid - 1, best_i + 2)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(60):
            if hi - lo < 1e-10:
                break
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            o1 = obj_at_t(t1)
            o2 = obj_at_t(t2)
            if o1 < o2:
                hi = t2
            else:
                lo = t1
        best_t = (lo + hi) / 2.0
        best_obj = obj_at_t(best_t)

    return best_t, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    if new_n > n:
        a_new = np.interp(x_new, x_old, a)
    else:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_structural_init(n, mode=0):
    """Generate a structured initial sequence."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif mode == 1:
        # Gaussian window
        a = np.exp(-2.0 * (x - 0.5) ** 2 / (0.1 * 0.1))
        a = np.maximum(a, 0.01)
    elif mode == 2:
        # Sinc-like (main lobe of Dirichlet kernel)
        k = np.arange(n) - n // 2
        a = np.sinc(k / (n * 0.1))
        a = np.abs(a)
        a = np.maximum(a, 0.01)
    elif mode == 3:
        # Flat with tapering
        taper = 0.1
        a = np.ones(n)
        for i in range(int(n * taper)):
            a[i] = i / (n * taper) + 0.1
            a[n - 1 - i] = a[i]
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
        a = a + 0.3 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    return a / np.sum(a)

        # Optimize: EGD + LP
        alloc1 = min(remaining * 0.04, 15.0)
        a_opt = energy_gradient_descent(a_r, n_iters=5000, time_budget=alloc1)

        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc2 = min(remaining * 0.06, 25.0)
        a_opt = lp_alternating(a_opt, time_budget=alloc2)

        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
