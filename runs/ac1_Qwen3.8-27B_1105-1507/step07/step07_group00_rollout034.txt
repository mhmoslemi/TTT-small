import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def grad_conv_k(a, k):
    """
    Gradient of (a*a)[k] with respect to a.
    grad[j] = a[k-j] (if valid) + a[j-k] (if valid)
    """
    n = len(a)
    g = np.zeros(n)
    # For each j in [0, n-1]:
    #   grad[j] = a[k-j] if 0 <= k-j < n, else 0
    #            + a[j-k] if 0 <= j-k < n, else 0
    j = np.arange(n)
    idx1 = k - j  # k - j
    idx2 = j - k  # j - k
    
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    
    g[mask1] += a[idx1[mask1]]
    g[mask2] += a[idx2[mask2]]
    return g

        return np.zeros(n), 0

    thresh = max_val * 0.5  # peaks above half the max

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:

    elif mode == 1:

    elif mode == 2:

        x = np.arange(n)

        a = a / a.sum()
    elif mode == 3:
        # Inverted U (peak in center)
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = 1.0 - ((x - center) / center) ** 2
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.sqrt(x * (1 - x))
        a = a / a.sum()
    elif mode == 5:
        # Smooth random
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Exponential decay from edges
        x = np.arange(n)
        center = (n - 1) / 2.0
        dist = np.abs(x - center) / center
        a = np.exp(dist)  # Higher at edges
        a = a / a.sum()
    elif mode == 7:

        a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2) + 0.01
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:

    x_old = np.linspace(0, 1, n)

def multi_peak_subgradient(a, num_peaks=10, alpha=2.0):
    """
    Compute subgradient using all significant peaks with power-weighted importance.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    if max_val < 1e-15:
        return np.zeros(n), 0
    
    # Weight peaks by (conv[k] / max_val)^alpha
    weights = (conv / max_val) ** alpha
    weights /= weights.sum()
    
    g = np.zeros(n)
    j = np.arange(n)
    
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        idx1 = k - j
        idx2 = j - k
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad_k = np.zeros(n)
        grad_k[mask1] += a[idx1[mask1]]
        grad_k[mask2] += a[idx2[mask2]]
        g += weights[k] * grad_k
    
    return g, int(np.argmax(conv))

"""
Optimized sequence search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp continuation, FISTA acceleration, mirror descent,

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_and_obj(a):
    """Returns (conv, obj) for normalized a."""
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    conv = compute_conv(a_norm)
    obj = 2.0 * n * float(np.max(conv))
    return conv, obj


def grad_conv_k(a, k):
    """
    Gradient of (a*a)[k] with respect to a.
    grad[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    """
    n = len(a)
    j = np.arange(n)
    idx1 = k - j
    idx2 = j - k
    g = np.zeros(n)
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask1] += a[idx1[mask1]]
    g[mask2] += a[idx2[mask2]]
    return g


def logsumexp_grad(a, t):
    """
    Gradient of (1/t)*log(sum_k exp(t*(a*a)[k])).
    This is a smooth approximation to the subgradient of max_k (a*a)[k].
    """
    n = len(a)
    conv = compute_conv(a)
    # Shift for numerical stability
    max_conv = float(np.max(conv))
    if max_conv < 1e-15:
        return np.zeros(n), 0
    shifted = t * (conv - max_conv)
    # Clip for stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    
    g = np.zeros(n)
    j = np.arange(n)
    # Only process significant peaks
    threshold = 1e-8
    active = np.where(weights > threshold)[0]
    
    for k in active:
        k = int(k)
        idx1 = k - j
        idx2 = j - k
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        g[mask1] += weights[k] * a[idx1[mask1]]
        g[mask2] += weights[k] * a[idx2[mask2]]
    
    return g, int(np.argmax(conv))


def topK_subgradient(a, K=5):
    """Subgradient using top-K peaks with equal weighting."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    g = np.zeros(n)
    for k in top_idx:
        g += grad_conv_k(a, int(k))
    g /= K
    return g, int(top_idx[0])


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_logsumexp(a, time_budget=100.0, t_start=2.0, t_end=20.0, verbose=False):
    """
    FISTA with log-sum-exp continuation.
    Starts with smooth t_start, gradually increases to t_end.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t = t_start
    t_k = 1.0
    y = a.copy()
    step_size = 1.0 / 2.0  # 1/L where L <= 2
    total_time = time_budget
    
    it = 0
    stall = 0
    patience = 400
    
    while time.monotonic() - start < time_budget:
        elapsed_frac = (time.monotonic() - start) / total_time
        # Continue t from t_start to t_end over the course of the run
        current_t = t_start + (t_end - t_start) * min(1.0, elapsed_frac * 1.5)
        
        g, k_star = logsumexp_grad(y, current_t)
        
        z = y - step_size * g
        x_new = project_simplex(z)
        
        obj_new = compute_obj(x_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
            step_size = min(step_size * 1.002, 0.8)
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 7919 + n)
                scale = min(0.3, 0.02 + 0.01 * (stall // patience))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                y = a.copy()
                t_k = 1.0
                step_size = 0.5
                stall = 0
            else:
                step_size = max(0.05, step_size * 0.998)
        
        a = x_new
        t_k = t_new
        it += 1
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTA-LS it{it}: obj={best_obj:.8f}, t={current_t:.1f}, step={step_size:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a


def fista_true_max(a, time_budget=100.0, K=5, verbose=False):
    """
    FISTA with true max subgradient (top-K peaks).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t_k = 1.0
    y = a.copy()
    step_size = 0.4
    prev_a = a.copy()
    prev_g = None
    prev_step = step_size
    
    it = 0
    stall = 0
    patience = 500
    
    while time.monotonic() - start < time_budget:
        g, k_star = topK_subgradient(y, K=K)
        
        # Barzilai-Borwein step size adjustment
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-12 and abs(sy) > 1e-12:
                bb_step1 = ss / sy if sy > 0 else step_size
                bb_step2 = ss / abs(sy) if sy < 0 else step_size
                new_step = bb_step1 if ss / abs(sy) > 0 else bb_step2
                step_size = 0.5 * step_size + 0.5 * min(max(new_step, 0.05), 0.8)
        
        z = y - step_size * g
        x_new = project_simplex(z)
        
        obj_new = compute_obj(x_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 104729 + n)
                scale = min(0.25, 0.02 + 0.01 * (stall // patience))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                y = a.copy()
                t_k = 1.0
                step_size = 0.4
                stall = 0
            else:
                step_size = max(0.05, step_size * 0.999)
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = x_new
        t_k = t_new
        it += 1
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTA-MX it{it}: obj={best_obj:.8f}, step={step_size:.4f}, t={t_k:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a


def mirror_descent_optimize(a, time_budget=100.0, K=5, verbose=False):
    """
    Mirror descent (multiplicative updates) on the probability simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n) * 1.5
    stall = 0
    momentum = 0.0
    v = np.zeros(n)
    
    it = 0
    while time.monotonic() - start < time_budget:
        g, k_star = topK_subgradient(a, K=K)
        
        v = momentum * v + g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = max(1e-6, lr * 0.9995)
            momentum = min(0.85, momentum + 0.003)
        else:
            stall += 1
            if stall > 400:
                rng = np.random.default_rng(it * 31 + n)
                scale = min(0.3, 0.03 + 0.01 * (stall // 400))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = max(1e-6, lr * 0.6)
                stall = 0
                v = np.zeros(n)
                momentum = 0.0
            else:
                lr = max(1e-6, lr * 0.997)
        
        it += 1
        
        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, mom={momentum:.3f}, elapsed={elapsed:.1f}s")
    
    return best_a


def symmetrize(a):
    """Symmetrize: a = (a + flip(a)) / 2."""
    return (a + a[::-1]) / 2.0


def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: move mass from peak contributors to non-contributors.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget and it < 200:
        k_star = int(np.argmax(conv))
        improved = False
        
        # Find contributors to the peak
        contrib_pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib_pairs.append((a[i] * a[j], i, j))
        contrib_pairs.sort(reverse=True)
        
        # Non-contributors
        contrib_set = set()
        for _, i, j in contrib_pairs:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [x for x in range(n) if x not in contrib_set]
        
        if not non_contrib:
            # Try all pairs if no non-contributors
            for i in range(0, n, max(1, n // 50)):
                for j in range(i + 1, n, max(1, n // 50)):
                    if time.monotonic() - start > time_budget:
                        return best_a
                    for frac in [0.01, 0.03, 0.05]:
                        delta = frac * a[i]
                        if a[i] - delta < 0:
                            continue
                        a[i] -= delta
                        a[j] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
            it += 1
            continue
        
        # Try moving mass from top contributors to non-contributors
        for contrib, i, j in contrib_pairs[:40]:
            if time.monotonic() - start > time_budget:
                return best_a
            if contrib < 1e-14:
                break
            
            for target in non_contrib[:10]:
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    delta = frac * max(a[i], a[j])
                    if a[i] >= a[j]:
                        if a[i] - delta < 0:
                            continue
                        a[i] -= delta
                        a[target] += delta
                        src, dst = i, target
                    else:
                        if a[j] - delta < 0:
                            continue
                        a[j] -= delta
                        a[target] += delta
                        src, dst = j, target
                    
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[dst] -= delta
                        break
        
        if not improved:
            break
        it += 1
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # U-shaped
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        a = 1.0 + 3.0 * ((x - center) / max(center, 1)) ** 2
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(0.02, 0.98, n)
        a = 1.0 / np.sqrt(x * (1 - x))
        a = a / a.sum()
    elif mode == 4:
        # Smooth random
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Exponential from edges
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        dist = np.abs(x - center) / max(center, 1)
        a = np.exp(1.5 * dist)
        a = a / a.sum()
    elif mode == 6:
        # Two peaks
        x = np.arange(n, dtype=np.float64)
        c1, c2 = n // 4, 3 * n // 4
        w = max(n // 8, 5)
        a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2) + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Power law from edges
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        dist = np.abs(x - center) / max(center, 1)
        a = dist ** 1.5 + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def quick_optimize(a, time_budget=30.0, K=5):
    """Quick optimization: FISTA logsumexp + mirror descent."""
    n = len(a)
    remaining = time_budget
    
    if remaining > 15:
        b = min(remaining * 0.5, 20.0)
        a = fista_logsumexp(a, time_budget=b, t_start=3.0, t_end=15.0)
        remaining = time_budget - b
    
    if remaining > 10:
        b = min(remaining * 0.6, 15.0)
        a2 = mirror_descent_optimize(a, time_budget=b, K=K)
        obj1 = compute_obj(a)
        obj2 = compute_obj(a2)
        if obj2 < obj1:
            a = a2
    
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: log-sum-exp continuation FISTA + mirror descent + n-sweep + coordinate refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_target = 800
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")
    
    # Phase 1: FISTA with log-sum-exp continuation on parent
    print(f"\n=== Phase 1: FISTA LogSumExp (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        budget = min(remaining * 0.2, 250.0)
        a_f = fista_logsumexp(best_seq, time_budget=budget, t_start=2.0, t_end=15.0, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  -> obj={best_score:.8f} **")
        else:
            print(f"  -> obj={obj_f:.8f} (no improvement)")
    
    # Phase 2: FISTA with true max
    print(f"\n=== Phase 2: FISTA True Max ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        budget = min(remaining * 0.15, 150.0)
        a_f = fista_true_max(best_seq, time_budget=budget, K=5, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  -> obj={best_score:.8f} **")
        else:
            print(f"  -> obj={obj_f:.8f}")
    
    # Phase 3: Mirror descent
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        budget = min(remaining * 0.15, 150.0)
        a_md = mirror_descent_optimize(best_seq, time_budget=budget, K=5, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  -> obj={best_score:.8f} **")
        else:
            print(f"  -> obj={obj_md:.8f}")
    
    # Try symmetrization
    a_sym = symmetrize(best_seq)
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 0.01:
        a_sym = a_sym / s
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  Symmetrized: obj={best_score:.8f} **")
    
    # Phase 4: Extended n-sweep
    print(f"\n=== Phase 4: N-Sweep ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    # Generate n candidates: nearby values plus some spread
    n_candidates = []
    for d in range(-300, 301, 25):
        n_try = n_current + d
        if 300 <= n_try <= 2000:
            n_candidates.append(n_try)
    # Add some specific values
    for n_try in [500, 600, 700, 750, 800, 850, 900, 950, 1000, 1100, 1200, 1500]:
        if 300 <= n_try <= 2000 and n_try not in n_candidates:
            n_candidates.append(n_try)
    
    n_candidates = sorted(set(n_candidates))
    # Sort by distance to current
    n_candidates.sort(key=lambda x: abs(x - n_current))
    
    tried_n = {n_current}
    n_tried = 0
    max_n_tries = 15
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= max_n_tries:
            break
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        n_tried += 1
        
        a_try = resample_sequence(best_seq, n_try)
        
        # Quick optimization
        alloc = min(remaining * 0.12, 50.0)
        a_opt = quick_optimize(a_try, time_budget=alloc, K=5)
        
        # Try symmetrization
        a_sym = symmetrize(a_opt)
        a_sym = np.maximum(a_sym, 0)
        s = a_sym.sum()
        if s > 0.01:
            a_sym = a_sym / s
            obj_sym = compute_obj(a_sym)
            if obj_sym < compute_obj(a_opt):
                a_opt = a_sym
        
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # Phase 5: Diverse restarts
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        
        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s = a_r.sum()
        if s > 0.01:
            a_r = a_r / s
        
        budget = min(remaining * 0.12, 70.0)
        a_opt = quick_optimize(a_r, time_budget=budget, K=5)
        
        # Symmetrize
        a_sym = symmetrize(a_opt)
        a_sym = np.maximum(a_sym, 0)
        s = a_sym.sum()
        if s > 0.01:
            a_sym = a_sym / s
            if compute_obj(a_sym) < compute_obj(a_opt):
                a_opt = a_sym
        
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
    
    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # FISTA logsumexp with high t
    if remaining > 25:
        budget = min(remaining * 0.2, 50.0)
        a_f = fista_logsumexp(best_seq, time_budget=budget, t_start=5.0, t_end=30.0)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final FISTA-LS: obj={best_score:.8f} **")
    
    # FISTA true max
    remaining = deadline - time.monotonic()
    if remaining > 20:
        budget = min(remaining * 0.2, 40.0)
        a_f = fista_true_max(best_seq, time_budget=budget, K=7)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final FISTA-MX: obj={best_score:.8f} **")
    
    # Mirror descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        budget = min(remaining * 0.2, 30.0)
        a_md = mirror_descent_optimize(best_seq, time_budget=budget, K=7)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")
    
    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 10:
        budget = min(remaining * 0.3, 25.0)
        a_cd = coordinate_descent_refine(best_seq, time_budget=budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
    
    # One more FISTA
    remaining = deadline - time.monotonic()
    if remaining > 15:
        budget = min(remaining * 0.4, 25.0)
        a_f = fista_true_max(best_seq, time_budget=budget, K=10)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final FISTA2: obj={best_score:.8f} **")
    
    # Symmetrization check
    a_sym = symmetrize(best_seq)
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 0.01:
        a_sym = a_sym / s
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  Final sym: obj={best_score:.8f} **")
    
    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
