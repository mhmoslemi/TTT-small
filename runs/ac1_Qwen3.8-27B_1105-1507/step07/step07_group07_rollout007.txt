def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)

"""
Hybrid spectral-temporal optimizer for low-autocorrelation non-negative sequences.
Combines spectral flattening (error reduction), L-BFGS-B gradient on smoothed objective,
LP alternating with golden-section line search, peak shaving, and multi-n exploration
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time


def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def spectral_flatten(a, n_iters=50, blend=0.3):
    """Error reduction: alternate between spectrum flattening and time-domain non-negativity."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    N = next_pow2(2 * n - 1)
    for it in range(n_iters):
        A = rfft(a, N)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(np.mean(P))
        P_target = P * (1.0 - blend) + P_mean * blend
        scale = np.sqrt(P_target / (P + 1e-30))
        A_new = A * scale
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    return a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def _line_search_obj(t, a_norm, g):
    """Objective at blending parameter t."""
    a_try = (1.0 - t) * a_norm + t * g
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 1e-10:
        return np.inf
    a_try = a_try / s
    return compute_obj(a_try)


def line_search_lp(a_norm, g):
    """Golden section search for best blending parameter t."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _line_search_obj(c, a_norm, g)
    fd = _line_search_obj(d, a_norm, g)
    for _ in range(35):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _line_search_obj(c, a_norm, g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _line_search_obj(d, a_norm, g)
    best_t = (lo + hi) / 2.0
    return best_t


def lp_alternate_improved(a, time_budget=60.0):
    """Improved LP alternating with golden-section line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()

        if remaining_step < 5.0:
            break
        g = solve_lp_minpeak(a)
        if g is None:
            break
        t = line_search_lp(a, g)
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break
    return best_a


def smoothed_obj_and_grad(x, tau):
    """Compute smoothed objective and gradient for a = exp(x)."""
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def _smoothed_fun(x, tau):
    val, _ = smoothed_obj_and_grad(x, tau)
    return val


def _smoothed_grad(x, tau):
    _, g = smoothed_obj_and_grad(x, tau)
    return g


def gradient_optimize(a_init, time_budget=60.0):
    """L-BFGS-B on smoothed objective with adaptive temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    taus = [10.0 / n, 5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
    x = np.log(a) - np.log(a).max()
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        res = minimize(_smoothed_fun, x, args=(tau,), jac=_smoothed_grad,
                       method='L-BFGS-B',
                       options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})
        if res.success or res.fun < _smoothed_fun(x, tau):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
    return best_a


def peak_shave(a, time_budget=30.0):
    """Iteratively shave the peak using subgradient descent with multiplicative updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()

    start = time.monotonic()
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        gmax = np.max(grad)
        if gmax < 1e-15:
            break
        for eta in [0.1, 0.05, 0.01]:
            a_new = best_a * np.exp(-eta * grad / gmax)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                break
    return best_a


def multi_peak_shave(a, time_budget=30.0):
    """Shave multiple near-peak lags simultaneously."""
    n = len(a)
    a = a.copy().astype(np.float64)

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def hanning_init(n):
    """Hanning window initialization."""
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / np.sum(a)


def gaussian_init(n):
    """Gaussian window initialization."""
    i = np.arange(n)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return a / np.sum(a)


def raised_cosine_init(n):
    """Raised cosine (Hanning-like) with wider support."""
    i = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / n))
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines spectral flattening, gradient optimization, LP alternating,
    peak shaving (single and multi), and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = hanning_init(800)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)

    # === Phase 0: Try different initializations at same n ===
    print("\n=== Phase 0: Initialization Comparison ===")
    inits = {
        'hanning': hanning_init(n_parent),
        'gaussian': gaussian_init(n_parent),
        'raised_cos': raised_cosine_init(n_parent),
    }
    for name, a_init in inits.items():
        a_try = spectral_flatten(a_init, n_iters=30, blend=0.2)
        obj = compute_obj(a_try)
        print(f"  {name}: obj={obj:.10f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_try.copy()

    # === Phase 1: Spectral flattening on best ===
    print("\n=== Phase 1: Spectral Flattening ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for blend in [0.1, 0.2, 0.3, 0.5]:
            a_spec = spectral_flatten(best_seq, n_iters=80, blend=blend)
            obj = compute_obj(a_spec)
            if obj < best_score:
                best_score = obj
                best_seq = a_spec.copy()
                print(f"  blend={blend}: obj={best_score:.10f} **")
            else:
                print(f"  blend={blend}: obj={obj:.10f}")

    # === Phase 2: Gradient optimization ===
    print("\n=== Phase 2: Gradient Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_grad = gradient_optimize(best_seq, time_budget=min(remaining * 0.3, 180.0))
        obj = compute_obj(a_grad)
        print(f"  After gradient: obj={obj:.10f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_grad.copy()

    # === Phase 3: LP Alternating ===
    print("\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_lp = lp_alternate_improved(best_seq, time_budget=min(remaining * 0.25, 150.0))
        obj = compute_obj(a_lp)
        print(f"  After LP: obj={obj:.10f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()

    # === Phase 4: Multi-peak shaving ===
    print("\n=== Phase 4: Multi-Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_mps = multi_peak_shave(best_seq, time_budget=min(remaining * 0.1, 40.0))
        obj = compute_obj(a_mps)
        print(f"  After multi-peak shave: obj={obj:.10f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_mps.copy()

    # === Phase 5: Single peak shaving ===
    print("\n=== Phase 5: Single Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_ps = peak_shave(best_seq, time_budget=min(remaining * 0.08, 30.0))
        obj = compute_obj(a_ps)
        print(f"  After peak shave: obj={obj:.10f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_ps.copy()

    # === Phase 6: Multi-n exploration ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        600, 700, 800, 900, 1000, 1100, 1200, 1400
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 1500 and n != n_parent]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        a_try = resample_sequence(best_seq, n_try)
        a_try = spectral_flatten(a_try, n_iters=20, blend=0.2)
        a_opt = gradient_optimize(a_try, time_budget=min(remaining * 0.08, 25.0))
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 7: Intensify on best ===
    print("\n=== Phase 7: Intensify ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_g2 = gradient_optimize(best_seq, time_budget=min(remaining * 0.25, 100.0))
        obj = compute_obj(a_g2)
        if obj < best_score:
            best_score = obj
            best_seq = a_g2.copy()
            print(f"  Gradient2: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_lp2 = lp_alternate_improved(best_seq, time_budget=min(remaining * 0.25, 80.0))
            obj = compute_obj(a_lp2)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp2.copy()
                print(f"  LP2: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_mps2 = multi_peak_shave(best_seq, time_budget=min(remaining * 0.15, 30.0))
            obj = compute_obj(a_mps2)
            if obj < best_score:
                best_score = obj
                best_seq = a_mps2.copy()
                print(f"  MPS2: obj={best_score:.10f}")

    # === Phase 8: Spectral + gradient cycle ===
    print("\n=== Phase 8: Spectral-Gradient Cycle ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for cycle in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_c = spectral_flatten(best_seq, n_iters=30, blend=0.15)
            a_c = gradient_optimize(a_c, time_budget=min(remaining * 0.25, 40.0))
            obj = compute_obj(a_c)
            if obj < best_score:
                best_score = obj
                best_seq = a_c.copy()
                print(f"  Cycle {cycle}: obj={best_score:.10f} **")
            else:
                print(f"  Cycle {cycle}: obj={obj:.10f}")

    # === Phase 9: Final polish ===
    print("\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_f = gradient_optimize(best_seq, time_budget=min(remaining * 0.3, 60.0))
        obj = compute_obj(a_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_f.copy()
            print(f"  Final grad: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_lp3 = lp_alternate_improved(best_seq, time_budget=min(remaining * 0.3, 45.0))
            obj = compute_obj(a_lp3)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp3.copy()
                print(f"  Final LP: obj={best_score:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_mps3 = multi_peak_shave(best_seq, time_budget=min(remaining * 0.4, 25.0))
            obj = compute_obj(a_mps3)
            if obj < best_score:
                best_score = obj
                best_seq = a_mps3.copy()
                print(f"  Final MPS: obj={best_score:.10f}")

    # Final gradient pass with very small tau
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final = gradient_optimize(best_seq, time_budget=remaining * 0.5)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Very final grad: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
