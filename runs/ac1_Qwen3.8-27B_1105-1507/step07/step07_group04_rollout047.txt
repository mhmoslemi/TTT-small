"""
Active-Set LP with FFT-Smoothed Gradient, Multi-Scale Construction, and Dual-Guided Updates.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovations over previous approach:
- Active-set LP (solves with only top-K binding constraints for speed)
- FFT-accelerated smoothed gradient for fast local improvements
- Multi-scale: start small, expand, re-optimize
- Dual-variable guided perturbation direction
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv)/(sum^2)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    return A


def solve_lp_full(a, solver_list=None):
    """Solve full LP: min M s.t. conv(a,g)[k]<=M, sum(g)=1, g>=0. Returns (g, M, duals)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    n_cons = 2 * n - 1

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)

    if solver_list is None:
        solver_list = [cp.PDLP, cp.SCS, cp.ECOS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value)
                    # Try to get duals (Lagrange multipliers for inequality constraints)
                    duals = None
                    try:
                        duals = np.zeros(n_cons)
                        for i, c in enumerate(cons[:-1]):
                            if c.dual_value is not None:
                                duals[i] = c.dual_value
                    except Exception:
                        pass
                    return g_val, M_val, duals
        except Exception:
            continue
    return None, None, None


def solve_lp_active(a, k_indices, solver_list=None):
    """Solve LP with only constraints at k_indices. Returns (g, M)."""
    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)
    A = A_full[k_indices, :]  # Only selected rows

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)

    if solver_list is None:
        solver_list = [cp.PDLP, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    return None, None


def fft_smoothed_gradient(a, top_k=20, temperature=5.0):
    """
    Compute smoothed gradient of max(conv(a,a)) using FFT.
    grad[j] = sum_k w[k] * (a[k-j] + a[j]) (with bounds)
    First term is cross-correlation: ifft(conj(FFT(w)) * FFT(a))
    """
    n = len(a)
    conv = compute_conv(a)
    # Smooth weights
    max_conv = np.max(conv)
    weights = np.exp(temperature * (conv - max_conv))
    weights = weights / np.sum(weights)

    # Compute grad[j] = sum_k w[k]*a[k-j] + sum_k w[k]*a[j]
    # Second term: a[j] * sum_k w[k] = a[j]
    # First term: cross-correlation of w and a
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    fw = rfft(weights, size)
    fa = rfft(a, size)
    # Cross-correlation: ifft(conj(FW) * FA) gives sum_k w[k]*a[k-j]
    corr = irfft(np.conj(fw) * fa, size)[:n]

    grad = corr + a  # + a[j]*sum(w) = a[j] since sum(w)=1
    return grad, conv, weights


def smoothed_gradient_descent(a, time_budget=100.0, lr=0.05, verbose=False):
    """Nesterov-accelerated smoothed gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.9
    x = a.copy()  # Nesterov lookahead
    temperature = 10.0
    top_k = 20

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        grad, conv, weights = fft_smoothed_gradient(x, top_k=top_k, temperature=temperature)
        grad_centered = grad - np.mean(grad)

        momentum = beta * momentum - lr * grad_centered
        new_a = a + momentum
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            new_a = a.copy()
            s_new = 1.0
        new_a = new_a / s_new

        # Nesterov update
        t_k = 2.0 / (it + 2.0)
        x = new_a + (1.0 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = new_a
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            if verbose and it % 200 == 0:
                print(f"      SGD iter {it}: obj={best_obj:.8f}")

        # Adaptive parameters
        if it % 100 == 99:
            lr = max(lr * 0.98, 0.001)
            temperature = max(temperature * 0.95, 2.0)

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_old = np.array([0.0, 1.0])
    else:
        x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)


def dual_guided_perturb(a, duals, n_perturb=10):
    """Perturb in direction guided by dual variables (most binding constraints)."""
    n = len(a)
    if duals is None or len(duals) == 0:
        return a
    # Find most binding constraints
    top_binding = np.argsort(duals)[::-1][:n_perturb]
    # Create perturbation that reduces these peaks
    rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
    perturbation = np.zeros(n)
    for k in top_binding:
        k = int(k)
        for j in range(n):
            ik = k - j
            if 0 <= ik < n:
                perturbation[j] += a[ik]
            perturbation[j] += a[j]
    perturbation = perturbation / (np.max(np.abs(perturbation)) + 1e-10)
    a_new = a * np.exp(-0.01 * perturbation + rng.normal(0, 0.01, n))
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting optimization from n={n_parent}, obj={best_score:.8f}")

    # === Phase 1: Quick smoothed gradient (120s) ===
    print(f"\n=== Phase 1: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_budget = min(120.0, remaining * 0.15)
        a_sg = smoothed_gradient_descent(best_seq, time_budget=sg_budget, lr=0.05, verbose=True)
        obj_sg = compute_obj(a_sg)
        print(f"  SGD result: obj={obj_sg:.8f}")
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 2: Active-set LP optimization (500s) ===
    print(f"\n=== Phase 2: Active-Set LP Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        lp_budget = min(500.0, remaining * 0.45)
        a_lp = active_set_lp_optimize(best_seq, time_budget=lp_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        print(f"  ActiveSet LP result: obj={obj_lp:.8f}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 3: Full LP polish (200s) ===
    print(f"\n=== Phase 3: Full LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        polish_budget = min(200.0, remaining * 0.2)
        a_polish = full_lp_polish(best_seq, time_budget=polish_budget, verbose=True)
        obj_polish = compute_obj(a_polish)
        print(f"  Full LP result: obj={obj_polish:.8f}")
        if obj_polish < best_score:
            best_score = obj_polish
            best_seq = a_polish.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 4: Multi-length exploration (250s) ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(100, n_best - 200),
            max(100, n_best - 100),
            n_best,
            n_best + 100,
            n_best + 200,
            n_best + 400,
        ]))
        n_options = [n for n in n_options if 100 <= n <= 3000]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if n_try == n_best:
                continue  # Already optimized
            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
                s = float(np.sum(a_try))
                if s > 1e-10:
                    a_try = a_try / s

            # Quick optimization: active-set LP + short gradient
            alloc_lp = min(30.0, remaining * 0.4)
            alloc_sg = min(20.0, remaining * 0.2)

            a_opt = active_set_lp_optimize(a_try, time_budget=alloc_lp)
            a_opt = smoothed_gradient_descent(a_opt, time_budget=alloc_sg, lr=0.03)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Second round of optimization on best (remaining) ===
    print(f"\n=== Phase 5: Second Optimization Round ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Active-set LP again (with fresh start from current best)
        alloc = min(remaining * 0.5, 150.0)
        a_round2 = active_set_lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_r2 = compute_obj(a_round2)
        print(f"  Round2 ActiveSet: obj={obj_r2:.8f}")
        if obj_r2 < best_score:
            best_score = obj_r2
            best_seq = a_round2.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 6: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.6, 80.0)
        a_final = full_lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        print(f"  Final polish: obj={obj_final:.8f}")
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 7: Random restarts (if time remains) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            scale = 0.02 + 0.015 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

            alloc = min(remaining * 0.35, 30.0)
            a_opt = active_set_lp_optimize(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
