"""
Active-set LP optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: cutting-plane active-set LP (solve small LPs with top-K constraints),
combined with aggressive multi-n exploration and dual-based directional updates.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_active_set(a, active_indices, time_limit=5.0):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for k in active_indices, sum(g)=1, g>=0.
    Returns (g, M, violated_indices) where violated_indices are indices not in 
    active_indices that violate the constraint.
    """
    n = len(a)
    a = a / np.sum(a)  # normalize
    K = len(active_indices)
    
    # Build the constraint matrix: K constraints of the form (a*g)[k_i] - M <= 0
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for row, k in enumerate(active_indices):
        # (a*g)[k] = sum_j a[j] * g[k-j] for valid j
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[row, j] = a[j]
        A_ub[row, n] = -1.0  # -M term
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))

Active-set LP optimizer with cutting planes for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: (1) cutting-plane active-set LP that solves small LPs with only K

import numpy as np

from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross-correlation: compute a*b convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            total_iters += 1

            # Periodically update active set to top-K of new peaks
            if it % 10 == 0:
                conv_new = compute_conv(a)
                K_cur = min(K, K_max)
                active_indices = list(np.argsort(conv_new)[-K_cur:])
                active_set = set(active_indices)

            if verbose and total_iters % 20 == 0:
                print(f"        AS-LP it{total_iters}: obj={best_obj:.9f}, K={K}, t={best_t_val:.3f}")
        else:
            stall += 1
            if stall > 15:
                # Perturb and reset
                rng = np.random.default_rng((int(time.monotonic() * 1e6) + it + n) % (2**31))
                scale = 0.01 + 0.005 * min(stall // 15, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                conv = compute_conv(a)
                K = min(K_init, 2 * n - 1)
                active_indices = list(np.argsort(conv)[-K:])
                active_set = set(active_indices)
                stall = 0
                last_t = 0.3

    return best_a


def smooth_max_grad_descent(a, time_budget=30.0, T_init=0.0002, T_min=0.00001):
    """Smooth max (log-sum-exp) gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.002
    momentum = 0.92
    v = np.zeros(n)
    stall = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -600, 600)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient: grad[i] = sum_j a[j] * weights[i+j]
        grad = np.correlate(weights, a, mode='valid')

        # Center for simplex
        grad_centered = grad - np.mean(grad)

        # Nesterov
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 200 == 0 and lr < 0.008:
                lr *= 1.01
        else:
            stall += 1
            if stall > 80:
                lr *= 0.65
                if lr < 1e-6:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 50:
            T = max(T_min, T_init * (0.997 ** it))

    return best_a


def dual_bound_check(a, y_distribution):
    """
    Compute a lower bound on the optimal max-convolution using dual certificate.
    y_distribution: probability vector of length 2n-1 (dual variables)
    Returns: lower bound on M* = max_k (a*g)_k over sum(g)=1, g>=0
    """
    n = len(a)
    a = a / np.sum(a)
    # (A^T y)_i = sum_k y_k * a[k-i] for valid k-i
    # = sum_k y[k] * a[k-i] where 0 <= k-i < n
    # For each i: sum over k from i to i+n-1 of y[k] * a[k-i]
    # This is a cross-correlation-like computation
    # (A^T y)_i = sum_{j=0}^{n-1} a[j] * y[i+j]
    # = same as gradient formula!
    bound = float(np.min(np.correlate(y_distribution, a, mode='valid')))
    return bound


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate random non-negative sequence (normalized to sum=1)."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Smooth random
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        center = n // 2
        width = max(2, n // 6)
        x = np.arange(n, dtype=float)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a += 0.4 * np.exp(-((x - center * 0.5) ** 2) / (2 * (width // 2 + 1) ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.2
        a = a / np.sum(a)
    elif mode == 5:
        # Triangle
        x = np.arange(n) / max(n - 1, 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / np.sum(a)
    elif mode == 6:
        # Random blocks
        n_blocks = rng.integers(3, 10)
        block_size = n // n_blocks
        a = np.zeros(n)
        for b in range(n_blocks):
            start = b * block_size
            end = min(start + block_size, n)
            a[start:end] = rng.uniform(0.1, 1.0)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: active-set LP alternating + smooth max gradient + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={obj:.9f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.9f}")

    # === Phase 1: Intensive Active-Set LP ===
    print(f"\n=== Phase 1: Active-Set LP (n={len(best_seq)}) ===")
    a = best_seq.copy()
    n_main = len(a)

    # Multiple rounds with increasing K
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        K_init = 10 + round_idx * 5
        K_max = 50 + round_idx * 50
        alloc = min(remaining * 0.3 / (4 - round_idx), 120.0)
        a_new = active_set_lp_alternating(a, time_budget=alloc, K_init=K_init, K_max=K_max, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  AS-LP Round {round_idx}: obj={best_score:.9f} (K_init={K_init})")
        else:
            print(f"  AS-LP Round {round_idx}: obj={obj:.9f} (no improvement)")

    # === Phase 2: Smooth Max Gradient ===
    print(f"\n=== Phase 2: Smooth Max Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(45.0, remaining * 0.2)
        a_sm = smooth_max_grad_descent(a, time_budget=alloc, T_init=0.0002, T_min=0.00001)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score - 1e-12:
            best_score = obj_sm
            best_seq = a_sm.copy()
            a = a_sm.copy()
            print(f"  SmoothMax: obj={best_score:.9f}")
        else:
            print(f"  SmoothMax: obj={obj_sm:.9f}")

        # Lower temperature
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(30.0, remaining * 0.15)
            a_sm2 = smooth_max_grad_descent(a, time_budget=alloc, T_init=0.00005, T_min=0.000005)
            obj_sm2 = compute_obj(a_sm2)
            if obj_sm2 < best_score - 1e-12:
                best_score = obj_sm2
                best_seq = a_sm2.copy()
                a = a_sm2.copy()
                print(f"  SmoothMax (low T): obj={best_score:.9f}")

    # === Phase 3: More LP after gradient ===
    print(f"\n=== Phase 3: Post-Gradient LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.3)
        a_lp = active_set_lp_alternating(a, time_budget=alloc, K_init=30, K_max=200, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  Post-LS LP: obj={best_score:.9f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        n_best + 600,
        500, 700, 1000, 1300, 1600, 2000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # LP optimization
        alloc_lp = min(remaining * 0.18, 40.0)
        a_opt = active_set_lp_alternating(a_try, time_budget=alloc_lp, K_init=15, K_max=100)
        obj_try = compute_obj(a_opt)

        # Quick gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.02:
            alloc_sm = min(remaining * 0.08, 15.0)
            a_opt2 = smooth_max_grad_descent(a_opt, time_budget=alloc_sm)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.9f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.9f}")

    # === Phase 5: Random Restarts ===
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        init_mode = r % 7
        n_r = n_best + int(rng.integers(-100, 101))
        n_r = max(300, min(2000, n_r))
        a_r = random_init(n_r, rng, mode=init_mode)

        alloc_r = min(remaining * 0.1, 25.0)
        a_opt = active_set_lp_alternating(a_r, time_budget=alloc_r * 0.7, K_init=10, K_max=80)

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = smooth_max_grad_descent(a_opt, time_budget=min(alloc_r * 0.3, remaining * 0.3))

        obj_r = compute_obj(a_opt)
        if obj_r < best_score - 1e-12:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.9f} **")
        else:
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_r:.9f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()

    # Large active set LP
    if remaining > 20:
        alloc = min(remaining * 0.35, 60.0)
        a_lp = active_set_lp_alternating(a_final, time_budget=alloc, K_init=40, K_max=400, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.9f}")

    # Smooth max very low T
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_sm = smooth_max_grad_descent(a_final, time_budget=min(remaining * 0.25, 30.0),
                                       T_init=0.00003, T_min=0.000003)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score - 1e-12:
            best_score = obj_sm
            best_seq = a_sm.copy()
            a_final = a_sm.copy()
            print(f"  Final SmoothMax: obj={best_score:.9f}")

    # One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_lp = active_set_lp_alternating(a_final, time_budget=min(remaining * 0.25, 30.0),
                                         K_init=50, K_max=500, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.9f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")

    return result
