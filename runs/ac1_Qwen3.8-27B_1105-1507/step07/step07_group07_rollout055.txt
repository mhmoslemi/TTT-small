    grad = conv_smoothed_grad(a, w) / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """Optimize using L-BFGS-B on smoothed objective."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    taus = [20.0/n, 10.0/n, 5.0/n, 2.5/n, 1.0/n, 0.5/n, 0.25/n, 0.1/n]
    
    x = np.log(a) - np.log(a).max()
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g
        
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 100})
        
        if res.success or res.fun < fun(x):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.5f}: obj={best_obj:.10f}")
    
    return best_a


def frank_wolfe_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Frank-Wolfe with away-steps for min 2*n*max(conv(a)) s.t. a in simplex.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Support set (indices where a > 0)
    support = set(np.where(a > 1e-10)[0].tolist())
    
    k_iter = 0
    while time.monotonic() - start < time_budget:
        k_iter += 1
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        g = conv_subgradient(a, k_star) * (2.0 * n)
        
        # Towards step: min_i g[i]
        j_towards = int(np.argmin(g))
        
        # Away step: max_{i in support} a[i] * g[i]
        if support:
            support_arr = np.array(list(support))
            away_scores = a[support_arr] * g[support_arr]
            j_away_local = int(np.argmax(away_scores))
            j_away = int(support_arr[j_away_local])
            
            # Combined step direction
            if j_away == j_towards:
                # Only towards step
                direction = np.zeros(n)
                direction[j_towards] = 1.0
                max_t = 1.0
            else:
                direction = np.zeros(n)
                direction[j_towards] = 1.0
                direction[j_away] = -a[j_away]
                max_t = 1.0 / a[j_away] if a[j_away] > 1e-15 else 1.0
                max_t = min(max_t, 1.0)
            
            # Line search
            best_t = 2.0 / (k_iter + 2.0)
            best_f = np.inf
            for t in [best_t, 0.5 * best_t, 2.0 * best_t, 0.1, 0.3, 0.5, 0.7, 1.0]:
                t = min(t, max_t)
                if t <= 0:
                    continue
                a_try = a + t * direction
                if np.any(a_try < -1e-15):
                    continue
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                f_try = compute_obj(a_try)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t
            
            a_new = a + best_t * direction
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    # Update support
                    support = set(np.where(a > 1e-10)[0].tolist())
        
        # Periodic restart for diversity
        if k_iter % 5000 == 0:
            support = set(np.where(a > 1e-10)[0].tolist())
    
    if verbose:
        print(f"    FW: {k_iter} iters, obj={best_obj:.10f}")
    return best_a


def coordinate_descent(a_init, time_budget=30.0, verbose=False):
    """
    Coordinate descent: iterate over coordinates, doing 1D optimization for each.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    n_iter = 0
    while time.monotonic() - start < time_budget:
        n_iter += 1
        i = int(rng.integers(0, n))
        old_val = a[i]
        
        # Try several values
        candidates = []
        # Geometric multiples
        for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
            candidates.append(old_val * factor)
        # Small perturbations
        for delta in [-0.1, -0.05, 0.05, 0.1] * (old_val + 1.0/n):
            candidates.append(max(0, old_val + delta))
        # Zero and max
        candidates.append(0.0)
        candidates.append(s / (len(support) if (support := np.where(a > 1e-10)[0]) else [0]))
        
        best_candidate = old_val
        best_f = compute_obj(a)
        
        for c in candidates:
            c = max(0.0, c)
            a[i] = c
            s_try = float(np.sum(a))
            if s_try < 1e-10:
                a[i] = old_val
                continue
            a_norm = a / s_try
            obj_try = compute_obj(a_norm)
            if obj_try < best_f:
                best_f = obj_try
                best_candidate = c
        
        a[i] = best_candidate
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    if verbose:
        print(f"    CD: {n_iter} iters, obj={best_obj:.10f}")
    return best_a


def peak_shaving(a_init, time_budget=30.0, verbose=False):
    """
    Iteratively identify peaks of conv and reduce contributing elements.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find all contributing indices
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        
        # Reduce elements that contribute to the peak
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                # Reduce a[i] proportional to a[i]*a[j]
                a[i] *= (1.0 - eta * a[j])
        
        a = np.maximum(a, 1e-30)
        a = a / np.sum(a)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            eta = max(eta * 0.999, 0.01)
        else:
            a = best_a.copy()
            eta *= 0.9
    
    if verbose:
        print(f"    Peak shaving: obj={best_obj:.10f}")
    return best_a
