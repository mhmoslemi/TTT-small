    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    w = w / w.sum() * s
    return w


def subgradient_optimize(a, n_iters=5000, time_budget=60.0, top_m=15, temp_scale=0.2):
    """Proximal subgradient with adaptive step size and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Parameters
    alpha = 0.1
    momentum = 0.7
    v = np.zeros(n)
    stall_count = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        grad = subgradient_of_max_conv(a, conv, top_m=top_m, temp_scale=temp_scale)

        # Normalize gradient for stable steps
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        # Adaptive step size with backtracking
        best_step_obj = best_obj
        best_step_a = best_a.copy()
        for alpha_try in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            a_try = a - alpha_try * v / v_norm
            a_try = project_simplex(a_try, 1.0)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()
                break

        if best_step_obj < best_obj:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_a
            alpha = min(alpha * 1.05, 1.0)
            stall_count = 0
        else:
            a = best_a.copy()
            alpha *= 0.8
            stall_count += 1
            if stall_count > 30:
                # Perturb to escape
                rng = np.random.default_rng(it * 13 + 7)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = project_simplex(a, 1.0)
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                alpha = 0.05
                stall_count = 0

        # Periodically refresh top_m and temp
        if it % 200 == 199:
            temp_scale = max(0.05, temp_scale * 0.95)

    return best_a


def lp_direction(a, use_cvxpy=True):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)

    # Build constraint matrix: (a*g)[k] = sum_j a[k-j]*g[j] for valid j
    # (a*g)[k] = sum_{j=0}^{min(k,n-1)} a_norm[max(0,k-n+1):k+1][::-1] dot g[valid]
    # More precisely: (a*g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a_norm[k-j]*g[j]
    # For fixed k: j ranges from max(0, k-n+1) to min(n-1, k)
    # (a*g)[k] = sum_{j=lo}^{hi} a_norm[k-j] * g[j]

    # This is a matrix A where A[k,j] = a_norm[k-j] if 0<=k-j<n, else 0
    # A is (2n-1) x n
    # For efficiency, use the Toeplitz structure
    # A[k,j] = a_norm[k-j] for max(0,k-n+1) <= j <= min(n-1,k)

    if use_cvxpy:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()

        # Build constraint A@g <= M
        # A is sparse Toeplitz-like
        # For each k, the row is a_norm[k-:k+1] reversed, padded
        A_rows = []
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            row = np.zeros(n)
            row[lo:hi+1] = a_norm[k-hi:k-lo+1][::-1]
            A_rows.append(row)
        A_mat = np.array(A_rows)

        constraints = [A_mat @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solvers = [cp.MOSEK, cp.SCS, cp.CLARABEL, cp.ECOS, cp.PDLP]
        for solver in solvers:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    # Fallback: scipy linprog
    try:
        # min M = min sum(c*x) where x = [g, M], c = [0,...,0, 1]
        c = np.zeros(n + 1)
        c[-1] = 1.0

        # (a*g)[k] - M <= 0 for all k
        # A_ub[k, :n] = row of A_mat, A_ub[k, n] = -1
        A_ub = np.zeros((2 * n - 1, n + 1))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_ub[k, lo:hi+1] = a_norm[k-hi:k-lo+1][::-1]
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=120.0, max_iters=200, verbose=False):
    """LP alternating with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = lp_direction(a)
        if g_lp is None:
            if verbose:
                print(f"        LP failed at iter {it}")
            break

        # Line search
        best_t_obj = best_obj
        best_t_a = best_a.copy()

        # Try various step sizes
        candidate_ts = np.array([0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0])
        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_a
            stall = 0
            if verbose and it % 5 == 0:
                print(f"        LP it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = project_simplex(a, 1.0)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

    return best_a


def peak_equalization(a, n_iters=500, time_budget=30.0):
    """
    Targeted peak equalization: identify top peaks and find direction
    that reduces them simultaneously using a small QP.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = np.max(conv)

        # Find peaks that are within 5% of max
        active_mask = conv >= max_val * 0.95
        active_idx = np.where(active_mask)[0]

        if len(active_idx) == 0:
            break
        if len(active_idx) == 1:
            # Just use subgradient for single peak
            break

        # Build small system: for each active peak k, 
        # (a*a)[k] = sum_i a[i]*a[k-i]
        # We want to perturb a by delta to reduce all active peaks
        # Linearize: (a+εδ)*(a+εδ)[k] ≈ (a*a)[k] + 2ε*(a*delta)[k]
        # We want (a*delta)[k] < 0 for all active k
        # Find delta of minimum norm satisfying this

        # This is: min ||delta||^2 s.t. A_active @ delta <= -eta, sum(delta)=0
        # where A_active[k, j] = a[k-j] for valid j (rows for active peaks)

        n_active = len(active_idx)
        if n_active > 20:
            active_idx = active_idx[np.argsort(conv[active_idx])[-20:]]
            n_active = 20

        A_active = np.zeros((n_active, n))
        for i, k in enumerate(active_idx):
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            A_active[i, lo:hi+1] = a[ki-hi:ki-lo+1][::-1]

        # Solve: min ||delta||^2 s.t. A_active @ delta <= -0.001, sum(delta) = 0
        # Using cvxpy
        try:
            delta = cp.Variable(n)
            obj = cp.sum_squares(delta)
            cons = [A_active @ delta <= -0.001, cp.sum(delta) == 0]
            prob = cp.Problem(cp.Minimize(obj), cons)
            prob.solve(solver=cp.SCS, verbose=False)
            if delta.value is not None:
                delta_val = delta.value
                # Line search
                for eps in [0.01, 0.005, 0.002, 0.001]:
                    a_try = a + eps * delta_val
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        break
        except Exception:
            pass

    return best_a


def nelder_mead_polish(a, time_budget=30.0, max_iters=200):
    """Nelder-Mead style final polish using evaluate_sequence."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Scale for NM
    scale = 0.01

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Try perturbations in random directions
        rng = np.random.default_rng(it * 17 + n)
        improved = False

        for _ in range(5):
            direction = rng.normal(0, 1, n)
            direction = direction - direction.mean()  # Keep sum approximately 0
            for step in [scale, scale * 0.5, scale * 0.1]:
                a_try = a + step * direction
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            scale *= 0.7
            if scale < 1e-6:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.arange(n) / (n - 1)
    x_new = np.arange(new_n) / (new_n - 1)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Smooth random
        freqs = rng.uniform(0.1, 5.0, 6)
        phases = rng.uniform(0, 2 * np.pi, 6)
        amps = rng.uniform(0.1, 0.5, 6)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 2:
        # Asymmetric bump
        center = rng.uniform(0.3, 0.7) * n
        width = rng.uniform(0.1, 0.4) * n
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 3:
        # Power law (favors one end)
        x = np.arange(1, n + 1)
        power = rng.uniform(0.5, 3.0)
        a = x ** power
        if rng.random() > 0.5:
            a = a[::-1]
        a = a / a.sum()
    elif mode == 4:
        # Two bumps (asymmetric)
        c1 = rng.uniform(0.2, 0.4) * n
        c2 = rng.uniform(0.6, 0.8) * n
        w1 = rng.uniform(0.05, 0.2) * n
        w2 = rng.uniform(0.05, 0.2) * n
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + 0.7 * np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    else:
        # Slightly perturbed uniform
        a = np.ones(n) + rng.normal(0, 0.3, n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining subgradient, LP alternating, peak equalization,
    multi-n exploration, and Nelder-Mead polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Fast subgradient on parent ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 1: Subgradient (n={len(best_seq)}) ===")
        alloc = min(remaining * 0.2, 120.0)
        a_sg = subgradient_optimize(best_seq, n_iters=10000, time_budget=alloc, top_m=15, temp_scale=0.15)
        obj = compute_obj(a_sg)
        print(f"  Subgradient 1: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()

        n_rounds = min(3, int((remaining - 30) / 60) + 1)

            remaining = deadline - time.monotonic()

            alloc = min(remaining * 0.2, 100.0)
            a_lp = lp_alternating(best_seq, time_budget=alloc, max_iters=300, verbose=True)
            obj = compute_obj(a_lp)
            print(f"  LP round {rnd}: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()

    # === Phase 3: Peak equalization ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: Peak Equalization ===")
        alloc = min(remaining * 0.08, 45.0)
        a_pe = peak_equalization(best_seq, n_iters=500, time_budget=alloc)
        obj = compute_obj(a_pe)
        print(f"  Peak eq: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_pe.copy()

    # === Phase 4: Another subgradient round ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Subgradient Round 2 ===")
        alloc = min(remaining * 0.1, 80.0)
        a_sg3 = subgradient_optimize(best_seq, n_iters=8000, time_budget=alloc, top_m=10, temp_scale=0.1)

        print(f"  Subgradient 3 (sharp): obj={obj:.8f}")

            best_score = obj
            best_seq = a_sg3.copy()

    # === Phase 5: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:

        n_best = len(best_seq)
        n_options = sorted(set([
            max(300, n_best - 200),

        n_options = [n for n in n_options if abs(n - n_best) >= 50]

                a_r = project_simplex(a_r, 1.0)

            if obj < best_score:
                best_score = obj
                best_seq = a_md.copy()

        # Peak equalization
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.2, 25.0)
            a_pe = peak_equalization(best_seq, n_iters=300, time_budget=alloc)
            obj = compute_obj(a_pe)
            print(f"  Final PE: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_pe.copy()

        # Nelder-Mead
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.2, 20.0)
            a_nm = nelder_mead_polish(best_seq, time_budget=alloc, max_iters=100)
            obj = compute_obj(a_nm)
            print(f"  Final NM: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_nm.copy()

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.35, 40.0)
            a_last = lp_alternating(best_seq, time_budget=alloc, max_iters=100, verbose=False)
            obj = compute_obj(a_last)
            print(f"  Final LP2: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_last.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
