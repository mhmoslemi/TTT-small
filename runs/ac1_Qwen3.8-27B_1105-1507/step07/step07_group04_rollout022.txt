import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def compute_conv(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def get_subgradient(a, k_indices):

    n = len(a)

            # a[i] appears as a[j] when j=i, contributing a[i+k] (if i+k < n)
            # a[i] appears as a[j+k] when j+k=i, i.e., j=i-k, contributing a[i-k] (if i-k >= 0)
            if i + k < n:  # wait, this is per-element, let me vectorize

    # Vectorized version:

def nesterov_subgradient(a, time_budget=100.0, top_k=10, verbose=False):

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # lookahead point
    t_k = 1.0
    step_size = 0.05
    step_min = 0.001
    no_improve = 0
    patience = 30

        s_y = float(np.sum(y_new))
        if s_y > 1e-10:

        # x_{k+1} = project(a - step_size * grad)

        sy = float(np.sum(y))

        a = x_new

            no_improve = 0

def active_set_lp(a, k_indices, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in k_indices, sum(g)=1, g>=0.
    Only use the active constraints specified in k_indices (plus maybe all others).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Build only the rows for the active constraints
    # conv(a,g)[k] = sum_j a[j] * g[j+k] + ... 
    # Actually conv(a,g)[k] = sum_j a[j] * g[k-j] where 0<=j<n, 0<=k-j<n
    # = sum_j a[j] * g[k-j] for j in [max(0,k-n+1), min(n-1, k)]
    
    # Let's build the matrix for the active constraints only
    A_active = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_active[idx, j] = a[k - j]
    
    # LP: min M s.t. A_active @ g <= M, sum(g) = 1, g >= 0
    # Variables: g[0..n-1], M
    # A_ub: A_active for g, -I for M
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((len(k_indices), n + 1))
    A_ub[:, :n] = A_active
    A_ub[:, n] = -1.0  # A_active @ g - M <= 0
    
    b_ub = np.zeros(len(k_indices))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def full_lp(a, verbose=False):

            print(f"      Nesterov: it={it}, obj={best_obj:.8f}, step={step_size:.4f}, t={elapsed:.0f}s")

    return best_a


def active_set_lp_with_violation(a, k_indices, max_iter=20):
    """
    Active-set LP: solve min M s.t. conv(a,g)[k]<=M for active k, sum(g)=1, g>=0.
    Then check for violations among non-active constraints and add them.
    """
    n = len(a)
    a = a / np.sum(a)
    active = list(k_indices)

    for iteration in range(max_iter):
        K = len(active)
        A_act = np.zeros((K, n))
        for idx, k in enumerate(active):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A_act[idx, j] = a[k - j]

        c = np.zeros(n + 1)
        c[n] = 1.0
        A_ub = np.zeros((K, n + 1))
        A_ub[:, :n] = A_act
        A_ub[:, n] = -1.0
        b_ub = np.zeros(K)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                return None, None
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs < 1e-10:
                return None, None
            g_val = g_val / gs
            M_val = float(res.x[n])
        except Exception:
            return None, None

        # Check for violations among non-active constraints
        conv_ig = compute_conv(g_val)
        # But we need conv(a, g_val), not conv(g_val, g_val)
        # conv(a, g)[k] = sum_j a[j] * g[k-j]
        # Let's compute this properly
        ag_conv = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            ag_conv[k] = np.dot(a[j_start:j_end + 1], g_val[k - j_end:k - j_start + 1][::-1])

        # Find most violated non-active constraint
        active_set = set(int(x) for x in active)
        violations = []
        for k in range(2 * n - 1):
            if k not in active_set and ag_conv[k] > M_val + 1e-10:
                violations.append((k, ag_conv[k] - M_val))

        if not violations:
            return g_val, M_val

        # Add the most violated constraint
        violations.sort(key=lambda x: -x[1])
        active.append(violations[0][0])

    # Return even with violations (best effort)
    return g_val, M_val


def lp_alternate_improved(a, time_budget=200.0, verbose=False):
    """LP alternating with proper active-set management and line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = symmetrize(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    k_count = 8

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_indices = np.argsort(conv)[::-1][:k_count]

        g_lp, M_lp = active_set_lp_with_violation(a, k_indices, max_iter=10)
        if g_lp is None:
            stall += 1
            if stall > 15:
                # Perturb and retry
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = symmetrize(a)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
            continue

        # Fine line search
        best_t = 0
        best_t_obj = best_obj
        t_grid = np.concatenate([
            np.linspace(0.05, 0.5, 8),
            np.linspace(0.5, 1.0, 6)
        ])
        for t in t_grid:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = symmetrize(a_try)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            a_try = (1.0 - best_t) * a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = symmetrize(a_try)
            a_try = a_try / np.sum(a_try)
            best_a = a_try.copy()
            a = a_try
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, K={k_count}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall == 5:
                k_count = min(k_count + 4, 40)
            elif stall == 10:
                k_count = max(k_count - 4, 5)
            if stall > 20:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 20, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = symmetrize(a)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                a = best_a.copy()
                stall = 0
                k_count = 8

    return best_a


def slsqp_refine(a, time_budget=60.0, verbose=False):
    """
    SLSQP-based refinement minimizing a smooth proxy for the max peak.
    Uses log-sum-exp smoothing of the top peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = symmetrize(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use log-sum-exp with increasing beta
    for beta in [5.0, 10.0, 20.0, 50.0]:
        def objective(x):
            x = np.maximum(x, 0)
            sx = float(np.sum(x))
            if sx < 1e-10:
                return 1e10
            x = x / sx
            conv = compute_conv(x)
            # Log-sum-exp of peaks (all of them, for safety)
            conv_shifted = conv - np.max(conv)
            return float(np.sum(np.exp(beta * conv_shifted)) / beta)

        def gradient(x):
            x = np.maximum(x, 0)
            sx = float(np.sum(x))
            if sx < 1e-10:
                return np.zeros(n)
            x = x / sx
            conv = compute_conv(x)
            conv_shifted = conv - np.max(conv)
            weights = np.exp(beta * conv_shifted)
            weights = weights / np.sum(weights)
            # Gradient of weighted sum of conv(a,a)[k]
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                w = weights[k]
                if abs(w) < 1e-15:
                    continue
                if k == 0:
                    grad += 2.0 * w * a
                else:
                    grad[:n - k] += w * a[k:]
                    grad[k:] += w * a[:n - k]
            grad_centered = grad - np.mean(grad)
            return grad_centered

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0,
                        'jac': lambda x: np.ones(n)}]
        bounds = [(0, None)] * n

        try:
            res = minimize(objective, a, method='SLSQP', jac=gradient,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 50, 'ftol': 1e-10})
            if res.x is not None:
                a_new = np.maximum(res.x, 0)
                a_new = symmetrize(a_new)
                sa = float(np.sum(a_new))
                if sa > 1e-10:
                    a_new = a_new / sa
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
                        if verbose:
                            print(f"      SLSQP beta={beta}: obj={best_obj:.8f}")
        except Exception:
            pass

        if time.monotonic() - start > time_budget:
            break

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """Fine coordinate descent on pairs of symmetric coordinates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = symmetrize(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    half_n = (n + 1) // 2  # number of free variables in symmetric param

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Try adjusting pairs of symmetric coordinates
        rng = np.random.default_rng(42 + it)
        for trial in range(20):
            i = rng.integers(0, half_n)
            j = rng.integers(0, half_n)
            if i == j:
                continue
            # a[i] and a[n-1-i] are the pair, same for j
            delta_fracs = [0.01, 0.02, 0.05]
            for frac in delta_fracs:
                delta = frac * a[i]
                if delta >= a[i]:
                    continue
                a[i] -= delta
                a[n - 1 - i] -= delta
                a[j] += delta
                a[n - 1 - j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                obj = compute_obj(a)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    # Revert
                    a[i] += delta
                    a[n - 1 - i] += delta
                    a[j] -= delta
                    a[n - 1 - j] -= delta
                    break

        if not improved and it > 100:
            break

    return best_a


def make_init(n, kind, **kw):
    """Generate various parametric initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'raised_cosine':
        p = kw.get('p', 1.0)
        x = np.linspace(0, np.pi, n)
        a = np.maximum(np.cos(x / 2) ** (2 * p), 0)
        return a / np.sum(a)
    elif kind == 'gaussian':
        alpha = kw.get('alpha', 2.0)
        x = np.linspace(-1, 1, n)
        a = np.exp(-alpha * x ** 2)
        return a / np.sum(a)
    elif kind == 'parabola':
        beta = kw.get('beta', 0.5)
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - beta * x ** 2, 0)
        return a / np.sum(a)
    elif kind == 'triangular':
        x = np.linspace(0, 1, n)
        a = np.minimum(x, 1 - x) * 2
        return a / np.sum(a)
    elif kind == 'hann':
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
        a = np.maximum(a, 0)
        return a / np.sum(a)
    elif kind == 'fejer':
        # Fejer window: triangle in frequency -> (sin/sin^2)^2 in time
        x = np.arange(n)
        a = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
        a = np.maximum(a, 0)
        return a / np.sum(a)
    elif kind == 'optimal_shape':
        # Empirically good shape: raised cosine with p~1.5
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 3
        a = np.maximum(a, 0)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    # Map from continuous [0,1] domain
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimizer with multiple phases.
    """
    total_start = time.monotonic()
    deadline = total_start + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = len(parent)
                print(f"Parent (n={n_parent}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = make_init(800, 'raised_cosine', p=1.5)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    a = best_seq.copy()
    n = len(a)

    # === Phase 1: Nesterov subgradient (250s) ===
    elapsed_remaining = deadline - time.monotonic()
    print(f"\n=== Phase 1: Nesterov (n={n}, budget={min(250, elapsed_remaining*0.25):.0f}s) ===")
    mwu_budget = min(250.0, elapsed_remaining * 0.25)

    for top_k in [8, 15, 25]:
        elapsed_remaining = deadline - time.monotonic()
        if elapsed_remaining < 30:
            break
        alloc = min(mwu_budget / 3.0, elapsed_remaining * 0.4)
        a_new = nesterov_optimize(a, time_budget=alloc, top_k=top_k, use_symmetry=True, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Nesterov K={top_k}: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov K={top_k}: obj={obj:.8f}")

    # === Phase 2: LP alternating (350s) ===
    elapsed_remaining = deadline - time.monotonic()
    print(f"\n=== Phase 2: LP Alternating (budget={min(350, elapsed_remaining*0.35):.0f}s) ===")
    lp_budget = min(350.0, elapsed_remaining * 0.35)

    for round_idx in range(3):
        elapsed_remaining = deadline - time.monotonic()
        if elapsed_remaining < 30:
            break
        alloc = min(lp_budget / (3 - round_idx), elapsed_remaining * 0.5)
        a_new = lp_alternate_improved(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP Round {round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP Round {round_idx}: obj={obj:.8f}")

    # === Phase 3: SLSQP refinement (100s) ===
    elapsed_remaining = deadline - time.monotonic()
    print(f"\n=== Phase 3: SLSQP (budget={min(100, elapsed_remaining*0.1):.0f}s) ===")
    slsqp_budget = min(100.0, elapsed_remaining * 0.1)
    if slsqp_budget > 15:
        a_new = slsqp_refine(a, time_budget=slsqp_budget, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  SLSQP: obj={best_score:.8f} **")
        else:
            print(f"  SLSQP: obj={obj:.8f}")

    # === Phase 4: Multi-start with different initializations and lengths (250s) ===
    elapsed_remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: Multi-start (budget={min(250, elapsed_remaining*0.25):.0f}s) ===")
    multi_budget = min(250.0, elapsed_remaining * 0.25)

    # Try different n values
    n_options = sorted(set([
        max(200, n - 100), n,
        n + 50, n + 100, n + 200,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 3000]

    # Also try different parametric inits at best n
    inits = [
        ('raised_cosine', {'p': 1.0}),
        ('raised_cosine', {'p': 2.0}),
        ('gaussian', {'alpha': 3.0}),
        ('parabola', {'beta': 0.3}),
        ('optimal_shape', {}),
    ]

    candidates = []
    for n_try in n_options:
        if n_try == n:
            candidates.append((best_seq, n_try))
        else:
            candidates.append((resample_sequence(best_seq, n_try), n_try))

    for kind, kw in inits:
        candidates.append((make_init(n, kind, **kw), n))

    # Shuffle and try
    rng_ms = np.random.default_rng(seed + 1)
    rng_ms.shuffle(candidates)

    per_candidate = multi_budget / max(len(candidates), 1)
    for idx, (a_init, n_init) in enumerate(candidates):
        elapsed_remaining = deadline - time.monotonic()
        if elapsed_remaining < 30:
            break
        alloc = min(per_candidate * 1.5, elapsed_remaining * 0.3)
        if alloc < 10:
            continue

        # Quick Nesterov + LP
        a_opt = nesterov_optimize(a_init, time_budget=alloc * 0.4, top_k=10, use_symmetry=True)
        a_opt = lp_alternate_improved(a_opt, time_budget=alloc * 0.4)
        # Quick SLSQP if time allows
        if elapsed_remaining - (deadline - time.monotonic()) < alloc * 0.7:
            rem = deadline - time.monotonic() - 20
            if rem > 10:
                a_opt = slsqp_refine(a_opt, time_budget=min(15, rem))

        obj = compute_obj(a_opt)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_opt.copy()
            n = n_init
            a = a_opt.copy()
            print(f"  Multi [{idx}] n={n_init}: obj={best_score:.8f} **")
        else:
            print(f"  Multi [{idx}] n={n_init}: obj={obj:.8f}")

    # === Phase 5: Final polish (remaining time) ===
    elapsed_remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Final Polish (remaining={elapsed_remaining:.0f}s) ===")
    n_final = len(best_seq)
    a_final = best_seq.copy()

    if elapsed_remaining > 20:
        alloc = min(elapsed_remaining * 0.25, 60.0)
        a_lp = lp_alternate_improved(a_final, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    if elapsed_remaining > 15:
        elapsed_remaining = deadline - time.monotonic()
        alloc = min(elapsed_remaining * 0.2, 30.0)
        a_sl = slsqp_refine(a_final, time_budget=alloc, verbose=True)
        obj = compute_obj(a_sl)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_sl.copy()
            a_final = a_sl.copy()
            print(f"  Final SLSQP: obj={best_score:.8f} **")

    if elapsed_remaining > 15:
        elapsed_remaining = deadline - time.monotonic()
        alloc = min(elapsed_remaining * 0.2, 20.0)
        a_cd = coordinate_polish(a_final, time_budget=alloc)
        obj = compute_obj(a_cd)
        if obj < best_score - 1e-10:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # === Phase 6: Additional LP rounds if time permits ===
    elapsed_remaining = deadline - time.monotonic()
    if elapsed_remaining > 30:
        print(f"\n=== Phase 6: Extended LP (remaining={elapsed_remaining:.0f}s) ===")
        for r in range(5):
            elapsed_remaining = deadline - time.monotonic()
            if elapsed_remaining < 20:
                break
            alloc = min(elapsed_remaining * 0.3, 50.0)
            a_new = lp_alternate_improved(best_seq, time_budget=alloc, verbose=True)
            obj = compute_obj(a_new)
            if obj < best_score - 1e-10:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Ext LP {r}: obj={best_score:.8f} **")
            else:
                print(f"  Ext LP {r}: obj={obj:.8f}")
                if obj >= best_score:
                    break

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100

    elapsed = time.monotonic() - total_start
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
