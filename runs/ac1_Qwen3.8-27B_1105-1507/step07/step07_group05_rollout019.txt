def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix_a(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_qp_hankel(q, tol=1e-8, max_iter=500):

    Uses projected gradient with backtracking.

    n = (len(q) + 1) // 2
    
    # Initialize with uniform
    a = np.ones(n) / n

    # For n up to ~2000, building the full matrix is 4M entries = 32MB, which is OK
    # But for very large n, use matvec
    if n <= 3000:
        idx = np.add.outer(np.arange(n), np.arange(n))  # idx[i,j] = i+j
        H = q[idx]
    else:
        H = None  # use matvec
    
    best_a = a.copy()
    best_val = a @ (H @ a) if H is not None else a @ hankel_matvec(q, a)
    
    eta = 0.1
    momentum_v = np.zeros(n)
    momentum = 0.5
    
    for it in range(max_iter):
        # Compute gradient: 2 * H @ a
        if H is not None:
            grad = 2.0 * (H @ a)
        else:
            grad = 2.0 * hankel_matvec(q, a)
        
        # Add Lagrange multiplier for sum constraint (approximate)
        # At optimum: H@a + lambda*1 = mu (for a>0), H@a + lambda*1 >= 0 (for a=0)
        # Approximate lambda as mean of grad at support
        support = a > 1e-10
        if support.any():
            lam = np.mean(grad[support])
        else:
            lam = np.mean(grad)
        
        # Gradient of the Lagrangian: grad - 2*lam (centered)
        grad_centered = grad - lam
        
        # Momentum update
        momentum_v = momentum * momentum_v - eta * grad_centered
        a_new = a + momentum_v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            eta *= 0.5
            momentum_v *= 0.5
            continue
        a_new = a_new / s
        
        # Evaluate
        if H is not None:
            val = a_new @ (H @ a_new)
        else:
            val = a_new @ hankel_matvec(q, a_new)
        
        if val < best_val - 1e-12:
            best_val = val
            if val < best_val + 1e-14:
                best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.9
            momentum_v *= 0.5
            if eta < 1e-10:
                break
    
    # Project best solution
    best_a = np.maximum(best_a, 0)
    s = best_a.sum()
    if s > 0:
        best_a = best_a / s
    return best_a

            a = a_new
            stall = 0
            eta = min(eta * 1.01, 3.0)
            tau = max(tau * 0.995, 0.001)  # slowly decrease temperature
            momentum = min(momentum + 0.001, 0.92)
        else:
            stall += 1
            eta *= 0.95
            v *= 0.5
            # Occasionally increase tau to escape
            if stall % 20 == 0:
                tau = min(tau * 2.0, 0.1)
            if stall > max_stall:
                # Perturb and restart
                rng_local = np.random.default_rng(it * 7919)
                a = best_a * np.exp(rng_local.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                tau = 0.05
                momentum = 0.7
                stall = 0
                max_stall += 50
        
        if verbose and it % 200 == 0:
            print(f"    SM-GD it{it}: obj={best_obj:.8f}, tau={tau:.5f}, eta={eta:.4f}")
    
    return best_a


def multiplicative_weights(a, time_budget=120.0, verbose=False):
    """
    Dual-fitting via multiplicative weights on the lag distribution.
    Maintenance strategy:
    1. Maintain q over lags k=0..2n-2
    2. Solve QP: min a^T H_q a s.t. sum(a)=1, a>=0
    3. Update q based on where the current a's auto-conv peaks
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Initialize q as uniform over lags
    K = 2 * n - 1
    q = np.ones(K) / K
    
    # Learning rate for MW
    mw_lr = 0.5
    
    while time.monotonic() - start < time_budget:
        # Solve QP for current q
        a_new = solve_qp_hankel(q, max_iter=200)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Update q: boost lags where a_new's auto-conv is large
        b = compute_conv(a_new)
        b_shifted = b - np.max(b)
        w = np.exp(mw_lr * b_shifted)
        w = w / w.sum()
        
        # Exponential interpolation
        alpha = 0.3
        q = (1 - alpha) * q + alpha * w
        
        # Also try: pure MW update
        # q[k] proportional to exp(cumulative_loss_k)
        # where loss_k = (a*a)[k]
        # For now, use the interpolation approach
        
        if verbose:
            print(f"    MW: obj={best_obj:.8f}")
    
    return best_a


def lp_alternating_v2(a, time_budget=60.0, verbose=False):
    """
    Improved LP alternating: solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Then blend and repeat.
    Uses scipy linprog for reliability.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Build constraint matrix
    A = build_conv_matrix_a(a)  # shape (2n-1, n)
    # Constraint: A @ g <= M, sum(g)=1, g>=0
    # Variables: [g_0, ..., g_{n-1}, M]
    # A_ub: [A, -I] @ [g; M] <= 0
    # A_eq: [1, ..., 1, 0] @ [g; M] = 1
    
    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        try:
            from scipy.optimize import linprog
            c = np.zeros(n + 1)
            c[-1] = 1.0  # minimize M
            
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            
            bounds = [(0, None)] * n + [(0, None)]
            
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            
            if res.status == 0:
                g_lp = np.maximum(res.x[:n], 0)
                s_g = g_lp.sum()
                if s_g > 1e-10:
                    g_lp = g_lp / s_g
                    
                    # Try different blend factors
                    best_t_obj = np.inf
                    best_t = 0.3
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * a + t * g_lp
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_t = t
                    
                    if best_t_obj < best_obj - 1e-11:
                        best_obj = best_t_obj
                        a = (1 - best_t) * a + best_t * g_lp
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        best_a = a.copy()
                        # Rebuild A for new a
                        A = build_conv_matrix_a(a)
                        if verbose and it % 5 == 0:
                            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
                    else:
                        break
        except Exception as e:
            break
    
    return best_a


def peak_reduction(a, time_budget=30.0):
    """
    Direct peak reduction: identify the peak lag and reduce contributions.
    Uses a "water-filling" approach on the contributing pairs.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    eta = 0.02
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        b_max = np.max(b)
        
        # Find all "hot" lags (within 5% of max)
        threshold = b_max * 0.95
        hot_lags = np.where(b >= threshold)[0]
        
        if len(hot_lags) == 0:
            break
        
        # Compute pressure on each a[i]
        # For each hot lag k, the contribution to b[k] from a[i] is a[k-i] (times 2)
        # We want to reduce a[i] where it contributes most to hot lags
        pressure = np.zeros(n)
        for k in hot_lags:
            weight = (b[k] - b_max * 0.5) / max(b_max, 1e-10)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += weight * a[j]
        
        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max
        
        # Update: reduce high-pressure elements, increase low-pressure
        grad = pressure - pressure.mean()
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        fa_new = rfft(a_new, N)
        b_new = irfft(fa_new * fa_new, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(b_new))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 0.1)
        else:
            eta *= 0.93
            if eta < 1e-8:
                break
    
    return best_a


def diverse_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = a.sum()
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Symmetric U-shape
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / a.sum()
    elif mode == 3:
        # Symmetric bell
        x = np.arange(n) / max(n - 1, 1)
        a = np.exp(-8 * (x - 0.5) ** 2) + 0.1
        a = a / a.sum()
    elif mode == 4:
        # Symmetric with sinusoidal modulation
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Two bumps (symmetric)
        centers = [n * 0.25, n * 0.75]
        sigma = n * 0.1
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Triangle wave (symmetric)
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0 - np.abs(2 * x - 1)
        a = a + 0.1
        a = a / a.sum()
    else:
        # Random symmetric
        half = n // 2
        a_half = rng.uniform(0.1, 1.0, half)
        a = np.zeros(n)
        a[:half] = a_half
        a[half:] = a_half[::-1]
        if n % 2 == 0:
            a[half - 1:half + 1] = a_half[half - 1]
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def combined_refine(a, time_budget=60.0, verbose=False):
    """
    Combined refinement: soft-max GD -> peak reduction -> LP -> soft-max GD.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Step 1: Soft-max GD
    remain = max(0, start + time_budget - time.monotonic())
    if remain > 10:
        a1 = soft_max_gd(a, time_budget=remain * 0.35, verbose=verbose)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            a = a1.copy()
            if verbose:
                print(f"      [SM-GD] {best_obj:.8f}")
    
    # Step 2: Peak reduction
    remain = max(0, start + time_budget - time.monotonic())
    if remain > 10:
        a2 = peak_reduction(a, time_budget=remain * 0.2)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            a = a2.copy()
            if verbose:
                print(f"      [PeakRed] {best_obj:.8f}")
    
    # Step 3: LP
    remain = max(0, start + time_budget - time.monotonic())
    if remain > 15:
        a3 = lp_alternating_v2(a, time_budget=remain * 0.25, verbose=verbose)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            a = a3.copy()
            if verbose:
                print(f"      [LP] {best_obj:.8f}")
    
    # Step 4: Final soft-max GD
    remain = max(0, start + time_budget - time.monotonic())
    if remain > 10:
        a4 = soft_max_gd(a, time_budget=remain * 0.3, verbose=verbose)
        obj4 = compute_obj(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            if verbose:
                print(f"      [SM-GD2] {best_obj:.8f}")
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: adaptive soft-max GD + MW dual fitting + LP + multi-scale.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: Intensive soft-max GD from parent ===
    print(f"\n=== Phase 1: Soft-Max GD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 250.0)
    a_sm = soft_max_gd(best_seq, time_budget=alloc, verbose=True)
    obj_sm = compute_obj(a_sm)
    if obj_sm < best_score:
        best_score = obj_sm
        best_seq = a_sm.copy()
        print(f"  ** SM-GD: obj={best_score:.8f} **")
    else:
        print(f"  SM-GD: obj={obj_sm:.8f}")
    
    # === Phase 2: Peak reduction ===
    print(f"\n=== Phase 2: Peak Reduction ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_pr = peak_reduction(best_seq, time_budget=alloc)
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  ** PeakRed: obj={best_score:.8f} **")
        else:
            print(f"  PeakRed: obj={obj_pr:.8f}")
    
    # === Phase 3: LP Alternating ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 100.0)
        a_lp = lp_alternating_v2(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")
    
    # === Phase 4: Combined refinement ===
    print(f"\n=== Phase 4: Combined Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 150.0)
        a_comb = combined_refine(best_seq, time_budget=alloc, verbose=True)
        obj_comb = compute_obj(a_comb)
        if obj_comb < best_score:
            best_score = obj_comb
            best_seq = a_comb.copy()
            print(f"  ** Combined: obj={best_score:.8f} **")
        else:
            print(f"  Combined: obj={obj_comb:.8f}")
    
    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best + 100, n_best + 200, n_best + 500,
        n_best - 100 if n_best > 500 else n_best + 50,
        600, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and n != n_best]
    n_options = n_options[:5]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.2, 80.0)
        a_opt = combined_refine(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 6: Random restarts with combined optimization ===
    print(f"\n=== Phase 6: Random Restarts ===")
    n_curr = len(best_seq)
    n_restarts = max(2, int((deadline - time.monotonic()) / 60.0))
    n_restarts = min(n_restarts, 8)
    
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        a_r = diverse_init(n_curr, rng)
        alloc = min(remaining * 0.2, 50.0)
        a_opt = combined_refine(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:

        else:

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.4, 120.0)
        a_final = combined_refine(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final: obj={best_score:.8f} **")
    
    # One more peak reduction
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_pc = peak_reduction(best_seq, time_budget=min(remaining * 0.3, 30.0))
        obj_pc = compute_obj(a_pc)
        if obj_pc < best_score:
            best_score = obj_pc
            best_seq = a_pc.copy()
            print(f"  ** Final Peak: obj={best_score:.8f} **")
    
    # Final soft-max GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sm = soft_max_gd(best_seq, time_budget=min(remaining * 0.3, 30.0))
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  ** Final SM: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
