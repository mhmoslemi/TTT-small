"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses log-sum-exp smoothed gradient descent with adaptive temperature,
LP alternating optimization, structured family exploration, and multi-restart.
Starts from parent sequence and explores structured initializations.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_gradient(a_norm, beta):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    a_norm should have sum=1, a_norm >= 0.
    Returns (lse_value, gradient_centered) where gradient_centered has sum=0.
    """
    n = len(a_norm)
    conv = compute_conv(a_norm)
    max_conv = np.max(conv)
    shifted = beta * (conv - max_conv)
    # Numerically stable log-sum-exp
    max_shifted = np.max(shifted)
    lse = max_conv * beta + max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    
    # Softmax weights
    weights = np.exp(shifted - max_shifted)
    weights /= np.sum(weights)
    
    # Gradient: d/d a[i] of lse(sum_k beta*conv[k]) 
    # = beta * sum_k weights[k] * d conv[k]/d a[i]
    # d conv(a,a)[k] / d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    # grad[i] = 2*beta * sum_{k: 0<=k-i<n} weights[k] * a[k-i]
    # Let m = k-i: grad[i] = 2*beta * sum_m a[m] * weights[i+m]
    # This is cross-correlation of a and weights at lag i
    
    a_rev = a_norm[::-1]
    conv_whole = np.convolve(a_rev, weights, mode='full')
    grad = 2.0 * beta * conv_whole[n - 1: n - 1 + n]
    
    # Center to preserve sum
    grad_centered = grad - np.mean(grad)
    return lse, grad_centered

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=60.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_work = a / s
    A = build_conv_matrix(a_work)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
        for solver in solvers:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None

        if g_lp is None:

        # Uniform
        a = np.ones(n)
    elif family_id == 1:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif family_id == 2:
        # Parabolic window
        a = 1 - (2 * t - 1)**2
    elif family_id == 3:
        # Asymmetric: high on left, low on right
        a = 1 - t
        a = np.maximum(a, 0.01)
    elif family_id == 4:
        # Double hump
        a = np.exp(-((t - 0.3)**2) / 0.05) + np.exp(-((t - 0.7)**2) / 0.05)
    elif family_id == 5:
        # S-curve (sigmoid-like)
        a = 1 / (1 + np.exp(-10 * (t - 0.5)))
    elif family_id == 6:
        # Asymmetric Gaussian
        a = np.exp(-((t - 0.35)**2) / 0.04)
    elif family_id == 7:
        # Cubic window
        a = t**2 * (3 - 2*t)  # smoothstep
    elif family_id == 8:
        # Random smooth
        n_knots = int(rng.integers(3, 8))
        knots = np.sort(rng.uniform(0, 1, n_knots))
        values = rng.uniform(0.1, 1.0, n_knots)
        a = np.interp(t, knots, values)
    else:
        # Triangle (peak in middle)
        a = 1 - np.abs(2 * t - 1)
    
    a = np.maximum(a, 0.001)
    a = a / np.sum(a)
    return a


def coordinate_descent(a, time_budget=30.0, verbose=False):
    """Optimize a few random coordinates at a time."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        # Pick 1-3 random coordinates
        n_coords = int(rng.integers(1, 4))
        indices = rng.choice(n, size=n_coords, replace=False)
        
        # For each coordinate, try a range of values
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        
        for idx in indices:
            current_val = best_local_a[idx]
            # Try multiplicative changes
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                trial = best_local_a.copy()
                trial[idx] = current_val * factor
                trial = np.maximum(trial, 0)
                s_trial = float(np.sum(trial))
                if s_trial < 1e-10:
                    continue
                trial = trial / s_trial
                obj_trial = compute_obj(trial)
                if obj_trial < best_local_obj:
                    best_local_obj = obj_trial
                    best_local_a = trial.copy()
        
        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
        else:
            # If no improvement in a while, try larger perturbations
            if it % 20 == 0:
                idx = int(rng.integers(n))
                trial = best_a.copy()
                trial[idx] *= rng.uniform(0.3, 3.0)
                trial = np.maximum(trial, 0)
                s_trial = float(np.sum(trial))
                if s_trial > 1e-10:
                    trial = trial / s_trial
                    obj_trial = compute_obj(trial)
                    if obj_trial < best_obj:
                        best_obj = obj_trial
                        best_a = trial.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def combined_optimize(a, time_budget=60.0, verbose=False):
    """
    Combined optimization: alternate between LSE gradient and LP steps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    phase_budget_lse = 0.6
    phase_budget_lp = 0.3
    phase_budget_cd = 0.1

    while True:
        remaining = deadline_helper(time_budget, start)
        if remaining < 10:
            break
        
        # LSE gradient phase
        lse_alloc = min(remaining * phase_budget_lse, 30.0)
        if lse_alloc > 5:
            a_lse = lse_optimize(best_a, time_budget=lse_alloc, verbose=verbose)
            obj_lse = compute_obj(a_lse)
            if obj_lse < best_obj:
                best_obj = obj_lse
                best_a = a_lse.copy()
                if verbose:
                    print(f"      After LSE: obj={best_obj:.8f}")
        
        remaining = deadline_helper(time_budget, start)
        if remaining < 10:
            break
        
        # LP phase
        lp_alloc = min(remaining * phase_budget_lp, 30.0)
        if lp_alloc > 5:
            a_lp = lp_alternate(best_a, time_budget=lp_alloc, verbose=verbose)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"      After LP: obj={best_obj:.8f}")
        
        remaining = deadline_helper(time_budget, start)
        if remaining < 5:
            break
        
        # Coordinate descent polish
        cd_alloc = min(remaining * phase_budget_cd, 15.0)
        if cd_alloc > 3:
            a_cd = coordinate_descent(best_a, time_budget=cd_alloc)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                if verbose:
                    print(f"      After CD: obj={best_obj:.8f}")

        # Check if we should continue
        remaining = deadline_helper(time_budget, start)
        if remaining < 15:
            break

    return best_a


def deadline_helper(total_budget, start_time):
    """Helper to compute remaining time."""
    return total_budget - (time.monotonic() - start_time)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses log-sum-exp smoothed gradient descent,
    LP alternating, coordinate descent, and structured family exploration.
    Returns the best sequence found.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Establish baseline
    print("=== Phase 0: Baseline ===")
    
    # Try parent sequence
    parent = None
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr.copy()
                n_parent = parent.size
                parent_list = [float(x) for x in parent]
                score = evaluate_sequence(parent_list)
                if score < np.inf:
                    print(f"  Parent (n={n_parent}): score={score:.8f}")
                    if score < best_score:
                        best_score = score
                        best_seq = parent.copy()
                else:
                    print(f"  Parent (n={n_parent}): INVALID")
    except NameError:
        print("  Parent not available (NameError)")
    except Exception as e:
        print(f"  Parent not available: {e}")

    if best_seq is None:
        n_parent = 800
        best_seq = np.ones(n_parent) / n_parent
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"  Default constant (n={n_parent}): score={best_score:.8f}")

    # Also try a few constant baselines
    for n_base in [600, 800, 1000, 1200]:
        if time.monotonic() > deadline:
            break
        seq_const = [1.0] * n_base
        score = evaluate_sequence(seq_const)
        if score < best_score:
            best_score = score
            best_seq = np.ones(n_base) / n_base
            print(f"  Constant n={n_base}: score={score:.8f} **")

    print(f"  Best so far: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: Optimize parent with combined method
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 1: Optimize parent (n={len(best_seq)}) ===")
        a_work = best_seq.copy()
        s = float(np.sum(a_work))
        if s > 0:
            a_work = a_work / s
        
        alloc = min(180.0, (deadline - time.monotonic()) * 0.3)
        a_opt = combined_optimize(a_work, time_budget=alloc, verbose=True)
        opt_list = [float(x) for x in a_opt]
        score = evaluate_sequence(opt_list)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  ** Improved: n={len(a_opt)}, score={best_score:.8f} **")
        else:
            print(f"  Score: {score:.8f} (no improvement)")

    # Phase 2: Explore structured families at same n
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 2: Structured families (n={len(best_seq)}) ===")
        n_cur = len(best_seq)
        
        for fam in range(9):
            if time.monotonic() > deadline - 15:
                break
            a_init = structured_family(n_cur, fam, rng)
            a_opt = combined_optimize(a_init, time_budget=min(40.0, (deadline - time.monotonic()) * 0.1), verbose=False)
            score = evaluate_sequence([float(x) for x in a_opt])
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Family {fam}: score={score:.8f} **")
            else:
                print(f"  Family {fam}: score={score:.8f}")

    # Phase 3: Explore different n values
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 3: Multi-n exploration ===")
        n_cur = len(best_seq)
        n_options = sorted(set([
            max(100, n_cur - 200), max(100, n_cur - 100),
            n_cur + 100, n_cur + 200, n_cur + 400,
            400, 600, 1000, 1500, 2000
        ]))
        n_options = [n for n in n_options if n != n_cur and 100 <= n <= 3000]
        
        for n_try in n_options:
            if time.monotonic() > deadline - 15:
                break
            a_try = resample_sequence(best_seq, n_try)
            a_opt = combined_optimize(a_try, time_budget=min(35.0, (deadline - time.monotonic()) * 0.1), verbose=False)
            score = evaluate_sequence([float(x) for x in a_opt])
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: score={score:.8f} **")
            else:
                print(f"  n={n_try}: score={score:.8f}")

    # Phase 4: Intensive optimization on best
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 4: Intensive optimization (n={len(best_seq)}) ===")
        a_work = best_seq.copy()
        s = float(np.sum(a_work))
        if s > 0:
            a_work = a_work / s
        
        # Multiple rounds of combined optimization
        for round_idx in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.3, 60.0)
            if alloc < 10:
                break
            a_opt = combined_optimize(a_work, time_budget=alloc, verbose=True)
            score = evaluate_sequence([float(x) for x in a_opt])
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                a_work = a_opt.copy()
                s = float(np.sum(a_work))
                if s > 0:
                    a_work = a_work / s
                print(f"  Round {round_idx}: score={best_score:.8f} **")
            else:
                print(f"  Round {round_idx}: score={score:.8f}")

    # Phase 5: Final coordinate descent polish
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 5: Final polish ===")
        a_polish = coordinate_descent(best_seq.copy(), time_budget=min(30.0, deadline - time.monotonic() - 5))
        score = evaluate_sequence([float(x) for x in a_polish])
        if score < best_score:
            best_score = score
            best_seq = a_polish.copy()
            print(f"  CD polish: score={best_score:.8f} **")
        else:
            print(f"  CD polish: score={score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800) / 800

    # Ensure all values are valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        print("  WARNING: Final sequence invalid, using constant fallback")
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
