import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)

def compute_obj_normalized(p):
    """Compute 2*n*max(conv(p,p)) where p is already normalized (sum=1)."""
    n = len(p)
    return 2.0 * n * float(np.max(compute_conv(p)))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Wang & Tibshirani (2015)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = len(a)
    a = a / np.sum(a)  # normalize
    
    # Build constraint matrix: A[k,j] = a[k-j] if 0 <= k-j < n
    # conv(a,g)[k] = sum_j a[k-j] * g[j]

        # A[k, j] = a[k-j] if 0 <= k-j < n

        A[j:j+n, j] = a

    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

                s = float(np.sum(g_val))

        A_ub[:, n] = -1.0

        c[-1] = 1.0
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1),

            s = float(np.sum(g_val))

    except Exception:

    return None, None

def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP alternating with golden-section line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Golden section line search on t in (0, 1)
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj_normalized(a_try)

        # Golden section search
        gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)
        
        for _ in range(10):  # 10 iterations of golden section ≈ good precision
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at_t(d)
        
        best_t = (lo + hi) / 2
        a_new = (1.0 - best_t) * a + best_t * g_lp
        a_new = np.maximum(a_new, 0)
        st = float(np.sum(a_new))
        if st > 1e-10:
            a_new = a_new / st
            obj_new = compute_obj_normalized(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall_count = 0
                if verbose and it % 20 == 0:
                    print(f"        LP it{it}: obj={best_obj:.8f}, t={best_t:.4f}")
            else:
                stall_count += 1
        else:
            stall_count += 1

        if stall_count > 40:
            # Perturb to escape
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            scale = 0.02 + 0.01 * min(stall_count // 40, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            stall_count = 0

"""
Step function optimizer using LP alternating, coordinate descent,
mirror descent, and structured multi-n search.

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(p):
    """p is normalized (sum=1). Returns 2*n*max(conv(p,p))."""
    n = len(p)
    if n == 0:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(p)))

def normalize(a):
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    return a / s

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_A(a):
    """Build the convolution matrix A (2n-1 x n) where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp(a):
    """Solve min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_A(a)
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in ['CLARABEL', 'MOSEK', 'SCS', 'ECOS']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass
    
    # Fallback
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1),
                 A_eq=np.ones((1, n+1)), b_eq=np.array([1.0]),
                 bounds=[(0, None)] * (n+1), method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

"""
Step function optimizer using LP alternating with golden-section line search,
subgradient methods with multiple peak tracking, mirror descent (multiplicative weights),
and structured multi-n exploration. Targets the objective 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: golden-section line search, multi-peak subgradient,
mirror descent complement, and adaptive time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(p):
    """Compute 2*n*max(conv(p,p)) assuming p is normalized (sum=1)."""
    n = len(p)
    if n == 0:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(p)))


def normalize(a):
    """Clip to non-negative and normalize to sum 1. Returns None if invalid."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    return a / s


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_A(a):
    """Build convolution matrix A (2n-1 x n) where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    if n < 2:
        return np.array([1.0]), 1.0
    A = build_A(a)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in ['CLARABEL', 'MOSEK', 'SCS', 'ECOS', 'PDLP']:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass

    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                  A_eq=np.ones((1, n + 1)), b_eq=np.array([1.0]),
                  bounds=[(0, None)] * (n + 1), method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def subgradient_multi_peak(p, top_k=5):
    """Compute subgradient using top_k argmax convolution bins weighted by value.
    Returns gradient vector g where g[j] = 2 * sum_k w_k * a[k-j] for contributing k."""
    n = len(p)
    conv = compute_conv(p)
    # Get top_k bins
    top_indices = np.argsort(conv)[-top_k:]
    top_vals = conv[top_indices]
    # Softmax weights on top bins
    w = np.exp(top_vals - np.max(top_vals))
    w = w / np.sum(w)

    grad = np.zeros(n)
    for idx, wi in zip(top_indices, w):
        k = int(idx)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                grad[j] += 2.0 * wi * p[m]
    return grad, conv


def lp_alternating(p, time_budget=60.0, verbose=False):
    """LP alternating with golden-section line search and adaptive perturbations."""
    n = len(p)
    p = np.maximum(p, 0).copy().astype(np.float64)
    s = float(np.sum(p))
    if s < 1e-10:
        return p
    p = p / s

    best_p = p.copy()
    best_obj = compute_obj(p)
    start = time.monotonic()
    stall_count = 0
    consecutive_no_lp = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp(p)
        if g_lp is None:
            consecutive_no_lp += 1
            if consecutive_no_lp > 10:
                break
            # Fallback: use subgradient direction
            grad, _ = subgradient_multi_peak(p, top_k=3)
            grad_c = grad - np.mean(grad)
            g_lp = simplex_project(p - 0.01 * grad_c)
        else:
            consecutive_no_lp = 0

        # Golden section line search on t in (0, 1)
        def eval_at_t(t):
            p_try = (1.0 - t) * p + t * g_lp
            p_try = np.maximum(p_try, 0)
            st = float(np.sum(p_try))
            if st < 1e-10:
                return np.inf
            p_try = p_try / st
            return compute_obj(p_try)

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = eval_at_t(c1)
        f2 = eval_at_t(c2)

        for _ in range(12):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = eval_at_t(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = eval_at_t(c2)

        best_t = (lo + hi) / 2.0
        p_new = (1.0 - best_t) * p + best_t * g_lp
        p_new = np.maximum(p_new, 0)
        st = float(np.sum(p_new))
        if st > 1e-10:
            p_new = p_new / st
            obj_new = compute_obj(p_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_p = p_new.copy()
                p = p_new
                stall_count = 0
                if verbose and it % 30 == 0:
                    print(f"        LP it{it}: obj={best_obj:.8f}, t={best_t:.4f}")
            else:
                stall_count += 1
        else:
            stall_count += 1

        if stall_count > 30:
            # Perturb to escape local optimum
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            scale = 0.02 + 0.015 * min(stall_count // 30, 6)
            p = best_p * np.exp(rng.normal(0, scale, n))
            p = np.maximum(p, 0)
            s2 = float(np.sum(p))
            if s2 > 1e-10:
                p = p / s2
            stall_count = 0

    return best_p


def mirror_descent(p, time_budget=40.0, eta_init=0.05, eta_min=0.001):
    """Mirror descent (multiplicative weights) for the normalized problem."""
    n = len(p)
    p = np.maximum(p, 1e-15).copy().astype(np.float64)
    s = float(np.sum(p))
    if s < 1e-10:
        return p
    p = p / s

    best_p = p.copy()
    best_obj = compute_obj(p)
    start = time.monotonic()
    eta = eta_init

    for it in range(8000):
        if time.monotonic() - start > time_budget:
            break

        grad, conv = subgradient_multi_peak(p, top_k=5)
        grad_c = grad - np.mean(grad)

        # Multiplicative weights update
        p_new = p * np.exp(-eta * grad_c)
        p_new = np.maximum(p_new, 1e-15)
        p_new = p_new / np.sum(p_new)

        obj_new = compute_obj(p_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_p = p_new.copy()
            p = p_new
            eta = max(eta_min, eta * 0.9995)
        else:
            eta *= 0.93
            if eta < eta_min:
                break
            p = best_p.copy()

    return best_p

def coordinate_descent_pass(p, time_budget=20.0, max_passes=3):
    """Cyclic coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(p)
    p = np.maximum(p, 0).copy().astype(np.float64)
    s = float(np.sum(p))
    if s < 1e-10:
        return p
    p = p / s

    best_p = p.copy()
    best_obj = compute_obj(p)
    start = time.monotonic()

    for cp in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - start > time_budget:
                break
            # Try a few perturbations of coordinate j
            base_val = p[j]
            if base_val < 1e-8:
                # Try adding mass
                for frac in [0.005, 0.01, 0.02]:
                    mass = frac * np.mean(p)
                    p_try = p.copy()

                    # Take mass proportionally from others
                    others = np.arange(n) != j
                    p_try[others] -= (mass * p[others]) / max(np.sum(p[others]), 1e-15)
                    p_try[j] += mass
                    p_try = np.maximum(p_try, 0)
                    st = float(np.sum(p_try))
                    if st < 1e-10:
                        continue
                    p_try = p_try / st
                    obj_try = compute_obj(p_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_p = p_try.copy()
                        p = p_try
                        improved = True
                        break
            else:
                # Try reducing or increasing
                for frac in [-0.5, -0.2, 0.2, 0.5]:
                    delta = frac * base_val
                    p_try = p.copy()
                    p_try[j] = max(0, p_try[j] + delta)
                    # Redistribute
                    others = np.arange(n) != j
                    diff = p[j] - p_try[j]
                    if abs(diff) > 1e-12:
                        p_try[others] += diff * (p[others] / max(np.sum(p[others]), 1e-15))
                    p_try = np.maximum(p_try, 0)
                    st = float(np.sum(p_try))
                    if st < 1e-10:
                        continue
                    p_try = p_try / st
                    obj_try = compute_obj(p_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_p = p_try.copy()
                        p = p_try
                        improved = True
                        break
        if not improved:
            break

    return best_p


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
