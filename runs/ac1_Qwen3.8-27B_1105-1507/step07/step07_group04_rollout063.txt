    while size < 2 * n - 1:
        size <<= 1

            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(None, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            pass

    return None


def lp_alternating(a, time_budget=100.0, K=1, verbose=False):
    """LP alternating optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp = solve_lp_minmax(a, K=K)
        if g_lp is None:
            # Perturb and retry
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            a_pert = a * np.exp(rng.normal(0, 0.01, n))
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 1e-10:
                a_pert /= s_p
                g_lp = solve_lp_minmax(a_pert, K=K)
            if g_lp is None:
                break

        # Adaptive line search
        candidate_ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP(K={K}): obj={best_obj:.8f}, t={best_t_val:.3f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 10, 4)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a /= s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

    return best_a


def coordinate_descent(a, time_budget=30.0):
    """Pair-exchange coordinate descent for final polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a /= s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    while time.monotonic() - start < time_budget and stall < 200:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from peak contributors to non-peak positions
        for _ in range(50):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = (i + 1 + rng.integers(0, n - 2)) % n
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.1)
            delta = frac * a[i]
            if delta <= 0 or delta >= a[i]:
                continue
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a /= s
            new_obj = compute_obj(a)
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                improved = True
            else:
                a[i] += delta
                a[j] -= delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a /= s

        if not improved:
            stall += 1
        else:
            stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_ramp(n, direction=1):
    """Generate a linear ramp sequence (non-uniform initialization)."""
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    return a / np.sum(a)


def generate_gaussian(n, sigma_frac=0.3):
    """Generate a Gaussian-shaped sequence."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / np.sum(a)


def generate_dual_ramp(n):
    """Generate a dual-ramp (high at both ends, low in middle)."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - 0.5 * np.cos(2 * np.pi * x / n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Log-sum-exp gradient + LP alternating + Multi-n + Polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = generate_ramp(800)
        best_score = compute_obj(best_seq)
        print(f"Default ramp init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Log-sum-exp optimization on parent (300s) ===
    print(f"\n=== Phase 1: LSE Optimization (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(300.0, remaining * 0.25)
    a_lse = lse_optimize(best_seq, time_budget=alloc, T_start=20.0, T_end=3000.0, verbose=True)
    obj_lse = compute_obj(a_lse)
    if obj_lse < best_score:
        best_score = obj_lse
        best_seq = a_lse.copy()
        print(f"  LSE: obj={best_score:.8f} **")
    else:
        print(f"  LSE: obj={obj_lse:.8f} (no improvement)")

    # === Phase 2: LP Alternating - K=1 (300s) ===
    print(f"\n=== Phase 2: LP Alternating K=1 ===")
    remaining = deadline - time.monotonic()
    alloc = min(300.0, remaining * 0.25)
    a_lp1 = lp_alternating(best_seq, time_budget=alloc, K=1, verbose=True)
    obj_lp1 = compute_obj(a_lp1)
    if obj_lp1 < best_score:
        best_score = obj_lp1
        best_seq = a_lp1.copy()
        print(f"  LP K=1: obj={best_score:.8f} **")
    else:
        print(f"  LP K=1: obj={obj_lp1:.8f}")

    # === Phase 3: LSE again with higher temperature (150s) ===
    print(f"\n=== Phase 3: LSE Sharp (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(150.0, remaining * 0.15)
    a_lse2 = lse_optimize(best_seq, time_budget=alloc, T_start=100.0, T_end=10000.0, verbose=True)
    obj_lse2 = compute_obj(a_lse2)
    if obj_lse2 < best_score:
        best_score = obj_lse2
        best_seq = a_lse2.copy()
        print(f"  LSE Sharp: obj={best_score:.8f} **")
    else:
        print(f"  LSE Sharp: obj={obj_lse2:.8f}")

    # === Phase 4: Multi-n exploration (250s) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        n_best,
        n_best + 100, n_best + 200, n_best + 400,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc_lp = min(30.0, remaining * 0.3)

            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")

    # === Phase 7: Final polish (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

        a_final = lse_optimize(best_seq, time_budget=alloc * 0.5, T_start=50.0, T_end=5000.0)
        a_final = lp_alternating(a_final, time_budget=alloc * 0.5, K=1)

            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 8: Random restarts from best (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(8):

            scale = 0.02 + 0.015 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
