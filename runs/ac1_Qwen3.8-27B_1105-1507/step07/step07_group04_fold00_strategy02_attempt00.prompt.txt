def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def mwu_gradient(a, k_indices):
    """Compute gradient of average b_k over k in k_indices w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    K = len(k_indices)
    for k in k_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        d_start = k + 1 - j_end
        d_end = k - j_start + 1
        grad[j_start:j_end] += a[d_start:d_end][::-1]
    grad /= K
    return grad

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]
    if solver:
        solvers = [solver] + solvers
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None

def subgradient_momentum(a, time_budget=60.0, top_k=5, verbose=False):
    """
    Nesterov-accelerated subgradient method with momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.9
    step_size = 0.02
    x = a.copy()  # Nesterov's lookahead point

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Compute gradient at lookahead point
        conv = compute_conv(x)
        k_indices = np.argsort(conv)[::-1][:top_k]
        grad = mwu_gradient(x, k_indices)
        
        # Center the gradient (project to sum=0 hyperplane)
        grad_centered = grad - np.mean(grad)

        # Momentum update
        momentum = beta * momentum - step_size * grad_centered
        new_a = a + momentum

        # Project to simplex (a >= 0, sum(a) = 1)
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            new_a = a.copy()
            s_new = 1.0
        new_a = new_a / s_new

        # Nesterov: update lookahead
        t_k = 2.0 / (it + 2.0)
        x = new_a + (1 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = new_a
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive step size
        if it % 100 == 99:
            step_size = max(step_size * 0.95, 0.001)

    return best_a


def coordinate_descent_polish(a, time_budget=30.0):
    """
    Coordinate descent: adjust pairs of coordinates to reduce peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        k_star = int(np.argmax(conv))

        # Try moving mass from high contributors to low contributors
        for i in range(0, n, 2):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i:
                # Also check other peaks
                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[i]
                    if delta > a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            for _ in range(10):
                idx = rng.integers(0, n)
                jdx = (idx + 1 + rng.integers(0, n - 1)) % n
                delta = 0.01 * a[idx]
                a[idx] -= delta
                a[jdx] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a /= s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 50:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
