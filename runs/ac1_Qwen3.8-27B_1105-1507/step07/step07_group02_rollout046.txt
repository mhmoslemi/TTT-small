import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses harmonic parameterization for diverse starts, LP alternating optimization,
subgradient descent, and multi-scale exploration. Key insight: asymmetric sequences

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, timeout=20.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s

    # Try cvxpy first (faster for larger n)
    try:
        import cvxpy as cp
        A = build_conv_matrix(a)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.PDLP, cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs
            except Exception:
                continue
    except Exception:
        pass

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    return None

def subgradient_optimize(a, time_budget=30.0):
    """Subgradient descent on max(conv(a,a)) with sum constraint."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient of max(conv) w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Center to preserve sum
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        a_new = a - lr * grad_centered / grad_norm
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / sn

            best_obj = obj_new
            best = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.1)

                        best = a_pert.copy()
                        best_obj = compute_obj(a_pert)
                a = best.copy()
                lr = 2.0 / (n * (1 + np.linalg.norm(a)))

    return best


def harmonic_sequence(n, K, alphas, phases):
    """Generate sequence as sum of harmonics: a[i] = (1/n)*(1 + sum_k alpha_k*cos(2*pi*k*i/n + phi_k))."""
    i = np.arange(n)
    a = np.ones(n)
    for k in range(K):
        a += alphas[k] * np.cos(2.0 * np.pi * k * i / n + phases[k])
    a = a / n
    return np.maximum(a, 0)

def gaussian_init(n, sigma_frac, center_frac, asymmetry=0.0):
    """Generate Gaussian window with optional asymmetry."""
    i = np.arange(n)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
    if asymmetry != 0:
        a = a * (1.0 + asymmetry * np.sin(2 * np.pi * i / n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def cosine_init(n, power=1.0, shift=0.0):
    """Cosine window with power and shift."""
    i = np.arange(n)
    x = (i + shift) / n
    a = np.cos(np.pi * (x - 0.5)) ** (2 * power)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            scored_inits.append((obj, a_init))
        scored_inits.sort(key=lambda x: x[0])

        # Optimize top candidates with LP
        for idx, (obj_init, a_init) in enumerate(scored_inits[:8]):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(30.0, remaining * 0.2)
            a_opt = lp_alternate(a_init, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            print(f"  Window {idx} (init_obj={obj_init:.6f}): LP result={ev:.8f}")
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"    ** NEW BEST: {best_score:.8f} **")

    # Phase 3: LP alternating on best sequence (main workhorse)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: LP alternating on best (remaining={remaining:.0f}s) ===")
        a_cur = best_seq.copy()
        s = float(np.sum(a_cur))
        if s > 0:
            a_cur = a_cur / s

        alloc = min(180.0, remaining * 0.4)
        a_lp = lp_alternate(a_cur, time_budget=alloc, verbose=True)
        ev_lp = evaluate_sequence([float(x) for x in a_lp])
        print(f"  LP result: {ev_lp:.8f}")
        if ev_lp < best_score:
            best_score = ev_lp
            best_seq = a_lp.copy()
            print(f"  ** NEW BEST: {best_score:.8f} **")

    # Phase 4: Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Subgradient polish (remaining={remaining:.0f}s) ===")
        a_cur = best_seq.copy()
        s = float(np.sum(a_cur))
        if s > 0:
            a_cur = a_cur / s
        alloc = min(60.0, remaining * 0.3)
        a_sg = subgradient_optimize(a_cur, time_budget=alloc)
        ev_sg = evaluate_sequence([float(x) for x in a_sg])
        print(f"  Subgradient result: {ev_sg:.8f}")
        if ev_sg < best_score:
            best_score = ev_sg
            best_seq = a_sg.copy()
            print(f"  ** NEW BEST: {best_score:.8f} **")

    # Phase 5: Multi-n exploration
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Multi-n exploration (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(100, n_best - 300), max(100, n_best - 150),
            n_best + 100, n_best + 200, n_best + 400,
            400, 600, 900, 1200, 1600, 2000
        ]))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            if n_try == len(best_seq):
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try

            alloc = min(30.0, remaining * 0.15)
            a_opt = lp_alternate(a_try, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: score={best_score:.8f} **")
            else:
                print(f"  n={n_try}: score={ev:.8f}")

    # Phase 6: Random restarts with diverse strategies
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Random restarts (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_restarts = min(10, int(remaining // 25))

        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            strategy = r % 5
            if strategy == 0:
                # Perturbed best
                a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
                a_r = np.maximum(a_r, 0)
            elif strategy == 1:
                # Gaussian init
                a_r = gaussian_init(n_best, 0.15 + 0.1 * (r % 3), 0.4 + 0.1 * (r % 4), 0.3 + 0.2 * (r % 3))
            elif strategy == 2:
                # Different n
                n_r = max(200, min(2500, n_best + int(rng.integers(-400, 400))))
                a_r = resample_sequence(best_seq, n_r)
            elif strategy == 3:
                # Harminic
                a_r = harmonic_sequence(n_best, 3, rng.uniform(0.2, 0.7, 3), rng.uniform(0, 2*np.pi, 3))
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                if sr > 0.01:
                    a_r = a_r / sr
                else:
                    a_r = best_seq.copy()
            else:
                # Sine-modulated
                i = np.arange(n_best)
                a_r = np.ones(n_best) * (1.0 + rng.uniform(0.2, 0.5) * np.sin(2*np.pi*i/n_best + rng.uniform(0, 2*np.pi)))
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                if sr > 0.01:
                    a_r = a_r / sr

            alloc = min(25.0, remaining * 0.3)
            a_opt = lp_alternate(a_r, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  Restart {r} (strat={strategy}): score={best_score:.8f} **")
            else:
                print(f"  Restart {r} (strat={strategy}): score={ev:.8f}")

    # Phase 7: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 7: Final polish (remaining={remaining:.0f}s) ===")
        a_final = best_seq.copy()
        s = float(np.sum(a_final))
        if s > 0:
            a_final = a_final / s

        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.35, 50.0)
            a_pol = lp_alternate(a_final, time_budget=alloc)
            ev_pol = evaluate_sequence([float(x) for x in a_pol])
            if ev_pol < best_score:
                best_score = ev_pol
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Polish {rnd}: score={best_score:.8f} **")
            else:
                print(f"  Polish {rnd}: score={ev_pol:.8f}")

            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # Subgradient
            a_sg2 = subgradient_optimize(a_final, time_budget=min(remaining * 0.3, 25.0))
            ev_sg2 = evaluate_sequence([float(x) for x in a_sg2])
            if ev_sg2 < best_score:
                best_score = ev_sg2
                best_seq = a_sg2.copy()
                a_final = a_sg2.copy()
                print(f"  Subgrad {rnd}: score={best_score:.8f} **")
            else:
                print(f"  Subgrad {rnd}: score={ev_sg2:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
