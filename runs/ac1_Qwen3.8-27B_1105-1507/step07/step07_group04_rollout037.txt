```python
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def conv_gradient_single(a, k):
    """Gradient of (conv(a,a))[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        # grad[j] = 2 * a[k-j] for j in [j_start, j_end]
        indices = np.arange(j_start, j_end + 1)
        grad[indices] = 2.0 * a[k - indices]
    return grad

def conv_subgradient(a, top_k=5):
    """Subgradient of max(conv(a,a)) at point a."""
    n = len(a)
    conv = compute_conv(a)
    peaks = np.argsort(conv)[::-1][:top_k]
    # Weight by how much each peak exceeds the second-highest
    peak_vals = conv[peaks]
    max_val = peak_vals[0]
    # Use softmax-like weighting
    weights = np.exp((peak_vals - max_val) / (0.01 * max_val + 1e-10))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for w, k in zip(weights, peaks):
        grad += w * conv_gradient_single(a, int(k))
    return grad, conv

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_for_g(a, top_k=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix A such that A @ g = conv(a, g)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Actually, conv(a,g)[k] = sum_j a[k-j] * g[j] where 0 <= k-j < n
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    if top_k is not None:
        conv = compute_conv(a)  # This is conv(a,a), not what we want
        # Actually we need to figure out which k positions are important
        # For now, use the top_k positions from conv(a,a)
        peak_positions = np.argsort(compute_conv(a))[::-1][:top_k]
        rows = peak_positions
    else:
        rows = np.arange(2 * n - 1)
    
    # Build A matrix (only the selected rows)
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[idx, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    
    # LP: min M s.t. A @ g - M <= 0, sum(g) = 1, g >= 0, M >= 0
    # Variables: [g_0, ..., g_{n-1}, M], total n+1 variables
    c = np.zeros(n + 1)
    c[-1] = 1.0  # Minimize M
    
    # Inequality constraints: A @ g - M <= 0
    A_ub = np.zeros((len(rows), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(rows))
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = np.sum(g)
        if gs > 1e-10:
            g = g / gs
        return g
    return None

            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:

            best_a = best_a_local.copy()
            a = best_a.copy()
            t = best_t
            no_improve = 0
        else:
            no_improve += 1
            t = max(t * 0.7, 0.05)
            if no_improve > 20:

    return best_a

def subgrad_optimize(a, time_budget=60, top_k=5):
    """Subgradient with Nesterov momentum."""
    n = len(a)
    a = simplex_project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    step = 0.05
    y = a.copy()  # Nesterov lookahead
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        grad, conv = conv_subgradient(y, top_k)
        # Center the gradient (remove mean to maintain sum constraint approximately)
        grad = grad - np.mean(grad)
        
        momentum = beta * momentum - step * grad
        new_a = simplex_project(a + momentum)
        
        # Nesterov update
        t_k = 2.0 / (it + 2.0)
        y = new_a + (1 - t_k) * (new_a - a)
        y = simplex_project(y)
        
        a = new_a
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Decay step size
        if it % 200 == 199:
            step *= 0.9
    
    return best_a
```

```python
    conv_ag = compute_conv(a * g_prev)  # Wait, this isn't right
    # Actually, conv(a, g)[k] = sum_j a[k-j] * g[j]
    # This is NOT the same as conv(a*g, 1) or anything simple
    # Let me compute it directly using FFT
    # conv(a, g) can be computed as ifft(fft(a) * fft(g))
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g_prev, size)
    conv_ag = irfft(fa * fg, size)[:2*n-1]
    
    # Find top-k positions
    peak_positions = np.argsort(conv_ag)[::-1][:top_k]
    
    # Build reduced A matrix
    A = np.zeros((len(peak_positions), n))
    for idx, k in enumerate(peak_positions):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[idx, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    
    # Solve reduced LP
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(peak_positions), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(peak_positions))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = np.sum(g)
        if gs > 1e-10:
            g = g / gs
        return g
    return None
```

Let me think of a simpler but effective approach:

**Key insight**: The LP alternating converges to a fixed point where a = g (i.e., the sequence is unchanged by the LP update). At this fixed point, a is a "KKT point" of the alternating optimization. The quality of the solution depends on how well we've converged.

To get a better solution:
1. Converge the LP alternating more completely (more iterations)
2. Use a better initial point
3. Perturb and re-optimize from different starting points

I think the most practical approach is:
1. Make the LP solve as fast as possible (use HiGHS, sparse if possible)
2. Do many iterations with adaptive step size
3. At the end, do a few rounds of "simultaneous" update (update both a and g at once)

For the "simultaneous" update, we can use:
- a_new = (1-t)*a + t*g_lp
- g_new = (1-t)*g + t*a_lp (where a_lp is the LP solution when g is fixed)
- This double sweep might converge faster

Actually, I just realized something important. The LP alternating I'm describing is:
- Fix a, find g = argmin max(conv(a,g)) s.t. sum(g)=1, g>=0
- Update a = (1-t)*a + t*g

This is a block coordinate descent on the function F(a) = min_g max(conv(a,g)) s.t. sum(g)=1, g>=0.

But the true objective is max(conv(a,a)), which is different from F(a). The LP gives a lower bound: max(conv(a,g_lp)) <= max(conv(a,a)) (since g=a is feasible). And after the update, the new objective is max(conv(a_new, a_new)), which we evaluate directly.

So the LP alternating is a heuristic, not an exact algorithm. The question is: can we do better?

**Alternative: Direct minimization of max(conv(a,a))**

This is a non-convex problem (max of quadratic forms). But we can use:
- Subgradient methods (as described earlier)
- Riemannian optimization (optimize on the simplex manifold)
- Iterative reweighting

Let me try a different approach: **iterative soft minimax**.

The idea: approximate the max with a soft version (e.g., log-sum-exp or power-sum), optimize the smooth approximation, then refine.

soft_max(x, p) = (sum x_k^p)^(1/p) for large p approximates max(x)

So we minimize (sum_k conv_k(a)^p)^(1/p) subject to a >= 0, sum(a) = 1.

The gradient of this is:
∇ = (1/p) * (sum_k conv_k^p)^(-1/p + 1) * p * sum_k conv_k^(p-1) * ∇conv_k

= (sum_k conv_k^p)^(-1/p + 1) * sum_k conv_k^(p-1) * ∇conv_k

The weights w_k = conv_k^(p-1) / (sum_j conv_j^(p-1)) give a weighted subgradient.

For large p, this concentrates on the maximum, recovering the hard max.

This approach has the advantage of being smooth (for finite p), which allows better gradient-based optimization.

Let me implement this:

```python
def soft_max_subgradient(a, top_k=10, p=100):
    """Compute soft-max subgradient of max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    
    # Use top-k peaks with power weighting
    peak_indices = np.argsort(conv)[::-1][:top_k]
    peak_vals = conv[peak_indices]
    
    # Power weighting
    weights = peak_vals ** (p - 1)
    weights /= np.sum(weights)
    
    # Weighted gradient
    grad = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end+1] = w * 2.0 * a[k - j_end:k - j_start + 1][::-1]
    
    return grad, conv

def smooth_optimize(a, time_budget=100, p=100, top_k=10):
    """Optimize soft-max of convolution using gradient descent with momentum."""
    n = len(a)
    a = simplex_project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    step = 0.1
    y = a.copy()
    
    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break
        
        grad, conv = soft_max_subgradient(y, top_k, p)
        grad = grad - np.mean(grad)  # Center for sum constraint
        
        momentum = beta * momentum - step * grad
        new_a = simplex_project(a + momentum)
        
        # Nesterov
        t_k = 2.0 / (it + 2.0)
        y = new_a + (1 - t_k) * (new_a - a)
        y = simplex_project(y)
        
        a = new_a
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adapt p: increase to focus on the max
        if it % 500 == 499 and p < 1000:
            p = min(p * 1.2, 1000)
        
        # Decay step
        if it % 100 == 99:
            step *= 0.95
    
    return best_a
```

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses fast LP alternating (HiGHS), smooth soft-max subgradient with Nesterov momentum,
and local perturbation search. Key improvements over prior: direct scipy HiGHS LP (faster
than cvxpy), soft-max gradient method for simultaneous two-sided updates, and active-set
LP for reduced constraint count.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    n = len(v)
    total = v.sum()
    if total <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_rows(a, rows):
    n = len(a)
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_full(a):
    n = len(a)
    a = a / np.sum(a)
    rows = np.arange(2 * n - 1)
    A = build_conv_matrix_rows(a, rows)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = np.sum(g)
        if gs > 1e-10:
            g = g / gs
        return g
    return None


def solve_lp_active(a, g_prev, n_top=30):
    n = len(a)
    a = a / np.sum(a)
    conv_ag = compute_conv_cross(a, g_prev)
    peak_positions = np.argsort(conv_ag)[::-1][:n_top]
    A = build_conv_matrix_rows(a, peak_positions)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(peak_positions), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(peak_positions))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = np.sum(g)
        if gs > 1e-10:
            g = g / gs
        # Verify no dropped constraint is violated
        conv_check = compute_conv_cross(a, g)
        if np.max(conv_check) <= float(res.x[n]) * 1.001 + 1e-10:
            return g
        else:
            return solve_lp_full(a)
    return None


def lp_alternating(a, time_budget=100.0, max_iters=500, use_active=True):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t = 0.5
    no_improve = 0
    g_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        if use_active and it > 0:
            g = solve_lp_active(a, g_prev, n_top=40)
        else:
            g = solve_lp_full(a)
        if g is None:
            break
        g_prev = g.copy()

        best_t_obj = np.inf
        best_t = t
        best_a_local = a.copy()
        candidates = sorted(set([max(0.05, t * 0.5), t, min(1.0, t * 1.5), 0.3, 0.7]))
        for t_try in candidates:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a.copy()
            t = best_t
            no_improve = 0
        else:
            no_improve += 1
            t = max(t * 0.7, 0.05)
            if no_improve > 30:
                break
    return best_a


def soft_max_subgradient(a, top_k=10, p=50):
    n = len(a)
    conv = compute_conv(a)
    peak_indices = np.argsort(conv)[::-1][:top_k]
    peak_vals = conv[peak_indices].copy()
    peak_vals = np.maximum(peak_vals, 1e-15)
    weights = peak_vals ** (p - 1)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.ones(len(peak_indices)) / len(peak_indices)
    else:
        weights /= wsum
    grad = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += w * 2.0 * a[k - j_end:k - j_start + 1][::-1]
    return grad, conv


def smooth_gradient_optimize(a, time_budget=100.0, top_k=10, p_init=20, p_max=500):
    n = len(a)
    a = simplex_project(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    step = 0.08
    y = a.copy()
    p = p_init

    for it in range(30000):
        if time.monotonic() - start > time_budget:
            break
        grad, conv = soft_max_subgradient(y, top_k, p)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm * min(gnorm, 1.0)
        momentum = beta * momentum - step * grad
        new_a = simplex_project(a + momentum)
        t_k = 2.0 / (it + 2.0)
        y = new_a + (1.0 - t_k) * (new_a - a)
        y = simplex_project(y)
        a = new_a
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        if it % 300 == 299:
            step *= 0.97
            p = min(p * 1.15, p_max)
        if it % 1000 == 999:
            beta = min(beta + 0.002, 0.98)
    return best_a


def coordinate_refine(a, time_budget=30.0):
    n = len(a)
    a = simplex_project(a.astype(np.float64).copy())
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        found = False
        for _ in range(20):
            i = int(rng.integers(0, n))
            j_candidate = k_peak - i
            if 0 <= j_candidate < n and j_candidate != i:
                j = j_candidate
            else:
                j = int(rng.integers(0, n))
                if j == i:
                    continue
            for frac in [0.02, 0.05, 0.01, 0.08]:
                delta = frac * a[i]
                if delta >= a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    break
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    found = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    a = a / np.sum(a)
                    break
        if not found and it > 100:
            break
    return best_a
