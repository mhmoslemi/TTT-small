```python
def nestorov_subgradient(a, time_budget=60.0):
    """
    Nesterov-accelerated subgradient descent for min max(conv(a,a)) s.t. sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    # Nesterov parameters
    beta = 1.0
    eta = 0.1  # step size
    x_prev = a.copy()
    x_curr = a.copy()
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = x_curr + (beta / beta_new) * (x_curr - x_prev)
        y = np.maximum(y, 0)
        ys = float(np.sum(y))
        if ys > 1e-10:
            y = y / ys
        
        # Compute subgradient
        conv = compute_conv(y)
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for j in range(n):
            v1 = y[k_star - j] if 0 <= k_star - j < n else 0.0
            v2 = y[j - k_star] if 0 <= j - k_star < n else 0.0
            grad[j] = v1 + v2
        
        # Projected gradient step
        x_new = y - eta * grad
        # Project onto {a >= 0, sum(a) = 1}
        x_new = np.maximum(x_new, 0)
        xs = float(np.sum(x_new))
        if xs > 1e-10:
            x_new = x_new / xs
        
        obj_new = compute_obj_normalized(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = x_new.copy()
        
        # Adaptive step size
        if obj_new < best_obj + 1e-8:
            eta = min(eta * 1.01, 1.0)
        else:
            eta *= 0.95
        
        x_prev = x_curr.copy()
        x_curr = x_new
        beta = beta_new
    
    return best_a
```

```python
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def obj_normalized(a):
    """2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    if abs(s - 1.0) > 1e-8:
        a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def safe_evaluate(seq):
    """Safely evaluate a sequence, returning inf for invalid inputs."""
    try:
        return evaluate_sequence(seq)
    except Exception:
        return np.inf

def build_A(a):
    """Build (2n-1) x n matrix A where A @ b = conv(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        e = min(j + n, 2 * n - 1)
        A[j:e, j] = a[:e - j]
    return A

def lp_minpeak(a, max_time=15.0):
    """LP: min M s.t. A@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return None, None
    a_n = a / s
    A = build_A(a_n)
    c = np.zeros(n + 1); c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.array([[1.0] * n + [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.status == 0 and res.x is not None:
            bv = np.maximum(res.x[:n], 0)
            bs = np.sum(bv)
            if bs > 1e-10:
                return bv / bs, float(res.x[n])
    except Exception:
        pass
    return None, None

def alt_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    stall = 0
    t_last = 0.5
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        g, M_lp = lp_minpeak(a)
        if g is None:
            break
        # Line search
        best_t = t_last
        best_t_obj = np.inf
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = np.sum(a_try)
            if st < 1e-10:
                continue
            a_try = a_try / st
            ot = obj_normalized(a_try)
            if ot < best_t_obj:
                best_t_obj = ot
                best_t = t
        if best_t_obj < best_obj - 1e-12:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            t_last = best_t
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    alt[{it}]: obj={best_obj:.8f} t={best_t:.2f}")
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(it * 7919)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
    return best_a

def subgradient_nesterov(a, time_budget=60.0):

    eta = 0.05

        if time.monotonic() - start > time_budget:
```

```python
def subgradient_at_kstar(y, k_star):
    n = len(y)
    grad = np.zeros(n)
    # First term: grad[j] += y[k_star - j] for j where 0 <= k_star - j < n
    # j ranges from max(0, k_star - n + 1) to min(n - 1, k_star)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    if j_lo <= j_hi:
        indices = np.arange(j_lo, j_hi + 1)
        grad[indices] += y[k_star - indices]
    # Second term: grad[j] += y[j - k_star] for j where 0 <= j - k_star < n
    # j ranges from max(0, k_star) to min(n - 1, k_star + n - 1)
    j_lo2 = max(0, k_star)
    j_hi2 = min(n - 1, k_star + n - 1)
    if j_lo2 <= j_hi2:
        indices2 = np.arange(j_lo2, j_hi2 + 1)
        grad[indices2] += y[indices2 - k_star]
    return grad
```

Actually, I can be even more clever. Note that:
- `y[k_star - j]` for `j = 0, ..., n-1` is `y` reversed and shifted.
