"""
Improved optimizer for low-autocorrelation non-negative sequences.
Key improvements over previous:
1. Spectral initialization using shaped power spectra
2. Multiplicative gradient descent with adaptive step size
3. Multi-scale logsumexp objective (penalizes top-K peaks, not just max)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def compute_obj_multiscale(a, K=5, beta=1.0):
    """Penalize top-K peaks with different weights."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    sorted_conv = np.sort(conv)[::-1]
    top_k = sorted_conv[:min(K, len(conv))]
    # Geometrically weighted sum of top peaks
    weights = np.array([0.5**i for i in range(len(top_k))])
    peak_sum = np.dot(weights, top_k) / np.sum(weights)
    return 2.0 * n * float(peak_sum) * beta + compute_obj(a) * (1 - beta)


def logsumexp_conv(a, tau):
    """Compute logsumexp of convolution / tau^2, which approximates max/tau + log(2n-1)/tau - 2*log(sum)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    conv = compute_conv(a)
    x = conv / tau
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    lse = x_max + np.log(np.sum(exp_x))
    F = lse / (s * s)
    return F, conv


def grad_logsumexp(a, tau):
    """Gradient of logsumexp(conv(a,a)/tau) / (sum a)^2 w.r.t. a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.zeros(n)
    conv = compute_conv(a)
    x = conv / tau
    x_max = np.max(x)
    w = np.exp(x - x_max) / np.sum(np.exp(x - x_max))
    
    # grad = 2 * (w * a_reversed correlation) / (tau * s^2) - 2 * F * a / s
    # where the correlation is: corr[i] = sum_k w[k] * a[k-i] (with bounds)
    # For conv(a,a), d(conv[k])/d(a[i]) = a[k-i] (if valid)
    # So sum_k w[k] * d(conv[k])/d(a[i]) = sum_k w[k] * a[k-i]
    
    size = next_pow2(2 * n - 1)
    # corr[i] = sum_k w[k] * a[k-i] for valid k,i
    # This is the cross-correlation of w and a
    # corr = ifft(fft(w) * conj(fft(a)))... but we need linear, not circular
    # Actually: corr[i] = sum_k w[k] * a[k-i] = (w * a_flipped)[i + n - 1]
    a_flip = a[::-1]
    fw = rfft(w, size)
    faf = rfft(a_flip, size)
    conv_w_af = irfft(fw * faf, size)[:2*n-1]
    corr = conv_w_af[n-1:n-1+n]
    
    F = (x_max + np.log(np.sum(np.exp(x - x_max)))) / (s * s)
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return grad


def multiplicative_descent(a_init, time_budget=60.0, tau_init=None, verbose=False):

    it = 0

        grad = grad_logsumexp(a, tau)
        
        # Multiplicative update: a <- a * exp(-step * grad)
        # Normalize grad for stability

        obj_new = compute_obj(a_new)

            best_a = a_new.copy()
            a = a_new
            # Gradually decrease step and tau

            b = barker.get(n, [1]*n)

            # Extended: tile a Barker and smooth

def lp_one_step(a, verbose=False):
    """One LP alternating step."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix: A @ g = conv(a_norm, g)
    # A is (2n-1) x n, A[k, i] = a_norm[min(n-1-k+i, ...)]... 
    # Actually: conv(a, g)[k] = sum_i a[i] * g[k-i]
    # So A[k, i] = a[i] if 0 <= k-i < n, else 0
    # i.e., A[k, i] = a[i] if max(0, k-n+1) <= i <= min(n-1, k)
    
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # a[i] contributes to conv[k] for k = i, i+1, ..., i+n-1
        A[i:i+n, i] = a_norm[:n]
    
    # LP: min M s.t. A@g <= M, sum(g) = 1, g >= 0
    # Variables: g[0..n-1], M
    # Using scipy linprog
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                return g_val
    except Exception:
        pass
    
    return None


def lp_alternate(a, time_budget=30.0, verbose=False):
    """LP alternating with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        g = lp_one_step(a)
        if g is None:
            break
        
        # Line search
        improved = False
        for t in [0.5, 0.3, 0.7, 1.0, 0.2, 0.8, 0.4, 0.6, 0.1, 0.9]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            # Try to perturb
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def cross_over(a1, a2, rng):
    """Geometric crossover of two sequences (must be same length)."""
    n = len(a1)
    t = rng.uniform(0.2, 0.8)
    a = a1**t * a2**(1-t)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)
    return a

"""
Improved optimizer for low-autocorrelation non-negative sequences.
Combines spectral initialization, multiplicative gradient descent with adaptive
temperature, LP alternating, spectral flattening, n-exploration, and population
diversity to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def next_pow2(m):
    """Smallest power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def logsumexp_obj_and_conv(a, tau):
    """Compute smoothed objective value and convolution."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(2 * n - 1)
    conv = compute_conv(a)
    x = conv / tau
    x_max = float(np.max(x))
    exp_x = np.exp(x - x_max)
    lse = x_max + float(np.log(np.sum(exp_x)))
    F = lse / (s * s)
    return F, conv, x_max, exp_x


def grad_logsumexp(a, tau):
    """Gradient of logsumexp(conv(a,a)/tau)/(sum a)^2 w.r.t. a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.zeros(n)
    
    conv = compute_conv(a)
    x = conv / tau
    x_max = float(np.max(x))
    exp_x = np.exp(x - x_max)
    w = exp_x / float(np.sum(exp_x))
    
    # corr[i] = sum_k w[k] * a[k-i] (cross-correlation)
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    
    lse = x_max + float(np.log(float(np.sum(exp_x))))
    F = lse / (s * s)
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return grad


def multiplicative_descent(a_init, time_budget=60.0, tau_init=None, verbose=False):
    """
    Multiplicative gradient descent on smoothed objective.
    Preserves non-negativity naturally.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / float(np.sum(a))
    
    best_a = a.copy()
    best_obj = float(compute_obj(a))
    start = time.monotonic()
    
    if tau_init is None:
        tau_init = 2.0 / n
    
    tau = tau_init
    step = 1.5 / n
    stall = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        grad = grad_logsumexp(a, tau)
        
        # Normalize gradient for stable steps
        gmax = float(np.max(np.abs(grad)))
        if gmax < 1e-18:
            break
        grad = grad / gmax
        
        # Multiplicative update
        a_new = a * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        
        obj_new = float(compute_obj(a_new))

            best_obj = obj_new
            best_a = a_new.copy()

            tau *= 0.999

        else:

                a = np.maximum(a, 1e-15)
                a = a / float(np.sum(a))
                step = 1.5 / n
                tau = max(tau, 0.1 / n)
                stall = 0
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it={it}: best={best_obj:.10f}, tau={tau:.6f}, step={step:.6f}, t={elapsed:.1f}s")

    return best_a


def spectral_flatten(a, n_iters=50, verbose=False):
    """
    Spectral flattening: iteratively damp frequency components that contribute
    to peaks in the convolution.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / float(np.sum(a))
    
    best_a = a.copy()
    best_obj = float(compute_obj(a))
    
    size = next_pow2(2 * n - 1)
    
    for it in range(n_iters):
        # Compute FFT of a (zero-padded for linear convolution)
        fa = rfft(a, size)
        # Power spectrum
        P = fa * np.conj(fa)
        
        # Compute convolution
        conv = irfft(fa * fa, size)[:2 * n - 1]
        
        # Find peak location
        k_peak = int(np.argmax(conv))
        
        # The peak at position k corresponds to constructive interference
        # at frequency 2*pi*k/(2*n). Damp that frequency in the spectrum.
        # Actually, the relationship is: conv[k] = IFFT(|A|^2)[k]
        # A peak in conv[k] means |A|^2 has a "fringe" at that position.
        # To flatten, we want to equalize |A|^2.
        
        # Strategy: compute target spectrum as geometric mean of current and flat
        P_flat = np.full(len(P), np.mean(P))
        # Blend: reduce peaks in power spectrum
        alpha = 0.1 + 0.3 * (it / max(n_iters, 1))
        P_new = (1 - alpha) * P + alpha * P_flat
        
        # But we need to preserve the phase and make DC large enough
        # New magnitude: sqrt(P_new)
        mag_new = np.sqrt(P_new)
        # Old magnitude
        mag_old = np.abs(fa)
        # Preserve phase
        phase = fa / np.maximum(mag_old, 1e-30)
        
        # New FFT
        fa_new = mag_new * phase
        fa_new[0] = np.abs(fa_new[0])  # Keep DC real and positive
        
        # Inverse FFT
        a_new = irfft(fa_new, size)[:n]
        a_new = np.maximum(a_new, 1e-15)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = float(compute_obj(a_new))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        
        if verbose and it % 10 == 0:
            print(f"    SF it={it}: best={best_obj:.10f}")
    
    return best_a


def lp_one_step(a):
    """One LP alternating step: solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build constraint matrix A (2n-1 x n): A[k,i] = a_norm[i] if k >= i and k-i < n
    # conv(a_norm, g)[k] = sum_i a_norm[i] * g[k-i] (valid when 0<=k-i<n)
    # So A[k,i] = a_norm[i] if max(0, k-n+1) <= i <= min(n-1, k)
    
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        lo = i
        hi = min(2 * n - 2, i + n - 1)
        A[lo:hi + 1, i] = a_norm[:hi - lo + 1]
    
    # LP: min M, variables = [g_0,...,g_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None


def lp_alternate(a, time_budget=30.0, verbose=False):

    n = len(a)

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)

            n_try = len(best_seq)
            a_p = spectral_init(n_try, mode=int(rng.integers(0, 8)), rng=rng)
            # Geometric blend
            t = rng.uniform(0.3, 0.7)
            a_p = best_seq ** t * a_p ** (1 - t)
        elif i == 3:
            n_try = len(best_seq)
            # Antipodal: try 1 - normalized
            a_anti = 1.0 - best_seq * float(np.sum(best_seq)) / float(np.sum(best_seq))
            a_anti = 1.0 - best_seq  # Since sum is 1, this gives 1-a
            a_anti = np.maximum(a_anti, 1e-5)
            a_p = 0.5 * best_seq + 0.5 * a_anti / float(np.sum(a_anti))
        else:
            n_try = max(400, min(2000, len(best_seq) + int(rng.integers(-200, 200))))
            a_p = spectral_init(n_try, mode=int(rng.integers(0, 8)), rng=rng)
            a_p = a_p * np.exp(rng.normal(0, 0.08, n_try))
        
        a_p = np.maximum(a_p, 1e-15)
        a_p = a_p / float(np.sum(a_p))
        
        a_opt = multiplicative_descent(a_p, time_budget=max(5.0, per_pop),
                                       tau_init=2.0/len(a_p))
        obj = float(compute_obj(a_opt))
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Pop {i} (n={len(a_opt)}): obj={best_score:.10f} **")
        else:
            print(f"  Pop {i} (n={len(a_opt)}): obj={obj:.10f}")
    
    # === Phase 8: Final Intensive Optimization ===
    print(f"\n=== Phase 8: Final Intensive ===")
    remaining = deadline - time.monotonic()
    
    # Multiple refinement rounds
    for rnd in range(4):
        if time.monotonic() > deadline - 20:
            break
        
        # MD with decreasing tau
        tau_r = (1.0 / (rnd + 1)) / len(best_seq)
        alloc = min(remaining * 0.15, 40.0)
        a_md = multiplicative_descent(best_seq, time_budget=alloc, tau_init=tau_r)
        obj = float(compute_obj(a_md))
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()
        
        # Spectral flatten
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_sf = spectral_flatten(best_seq, n_iters=min(100, int(remaining * 5)))
            obj = float(compute_obj(a_sf))
            if obj < best_score:
                best_score = obj
                best_seq = a_sf.copy()
                print(f"  Round {rnd} SF: obj={best_score:.10f}")
        
        # LP if n is small enough
        remaining = deadline - time.monotonic()
        if remaining > 40 and len(best_seq) <= 1200:
            alloc_lp = min(remaining * 0.2, 25.0)
            a_lp = lp_alternate(best_seq, time_budget=alloc_lp)
            obj = float(compute_obj(a_lp))
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  Round {rnd} LP: obj={best_score:.10f}")
    
    # === Phase 9: Coordinate Descent Polish ===
    print(f"\n=== Phase 9: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 2000:
        cd_budget = min(remaining * 0.2, 40.0)
        a_cd = coordinate_descent(best_seq, time_budget=cd_budget, verbose=True)
        obj = float(compute_obj(a_cd))
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.10f}")
    
    # === Phase 10: Final Fine MD ===
    print(f"\n=== Phase 10: Final Fine MD ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        tau_fine = 0.05 / len(best_seq)
        alloc = min(remaining * 0.6, 60.0)
        a_final = multiplicative_descent(best_seq, time_budget=alloc, tau_init=tau_fine, verbose=True)
        obj = float(compute_obj(a_final))
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
