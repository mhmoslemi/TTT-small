```python
"""
LP-based alternating minimization with active set selection and subgradient polishing.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_for_g(a, active_k, M_init=None):
```

```python
"""
LP-based alternating minimization with active-set selection, Nesterov subgradient polishing,
and multi-length exploration. Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements over prior approach: faster active-set LP, dual LP alternation,
adaptive K, diverse structured initializations, and Nesterov-accelerated subgradient.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import math


def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective value 2*n*max_conv/(sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_lp_matrices(a, active_k):
    """Build LP matrices for min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    K = len(active_k)

    A_ub = np.zeros((K, n + 1))
    for i, k in enumerate(active_k):
        k = int(k)
        # conv(a,g)[k] = sum_lj a[k-lj] * g[lj] for valid indices
        # Coefficient of g[lj] is a[k-lj] if 0 <= k-lj < n
        lj_min = max(0, k - (n - 1))
        lj_max = min(n - 1, k)
        if lj_max >= lj_min:
            # a[k-lj] for lj in [lj_min, lj_max]
            # When lj = lj_min: a[k - lj_min] = a[k - max(0, k-n+1)]
            # When lj = lj_max: a[k - lj_max] = a[k - min(n-1, k)]
            idx_start = k - lj_max
            idx_end = k - lj_min
            A_ub[i, lj_min:lj_max + 1] = a[idx_start:idx_end + 1]
        A_ub[i, n] = -1.0

    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0

    return A_ub, b_ub, A_eq, b_eq, c


def solve_lp_for_g(a, active_k):
    """Solve LP to find g minimizing max conv(a,g) over active positions."""
    n = len(a)
    A_ub, b_ub, A_eq, b_eq, c = build_lp_matrices(a, active_k)
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={"presolve": True, "dual_feasibility_tolerance": 1e-9})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                g = g / gs
                M = float(res.x[n])
                return g, M
    except Exception:
        pass
    return None, None


def solve_lp_for_a(g, active_k):
    """Solve dual LP: find a minimizing max conv(g,a) over active positions."""
    # Same formulation but with g fixed and a as variable
    return solve_lp_for_g(g, active_k)


def nesterov_subgradient(a, time_budget, top_k=10, verbose=False):
    """Nesterov-accelerated subgradient method."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()  # Nesterov lookahead
    momentum = np.zeros(n)
    beta = 0.85
    step_size = 0.03

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv_x = compute_conv(x)
        k_indices = np.argsort(conv_x)[::-1][:top_k]

        # Compute average subgradient
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            lj_min = max(0, k - (n - 1))
            lj_max = min(n - 1, k)
            if lj_max >= lj_min:
                idx_start = k - lj_max
                idx_end = k - lj_min
                grad[lj_min:lj_max + 1] += 2.0 * a[idx_start:idx_end + 1]
        grad /= len(k_indices)

        # Center gradient
        grad_centered = grad - np.mean(grad)

        # Momentum update
        momentum = beta * momentum - step_size * grad_centered
        new_a = a + momentum
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            new_a = a.copy()
            s_new = 1.0
        new_a = new_a / s_new

        # Nesterov lookahead
        t_k = 2.0 / (it + 2.0)
        x = new_a + (1.0 - t_k) * (new_a - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = new_a
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step size
        if it % 200 == 199:
            step_size = max(step_size * 0.97, 0.002)
            beta = min(beta + 0.001, 0.95)

        if verbose and it % 500 == 0:
            print(f"      NS: it={it}, obj={best_obj:.8f}, step={step_size:.5f}")

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, mode, rng):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # Cosine window
        i = np.arange(n)
        a = np.cos(np.pi * i / (2 * n)) ** 2
        return a / np.sum(a)
    elif mode == 2:
        # Exponential
        a = np.exp(-2.0 * np.arange(n) / n)
        return a / np.sum(a)
    elif mode == 3:
        # Two peaks
        a = np.zeros(n)
        a[:n // 4] = 1.0
        a[3 * n // 4:] = 1.0
        return a / np.sum(a)
    elif mode == 4:
        # Symmetric bump
        i = np.arange(n)
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - center) / (n * 0.15)) ** 2)
        return a / np.sum(a)
    elif mode == 5:
        # Random
        a = rng.random(n)
        return a / np.sum(a)
    elif mode == 6:
        # Alternating high/low
        a = np.ones(n)
        a[::2] = 1.5
        a[1::2] = 0.5
        return a / np.sum(a)
    elif mode == 7:
        # Triangular
        i = np.arange(n)
        a = 1.0 - np.abs(2.0 * i / (n - 1) - 1.0)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    else:
        return np.ones(n) / n

            best_score = obj_n
            best_seq = a_opt_n.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_n:.8f}")

    # Phase 6: Final refinement on best found
    print(f"\n=== Phase 6: Final Refinement (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP with moderate K
        alloc = min(remaining * 0.4, 100.0)
        a_final = lp_alternate(best_seq, time_budget=alloc, K=40, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    if remaining > 20:
        remaining = deadline - time.monotonic()
        # Subgradient polish
        alloc = min(remaining * 0.4, 60.0)
        a_sg2 = nesterov_subgradient(best_seq, time_budget=alloc, top_k=8, verbose=False)
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  Final SG: obj={best_score:.8f} **")
        else:
            print(f"  Final SG: obj={obj_sg2:.8f}")

    # Phase 7: Random restarts near best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            scale = 0.02 + 0.01 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

            alloc = min(remaining * 0.3, 30.0)
            a_opt_r = lp_alternate(a_r, time_budget=alloc, K=40, verbose=False)
            obj_r = compute_obj(a_opt_r)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with the actual evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
