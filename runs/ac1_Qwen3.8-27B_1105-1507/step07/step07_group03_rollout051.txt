def lp_step(a, solver='highs'):
    """Given a, find g that minimizes max conv(a,g) subject to g>=0, sum(g)=1."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build the constraint matrix
    # conv(a, g)[k] = sum_i g[i] * a[k-i] for valid i, k-i
    # We want: sum_i g[i] * a[k-i] <= M for all k
    # sum(g) = 1, g >= 0
    
    # Use the structure: A is (2n-1) x n where A[k,i] = a[k-i] if 0<=k-i<n
    
    from scipy.optimize import linprog
    # Variables: g[0..n-1], M
    # Minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality: A @ g - M <= 0 for each k
    # i.e., A @ g + (-1) * M <= 0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = build_conv_matrix(a)
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        g = g / np.sum(g)
        return g, res.x[n]
    return None, None

```python
def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_fw(a, n_iters=5000, time_budget=60.0):
    """Subgradient method with simplex projection and Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a /= a.sum()
    
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    
    x = a.copy()  # current iterate
    y = a.copy()  # proximal point (for Nesterov)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute conv at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        
        # Subgradient: g[i] = 2*y[k-i] if 0<=k-i<n else 0
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = y[j]
        
        # Normalized step size
        step = 2.0 / ((it + 1) * (np.linalg.norm(g) + 1e-8))
        
        # Update
        x_new = x - step * g
        x_new = project_simplex(x_new)
        
        # Nesterov update
        t1 = 2.0 / (it + 3.0)
        y_new = (1 - t1) * x_new + t1 * (2*x_new - x)
        y_new = np.maximum(y_new, 0)
        y_new /= y_new.sum()
        
        x = x_new
        y = y_new
        
        # Check objective
        o = obj_val(x)
        if o < best_obj:
            best_obj = o
            best_a = x.copy()
    
    return best_a

def build_A(a):
    n = len(a)
    # A[k, i] = a[k-i] if 0 <= k-i < n else 0
    # For fixed i, A[:, i] has a[0], ..., a[n-1] starting at row i

    for i in range(n):
        A[i:i+n, i] = a

```

"""
Optimized minimization of 2*n*max(conv(a,a))/(sum(a))^2 using
fast LP-alternating (HiGHS), LP-informed subgradient, multi-n exploration,
and diverse random restarts with adaptive time allocation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_val(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    an = a / s
    c = compute_conv(an)
    return 2.0 * n * float(np.max(c))


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def lp_solve(a):
    n = len(a)
    a = a / np.sum(a)
    A = build_A(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.column_stack([A, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.column_stack([np.ones(n), np.zeros(1)])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = g.sum()
        if gs < 1e-10:
            return None, None, None
        g /= gs
        # Extract dual variables for subgradient weighting
        dual = None
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            dual = np.maximum(res.ineqlin, 0)
        return g, float(res.x[-1]), dual
    return None, None, None


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def lp_alternating(a, max_iters=5000, time_budget=60.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    stall = 0
    rng_local = np.random.default_rng(hash((n, int(best_obj * 1e6))) % (2**32))

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g, M_lp, dual = lp_solve(a)
        if g is None:
            break

        # Adaptive blending
        ts = [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]
        best_local = best_obj
        best_local_a = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try /= st
            o = obj_val(a_try)
            if o < best_local:
                best_local = o
                best_local_a = a_try.copy()

        if best_local < best_obj - 1e-13:
            best_obj = best_local
            best_a = best_local_a
            a = best_a
            stall = 0
            if verbose and it % 200 == 0:
                print(f"    LP it={it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 50:
                # Perturb to escape
                scale = 0.02 + 0.01 * min(stall // 50, 5)
                a = best_a * np.exp(rng_local.normal(0, scale, n))
                a = np.maximum(a, 0)
                s2 = a.sum()
                if s2 > 1e-10:
                    a /= s2
                best_a = a.copy()
                best_obj = obj_val(a)
                stall = 0

    return best_a


def subgradient_lp_informed(a, n_iters=3000, time_budget=45.0):
    """Subgradient method using LP dual weights for multi-lag gradient."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    x = a.copy()
    momentum = np.zeros(n)
    beta = 0.9

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Get LP dual for weighted subgradient (compute less frequently for speed)
        if it % 10 == 0:
            g_lp, M_lp, dual = lp_solve(x)
            if dual is not None and g_lp is not None:
                # Build weighted subgradient from all active lags
                g = np.zeros(n)
                total_w = 0.0
                for k in range(len(dual)):
                    dk = dual[k]
                    if dk > 1e-6:
                        for i in range(n):
                            j = k - i
                            if 0 <= j < n:
                                g[i] += dk * x[j]
                        total_w += dk
                if total_w > 1e-10:
                    g /= total_w
                else:
                    g = None
            else:
                g = None
        else:
            g = None

        if g is None:
            # Fall back to single-peak subgradient
            conv_x = compute_conv(x)
            k_star = int(np.argmax(conv_x))
            g = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    g[i] = x[j]

        # Nesterov-style momentum
        momentum = beta * momentum + (1 - beta) * g
        step = 2.0 / ((it + 10) * (np.linalg.norm(momentum) + 1e-8))
        step = min(step, 0.5)

        x_new = x - step * momentum
        x_new = project_simplex(x_new)

        o = obj_val(x_new)
        if o < best_obj:
            best_obj = o
            best_a = x_new.copy()
        x = x_new

    return best_a


def random_init(n, rng):
    mode = rng.integers(0, 7)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a /= a.sum()
    elif mode == 2:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = a ** 2 + 0.05
        a /= a.sum()
    elif mode == 3:
        x = np.arange(n) / n
        a = np.exp(-3 * (x - 0.5) ** 2) + 0.05
        a /= a.sum()
    elif mode == 4:
        x = np.arange(n) / n
        a = np.exp(-6 * x ** 2) + np.exp(-6 * (x - 1) ** 2) + 0.03
        a /= a.sum()
    elif mode == 5:
        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2)) + 0.02
        a /= a.sum()
    else:
        # Smooth random via low-freq sine
        freqs = rng.uniform(0.5, 3.0, 4)
        phases = rng.uniform(0, 2 * np.pi, 4)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a /= a.sum()
    return a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if len(a) > 0 else 1.0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
