"""
Optimized step function sequence generator minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast sparse LP alternating with momentum, peak-shaving local search,
frequency-domain gradient methods, and adaptive multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy import sparse
import warnings
warnings.filterwarnings('ignore')


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_sparse_conv_matrix(a):
    """Build sparse convolution matrix A (2n-1 x n) where A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_cross_conv(a):
    """
    Solve LP: min M s.t. conv(a, g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    A = build_sparse_conv_matrix(a)
    
    # Build full problem in standard form for linprog
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ g - M <= 0 for each k
    A_ub = A.copy().tocsc()
    col_M = sparse.csc_matrix((-np.ones(2 * n - 1), (np.arange(2 * n - 1), np.zeros(2 * n - 1, dtype=int))),
                               shape=(2 * n - 1, 1))
    A_ub_full = sparse.hstack([A_ub, col_M]).tocsr()
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

    """Faster LP solve using dense matrix for small n, sparse for large n."""
    n = len(a)
    if n > 1500:
        return solve_lp_cross_conv(a)

    A_dense = np.zeros((2 * n - 1, n))
    for i in range(n):
        for j in range(n):
            k = i + j
            if k < 2 * n - 1:
                A_dense[k, i] += a[j]
                # Actually conv(a,g)[k] = sum_i a[i]*g[k-i], so A[k,i] = a[k-i]
    # Redo properly
    A_dense = np.zeros((2 * n - 1, n))
    for i in range(n):
        for j in range(n):

    c = np.zeros(n + 1)

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_dense
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:

def peak_shaving(a, time_budget=30.0):
    """
    Local search that directly reduces contributions to peak convolution bins.
    For each peak bin k, reduce a[i] where a[i]*a[k-i] contributes most.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find pairs (i, k_star-i) that contribute most
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Reduce the top contributors slightly, redistribute to others
        top_frac = max(1, n // 20)
        top_idx = np.argsort(contributions)[-top_frac:]
        bottom_idx = np.argsort(contributions)[:, :n - top_frac] if n > top_frac else np.arange(n)
        
        if len(bottom_idx) == 0:
            break
        
        # Compute amount to shift
        top_mean = np.mean(a[top_idx])
        bot_mean = np.mean(a[bottom_idx])
        if top_mean <= bot_mean:
            break
        
        shift = 0.02 * (top_mean - bot_mean)
        a_new = a.copy()
        a_new[top_idx] -= shift
        # Redistribute proportionally to bottom
        bot_sum = np.sum(a[bottom_idx])
        if bot_sum < 1e-10:
            break
        a_new[bottom_idx] += shift * a[bottom_idx] / bot_sum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller shift
            shift *= 0.5
            a_new = a.copy()
            a_new[top_idx] -= shift
            a_new[bottom_idx] += shift * a[bottom_idx] / max(bot_sum, 1e-10)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    it += 1
                    if it > 200:
                        break
            else:
                break
        it += 1

    return best_a


def freq_domain_descent(a, time_budget=40.0):
    """
    Gradient descent in frequency domain. 
    Minimize max(ifft(|A|^2)) by adjusting |A|^2 to be flatter,
    while maintaining a = ifft(A) >= 0.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = 0.001  # temperature for smooth max
    lr = 0.01
    stall = 0
    
    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        
        # Gradient of smooth max w.r.t. a
        # d(smooth_max)/d(a[i]) = sum_k weights[k] * a[k-i] (for valid k-i)
        grad = np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)
        
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0 and lr < 0.05:
                lr = min(lr * 1.02, 0.05)
        else:
            stall += 1
            if stall > 50:
                lr *= 0.7
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break
        
        if it > 100:
            T = max(0.0001, T * 0.995)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng, mode=0):
    """Generate diverse random initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Smooth via cumulative sum of noise
        noise = rng.normal(0, 1, n)
        a = np.cumsum(noise) - np.min(np.cumsum(noise)) + 0.1
        a = a / np.sum(a)
    elif mode == 3:
        # Periodic + noise
        period = max(2, n // rng.integers(3, 8))
        x = np.arange(n)
        a = 1 + 0.3 * np.sin(2 * np.pi * x / period) + 0.1 * rng.normal(0, 1, n)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    elif mode == 4:
        # Multi-scale
        scales = [n // 3, n // 5, n // 7]
        x = np.arange(n)
        a = np.ones(n)
        for s in scales:
            a += 0.5 * np.sin(2 * np.pi * x / s + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 5:
        # Heavy tail in middle
        center = n / 2
        x = np.arange(n)
        a = 1.0 / (1.0 + 0.1 * (x - center) ** 2 / n)
        a = a / np.sum(a)
    elif mode == 6:
        # Alternating high/low
        x = np.arange(n)
        a = np.where(x % 2 == 0, 1.2, 0.8) + 0.1 * rng.normal(0, 1, n)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines fast LP alternating with momentum, peak shaving,
    frequency-domain descent, and adaptive multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.8f}")

    a = best_seq.copy()
    n_cur = len(a)

    # === Phase 1: Fast LP Alternating with Momentum (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating Momentum (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    phase1_total = min(500.0, remaining * 0.45)
    
    # Split into multiple rounds with perturbation between
    n_rounds = 4
    for round_idx in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = phase1_total / (n_rounds - round_idx)
        alloc = min(alloc, remaining * 0.5)
        
        a_new = lp_alternating_momentum(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP-Round {round_idx}: obj={best_score:.8f}")
        else:
            print(f"  LP-Round {round_idx}: obj={obj:.8f} (no improvement)")
        
        # Perturb between rounds
        if round_idx < n_rounds - 1:
            scale = 0.02
            a = best_seq * np.exp(rng.normal(0, scale, n_cur))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

    # === Phase 2: Peak Shaving ===
    print(f"\n=== Phase 2: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(40.0, remaining * 0.1)
        a_ps = peak_shaving(best_seq, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            a = a_ps.copy()
            print(f"  Peak Shaving: obj={best_score:.8f}")
        else:
            print(f"  Peak Shaving: obj={obj_ps:.8f} (no improvement)")

    # === Phase 3: Frequency Domain Descent ===
    print(f"\n=== Phase 3: Frequency Domain Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(35.0, remaining * 0.1)
        a_fd = freq_domain_descent(best_seq, time_budget=alloc)
        obj_fd = compute_obj(a_fd)
        if obj_fd < best_score:
            best_score = obj_fd
            best_seq = a_fd.copy()
            a = a_fd.copy()
            print(f"  Freq Descent: obj={best_score:.8f}")
        else:
            print(f"  Freq Descent: obj={obj_fd:.8f} (no improvement)")

    # === Phase 4: LP Alternating on best (second pass) ===
    print(f"\n=== Phase 4: LP Pass 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(100.0, remaining * 0.25)
        a_lp2 = lp_alternating_momentum(best_seq, time_budget=alloc, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            a = a_lp2.copy()
            print(f"  LP Pass 2: obj={best_score:.8f}")
        else:
            print(f"  LP Pass 2: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    # Focus on n values near the best, plus some spread
    n_options = sorted(set([
        max(300, n_best - 200),
        max(300, n_best - 100),
        max(300, n_best - 50),
        n_best - 10,
        n_best,
        n_best + 10,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        min(2000, n_best + 400),
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        # Quick LP
        alloc = min(remaining * 0.12, 40.0)
        a_opt = lp_alternating_momentum(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        
        # Peak shave if promising
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.005:
            a_opt = peak_shaving(a_opt, time_budget=min(remaining * 0.05, 10.0))
            obj_try = compute_obj(a_opt)
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Random Restarts with Structured Inits ===
    print(f"\n=== Phase 6: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        init_mode = r % 7
        n_r = n_best + int(rng.integers(-100, 101))
        n_r = max(400, min(1600, n_r))
        a_r = random_init(n_r, rng, mode=init_mode)
        
        alloc_r = min(remaining * 0.1, 30.0)
        a_opt = lp_alternating_momentum(a_r, time_budget=alloc_r)
        
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt = peak_shaving(a_opt, time_budget=min(remaining * 0.03, 8.0))
        
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()
        
        # LP polish
        alloc = min(remaining * 0.3, 60.0)
        a_lp = lp_alternating_momentum(a_final, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")
        
        # Peak shave
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_ps = peak_shaving(a_final, time_budget=min(remaining * 0.2, 20.0))
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                a_final = a_ps.copy()
                print(f"  Final PeakShave: obj={best_score:.8f}")
        
        # Freq descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_fd = freq_domain_descent(a_final, time_budget=min(remaining * 0.2, 15.0))
            obj_fd = compute_obj(a_fd)
            if obj_fd < best_score:
                best_score = obj_fd
                best_seq = a_fd.copy()
                print(f"  Final Freq: obj={best_score:.8f}")
        
        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_lp2 = lp_alternating_momentum(best_seq, time_budget=min(remaining * 0.3, 20.0))
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  Final LP2: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
