import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, timeout=60.0):
    """
    Solve LP: min M s.t. conv(a,b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b_normalized, M_opt) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    # Try cvxpy first
    try:
        import cvxpy as cp
        b = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ b <= M, cp.sum(b) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:

                prob.solve(solver=solver, verbose=False)
                if b.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    s = float(np.sum(a))

"""
Minimize evaluate_sequence = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses LP alternating optimization with high-precision solves, diverse initializations,
adaptive line search, n-exploration, and perturbation-based diversification.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, timeout=60.0):
    """
    Solve LP: min M s.t. conv(a,b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b_normalized, M_opt) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None

    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        b = cp.Variable(n, nonneg=True)
        M = cp.Variable()

        prob = cp.Problem(cp.Minimize(M), constraints)

        # Try solvers in order of expected performance

            except Exception:

    except Exception:

        c = np.zeros(n + 1)
        c[-1] = 1.0

        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):

def constrained_looking_optimize(a, time_budget=30.0):

            a_try = a.copy()
            for idx in high_idx:
                a_try[idx] -= epsilon
            for idx in low_idx:
                a_try[idx] += epsilon
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP alternating with:
    1. Diverse initializations and multi-start
    2. Systematic n-exploration
    3. Perturbation-based diversification
    4. Peak deflation as auxiliary method
    5. Adaptive time allocation
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Establish baseline
    print("=== Phase 0: Baseline ===")

    parent = None
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr
                n_parent = parent.size
                score = evaluate_sequence([float(x) for x in parent])
                if score < np.inf:
                    print(f"  Parent (n={n_parent}): score={score:.8f}")
                    best_score = score
                    best_seq = parent.copy()
    except Exception as e:
        print(f"  Parent not available: {e}")

    if best_seq is None or best_score >= 2.0:
        seq_const = [1.0] * 800
        score_const = evaluate_sequence(seq_const)
        print(f"  Constant n=800: score={score_const:.8f}")
        if score_const < best_score:
            best_score = score_const
            best_seq = np.ones(800)

    if best_seq is None:
        best_seq = np.ones(800)
        best_score = evaluate_sequence([float(x) for x in best_seq])

    print(f"  Initial best: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: Intensive LP optimization of parent
    remaining = deadline - time.monotonic()
    if remaining > 60 and parent is not None:
        print(f"\n=== Phase 1: LP optimize parent (remaining={remaining:.0f}s) ===")
        a_p = parent / np.sum(parent)
        alloc = min(200.0, remaining * 0.2)
        a_opt = lp_alternate(a_p, time_budget=alloc, verbose=True)
        ev = evaluate_sequence([float(x) for x in a_opt])
        print(f"  Parent LP: score={ev:.8f}")
        if ev < best_score:
            best_score = ev
            best_seq = a_opt.copy()

    # Phase 2: N-exploration with LP optimization
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: N-exploration (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_candidates = sorted(set([
            200, 300, 400, 500, 600, 700, 800, 900, 1000,
            1100, 1200, 1500, 2000, 2500,
            max(100, n_best - 100), max(100, n_best + 100),
            max(100, n_best - 200), max(100, n_best + 200),
        ]))

        per_budget = max(20.0, remaining * 0.35 / len(n_candidates))

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < per_budget + 10:
                break
            if n_try == n_best and best_seq is not None:
                continue

            a_try = resample_sequence(best_seq, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try

            alloc = min(per_budget, remaining * 0.1)
            a_opt = lp_alternate(a_try, time_budget=alloc)
            ev_try = evaluate_sequence([float(x) for x in a_opt])
            if ev_try < best_score:
                best_score = ev_try
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: score={best_score:.8f} **")
            else:
                print(f"  n={n_try}: score={ev_try:.8f}")

    # Phase 3: Random restarts with LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: Random restarts (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)

        per_restart = 35.0
        max_restarts = max(5, int((remaining - 80) / per_restart))
        max_restarts = min(max_restarts, 40)

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < per_restart + 15:
                break

            if r % 5 == 0:
                a_r = random_init(n_best, rng)
            elif r % 5 == 1:
                # Small perturbation of best
                a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                a_r = a_r / sr if sr > 0 else np.ones(n_best) / n_best
            elif r % 5 == 2:
                # Larger perturbation
                a_r = best_seq * np.exp(rng.normal(0, 0.1, n_best))
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                a_r = a_r / sr if sr > 0 else np.ones(n_best) / n_best
            elif r % 5 == 3:
                # Blend with random
                a_rand = random_init(n_best, rng)
                mix = rng.uniform(0.3, 0.7)
                a_r = mix * best_seq + (1 - mix) * a_rand
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                a_r = a_r / sr if sr > 0 else np.ones(n_best) / n_best
            else:
                # Different n
                n_r = max(200, min(2000, n_best + int(rng.integers(-200, 200))))
                a_r = resample_sequence(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.05, n_r))
                a_r = np.maximum(a_r, 0)
                sr = float(np.sum(a_r))
                a_r = a_r / sr if sr > 0 else np.ones(n_r) / n_r

            alloc = min(per_restart, remaining * 0.12)
            a_opt = lp_alternate(a_r, time_budget=alloc)
            ev_r = evaluate_sequence([float(x) for x in a_opt])
            if ev_r < best_score:
                best_score = ev_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: n={len(a_opt)}, score={best_score:.8f} **")
            else:
                print(f"  Restart {r}: score={ev_r:.8f}")

    # Phase 4: Peak deflation on best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 4: Peak deflation (remaining={remaining:.0f}s) ===")
        a_def = best_seq.copy()
        s_def = float(np.sum(a_def))
        if s_def > 0:
            a_def = a_def / s_def
        alloc = min(30.0, remaining * 0.15)
        a_opt_def = constrained_looking_optimize(a_def, time_budget=alloc)
        ev_def = evaluate_sequence([float(x) for x in a_opt_def])
        print(f"  Deflation: score={ev_def:.8f}")
        if ev_def < best_score:
            best_score = ev_def
            best_seq = a_opt_def.copy()

    # Phase 5: Intensive final polishing
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 5: Final polish (remaining={remaining:.0f}s) ===")
        a_final = best_seq.copy()
        s = float(np.sum(a_final))
        if s > 0:
            a_final = a_final / s

        for round_idx in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.2, 50.0)
            a_pol = lp_alternate(a_final, time_budget=alloc)
            ev_pol = evaluate_sequence([float(x) for x in a_pol])
            if ev_pol < best_score:
                best_score = ev_pol
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Polish {round_idx}: score={best_score:.8f} **")
            else:
                print(f"  Polish {round_idx}: score={ev_pol:.8f}")
                if ev_pol >= best_score - 1e-12:
                    break

    # Phase 6: Ultra-fine perturbation search
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Phase 6: Ultra-fine tune (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)

        for trial in range(30):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            # Very small perturbation
            scale = 0.002 + 0.001 * trial
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_pert = np.maximum(a_pert, 0)
            sp = float(np.sum(a_pert))
            a_pert = a_pert / sp if sp > 0 else best_seq.copy()

            alloc = min(12.0, remaining * 0.08)
            a_tuned = lp_alternate(a_pert, time_budget=alloc)
            ev_tuned = evaluate_sequence([float(x) for x in a_tuned])
            if ev_tuned < best_score:
                best_score = ev_tuned
                best_seq = a_tuned.copy()
                print(f"  Fine {trial}: score={best_score:.8f} **")
            elif trial < 5:
                # Also try the perturbation directly with quick LP
                pass

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
