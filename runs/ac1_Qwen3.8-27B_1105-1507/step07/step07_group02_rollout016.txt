    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses HiGHS solver. Returns normalized g or None.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def alternating_lp(a, time_budget=60.0, verbose=False):
    """LP alternating optimization with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    consecutive_full = 0

    max_iters = 500
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        g = solve_lp_fast(a)
        if g is None:
            break

        # Try blending: prefer full replacement, fall back to partial
        t_list = [1.0, 0.75, 0.5, 0.3, 0.15]
        if consecutive_full >= 3:
            t_list = [1.0, 0.85, 0.6, 0.4]

        best_t_obj = np.inf
        best_t_val = 1.0
        best_a_local = best_a.copy()

        for t in t_list:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if best_t_val >= 0.99:
                consecutive_full += 1
            else:
                consecutive_full = 0
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.10f}, t={best_t_val:.2f}")
        else:
            stall += 1
            consecutive_full = 0
            if stall > 8:
                rng = np.random.default_rng(it * 104729 + n * 31)
                scale = 0.03 + 0.02 * min(stall // 8, 4)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_p = float(np.sum(a))
                if s_p > 1e-10:
                    a = a / s_p
                stall = 0

    return best_a


def gradient_polish(a, time_budget=30.0):
    """Subgradient descent on the peak convolution value."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.03

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[i] = a[k_star - i] if valid
        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        grad_centered = grad - np.mean(grad)
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.15)
        else:
            lr *= 0.92
            if lr < 1e-6:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def structured_init(n, mode, rng):
    """Generate structured initial sequences."""
    x = np.arange(n)
    if mode == 0:
        # Constant
        return np.ones(n) / n
    elif mode == 1:
        # Cosine bell
        a = np.cos(np.pi * x / n)**2 + 0.01
    elif mode == 2:
        # Two humps
        c1, c2 = n * 0.25, n * 0.75
        w = n * 0.12
        a = np.exp(-((x - c1)**2) / (2 * w**2)) + np.exp(-((x - c2)**2) / (2 * w**2))
    elif mode == 3:
        # Raised cosine with noise
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
        a += rng.normal(0, 0.02, n)
    elif mode == 4:
        # Smooth random
        a = rng.uniform(0.5, 1.0, n)
        k = max(3, n // 40)
        kernel = np.exp(-0.5 * (np.arange(-k, k + 1) / (k * 0.5))**2)
        a = np.convolve(a, kernel / kernel.sum(), mode='same')
    elif mode == 5:
        # Triangular wave
        a = 1.0 - 2.0 * np.abs(x / n - 0.5)
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Sinusoidal modulation
        freq = rng.uniform(1, 4)
        a = 0.7 + 0.3 * np.sin(2 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
    else:
        # Near-uniform with structured noise
        a = 1.0 + 0.1 * np.sin(2 * np.pi * rng.uniform(1, 5) * x / n)
        a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s if s > 0 else np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + gradient polish + multi-n exploration.
    Returns best sequence found within budget.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Load parent and establish baseline
    print("=== Phase 0: Initialization ===")
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr.copy()
                score = evaluate_sequence([float(x) for x in parent])
                if score < np.inf:
                    best_score = score
                    best_seq = parent.copy()
                    print(f"  Parent: n={parent.size}, score={score:.8f}")
    except Exception as e:
        print(f"  No parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"  Baseline constant n=800: score={best_score:.8f}")

    # Phase 1: Main LP optimization on best (parent or baseline)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 1: LP Optimize (n={len(best_seq)}, budget={remaining:.0f}s) ===")
        a_work = best_seq.copy()
        s = float(np.sum(a_work))
        if s > 0:
            a_work = a_work / s

        # Give most time to this phase
        alloc = min(remaining * 0.55, 450.0)
        a_opt = alternating_lp(a_work, time_budget=alloc, verbose=True)
        ev = evaluate_sequence([float(x) for x in a_opt])
        print(f"  LP result: score={ev:.8f}")
        if ev < best_score:
            best_score = ev
            best_seq = a_opt.copy()

    # Phase 2: Gradient polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 2: Gradient Polish (remaining={remaining:.0f}s) ===")
        a_work = best_seq.copy()
        s = float(np.sum(a_work))
        if s > 0:
            a_work = a_work / s
        alloc = min(remaining * 0.15, 40.0)
        a_gp = gradient_polish(a_work, time_budget=alloc)
        ev = evaluate_sequence([float(x) for x in a_gp])
        print(f"  Gradient result: score={ev:.8f}")
        if ev < best_score:
            best_score = ev
            best_seq = a_gp.copy()

    # Phase 3: Multi-n exploration
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: Multi-n Exploration (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 300), max(200, n_best - 150),
            n_best,
            n_best + 150, n_best + 300, n_best + 500,
            600, 1000, 1500, 2000, 2500, 3000
        ]))
        n_options = [n for n in n_options if 200 <= n <= 4000 and n != n_best]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_try = resample_sequence(best_seq, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try

            alloc = min(25.0, remaining * 0.08)
            a_opt = alternating_lp(a_try, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: score={best_score:.8f} **")
            else:
                print(f"  n={n_try}: score={ev:.8f}")

    # Phase 4: Structured restarts
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: Structured Restarts (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        num_restarts = min(15, int(remaining / 25))

        for r in range(num_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            mode = r % 7
            n_r = n_best
            if mode == 3:
                n_r = max(400, n_best + int(rng.integers(-200, 400)))
            elif mode == 5:
                n_r = max(400, n_best + int(rng.integers(-100, 300)))

            a_r = structured_init(n_r, mode, rng)

            # Small perturbation of best for some modes
            if mode in [0, 1, 6]:
                a_r = 0.5 * a_r + 0.5 * resample_sequence(best_seq, n_r)
                s_r = float(np.sum(a_r))
                if s_r > 0:
                    a_r = a_r / s_r

            alloc = min(remaining * 0.08, 25.0)
            a_opt = alternating_lp(a_r, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_opt])
            if ev < best_score:
                best_score = ev
                best_seq = a_opt.copy()
                print(f"  Restart {r} (mode={mode}, n={n_r}): {best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): {ev:.8f}")

    # Phase 5: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 5: Final Polish (remaining={remaining:.0f}s) ===")
        a_final = best_seq.copy()
        s = float(np.sum(a_final))
        if s > 0:
            a_final = a_final / s

        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.25, 40.0)
            a_pol = alternating_lp(a_final, time_budget=alloc)
            ev = evaluate_sequence([float(x) for x in a_pol])
            if ev < best_score:
                best_score = ev
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Polish {r}: {best_score:.8f} **")
            else:
                print(f"  Polish {r}: {ev:.8f}")

        # Quick gradient
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_gp = gradient_polish(a_final, time_budget=min(remaining * 0.3, 15.0))
            ev = evaluate_sequence([float(x) for x in a_gp])
            if ev < best_score:
                best_score = ev
                best_seq = a_gp.copy()
                print(f"  Grad polish: {best_score:.8f} **")
            else:
                print(f"  Grad polish: {ev:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 800
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
