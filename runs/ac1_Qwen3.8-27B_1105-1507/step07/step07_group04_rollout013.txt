"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_obj_and_grad(a, beta):
    """
    Compute smooth objective and gradient.
    f(a) = (2n/beta) * logsumexp(beta * c[k] / s^2)
    where c = conv(a,a), s = sum(a)
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    
    c = compute_conv(a)
    u = c / (s * s)
    
    # Softmax weights
    max_u = np.max(u)
    exp_u = np.exp(beta * (u - max_u))
    Z = np.sum(exp_u)
    w = exp_u / Z  # softmax weights, sum to 1
    
    # Objective
    obj = (2.0 * n / beta) * (np.log(Z) + beta * max_u)
    
    # Gradient: df/da[j] = (2n/s^2) * (w convolve a)[j] - (4n*(w·c))/s^3
    # Compute (w ⊛ a)[j] = sum_k w[k] * a[k-j]
    # Using FFT: ifft(fft(w) * conj(fft(a)))
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    W = fft(w, size)
    A = fft(a, size)
    corr = np.real(ifft(W * np.conj(A), size))[:n]
    
    wc = np.dot(w, c)
    
    grad = (2.0 * n / (s * s)) * corr - (4.0 * n * wc) / (s ** 3)
    
    return obj, grad

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def nesterov_smooth_opt(a, time_budget=300.0, beta_start=10.0, beta_end=100.0,

    Nesterov-accelerated gradient descent on the smooth objective.
    Beta is gradually increased for sharper max approximation.
    """

    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()  # current iterate
    y = a.copy()  # Nesterov lookahead
    t = 1.0
    eta = 1.0 / np.sqrt(n)  # initial step size
    
    beta = beta_start
    beta_ratio = (beta_end / beta_start) ** (1.0 / 100.0)
    
    no_improve = 0
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        # Update beta periodically
        if it % 500 == 499:
            beta *= beta_ratio
            beta = min(beta, beta_end)
            eta = 1.0 / np.sqrt(n * beta)
            t = 1.0  # reset momentum
        
        # Compute gradient at y
        obj_y, grad_y = smooth_obj_and_grad(y, beta)
        
        # Nesterov update
        x_new = y - eta * grad_y
        x_new = project_simplex(x_new, 1.0)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = np.maximum(y_new, 0)
        s_y = float(np.sum(y_new))
        if s_y > 1e-10:
            y_new = y_new / s_y
        
        x = x_new
        y = y_new
        t = t_new
        
        # Evaluate actual objective

            obj_actual = compute_obj(x)
            if obj_actual < best_obj - 1e-12:
                best_obj = obj_actual
                best_a = x.copy()
                no_improve = 0
            else:
                no_improve += 1

                x = best_a * np.exp(rng.normal(0, 0.02, n))

                t = 1.0

def peak_directed_mwu(a, time_budget=200.0, verbose=True):

        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        obj = 2.0 * n * max_b
        
        # Soft targeting: weight each lag by exp(beta * conv[k] / max_b)
        w = np.exp(beta * (conv - max_b) / max_b)
        w /= np.sum(w)
        
        # Gradient of weighted sum: sum_k w[k] * a[k-j]
        size = 1
        while size < len(w) + n - 1:
            size <<= 1

        A = fft(a, size)
        grad = np.real(ifft(W * np.conj(A), size))[:n]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad / max(grad) if max(grad) > 0 else 0)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.2)
        else:
            eta *= 0.95
            if eta < 0.001:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 100) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                eta = 0.03
                t = 1.0

            print(f"    PeakMWU: obj={best_obj:.8f}, eta={eta:.5f}, t={time.monotonic()-start:.0f}s")
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception as e:
        print(f"    LP error: {e}")
    
    return None, None

def lp_alternating(a, time_budget=200.0, verbose=True):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = np.inf
        best_a_local = best_a.copy()
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
    
    return best_a

def generate_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size == n and np.sum(parent) > 0.01:
            inits.append(('parent', parent / np.sum(parent)))
    except:
        pass
    
    # Random uniform
    a = rng.uniform(0.1, 1.0, n)
    inits.append(('uniform', a / np.sum(a)))
    
    # Exponential
    a = np.exp(rng.normal(0, 0.5, n))
    inits.append(('exp_random', a / np.sum(a)))
    
    # Linear ramp
    a = np.linspace(0.1, 1.0, n)
    inits.append(('ramp_up', a / np.sum(a)))
    a = np.linspace(1.0, 0.1, n)
    inits.append(('ramp_down', a / np.sum(a)))
    
    # Two-level
    a = np.ones(n)
    a[rng.integers(0, n, n // 2)] = 0.2
    inits.append(('two_level', a / np.sum(a)))
    
    # Cosine bump
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    inits.append(('cosine', a / np.sum(a)))
    
    # Sparse peak
    a = np.zeros(n)
    a[n // 4:3 * n // 4] = 1.0
    inits.append(('block', a / np.sum(a)))
    
    # Log-normal
    a = np.exp(rng.normal(0, 1.0, n))
    inits.append(('lognormal', a / np.sum(a)))
    
    return inits

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
