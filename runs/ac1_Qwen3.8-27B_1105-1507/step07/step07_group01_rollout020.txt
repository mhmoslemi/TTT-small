"""
Optimized sequence search using multi-peak subgradient descent with Nesterov momentum,
spectral equalization (Blahut-type), coordinate descent polishing, and aggressive
multi-n exploration. Designed to beat the 1.506340 baseline.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_multi_peak(a, T, top_k_frac=0.1):

    Blahut-type spectral equalization: modify the spectrum to flatten the autocorrelation.
    """

        # Modify the power spectrum

        new_power = power - alpha * np.real(E)
        new_power = np.maximum(new_power, 1e-12)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)

        return np.array([0.0])

    while size < 2 * n - 1:

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a = np.ones(n)
    elif mode == 1:
        a = np.exp(-0.8 * x / n)
    elif mode == 2:
        a = np.exp(-0.8 * (n - 1 - x) / n)
    elif mode == 3:
        a = np.cos(np.pi * x / (2 * n)) ** 2 + 0.05
    elif mode == 4:
        center = n * 0.35
        width = n * 0.25
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 5:
        a = 1.0 - 0.7 * np.cos(4 * np.pi * x / n)
    elif mode == 6:
        a = rng.dirichlet(np.ones(n) * 0.3) * n
    elif mode == 7:
        a = np.minimum(x / (0.3 * n), (n - 1 - x) / (0.7 * n)) + 0.05
    elif mode == 8:
        a = np.exp(-((x - n * 0.3) ** 2) / (2 * (n * 0.15) ** 2)) + \
            0.6 * np.exp(-((x - n * 0.7) ** 2) / (2 * (n * 0.12) ** 2))
    elif mode == 9:
        a = np.sin(np.pi * x / n) ** 1.5 + 0.1
    else:
        a = np.ones(n) * (1 + 0.3 * rng.standard_normal(n))
        a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def anneal_and_optimize(a, time_budget=30.0):
    """
    Annealing approach: start with smooth (low-frequency) modifications,
    gradually add higher frequency detail.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    n_frequencies = size // 2 + 1
    for stage in range(5):
        stages = [5, 20, 50, 100, n_frequencies]
        max_freq = stages[stage]
        for inner in range(10):
            if time.monotonic() - start_time_global > deadline_global:
                return best_a
            A = rfft(a, size)
            mask = np.zeros(n_frequencies, dtype=bool)
            mask[:max_freq] = True
            conv = compute_conv(a)
            peak_k = int(np.argmax(conv))
            grad = np.zeros(n)
            offsets = peak_k - np.arange(n)
            valid = (offsets >= 0) & (offsets < n)
            grad[valid] = 2.0 * a[offsets[valid]]
            grad_centered = grad - np.mean(grad)
            G = rfft(grad_centered, size)
            G[~mask] = 0
            step = 0.01 * (1 + 0.5 * stage)
            A_new = A - step * G
            a_new = irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    a = best_a.copy()
    return best_a


def try_lp_fast(a, n_iters=50):
    """
    Fast LP approximation using iterative "chase the peak" on the dual.
    Instead of solving LP exactly, use subgradient iterations to approximate.
    """
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    a = a / s
    g = np.ones(n) / n
    for it in range(n_iters):
        A_g = compute_conv(a, g) if hasattr(compute_conv, 'two_args') else None
        # Compute conv(a, g) manually
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a, size)
        fg = rfft(g, size)
        conv_ag = irfft(fa * fg, size)[:2 * n - 1]
        k_star = int(np.argmax(conv_ag))
        # Subgradient w.r.t. g: dg[k_star]/d[g[i]] = a[k_star - i]
        grad_g = np.zeros(n)
        offsets = k_star - np.arange(n)
        valid = (offsets >= 0) & (offsets < n)
        grad_g[valid] = a[offsets[valid]]
        grad_g_centered = grad_g - np.mean(grad_g)
        lr = 0.1 / (1 + it * 0.05)
        g_new = simplex_project(g - lr * grad_g_centered)
        g = g_new
    return g


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining subgradient,
    spectral methods, coordinate descent, and multi-n exploration.
    """
    global start_time_global, deadline_global
    start_time_global = time.monotonic()
    deadline_global = start_time_global + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"[Parent] n={parent_n}, obj={obj:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        parent_n = 800
        print(f"[Init] n={parent_n}, obj={best_score:.8f}")

    # === Phase 1: Intensive subgradient on parent ===
    print(f"\n[Phase 1] Subgradient optimization (n={len(best_seq)})")
    a = best_seq.copy()
    budget_p1 = min(200.0, (deadline_global - time.monotonic()) * 0.25)
    a_opt = subgradient_descent(a, time_budget=budget_p1, verbose=True)
    obj = compute_obj(a_opt)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  -> obj={best_score:.8f}")
    else:
        print(f"  -> obj={obj:.8f} (no improvement)")

    # === Phase 2: Spectral equalization ===
    print(f"\n[Phase 2] Spectral equalization")
    remaining = deadline_global - time.monotonic()
    if remaining > 50:
        a_spec = spectral_equalize(best_seq, n_iters=30, alpha=0.05)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  -> obj={best_score:.8f} (power eq)")

        a_phase = spectral_phase_equalize(best_seq, n_iters=20, gamma=0.2)
        obj_phase = compute_obj(a_phase)
        if obj_phase < best_score:
            best_score = obj_phase
            best_seq = a_phase.copy()
            print(f"  -> obj={best_score:.8f} (phase eq)")

        # Follow up with subgradient
        remaining = deadline_global - time.monotonic()
        if remaining > 30:
            a_follow = subgradient_descent(best_seq, time_budget=min(40.0, remaining * 0.5))
            obj_follow = compute_obj(a_follow)
            if obj_follow < best_score:
                best_score = obj_follow
                best_seq = a_follow.copy()
                print(f"  -> follow-up subgrad: obj={best_score:.8f}")

    # === Phase 3: Coordinate descent polish ===
    print(f"\n[Phase 3] Coordinate descent polish")
    remaining = deadline_global - time.monotonic()
    if remaining > 30:
        a_cd = coordinate_polish(best_seq, time_budget=min(25.0, remaining * 0.3), multi_pass=1)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  -> obj={best_score:.8f}")
        else:
            print(f"  -> obj={obj_cd:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[Phase 4] Multi-n exploration")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50),
        n_best, n_best + 50, n_best + 100, n_best + 200,
        n_best + 400,
        600, 800, 1000, 1200, 1400, 1600, 1800
    ]))
    n_candidates = [x for x in n_candidates if 400 <= x <= 2000]

    for n_try in n_candidates:
        remaining = deadline_global - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick subgradient
        alloc = min(remaining * 0.12, 25.0)
        a_opt_n = subgradient_descent(a_try, time_budget=alloc)
        obj_n = compute_obj(a_opt_n)

        # Quick spectral
        remaining = deadline_global - time.monotonic()
        if remaining > 20 and obj_n < best_score + 0.01:
            a_spec_n = spectral_equalize(a_opt_n, n_iters=10, alpha=0.05)
            obj_spec_n = compute_obj(a_spec_n)
            if obj_spec_n < obj_n:
                a_opt_n = a_spec_n
                obj_n = obj_spec_n

        if obj_n < best_score:
            best_score = obj_n
            best_seq = a_opt_n.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_n:.8f}")

    # === Phase 5: Diverse random restarts ===
    print(f"\n[Phase 5] Diverse random restarts")
    n_best = len(best_seq)
    for r in range(15):
        remaining = deadline_global - time.monotonic()
        if remaining < 25:
            break

        init_mode = r % 10
        n_r = int(n_best + rng.integers(-100, 150))
        n_r = max(500, min(1800, n_r))
        a_r = diverse_init(n_r, init_mode, rng)

        # Blend with current best for potential synergy
        if best_seq is not None and r % 3 == 0:
            a_resampled = resample_sequence(best_seq, n_r)
            a_r = 0.5 * a_r + 0.5 * a_resampled
            a_r = a_r / np.sum(a_r)

        alloc_r = min(remaining * 0.1, 30.0)
        a_opt = subgradient_descent(a_r, time_budget=alloc_r)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r} (mode={init_mode}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={init_mode}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n[Phase 6] Final intensive polish")
    remaining = deadline_global - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        # Multiple subgradient passes with different T
        for t_init in [0.005, 0.001, 0.0005]:
            remaining = deadline_global - time.monotonic()
            if remaining < 15:
                break
            a_sm = subgradient_descent(a_final, time_budget=min(remaining * 0.25, 40.0))
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                a_final = a_sm.copy()
                print(f"  Subgrad (T={t_init}): obj={best_score:.8f}")

        # Spectral final
        remaining = deadline_global - time.monotonic()
        if remaining > 15:
            a_sf = spectral_equalize(a_final, n_iters=20, alpha=0.04)
            obj_sf = compute_obj(a_sf)
            if obj_sf < best_score:
                best_score = obj_sf
                best_seq = a_sf.copy()
                print(f"  Spectral final: obj={best_score:.8f}")

        # Coordinate final
        remaining = deadline_global - time.monotonic()
        if remaining > 15:
            a_cf = coordinate_polish(best_seq, time_budget=min(remaining * 0.3, 20.0), multi_pass=1)
            obj_cf = compute_obj(a_cf)
            if obj_cf < best_score:
                best_score = obj_cf
                best_seq = a_cf.copy()
                print(f"  Coord final: obj={best_score:.8f}")

        # One more subgradient
        remaining = deadline_global - time.monotonic()
        if remaining > 10:
            a_final2 = subgradient_descent(best_seq, time_budget=min(remaining * 0.4, 20.0))
            obj_f2 = compute_obj(a_final2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_final2.copy()
                print(f"  Final subgrad: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time_global
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
