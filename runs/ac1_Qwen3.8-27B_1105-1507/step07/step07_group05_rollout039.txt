```python
def smooth_max_gradient(a, T):
    """Compute gradient of smooth max of conv(a,a) using log-sum-exp."""
    n = len(a)
    N = 1
    while N < 3*n - 1:
        N <<= 1
    
    # Compute b = conv(a, a)
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2*n-1]
    
    # Compute weights w[k] = exp(b[k]/T) / sum(exp(b/T))
    b_shifted = b - np.max(b)
    w = np.exp(b_shifted / T)
    w = w / np.sum(w)
    
    # Compute gradient
    a_flip = a[::-1]
    fw = rfft(w, N)
    faf = rfft(a_flip, N)
    conv_result = irfft(fw * faf, N)
    # g[j] = 2 * conv(w, a_flip)[n-1+j]
    g = 2.0 * conv_result[n-1:n+n-1]
    
    # Also compute the smooth max value
    smooth_max_val = T * np.log(np.sum(np.exp(b/T - np.max(b)/T)))  # shift for numerical stability
    # Actually: T * log(sum(exp(b/T))) = T * log(sum(exp((b - max_b)/T))) + max_b
    smooth_max_val = np.max(b) + T * np.log(np.sum(np.exp(b_shifted / T)))
    
    return g, smooth_max_val
```

        # If we normalize sum_a = 1, obj = 2n * max_b
        # So we just need to minimize max_b (or smooth_max_b)
        
        # Center the gradient (to maintain normalization)
        g = g - np.mean(g)
        
        # Momentum update
        v = momentum * v - eta * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Adapt parameters
            eta = min(eta * 1.02, 1.0)
            T = max(T * 0.99, 0.01)  # slowly decrease temperature
        else:
            eta *= 0.95
            v *= 0.5
            if eta < 1e-8:
                break
    
    return best_a

Let me also think about the temperature schedule. The temperature T controls the "softness" of the max:
- Large T: all peaks contribute to gradient → more global
- Small T: only the max peak contributes → more local, sharper

A good strategy: start with moderate T, decrease over time.

Another idea: **Adaptive temperature** based on the "effective number of peaks" contributing:
`N_eff = 1 / sum(w[k]^2)` (inverse participation ratio)
If N_eff is too large (too many peaks contributing), decrease T.
If N_eff is too small (only 1 peak), maybe increase T slightly for exploration.

Let me also think about the step size / line search:
- After computing the gradient direction d, do a backtracking line search:
  - Start with large step
  - Halve until we get improvement
  - This is robust and doesn't require knowing the Lipschitz constant

Let me also add a few more tricks:
- **Normalization trick**: Since the objective is homogeneous (scaling a by c multiplies both max_b and sum_a^2 by c^2, so the objective is invariant), we can normalize a to have sum=1 and just minimize 2n*max_b.
- **Clipping strategy**: After gradient step, clip to [0, ∞) and renormalize.
- **Annealing**: Periodically perturb the solution to escape local minima.

"""
Smooth max-convolution minimizer using log-sum-exp gradient with adaptive temperature,
momentum, and line search. Combined with LP polishing and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv_ffft(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = compute_conv_ffft(a)
    return 2.0 * n * float(np.max(b))


def compute_obj(a):
    """Compute full objective 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def smooth_max_and_gradient(a, T):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Uses log-sum-exp for numerical stability.
    Returns (smooth_max_value, gradient_vector).
    """
    n = len(a)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    
    # Compute b = conv(a, a)
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]
    
    # Softmax weights
    b_max = float(np.max(b))
    b_shifted = b - b_max
    exp_vals = np.exp(b_shifted / T)
    w = exp_vals / np.sum(exp_vals)
    
    # Smooth max value
    smooth_val = b_max + T * float(np.log(np.sum(exp_vals)))
    
    # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # = 2 * conv(w, a_flip)[n-1+j]
    a_flip = a[::-1]
    fw = rfft(w, N)
    faf = rfft(a_flip, N)
    conv_w_af = irfft(fw * faf, N)
    g = 2.0 * conv_w_af[n - 1:2 * n - 1]
    
    return smooth_val, g


def smooth_max_gd(a, time_budget=100.0, verbose=False):
    """
    Gradient descent on smooth max of auto-convolution.
    Adaptive temperature, momentum, and backtracking line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    T = 0.05  # temperature
    eta = 0.5  # step size
    momentum = 0.6
    v = np.zeros(n)
    stall = 0
    max_stall = 300
    
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    
    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient
        _, g = smooth_max_and_gradient(a, T)
        
        # Center gradient (preserve normalization)
        g_centered = g - np.mean(g)
        
        # Momentum update
        v = momentum * v - eta * g_centered
        
        # Try multiple step sizes (backtracking)
        found_improvement = False
        step = eta
        for trial in range(10):
            a_try = a + step * v
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                stall = 0
                found_improvement = True
                # Adapt
                eta = min(eta * 1.01, 2.0)
                T = max(T * 0.9995, 0.005)
                momentum = min(momentum + 0.0005, 0.9)
                break
            step *= 0.5
        
        if not found_improvement:
            stall += 1
            eta *= 0.9
            v *= 0.5
            if stall > max_stall:
                # Perturb to escape local minimum
                rng_local = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
                a = best_a * np.exp(rng_local.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.6
                stall = 0
                max_stall = min(max_stall + 200, 2000)
            if eta < 1e-6:
                break
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SMGD it{it}: obj={best_obj:.8f}, T={T:.4f}, eta={eta:.4f}, time={elapsed:.1f}s")
    
    return best_a


def chase_top_k_peaks(a, time_budget=30.0, k=5):
    """
    Subgradient descent targeting the top-k peaks of the convolution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    eta = 0.1
    
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]
        
        # Find top-k peaks
        top_k_indices = np.argsort(b)[-k:][::-1]
        
        # Average gradient from top-k peaks
        g = np.zeros(n)
        for kp in top_k_indices:
            for j in range(n):
                diff = int(kp) - j
                if 0 <= diff < n:
                    g[j] += a[diff]
        g *= 2.0 / k
        
        # Center
        g = g - np.mean(g)
        
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.5)
        else:
            eta *= 0.95
            if eta < 1e-7:
                break
    
    return best_a


def solve_lp_minpeak(a):
    """Solve LP to minimize peak of conv(a, g) subject to sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s
    
    # Build constraint matrix: conv(a, g)[k] = sum_j A[k,j] * g[j]
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For efficiency, use structure
    
    try:
        from scipy.optimize import linprog
        # Variables: g[0], ..., g[n-1], M
        # Minimize M
        # Subject to: conv(a,g)[k] <= M for k=0..2n-2
        #             sum(g) = 1
        #             g >= 0, M >= 0
        
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M
        
        # A_ub: (2n-1) x (n+1)
        # Row k: conv(a,g)[k] - M <= 0
        # conv(a,g)[k] = sum_j A[k,j] * g[j]
        # A[k,j] = a[k-j] if 0<=k-j<n
        
        A_ub = np.zeros((2 * n - 1, n + 1))
        b_ub = np.zeros(2 * n - 1)
        
        for k in range(2 * n - 1):
            for j in range(n):
                diff = k - j
                if 0 <= diff < n:
                    A_ub[k, j] = a[diff]
            A_ub[k, n] = -1.0  # -M
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g
    except Exception:
        pass
    
    return None


def lp_polish(a, time_budget=60.0):
    """LP-based alternating polish."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search for mixing parameter
        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = (1 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break
    
    return best_a


def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        sigma = max(n // 8, 1)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * sigma ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        a = a / np.sum(a)
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / x**0.3
        a = a / np.sum(a)
    elif mode == 6:
        # Smooth random (low-pass filtered noise)
        a = rng.normal(0, 1, n)
        # Simple moving average smoothing
        kernel_size = max(n // 20, 1)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.abs(a) + 0.1
        a = a / np.sum(a)
    else:
        # Multi-modal
        n_modes = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(n_modes):
            c = rng.integers(0, n)
            sigma = rng.uniform(n // 20, n // 5)
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * sigma ** 2))
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
Smooth max-convolution minimizer using log-sum-exp gradient with adaptive temperature,
conjugate gradient momentum, backtracking line search, multi-k peak chasing,
LP polishing, and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv_fft(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = compute_conv_fft(a)
    return 2.0 * n * float(np.max(b))


def compute_obj_full(a):
    """Compute full objective 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def smooth_max_and_gradient(a, T, N=None):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a using log-sum-exp.
    Returns (smooth_max_value, gradient_vector, b_array).
    """
    n = len(a)
    if N is None:
        N = 1
        while N < 3 * n - 1:
            N <<= 1

    # Compute b = conv(a, a)
    fa = rfft(a, N)
    b = irfft(fa * fa, N)[:2 * n - 1]

    # Softmax weights for smooth max
    b_max = float(np.max(b))
    b_shifted = b - b_max
    exp_vals = np.exp(b_shifted / T)
    exp_sum = float(np.sum(exp_vals))
    if exp_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = exp_vals / exp_sum

    # Smooth max value
    smooth_val = b_max + T * float(np.log(exp_sum))

    # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (for valid 0<=k-j<n)
    # = 2 * conv(w, a_flip)[n-1+j]
    a_flip = a[::-1]
    fw = rfft(w, N)
    faf = rfft(a_flip, N)
    conv_w_af = irfft(fw * faf, N)
    g = 2.0 * conv_w_af[n - 1:2 * n - 1]

    return smooth_val, g, b

def chase_top_k_peaks(a, time_budget=30.0, k=5):
    """Subgradient descent targeting top-k peaks of the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    eta = 0.08

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:2 * n - 1]

        # Top-k peaks
        top_k = np.argsort(b)[-k:][::-1]

        # Build weight vector
        w = np.zeros(2 * n - 1)
        for kp in top_k:
            w[int(kp)] = 1.0 / k

        # Gradient via FFT: g[j] = 2 * conv(w, a_flip)[n-1+j]
        a_flip = a[::-1]
        fw = rfft(w, N)
        faf = rfft(a_flip, N)
        conv_result = irfft(fw * faf, N)
        g = 2.0 * conv_result[n - 1:2 * n - 1]
        g = g - np.mean(g)

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.003, 0.4)
        else:
            eta *= 0.95
            if eta < 1e-7:
                break

    return best_a


def solve_lp_minpeak(a, timeout=20.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s

    try:
        from scipy.optimize import linprog

        c = np.zeros(n + 1)
        c[-1] = 1.0

        A_ub = np.zeros((2 * n - 1, n + 1))
        b_ub = np.zeros(2 * n - 1)

        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A_ub[k, j] = a[k - j]
            A_ub[k, n] = -1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0.0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g
    except Exception:
        pass
    return None

        if remaining_lp < 5.0:

        g_lp = solve_lp_minpeak(a, timeout=min(remaining_lp * 0.7, 15.0))

        best_t_obj = np.inf
        best_t = 0.5
        for t in [0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:

            best_a = (1.0 - best_t) * a + best_t * g_lp
            best_a = np.maximum(best_a, 0.0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)

    print(f"\n=== Phase 1: Smooth Max GD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 250.0)
    a_sm = smooth_max_gd(best_seq, time_budget=alloc, verbose=True)
    obj_sm = compute_obj_normalized(a_sm)
    if obj_sm < best_score:
        best_score = obj_sm
        best_seq = a_sm.copy()
        print(f"  ** SMGD: obj={best_score:.8f} **")
    else:
        print(f"  SMGD: obj={obj_sm:.8f}")

    # === Phase 2: Conjugate Gradient ===
    print(f"\n=== Phase 2: Smooth Max CG ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 120.0)
        a_cg = smooth_max_cg(best_seq, time_budget=alloc, verbose=True)
        obj_cg = compute_obj_normalized(a_cg)
        if obj_cg < best_score:
            best_score = obj_cg
            best_seq = a_cg.copy()
            print(f"  ** CG: obj={best_score:.8f} **")
        else:
            print(f"  CG: obj={obj_cg:.8f}")

    # === Phase 3: Multi-k Peak Chase ===
    print(f"\n=== Phase 3: Multi-k Peak Chase ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for k in [3, 7, 15]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.1, 30.0)
            a_pk = chase_top_k_peaks(best_seq, time_budget=alloc, k=k)
            obj_pk = compute_obj_normalized(a_pk)
            if obj_pk < best_score:
                best_score = obj_pk
                best_seq = a_pk.copy()
                print(f"  ** Peak k={k}: obj={best_score:.8f} **")
            else:
                print(f"  Peak k={k}: obj={obj_pk:.8f}")

    # === Phase 4: LP Polish ===
    print(f"\n=== Phase 4: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.12, 90.0)
        a_lp = lp_polish(best_seq, time_budget=alloc)
        obj_lp = compute_obj_normalized(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 5: SMGD Round 2 ===
    print(f"\n=== Phase 5: SMGD Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.18, 150.0)
        a_sm2 = smooth_max_gd(best_seq, time_budget=alloc, verbose=True)
        obj_sm2 = compute_obj_normalized(a_sm2)
        if obj_sm2 < best_score:
            best_score = obj_sm2
            best_seq = a_sm2.copy()
            print(f"  ** SMGD2: obj={best_score:.8f} **")
        else:
            print(f"  SMGD2: obj={obj_sm2:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_curr = len(best_seq)
    n_options = sorted(set([
        n_curr + 50, n_curr + 100, n_curr + 200, n_curr + 400,
        n_curr - 100, n_curr - 200,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    n_options = [x for x in n_options if 300 <= x <= 2500 and x != n_curr]
    n_options = n_options[:7]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 50.0)
        a_opt = smooth_max_gd(a_try, time_budget=alloc)
        obj_try = compute_obj_normalized(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Random Restarts ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_curr = len(best_seq)
    n_restarts = max(2, int((deadline - time.monotonic()) / 35.0))
    n_restarts = min(n_restarts, 8)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_r = random_init(n_curr, rng)
        alloc = min(remaining * 0.15, 35.0)
        a_opt = smooth_max_gd(a_r, time_budget=alloc)
        obj_r = compute_obj_normalized(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 60.0)
        a_final = smooth_max_gd(best_seq, time_budget=alloc)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final SMGD: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_lp = lp_polish(best_seq, time_budget=min(remaining * 0.4, 25.0))
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  ** Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_cg = smooth_max_cg(best_seq, time_budget=min(remaining * 0.4, 20.0))
            obj_cg = compute_obj_normalized(a_cg)
            if obj_cg < best_score:
                best_score = obj_cg
                best_seq = a_cg.copy()
                print(f"  ** Final CG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
