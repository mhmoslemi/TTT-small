"""
Improved optimizer for low-autocorrelation non-negative step functions.
Key improvements: asymmetric initializations, sharper temperature schedule,
FFT-based peak-pushing gradient, LP alternating refinement, local search,
and systematic n-exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(x, tau):
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)

    size2 = next_pow2(3 * n)
    a_rev = a[::-1]
    fa_rev = rfft(a_rev, size2)
    fw = rfft(w, size2)
    corr_full = irfft(fa_rev * fw, size2)
    corr = corr_full[n - 1:n - 1 + n]

    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    taus = [5.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n, 0.02 / n]

    x = np.log(a) - np.log(a).max()

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break

        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val

        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g

        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxls': 50})

        if res.fun < fun(x):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = x_opt
                    if verbose:
                        print(f"    grad tau={tau:.4f}: obj={best_obj:.8f}")

    return best_a


def peak_push_optimize(a_init, time_budget=30.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_peaks = min(3, 2 * n - 1)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        current_obj = 2.0 * n * float(np.max(conv))

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()

        peak_indices = np.argsort(conv)[-n_peaks:]

        size2 = next_pow2(3 * n)
        w_peak = np.zeros(2 * n - 1)
        for k in peak_indices:
            w_peak[k] = 1.0

        a_rev = a[::-1]
        fa_rev = rfft(a_rev, size2)
        fw = rfft(w_peak, size2)
        corr_full = irfft(fa_rev * fw, size2)
        grad = corr_full[n - 1:n - 1 + n]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        best_try = None
        for lr in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_try = a - lr * grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if best_try is None or obj_try < best_try[1]:
                best_try = (a_try, obj_try)

        if best_try is None:
            break

        a_new, obj_new = best_try
        if obj_new < best_obj - 1e-12:
            a = a_new
        else:
            break

    return best_a


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, use_cvxpy=True):
    n = len(a)
    a_norm = a / np.sum(a)

    if use_cvxpy:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    try:
        A = build_conv_matrix(a_norm)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternate_one_step(a):
    n = len(a)
    a_norm = a / np.sum(a)

    g_lp, M_lp = solve_lp_minpeak(a_norm)
    if g_lp is None:
        return a, compute_obj(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8]:
        a_try = (1.0 - t) * a_norm + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def lp_alternate_optimize(a, time_budget=60.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        a_new, obj_new = lp_alternate_one_step(a)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 8:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 8, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def local_search(a_init, time_budget=30.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        delta = rng.uniform(-0.02, 0.02)
        a_try = a.copy()
        a_try[i] += delta
        a_try[j] -= delta
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def asymmetric_init(n, rng):
    mode = rng.integers(0, 8)
    x = np.arange(n) / max(n - 1, 1)
    if mode == 0:
        a = 1.0 - x
    elif mode == 1:
        a = np.exp(-3.0 * x)
    elif mode == 2:
        a = (1.0 - x) ** 2
    elif mode == 3:
        a = np.exp(-5.0 * x) + 0.3
    elif mode == 4:
        a = np.sqrt(1.0 - x)
    elif mode == 5:
        a = 1.0 / (1.0 + 5.0 * x)
    elif mode == 6:
        a = (1.0 - x) ** 0.5 + 0.2 * np.exp(-10.0 * x)
    else:
        a = 1.0 - x + 0.3 * np.sin(np.pi * x)
    a = a + 0.05 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.001)
    a = a / np.sum(a)
    return a


def random_init(n, rng):
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        a = rng.uniform(0, 1, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)
    elif mode == 3:
        a = 1.0 - np.abs(np.arange(n) - n / 2) / (n / 2)
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 4:
        c1, c2 = n // 3, 2 * n // 3
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n * n / 20)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n * n / 20))
        a = a / np.sum(a)
    else:
        a = np.ones(n) + 0.3 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    return a
