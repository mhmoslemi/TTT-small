Advanced optimizer for low-autocorrelation non-negative sequences.
Uses Frank-Wolfe (conditional gradient), LP alternating, spectral methods,

Key improvements over previous version:
- Fast O(n) Frank-Wolfe iterations
- Better multi-n exploration (up to 2500)
- Spectral smoothing for escaping local minima
- Adaptive step sizes with backtracking

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def frank_wolfe_optimize(a_init, time_budget=60.0, verbose=False):
    """
    Frank-Wolfe (Conditional Gradient) for min max_k conv_k(a) on the simplex.
    Very fast: O(n log n) per iteration (FFT for conv + O(n) for argmax/argmin).
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Frank-Wolfe step size schedule: t_k = 2/(k+2)
    it = 0
    eta = 1.0  # initial step for the "gap" based step
    
    while time.monotonic() - start < time_budget:
        batch_end = time.monotonic() + 0.01  # do a batch of iterations
        
        while time.monotonic() < batch_end:
            it += 1
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
            
            # Compute subgradient: grad[i] = a[k_star - i] + a[i - k_star] (with bounds)
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += a[j]
                j2 = i - k_star
                if 0 <= j2 < n:
                    grad[i] += a[j2]
            
            # Linear minimization oracle: find vertex e_{i*} minimizing grad
            i_star = int(np.argmin(grad))
            
            # Frank-Wolfe step size: t = 2/(it+2)
            t = 2.0 / (it + 2)
            
            # Update
            a_new = (1.0 - t) * a + t * (1.0 / n)  # wait, the vertex on the simplex
            # Actually the vertex is e_{i*} scaled... no.

            # But e_i has sum 1 only if it's the standard basis vector.
            # So the LW (linear minimization) gives us v = e_{i*} (the unit vector at i*).
            # Update: a_new = (1-t) * a + t * v
            
            v = np.zeros(n)
            v[i_star] = 1.0
            a_new = (1.0 - t) * a + t * v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            a_new = a_new / s_new

            if obj_new < best_obj:

                best_a = a_new.copy()
            
            a = a_new
    
    return best_a

"""
Advanced hybrid optimizer for low-autocorrelation non-negative sequences.
Combines Frank-Wolfe, LP alternating, smoothed gradient (L-BFGS-B),
spectral smoothing, and multi-n exploration to minimize
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(x, tau):
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))
    conv = compute_conv(a)
    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = np.zeros(size // 2 + 1)
    # Need w padded to size for FFT
    w_padded = np.zeros(size)
    w_padded[:len(w)] = w
    fw = rfft(w_padded, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n]
    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    taus = [10.0/n, 5.0/n, 2.0/n, 1.0/n, 0.5/n, 0.2/n, 0.1/n, 0.05/n]
    x = np.log(a) - np.log(a).max()
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g
        
        res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                       options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-11,
                                'maxls': 50})
        
        if res.success or res.fun < fun(x):
            x_opt = res.x - res.x.max()
            a_new = np.exp(x_opt)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = x_opt
                if verbose:
                    print(f"    grad tau={tau:.5f}: obj={best_obj:.8f}")
    
    return best_a


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    n = len(a)
    a_norm = a / np.sum(a)
    
    try:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass
    
    return None, None

def spectral_smooth(a, sigma):
    """Apply Gaussian smoothing in frequency domain."""
    n = len(a)
    size = next_pow2(n)
    fa = rfft(a, size)
    freqs = np.fft.rfftfreq(size)
    kernel = np.exp(-2 * np.pi**2 * (sigma**2) * freqs**2)
    fa_smooth = fa * kernel
    a_smooth = irfft(fa_smooth, size)[:n]
    return np.maximum(a_smooth, 0)

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
