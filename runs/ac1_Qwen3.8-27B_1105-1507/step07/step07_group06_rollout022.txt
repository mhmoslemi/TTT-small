"""
LP-Alternating + FFT-accelerated Smoothed Gradient with ramp initializations.
Key insight: Linear ramps give theoretical objective ~4/3 (vs 2 for uniform).
We initialize with power-law ramps and polish aggressively with LP alternating,
using FFT-accelerated smoothed gradient for escaping local minima.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return np.real(ifft(fa * fa, size))[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def fft_smoothed_grad(a, tau):
    """
    Compute gradient of smooth_max(conv(a,a), tau) w.r.t. a using FFT.
    Returns gradient array of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)

    # Softmax weights for smooth max
    w = tau * np.exp((conv - conv_max) * tau)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.zeros(n), conv_max
    w = w / w_sum

    # grad[i] = 2 * sum_{k=i}^{i+n-1} w[k] * a[k-i]
    # This is cross-correlation: (w * reversed_a)[n-1+i]
    b = a[::-1]  # reversed a, length n
    # conv(w, b) has length (2n-1) + n - 1 = 3n-2
    # We need positions n-1, n, ..., 2n-2 (which is indices n-1 to 2n-2)
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    W = fft(w, size)
    B = fft(b, size)
    conv_wb = np.real(ifft(W * B, size))
    # grad[i] = 2 * conv_wb[i + n - 1] for i = 0,...,n-1
    grad = 2.0 * conv_wb[n - 1:2 * n - 1]
    return grad, conv_max


def smoothed_gradient_descent(a, tau, steps=500, step_size=0.1, time_budget=30.0):
    """Smoothed gradient descent on simplex with FFT-accelerated gradients."""
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    ss = step_size
    patience = 0

    for it in range(steps):
        if time.monotonic() - start > time_budget:
            break

        grad, _ = fft_smoothed_grad(a, tau)

        # Project to tangent space of simplex (sum-preserving)
        grad = grad - np.mean(grad)

        a_new = a - ss * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                ss = min(ss * 1.05, 1.0)
                patience = 0
            else:
                patience += 1
                ss *= 0.9
                if patience > 50:
                    break
        else:
            ss *= 0.7
            patience += 1
            if patience > 50:
                break

    return best_a, best_obj

    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_ramp(n, power=1.0, rng=None):
    """Generate a power-law ramp: a[i] = (i+1)^power."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    return a / np.sum(a)


def generate_exponential(n, alpha=1.0, rng=None):
    """Generate exponential ramp: a[i] = exp(alpha * i/(n-1))."""
    i = np.linspace(0, 1, n)
    a = np.exp(alpha * i)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: ramp initializations + LP alternating + smoothed gradient escape.
    Key insight: linear ramps give theoretical obj ~4/3, much better than uniform (~2).
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = generate_ramp(800, power=1.0)
        best_score = compute_obj(best_seq)
        print(f"Default ramp (n=800): obj={best_score:.8f}")

    # === Phase 1: Evaluate diverse initializations ===
    print(f"\n=== Phase 1: Initialization Evaluation ===")
    n_base = len(best_seq)
    candidates = []

    # Power-law ramps with different exponents
    for power in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        for n_try in [n_base]:
            a_ramp = generate_ramp(n_try, power=power)
            obj_ramp = compute_obj(a_ramp)
            candidates.append((obj_ramp, a_ramp, f"ramp_p{power}_n{n_try}"))
            print(f"  Ramp p={power}, n={n_try}: obj={obj_ramp:.8f}")

    # Exponential ramps
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        a_exp = generate_exponential(n_base, alpha=alpha)
        obj_exp = compute_obj(a_exp)
        candidates.append((obj_exp, a_exp, f"exp_a{alpha}"))
        print(f"  Exp alpha={alpha}: obj={obj_exp:.8f}")

    # Truncated ramps (zero out some portion)
    for frac in [0.1, 0.2, 0.3]:
        a_trunc = generate_ramp(n_base, power=1.0)
        a_trunc[:int(frac * n_base)] = 0
        s_t = np.sum(a_trunc)
        if s_t > 0:
            a_trunc = a_trunc / s_t
            obj_trunc = compute_obj(a_trunc)
            candidates.append((obj_trunc, a_trunc, f"trunc_{frac}"))
            print(f"  Trunc frac={frac}: obj={obj_trunc:.8f}")

    # Inverted ramps (decreasing)
    for power in [1.0, 1.5]:
        a_inv = generate_ramp(n_base, power=power)
        a_inv = a_inv[::-1]
        obj_inv = compute_obj(a_inv)
        candidates.append((obj_inv, a_inv, f"inv_ramp_p{power}"))
        print(f"  Inverted ramp p={power}: obj={obj_inv:.8f}")

    # Parent (if available)
    if best_seq is not None:
        candidates.append((best_score, best_seq.copy(), "parent"))

    # Sort by objective
    candidates.sort(key=lambda x: x[0])
    print(f"  Best init: {candidates[0][2]} with obj={candidates[0][0]:.8f}")

    # === Phase 2: Intensive optimization on top candidates ===
    print(f"\n=== Phase 2: Intensive LP + Gradient Optimization ===")
    top_k = min(3, len(candidates))
    for idx in range(top_k):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        obj_init, a_init, name = candidates[idx]
        alloc = min(remaining * 0.4, 500.0)
        print(f"  Optimizing '{name}' (obj={obj_init:.6f}), budget={alloc:.0f}s")

        a_opt, obj_opt = lp_with_gradient_escape(a_init, time_budget=alloc, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: obj={best_score:.8f} (from {name})")
        else:
            print(f"  No improvement: obj={obj_opt:.8f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 200), max(400, n_cur - 100),
        n_cur, n_cur + 100, n_cur + 200,
        min(1500, n_cur + 400),
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.2, 80.0)
        a_opt, obj_opt = lp_with_gradient_escape(a_try, time_budget=alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Targeted peak reduction ===
    print(f"\n=== Phase 4: Targeted Peak Reduction ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_best = len(best_seq)
        a_cur = best_seq.copy() / np.sum(best_seq)
        conv = compute_conv(a_cur)
        peak_idx = int(np.argmax(conv))
        print(f"  Peak at index {peak_idx}, value {conv[peak_idx]:.10f}")

        # Try to reduce the peak by adjusting specific coordinates
        alloc = min(remaining * 0.3, 60.0)
        start_p4 = time.monotonic()

        # Identify which pairs (i, j) contribute to the peak
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i]
        contributors = []
        for i in range(n_best):
            j = peak_idx - i
            if 0 <= j < n_best:
                contributors.append((i, a_cur[i] * a_cur[j]))
        contributors.sort(key=lambda x: -x[1])

        # Reduce the largest contributors
        for i, contrib_val in contributors[:30]:
            if time.monotonic() - start_p4 > alloc:
                break
            # Try reducing a[i] slightly and distributing to a "quiet" position
            for reduce_frac in [0.02, 0.05, 0.1]:
                delta = reduce_frac * a_cur[i]
                if delta <= 0:
                    continue
                # Find a position where adding mass won't increase the peak
                a_try = a_cur.copy()
                a_try[i] -= delta
                # Add to position 0 (or wherever it's smallest contribution)
                a_try[0] += delta
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score - 1e-12:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        a_cur = a_try
                        break

    # === Phase 5: Final intensive LP polish ===
    print(f"\n=== Phase 5: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 120.0)
        print(f"  Running final LP with budget={alloc:.0f}s")
        a_final, obj_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 6: One more gradient pass ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for tau in [100.0, 300.0]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_g, obj_g = smoothed_gradient_descent(
                best_seq, tau=tau, steps=300, step_size=0.05,
                time_budget=min(remaining * 0.3, 20.0))
            if obj_g < best_score:
                best_score = obj_g
                best_seq = a_g.copy()
                print(f"  Final grad tau={tau}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = generate_ramp(800)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluate_sequence
    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
