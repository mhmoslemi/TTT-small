"""
Optimized search for low-autocorrelation non-negative sequences.
Uses active-set subgradient, LP polishing, and diverse initialization.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def active_set_subgradient(a, time_budget=60.0, K=5, verbose=False):
    """
    Subgradient method focusing on the top-K convolution terms.
    This is more effective than the full max subgradient because it
    spreads the descent direction over multiple active constraints.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.1
    patience = 0
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Get top-K indices
        top_k = np.argsort(conv)[-K:]
        
        # Compute subgradient: gradient of sum of top-K conv terms
        grad = np.zeros(n)
        for k in top_k:
            # d(conv[k])/da[i] = a[k-i] if 0 <= k-i < n
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += a[k - i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Try a few step sizes
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1]:
            step_try = step * factor
            a_new = a - step_try * grad
            a_new = np.maximum(a_new, 1e-15)
            s = float(np.sum(a_new))
            if s < 1e-10:
                continue
            a_new = a_new / s
            conv_new = compute_conv(a_new)
            max_new = float(np.max(conv_new))
            if max_new < max_val - 1e-12:
                a = a_new
                obj_new = 2.0 * n * max_new
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    patience = 0
                else:
                    patience += 1
                improved = True
                step = min(step * 1.05, 1.0)
                break
        
        if not improved:
            step *= 0.7
            patience += 1
            if patience > 50:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                step = 0.1
                patience = 0
    
    return best_a

"""
Hybrid optimizer for low-autocorrelation non-negative sequences.
Combines L-BFGS-B (log-sum-exp smoothed), Frank-Wolfe/conditional gradient,
momentum subgradient, LP alternating, and diverse asymmetric initialization
to minimize 2*n*max(convolve(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def smoothed_obj_and_grad(x, tau):
    """
    Smoothed log-sum-exp objective and gradient for a = exp(x - max(x)).
    Returns (F, grad_x) where F approximates max(conv(exp(x)))/sum(exp(x))^2.
    """
    n = len(x)
    x = x - np.max(x)
    a = np.exp(x)
    s = float(np.sum(a))

    conv = compute_conv(a)

    c_scaled = conv / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

    logsumexp_val = c_max + np.log(sum_exp)
    F = logsumexp_val / (s * s)

    # Gradient: dF/dx_i = 2*corr_i/(tau*s^2) - 2*F*a_i/s
    size = next_pow2(2 * n - 1)
    a_flip = a[::-1]
    fa_flip = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_w_af = irfft(fw * fa_flip, size)
    corr = conv_w_af[n - 1:n - 1 + n].copy()

    grad = 2.0 * corr / (tau * s * s) - 2.0 * F * a / s
    return F, grad


def gradient_optimize(a_init, time_budget=60.0, verbose=False):
    """L-BFGS-B on smoothed log-sum-exp objective with log-parametrization."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = np.log(a) - np.log(a).max()

    taus = [20.0 / n, 10.0 / n, 5.0 / n, 2.0 / n, 1.0 / n,
            0.5 / n, 0.25 / n, 0.1 / n, 0.05 / n]

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break

        def fun(xv):
            xv = xv - np.max(xv)
            val, _ = smoothed_obj_and_grad(xv, tau)
            return val

        def fgrad(xv):
            xv = xv - np.max(xv)
            _, g = smoothed_obj_and_grad(xv, tau)
            return g

        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=fgrad,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12,
                                    'maxls': 50})
            if res.fun < fun(x) + 1e-15:
                x_opt = res.x - res.x.max()
                a_new = np.exp(x_opt)
                a_new = a_new / np.sum(a_new)

                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = x_opt
                    if verbose:
                        print(f"    grad tau={tau:.4f}: obj={best_obj:.10f}")
        except Exception:
            pass

    return best_a

    t = 1.0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov momentum
        alpha = (t - 1) / (t + 2) if t > 1 else 0.0
        y = a + alpha * (a - a_prev)
        y = project_simplex(y)

        # Compute subgradient at y
        conv_y = compute_conv(y)
        top_k = np.argsort(conv_y)[-K:]
        g = np.zeros(n)
        for k in top_k:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            g[lo:hi + 1] += y[k - hi:k - lo + 1][::-1]
        g /= K

        # Find vertex: s = e_{argmin g}
        j_star = int(np.argmin(g))

        # Line search: minimize f((1-tau)*y + tau*e_{j_star})
        e_js = np.zeros(n)
        e_js[j_star] = 1.0

        best_tau = 0.0
        best_f = float(np.max(compute_conv(y)))

        # Try a few tau values
        for tau_try in [0.5, 0.4, 0.3, 0.2, 0.1, 0.7, 0.8, 0.9, 0.05]:
            y_try = (1.0 - tau_try) * y + tau_try * e_js
            conv_try = compute_conv(y_try)
            f_try = float(np.max(conv_try))
            if f_try < best_f:
                best_f = f_try
                best_tau = tau_try

        # Nesterov step size
        t_new_step = 2.0 / (t + 3.0)
        a_new = (1.0 - t_new_step) * y + t_new_step * e_js
        a_new = project_simplex(a_new)

        # Also try the line search result
        a_ls = (1.0 - best_tau) * y + best_tau * e_js
        a_ls = project_simplex(a_ls)

        # Pick the better one
        obj_nest = compute_obj(a_new)
        obj_ls = compute_obj(a_ls)

        if obj_ls < obj_nest:
            a_new = a_ls
            obj_new = obj_ls
        else:
            obj_new = obj_nest

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and it % 1000 == 0:
                print(f"    FW it={it}: obj={best_obj:.10f}")
        a_prev = a
        a = a_new
        t += 1.0

    return best_a


def momentum_subgradient(a_init, time_budget=60.0, K=5, verbose=False):
    """
    Subgradient method with Nesterov momentum on the simplex.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    beta = 0.9
    step = 0.05
    patience = 0

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        if it > 0:
            m = (np.sqrt(1 + 4 * it * it) - 1) / 2
            y = a + ((m - 1) / m) * v
            y = project_simplex(y)
        else:
            y = a.copy()

        # Subgradient at y
        conv_y = compute_conv(y)
        top_k = np.argsort(conv_y)[-K:]
        g = np.zeros(n)
        for k in top_k:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            g[lo:hi + 1] += y[k - hi:k - lo + 1][::-1]
        g /= K

        # Update
        step_new = 2.0 / (2.0 + it + 1)
        v_new = y - a
        a_new = y - step_new * g
        a_new = project_simplex(a_new)
        v = v_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 200:
                # Perturb
                rng = np.random.default_rng(it * 7 + 3)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = project_simplex(a)
                v = np.zeros(n)
                patience = 0

        a = a_new

    return best_a


def build_conv_matrix_sparse_info(a):
    """Return the convolution matrix A (2n-1 x n) where A@g = conv(a,g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_alternate_one_step(a):
    """One step of LP alternating. Returns improved a or None."""
    n = len(a)
    a_norm = a / np.sum(a)

    try:
        A = build_conv_matrix_sparse_info(a_norm)
        # LP: min M s.t. A@g <= M*1, sum(g)=1, g>=0
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 10})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                g_val = g_val / s_g
                M_opt = float(res.x[n])
                return g_val, M_opt
    except Exception:
        pass
    return None, None


def lp_optimize(a_init, time_budget=60.0, verbose=False):
    """LP alternating optimization."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        result = lp_alternate_one_step(a)
        g_lp, M_lp = result
        if g_lp is None:
            break

        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.8, 0.6, 0.4, 0.2, 0.9]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                stall = 0
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 5:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

    return best_a


def make_init(n, mode, rng):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / (np.sum(a) + 1e-15)
    elif mode == 2:
        # Exponential decay (asymmetric)
        alpha = rng.uniform(0.5, 2.0)
        a = np.exp(-alpha * np.arange(n) / n)
        a = a / np.sum(a)
    elif mode == 3:
        # Off-center Gaussian
        center = n * rng.uniform(0.1, 0.4)
        width = n * rng.uniform(0.1, 0.3)
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        a = a / (np.sum(a) + 1e-15)
    elif mode == 4:
        # Two bumps
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.6, 0.9)
        w = n * 0.1
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            0.5 * np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / (np.sum(a) + 1e-15)
    elif mode == 5:
        # Raised cosine
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / (np.sum(a) + 1e-15)
    elif mode == 6:
        # Triangular (asymmetric)
        peak = n * rng.uniform(0.2, 0.8)
        a = 1.0 - np.abs(np.arange(n) - peak) / max(n * 0.6, 1)
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n)
        a = a / np.sum(a)
    elif mode == 7:
        # Smooth random with asymmetry
        a = rng.uniform(0, 1, n)
        kernel_size = max(3, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        # Add asymmetry
        a = a * np.exp(-0.5 * np.arange(n) / n)
        a = a / (np.sum(a) + 1e-15)
    elif mode == 8:
        # Sine modulated
        freq = rng.uniform(2, 8)
        a = 0.5 + 0.3 * np.sin(freq * np.arange(n) / n * 2 * np.pi)
        a = np.maximum(a, 0)
        a = a / (np.sum(a) + 1e-15)
    else:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(budget_s=1500):
    """
    Main search function. Combines multiple optimization strategies:
    L-BFGS-B (smoothed), Frank-Wolfe, momentum subgradient, LP alternating,
    with diverse asymmetric initializations and n-exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)

    # === Phase 1: Quick gradient optimization on parent ===
    print(f"\n=== Phase 1: Gradient on parent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 120.0)
        a_grad = gradient_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_grad)
        if obj < best_score:
            best_score = obj
            best_seq = a_grad.copy()
            print(f"  After gradient: obj={best_score:.10f}")
        else:
            print(f"  Gradient: obj={obj:.10f} (no improve)")

    # === Phase 2: Frank-Wolfe on parent ===
    print(f"\n=== Phase 2: Frank-Wolfe (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 150.0)
        a_fw = frank_wolfe_optimize(best_seq, time_budget=alloc, K=3, verbose=True)
        obj = compute_obj(a_fw)
        if obj < best_score:
            best_score = obj
            best_seq = a_fw.copy()
            print(f"  After FW: obj={best_score:.10f}")
        else:
            print(f"  FW: obj={obj:.10f} (no improve)")

    # === Phase 3: Diverse initializations + optimization ===
    print(f"\n=== Phase 3: Diverse Initializations ===")
    n_best = len(best_seq)
    inits_to_try = [
        (n_best, 2), (n_best, 3), (n_best, 4), (n_best, 7),
        (n_best, 0), (n_best, 8), (n_best, 5), (n_best, 6),
    ]
    # Also try different n values
    for n_try in [n_best - 100, n_best + 100, 600, 1000]:
        if 300 <= n_try <= 1500 and n_try != n_best:
            inits_to_try.append((n_try, 2))
            inits_to_try.append((n_try, 4))

    for idx, (n_try, mode) in enumerate(inits_to_try):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_init = make_init(n_try, mode, rng)

        # Quick gradient
        grad_alloc = min(remaining * 0.05, 40.0)
        a_opt = gradient_optimize(a_init, time_budget=grad_alloc)
        obj = compute_obj(a_opt)

        # If promising, try Frank-Wolfe
        if obj < best_score + 0.05:
            remaining = deadline - time.monotonic()
            if remaining > 60:
                fw_alloc = min(remaining * 0.04, 50.0)
                a_fw = frank_wolfe_optimize(a_opt, time_budget=fw_alloc, K=3)
                obj_fw = compute_obj(a_fw)
                if obj_fw < obj:
                    a_opt = a_fw
                    obj = obj_fw

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try

        else:
            print(f"  Init {idx} (n={n_try}, mode={mode}): obj={obj:.10f}")

    # === Phase 4: Momentum subgradient polish ===
    print(f"\n=== Phase 4: Momentum Subgradient (n={len(best_seq)}) ===")

        alloc = min(remaining * 0.15, 120.0)
        a_ms = momentum_subgradient(best_seq, time_budget=alloc, K=5, verbose=True)

            best_seq = a_ms.copy()
            print(f"  After momentum: obj={best_score:.10f}")

    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Gradient round 2
    if remaining > 60:
        alloc = min(remaining * 0.15, 100.0)
        a_g2 = gradient_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_g2)
        if obj < best_score:
            best_score = obj
            best_seq = a_g2.copy()
            n_best = len(best_seq)

        a_fw2 = frank_wolfe_optimize(best_seq, time_budget=alloc, K=5, verbose=True)
        obj = compute_obj(a_fw2)
        if obj < best_score:
            best_score = obj
            best_seq = a_fw2.copy()
            print(f"  FW2: obj={best_score:.10f}")

    # Momentum round 2
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.12, 80.0)
        a_ms2 = momentum_subgradient(best_seq, time_budget=alloc, K=3)
        obj = compute_obj(a_ms2)
        if obj < best_score:
            best_score = obj
            best_seq = a_ms2.copy()
            print(f"  Momentum2: obj={best_score:.10f}")

    # === Phase 7: n-exploration ===
    print(f"\n=== Phase 7: n-Exploration ===")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        n_best - 50, n_best - 100, n_best,
        n_best + 50, n_best + 100, n_best + 150,
        700, 850, 950, 1100

    n_candidates = [n for n in n_candidates if 400 <= n <= 1400]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)
        grad_alloc = min(remaining * 0.06, 30.0)
        a_opt = gradient_optimize(a_try, time_budget=grad_alloc)

            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj:.10f}")

    # === Phase 8: Final intensification ===
    print(f"\n=== Phase 8: Final Polish ===")
    n_final = len(best_seq)
    remaining = deadline - time.monotonic()

    # Multiple short rounds of different methods
    for round_i in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:

        alloc = min(remaining * 0.1, 45.0)

            a_r = gradient_optimize(best_seq, time_budget=alloc)
        elif round_i == 1:
            a_r = frank_wolfe_optimize(best_seq, time_budget=alloc, K=2)
        elif round_i == 2:
            a_r = momentum_subgradient(best_seq, time_budget=alloc, K=3)
        else:
            a_r = lp_optimize(best_seq, time_budget=alloc)

        obj = compute_obj(a_r)
        if obj < best_score:
            best_score = obj
            best_seq = a_r.copy()
            print(f"  Round {round_i} (n={len(a_r)}): obj={best_score:.10f} **")
        else:
            print(f"  Round {round_i}: obj={obj:.10f}")

    # === Phase 9: Very final gradient with lowest temperature ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = gradient_optimize(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final grad: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
