This is a non-convex QCQP, but if we fix the active set, we can use KKT conditions.

At the KKT point for the active set S = {k1, k2, ...} (all peak bins have the same value t):
- Gradient of L = sum_{k in S} lambda_k * grad (p*p)[k] + mu * 1 + nu = 0 (where nu >= 0 is the multiplier for p >= 0)
- (p*p)[k_i] = t for k_i in S
- (p*p)[k] <= t for k not in S
- sum(p) = 1, p >= 0

This gives us a system of equations that we can solve given the active set S.

**Algorithm: Active Set Method**
1. Start with some initial p.
2. Find the peak bin(s) S.
3. Solve the KKT system for the given active set.
4. Check if the solution is feasible (all other bins <= t).
5. If not, update the active set (add the most violated bin) and go to 3.
6. If yes, we have the optimum for this active set.

Let me now think about the "direct peak shaving" algorithm more carefully.

**Direct Peak Shaving Algorithm:**
1. Compute conv = p*p. Find the peak bin k*.
2. The peak value is t = conv[k*] = sum_j p[j]*p[k*-j].
3. For each j, the "responsibility" is r[j] = p[j]*p[k*-j] (contribution to the peak).
4. We want to reduce t. The most effective way is to reduce the products p[j]*p[k*-j].
5. For each j, if p[j] > p[k*-j], reducing p[j] by delta reduces the peak by delta*p[k*-j].
6. But reducing p[j] also affects other convolution bins.
7. A good heuristic: move mass from position j to position j' where p[j']*p[k*-j'] is small (so the move doesn't increase the peak much).
8. Specifically: move delta from j to j' where j' is chosen to minimize the increase in other peaks.

A simpler version:
1. Find peak bin k*.
2. For each pair (j, k*-j) that contributes to the peak, compute the "cost" of moving mass from j to some other position.
3. Move a small amount of mass from the "most responsible" position to a "safe" position.
4. Repeat.

This is essentially coordinate descent but with a smarter choice of which position to move to.

Actually, the simplest version is:

        try:
            result = minimize(obj_func, theta, jac=grad_func, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-7, 'maxfun': 800})
            if result.x is not None:
                a_new = softmax_from_theta(result.x)
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        theta = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_restricted(a, k_constraints=300):
    """Solve restricted LP: min M s.t. top-k conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)

    conv_current = compute_conv(a)
    top_k = np.argsort(conv_current)[::-1][:min(k_constraints, 2 * n - 1)]

    A_ub = np.zeros((len(top_k), n + 1))
    A_ub[:, :n] = A_full[top_k, :]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(top_k))

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def solve_lp_full(a):
    """Solve full LP with all constraints."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_alternating(a, time_budget=60.0, use_full=False, k_constraints=300):
    """LP alternating optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        if use_full and n <= 1500:
            g_lp = solve_lp_full(a)
        else:
            g_lp = solve_lp_restricted(a, k_constraints)

        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
        else:
            if it % 5 == 0:
                rng = np.random.default_rng(it * 104729 + n)
                a_pert = best_a * np.exp(rng.normal(0, 0.03, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    if compute_obj(a_pert) < best_obj:
                        best_obj = compute_obj(a_pert)
                        best_a = a_pert.copy()

    return best_a


def peak_shaving(a, time_budget=60.0):
    """Direct peak shaving: move mass from peak contributors to safe positions."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        t_star = conv[k_star]

        # Compute responsibility for each position
        r = np.zeros(n)
        for j in range(n):
            kp = k_star - j
            if 0 <= kp < n:
                r[j] = a[j] * a[kp]

        # Source: position with highest responsibility
        j_src = int(np.argmax(r))
        if r[j_src] < 1e-15:
            break

        # Destination: position with low responsibility AND low "secondary" impact
        # Secondary impact: moving to j' increases conv[k'] for k' near j' + k_star - j_src
        # Simplified: just pick position with lowest r
        candidates = np.where(r < 0.3 * r[j_src])[0]
        if len(candidates) == 0:
            candidates = np.where(r == r.min())[0]
        j_dst = candidates[rng.integers(0, len(candidates))]

        # Multiple delta sizes
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.01]:
            delta = frac * a[j_src]
            if delta <= 0:
                continue
            a[j_src] -= delta
            a[j_dst] += delta
            a = np.maximum(a, 0)
            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                improved = True
                break
            else:
                a[j_dst] -= delta
                a[j_src] += delta

        if not improved and it % 100 == 0:
            # Random perturbation to escape
            pert = rng.normal(0, 0.02, n)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

    return best_a


def coordinate_descent_block(a, time_budget=30.0, block_size=10):
    """Block coordinate descent: optimize blocks of variables at a time."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    n_blocks = max(1, n // block_size)

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try random block moves
        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            kp_i = k_star - i
            kp_j = k_star - j
            # Only move if it reduces the peak
            if 0 <= kp_i < n and 0 <= kp_j < n:
                if a[i] * a[kp_i] > a[j] * a[kp_j]:
                    for frac in [0.01, 0.03, 0.05]:
                        delta = frac * a[i]
                        a[i] -= delta
                        a[j] += delta
                        a = np.maximum(a, 0)
                        obj_new = compute_obj(a)
                        if obj_new < best_obj - 1e-14:
                            best_obj = obj_new
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[j] -= delta
                            a[i] += delta
                            break
            if improved:
                break

        if not improved and it % 50 == 0:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, rng, style=0):
    """Generate structured asymmetric initialization."""
    if style == 0:
        # Linear ramp with noise
        slope = rng.uniform(1.5, 3.0)
        a = np.linspace(1.0, slope, n)
        a *= np.exp(rng.normal(0, 0.1, n))
    elif style == 1:
        # Exponential
        rate = rng.uniform(1.0, 4.0)
        a = np.exp(np.linspace(0, rate, n))
        a *= np.exp(rng.normal(0, 0.05, n))
    elif style == 2:
        # Cosine window (asymmetric) + ramp
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * t)) + rng.uniform(0.3, 1.0) * t
        a *= np.exp(rng.normal(0, 0.05, n))
    elif style == 3:
        # Two-bump asymmetric
        center1 = rng.uniform(0.2, 0.4)
        center2 = rng.uniform(0.6, 0.8)
        width1 = rng.uniform(0.05, 0.15)
        width2 = rng.uniform(0.05, 0.15)
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - center1) / width1) ** 2) + 0.7 * np.exp(-((t - center2) / width2) ** 2)
    else:
        # Random smooth (low-pass filtered random)
        a = rng.random(n)
        kernel = np.exp(-np.linspace(-5, 5, min(n, 21)) ** 2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a += 0.1 * np.linspace(1, 2, n)

    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Wide n exploration with structured inits + quick optimization
    Phase 2: Intensive optimization on best (Nesterov + L-BFGS-B + LP)
    Phase 3: Peak shaving and coordinate descent polish
    Phase 4: Final refinement with full LP and intensive local search
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    best_n = 800
    rng = np.random.default_rng(seed)

    # Phase 0: Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                best_n = len(best_seq)
                print(f"Parent (n={best_n}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(800, rng, style=0)
        best_score = compute_obj(best_seq)
        best_n = len(best_seq)
        print(f"Default init (n={best_n}): obj={best_score:.8f}")

    # Phase 1: Wide n exploration (250s)
    print(f"\n=== Phase 1: Wide n Exploration ===")
    phase1_budget = min(250.0, (deadline - time.monotonic()) * 0.25)
    n_candidates = [500, 600, 700, 800, 900, 1000, 1200, 1500, 2000]
    # Include parent n
    if best_n not in n_candidates:
        n_candidates.append(best_n)
    n_candidates = sorted(set([n for n in n_candidates if 300 <= n <= 3000]))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < phase1_budget / max(1, len(n_candidates)):
            break
        if remaining < 20:
            break

        per_n_budget = min(phase1_budget / max(1, len(n_candidates)), remaining * 0.4, 30.0)

        if n_try == best_n and best_seq is not None:
            a_init = best_seq.copy()
        else:
            # Try 2 different initializations
            a_init = structured_init(n_try, rng, style=rng.integers(0, 5))

        # Quick Nesterov optimization
        a_opt, obj_opt = nesterov_smoothed(a_init, tau_start=20.0, tau_end=200.0,
                                            time_budget=per_n_budget, max_steps=2000)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            best_n = n_try
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 2: Intensive optimization on best (500s)
    print(f"\n=== Phase 2: Intensive Optimization (n={best_n}) ===")
    phase2_budget = min(500.0, (deadline - time.monotonic()) * 0.35)
    n_cur = len(best_seq)

    # 2a: L-BFGS-B with annealing
    alloc = min(phase2_budget * 0.3, 120.0)
    if deadline - time.monotonic() > alloc + 10:
        a_opt, obj_opt = lbfgs_smoothed(best_seq, tau_start=10.0, tau_end=500.0,
                                         time_budget=alloc, max_iter=200)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  L-BFGS-B: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS-B: obj={obj_opt:.8f}")

    # 2b: Nesterov with different tau range
    alloc = min(phase2_budget * 0.25, 100.0)
    if deadline - time.monotonic() > alloc + 10:
        a_opt, obj_opt = nesterov_smoothed(best_seq, tau_start=50.0, tau_end=500.0,
                                            time_budget=alloc, max_steps=3000)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_opt:.8f}")

    # 2c: LP alternating (restricted)
    alloc = min(phase2_budget * 0.25, 100.0)
    if deadline - time.monotonic() > alloc + 10:
        a_lp = lp_alternating(best_seq, time_budget=alloc, use_full=False, k_constraints=300)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP restricted: obj={best_score:.8f} **")
        else:
            print(f"  LP restricted: obj={obj_lp:.8f}")

    # 2d: Peak shaving
    alloc = min(phase2_budget * 0.2, 80.0)
    if deadline - time.monotonic() > alloc + 10:
        a_ps = peak_shaving(best_seq, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Peak shaving: obj={best_score:.8f} **")
        else:
            print(f"  Peak shaving: obj={obj_ps:.8f}")

    # Phase 3: Neighbor n exploration + refinement (300s)
    print(f"\n=== Phase 3: Neighbor n + Refinement (n={best_n}) ===")
    phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.3)
    n_cur = len(best_seq)

    n_neighbors = sorted(set([
        n_cur - 100, n_cur - 50, n_cur, n_cur + 50, n_cur + 100, n_cur + 200,
    ]))
    n_neighbors = [n for n in n_neighbors if 300 <= n <= 3000]

    for n_try in n_neighbors:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization
        per_budget = min(phase3_budget / len(n_neighbors), remaining * 0.3, 40.0)
        a_opt, obj_opt = nesterov_smoothed(a_try, tau_start=30.0, tau_end=400.0,
                                            time_budget=per_budget, max_steps=2000)

        # Follow with LP if good
        if obj_opt < best_score + 0.01 and per_budget > 10:
            a_lp = lp_alternating(a_opt, time_budget=min(per_budget * 0.5, 20.0),
                                  use_full=False, k_constraints=200)
            obj_lp = compute_obj(a_lp)
            if obj_lp < obj_opt:
                obj_opt = obj_lp
                a_opt = a_lp

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = n_try
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # 4a: Full LP (if n is small enough)
    if n_final <= 1500 and remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, use_full=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Full LP: obj={best_score:.8f} **")
        else:
            print(f"  Full LP: obj={obj_lp:.8f}")

    # 4b: Intensive peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 50.0)
        a_ps = peak_shaving(best_seq, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Peak shave: obj={best_score:.8f} **")
        else:
            print(f"  Peak shave: obj={obj_ps:.8f}")

    # 4c: L-BFGS-B final (high tau = close to exact)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 40.0)
        a_opt, obj_opt = lbfgs_smoothed(best_seq, tau_start=100.0, tau_end=1000.0,
                                         time_budget=alloc, max_iter=150)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Final L-BFGS: obj={obj_opt:.8f}")

    # 4d: Coordinate descent block
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 30.0)
        a_cd = coordinate_descent_block(best_seq, time_budget=alloc, block_size=5)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord descent: obj={best_score:.8f} **")
        else:
            print(f"  Coord descent: obj={obj_cd:.8f}")

    # 4e: One more Nesterov pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_opt, obj_opt = nesterov_smoothed(best_seq, tau_start=100.0, tau_end=800.0,
                                            time_budget=alloc, max_steps=2000)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Final Nesterov: obj={obj_opt:.8f}")

    # 4f: Random restarts near best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_restarts = min(4, int(remaining / 20))
        best_n = len(best_seq)
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # Small perturbation
            a_r = best_seq * np.exp(rng.normal(0, 0.05, len(best_seq)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            alloc = min(remaining * 0.3, 20.0)
            a_opt, obj_opt = nesterov_smoothed(a_r, tau_start=50.0, tau_end=500.0,
                                                time_budget=alloc, max_steps=1000)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
