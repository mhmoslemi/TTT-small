Hybrid QP + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))

        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def subgradient_topK(a, K=5):
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n)
    # Exponential weighting
    weights = np.exp((peak_vals - max_val) / (max_val * 0.1))
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
    return g

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def two_point_refine(a, time_budget=30):
    """Efficient two-point move refinement."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    # Precompute contributions to peak
    # conv[k_star] = sum_i a[i] * a[k_star - i]
    # For each position i, its "partner" is j = k_star - i
    
    it = 0
    while time.monotonic() - start < time_budget:
        improved = False
        k_star = int(np.argmax(conv))
        
        # Get all (i, j) pairs contributing to the peak, sorted by contribution
        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        pairs.sort(reverse=True)
        
        # Non-contributors
        contrib_set = set()
        for _, i, j in pairs:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_set]
        
        for contrib, i, j in pairs[:10]:
            if time.monotonic() - start > time_budget:
                break
            for target in non_contrib[:3]:
                if target == i or target == j:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            break
        it += 1
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Hybrid QP + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses reduced-dimension QP direction finding (targeting flat autocorrelation),
LP-based peak minimization, accelerated projected gradient, two-point moves,
and multi-n exploration with simulated annealing restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def build_reduced_conv_matrix(a, top_k):
    """Build reduced matrix using only top-k convolution bins."""
    n = len(a)
    conv_a = compute_conv(a)
    top_indices = np.argsort(conv_a)[::-1][:top_k]

    A = np.zeros((top_k, n))
    for idx, k in enumerate(top_indices):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            A[idx, i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return A, top_indices, conv_a


def subgradient_topK(a, K=5, temperature=0.05):
    """Compute subgradient using top-K peaks with softmax weighting."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n)
    weights = np.exp((peak_vals - max_val) / (max_val * temperature))
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_qp_reduced(a, n_top=150, target_type='flat'):
    """Solve reduced QP: min ||A_reduced @ g - target||^2 s.t. g>=0, sum(g)=1."""
    n = len(a)
    a = a / a.sum()
    A, top_indices, conv_a = build_reduced_conv_matrix(a, n_top)

    if target_type == 'flat':
        tgt = np.ones(n_top) / (2 * n - 1)
    elif target_type == 'capped':
        M = conv_a.max()
        tgt = conv_a[top_indices].copy()
        tgt = np.minimum(tgt, M * 0.75)
        tgt = tgt + M * 0.005
    elif target_type == 'current':
        tgt = conv_a[top_indices].copy()
    else:
        tgt = np.ones(n_top) / (2 * n - 1)

    g = cp.Variable(n, nonneg=True)
    cost = cp.sum_squares(A @ g - tgt)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(cost), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(g_val.sum())
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_qp_full(a, target_type='flat'):
    """Solve full QP: min ||A @ g - target||^2 s.t. g>=0, sum(g)=1."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    if target_type == 'flat':

    elif target_type == 'capped':
        conv_a = compute_conv(a)
        M = conv_a.max()
        tgt = np.minimum(conv_a, M * 0.7)
        tgt = tgt + M * 0.005
    else:
        tgt = np.ones(2 * n - 1) / (2 * n - 1)

    g = cp.Variable(n, nonneg=True)
    cost = cp.sum_squares(A @ g - tgt)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(cost), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(g_val.sum())
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(g_val.sum())
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def do_line_search(a_base, direction, best_obj, n_ts=8):
    """Line search: find best t in (0,1) for a_base + t*(direction - a_base)."""
    best_t = 0.5
    best_t_obj = best_obj
    best_a = a_base.copy()

    ts = np.linspace(0.1, 1.0, n_ts)
    for t in ts:
        candidate = (1 - t) * a_base + t * direction
        candidate = np.maximum(candidate, 0)
        cs = float(candidate.sum())
        if cs < 1e-10:
            continue
        candidate = candidate / cs
        obj_try = compute_obj(candidate)
        if obj_try < best_t_obj:
            best_t_obj = obj_try
            best_t = t
            best_a = candidate.copy()

    return best_a, best_t_obj, best_t


def qp_alternating(a, time_budget=200, verbose=False, use_reduced=True, n_top=150):
    """QP-based alternating optimization with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0
    last_t = 0.5
    target_cycle = ['flat', 'capped', 'flat', 'flat', 'capped']
    target_idx = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Choose target type cyclically
        target_type = target_cycle[target_idx % len(target_cycle)]
        target_idx += 1

        # Solve for direction
        if use_reduced:
            g = solve_qp_reduced(a, n_top=n_top, target_type=target_type)
        else:
            g = solve_qp_full(a, target_type=target_type)

        if g is None:
            g, _ = solve_lp_direction(a)

        if g is None:
            break

        # Adaptive line search (centered around last best t)
        if 0.2 < last_t < 0.8:
            ts_custom = sorted(set(
                [last_t * 0.5, last_t * 0.75, last_t, last_t * 1.25, last_t * 1.5] +
                [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]
            ))
            ts_custom = [t for t in ts_custom if 0.05 <= t <= 1.0][:10]
            best_a_new, best_obj_new, best_t = do_line_search(a, g, best_obj, n_ts=len(ts_custom))
            # Override with custom ts
            for t in ts_custom:
                candidate = (1 - t) * a + t * g
                candidate = np.maximum(candidate, 0)
                cs = float(candidate.sum())
                if cs < 1e-10:
                    continue
                candidate = candidate / cs
                obj_try = compute_obj(candidate)
                if obj_try < best_obj_new:
                    best_obj_new = obj_try
                    best_t = t
                    best_a_new = candidate.copy()
        else:
            best_a_new, best_obj_new, best_t = do_line_search(a, g, best_obj, n_ts=8)

        if best_obj_new < best_obj - 1e-10:
            best_obj = best_obj_new
            best_a = best_a_new
            a = best_a_new
            last_t = best_t
            stall = 0
            it += 1
            if verbose and (it % 10 == 0 or it < 5):
                elapsed = time.monotonic() - start
                print(f"    QP alt it{it}: obj={best_obj:.8f}, t={best_t:.2f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 20:
                # Perturb and retry
                rng = np.random.default_rng(it * 7919 + n + int(time.monotonic() * 1000))
                scale = min(0.3, 0.05 + 0.01 * (stall // 20))
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert
                # Try one more iteration from perturbed
                g2 = solve_qp_reduced(a, n_top=n_top, target_type='flat')
                if g2 is not None:
                    a_try, obj_try, t_try = do_line_search(a, g2, best_obj, n_ts=6)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try
                        a = a_try
                        stall = 0
                        it += 1
                if stall > 40:
                    break

    return best_a


def projected_gradient_optimize(a, time_budget=100, verbose=False):
    """FISTA-style accelerated projected gradient with top-K subgradient."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    tk = 1.0
    lr = 10.0 / n
    K = 5

    it = 0
    while time.monotonic() - start < time_budget:
        g = subgradient_topK(y, K=K, temperature=0.03)
        x_new = y - lr * g
        x_new = project_simplex(x_new)

        # FISTA momentum
        tk_new = (1 + np.sqrt(1 + 4 * tk * tk)) / 2
        y_new = x_new + ((tk - 1) / tk_new) * (x_new - y)

        y = y_new
        tk = tk_new

        obj = compute_obj(x_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = x_new.copy()
            a = x_new.copy()
        else:
            a = x_new.copy()
            lr = max(1.0 / n, lr * 0.97)

        it += 1
        if verbose and it % 200 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def two_point_refine(a, time_budget=30):
    """Targeted two-point moves to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    it = 0
    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))

        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        pairs.sort(reverse=True)

        contrib_set = set()
        for _, i, j in pairs:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_set]

        if not non_contrib:
            break

        improved = False
        for contrib, i, j in pairs[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            for target in non_contrib[:5]:
                if target == i or target == j:
                    continue
                for frac in [0.03, 0.07, 0.15, 0.3]:
                    delta = frac * a[i]
                    if a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try moving from j instead
            for contrib, i, j in pairs[:10]:
                if time.monotonic() - start > time_budget:
                    break
                if j == i:
                    continue
                for target in non_contrib[:3]:
                    if target == i or target == j:
                        continue
                    for frac in [0.05, 0.1, 0.2]:
                        delta = frac * a[j]
                        if a[j] - delta < 0:
                            continue
                        a[j] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                            improved = True
                            break
                        else:
                            a[j] += delta
                            a[target] -= delta
                            break
                if improved:
                    break

        if not improved:
            break
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01) + 0.1)
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x**3 + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Hybrid QP + LP + PG optimization for minimizing autocorrelation peak.
    Strategy: reduced QP alternating (fast) -> full QP/LP polish -> multi-n sweep -> final refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n = 800
        best_seq = np.ones(n) / n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Rapid QP alternating (reduced) ===
    print(f"\n=== Phase 1: Reduced QP Alternating (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 400.0)

    if p1_budget > 30:
        a_opt = qp_alternating(best_seq, time_budget=p1_budget, verbose=True,
                               use_reduced=True, n_top=120)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 result: obj={obj_opt:.8f} (no improvement)")

    # === Phase 2: Full QP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.20, 250.0)
        print(f"\n=== Phase 2: Full QP Polish (budget={p2_budget:.0f}s) ===")
        a_opt = qp_alternating(best_seq, time_budget=p2_budget, verbose=True,
                               use_reduced=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 2 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 result: obj={obj_opt:.8f}")

    # === Phase 3: FISTA polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.10, 100.0)
        print(f"\n=== Phase 3: FISTA (budget={p3_budget:.0f}s) ===")
        a_opt = projected_gradient_optimize(best_seq, time_budget=p3_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 3 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 result: obj={obj_opt:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    n_options = []
    for dn in [-150, -100, -50, 50, 100, 150, 200, 300]:
        n_try = n_best + dn
        if 250 <= n_try <= 2000:
            n_options.append(n_try)
    for n_fixed in [400, 500, 600, 700, 900, 1000, 1200]:
        if 250 <= n_fixed <= 2000 and n_fixed not in n_options:
            n_options.append(n_fixed)

    n_options = sorted(set(n_options))
    n_options = [n for n in n_options if n != n_best]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 80.0)

        # Quick reduced QP
        a_opt = qp_alternating(a_try, time_budget=alloc, verbose=False,
                               use_reduced=True, n_top=100)
        obj_opt = compute_obj(a_opt)

        # If promising, do more
        if obj_opt < best_score + 0.005 and remaining > 30:
            remaining = deadline - time.monotonic()
            extra_alloc = min(remaining * 0.05, 30.0)
            if extra_alloc > 10:
                a_opt2 = qp_alternating(a_opt, time_budget=extra_alloc, verbose=False,
                                        use_reduced=True, n_top=100)
                obj_opt2 = compute_obj(a_opt2)
                if obj_opt2 < obj_opt:
                    a_opt = a_opt2
                    obj_opt = obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Intensive final optimization ===
    print(f"\n=== Phase 5: Final Intensive Optimization (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p5_budget = min(remaining * 0.35, 300.0)
        # Alternate between reduced QP and LP
        a_opt = qp_alternating(best_seq, time_budget=p5_budget * 0.6, verbose=True,
                               use_reduced=True, n_top=100)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

        remaining = deadline - time.monotonic()
        if remaining > 30:
            p5b_budget = min(remaining * 0.3, 100.0)
            a_opt2 = solve_lp_direction(best_seq)
            if a_opt2[0] is not None:
                g_lp = a_opt2[0]
                a_try, obj_try, _ = do_line_search(best_seq, g_lp, best_objective if False else best_score, n_ts=8)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  LP direction: obj={best_score:.8f} **")

    # === Phase 6: Two-point refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        tp_budget = min(remaining * 0.2, 40.0)
        print(f"\n=== Phase 6: Two-Point Refinement ===")
        a_opt = two_point_refine(best_seq, time_budget=tp_budget)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Two-point: obj={best_score:.8f} **")
        else:
            print(f"  Two-point: obj={obj_opt:.8f}")

    # === Phase 7: Final quick QP ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        fin_budget = min(remaining * 0.6, 60.0)
        a_opt = qp_alternating(best_seq, time_budget=fin_budget, verbose=False,
                               use_reduced=True, n_top=80)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final QP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
