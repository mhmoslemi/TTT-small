import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def conv_matrix_times_vec(a, g):
    """Compute conv(a, g) efficiently."""
    return np.convolve(a, g, mode='full')

def solve_lp_for_g(a, M_bound=None):
    """
    Solve: max sum(g) s.t. conv(a,g) <= M_bound, g >= 0
    Or if M_bound is None: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0
    Returns g (normalized to sum=1) and M.
    """
    n = len(a)
    a = a / np.sum(a)
    
    if M_bound is not None:
        # Maximize sum(g) subject to A@g <= M_bound, g >= 0
        # Variables: g[0], ..., g[n-1]
        c = -np.ones(n)  # minimize -sum(g)
        A_ub = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            j_start = k - i_end
            j_end = k - i_start
            A_ub[k, i_start:i_end+1] = a[j_start:j_end+1]
        b_ub = np.full(2 * n - 1, M_bound)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, M_bound
        return None, None
    else:
        # Minimize M s.t. A@g <= M, sum(g)=1, g>=0
        # Variables: g[0], ..., g[n-1], M
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        A_ub = np.zeros((2 * n - 1, n + 1))
        for k in range(2 * n - 1):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            j_start = k - i_end
            j_end = k - i_start
            A_ub[k, i_start:i_end+1] = a[j_start:j_end+1]
            A_ub[k, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            M = float(res.x[n])

def lp_alternating_fast(a, time_budget=100.0, verbose=False, max_iters=200):
    """
    Fast LP alternating optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g, M_lp = solve_lp_for_g(a, M_bound=None)
        if g is None:
            break
        
        # Direct replacement or blend
        # Try full replacement first
        obj_new = compute_obj(g)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = g.copy()
            a = g.copy()
        else:
            # Try blending
            for t in [0.3, 0.5, 0.7]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
        
        if verbose and it % 10 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, elapsed={time.monotonic()-start:.1f}s")

def compute_conv(a):

    n = len(a)

        return np.array([], dtype=np.float64)

    step = 0.1
    it = 0

    while time.monotonic() - start < time_budget:
        g, conv, top_idx = subgradient_topK(a, K=3)
        g = g / max(np.linalg.norm(g), 1e-15)

        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = g - prev_grad
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and sy > 1e-15:
                step_bb = ss / sy
                step = min(max(step * 0.5, 1e-6), step_bb, 0.5)

        prev_a = a.copy()
        prev_grad = g.copy()

        a_new = a - step * g
        # Project onto simplex
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        idx = np.arange(1, n + 1)
        cond = u * idx > (cssv - 1.0)
        rho_arr = np.where(cond)[0]
        if len(rho_arr) > 0:
            rho = rho_arr[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            step *= 0.8

        it += 1

    return best_a


def lp_alternating(a, time_budget=200.0, verbose=False, use_reduced=False, K_reduced=30):
    """LP alternating optimization with adaptive strategy."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_progress = 0

    while time.monotonic() - start < time_budget:
        # Choose LP type
        if use_reduced and it > 20:
            conv_cur = compute_conv(a)
            top_idx = np.argsort(conv_cur)[::-1][:K_reduced]
            g = solve_lp_reduced(a, top_idx)
        else:
            g, M_lp = solve_lp_min_m(a)

        if g is None:
            no_progress += 1
            if no_progress > 10:
                break
            continue

        # Try direct replacement first
        obj_g = compute_obj(g)
        if obj_g < best_obj - 1e-13:
            best_obj = obj_g
            best_a = g.copy()
            a = g.copy()
            no_progress = 0
        else:
            # Try blending
            improved = False
            for t in [0.5, 0.3, 0.7, 0.2]:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
            if not improved:
                no_progress += 1
                if no_progress > 30:
                    # Perturb to escape
                    rng = np.random.default_rng(it * 104729 + n)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    no_progress = 0

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def coordinate_descent_refine(a, time_budget=30.0):
    """Paired coordinate descent: move mass from peak contributors to non-contributors."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    it = 0

    while time.monotonic() - start < time_budget and it < 500:
        k_star = int(np.argmax(conv))
        improved = False

        # Find contributors to peak
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        non_contrib = set(range(n))
        for _, i, j in contribs:
            non_contrib.discard(i)
            non_contrib.discard(j)
        non_contrib_list = list(non_contrib)

        if not non_contrib_list:
            break

        for ci, (val, i, j) in enumerate(contribs[:10]):
            if time.monotonic() - start > time_budget:
                break
            if ci >= len(contribs) - 1 and i == j:
                break
            for target in non_contrib_list[:3]:
                for frac in [0.05, 0.1, 0.15]:
                    delta = frac * a[i]
                    if delta <= 0 or a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            break
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        x = np.arange(n, dtype=np.float64)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 0.01))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.05
        a = a / a.sum()
    elif mode == 6:
        a = rng.exponential(1, n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, subgradient mirror descent,
    multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_best = 800
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(parent.sum())
        if s > 0.01:
            best_seq = parent / s
            best_score = compute_obj(best_seq)
            n_best = len(best_seq)
            print(f"Parent: n={n_best}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        n_best = 800
        print(f"Default init: n={n_best}, score={best_score:.8f}")

    def check_time():
        return (deadline - time.monotonic()) > 20

    def try_improve(candidate, label=""):
        nonlocal best_seq, best_score
        obj = compute_obj(candidate)
        if obj < best_score:
            best_score = obj
            best_seq = candidate.copy()
            print(f"  ** {label}: obj={best_score:.8f} **")
        return obj

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating (n={n_best}) ===")
    if check_time():
        alloc = min(300.0, (deadline - time.monotonic()) * 0.35)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        try_improve(a_lp, "LP-alt")

    # === Phase 2: Subgradient Mirror Descent ===
    print(f"\n=== Phase 2: Subgradient MD ===")
    if check_time():
        alloc = min(150.0, (deadline - time.monotonic()) * 0.2)
        a_md = mirror_descent_optimize(best_seq, time_budget=alloc, topK=7, verbose=True)
        try_improve(a_md, "SubMD")

    # === Phase 3: Linear Subgradient ===
    print(f"\n=== Phase 3: Linear Proj Subgrad ===")
    if check_time():
        alloc = min(100.0, (deadline - time.monotonic()) * 0.1)
        a_ps = subgradient_proj_optimize(best_seq, time_budget=alloc)
        try_improve(a_ps, "LinSG")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200,
        500, 600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        if not check_time():
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)
        obj_init = compute_obj(a_try)

        # Quick LP
        alloc = min(60.0, (deadline - time.monotonic()) * 0.1)
        if alloc > 15:
            a_opt = lp_alternating(a_try, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
            if obj_opt < obj_init:
                try_improve(a_opt, f"n={n_try} LP")
                # If improved, do more MD
                if check_time():
                    alloc2 = min(40.0, (deadline - time.monotonic()) * 0.05)
                    if alloc2 > 10:
                        a_md2 = mirror_descent_optimize(a_opt, time_budget=alloc2, topK=5)
                        try_improve(a_md2, f"n={n_try} MD")
            else:
                print(f"  n={n_try}: init={obj_init:.6f}, LP={obj_opt:.6f} (no imp)")
        else:
            print(f"  n={n_try}: init={obj_init:.6f} (no time)")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    n_restarts = 5
    for r in range(n_restarts):
        if not check_time():
            break
        mode = (r * 3 + 1) % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(80.0, (deadline - time.monotonic()) * 0.15)
        if alloc < 20:
            break

        # Quick LP then MD
        half = alloc / 2
        a_lp_r = lp_alternating(a_r, time_budget=half)
        obj_lp_r = compute_obj(a_lp_r)
        if obj_lp_r < compute_obj(a_r):
            a_r = a_lp_r

        if check_time():
            alloc2 = min(40.0, (deadline - time.monotonic()) * 0.05)
            if alloc2 > 10:
                a_md_r = mirror_descent_optimize(a_r, time_budget=alloc2, topK=5)
                obj_md_r = compute_obj(a_md_r)
                if obj_md_r < obj_lp_r:
                    a_r = a_md_r

        try_improve(a_r, f"Restart{r}(m{mode})")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    if check_time():
        # LP polish
        alloc = min(100.0, (deadline - time.monotonic()) * 0.4)
        if alloc > 20:
            a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
            try_improve(a_final, "Polish-LP")

    if check_time():
        # MD polish with larger K
        alloc = min(60.0, (deadline - time.monotonic()) * 0.3)
        if alloc > 10:
            a_md_final = mirror_descent_optimize(best_seq, time_budget=alloc, topK=10)
            try_improve(a_md_final, "Polish-MD")

    if check_time():
        # Coordinate descent
        alloc = min(30.0, (deadline - time.monotonic()) * 0.3)
        if alloc > 5:
            a_cd = coordinate_descent_refine(best_seq, time_budget=alloc)
            try_improve(a_cd, "Polish-CD")

    if check_time():
        # One more LP
        alloc = min(40.0, (deadline - time.monotonic()) * 0.5)
        if alloc > 10:
            a_last = lp_alternating(best_seq, time_budget=alloc)
            try_improve(a_last, "Polish-LP2")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
