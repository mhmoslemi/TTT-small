   def next_pow2(m):
       p = 1
       while p < m:
           p <<= 1
       return p
   
   
   def compute_conv(a):
       """Full linear convolution a*a."""
       n = len(a)
       size = next_pow2(2 * n - 1)
       fa = rfft(a, size)
       return irfft(fa * fa, size)[:2 * n - 1]
   
   
   def compute_cross_corr(u, v):
       """Cross-correlation-like: (u*v)[k] = sum_j u[k-j]*v[j] for valid j."""
       n = len(u)
       m = len(v)
       size = next_pow2(n + m - 1)
       fu = rfft(u, size)
       fv = rfft(v, size)
       return irfft(fu * fv, size)[:n + m - 1]
   
   
   def compute_obj(a):
       """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
       n = len(a)
       if n == 0:
           return np.inf
       s = float(np.sum(a))
       if s < 0.01:
           return np.inf
       a_norm = a / s
       conv = compute_conv(a_norm)
       return 2.0 * n * float(np.max(conv))

       """
       Solve QP: min 0.5*||a - a_old||^2
       s.t. cross_corr(a_old, a)[k] <= target_M for k in peak_indices

            a >= 0

           cons = [A @ a_var <= b, cp.sum(a_var) == 1, a_var >= 0]
           prob = cp.Problem(cp.Minimize(obj), cons)

                   if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                       a_sol = np.maximum(a_var.value, 0)
                       s = float(np.sum(a_sol))
                       if s > 1e-10:

               except Exception:

               viol = np.maximum(0, Av - b)
               val += 1e6 * np.sum(viol ** 2)
               # Penalty for sum constraint
               s = np.sum(xv)
               val += 1e6 * (s - 1.0) ** 2
               # Penalty for non-negativity
               neg = np.maximum(0, -xv)
               val += 1e6 * np.sum(neg ** 2)
               return val

               s = np.sum(xv)
               g += 2e6 * (s - 1.0)
               neg = np.maximum(0, -xv)
               g += 2e6 * neg
               return g

               s = float(np.sum(a_sol))
               if s > 1e-10:
                   return a_sol / s
       except Exception:
           pass
       
       return None

def build_constraint_matrix(a_base, peak_indices):

        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = a_base[k - j]
    return A


def qp_step(a_old, n_peaks=10, reduction=0.995):
    n = len(a_old)
    conv = compute_conv(a_old)
    sorted_idx = np.argsort(conv)[::-1]
    peak_indices = sorted_idx[:n_peaks]
    peak_vals = conv[peak_indices]
    M_target = np.max(peak_vals) * reduction
    M_target = max(M_target, np.min(peak_vals) * 0.95)
    A = build_constraint_matrix(a_old, peak_indices)
    b = np.full(n_peaks, M_target)
    a_var = cp.Variable(n)
    obj = cp.sum_squares(a_var - a_old)
    cons = [A @ a_var <= b, cp.sum(a_var) == 1, a_var >= 0]
    prob = cp.Problem(cp.Minimize(obj), cons)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_sol = np.maximum(a_var.value, 0)
                s = float(np.sum(a_sol))
                if s > 1e-10:
                    return a_sol / s
        except Exception:
            continue
    return None

        best_step = a.copy()
        best_step_obj = best_obj
        for t in [0.3, 0.5, 0.7, 1.0, 0.9, 0.6, 0.4, 0.8, 0.2]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step = a_try.copy()
        if best_step_obj < best_obj - 1e-12:
            best_obj = best_step_obj
            best_a = best_step.copy()
            a = best_step
            stall = 0
            if verbose:
                print(f"    lp: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.10f}")

    # Phase 1: Nesterov smoothed gradient
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: Nesterov Gradient (n={len(best_seq)}) ===")
        alloc = min(remaining * 0.25, 200.0)
        a_g = nesterov_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_g)
        if obj < best_score:
            best_score = obj
            best_seq = a_g.copy()
            print(f"  After nesterov: obj={best_score:.10f}")

    # Phase 2: QP optimization
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: QP Optimization (n={len(best_seq)}) ===")
        alloc = min(remaining * 0.25, 250.0)
        a_qp = qp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_qp)
        if obj < best_score:
            best_score = obj
            best_seq = a_qp.copy()
            print(f"  After QP: obj={best_score:.10f}")
        else:
            print(f"  QP result: obj={obj:.10f} (no improvement)")

    # Phase 3: LP alternating
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: LP Alternating ===")
        alloc = min(remaining * 0.2, 120.0)
        a_lp = lp_alternate_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_qp.copy()
            best_seq = a_lp.copy()
            print(f"  After LP: obj={best_score:.10f}")

    # Phase 4: Second round of QP + gradient
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Refinement (n={len(best_seq)}) ===")
        alloc = min(remaining * 0.3, 150.0)
        a_g2 = nesterov_optimize(best_seq, time_budget=alloc * 0.5, verbose=True)
        obj = compute_obj(a_g2)
        if obj < best_score:
            best_score = obj
            best_seq = a_g2.copy()
            print(f"  Gradient2: obj={best_score:.10f}")
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc2 = min(remaining * 0.4, 100.0)
            a_qp2 = qp_optimize(best_seq, time_budget=alloc2, verbose=True)
            obj = compute_obj(a_qp2)
            if obj < best_score:
                best_score = obj
                best_seq = a_qp2.copy()
                print(f"  QP2: obj={best_score:.10f}")

    # Phase 5: Multi-n exploration
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: Multi-n Exploration ===")
        n_parent = len(best_seq)
        n_candidates = sorted(set([
            max(400, n_parent - 100),
            max(400, n_parent - 50),
            n_parent,
            n_parent + 50,
            n_parent + 100,
            n_parent + 150,
            n_parent + 200,
            900, 1000, 1100, 1200, 1300, 1400
        ]))
        n_candidates = [n for n in n_candidates if 400 <= n <= 1500]
        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            if n_try == len(best_seq):
                continue
            a_try = resample_sequence(best_seq, n_try)
            alloc = min(remaining * 0.15, 40.0)
            a_opt = nesterov_optimize(a_try, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj:.10f}")

    # Phase 6: Final QP polish on best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Final Polish ===")
        alloc = min(remaining * 0.4, 80.0)
        a_final = qp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final QP: obj={best_score:.10f}")

    # Phase 7: Random restarts
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_restarts = min(max(2, int(remaining / 30)), 6)
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            if r == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            elif r == 1:
                n_r = n_best + int(rng.integers(-100, 150))
                n_r = max(500, min(1400, n_r))
                a_r = resample_sequence(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            elif r == 2:
                x = np.linspace(0, 1, n_best)
                a_r = (1 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * rng.normal(0, 1, n_best))
            else:
                a_r = np.ones(n_best) * (1 + 0.2 * rng.normal(0, 1, n_best))
            a_r = np.maximum(a_r, 0.001)
            a_r = a_r / np.sum(a_r)
            alloc = min(remaining * 0.25, 30.0)
            a_opt = nesterov_optimize(a_r, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                n_best = len(best_seq)
                print(f"  Restart {r}: obj={best_score:.10f} **")
            else:
                print(f"  Restart {r}: obj={obj:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
