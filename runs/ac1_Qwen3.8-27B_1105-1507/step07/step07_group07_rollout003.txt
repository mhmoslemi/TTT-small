    p = 1
    while p < m:
        p <<= 1
    return p

def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # conv(a,g)[k] = sum_i a[i]*g[k-i]
        # For fixed j (index of g), g[j] appears in conv(a,g)[k] when k-i=j, i.e., k=i+j
        # So conv(a,g)[i+j] += a[i]*g[j] for i=0,...,min(n-1, 2n-2-j)
        for i in range(min(n, 2*n-1-j)):
            A[i+j, j] += a[i]
    return A

    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # Variables: g[0],...,g[n-1], M
    # Constraints: A @ g - M <= 0 (2n-1 constraints)
    #              sum(g) = 1
    #              g >= 0, M >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq[0, :n] = 1.0
