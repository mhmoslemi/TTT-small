```python
LP-based optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with golden-section line search, top-K LP variants,

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1 for speed."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Variables: g_0, ..., g_{n-1}, M
    # Minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub @ x <= b_ub: A@g - M <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq @ x = b_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
def solve_qp_direction(a):
    """Solve QP: min ||A@g||^2 s.t. 1^T g = 1, g >= 0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # A^T A is n x n
    ATA = A.T @ A
    ones = np.ones(n)
    
    # Try unconstrained: ATA @ g = mu * ones, 1^T g = 1
    # g = mu * (ATA)^{-1} ones
    # mu = 1 / (ones^T (ATA)^{-1} ones)
    try:
        sol = np.linalg.solve(ATA, ones)
        mu = 1.0 / np.sum(sol)
        g = mu * sol
        if np.all(g >= -1e-10):
            g = np.maximum(g, 0)
            return g / np.sum(g)
    except np.linalg.LinAlgError:
        pass
    
    # Fall back to scipy QP
    from scipy.optimize import minimize
    def qp_obj(g):
        Ag = A @ g
        return np.sum(Ag * Ag)
    def qp_grad(g):
        return 2 * A.T @ (A @ g)
    
    res = minimize(qp_obj, np.ones(n)/n, jac=qp_grad,
                   constraints=[{'type': 'eq', 'fun': lambda g: np.sum(g) - 1}],
                   bounds=[(0, None)] * n, method='SLSQP')
    if res.success:
        g = np.maximum(res.x, 0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None
```

```python
"""
LP-alternating optimizer with top-K LP variants, Nesterov subgradient,
and multi-n exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: HiGHS-accelerated LP with fine line search, top-K peak
balancing LP, dual-guided perturbation, and aggressive multi-n search.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None


def solve_lp_topk(a, K):
    """Solve LP: min M + (1/K)*sum_k max(0, (A@g)_k - M) s.t. sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    m = 2 * n - 1

    nv = n + 1 + m
    c = np.zeros(nv)
    c[n] = 1.0
    c[n+1:] = 1.0 / K

    A_ub = np.zeros((m, nv))
    for k in range(m):
        A_ub[k, :n] = A[k, :]
        A_ub[k, n] = -1.0
        A_ub[k, n + 1 + k] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)] + [(0, None)] * m

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None


def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    sv = np.sum(v)
    if sv <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def line_search(a, g, t_values):
    """Find best blend factor t. Returns (best_a, best_obj)."""
    best_obj = np.inf
    best_a = a.copy()
    for t in t_values:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            continue
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    return best_a, best_obj


def lp_alternating(a, time_budget=100.0, verbose=False, use_topk=False, topk_K=20):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_values = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]
    stall = 0
    perturb_scale = 0.03

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        if use_topk:
            g = solve_lp_topk(a, topk_K)
        else:
            g = solve_lp(a)
        if g is None:
            break

        a_new, obj_new = line_search(a, g, t_values)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new
            a = a_new
            stall = 0
            if verbose and it % 50 == 0:

        else:
            stall += 1
            if stall > 30:

                a = best_a * np.exp(rng.normal(0, perturb_scale, n))

def nesterov_subgrad(a, time_budget=60.0):
    """Nesterov accelerated subgradient descent on max-conv objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    y = a.copy()

        a = a_new

            best_a = a_new.copy()
            y = a_new.copy()
            stall = 0

                v = np.zeros(n)
                a = best_a.copy()
                y = best_a.copy()
                stall = 0
                if lr < 1e-5:

def smooth_max_grad(a, time_budget=60.0, T=0.001):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    momentum = 0.85
    v = np.zeros(n)

        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = np.clip((conv - max_c) / T, -500, 500)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)

        grad = np.correlate(weights, a, mode='valid')
        grad_c = grad - np.mean(grad)

        v = momentum * v - lr * grad_c
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-5:
                    break

        if it > 100:
            T = max(0.0001, T * 0.995)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            print(f"  Nesterov: obj={obj_ns:.8f}")

    # === Phase 4: Smooth Max Gradient ===
    print(f"\n=== Phase 4: Smooth Max Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 40.0)
        a_sm = smooth_max_grad(best_seq, time_budget=alloc)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score - 1e-8:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  SmoothMax: obj={best_score:.8f} **")
        else:
            print(f"  SmoothMax: obj={obj_sm:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best - 20,
        n_best + 20, n_best + 50, n_best + 100, n_best + 200,
        400, 500, 600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and n != n_best]
    # Sort by proximity to n_best first
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 50.0)
        a_opt = lp_alternating(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        # Quick polish with smooth max if promising
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.005:
            alloc_sm = min(remaining * 0.1, 20.0)
            a_opt2 = smooth_max_grad(a_opt, time_budget=alloc_sm)
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        if obj_try < best_score - 1e-8:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Random Restarts ===
    print(f"\n=== Phase 6: Random Restarts ===")
    n_best = len(best_seq)

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        n_r = max(400, min(1500, n_best + int(rng.integers(-100, 101))))
        mode = r % 5
        if mode == 0:
            a_r = np.ones(n_r) / n_r
        elif mode == 1:
            a_r = rng.dirichlet(np.ones(n_r) * 0.3)
        elif mode == 2:
            x = np.arange(n_r)
            a_r = np.exp(-0.5 * ((x - n_r / 2) / (n_r / 5)) ** 2)
            a_r = a_r / np.sum(a_r)
        elif mode == 3:
            x = np.linspace(0, 1, n_r)
            a_r = (1 - np.abs(2 * x - 1)) ** 2 + 0.05
            a_r = a_r / np.sum(a_r)
        else:
            a_r = np.cos(np.linspace(0, np.pi, n_r)) ** 2 + 0.1
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.12, 30.0)
        a_opt = lp_alternating(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score - 1e-8:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (n={n_r}, mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}, mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    a_final = best_seq.copy()
    for polish in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.25, 40.0)
        a_lp = lp_alternating(a_final, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-8:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Polish {polish}: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_sm2 = smooth_max_grad(a_final, time_budget=min(remaining * 0.2, 20.0), T=0.0005)
        obj_sm2 = compute_obj(a_sm2)
        if obj_sm2 < best_score - 1e-8:
            best_score = obj_sm2
            best_seq = a_sm2.copy()
            print(f"  Final Smooth: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_ns2 = nesterov_subgrad(a_final, time_budget=min(remaining * 0.2, 15.0))
        obj_ns2 = compute_obj(a_ns2)
        if obj_ns2 < best_score - 1e-8:
            best_score = obj_ns2
            best_seq = a_ns2.copy()
            print(f"  Final Nesterov: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.ones(100))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
