"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            s = float(np.sum(g_val))

    except Exception:

    return None, None


def compute_smoothed_gradient(a, tau, n):

    Returns (objective_value, gradient_array).
    Uses FFT-based cross-correlation for speed.
    """

    conv_max = np.max(conv)
    # Softmax weights with numerical stability

    w = exp_shifted / np.sum(exp_shifted)

            for shrinks in range(20):
                alpha = step_size / (2.0 ** shrinks)
                a_new = a - alpha * grad_proj
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new < 1e-10:
                    break
                a_new = a_new / s_new
                new_obj = compute_obj(a_new)
                # Armijo condition (loose)
                if new_obj < current_obj - 1e-10 or new_obj < best_obj:
                    a = a_new
                    current_obj = new_obj
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = a.copy()
                    improved = True
                    step_size = min(step_size * 1.05, 2.0)
                    break
            if not improved:
                step_size *= 0.7
                if step_size < 1e-6:
                    break
        
        # After each tau stage, update best
        a = best_a.copy()
    
    return best_a, best_obj


def lp_alternating(a, time_budget=60.0):
    """LP alternating optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            # Perturb to escape
            if it % 5 == 0:
                rng = np.random.default_rng(it * 104729 + n)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length using cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use linear interpolation (cubic can produce negatives)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def parametric_optimize(n, n_basis, time_budget, rng=None):
    """
    Optimize over an exponential polynomial parameterization:
    a[i] = exp(sum_k c_k * (i/n)^k) / normalization
    This reduces dimensionality and may find good basins.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    start = time.monotonic()
    
    def param_to_seq(c):
        x = np.linspace(0, 1, n)
        log_a = np.zeros(n)
        for k in range(len(c)):
            log_a += c[k] * x ** (k + 1)
        # Shift for numerical stability
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a = a / np.sum(a)
        return a
    
    best_a = None
    best_obj = np.inf
    
    for trial in range(30):
        if time.monotonic() - start > time_budget:
            break
        
        # Random initialization of parameters
        c0 = rng.uniform(-3, 3, n_basis)
        a_init = param_to_seq(c0)
        obj_init = compute_obj(a_init)
        
        def obj_func(c):
            a = param_to_seq(c)
            return compute_obj(a)
        
        def grad_func(c):
            # Numerical gradient
            eps = 1e-5
            g = np.zeros(n_basis)
            for i in range(n_basis):
                c_plus = c.copy()
                c_plus[i] += eps
                c_minus = c.copy()
                c_minus[i] -= eps
                g[i] = (obj_func(c_plus) - obj_func(c_minus)) / (2 * eps)
            return g
        
        try:
            result = minimize(obj_func, c0, jac=grad_func, method='Nelder-Mead',
                            options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-8})
            if result.x is not None:
                a_opt = param_to_seq(result.x)
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt.copy()
        except Exception:
            pass
    
    if best_a is not None:
        return best_a, best_obj
    return None, np.inf


def diverse_random_init(n, rng, init_type=0):
    """Generate diverse random initializations."""
    if init_type == 0:
        # Linear ramp
        slope = rng.uniform(1.5, 4.0)
        a = np.linspace(1.0, slope, n)
    elif init_type == 1:
        # Exponential
        rate = rng.uniform(0.5, 4.0)
        if rng.random() < 0.5:
            a = np.exp(np.linspace(0, rate, n))
        else:
            a = np.exp(np.linspace(rate, 0, n))
    elif init_type == 2:
        # Quadratic (U-shaped or inverted)
        x = np.linspace(0, 1, n)
        curv = rng.uniform(-3, 3)
        a = 1.0 + curv * (x - 0.5) ** 2
        a = np.maximum(a, 0.01)
    elif init_type == 3:
        # Sine modulation
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.sin(freq * np.pi * x + phase)
        a = np.maximum(a, 0.01)
    elif init_type == 4:
        # Random smooth (low-pass filtered noise)
        raw = rng.normal(0, 1, n)
        # Simple moving average as low-pass
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(raw, kernel, mode='same')
        a = a - np.min(a) + 0.1
    elif init_type == 5:
        # Two-hump
        center1 = rng.uniform(0.1, 0.4)
        center2 = rng.uniform(0.6, 0.9)
        width = rng.uniform(0.05, 0.2)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - center1) / width) ** 2) + 0.5 * np.exp(-((x - center2) / width) ** 2)
    else:
        # Cosine ramp (smooth)
        a = 0.5 + 0.5 * np.cos(np.pi * np.linspace(0, 1, n))
    
    a = np.maximum(a, 0.001)
    a = a / np.sum(a)
    return a


def coordinate_descent_fast(a, time_budget=30.0):
    """Fast coordinate descent targeting peak contributors."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345 + n)
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        
        # Try moving mass from peak contributors to other positions
        for _ in range(100):
            # Pick a source that contributes to the peak
            valid_i = []
            for i in range(n):
                if k_star - 0 <= i < k_star + n:  # i contributes to conv[k_star]
                    if a[i] > 1e-6:
                        valid_i.append(i)
            if not valid_i:
                break
            i = valid_i[rng.integers(0, len(valid_i))]
            j = rng.integers(0, n)
            if i == j:
                continue
            
            for frac in [0.02, 0.05, 0.1]:
                delta = frac * a[i]
                if delta >= a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    break
        
        if not improved:
            # Random pair swap
            for _ in range(30):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-6:
                    continue
                delta = 0.03 * a[i] * rng.uniform(0.5, 1.5)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                # Larger perturbation
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
    
    return best_a


def subgradient_push(a, time_budget=30.0):
    """
    Subgradient method: push mass away from peak contributors.
    At each step, move mass from positions that contribute to the current peak.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999 + n)
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Compute contribution of each a[i] to conv[k_star]
        # conv[k_star] = sum_i a[i] * a[k_star - i] (valid i)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[j]
        
        # Move mass from high-contributing positions to low-contributing ones
        total_contrib = np.sum(contrib)
        if total_contrib < 1e-10:
            break
        
        # Target: reduce positions where contrib is high, increase where it's low
        # Move a small fraction of mass
        step = 0.02
        top_k = max(3, n // 20)
        bottom_k = max(3, n // 30)
        
        top_idx = np.argsort(contrib)[-top_k:]
        bottom_idx = np.argsort(contrib)[:bottom_k]
        
        # Average mass from top, distribute to bottom
        mass_to_move = step * np.sum(a[top_idx]) / len(top_idx)
        
        a_new = a.copy()
        for i in top_idx:
            a_new[i] -= mass_to_move / len(top_idx)
        for j in bottom_idx:
            a_new[j] += mass_to_move / len(bottom_idx)
        
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller step
                a_new2 = a - 0.005 * (np.maximum(a[top_idx], 0) - a) 
                # Just add random noise
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    # Phase 1: Smoothed gradient descent with tau annealing (multiple starts)
    print(f"\n=== Phase 1: Smoothed Gradient Descent ===")
    tau_schedules = [
        [5, 20, 80, 300, 800],
        [10, 50, 200, 600],
        [3, 15, 60, 250, 1000],
    ]
    
    for sched_idx, taus in enumerate(tau_schedules):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        if sched_idx == 0:
            a_init = best_seq.copy()
            alloc = min(remaining * 0.3, 200.0)
        elif sched_idx == 1:
            a_init = best_seq * np.exp(rng.normal(0, 0.08, n_parent))
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / np.sum(a_init)
            alloc = min(remaining * 0.25, 150.0)
        else:
            # Bidirectional perturbation
            ramp = np.linspace(0.7, 1.3, n_parent)
            a_init = best_seq * ramp
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / np.sum(a_init)
            alloc = min(remaining * 0.2, 100.0)
        
        a_opt, obj_opt = smoothed_grad_descent(a_init, taus, time_budget=alloc,
                                                max_iter_per_tau=400, rng=rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_parent = len(best_seq)
            print(f"  SGD sched{sched_idx}: obj={best_score:.8f} **")
        else:
            print(f"  SGD sched{sched_idx}: obj={obj_opt:.8f}")
    
    # Phase 2: LP Alternating
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc_lp = min(remaining * 0.2, 120.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc_lp)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            n_parent = len(best_seq)
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")
    
    # Phase 3: Parametric optimization (reduced dimensionality)
    print(f"\n=== Phase 3: Parametric Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for n_basis in [5, 8, 12]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_param, obj_param = parametric_optimize(n_parent, n_basis,
                                                     time_budget=min(remaining * 0.15, 30.0),
                                                     rng=rng)
            if a_param is not None and obj_param < best_score:
                best_score = obj_param
                best_seq = a_param.copy()
                print(f"  Parametric (basis={n_basis}): obj={best_score:.8f} **")
            else:
                print(f"  Parametric (basis={n_basis}): obj={obj_param:.8f}")
    
    # Phase 4: Subgradient push + coordinate descent
    print(f"\n=== Phase 4: Subgradient + Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 40.0)
        a_sg = subgradient_push(best_seq, time_budget=alloc)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgradient: obj={best_score:.8f} **")
        else:
            print(f"  Subgradient: obj={obj_sg:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd = coordinate_descent_fast(best_seq, time_budget=min(remaining * 0.2, 30.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord Descent: obj={best_score:.8f} **")
        else:
            print(f"  Coord Descent: obj={obj_cd:.8f}")
    
    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(400, n_cur - 200), max(400, n_cur - 100),
        n_cur - 50, n_cur, n_cur + 50,
        n_cur + 100, n_cur + 200, n_cur + 400,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if n_try == n_cur:
            continue  # Already optimized at this n
        
        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / np.sum(a_try)
        
        # Quick optimization at this n
        alloc = min(remaining * 0.15, 40.0)
        a_opt, obj_opt = smoothed_grad_descent(a_try, [20, 100, 400],
                                                time_budget=alloc, max_iter_per_tau=200, rng=rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = len(best_seq)
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # Phase 6: Random restarts with diverse initializations
    print(f"\n=== Phase 6: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    for r in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        
        init_type = r % 6
        a_r = diverse_random_init(n_best, rng, init_type)
        
        # Blend with current best for warm start
        blend = rng.uniform(0.3, 0.6)
        a_r = blend * best_seq + (1 - blend) * a_r
        a_r = np.maximum(a_r, 1e-10)
        a_r = a_r / np.sum(a_r)
        
        alloc = min(remaining * 0.12, 30.0)
        a_opt, obj_opt = smoothed_grad_descent(a_r, [15, 80, 300],
                                                time_budget=alloc, max_iter_per_tau=150, rng=rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (type={init_type}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_opt:.8f}")
    
    # Phase 7: Final intensive polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 40:
        # LP alternating (final)
        alloc = min(remaining * 0.25, 80.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # High-tau gradient descent (very sharp peak targeting)
        alloc = min(remaining * 0.25, 50.0)
        a_opt, obj_opt = smoothed_grad_descent(best_seq, [100, 400, 1500],
                                                time_budget=alloc, max_iter_per_tau=300, rng=rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final High-Tau SGD: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more LP pass
        alloc = min(remaining * 0.3, 40.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f}")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Final coordinate descent
        a_cd = coordinate_descent_fast(best_seq, time_budget=min(remaining * 0.4, 20.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
