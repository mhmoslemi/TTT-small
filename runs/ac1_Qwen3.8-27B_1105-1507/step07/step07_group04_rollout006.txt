"""
Optimized LP alternating + smooth refinement for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses fast LP solving with warm starts, log-sum-exp smoothing for polish,
and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_fast(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build convolution matrix A (2n-1 x n) where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - (n - 1))  # k - j >= 0 => k >= j
        hi = min(2 * n - 2, j + (n - 1))  # k - j <= n-1 => k <= j + n - 1
        k_range = np.arange(lo, hi + 1)
        A[k_range, j] = a[k_range - j]
    return A


def solve_lp(a, solver=None):
    """
    Solve LP: min M s.t. A@z <= M*1, sum(z)=1, z>=0.
    Returns (z, M) or (None, None).
    """

    A = build_A(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers
    
    for s_name in solvers:
        try:
            prob.solve(solver=s_name)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    return z_val / zs, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                return z_val / zs, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternate(a, time_budget=100.0, verbose=False):
    """LP alternating optimization with adaptive step size."""
    n = len(a)

        else:

def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

And the second sum: sum_k softmax[k] * a[i-k]
Let l = i-k, then k = i-l:
= sum_l softmax[i-l] * a[l]

This is the cross-correlation of a and softmax, evaluated at offset i.

So grad_a[i] = (1/tau) * (cross_corr(softmax, a)[i] + cross_corr(a, softmax)[i])

And cross_corr(x, y)[i] = sum_j x[j+i] * y[j]

This can be computed using FFT:
cross_corr(x, y) = ifft(fft(x_padded) * conj(fft(y_padded)))
