import numpy as np

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    m = 2 * n - 1
    while size < m:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:m]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    an = a / s
    conv = compute_conv(an)
    return 2.0 * n * float(np.max(conv))

def solve_lp_direction(a):
    """
    Given normalized a (sum=1), solve:
      maximize sum(g)
      subject to: conv(a, g)[k] <= 1 for all k = 0, ..., 2n-2
                  g >= 0
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    
    c = -np.ones(n)
    b_ub = np.ones(2 * n - 1)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        g = np.maximum(res.x, 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs
    return None

def line_search(a, g, n_evals=30):
    """Find best t in [0,1] minimizing f((1-t)*a + t*g)."""
    n = len(a)
    best_t = 0.0

        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return a * (1 - best_t) + g * best_t if best_t > 0 else a.copy(), best_obj

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    mu = 0.0
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) > 0:
        rho = rho[-1]
        mu = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - mu, 0)

def random_init(n, rng):

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses fast alternating LP (HiGHS), golden-section line search on the convex
1D objective, multiple diverse initializations, periodic restarts, gradient
polish via subgradient + simplex projection, and n-sweep.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    m = 2 * n - 1
    size = 1
    while size < m:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:m]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    an = a / s
    conv = compute_conv(an)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build (2n-1) x n matrix A where (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_direction(a):
    """
    Given normalized a (sum=1, a>=0), solve:
      maximize sum(g)  s.t.  conv(a,g)[k] <= 1 for all k,  g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.ones(2 * n - 1)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.status == 0:
            g = np.maximum(res.x, 0)
            gs = float(np.sum(g))
            if gs > 1e-12:
                return g / gs
    except Exception:
        pass
    return None


def line_search_golden(a, g, max_evals=25):
    """
    Golden-section search to minimize f(t) = obj((1-t)*a + t*g) for t in [0,1].
    f(t) is convex (max of convex quadratics), so golden section is optimal.
    Returns (best_a, best_obj, best_t).
    """
    n = len(a)

    def f(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-12:
            return np.inf
        a_t = a_t / s
        return compute_obj(a_t)

    # First check endpoints
    f0 = f(0.0)
    f1 = f(1.0)
    if f1 <= f0:
        return g.copy(), f1, 1.0

    # Golden section on [0, 1] (we know f(0) < f(1))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    f_c = f(c_pt)
    f_d = f(d_pt)
    evals = 3

    for _ in range(max_evals):
        if f_c < f_d:
            a_hi = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = a_hi - gr * (a_hi - a_lo)
            f_c = f(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = a_lo + gr * (a_hi - a_lo)
            f_d = f(d_pt)
        evals += 1
        if a_hi - a_lo < 1e-8:
            break

    best_t = (a_lo + a_hi) / 2.0
    best_obj = f(best_t)
    a_best = (1.0 - best_t) * a + best_t * g
    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s > 1e-12:
        a_best = a_best / s
    return a_best, best_obj, best_t


def alternating_optimize(a, time_budget, max_iters=200000, verbose=False):
    """
    Alternating LP optimization with golden-section line search.
    Periodically restarts with perturbations to escape local minima.
    Returns (best_a, best_obj).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n, 2.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    eval_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g = solve_lp_direction(a)
        if g is None or np.allclose(g, 0):
            stall += 1
            if stall > 30:
                break
            if stall % 10 == 0:
                rng = np.random.default_rng((it * 7919 + n) % (2**32))
                scale = min(0.05 + stall * 0.005, 0.4)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
            continue

        a_new, obj_new, t_used = line_search_golden(a, g)
        eval_count += 1

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and (it % 200 == 0 or it < 5):
                print(f"    alt it{it}: obj={best_obj:.10f}, t={t_used:.4f}")
        else:
            stall += 1
            if stall > 40:
                rng = np.random.default_rng((it * 31 + 13) % (2**32))
                scale = min(0.03 + stall * 0.003, 0.3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a, best_obj


def gradient_polish(a, time_budget):
    """
    Subgradient descent with Nesterov momentum and simplex projection.
    Exploits the structure: grad[i] = 2*a[k*-i] at peak position k*.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.02
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * a[k_star - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        # Nesterov momentum
        v = momentum * v - lr * grad
        a_new = a + v

        # Simplex projection
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.005, 0.3)
        else:
            stall += 1
            lr *= 0.97
            a = best_a.copy()
            if stall % 100 == 0 and lr > 1e-6:
                rng = np.random.default_rng(it * 7 + 3)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
            if lr < 1e-8:
                break

    return best_a, best_obj


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_arr = np.arange(1, n + 1)
    cond = v_sorted * rho_arr > (cssv - 1.0)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) > 0:
        rho = rho_idx[-1]
        mu = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        mu = 0.0
    return np.maximum(v - mu, 0)
