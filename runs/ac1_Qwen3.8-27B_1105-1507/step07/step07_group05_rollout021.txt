import numpy as np

from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def compute_grad_smoothed(a, p=4.0):
    """
    Compute gradient of sum_k (b_k/b_max)^p * b_k w.r.t. a_i.
    This approximates the max as p -> infinity.
    Returns (gradient, b).
    """
    n = len(a)
    b = compute_conv(a)
    b_max = np.max(b)
    if b_max < 1e-30:
        return np.zeros(n), b
    # Power weights
    w = (b / b_max) ** p
    w = w / np.sum(w)
    # grad[i] = sum_k w[k] * a[k-i] (valid indices only)
    # = conv(w, a_flip)[n-1 : 2n-1]
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    fw = rfft(w, N)
    faf = rfft(a[::-1], N)
    c = irfft(fw * faf, N)[:3 * n - 2]
    grad = c[n - 1:2 * n - 1]
    return grad, b


def compute_grad_exp(a, T=0.05):
    """
    Compute gradient using exponential (softmax) smoothing.
    Returns (gradient, b).
    """
    n = len(a)
    b = compute_conv(a)
    b_shifted = b - np.max(b)
    w = np.exp(b_shifted / T)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.zeros(n), b
    w = w / w_sum
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    fw = rfft(w, N)
    faf = rfft(a[::-1], N)
    c = irfft(fw * faf, N)[:3 * n - 2]
    grad = c[n - 1:2 * n - 1]
    return grad, b


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rh О + 1.0) if False else (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gd_nesterov(a, time_budget=120.0, p_initial=2.0, p_final=30.0, verbose=False):
    """
    Smoothed gradient descent with Nesterov momentum and power-based cooling.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    momentum = 0.7
    v = np.zeros(n)
    y = a.copy()  # Nesterov look-ahead

    max_iters = 30000
    stall = 0
    max_stall = 300

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Cooling schedule for power p
        progress = min(it / max_iters, 1.0)
        # Use a smooth transition
        p = p_initial + (p_final - p_initial) * progress ** 0.7

        # Nesterov step: evaluate gradient at y (look-ahead point)
        if momentum > 0:
            grad_y, b_y = compute_grad_smoothed(y, p)
        else:
            grad_y, b_y = compute_grad_smoothed(a, p)

        # Center gradient (we're optimizing on the simplex)
        grad_y = grad_y - np.mean(grad_y)

        # Momentum update
        v = momentum * v - eta * grad_y
        a_new = a + v
        a_new = project_simplex(a_new)

        # Evaluate
        b_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.003, 5.0)
            momentum = min(momentum + 0.0002, 0.92)
        else:
            stall += 1
            eta *= 0.98
            v *= 0.5
            if stall > max_stall:
                # Perturbation restart
                rng_local = np.random.default_rng(it * 7 + 13)
                noise = rng_local.normal(0, 0.03 * (1 + stall / max_stall), n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                v = np.zeros(n)
                eta = 0.5
                momentum = 0.7
                stall = 0
                max_stall = min(max_stall + 100, 800)
            if eta < 1e-10:
                break

        # Update Nesterov look-ahead
        y = a + momentum * v

        if verbose and it % 2000 == 0:
            print(f"            GD-Nest it{it}: obj={best_obj:.8f}, p={p:.1f}, eta={eta:.4f}, mom={momentum:.3f}")

    return best_a


def dual_mwu(a, time_budget=60.0, verbose=False):
    """
    Dual multiplicative weight update for the max-min problem:
    max_y min_i (A^T y)_i s.t. y in simplex.
    The primal recovery: set a_i proportional to indicator of argmin.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Dual variable: distribution over k = 0, ..., 2n-2
    m = 2 * n - 1
    y = np.ones(m) / m  # uniform initialization
    a_avg = a.copy()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta_dual = 0.1
    T_avg = 100

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        # Compute A^T y: (A^T y)_i = sum_k y[k] * a[k-i] (valid)
        # This is conv(y, a_flip)[n-1:2n-1]
        N = 1
        while N < 3 * n - 1:
            N <<= 1
        fy = rfft(y, N)
        faf = rfft(a[::-1], N)
        c = irfft(fy * faf, N)[:3 * n - 2]
        At_y = c[n - 1:2 * n - 1]

        # Player 2 (minimizer) best response
        i_star = int(np.argmin(At_y))

        # Update dual: multiplicative weight
        # y_k *= exp(eta * (At_y[i_star] contribution to k))
        # (A^T y)_i includes y[k] for k where a[k-i] != 0
        # For MWU on dual: y_k *= exp(eta * A[k, i_star] / max_z A[z, i_star])
        # A[k, i_star] = a[k - i_star] if 0 <= k - i_star < n else 0
        k_valid = np.arange(i_star, i_star + n)
        k_valid = k_valid[k_valid < m]
        if len(k_valid) > 0:
            vals = a[k_valid - i_star]
            max_val = np.max(vals)
            if max_val > 1e-30:
                y[k_valid] *= np.exp(eta_dual * vals / max_val)
                y = y / np.sum(y)

        # Player 1 (maximizer) best response on current a
        b = compute_conv(a)
        k_star = int(np.argmax(b))

        # Update a toward the support of near-optimal coordinates
        # Use the dual solution to guide a
        threshold = At_y.max() - 0.1 * (At_y.max() - At_y.min())
        support = At_y >= threshold
        if np.sum(support) > 0:
            a_dual = np.where(support, 1.0, 0.0)
            a_dual = a_dual / np.sum(a_dual)
        else:
            a_dual = np.ones(n) / n

        # Mix
        t = 2.0 / (it + 2)
        a_new = (1 - t) * a + t * a_dual
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Still update a with smaller step
            t_small = 1.0 / (it + 2)
            a = (1 - t_small) * a + t_small * a_dual
            a = np.maximum(a, 0)
            a = a / np.sum(a)

        if verbose and it % 500 == 0:
            print(f"            MWU it{it}: obj={best_obj:.8f}, min_At_y={At_y.min():.6f}")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Build convolution matrix A (2n-1 x n)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        try:
            from scipy.optimize import linprog
            # min M s.t. A_g @ g <= M, sum(g) = 1, g >= 0
            # Variables: [g_0, ..., g_{n-1}, M]
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((m, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(m)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(0, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    g_val = g_val / s_g
                    # Line search
                    best_t_obj = np.inf
                    best_t = 0.3
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * a + t * g_val
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_t = t

                    if best_t_obj < best_obj - 1e-11:
                        best_obj = best_t_obj
                        best_a = (1 - best_t) * a + best_t * g_val
                        best_a = np.maximum(best_a, 0)
                        best_a = best_a / np.sum(best_a)
                        a = best_a.copy()
                        if verbose:
                            print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
                    else:
                        break
        except Exception:
            break

    return best_a


def peak_chase_accelerated(a, time_budget=40.0):
    """Accelerated peak chasing with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.1
    v = np.zeros(n)
    momentum = 0.8

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv(a)
        b_max = np.max(b)

        # Find all near-peak positions
        threshold = 0.95 * b_max
        peaks = np.where(b >= threshold)[0]

        # Average subgradient over near-peaks
        grad = np.zeros(n)
        for k in peaks:
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(peaks) if len(peaks) > 0 else 1
        grad -= np.mean(grad)

        v = momentum * v - eta * grad
        a_new = a + v
        a_new = project_simplex(a_new)

        b_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 1.0)
        else:
            eta *= 0.97
            v *= 0.5
            if eta < 1e-9:
                break

    return best_a


def diverse_init(n, mode, rng):
    """Generate diverse initial sequences."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # U-shaped
        x = np.arange(n) / max(n - 1, 1)
        a = (x - 0.5) ** 2 * 4 + 0.05
        return a / np.sum(a)
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / max((n // 8) ** 2, 1))
        return a / np.sum(a)
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        return a / np.sum(a)
    elif mode == 5:
        # Asymmetric ramp
        x = np.arange(n, dtype=float)
        a = np.exp(-x / (n * 0.5))
        return a / np.sum(a)
    elif mode == 6:
        # Triangular
        x = np.arange(n, dtype=float)
        a = 1.0 - np.abs(x - (n - 1) / 2) / ((n - 1) / 2 + 1)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif mode == 7:
        # Sparse random
        a = np.zeros(n)
        k = max(n // 4, 10)
        idx = rng.choice(n, size=k, replace=False)
        a[idx] = rng.uniform(0.5, 1.0, k)
        return a / np.sum(a)
    else:
        # Concentrated center
        x = np.arange(n, dtype=float)
        a = np.exp(-((x - n / 2) ** 2) / (n * 0.1))
        return a / np.sum(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def exp_smooth_gd(a, time_budget=60.0, T_initial=0.1, T_final=0.005, verbose=False):
    """Exponential smoothing gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.3
    momentum = 0.75
    v = np.zeros(n)
    y = a.copy()

    max_iters = 30000
    stall = 0
    max_stall = 400

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = min(it / max_iters, 1.0)
        T = T_initial * (T_final / T_initial) ** progress

        if momentum > 0:
            grad, b = compute_grad_exp(y, T)
        else:
            grad, b = compute_grad_exp(a, T)

        grad = grad - np.mean(grad)

        v = momentum * v - eta * grad
        a_new = a + v
        a_new = project_simplex(a_new)

        b_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(b_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.002, 3.0)
            momentum = min(momentum + 0.0001, 0.9)
        else:
            stall += 1
            eta *= 0.985
            v *= 0.3
            if stall > max_stall:
                rng_local = np.random.default_rng(it * 3 + 7)
                noise = rng_local.normal(0, 0.02, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                v = np.zeros(n)
                eta = 0.3
                momentum = 0.75
                stall = 0
                max_stall = min(max_stall + 200, 1000)
            if eta < 1e-10:
                break

        y = a + momentum * v

        if verbose and it % 5000 == 0:
            print(f"            ExpGD it{it}: obj={best_obj:.8f}, T={T:.5f}, eta={eta:.4f}")

    return best_a

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(time_left() * 0.12, 60.0)
        a_opt = smoothed_gd_nesterov(a_try, time_budget=alloc, p_initial=3.0, p_final=30.0, verbose=False)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Diverse multi-start ===
    print("\n=== Phase 8: Diverse Multi-Start ===")
    n_use = len(best_seq)
    n_starts = max(3, int(time_left() / 50.0))
    n_starts = min(n_starts, 10)

    for r in range(n_starts):
        if time_left() < 30:
            break
        mode = int(rng.integers(0, 9))
        a_r = diverse_init(n_use, mode, rng)
        alloc = min(time_left() * 0.1, 40.0)
        a_opt = smoothed_gd_nesterov(a_r, time_budget=alloc, p_initial=2.0, p_final=25.0, verbose=False)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  ** Start {r} (mode {mode}): obj={best_score:.8f} **")
        else:
            print(f"  Start {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 9: Final combined polish ===
    print("\n=== Phase 9: Final Polish ===")
    if time_left() > 60:
        # Power GD
        alloc = min(time_left() * 0.25, 80.0)
        a_f1 = smoothed_gd_nesterov(best_seq, time_budget=alloc, p_initial=4.0, p_final=50.0, verbose=False)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  ** Final GD: obj={best_score:.8f} **")

        # LP
        if time_left() > 40:
            alloc = min(time_left() * 0.2, 60.0)
            a_f2 = lp_polish(best_seq, time_budget=alloc, verbose=False)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  ** Final LP: obj={best_score:.8f} **")

        # Peak chase
        if time_left() > 20:
            a_f3 = peak_chase_accelerated(best_seq, time_budget=min(time_left() * 0.15, 25.0))
            obj_f3 = compute_obj(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  ** Final Peak: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
