    n = len(a)
    a = a / np.sum(a)  # normalize
    best = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iters):
        conv = np.convolve(a, a)
        target = 1.0 / (2*n - 1)  # ideal flat value
        
        # Compute "excess" at each position
        excess = np.maximum(conv - target, 0)
        
        # For each position with excess, the gradient is a shifted version of a
        # We want to reduce a where it contributes most to the excess
        grad = np.zeros(n)
        for k in range(2*n-1):
            if excess[k] > 0:
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        grad[i] += excess[k] * a[j]
        
        # Gradient step: reduce a where grad is high
        eta = 0.1
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best = a_new.copy()
                a = a_new
            else:
                eta *= 0.9
    
    return best

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    L = next_fast_len(2 * n - 1)
    fa = rfft(a, L)
    return irfft(fa * fa, L)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, max_iters_inner=50):
    """Solve LP: min t s.t. conv(a,g)[k] <= t, sum(g)=1, g>=0.
    Returns (g, t) or (None, None)."""

    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))

                      bounds=bounds, method='highs',
                      options={'maxiter': max_iters_inner})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

    except Exception:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, rng):

        a = a / np.sum(a)
    elif mode == 2:

        x = np.linspace(0, np.pi, n)

        a = a / np.sum(a)
    elif mode == 3:
        # Triangle window
        center = n // 2
        a = 1.0 - np.abs(np.arange(n) - center) / (center + 1)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 4:
        # Exponential decay pair
        c = n // 2
        sigma = n // 6
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * sigma ** 2))
        a = a / np.sum(a)
    elif mode == 5:
        # Binary-ish (rounded)
        a = rng.integers(0, 3, n).astype(float)

        x = np.arange(1, n + 1)
        alpha = rng.uniform(0.5, 2.0)

                    a[i1] = 0
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    found = True
                    break
                else:
                    a[i1] += delta
                    a[i2] -= delta
            if not found and it > 100:
                break

    return best_a
