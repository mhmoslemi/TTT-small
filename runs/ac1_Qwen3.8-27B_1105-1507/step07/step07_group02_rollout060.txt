Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses symmetry exploitation, fast subgradient with momentum, 
structured initializations, and LP alternating for escaping local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv_fast(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_fast(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fast(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # First, try: x = max(v - t, 0) where t is chosen so sum(x) = target_sum
    # Sort v in descending order
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.max(np.where(v_sorted > 0)[0]) if np.any(v_sorted > 0) else 0
    # Find the right t
    # sum(max(v - t, 0)) = target_sum
    # For v > t: (v_i - t) + for v <= t: 0
    # If we sort descending, find rho such that sum_{i<=rho}(v_i - t) = target_sum
    # t = (sum_{i<=rho}(v_i) - target_sum) / (rho+1)
    # Binary search for rho
    cs = np.concatenate([[0], cumsum])
    vals = (cs[:-1] - target_sum) / np.arange(1, n+1)

    rho = int(np.max(np.where(mask)[0]))
    t = (cs[rho+1] - target_sum) / (rho + 1)
    return np.maximum(v - t, 0)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses fast LP alternating (scipy HiGHS), subgradient descent, multi-scale
initialization, and adaptive n exploration to beat the 1.5027 target.
Key improvements: direct linprog calls (no cvxpy overhead), dual-guided
line search, multi-peak flattening, and symmetry-aware perturbations.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1 << (2 * n - 2).bit_length()
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_mat(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    A = build_conv_mat(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP alternating optimization with line search and stall detection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    rng = np.random.default_rng(12345)

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g = solve_lp_fast(a)
        if g is None:
            break

        # Line search
        t_list = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        if 0.05 < last_t < 1.0:
            t_list = sorted(set(t_list + [max(0.05, last_t * 0.5), min(1.0, last_t * 1.5)]))

        best_t_obj = np.inf
        best_t_val = 0.5
        for t in t_list:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t

        if best_t_obj < best_obj - 1e-10:
            a = (1.0 - best_t_val) * a + best_t_val * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            best_obj = best_t_obj
            last_t = best_t_val
            stall = 0
            if verbose and it % 20 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall += 1
            if stall > 40:
                scale = 0.03 + 0.02 * min(stall // 40, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0

def subgradient_optimize(a, time_budget=60.0, verbose=False):
    """Subgradient descent with momentum on normalized objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.02
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    rng = np.random.default_rng(999)

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient at k_star
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
                if i == j:
                    grad[i] += a[i]

        # Center to preserve sum
        grad = grad - np.mean(grad)

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad

        # Step
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.03, 0.15)
            if verbose and it % 200 == 0:
                print(f"      SGD it{it}: obj={best_obj:.8f}, lr={lr:.5f}")
        else:
            stall += 1
            if stall > 300:
                a = best_a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                lr = 0.02
                stall = 0
            else:
                lr = max(lr * 0.97, 0.003)

    return best_a


def multi_peak_optimize(a, time_budget=60.0, K=3):
    """Optimize to flatten top-K peaks of the convolution simultaneously."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    alpha = 0.05

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Find top K peak positions
        top_k = np.argsort(conv)[::-1][:K]

        # Build combined subgradient from top-K peaks
        grad = np.zeros(n)
        weights = np.zeros(2 * n - 1)
        for rank, k_pos in enumerate(top_k):
            w = (K - rank) / (K * (K + 1) / 2.0)
            weights[k_pos] = w

        for k_pos in top_k:
            w = weights[k_pos]
            for i in range(n):
                j = k_pos - i
                if 0 <= j < n:
                    grad[i] += w * a[j]

        grad = grad - np.mean(grad)

        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.02, 0.2)
        else:
            alpha *= 0.95
            if alpha < 0.005:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                alpha = 0.05

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, mode, rng):
    """Generate structured initial sequences."""
    x = np.arange(n) / max(n - 1, 1)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 2:
        # Asymmetric ramp with cosine taper
        a = x * (1 - x) + 0.1
        a = a ** rng.uniform(0.5, 3.0)
    elif mode == 3:
        # Gaussian bump (off-center)
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((x - center) / width) ** 2) + 0.05
    elif mode == 4:
        # Two bumps
        c1, c2 = 0.3, 0.7
        w = rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2) + 0.05
    elif mode == 5:
        # Sine-modulated
        freq = rng.uniform(1, 5)
        a = 1 + 0.4 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0)
    elif mode == 6:
        # Power law (asymmetric)
        p = rng.uniform(0.3, 2.0)
        a = (x + 0.1) ** p + 0.05
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

        alloc2 = min(remaining * 0.2, 200.0)
        print(f"\n=== Phase 2: Multi-Peak (budget={alloc2:.0f}s) ===")
        a_best = best_seq.astype(np.float64) / np.sum(best_seq)
        for K in [2, 3, 5]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_mp = multi_peak_optimize(a_best, time_budget=min(alloc2 / 3, remaining * 0.3), K=K)
            try_update(a_mp)
            if np.sum(a_mp) > 0:
                a_best = a_mp / np.sum(a_mp)

    # Phase 3: Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc3 = min(remaining * 0.25, 200.0)
        print(f"\n=== Phase 3: Subgradient (budget={alloc3:.0f}s) ===")
        a_best = best_seq.astype(np.float64) / np.sum(best_seq)
        a_sg = subgradient_optimize(a_best, time_budget=alloc3, verbose=True)
        try_update(a_sg)

    # Phase 4: Block coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc4 = min(remaining * 0.15, 100.0)
        print(f"\n=== Phase 4: Block Coordinate (budget={alloc4:.0f}s) ===")
        a_best = best_seq.astype(np.float64) / np.sum(best_seq)
        a_bc = block_coord_descent(a_best, time_budget=alloc4)
        try_update(a_bc)

    # Phase 5: Multi-n exploration
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Multi-n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(100, n_best - 200), max(100, n_best - 100),
            n_best,
            min(3000, n_best + 100), min(3000, n_best + 200),
            500, 600, 700, 800, 900, 1000, 1200, 1500, 2000
        ]))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            if n_try == n_best:
                a_try = best_seq.astype(np.float64)
            else:
                a_try = resample_sequence(best_seq.astype(np.float64), n_try)
            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try

            alloc = min(30.0, remaining * 0.1)
            a_opt = lp_alternating(a_try, time_budget=alloc)
            sc = try_update(a_opt)
            if sc is not None:
                print(f"  n={n_try}: score={sc:.8f}")

    # Phase 6: Random restarts with structured inits
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Random Restarts ===")
        n_best = len(best_seq)
        n_restarts = max(3, int(remaining / 25))
        restart_budget = min(20.0, remaining / max(n_restarts, 1))

        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < restart_budget + 5:
                break

            mode = r % 7
            n_r = n_best
            if r % 3 == 1:
                n_r = max(200, min(2000, n_best + int(rng.integers(-200, 200))))
            if r % 3 == 2:
                n_r = max(200, min(2000, int(rng.integers(400, 1500))))

            a_r = structured_init(n_r, mode, rng)
            # Mix with best
            if r % 2 == 0 and len(best_seq) == n_r:
                a_r = 0.7 * best_seq.astype(np.float64) / np.sum(best_seq) + 0.3 * a_r
                a_r = np.maximum(a_r, 0)
                a_r = a_r / np.sum(a_r)

            a_opt = lp_alternating(a_r, time_budget=restart_budget)
            try_update(a_opt)

    # Phase 7: Final intensive polish on best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 7: Final Polish (remaining={remaining:.0f}s) ===")
        a_final = best_seq.astype(np.float64) / np.sum(best_seq)

        for round_idx in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.25, 100.0)
            a_pol = lp_alternating(a_final, time_budget=alloc, verbose=(round_idx == 0))
            sc = try_update(a_pol)
            if sc is not None and sc < best_score:
                a_final = a_pol.astype(np.float64) / np.sum(a_pol)

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sg = subgradient_optimize(a_final, time_budget=min(30.0, remaining * 0.5))
            try_update(a_sg)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val >= np.inf:
        result = [1.0] * 800
        final_val = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_val:.8f}, time={elapsed:.1f}s ===")
    return result
