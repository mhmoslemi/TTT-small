```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses alternating LP optimization with asymmetric initialization,
subgradient refinement, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = fast_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    
    A = build_conv_matrix(a_norm)
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.hstack([np.ones((1, n)), np.array([[0.0]])])
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None, None


def alternating_optimize(a, time_budget=60.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_t = 0
        for t in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a = (1.0 - best_t) * best_a + best_t * g_lp
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    alt it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 10:
                # Perturbation
                rng = np.random.default_rng(it * 7919 + n)
                a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert.copy()
                stall = 0
    
    return best_a


def subgradient_optimize(a, time_budget=30.0, verbose=False):
    """Subgradient descent on the max of conv."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / n
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = fast_conv(a)
        k_star = int(np.argmax(conv))
        
        # Gradient of conv[k_star] w.r.t. a
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        for j in range(j_start, j_end + 1):
            grad[j] = 2.0 * a[k_star - j]
        
        # Projected subgradient: reduce gradient by its mean (to preserve sum)
        grad_centered = grad - np.mean(grad)
        
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 5.0 / n)
        else:
            lr *= 0.95
            if lr < 1e-6 / n:
                rng = np.random.default_rng(it * 31)
                a_pert = best_a * np.exp(rng.normal(0, 0.03, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / np.sum(a_pert)
                if compute_obj(a_pert) < best_obj:
                    best_obj = compute_obj(a_pert)
                    best_a = a_pert.copy()
                    a = a_pert
                lr = 1.0 / n
    
    return best_a


def prop_candidates(n, rng):
    """Generate diverse initial candidates for given n."""
    candidates = []
    
    # Ramp
    p = np.arange(1, n + 1, dtype=np.float64)
    candidates.append(p / p.sum())
    
    # Inverse ramp
    p = np.arange(n, 0, -1, dtype=np.float64)
    candidates.append(p / p.sum())
    
    # Exponential (various rates)
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        p = np.exp(alpha * np.arange(n) / n)
        candidates.append(p / p.sum())
    
    # Power ramp
    for power in [0.5, 1.5, 2.0, 3.0]:
        p = np.arange(1, n + 1, dtype=np.float64) ** power
        candidates.append(p / p.sum())
    
    # Half ramp up, half ramp down (asymmetric)
    half = n // 2
    p1 = np.arange(1, half + 1, dtype=np.float64)
    p2 = np.arange(int(half * 0.5), 0, -1, dtype=np.float64)
    if len(p2) < n - half:
        p2 = np.append(p2, np.ones(n - half - len(p2)) * p2[-1])
    p2 = p2[:n - half]
    p = np.concatenate([p1, p2])
    candidates.append(p / p.sum())
    
    # Sine-squared (symmetric, for comparison)
    x = np.linspace(0, np.pi, n)
    p = np.sin(x) ** 2
    p = np.maximum(p, 1e-10)
    candidates.append(p / p.sum())
    
    # Random smooth
    p = rng.uniform(0.5, 1.5, n)
    kernel_size = max(3, n // 20)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1)) ** 2 / (kernel_size / 2) ** 2)
    p = np.convolve(p, kernel / kernel.sum(), mode='same')
    p = np.maximum(p, 0)
    candidates.append(p / p.sum())
    
    # Near-constant with slight asymmetry
    p = np.ones(n) + 0.1 * np.linspace(-1, 1, n)
    p = np.maximum(p, 0)
    candidates.append(p / p.sum())
    
    return candidates
