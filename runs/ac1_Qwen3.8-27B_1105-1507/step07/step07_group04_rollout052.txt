"""
Hybrid optimizer: Log-sum-exp Smoothed Gradient Descent (with Nesterov acceleration)
+ LP Alternating Optimization + Pair-Swap Local Search + Multi-length Exploration.

Key novel contribution: Uses the softmax-weighted gradient of the log-sum-exp
smoothing of the peak convolution value, providing a globally-informed descent
direction (weighted average of ALL peak directions) rather than just top-K.
Combined with adaptive temperature annealing and Nesterov momentum for fast convergence.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx - (cssv - 1.0) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, w):
    """
    Compute the smoothed gradient of log-sum-exp(conv/T).
    grad[i] = sum_k w[k] * a[k-i] (with a[j]=0 outside [0,n-1]).
    w is the softmax vector of length 2n-1.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        length = j_end - j_start
        if length <= 0:
            continue
        # a[k-j] for j in [j_start, j_end)
        # = a[k - j_start], a[k - j_start - 1], ..., a[k - (j_end-1)]
        # = a[k+1-j_end : k+1-j_start][::-1]
        lo = k + 1 - j_end
        hi = k + 1 - j_start
        grad[j_start:j_end] += wk * a[lo:hi][::-1]
    return grad


def nesterov_smoothed_gd(a, time_budget=300.0, T_init=0.05, T_min=0.001,
                         T_decay=0.95, lr_init=0.01, verbose=False):
    """
    Nesterov-accelerated gradient descent on the log-sum-exp smoothed objective.
    a is assumed normalized (sum=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # Nesterov lookahead point
    momentum = np.zeros(n)
    beta = 0.9

    T = T_init * float(np.max(compute_conv(a)))
    lr = lr_init
    T_floor = T_min * float(np.max(compute_conv(a)))

    no_improve = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Compute smoothed gradient at lookahead point y
        conv_y = compute_conv(y)
        max_conv_y = float(np.max(conv_y))
        T = max(T_floor, T)
        # Numerically stable softmax
        shifted = conv_y / T - np.max(conv_y) / T
        exp_vals = np.exp(shifted)
        w = exp_vals / float(np.sum(exp_vals))

        grad = smoothed_gradient(y, w)
        # Center the gradient (project to sum=0 subspace) for stability
        grad_centered = grad - float(np.mean(grad))

        # Nesterov momentum update
        momentum = beta * momentum - lr * grad_centered
        a_new = a + momentum

        # Project to simplex
        a_new = project_simplex(a_new)

        # Update Nesterov weights
        t_k = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if it % 200 == 199:
            if no_improve > 100:
                lr = max(lr * 0.8, 1e-5)
                beta = max(beta * 0.99, 0.5)
            elif no_improve == 0:
                lr = min(lr * 1.05, 0.1)

        # Anneal temperature
        if it % 100 == 99:
            T = max(T_floor, T * T_decay)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSGD: it={it}, obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        end = min(m + n, 2 * n - 1)
        A[m:end, m] = a[:end - m]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M_var.value)
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=300.0, verbose=False):

    n = len(a)

        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    solver_count = 0

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            # Perturb and retry
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            a_pert = a * np.exp(rng.normal(0, 0.02, n))
            a_pert = np.maximum(a_pert, 0)
            s_pert = float(np.sum(a_pert))
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
            g_lp, M_lp = solve_lp_minpeak(a_pert)
            if g_lp is None:
                break

        solver_count += 1

        # Adaptive line search
        candidate_ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        if 0 < last_t < 1.0:
            candidate_ts = sorted(set(candidate_ts + [last_t * 0.5, last_t * 1.3]))
            candidate_ts = [t for t in candidate_ts if 0 < t <= 1.0]

        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and solver_count % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={best_t_val:.3f}, iter={solver_count}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 10:
                # Perturb to escape local minimum
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                scale = 0.02 + 0.005 * min(stall // 10, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                last_t = 0.5

    return best_a


def pair_swap_local_search(a, time_budget=100.0, verbose=False):
    """
    Local search: try moving mass between pairs of coordinates.
    Focus on coordinates that contribute most to the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    conv = compute_conv(a)
    peak_idx = int(np.argmax(conv))

    # Sort coordinates by "importance" (contribution to peak)
    importance = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            importance[i] = a[j]

    order = np.argsort(importance)[::-1]

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        improved = False
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))

        # Try moves from high-importance to low-importance coordinates
        num_tries = 200
        for _ in range(num_tries):
            if time.monotonic() - start > time_budget:
                break

            # Pick source from top half, target from bottom half
            src_pool = order[:n // 2]
            tgt_pool = order[n // 2:]

            i = int(rng.choice(src_pool))
            j = int(rng.choice(tgt_pool))
            if i == j or a[i] < 1e-10:
                continue

            for frac in [0.005, 0.01, 0.02, 0.05]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                obj_try = compute_obj(a)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    break

        if not improved and it > 50:
            break

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    PSL: it={it}, obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def subgradient_momentum(a, time_budget=100.0, top_k=10, verbose=False):
    """
    Nesterov-accelerated subgradient method targeting top-K peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.03
    x = a.copy()

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        k_indices = np.argsort(conv)[::-1][:top_k]

        # Compute subgradient (average over top-K peaks)
        grad = np.zeros(n)
        for k in k_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            lo = k + 1 - j_end
            hi = k + 1 - j_start
            grad[j_start:j_end] += x[lo:hi][::-1]
        grad /= len(k_indices)

        # Center gradient
        grad_centered = grad - float(np.mean(grad))

        # Momentum update
        momentum = beta * momentum - lr * grad_centered
        a_new = project_simplex(a + momentum)

        # Nesterov
        t_k = 2.0 / (np.sqrt(it + 1) + 1.0)
        x = project_simplex(a_new + (1.0 - t_k) * (a_new - a))

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        if it % 500 == 499:
            lr = max(lr * 0.95, 0.001)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SubGrad: it={it}, obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines smoothed GD, LP alternating,
    local search, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    a = best_seq.copy()
    n_parent = len(a)

    # === Phase 1: Nesterov Smoothed Gradient Descent (350s) ===
    print(f"\n=== Phase 1: Nesterov Smoothed GD (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    sgd_budget = min(350.0, remaining * 0.25)
    if sgd_budget > 20:
        a_sgd = nesterov_smoothed_gd(a, time_budget=sgd_budget,
                                      T_init=0.05, T_min=0.001,
                                      T_decay=0.93, lr_init=0.008,
                                      verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            a = a_sgd.copy()
            print(f"  NSGD: obj={best_score:.8f} **")
        else:
            print(f"  NSGD: obj={obj_sgd:.8f}")

    # === Phase 2: LP Alternating (round 1) (400s) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(400.0, remaining * 0.3)
    if lp_budget > 20:
        a_lp = lp_alternating(a, time_budget=lp_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  LP1: obj={best_score:.8f} **")
        else:
            print(f"  LP1: obj={obj_lp:.8f}")

    # === Phase 3: Nesterov Smoothed GD (round 2, lower T) (200s) ===
    print(f"\n=== Phase 3: NSGD Round 2 ===")
    remaining = deadline - time.monotonic()
    sgd2_budget = min(200.0, remaining * 0.15)
    if sgd2_budget > 20:
        a_sgd2 = nesterov_smoothed_gd(a, time_budget=sgd2_budget,
                                       T_init=0.02, T_min=0.0005,
                                       T_decay=0.9, lr_init=0.005,
                                       verbose=False)
        obj_sgd2 = compute_obj(a_sgd2)
        if obj_sgd2 < best_score:
            best_score = obj_sgd2
            best_seq = a_sgd2.copy()
            a = a_sgd2.copy()
            print(f"  NSGD2: obj={best_score:.8f} **")
        else:
            print(f"  NSGD2: obj={obj_sgd2:.8f}")

    # === Phase 4: LP Alternating (round 2) (250s) ===
    print(f"\n=== Phase 4: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    lp2_budget = min(250.0, remaining * 0.2)
    if lp2_budget > 20:
        a_lp2 = lp_alternating(a, time_budget=lp2_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:

    # === Phase 5: Multi-n Exploration (200s) ===

        n_parent + 200,

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    print(f"\n=== Phase 6: Pair Swap Local Search ===")
    remaining = deadline - time.monotonic()

    if ps_budget > 10:

            best_seq = a_ps.copy()
            a = a_ps.copy()
            print(f"  PSL: obj={best_score:.8f} **")
        else:
            print(f"  PSL: obj={obj_ps:.8f}")

    # === Phase 7: Subgradient Momentum (100s) ===
    print(f"\n=== Phase 7: Subgradient Momentum ===")

    print(f"\n=== Phase 8: Final LP Polish ===")

    if remaining > 20:
        a_final = lp_alternating(a, time_budget=remaining * 0.7, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 9: One more NSGD with very low T ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_last = nesterov_smoothed_gd(a, time_budget=remaining * 0.6,
                                       T_init=0.005, T_min=0.0001,
                                       T_decay=0.85, lr_init=0.003,
                                       verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final NSGD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:

    return result
