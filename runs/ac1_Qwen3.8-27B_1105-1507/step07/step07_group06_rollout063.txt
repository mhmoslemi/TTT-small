import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_from_theta(theta):
    """Compute softmax with numerical stability."""
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def lse_value(conv, tau):
    """Compute log-sum-exp smoothed max."""
    m = np.max(conv)
    return m + np.log(np.sum(np.exp((conv - m) * tau))) / tau


def smooth_obj_and_grad_theta(theta, tau):
    """
    Compute smoothed objective and gradient w.r.t. log-parameters theta.
    a = softmax(theta), obj = 2n * LSE_tau(conv(a))
    """
    n = len(theta)
    a = softmax_from_theta(theta)
    conv = compute_conv(a)
    
    m = np.max(conv)
    obj_val = 2.0 * n * (m + np.log(np.sum(np.exp((conv - m) * tau))) / tau)
    
    # LSE gradient weights
    exp_t = np.exp((conv - m) * tau)
    w = exp_t / np.sum(exp_t)
    
    # g_a[i] = sum_k w[k] * a[k-i] (cross-correlation of w and a, first n values)
    N = 2 * n
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    g_a = np.fft.ifft(W * np.conj(A)).real[:n]
    
    # Gradient w.r.t. theta: df/dtheta_j = 2n * a_j * (g_a_j - sum_i a_i * g_a_i)
    mean_g = np.dot(a, g_a)
    grad_theta = 2.0 * n * a * (g_a - mean_g)
    
    return obj_val, grad_theta


def nesterov_smoothed(theta_init, n, tau_schedule, step_size_init, momentum, time_budget, max_iter=500):
    """
    Nesterov-accelerated gradient descent on the smoothed objective.
    theta_schedule: list of (tau, n_iters) tuples
    """
    best_a = softmax_from_theta(theta_init)
    best_obj = compute_obj(best_a)
    theta = theta_init.copy()
    v = np.zeros_like(theta)
    start = time.monotonic()
    
    for tau, n_iters in tau_schedule:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break
        
        step_size = step_size_init / tau  # step size scales with 1/tau
        
        for it in range(n_iters):
            if time.monotonic() - start > time_budget:
                break
            
            # Nesterov step
            theta_y = theta + momentum * v
            obj_val, grad = smooth_obj_and_grad_theta(theta_y, tau)
            
            # Update velocity and position
            v = momentum * v - step_size * grad
            theta_new = theta_y + v
            
            # Check improvement
            a_new = softmax_from_theta(theta_new)
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            
            theta = theta_new
        
        # Reset momentum at stage transitions
        v = np.zeros_like(theta)
    
    return best_a, best_obj

        if time.monotonic() - start > time_budget:

import numpy as np

from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_from_theta(theta):
    theta_shift = theta - np.max(theta)
    exp_t = np.exp(theta_shift)
    return exp_t / np.sum(exp_t)


def smooth_obj_grad(theta, tau):
    n = len(theta)
    a = softmax_from_theta(theta)
    conv = compute_conv(a)
    m = np.max(conv)
    obj_val = 2.0 * n * (m + np.log(np.sum(np.exp((conv - m) * tau))) / tau)
    exp_t = np.exp((conv - m) * tau)
    w = exp_t / np.sum(exp_t)
    N = 2 * n
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    g_a = np.fft.ifft(W * np.conj(A)).real[:n]
    mean_g = float(np.dot(a, g_a))
    grad = 2.0 * n * a * (g_a - mean_g)
    return obj_val, grad

```python
def adam_optimize(a_init, time_budget, tau_stages, lr=0.01, beta1=0.9, beta2=0.999):
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / np.sum(a_init)
    theta = np.log(a_init)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    best_a = a_init.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    
    for tau, n_iters in tau_stages:
        for it in range(n_iters):
            if time.monotonic() - start > time_budget:
                break
            
            t += 1
            obj_val, grad = smooth_obj_grad(theta, tau)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            theta = theta - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            a_new = softmax_from_theta(theta)
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        
        # Reset Adam state at stage transitions
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
    
    return best_a, best_obj
```

Adam is stateless (well, it has momentum and variance estimates) and adapts per-coordinate. The learning rate `lr` should be on the order of 0.01-0.1 for theta-space.
