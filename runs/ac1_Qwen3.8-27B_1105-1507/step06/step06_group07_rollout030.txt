"""
Improved optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses L-BFGS on log-sum-exp smoothed objective in log-space, combined with
LP refinement, subgradient polishing, and multi-n exploration.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, beta):
    """
    Compute log-sum-exp smoothed objective and gradient.
    a is assumed to be non-negative with sum(a) = 1.
    Returns (obj, grad) where grad = d(obj)/d(a).
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # Compute gradient using FFT cross-correlation
    size = 1
    while size < 2 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    # cc[m] = sum_j w[j] * a[j+m] (circular, with zero padding)
    cc = irfft(np.conj(W) * A, size)
    
    # d(ls)/d(a[i]) = 2 * cc[i] / beta
    grad_ls = 2.0 * cc[:n] / beta
    
    # d(obj)/d(a[i]) = (2n/sum_a^2) * (grad_ls[i] - ls/sum_a)
    grad = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad


def log_space_lbfgs(x_init, beta, time_budget=30.0, max_iter=500):
    """
    L-BFGS optimization in log-space.
    x is log(a), a = exp(x - max(x)) / sum(exp(x - max(x)))
    """
    n = len(x_init)
    x = x_init.copy()
    
    def objective(x):
        xc = x - np.max(x)
        a = np.exp(xc)
        a = a / np.sum(a)
        obj, _ = smooth_obj_and_grad(a, beta)
        return obj
    
    def grad(x):
        xc = x - np.max(x)
        a = np.exp(xc)
        a = a / np.sum(a)
        _, grad_a = smooth_obj_and_grad(a, beta)
        # Chain rule: d/dx of f(a(x)) where a = softmax(x)
        # a[i] = exp(x[i]) / sum(exp(x[j]))
        # d(a[i])/d(x[j]) = a[i] * (delta(i,j) - a[j])
        # d f / d x[j] = sum_i (df/da[i]) * d(a[i])/d(x[j])
        # = sum_i grad_a[i] * a[i] * (delta(i,j) - a[j])
        # = grad_a[j] * a[j] - a[j] * sum_i grad_a[i] * a[i]
        ga = grad_a * a
        return ga - np.sum(ga) * a
    
    start = time.monotonic()
    # Use a wrapper to enforce time limit
    try:
        res = minimize(objective, x, jac=grad, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': 10000,
                                'ftol': 1e-15, 'gtol': 1e-10})
    except Exception:
        return x
    
    return res.x

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def resample_sequence(a, new_n):

    x_old = np.linspace(0, 1, n)

    return np.interp(x_new, x_old, a)
