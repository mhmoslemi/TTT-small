Optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key insight: For non-negative sequences, max(conv(a,a)) is always at lag 0 by Cauchy-Schwarz,
so J = 2n*sum(a_i^2)/(sum(a_i))^2 >= 2, with equality at uniform.
We verify this computationally and try local refinements in case of edge cases.

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_local(a):
    """Local computation of J for comparison."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_descent_fast(a, max_sweeps=500, deadline=None, seed=42):
    """Fast coordinate descent with ratio-based candidates."""
    n = len(a)

    else:

        s = 1.0

    best_obj = compute_J_local(a)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

            break

        order = rng.permutation(n)

        for i in order:

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def lp_best_response(a, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= conv(a,a).max(), g >= 0."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    conv = np.convolve(a, a)
    M = conv.max()

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                    method='highs', options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None
