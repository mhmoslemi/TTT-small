def fast_conv(a):

    while size < 2 * n - 1:
        size <<= 1

    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(n, a, active_K):
    """Build sparse matrix for LP using only active constraints at indices active_K."""
    rows = []
    cols = []
    vals = []
    row_idx = 0
    for k in active_K:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(a[k - j])
        row_idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(len(active_K), n))

def lp_solve(n, a, M, active_K=None):
    """Solve the LP: max sum(g) s.t. (a*g)(k) <= M for k in active_K, g >= 0"""
    if active_K is None:
        active_K = list(range(2 * n - 1))
    A = build_lp_matrix(n, a, active_K)
    c = -np.ones(n)
    b_ub = np.full(len(active_K), M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None and r.x.sum() > 1e-15:
        return np.maximum(r.x, 0)
    return None

    """LP with constraint generation."""

    if init_constraints is None:

    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-15:
            break
        a_new = a_new / sc
        M_c = fast_conv(a_new).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_new.copy()
        a = a_new
    return best_a


def smooth_grad(a, tau):
    """Compute gradient of smooth max approximation."""
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(len(a))
    active = np.where(p > 1e-10)[0]
    n = len(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad


def run_smooth_nesterov(a, deadline, max_iters=3000, tau=50.0, lr=0.3):
    """Nesterov-accelerated gradient descent on smooth max approximation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        cur_tau = tau * (1 + 5 * frac)
        cur_lr = lr * (1 - 0.8 * frac)
        grad = smooth_grad(y, cur_tau)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - cur_lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M:
            best_M = M
            best_a = a.copy()
    return best_a


def lp_solve_subset(n, a, M, active_K):
    """Solve LP with only constraints at indices in active_K."""
    if len(active_K) == 0:
        return np.ones(n) / n
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_K):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_K), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_K), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_lazy(n, a, M, max_rounds=30, init_active=None):
    """LP with lazy constraint generation."""
    b = fast_conv(a)
    if init_active is not None and len(init_active) > 0:
        active = list(init_active)
    else:
        n_init = min(30, 2 * n - 1)
        active = [int(k) for k in np.argsort(b)[-n_init:]]

    g_best = None
    for rnd in range(max_rounds):
        g = lp_solve_subset(n, a, M, active)
        if g is None:
            if g_best is not None:
                return g_best
            return None
        ag = np.convolve(a, g)
        viol_threshold = M * (1 + 1e-8)
        violated = np.where(ag > viol_threshold)[0]
        if len(violated) == 0:
            return g
        # Add the most violated constraints
        sorted_viol = violated[np.argsort(ag[violated])]
        to_add = sorted_viol[-10:]
        for k in to_add:
            k = int(k)
            if k not in active:
                active.append(k)
        if len(active) > 300:
            # Thin out: keep most violated + original top
            ag_all = ag.copy()
            active_set = set(active)
            active = [int(k) for k in np.argsort(ag_all)[-min(200, 2*n-1):]]
        g_best = g
    return g_best


def anderson_mix(hist_x, hist_Fx, m):
    """Anderson mixing: find optimal convex combination of recent iterates."""
    if m < 2:
        return hist_Fx[-1].copy()
    residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(hist_Fx[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0)


def run_lp_mirror(a, deadline, max_iters=500, t_start=0.4, t_end=0.01,
                  anderson_win=5, perturb_every=40, perturb_mag=0.015,
                  n_constraints=50):
    """LP mirror descent with lazy constraints, Anderson acceleration, perturbation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0
    active_cache = None
    hist_x = []
    hist_Fx = []
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break

        # Solve LP with lazy constraints
        g = lp_lazy(n, a, M, max_rounds=20, init_active=active_cache)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Update active cache
        ag = np.convolve(a, g)
        active_cache = [int(k) for k in np.argsort(ag)[-n_constraints:]]

        # Step size with slow decay
        frac = it / max(max_iters - 1, 1)
        t = t_start * (t_end / t_start) ** (frac ** 0.7)
        t = max(t, 0.005)

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_win:
            hist_x.pop(0)
            hist_Fx.pop(0)

        if len(hist_x) >= 3:
            a_anderson = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_anderson.sum()
            if sa > 1e-10:
                a_anderson = a_anderson / sa
                M_and = float(fast_conv(a_anderson).max())
                M_new = float(fast_conv(a_new).max())
                if M_and <= M_new * 1.02:
                    a_new = a_anderson

        # Perturbation to escape local minima
        if it > 0 and it % perturb_every == 0 and no_improve > 15:
            eps = rng.uniform(-perturb_mag, perturb_mag, n)
            a_new = np.maximum(a_new + eps, 0)
            a_new = a_new / a_new.sum()

        a = a_new

    return best_a


def run_full_lp(a, deadline, max_iters=100, t=0.2):
    """Fallback: full LP mirror descent (all constraints)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

        # Full LP
        g = lp_solve_subset(n, a, M, list(range(2 * n - 1)))
        if g is None:
            break
        g = g / g.sum()
        a_new = np.maximum((1 - t) * a + t * g, 0)
        a_new = a_new / a_new.sum()
        a = a_new

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    t0 = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a.copy()
            el = time.monotonic() - t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        try_update(parent)

    n_parent = parent.size
    print(f"Starting optimization, budget={budget_s-30:.0f}s", flush=True)

    # === PHASE 1: GS initialization + smooth optimization (120s) ===
    print(f"\n--- Phase 1: Initialization ({time.monotonic()-t0:.0f}s) ---", flush=True)
    p1_deadline = time.monotonic() + 80

    # GS from parent
    a_gs = gs_flat(parent, n_iters=300, beta=0.85)
    try_update(a_gs)

    # Smooth optimization from GS result
    a_smooth = run_smooth_nesterov(a_gs, p1_deadline, max_iters=3000, tau=20, lr=0.4)
    try_update(a_smooth)

    # Also try smooth from parent directly
    a_smooth2 = run_smooth_nesterov(parent, min(time.monotonic() + 30, p1_deadline),
                                    max_iters=1000, tau=10, lr=0.3)
    try_update(a_smooth2)

    # Use best as working sequence
    work = best_seq.copy() if best_seq is not None else a_gs

    # === PHASE 2: LP mirror descent on best length (500s) ===
    print(f"\n--- Phase 2: LP Mirror Descent ({time.monotonic()-t0:.0f}s) ---", flush=True)
    p2_deadline = time.monotonic() + 450

    # First pass: aggressive
    a_lp1 = run_lp_mirror(work, min(time.monotonic() + 250, p2_deadline),
                          max_iters=300, t_start=0.45, t_end=0.02,
                          anderson_win=5, perturb_every=35, perturb_mag=0.02,
                          n_constraints=60)
    try_update(a_lp1)

    # Second pass: conservative (from best)
    if best_seq is not None and time.monotonic() < p2_deadline:
        a_lp2 = run_lp_mirror(best_seq, p2_deadline,
                              max_iters=200, t_start=0.25, t_end=0.01,
                              anderson_win=6, perturb_every=50, perturb_mag=0.01,
                              n_constraints=80)
        try_update(a_lp2)

    # === PHASE 3: Multi-length exploration (350s) ===
    print(f"\n--- Phase 3: Multi-length ({time.monotonic()-t0:.0f}s) ---", flush=True)
    p3_deadline = time.monotonic() + 300

    # Try lengths around the best
    n_best = best_seq.size if best_seq is not None else n_parent
    lengths_to_try = []
    for delta in [-96, -64, -32, 32, 64, 96, 128, 192, 256]:
        n_try = n_best + delta
        if 256 <= n_try <= 2560:
            lengths_to_try.append(n_try)
    # Always include some standard lengths
    for n_std in [768, 1024, 1280, 1536]:
        if n_std not in lengths_to_try:
            lengths_to_try.append(n_std)
    lengths_to_try = sorted(set(lengths_to_try))

    time_per_len = max(20, 300.0 / len(lengths_to_try))

    for n_try in lengths_to_try:
        if time.monotonic() > p3_deadline:
            break
        if best_seq is not None and n_try == best_seq.size:
            continue

        # Resample from best
        a_res = resample_seq(best_seq, n_try)
        a_res = np.maximum(a_res, 1e-6)
        a_res = a_res / a_res.sum()

        # Quick smooth
        sub_dl = min(time.monotonic() + time_per_len * 0.4, p3_deadline)
        a_sm = run_smooth_nesterov(a_res, sub_dl, max_iters=500, tau=30, lr=0.25)
        try_update(a_sm)

        # LP on this length
        if time.monotonic() < p3_deadline - 10:
            lp_dl = min(time.monotonic() + time_per_len * 0.6, p3_deadline)
            a_l = run_lp_mirror(a_sm, lp_dl, max_iters=100,
                                t_start=0.35, t_end=0.02,
                                anderson_win=4, perturb_every=30,
                                perturb_mag=0.015, n_constraints=50)
            try_update(a_l)

    # === PHASE 4: Perturbation + restart (200s) ===
    print(f"\n--- Phase 4: Perturbation ({time.monotonic()-t0:.0f}s) ---", flush=True)
    p4_deadline = time.monotonic() + 180

    if best_seq is not None:
        n_cur = best_seq.size
        # Several perturbation attempts
        for attempt in range(6):
            if time.monotonic() > p4_deadline:
                break
            if time.monotonic() > deadline:
                break
            # Different perturbation strategies
            if attempt == 0:
                # Small uniform noise
                a_p = best_seq + rng.uniform(-0.02, 0.02, n_cur)
            elif attempt == 1:
                # Gaussian noise
                a_p = best_seq + rng.normal(0, 0.015, n_cur)
            elif attempt == 2:
                # Multiplicative noise
                a_p = best_seq * (1 + rng.normal(0, 0.03, n_cur))
            elif attempt == 3:
                # Smooth random perturbation
                k = max(3, n_cur // 30)
                kern = np.ones(k) / k
                noise = rng.uniform(-1, 1, n_cur)
                a_p = best_seq + 0.02 * np.convolve(noise, kern, 'same')
            elif attempt == 4:
                # GS perturbation
                a_p = gs_flat(best_seq + rng.normal(0, 0.03, n_cur), n_iters=100, beta=0.7)
            else:
                # Freq domain perturbation
                size = 1
                while size < 2 * n_cur - 1:
                    size <<= 1
                F = np.fft.rfft(best_seq, size)
                F += rng.normal(0, 0.1, F.shape) * np.abs(F)
                a_p = np.fft.irfft(F, size)[:n_cur]

            a_p = np.maximum(a_p, 0)
            sp = a_p.sum()
            if sp < 1e-10:
                continue
            a_p = a_p / sp

            # Quick LP from perturbed start
            sub_dl = min(time.monotonic() + 25, p4_deadline)
            a_l = run_lp_mirror(a_p, sub_dl, max_iters=80,
                                t_start=0.3, t_end=0.02,
                                anderson_win=4, perturb_every=40,
                                perturb_mag=0.01, n_constraints=50)
            try_update(a_l)
            try_update(a_p)

    # === PHASE 5: Final polish (remaining time) ===
    print(f"\n--- Phase 5: Final Polish ({time.monotonic()-t0:.0f}s) ---", flush=True)

    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very careful LP with small steps
        a_fin = run_lp_mirror(best_seq, deadline - 2, max_iters=200,
                              t_start=0.15, t_end=0.005,
                              anderson_win=6, perturb_every=100,
                              perturb_mag=0.005, n_constraints=100)
        try_update(a_fin)

        # One more Anderson pass
        if time.monotonic() < deadline - 3:
            a_fin2 = run_lp_mirror(a_fin, deadline - 1, max_iters=50,
                                   t_start=0.1, t_end=0.003,
                                   anderson_win=5, perturb_every=200,
                                   perturb_mag=0.003, n_constraints=80)
            try_update(a_fin2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = compute_J(best_seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best_seq]
