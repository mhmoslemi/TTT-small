    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Pick a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        ai = a[i]
        aj = a[j]
        if ai < 1e-14 and aj < 1e-14:
            continue

        # 2D search: vary a[i] and a[j]
        # Use a small grid + refinement
        best_2d_obj = 2.0 * n * conv.max() / (s * s)
        best_xi = ai
        best_xj = aj

        # Coarse grid
        ri = [0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5] if ai > 1e-14 else [0.0, 1e-4, 1e-3, 1e-2]
        rj = [0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5] if aj > 1e-14 else [0.0, 1e-4, 1e-3, 1e-2]

        for xi_r in ri:
            xi = xi_r * ai
            for xj_r in rj:
                xj = xj_r * aj
                xi = max(0.0, xi)
                xj = max(0.0, xj)

                di = xi - ai
                dj = xj - aj
                if abs(di) < 1e-16 and abs(dj) < 1e-16:
                    continue

                # Update conv for both i and j
                # i affects conv[i..i+n-1]
                lo_i, hi_i = i, min(2 * n - 2, i + n - 1)
                lo_j, hi_j = j, min(2 * n - 2, j + n - 1)

                original_lo_i = conv[lo_i:hi_i + 1].copy() if False else None
                # Just do it directly and undo
                conv[lo_i:hi_i + 1] += 2.0 * di * a[lo_i - i:hi_i - i + 1]
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += di * di
                conv[lo_j:hi_j + 1] += 2.0 * dj * a[lo_j - j:hi_j - j + 1]
                if 2 * j <= 2 * n - 2:
                    conv[2 * j] += dj * dj
                # Correction for overlap: if i and j are close, the cross term is double-counted
                # Actually, the conv update for i uses a[j] which hasn't changed yet in our simulation
                # This is an approximation - for exactness we'd need to handle cross terms
                # For now, this is a heuristic approximation

                new_s = s - ai - aj + xi + xj
                if new_s > 1e-10:
                    f_val = 2.0 * n * conv.max() / (new_s * new_s)
                else:
                    f_val = float('inf')

                if f_val < best_2d_obj:
                    best_2d_obj = f_val
                    best_xi = xi
                    best_xj = xj

                # Undo
                conv[lo_i:hi_i + 1] -= 2.0 * di * a[lo_i - i:hi_i - i + 1]
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] -= di * di
                conv[lo_j:hi_j + 1] -= 2.0 * dj * a[lo_j - j:hi_j - j + 1]
                if 2 * j <= 2 * n - 2:
                    conv[2 * j] -= dj * dj

        # Apply best pair update
        if abs(best_xi - ai) > 1e-14 or abs(best_xj - aj) > 1e-14:
            di = best_xi - ai
            dj = best_xj - aj
            lo_i, hi_i = i, min(2 * n - 2, i + n - 1)
            lo_j, hi_j = j, min(2 * n - 2, j + n - 1)
            a_i_copy = a[lo_i - i:hi_i - i + 1].copy()
            a_j_copy = a[lo_j - j:hi_j - j + 1].copy()
            conv[lo_i:hi_i + 1] += 2.0 * di * a_i_copy
            if 2 * i <= 2 * n - 2:
                conv[2 * i] += di * di
            conv[lo_j:hi_j + 1] += 2.0 * dj * a_j_copy
            if 2 * j <= 2 * n - 2:
                conv[2 * j] += dj * dj
            a[i] = best_xi
            a[j] = best_xj
            s = s - ai - aj + best_xi + best_xj

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and rnd % 50 == 0:
            print(f"    Pairwise round {rnd}: obj={best_obj:.10f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def active_set_lp(a, deadline, top_k=15, max_iters=50, verbose=False):
    """
    LP-based step using only the top-K active constraints.
    Solves: max sum(g) s.t. (a*g)[k] <= conv_a[k] for top-K k, g >= 0.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv_a = fast_conv(a)
        # Get top K indices
        top_idx = np.argsort(conv_a)[::-1][:top_k]
        bounds_k = conv_a[top_idx]

        # Build LP: max sum(g) s.t. (a*g)[k] <= bound_k for k in top_idx, g >= 0
        rows = []
        cols = []
        vals = []
        for ci, k in enumerate(top_idx):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(ci)
                cols.append(j)
                vals.append(a[k - j])

        A = coo_matrix((vals, (rows, cols)), shape=(len(top_idx), n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = bounds_k.copy()
        bl = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bl, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Try different blend fractions
                    found = False
                    for frac in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-14:
                            a = a_try
                            best_obj = J_try
                            best_a = a.try_copy() if hasattr(a, 'try_copy') else a.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 5 == 0:
            print(f"    LP it {it}: obj={best_obj:.10f}", flush=True)

        if no_improve > 15:
            break

    return best_a, best_obj


def parametric_search(n, deadline, verbose=False):
    """
    Search over parametric families to find good initial structures.
    a[i] = exp(theta dot features(i)) normalized.
    """
    best_a = np.ones(n) / n
    best_obj = compute_J(best_a)
    t = np.linspace(0, 1, n, endpoint=False)
    phase = 2 * np.pi * t

    # Feature functions
    def make_features():
        return [
            np.ones(n),
            np.cos(phase), np.sin(phase),
            np.cos(2 * phase), np.sin(2 * phase),
            np.cos(3 * phase), np.sin(3 * phase),
            (t - 0.5) ** 2,
            (t - 0.5) ** 3,
            np.cos(4 * phase), np.sin(4 * phase),
        ]

    features = make_features()
    n_feat = len(features)
    rng = np.random.default_rng(123)

    # Random search in parameter space
    best_theta = np.zeros(n_feat)

    for it in range(2000):
        if time.monotonic() > deadline:
            break

        if it < 100:
            # Explore with large steps
            theta = rng.normal(0, 0.5, n_feat)
        else:
            # Exploit around best
            theta = best_theta + rng.normal(0, 0.1, n_feat)

        # Compute a from theta
        log_a = np.zeros(n)
        for f_idx in range(n_feat):
            log_a += theta[f_idx] * features[f_idx]

        # Shift to avoid overflow
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            continue
        a /= a.sum()

        obj = compute_J(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_theta = theta.copy()
            if verbose:
                print(f"    Param it {it}: J={obj:.10f}", flush=True)

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initialization candidates."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # From parent
    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            inits.append(p.copy())
            inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
            inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
            inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine modulated
    for freq in [1, 2, 3, 4]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            for phase_offset in [0, 0.5]:
                sig = 1.0 + amp * np.cos(2 * np.pi * freq * t + np.pi * phase_offset)
                a = project_simplex(sig)
                inits.append(a)

    # Beta-like distributions
    for alpha in [0.3, 0.5, 0.8, 1.5, 2.0, 3.0]:
        for beta_param in [0.3, 0.5, 0.8, 1.5, 2.0, 3.0]:
            if abs(alpha - beta_param) < 0.1:
                continue
            tt = t + 0.01
            sig = tt ** alpha * (1 - tt) ** beta_param
            a = project_simplex(sig)
            inits.append(a)

    # Smooth random
    ksize = max(3, n // 8)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) * 2.0 / ksize) ** 2)
    kernel /= kernel.sum()
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        a = project_simplex(smooth + 0.01)
        inits.append(a)

    # "Anti-triangle" - U-shaped
    u_shape = 1.0 / (0.1 + np.abs(t - 0.5) ** 0.5)
    inits.append(project_simplex(u_shape))

    # Multi-bump
    for nb in [2, 3, 4]:
        centers = rng.uniform(0.1, 0.9, nb)
        sig = np.zeros(n)
        for c in centers:
            sig += np.exp(-((t - c) / 0.15) ** 2)
        inits.append(project_simplex(sig + 0.01))

    return inits


def solve_pipeline(a, deadline, verbose=False):
    """
    Full optimization pipeline: CD -> Pairwise -> LP -> CD polish.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a /= a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            if verbose:
                print(f"      Updated: J={obj:.10f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 1: CD (50% of time)
    t1 = time.monotonic() + 0.50 * remaining
    if verbose:
        print(f"    [CD] budget={t1 - time.monotonic():.0f}s", flush=True)
    a_cd = cd_with_linesearch(a, t1, random_order=True, seed=42)
    try_update(a_cd)
    a_curr = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 2: Pairwise (20% of time)
    t2 = time.monotonic() + 0.20 * remaining
    if verbose:
        print(f"    [Pairwise] budget={t2 - time.monotonic():.0f}s", flush=True)
    a_pw = pairwise_cd(a_curr, t2, seed=77, max_rounds=3000)
    try_update(a_pw)
    a_curr = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 3: Active-set LP (15% of time)
    t3 = time.monotonic() + 0.15 * remaining
    if verbose:
        print(f"    [LP] budget={t3 - time.monotonic():.0f}s", flush=True)
    a_lp, obj_lp = active_set_lp(a_curr, t3, top_k=20, max_iters=100)
    try_update(a_lp)
    a_curr = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 4: Final CD polish (15% of time)
    t4 = time.monotonic() + 0.15 * remaining
    if verbose:
        print(f"    [Final CD] budget={t4 - time.monotonic():.0f}s", flush=True)
    a_final = cd_with_linesearch(a_curr, t4, random_order=True, seed=999)
    try_update(a_final)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Multi-phase approach:
    1. Load parent, parametric search for good structure
    2. Main pipeline on best initial
    3. Multi-start with diverse inits
    4. Final intensive polish on global best
    5. Try different lengths
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0  # safety margin

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_base = parent.size

    # === PHASE 1: Parametric search + diverse init selection ===
    phase1_deadline = t_start + 120.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Initialization (deadline at {phase1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Parametric search at base length
    t_param = min(phase1_deadline - 5, t_start + 60.0)
    print(f"  Parametric search at n={n_base}...", flush=True)
    a_param, obj_param = parametric_search(n_base, t_param, verbose=True)
    update_best(a_param)

    # Generate diverse inits and pick top few by quick CD
    inits = generate_diverse_inits(n_base, parent, rng)
    print(f"  Generated {len(inits)} diverse inits", flush=True)

    # Quick evaluation of all inits
    init_scores = []
    for idx, ini in enumerate(inits):
        obj = compute_J(ini)
        init_scores.append((obj, idx, ini))
    init_scores.sort(key=lambda x: x[0])

    # Take top 5 diverse inits (avoid near-duplicates)
    top_inits = []
    for obj, idx, ini in init_scores:
        if len(top_inits) >= 5:
            break
        # Check diversity from already selected
        diverse = True
        for _, _, sel_ini in top_inits:
            if np.linalg.norm(ini - sel_ini) / max(1, np.linalg.norm(ini)) < 0.1:
                diverse = False
                break
        if diverse:
            top_inits.append((obj, idx, ini))

    print(f"  Selected {len(top_inits)} diverse inits for optimization", flush=True)

    remaining_p1 = phase1_deadline - time.monotonic()
    if remaining_p1 > 20:
        # Quick CD on each top init
        per_init = max(10, remaining_p1 / max(1, len(top_inits)))
        for obj, idx, ini in top_inits:
            if time.monotonic() > phase1_deadline - 5:
                break
            a_quick = cd_with_linesearch(ini, time.monotonic() + per_init * 0.5,
                                         random_order=True, seed=idx + 100)
            update_best(a_quick)

    # === PHASE 2: Main pipeline on best ===
    phase2_deadline = t_start + 800.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Main pipeline (deadline at {phase2_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        t_main = min(phase2_deadline - 5, time.monotonic() + 0.7 * (phase2_deadline - time.monotonic()))
        print(f"  Running solve_pipeline on best (n={best_seq.size}, budget={t_main - time.monotonic():.0f}s)", flush=True)
        a_opt, obj_opt = solve_pipeline(best_seq, t_main, verbose=True)
        update_best(a_opt)
        print(f"  Pipeline result: J={obj_opt:.10f}", flush=True)

    # === PHASE 3: Multi-start optimization ===
    phase3_deadline = t_start + 1150.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-start (deadline at {phase3_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try a few of the diverse inits with full pipeline
    remaining_p3 = phase3_deadline - time.monotonic()
    if remaining_p3 > 60:
        n_starts = min(4, len(top_inits))
        per_start = max(20, remaining_p3 / max(1, n_starts))
        for s_idx in range(n_starts):
            if time.monotonic() > phase3_deadline - 10:
                break
            obj, idx, ini = top_inits[s_idx]
            alloc = min(per_start, phase3_deadline - time.monotonic() - 10)
            if alloc < 15:
                break
            print(f"  Multi-start {s_idx} (init score={obj:.6f}, budget={alloc:.0f}s)...", flush=True)
            a_opt3, obj_opt3 = solve_pipeline(ini, time.monotonic() + alloc, verbose=False)
            update_best(a_opt3)
            print(f"  Multi-start {s_idx} result: J={obj_opt3:.10f}", flush=True)

    # === PHASE 4: Try different lengths ===
    phase4_deadline = t_start + 1300.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-length (deadline at {phase4_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        candidate_ns = [512, 640, 768, 896, 1024, 1280]
        candidate_ns = [nn for nn in candidate_ns if nn != n_base and nn <= 2048]
        candidate_ns.sort(key=lambda nn: abs(nn - n_base))

        for new_n in candidate_ns[:4]:
            if time.monotonic() > phase4_deadline - 30:
                break
            alloc = min(60.0, phase4_deadline - time.monotonic() - 20)
            if alloc < 20:
                break
            print(f"  Trying n={new_n} (budget={alloc:.0f}s)...", flush=True)
            a_ext = resample_seq(best_seq, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()
            a_opt4, obj_opt4 = solve_pipeline(a_ext, time.monotonic() + alloc, verbose=False)
            update_best(a_opt4)
            print(f"  n={new_n}: J={obj_opt4:.10f}", flush=True)

    # === PHASE 5: Final intensive polish ===
    print(f"\n{'=' * 60}", flush=True)
    remaining_final = deadline - time.monotonic()
    print(f"PHASE 5: Final polish (budget={remaining_final:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining_final > 30:
        a_curr = best_seq.copy()

        # Long CD
        t_cd = time.monotonic() + 0.4 * remaining_final
        print(f"  Final CD (budget={t_cd - time.monotonic():.0f}s)...", flush=True)
        a_polished = cd_with_linesearch(a_curr, t_cd, random_order=True, seed=2024)
        update_best(a_polished)

        remaining = deadline - time.monotonic()
        if remaining > 20:
            # LP
            t_lp = time.monotonic() + 0.3 * remaining
            print(f"  Final LP (budget={t_lp - time.monotonic():.0f}s)...", flush=True)
            a_lp, obj_lp = active_set_lp(best_seq, t_lp, top_k=25, max_iters=200)
            update_best(a_lp)

        remaining = deadline - time.monotonic()
        if remaining > 15:
            # One more CD on the LP result
            a_final2 = cd_with_linesearch(best_seq, deadline - 3, random_order=True, seed=31415)
            update_best(a_final2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.10f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
