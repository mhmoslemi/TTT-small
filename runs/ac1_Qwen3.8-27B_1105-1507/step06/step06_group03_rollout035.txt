def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_safe(x, tau=1.0):
    x = tau * (x - x.max())
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_grad_fft(a, w):
    """Compute grad[i] = 2 * sum_j w[i+j] * a[j] using FFT cross-correlation."""
    n = len(a)
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    corr = np.fft.irfft(A * np.conj(W), size)
    c = np.empty(n)
    c[0] = corr[0]
    if n > 1:
        c[1] = corr[size - 1]
        if n > 2:
            c[2:] = corr[size - 2 : size - n : -1]
    return 2.0 * c


def compute_grad_active(a, conv, threshold=0.0):
    """Gradient using active constraints (conv[k] >= max - threshold)."""
    n = len(a)
    M = conv.max()
    active = np.where(conv >= M - threshold)[0]
    grad = np.zeros(n)
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            i = k - j
            if 0 <= i < n:
                grad[i] += 2.0 * a[j]
    return grad

def smooth_grad_optimize(a, max_iters, deadline, tau_init=20.0, tau_max=300.0,
                         lr_init=0.1, momentum=0.9, verbose=False):
    """Smoothed gradient descent with Nesterov momentum and adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_obj = compute_J(a)

    v = np.zeros(n)  # velocity (momentum)
    m = np.zeros(n)  # first moment
    v2 = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_max - tau_init) * frac * frac  # quadratic annealing

        conv = fast_conv(a)
        w = softmax_safe(conv, tau)
        grad = compute_grad_fft(a, w)

        # Subtract mean for scale invariance (J is scale-invariant, so uniform shift doesn't matter)
        grad = grad - grad.mean()

        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)

        m = beta1 * m + (1 - beta1) * grad
        v2 = beta2 * v2 + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v2_hat = v2 / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v2_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart from best with perturbation
                rng_restart = np.random.default_rng(it)
                a = best_a + rng_restart.normal(0, 0.01 / n ** 0.5, n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                m = np.zeros(n)
                v2 = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

        if verbose and it % 500 == 0:
            print(f"    Grad it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def subgradient_multi_active(a, max_iters, deadline, eps_frac=0.05, lr_init=0.05,
                             verbose=False):
    """Subgradient method using multiple active constraints."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    v = np.zeros(n)
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()
        threshold = M * (1 - eps_frac)
        active = np.where(conv >= threshold)[0]

        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient: average of gradients over active constraints
        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                i = k - j
                if 0 <= i < n:
                    grad[i] += 2.0 * a[j]
        grad = grad / len(active)
        grad = grad - grad.mean()  # scale invariance

        lr = lr_init / (1.0 + 0.1 * t_count ** 0.5)
        t_count += 1

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                a = best_a + np.random.default_rng(it).normal(0, 0.005 / n ** 0.5, n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                t_count = 0
                no_improve = 0

        a = a_new
        if verbose and it % 100 == 0:
            print(f"    Subgrad it {it}: obj={best_obj:.8f} n_active={len(active)}", flush=True)

    return best_a, best_obj


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = parent.copy()
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.01, 0.02, 0.05, 0.1]:
            inits.append(np.maximum(p + rng.normal(0, sigma / n ** 0.5, n), 0) / n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            base = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            base = np.maximum(base, 0.01)
            inits.append(base / base.sum())
            base2 = 1.0 + amp * np.sin(2 * np.pi * freq * t)
            base2 = np.maximum(base2, 0.01)
            inits.append(base2 / base2.sum())

    # Exponential tilts
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        for direction in [1, -1]:
            base = np.exp(direction * alpha * (t - 0.5))
            inits.append(base / base.sum())

    # Bimodal
    for center1, center2 in [(0.25, 0.75), (0.2, 0.8), (0.3, 0.7)]:
        for width in [0.05, 0.1, 0.15]:
            base = np.exp(-((t - center1) / width) ** 2) + np.exp(-((t - center2) / width) ** 2) + 0.01
            inits.append(base / base.sum())

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        smooth = np.maximum(smooth, 0.01)
        inits.append(smooth / smooth.sum())

    # Power law
    for alpha in [-0.5, -1.0, 0.5, 1.0]:
        base = (t + 0.01) ** alpha
        base = np.maximum(base, 0.01)
        inits.append(base / base.sum())

    return inits


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def full_pipeline(a, deadline, verbose=False):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a_new
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 1: Smooth gradient (40% of time)
    t1 = time.monotonic() + 0.35 * remaining
    a1, obj1 = smooth_grad_optimize(a, max_iters=50000, deadline=t1,
                                     tau_init=15.0, tau_max=200.0, lr_init=0.08)
    a1 = try_update(a1)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 2: Coordinate descent (30% of time)
    t2 = time.monotonic() + 0.30 * remaining
    a2, obj2 = coord_descent_incremental(best_a, fast_conv(best_a),
                                          max_sweeps=5000, deadline=t2,
                                          random_order=True, seed=42)
    a2 = try_update(a2)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 3: Smooth gradient again with different params (20%)
    t3 = time.monotonic() + 0.20 * remaining
    a3, obj3 = smooth_grad_optimize(best_a, max_iters=50000, deadline=t3,
                                     tau_init=50.0, tau_max=500.0, lr_init=0.03)
    a3 = try_update(a3)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 4: Final coordinate descent polish (15%)
    t4 = deadline - 2
    a4, obj4 = coord_descent_incremental(best_a, fast_conv(best_a),
                                          max_sweeps=10000, deadline=t4,
                                          random_order=True, seed=123)
    a4 = try_update(a4)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses diverse inits, FFT-based smoothed gradient descent, coordinate descent,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        # Normalize to a reasonable scale
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a_norm.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            ps = parent.sum()
            if ps > 1e-10:
                parent = parent / ps
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Optimize at parent length with the parent as init ===
    phase1_deadline = t_start + 350.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Optimize parent (n={n_parent}) with pipeline", flush=True)
    print(f"{'=' * 60}", flush=True)

    a1, obj1 = full_pipeline(parent, phase1_deadline, verbose=False)
    update_best(a1)
    print(f"  Phase 1 done: J={obj1:.8f}", flush=True)

    # === PHASE 2: Try diverse initializations at parent length ===
    phase2_deadline = t_start + 700.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Diverse inits at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_diverse_inits(n_parent, rng, best_seq)
    # Remove duplicates (very similar inits)
    n_init = min(len(inits), 8)
    per_init_time = (phase2_deadline - time.monotonic()) / max(1, n_init) - 5

    for idx in range(n_init):
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 40:
            break
        alloc = min(per_init_time, remaining_check - 5)
        if alloc < 30:
            break

        a0 = inits[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        # Quick check: skip if too similar to best
        if best_seq is not None:
            sim = np.dot(a0, best_seq) / (np.linalg.norm(a0) * np.linalg.norm(best_seq))
            if sim > 0.999:
                continue

        a_opt, obj_opt = full_pipeline(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f} n={n_parent}", flush=True)

    # === PHASE 3: Multi-length exploration ===
    phase3_deadline = t_start + 1050.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = [256, 384, 512, 640, 768, 896, 1024, 1280, 1536]
    candidate_ns = [n for n in candidate_ns if n != n_parent and n <= 2048]
    # Prioritize lengths near parent
    candidate_ns.sort(key=lambda n: abs(n - n_parent))

    for new_n in candidate_ns[:4]:
        remaining_check = phase3_deadline - time.monotonic()
        if remaining_check < 80:
            break
        alloc = min(120.0, remaining_check - 20)
        if alloc < 50:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = full_pipeline(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 4: Intensive local optimization on best ===
    print(f"\n{'=' * 60}", flush=True)
    remaining = deadline - time.monotonic()
    print(f"PHASE 4: Intensive polish (budget {remaining:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining > 30:
        # Multiple rounds of different methods
        a_curr = best_seq.copy()

        # Round 1: Long smooth gradient
        t_r1 = time.monotonic() + 0.25 * remaining
        print(f"  Smooth grad round (budget {t_r1 - time.monotonic():.0f}s)...", flush=True)
        a_r1, obj_r1 = smooth_grad_optimize(a_curr, max_iters=100000, deadline=t_r1,
                                             tau_init=10.0, tau_max=100.0, lr_init=0.05)
        update_best(a_r1)
        a_curr = best_seq.copy()

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Round 2: Coordinate descent
            t_r2 = time.monotonic() + 0.30 * remaining
            print(f"  CD round (budget {t_r2 - time.monotonic():.0f}s)...", flush=True)
            conv_curr = fast_conv(a_curr)
            a_r2, obj_r2 = coord_descent_incremental(a_curr, conv_curr,
                                                      max_sweeps=20000, deadline=t_r2,
                                                      random_order=True, seed=777)
            update_best(a_r2)
            a_curr = best_seq.copy()

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Round 3: Subgradient multi-active
            t_r3 = time.monotonic() + 0.20 * remaining
            print(f"  Subgrad round (budget {t_r3 - time.monotonic():.0f}s)...", flush=True)
            a_r3, obj_r3 = subgradient_multi_active(a_curr, max_iters=50000, deadline=t_r3,
                                                     eps_frac=0.02, lr_init=0.03)
            update_best(a_r3)
            a_curr = best_seq.copy()

        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Round 4: Final CD polish
            t_r4 = deadline - 5
            print(f"  Final CD (budget {t_r4 - time.monotonic():.0f}s)...", flush=True)
            conv_curr = fast_conv(a_curr)
            a_r4, obj_r4 = coord_descent_incremental(a_curr, conv_curr,
                                                      max_sweeps=20000, deadline=t_r4,
                                                      random_order=True, seed=999)
            update_best(a_r4)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
