def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response_reduced(a, M, n_top=None, tolerance=1e-8):
    """LP with only top-m binding constraints for speed."""
    n = len(a)

    b = fast_conv(a)
    # Select lags with highest values (likely to be binding)
    top_k = np.argsort(b)[::-1][:n_top]
    
    # Build sparse matrix for selected constraints
    rows = []
    cols = []
    vals = []
    for k in top_k:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_k), n))

        if r.success and r.x is not None and r.x.sum() > 1e-15:

def run_mirror_descent(a, deadline, max_iters=500, n_top=64, t_base=0.3, min_t=0.003,
                       use_reduced=True):
    """LP mirror descent with reduced constraints."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        
        # Choose LP type
        if use_reduced:
            # Adaptive n_top: start small, grow as we converge
            current_top = min(n_top, 16 + it * 4)
            g = lp_best_response_reduced(a, M, n_top=current_top)
        else:
            g = lp_best_response_full(a, M)
        
        if g is None:
            # Fallback to full LP
            g = lp_best_response_full(a, M)
            if g is None:
                break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Adaptive step size
        t = max(min_t, t_base / (1 + 0.03 * it))
        
        # Line search
        a_new = None
        for t_try in [t, t*0.5, t*2, t*0.25]:
            a_cand = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s = a_cand.sum()
            if s < 1e-10:
                continue
            a_cand /= s
            M_cand = fast_conv(a_cand).max()
            if M_cand < M:
                a_new = a_cand
                break
        
        if a_new is None:
            # Accept anyway if it's close
            a_cand = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_cand.sum()
            if s < 1e-10:
                break
            a = a_cand / s
        else:
            a = a_new
    
    return best_a

def smooth_max_grad(a, tau):
    """Compute gradient of smooth max of convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-11)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max

def run_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                 lr_init=0.3, lr_end=0.001):
    """Nesterov-accelerated smooth max minimization."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a

def gs_flat_spectral(init, n_iters=200, beta=0.8):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Power compression
        power = 0.7
        mag_comp = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_comp
        a_new = np.fft.irfft(F_new, size)[:n]
        a_cand = np.maximum(a_new, 0)
        sc = a_cand.sum()
        if sc < 1e-15:
            break
        a_cand /= sc
        J_c = compute_J(a_cand)
        if J_c < best_J:
            best_J = J_c
            best_a = a_cand.copy()
        a = a_cand
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coordinate_descent_polish(a, deadline, n_passes=5):
    """Simple coordinate descent: try small perturbations of individual elements."""
    a = np.maximum(a.astype(np.float64), 0)

    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(0)

        if time.monotonic() > deadline:
            break
        for _ in range(n * 2):

                break
            i = rng.integers(0, n)

                M_t = fast_conv(a_trial).max()
                if M_t < best_M - 1e-14:

                    best_a = a_trial.copy()

                    break
    
    return best_a
