   ```
       if violated:
           active_set.update(violated)

       # Update a
       ...
       # Update active set for next iteration
       active_set = {k for k in range(2*n-1) if ag[k] > M*(1-0.1)}
   ```

   This should significantly speed up the LP in later iterations when only a few constraints are active.

   Now let me think about the step size. In the previous code, the step size decays as t_base / (1 + 0.05 * it). This is a standard diminishing step. But we can do better:

   **Line search approach**:
   Given a and g (from LP), the update is a_new = (1-t)*a + t*g where g is normalized to sum=1.
   We want to choose t to minimize J(a_new) = 2n * max_k (a_new * a_new)(k) / (sum(a_new))^2.
   
   Since sum(a_new) = (1-t)*sum(a) + t*sum(g) = 1 (if both are normalized), we just need to minimize max_k (a_new * a_new)(k).
   
   Let d = g - a. Then a_new = a + t*d.
   (a_new * a_new)(k) = (a*a)(k) + 2t*(a*d)(k) + t^2*(d*d)(k)
   
   We want to find t in [0, 1] that minimizes max_k [(a*a)(k) + 2t*(a*d)(k) + t^2*(d*d)(k)].
   
   This is a max of quadratics, which is convex in t. We can find the minimum using:
   - Ternary search (since it's unimodal)
   - Or bisection on the derivative

   For ternary search, we need to evaluate the objective at a few points. Each evaluation requires computing the max of 2n-1 quadratic values, which is O(n). So ternary search takes O(n * log(1/eps)) time.

   This is cheap! Let's use it.

   ```python
   def line_search(a, g, n_evals=30):
       # Find t in [0,1] minimizing max_k (a + t*(g-a))*(a + t*(g-a))(k)
       d = g - a
       lo, hi = 0.0, 1.0
       for _ in range(n_evals):
           m1 = lo + (hi - lo) / 3
           m2 = hi - (hi - lo) / 3
           f1 = max_conv(a + m1 * d)
           f2 = max_conv(a + m2 * d)
           if f1 < f2:
               hi = m2
           else:
               lo = m1
       return (lo + hi) / 2
   ```

   But we need to ensure a + t*d remains non-negative. Since g ≥ 0 and a ≥ 0, and d = g - a, we have a + t*d = (1-t)*a + t*g ≥ 0 for t in [0,1]. Good.

   Also need sum(a + t*d) = (1-t)*1 + t*1 = 1. Good.

   So the line search is clean.

   However, computing max_conv at each ternary step requires an FFT, which costs O(n log n). For n=1024, that's about 1024*10 ≈ 10K operations. 30 ternary steps = 300K operations per line search. With each LP iteration taking maybe 5ms for the LP itself, the line search adds maybe 1-2ms. Acceptable.

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth Nesterov acceleration with adaptive tau for fast initial convergence
2. LP mirror descent with lazy constraints (active set) and ternary-search line search
3. Multi-length exploration around the best length found
4. Final polish with very small LP steps
Key improvements over previous: active-set LP for speed, exact line search for step size,
symmetric variant exploration, and more aggressive length sweeping.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import time


def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def max_conv(a):
    """Return max of self-convolution of a."""
    return float(fast_conv(a).max())


def compute_J_direct(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2 directly."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize a to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def smooth_max_subgrad(a, tau):
    """Compute smooth approximation gradient of max_k (a*a)(k)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Use shifted exponentials for numerical stability
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad_j += 2 * p[k] * a[k-j] for j in [lo, hi]
        # a[k-j] for j in [lo,hi] is a[hi] down to a[lo] reversed... 
        # Actually: for fixed k, (a*a)(k) = sum_j a[j]*a[k-j]
        # d/db_k/d a[j] = a[k-j]
        # So grad[j] = sum_k 2*p[k]*a[k-j]
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] += 2.0 * p[k] * a[idx]
    return grad, p, b


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=2.0, tau_end=300.0,
                        lr_init=0.3, lr_end=0.002):
    """Nesterov-accelerated smooth descent on max autocorrelation."""
    a = normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = max_conv(a)
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b = smooth_max_subgrad(y, tau)
        b_max = float(b.max())
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new
    return best_a


def build_conv_matrix(a, n, active_idx=None):
    """Build sparse matrix A where (A @ g)[k] = (a*g)[k] for k in active_idx."""
    if active_idx is None:
        active_idx = list(range(2 * n - 1))
    rows = []
    cols = []
    vals = []
    for row_pos, k in enumerate(active_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(row_pos)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    return A


def solve_lp(a, M, active_idx=None, n=None):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for k in active_idx, g >= 0."""
    if n is None:
        n = len(a)
    if active_idx is None:
        active_idx = list(range(2 * n - 1))
    A = build_conv_matrix(a, n, active_idx)
    c = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def line_search_step(a, g):
    """Ternary search for optimal t in [0,1] for a_new = (1-t)*a + t*g."""
    d = g - a
    lo, hi = 0.0, 1.0
    for _ in range(25):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = max_conv(a + m1 * d)
        f2 = max_conv(a + m2 * d)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def run_lp_mirror(a, deadline, max_iters=500, use_line_search=True,
                  active_margin=0.85, min_active_frac=0.1):
    """LP mirror descent with lazy constraints and optional line search."""
    a = normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = max_conv(a)
    active_idx = list(range(2 * n - 1))
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break

        # Solve LP with current active set
        g = solve_lp(a, M, active_idx, n)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Check if solution violates any non-active constraints
        ag = fast_conv(a) * 0 + np.convolve(a, g)  # (a*g)
        # Actually compute (a*g) properly
        ag = np.convolve(a, g)
        non_active = [k for k in range(2 * n - 1) if k not in set(active_idx)]
        violated = [k for k in non_active if ag[k] > M * 1.001]
        if violated:
            active_idx.extend(violated)
            active_idx = list(set(active_idx))
            continue

        # Update active set: keep constraints near the boundary
        min_active = max(10, int(min_active_frac * (2 * n - 1)))
        ag_vals = ag[active_idx]
        threshold = M * active_margin
        new_active = [active_idx[i] for i in range(len(active_idx)) if ag_vals[i] >= threshold]
        if len(new_active) < min_active:
            # Keep the top min_active
            sorted_idx = np.argsort(ag_vals)[::-1][:min_active]
            new_active = [active_idx[i] for i in sorted_idx]
        active_idx = new_active

        # Step size
        if use_line_search:
            t = line_search_step(a, g)
            t = max(0.001, min(1.0, t))
        else:
            t = max(0.01, 0.3 / (1 + 0.1 * it))

        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def run_lp_mirror_adaptive(a, deadline, max_iters=300):
    """LP mirror descent with adaptive step size and full constraint check."""
    a = normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = max_conv(a)
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break

        g = solve_lp(a, M, list(range(2 * n - 1)), n)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Try multiple step sizes and pick the best
        best_t = 0.01
        best_M_try = M
        for t_try in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_try = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            M_try = max_conv(a_try)
            if M_try < best_M_try:
                best_M_try = M_try
                best_t = t_try

        a_new = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
        a_new = normalize(a_new)
        a = a_new

    return best_a


def resample_seq(src, n):
    """Resample sequence to length n via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent):
    """Generate diverse initializations for sequence of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p = p / p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.001))
        # Partial randomization
        frac = 0.3
        mask = rng.random(n) < frac
        pert = p.copy()
        pert[mask] = rng.uniform(0.1, 1.0, mask.sum())
        inits.append(np.maximum(pert, 0.001))

    # Structured inits
    inits.append(np.ones(n) / n)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        v = t**alpha + (1 - t)**alpha
        inits.append(np.maximum(v - v.min() + 0.05, 0.01))
    for f in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-3 * (t - 0.5)**2) + 0.05)
    # Random smooth
    k = max(5, n // 16)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Random
    inits.append(np.maximum(rng.uniform(0.1, 1.0, n), 0.01))

    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

                                     use_line_search=True,
                                     active_margin=0.95, min_active_frac=0.3)
            best.update(a_final2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Length perturbation + quick re-opt ===
    if best.seq is not None and time.monotonic() < deadline - 15:
        n_best = best.seq.size
        for delta in [-32, -16, 16, 32, 64]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = normalize(np.maximum(a_ext, 1e-6))
            a_ext = run_nesterov_smooth(a_ext, min(time.monotonic() + 10, deadline - 8),
                                        max_iters=200, tau_start=5, tau_end=80,
                                        lr_init=0.1, lr_end=0.01)
            best.update(a_ext)
            if time.monotonic() < deadline - 8:
                a_lp = run_lp_mirror(a_ext, deadline - 5, max_iters=20,
                                     use_line_search=False,
                                     active_margin=0.9, min_active_frac=0.2)
                best.update(a_lp)

    # === Fallback ===
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J_direct(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
```
