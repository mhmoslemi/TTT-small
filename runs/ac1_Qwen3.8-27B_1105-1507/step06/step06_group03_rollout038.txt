"""
Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses fast coordinate descent with peak-balancing heuristics, SCA-LP integration,
subgradient-inspired steps, stochastic restarts, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_internal(a):
    """Compute J(a) assuming sum(a)=1, so J = 2n*max(conv)."""
    a = np.maximum(a, 0)
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def compute_J_full(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def cd_optimize(a, deadline, seed=42, max_no_improve=150, verbose=False):
    """
    Fast coordinate descent with peak-balancing heuristic.
    Operates on normalized (sum=1) sequences. J = 2n*max(conv).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    sweep = 0

    while time.monotonic() < deadline:
        sweep += 1
        order = rng.permutation(n)
        improved_sweep = 0

        for i in order:
            old = a[i]
            cmax = conv.max()

            # Generate candidates
            if old < 1e-12:
                candidates = [0.0, 1e-4, 5e-4, 1e-3]
            else:
                # Find peak and compute balancing delta
                k_peak = np.argmax(conv)
                ki = k_peak - i
                cands = [0.0, old * 0.4, old * 0.7, old * 1.3]

                if 0 <= ki < n and a[ki] > 1e-14:
                    # Compute second max (approximate)
                    # To save time, just use cmax and a fraction
                    # Balance: reduce conv[k_peak] by 2*delta*a[ki]
                    # Target: reduce to 0.95*cmax
                    delta_balance = 0.05 * cmax / (2.0 * a[ki])
                    if old - delta_balance > 0:
                        cands.append(old - delta_balance)
                    cands.append(old * 0.1)

                candidates = np.array(cands)
                candidates = np.clip(candidates, 0, max(old * 3, 0.01))
                candidates = np.unique(candidates)

            best_local = best_obj if best_obj < 2.0*n*conv.max() else 2.0*n*conv.max()
            cur_obj = 2.0 * n * conv.max()
            best_val = old

            for x in candidates:
                x = float(x)
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                # Apply: conv[i:i+n] += 2*delta*a, conv[2*i] += delta^2
                conv[i:i + n] += 2.0 * delta * a
                if 2 * i < 2 * n - 1:
                    conv[2 * i] += delta * delta

                new_s = 1.0 + delta  # since we started with sum=1
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                # Undo
                conv[i:i + n] -= 2.0 * delta * a
                if 2 * i < 2 * n - 1:
                    conv[2 * i] -= delta * delta

            if abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[i:i + n] += 2.0 * delta * a
                if 2 * i < 2 * n - 1:
                    conv[2 * i] += delta * delta
                a[i] = best_val
                improved_sweep += 1

        # Check improvement
        cur_s = a.sum()
        if cur_s > 1e-10:
            current_obj = 2.0 * n * conv.max() / (cur_s * cur_s)
        else:
            current_obj = float('inf')

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 100 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def subgradient_step(a, n_steps=50, deadline=None, lr_init=0.02, verbose=False):
    """
    Subgradient-inspired descent: at each step, find the peak lag k*,
    then decrease a[i] proportionally to a[k*-i] (the subgradient direction).
    """
    n = len(a)
    a = np.maximum(a.copy(), 1e-12)
    a /= a.sum()
    best_a = a.copy()
    best_obj = compute_J_internal(a)
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        k_peak = np.argmax(conv)
        cmax = conv.max()

        # Subgradient: grad[i] = 2*a[k_peak - i] if 0 <= k_peak - i < n
        grad = np.zeros(n)
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * a[k_peak - i]

        # Project gradient to maintain sum=1: subtract mean
        grad = grad - grad.mean()

        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.05 * step)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            break
        a_new /= a_new.sum()

        obj_new = compute_J_internal(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = a - lr * 0.3 * grad
            a_try = np.maximum(a_try, 0)
            if a_try.sum() > 1e-10:
                a_try /= a_try.sum()
                obj_try = compute_J_internal(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    a = best_a.copy()

        if verbose and step % 20 == 0:
            print(f"    SubGrad step {step}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def lp_sca_step(a, deadline, max_iters=100, verbose=False):
    """
    SCA-LP: Solve LP to find best response direction, then take a convex step.
    LP: min t s.t. (g * a)[k] <= t, sum(g)=1, g>=0
    """
    n = len(a)
    a = np.maximum(a.copy(), 1e-10)
    a /= a.sum()
    best_a = a.copy()
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    no_improve = 0

    # Precompute the constraint structure
    # (g * a)[k] = sum_j g[j] * a[k-j]
    # For each k, the coefficients are a[k-j] for valid j

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        rows = []
        cols = []
        vals = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        b_ub = np.zeros(2 * n - 1)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 20, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [1.0, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1]:
                        a_try = (1 - frac) * best_a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        obj_try = compute_J_internal(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_a = a_try.copy()
                            best_obj = obj_try
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 25:
            break

        if verbose and it % 10 == 0:
            print(f"    SCA-LP it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            p /= ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
        # Asymmetric perturbations
        asym = t * 0.3
        inits.append(project_simplex(p + asym))
        inits.append(project_simplex(p - asym + 0.15))

    # Linear decreasing
    inits.append(project_simplex(n - np.arange(n, dtype=np.float64) + 1))
    # Quadratic decreasing
    inits.append(project_simplex((n - np.arange(n, dtype=np.float64)) ** 2 + 1))
    # Two-block (high first half, low second)
    two_block = np.concatenate([np.ones(n // 2) * 2, np.ones(n - n // 2) * 0.3])
    inits.append(project_simplex(two_block))
    # Sine-modulated
    for freq in [1, 2, 3]:
        inits.append(project_simplex(1.0 + 0.4 * np.cos(2 * np.pi * freq * t)))
        inits.append(project_simplex(1.0 + 0.4 * np.sin(2 * np.pi * freq * t)))
    # Random smooth
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0.3, 1.0, n), kernel, mode='same')
        inits.append(project_simplex(smooth))
    # Exponential decay
    for rate in [1.0, 2.0, 3.0]:
        inits.append(project_simplex(np.exp(-rate * t)))
        inits.append(project_simplex(np.exp(-rate * (1 - t))))

    return inits

            remaining = phase3_end - time.monotonic()

                # Pass 3: Subgradient + CD
                t_p3 = time.monotonic() + 0.20 * remaining
                print(f"  SubGrad pass ({t_p3 - time.monotonic():.0f}s)...", flush=True)
                a_sg, obj_sg = subgradient_step(a_curr, n_steps=300, deadline=t_p3)
                update_best(a_sg)
                a_curr = best_seq.copy()
                a_curr = np.maximum(a_curr, 0)
                a_curr /= a_curr.sum()

            remaining = phase3_end - time.monotonic()
            if remaining > 30:
                # Pass 4: Final CD polish
                t_p4 = deadline - 5
                print(f"  Final CD ({t_p4 - time.monotonic():.0f}s)...", flush=True)
                a_final = cd_optimize(a_curr, t_p4, seed=7777, max_no_improve=500)
                update_best(a_final)

    # === PHASE 4: Stochastic restarts (if time remains) ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 4: Stochastic restarts", flush=True)
        print(f"{'=' * 60}", flush=True)

        if best_seq is not None:
            n_best = len(best_seq)
            best_norm = np.maximum(best_seq, 0)
            best_norm /= best_norm.sum()

            for restart in range(5):
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                noise_scale = 0.05 + 0.03 * restart
                a_pert = best_norm + rng.normal(0, noise_scale, n_best)
                a_pert = project_simplex(a_pert)

                per_restart = min(40.0, remaining - 10)
                a_opt, obj_opt = optimize_single(a_pert, time.monotonic() + per_restart,
                                                 strategy='cd_only', seed=5000 + restart)
                update_best(a_opt)
                print(f"  Restart {restart}: J={obj_opt:.8f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
