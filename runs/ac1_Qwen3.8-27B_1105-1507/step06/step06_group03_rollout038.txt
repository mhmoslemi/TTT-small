"""
Optimized sequence search combining:
1. Spectral-shaping initialization (asymmetric shapes)
2. L-BFGS on log-sum-exp smoothed objective
3. Active-set-aware subgradient method
4. Fast coordinate descent with analytic 1D optimization
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_smooth_max(a, tau):
    """Log-sum-exp approximation to max."""
    m = a.max()
    return m + np.log(np.exp((a - m) * tau).sum()) / tau

def lse_smooth_grad(a, tau):
    """Gradient of log-sum-exp approximation."""
    m = a.max()
    w = np.exp((a - m) * tau)
    w /= w.sum()
    return w

def smooth_objective(x, tau, n):
    """Smooth objective: 2n * LSE(conv(exp(x), exp(x))) / (sum(exp(x)))^2"""
    a = np.exp(x)
    conv = fast_conv(a)
    lse = lse_smooth_max(conv, tau)
    s = a.sum()
    return 2 * n * lse / (s * s)

def smooth_gradient(x, tau, n):
    """Gradient of smooth objective w.r.t. x (log-parameterization)."""
    a = np.exp(x)
    s = a.sum()
    conv = fast_conv(a)
    
    w = lse_smooth_grad(conv, tau)
    
    # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with boundary handling)

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1].astype(np.float64)

def conv_full(a, b):
    """Full convolution of a and b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1].astype(np.float64)

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex_v(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Scale to sum 1 first, then scale to target
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.full(n, target_sum / n)
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

            return smooth_gradient(a_, tau) * a_
        
        res = minimize(obj, x, jac=grad, method='L-BFGS-B',
                       options={'maxiter': 30, 'ftol': 1e-12, 'gtol': 1e-8})
        
        if res.success and res.fun < float('inf'):
            a_new = np.exp(res.x)
            a_new = np.maximum(a_new, 0)
            if a_new.sum() > 1e-10:
                a_new = a_new / a_new.sum()
                obj_new = compute_J(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose and it % 20 == 0:
                        print(f"      L-BFGS it {it}: tau={tau:.0f} J={best_obj:.8f}", flush=True)
        
        x = np.log(best_a)
    
    return best_a, best_obj

def coord_descent_fast(a, max_sweeps=5000, deadline=None, random_order=False, seed=42, verbose=False):
    """
    Fast coordinate descent with adaptive 1D search.
    For each coordinate, use a small set of candidate values including
    an analytic estimate of the optimum.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 500
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        improved = False
        
        for i in order:
            old = a[i]
            
            # Determine candidate values
            # The optimal a[i] balances reducing the max conv vs. the sum change
            # Quick estimate: try to equalize the top few conv values
            if old < 1e-12:
                candidates = np.array([0.0, 0.01, 0.02, 0.05, 0.1])
            else:
                # Try ratios around current value
                ratios = np.array([0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
                candidates = ratios * old
                # Also try "equalization" values
                # If increasing a[i] reduces the max, try values that would
                # bring the max down to the second-max level
                top2 = np.sort(conv)[-2:]
                if top2[0] > 0:
                    # Rough estimate of useful values
                    candidates = np.append(candidates, old * np.array([top2[1]/top2[0], (top2[1]/top2[0])**0.5]))
            
            candidates = np.unique(np.maximum(candidates, 0.0))
            # Limit to reasonable number
            if len(candidates) > 15:
                # Keep the most spread out ones
                idx = np.linspace(0, len(candidates)-1, 15).astype(int)
                candidates = candidates[idx]
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)
            
            for x in candidates:
                x = float(x)
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        if verbose and sweep % 1000 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a

def active_set_qp(a, max_iters=50, deadline=None, verbose=False):
    """
    Active-set QP: at each step, identify the active set (top lags)
    and solve a QP that minimizes the max over the active set.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        M = conv.max()
        
        # Identify active set: lags where conv[k] > 0.9 * M
        active_mask = conv > 0.9 * M
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) == 0:
            break
        
        # Build QP: minimize t
        # s.t. conv_new[k] <= t for k in active set
        #      sum(g) = 1
        #      g >= 0
        # where conv_new[k] = sum_i g[i]*a[k-i] + sum_i a[i]*g[k-i]
        #                    = sum_i (a[k-i] + a[i-k]) * g[i]  (linear in g!)
        
        # Wait, that's not right. conv_new is the self-convolution of g, not the cross-convolution.
        # We're trying to find g such that conv(g,g) is "flat" on the active set.
        # But conv(g,g) is quadratic in g, so we can't directly solve a QP.
        
        # Instead, use the SCA approach: linearize around current a.
        # conv(a + d, a + d)[k] ≈ conv(a,a)[k] + 2*(a*d)[k] + (d*d)[k]
        # For the first-order approximation: ≈ conv(a,a)[k] + 2*(a*d)[k]
        
        # So the LP becomes:
        # minimize t
        # s.t. conv(a,a)[k] + 2*(a*g)[k] <= t for k in active set
        #      sum(g) = 1
        #      g >= 0
        # where (a*g)[k] = sum_i a[i]*g[k-i] (cross-correlation, linear in g)
        
        # Actually, the linearization of conv(g,g) around a is:
        # conv(a + (g-a), a + (g-a))[k] ≈ conv(a,a)[k] + 2*(a*(g-a))[k]
        # Let d = g - a: conv ≈ conv(a,a) + 2*(a*d) = conv(a,a) + 2*(a*g) - 2*conv(a,a)
        # = 2*(a*g) - conv(a,a)
        
        # So the constraint is: 2*(a*g)[k] - conv(a,a)[k] <= t
        # i.e., 2*(a*g)[k] <= t + conv(a,a)[k]
        
        # (a*g)[k] = sum_i a[i]*g[k-i] (with appropriate bounds)
        # For fixed k: (a*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*g[k-i]
        # Let j = k-i: = sum_{j=k-(n-1)}^{k} a[k-j]*g[j] (with bounds)
        # = sum_j a[k-j]*g[j]
        
        # So (a*g)[k] is linear in g with coefficient a[k-j] for g[j].
        
        # Build the LP:
        # Variables: g[0..n-1], t (total n+1 variables)
        # Constraints: 2*sum_j a[k-j]*g[j] - t <= conv[k] for k in active set
        #              sum(g) = 1
        #              g >= 0
        
        rows = []
        cols = []
        vals = []
        b_ub = []
        
        for k in active_indices:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(len(b_ub))
            cols.append(n)  # t variable
            vals.append(-1.0)
            b_ub.append(conv[k])
        
        A_ub = coo_matrix((vals, (rows, cols)), shape=(len(active_indices), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t

        b_eq = [1.0]
