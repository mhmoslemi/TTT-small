Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Dual-gap-based convergence detection
3. Multi-scale initialization (coarse-to-fine)

5. Block coordinate descent polish
6. Multi-restart with diverse initializations

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J_normalized(a):
    """Compute J for a normalized (sum=1) sequence: J = 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * b.max()

def build_full_lp_matrix(n, a):
    """Build the full constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    rows = []
    cols = []
    vals = []
    conv_len = 2 * n - 1
    for k in range(conv_len):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    return A

def lp_best_response_full(a, M, time_limit=10):
    """Solve LP best response with ALL constraints.
    max sum(g) s.t. A@g <= M*ones, g >= 0
    """
    n = len(a)
    conv_len = 2 * n - 1
    A = build_full_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, r.fun  # r.fun = -optimal value of sum(g)
    except Exception as e:
        pass
    return None, None

def lp_best_response_topk(a, M, k=100, time_limit=5):
    """Solve LP best response with top-k constraints only."""
    n = len(a)
    conv_len = 2 * n - 1
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:k]
    
    rows = []
    cols = []
    vals = []
    for idx, k_val in enumerate(top_indices):
        j_lo = max(0, int(k_val) - n + 1)
        j_hi = min(n - 1, int(k_val))
        for j in range(j_lo, j_hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[int(k_val) - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:

    except Exception:
        pass
    return None

    n = len(a)

    best_M = fast_conv(a).max()
    no_improve = 0

        if time.monotonic() > deadline:

        if use_full:
            g, dual_val = lp_best_response_full(a, best_M, time_limit=5)
        else:
            g = lp_best_response_topk(a, best_M, k=topk, time_limit=3)
            dual_val = None
        
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break

        improved = False
        # Backtracking line search
        t = 0.5

            a_trial = (1.0 - t) * a + t * g

            if s_t < 1e-10:

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_normalized(a):

def proj_simplex(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho + 1.0 > cssv
    if not cond.any():
        theta = 0.0
    else:
        rho_idx = int(np.argmax(~cond)) if (~cond).any() else n - 1
        theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:

def build_lp_matrix_sparse(n, a, constraint_indices):
    """Build sparse LP constraint matrix for given constraint indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    nc = len(constraint_indices)
    A = csr_matrix((vals, (rows, cols)), shape=(nc, n))
    return A


def solve_lp_best_response(n, a, M, constraint_indices, time_limit=10.0):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for k in indices, g >= 0.
    Returns (normalized_g, optimal_value) or (None, None).
    """
    A = build_lp_matrix_sparse(n, a, constraint_indices)
    if A.nnz == 0:
        return None, None
    c = -np.ones(n)
    b_ub = np.full(len(constraint_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.fun is not None:
            opt_val = -r.fun  # optimal value of max sum(g)
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, opt_val
    except Exception:
        pass
    return None, None

    n = len(a)
    conv_len = 2 * n - 1

    if use_full_constraints:
        constraint_indices = list(range(conv_len))
    else:
        constraint_indices = list(range(conv_len))  # start with all, filter later

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M_cur = b.max()

        # Determine active set
        if use_full_constraints:
            active = list(range(conv_len))
        else:
            # Use top-k plus some margin
            k_eff = min(topk + it * 2, conv_len)
            top_idx = np.argsort(b)[::-1][:k_eff]
            active = top_idx.tolist()

        g, lp_opt = solve_lp_best_response(n, a, M_cur, active, time_limit=lp_time_limit)

        if g is None:
            no_improve += 2
            if no_improve > max_no_improve:
                break
            continue

        # Check dual gap: if lp_opt is very close to 1, we're near optimal
        if lp_opt is not None and lp_opt < 1.0 + 1e-8:
            if verbose:
                print(f"    LP dual gap ~0 at iter {it}, M={M_cur:.10f}", flush=True)
            break

        # Backtracking line search
        improved = False
        t = 0.5
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < M_cur * (1.0 - 1e-10):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
            t *= 0.5

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: M={M_cur:.10f} lp_opt={lp_opt:.6f}", flush=True)

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def fista_smooth_max(a_init, tau, max_iters, deadline, lr=0.05):
    """FISTA on log-sum-exp smooth approximation of max(conv(a,a))."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(y)
        b_max = b.max()
        # Numerically stable log-sum-exp
        exp_b = np.exp(tau * (b - b_max))
        exp_sum = exp_b.sum()
        if exp_sum < 1e-300:
            break
        p = exp_b / exp_sum

        # Gradient: grad[i] = 2 * sum_{m=0}^{n-1} p[i+m] * a[m] (where i+m < 2n-1)
        grad = np.zeros(n)
        for i in range(n):
            hi = min(2 * n - 1, i + n)
            if i < hi:
                grad[i] = 2.0 * np.dot(p[i:hi], a[:hi - i])

        grad_tan = grad - grad.mean()
        step = lr / (1.0 + 0.005 * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new

        if it % 20 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def subgradient_averaging(a_init, max_iters, deadline):
    """Subgradient method with Polyak step size and iterate averaging."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    # Lower bound on optimal M: average of b = 1/(2n-1) since sum(b) = (sum a)^2 = 1
    M_lb = 1.0 / (2 * n - 1)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    a_avg = a.copy()
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M_cur = b.max()
        k_star = int(np.argmax(b))

        # Subgradient of max at k_star
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            grad[j] = 2.0 * a[k_star - j]

        grad_tan = grad - grad.mean()
        grad_norm_sq = np.dot(grad_tan, grad_tan)
        if grad_norm_sq < 1e-15:
            break

        # Polyak step size
        step = (M_cur - M_lb) / (grad_norm_sq + 1e-15)
        step = min(step, 1.0 / (0.5 * it + 1))

        a_new = a - step * grad_tan
        a_new = proj_simplex(a_new)

        count += 1
        a_avg = a_avg + (a_new - a_avg) / count
        a = a_new

        if it % 100 == 0:
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = a_avg.copy()
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()

    M_avg = fast_conv(a_avg).max()
    if M_avg < best_M:
        best_M = M_avg
        best_a = a_avg.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = parent.copy()
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            p = np.maximum(np.interp(t, t_old, p), 0.0)
        if p.sum() > 0:
            p = p / p.sum()
            inits.append(p)
            for sigma in [0.005, 0.02, 0.05, 0.1]:
                p_pert = np.maximum(p + rng.normal(0, sigma, n), 1e-8)
                inits.append(p_pert / p_pert.sum())

    inits.append(np.ones(n) / n)

    gauss = np.exp(-6.0 * (t - 0.5) ** 2) + 0.02
    inits.append(gauss / gauss.sum())

    para = t * (1 - t) + 0.05
    inits.append(para / para.sum())

    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


def run_combined_optimization(a0, deadline, verbose=True):
    """Run multi-stage optimization on a single sequence."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    # Stage 1: LP with small active set (fast)
    t1 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
    a = run_lp_mirror_descent(a, t1, use_full_constraints=False, topk=30,
                              max_no_improve=15, lp_time_limit=3.0, verbose=verbose)

    # Stage 2: FISTA for smoothing
    t2 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.25)
    a = fista_smooth_max(a, 200, 2000, t2, lr=0.03)

    # Stage 3: LP with larger active set
    t3 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
    a = run_lp_mirror_descent(a, t3, use_full_constraints=False, topk=150,
                              max_no_improve=20, lp_time_limit=4.0, verbose=verbose)

    # Stage 4: Full constraint LP
    t4 = deadline
    a = run_lp_mirror_descent(a, t4, use_full_constraints=True,
                              max_no_improve=30, lp_time_limit=8.0, verbose=verbose)

    return a


def propose_candidate(budget_s=1500):
    """
    Multi-phase search for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    - Phase 1 (50%): LP mirror descent from parent with growing active sets
    - Phase 2 (20%): FISTA refinement with multiple tau values
    - Phase 3 (10%): Subgradient averaging
    - Phase 4 (15%): Multi-restart from diverse inits
    - Phase 5 (5%): Final full-constraint LP polish
    """
    rng = np.random.default_rng(12345)
    t_start = time.monotonic()
    total_budget = budget_s - 30.0
    deadline = t_start + total_budget

    best_seq = None
    best_score = float('inf')

    def track(a):
        nonlocal best_seq, best_score
        a_c = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a_c.size == 0 or a_c.sum() < 0.01:
            return
        sc = float(evaluate_sequence(a_c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            el = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={sc:.8f} n={a_c.size} t={el:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            track(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        n = 768
        parent = np.ones(n)
        track(parent)
        print(f"No parent, uniform n=768, J={best_score:.8f}", flush=True)

    n = best_seq.size if best_seq is not None else 768
    print(f"\nStarting optimization with n={n}", flush=True)

    # === PHASE 1: LP Mirror Descent from parent (50% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent (growing active set)", flush=True)
    print(f"{'='*60}", flush=True)

    a_work = parent.copy() if parent.size == n else np.ones(n) / n
    a_work = a_work / a_work.sum()

    # 1a: Top-20 LP (fast, many iterations)
    t1a_dl = t_start + 0.12 * total_budget
    print(f"  1a: topk=20, budget={t1a_dl - time.monotonic():.0f}s", flush=True)
    a_work = run_lp_mirror_descent(a_work, t1a_dl, use_full_constraints=False,
                                   topk=20, max_no_improve=12, lp_time_limit=2.0)
    track(a_work)
    print(f"  1a done: M={fast_conv(a_work).max():.10f}", flush=True)

    # 1b: Top-50 LP
    t1b_dl = t_start + 0.22 * total_budget
    print(f"  1b: topk=50, budget={t1b_dl - time.monotonic():.0f}s", flush=True)
    a_work = run_lp_mirror_descent(a_work, t1b_dl, use_full_constraints=False,
                                   topk=50, max_no_improve=15, lp_time_limit=3.0)
    track(a_work)
    print(f"  1b done: M={fast_conv(a_work).max():.10f}", flush=True)

    # 1c: Top-100 LP
    t1c_dl = t_start + 0.32 * total_budget
    print(f"  1c: topk=100, budget={t1c_dl - time.monotonic():.0f}s", flush=True)
    a_work = run_lp_mirror_descent(a_work, t1c_dl, use_full_constraints=False,
                                   topk=100, max_no_improve=15, lp_time_limit=4.0)
    track(a_work)
    print(f"  1c done: M={fast_conv(a_work).max():.10f}", flush=True)

    # 1d: Top-250 LP
    t1d_dl = t_start + 0.42 * total_budget
    print(f"  1d: topk=250, budget={t1d_dl - time.monotonic():.0f}s", flush=True)
    a_work = run_lp_mirror_descent(a_work, t1d_dl, use_full_constraints=False,
                                   topk=250, max_no_improve=20, lp_time_limit=5.0)
    track(a_work)
    print(f"  1d done: M={fast_conv(a_work).max():.10f}", flush=True)

    # 1e: Full constraint LP
    t1e_dl = t_start + 0.50 * total_budget
    print(f"  1e: full constraints, budget={t1e_dl - time.monotonic():.0f}s", flush=True)
    a_work = run_lp_mirror_descent(a_work, t1e_dl, use_full_constraints=True,
                                   max_no_improve=25, lp_time_limit=8.0)
    track(a_work)
    print(f"  1e done: M={fast_conv(a_work).max():.10f}", flush=True)

    # === PHASE 2: FISTA refinement (20% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)

    current = best_seq if best_seq is not None else a_work
    t2_dl = t_start + 0.70 * total_budget

    for tau in [50, 150, 400, 1000]:
        if time.monotonic() > t2_dl:
            break
        sub_dl = min(t2_dl, time.monotonic() + max(10, (t2_dl - time.monotonic()) / 2.0))
        a_f = fista_smooth_max(current, tau, 5000, sub_dl, lr=0.03)
        track(a_f)
        if best_seq is not None:
            current = best_seq
        else:
            current = a_f
        print(f"  tau={tau}: J={best_score:.8f}", flush=True)

    # === PHASE 3: Subgradient averaging (10% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Subgradient averaging", flush=True)
    print(f"{'='*60}", flush=True)

    t3_dl = t_start + 0.80 * total_budget
    if best_seq is not None:
        a_sg = subgradient_averaging(best_seq, 10000, t3_dl)
        track(a_sg)
    print(f"  Subgrad done: J={best_score:.8f}", flush=True)

    # === PHASE 4: Multi-restart (15% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-restart", flush=True)
    print(f"{'='*60}", flush=True)

    t4_dl = t_start + 0.95 * total_budget
    if best_seq is not None:
        inits = generate_inits(n, rng, best_seq)
        n_inits = min(len(inits), 5)
        time_per_init = max(5.0, (t4_dl - time.monotonic()) / n_inits)

        for ii in range(n_inits):
            if time.monotonic() > t4_dl:
                break
            a0 = inits[ii]
            sub_dl = min(t4_dl, time.monotonic() + time_per_init)
            print(f"  Restart {ii+1}/{n_inits} (budget {time_per_init:.0f}s)...", flush=True)

            # Quick LP + FISTA combo
            a_r = run_lp_mirror_descent(a0, min(sub_dl, time.monotonic() + time_per_init * 0.5),
                                        use_full_constraints=False, topk=50,
                                        max_no_improve=10, lp_time_limit=3.0, verbose=False)
            a_r = fista_smooth_max(a_r, 200, 1000, sub_dl, lr=0.03)
            track(a_r)

    # === PHASE 5: Final full-constraint LP polish (5% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_final = run_lp_mirror_descent(best_seq, deadline, use_full_constraints=True,
                                        max_no_improve=30, lp_time_limit=10.0, verbose=True)
        track(a_final)
        print(f"  Final LP: J={best_score:.8f}", flush=True)

        # One more FISTA pass
        if time.monotonic() < deadline - 5:
            a_f2 = fista_smooth_max(best_seq, 500, 2000, deadline - 2, lr=0.02)
            track(a_f2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768)
        best_score = 2.0

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
