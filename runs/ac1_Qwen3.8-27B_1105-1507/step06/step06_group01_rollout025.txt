import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    """Compute max of conv(a,a) for a on the simplex."""
    b = fast_conv(a)
    return float(b.max())

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(a):
    """Build the convolution matrix C where (C*g)[k] = (a*g)[k]."""
    n = len(a)
    conv_len = 2 * n - 1
    # C is conv_len x n
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                if a[m] > 1e-15:
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[m])
    C = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    return C

def lp_best_response_full(a, M):
    """Solve the LP best response with ALL constraints.
    max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
    Returns (g_normalized, primal_optimal_value).
    """
    n = len(a)
    conv_len = 2 * n - 1
    C = build_conv_matrix(a)
    
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                # The optimal value is -r.fun (since we minimize -sum(g))
                # The "improvement ratio" is g_sum (since a is normalized to sum 1)
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, 0.0

def golden_section_search(a, g, best_M, max_evals=40):
    """Golden section line search for t in [0,1] minimizing M((1-t)*a + t*g)."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def obj(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            return best_M + 1
        a_t = a_t / s
        return compute_M(a_t)
    
    lo = 0.0
    hi = 1.0
    
    # First check endpoints
    f_lo = obj(lo)  # should be ~best_M
    f_hi = obj(hi)
    
    if f_hi >= f_lo:
        # Minimum is at lo or near lo
        # Try a small t
        for t_test in [0.01, 0.005, 0.001]:
            f_t = obj(t_test)
            if f_t < f_lo:
                return t_test, f_t
        return 0.0, f_lo
    
    # Golden section
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(max_evals):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    t_opt = (lo + hi) / 2.0
    f_opt = obj(t_opt)
    return t_opt, f_opt

def run_fista_smooth(a_init, tau_start, max_iters, deadline, lr=0.3):
    """FISTA on log-sum-exp smooth approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        tau = tau_start
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        n_conv = len(b)
        grad = np.zeros(n)
        for i in range(n):
            end = min(i + n, n_conv)
            if i < n_conv:
                grad[i] = 2.0 * np.dot(p[i:end], a[:end-i])
        
        grad_tan = grad - grad.mean()
        step = lr / (1.0 + 0.01 * it)
        
        a_new = y - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        if it % 10 == 0:
            M_cur = compute_M(a)
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = compute_M(a)
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with ALL constraints (2n-1) + golden section line search
2. FISTA on smooth max approximation for diversification
3. Perturbation-based restarts to escape local minima
4. Multi-length search with asymmetric initializations
Key improvement over previous: full-constraint LP gives exact best-response
direction (tighter than top-k), and golden section extracts max progress per step.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of conv(a,a) assuming a is on simplex (sum=1)."""
    b = fast_conv(a)
    return float(b.max())


def compute_J_from_M(a, M):
    """Compute J from M (max conv) assuming sum(a)=1."""
    n = len(a)
    return float(2 * n * M)


def build_conv_matrix(a):
    """Build sparse convolution matrix C where (C*g)[k] = (a*g)[k].
    C has shape (2n-1, n). C[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    conv_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            m = k - j
            if a[m] > 1e-14:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
    C = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    return C


def lp_best_response_full(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, optimal_sum) or (None, 0)."""
    n = len(a)

    C = build_conv_matrix(a)

    b_ub = np.full(conv_len, M)

        r = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'mip_rel_gap': 0.0})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = float(g.sum())
            if g_sum > 1e-10:
                return g / g_sum, g_sum
            else:
                return None, 0.0
        else:
            # Try with relaxed tolerance
            r = linprog(c, A_ub=C, b_ub=b_ub * (1 + 1e-8), bounds=bounds,
                        method='highs', options={'time_limit': time_limit, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0.0)
                g_sum = float(g.sum())
                if g_sum > 1e-10:
                    return g / g_sum, g_sum
    except Exception:
        pass
    return None, 0.0


def lp_best_response_topk(a, M, top_k=50, time_limit=10):

def golden_section_search(a, g, best_M, max_evals=35):
    """Golden section line search: min M((1-t)*a + t*g) for t in [0,1]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def obj(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            return best_M + 1.0
        a_t = a_t / s
        return compute_M(a_t)
    
    lo = 0.0
    hi = 1.0
    f_hi = obj(hi)
    
    if f_hi >= best_M:
        # Check small t values
        best_t = 0.0
        best_f = best_M
        for t_test in [0.05, 0.02, 0.01, 0.005]:
            f_t = obj(t_test)
            if f_t < best_f:
                best_f = f_t
                best_t = t_test
        return best_t, best_f
    
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa

            fb = obj(b_pt)

    """LP best-response optimization loop with golden section line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        if use_full:
            g, g_sum = lp_best_response_full(a, best_M, time_limit=25)
        else:
            g, g_sum = lp_best_response_topk(a, best_M, top_k=top_k)
        
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # If g_sum is close to 1, we're near a local minimum
        if g_sum < 1.0 + 1e-6:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        # Golden section line search
        t_opt, f_opt = golden_section_search(a, g, best_M, max_evals=30)
        
        if f_opt < best_M - 1e-13:
            a_new = (1.0 - t_opt) * a + t_opt * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            a = a_new
            best_M = f_opt
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                break
    
    return best_a


def run_fista_smooth(a_init, tau, max_iters, deadline, lr=0.3):
    """FISTA on log-sum-exp smooth approximation of max(conv)."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        n_conv = len(b)
        grad = np.zeros(n)
        for i in range(n):
            end = min(i + n, n_conv)
            if i < n_conv:
                grad[i] = 2.0 * np.dot(p[i:end], a[:end - i])
        
        grad_tan = grad - grad.mean()
        step = lr / (1.0 + 0.005 * it)
        
        a_new = y - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-10:
            y_new = a_new.copy()
        else:
            y_new = y_new / s_y
        
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        
        if it % 10 == 0:
            M_cur = compute_M(a)
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = compute_M(a)
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def run_subgrad_descent(a_init, max_iters, deadline, step_init=0.5):
    """Subgradient descent using active constraint gradients."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Use all near-active constraints for subgradient
        threshold = M * 0.995
        active = np.where(b >= threshold)[0]
        
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)
        
        # Project to tangent (sum-zero) direction
        grad_tan = grad - grad.mean()
        
        # Decreasing step
        step = step_init / (1.0 + 0.05 * it)
        
        a_new = a - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        
        M_new = compute_M(a)
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences including asymmetric ones."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for noise in [0.02, 0.05, 0.1, 0.15]:
            p_n = np.maximum(p + rng.normal(0, noise, n), 1e-4)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Asymmetric ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    inits.append(ramp[::-1].copy() / ramp.sum())
    
    # Quadratic asymmetric
    quad = (t**2 + 0.1)
    inits.append(quad / quad.sum())
    inits.append((1 - t)**2 + 0.1)
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # Exponential decay
    exp_dec = np.exp(-3.0 * t) + 0.05
    inits.append(exp_dec / exp_dec.sum())
    inits.append(exp_dec[::-1].copy() / exp_dec.sum())
    
    # Centered Gaussian
    gauss = np.exp(-4.0 * (t - 0.5)**2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Skewed Gaussian
    skew = np.exp(-3.0 * (t - 0.35)**2) + 0.05
    inits.append(skew / skew.sum())
    
    # Cosine window
    cos_w = 0.5 + 0.4 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for min J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Full-constraint LP best-response + golden section on parent
    Phase 2: Diversification with FISTA + subgradient from multiple inits
    Phase 3: LP optimization with perturbation restarts (main improvement phase)
    Phase 4: Length exploration
    Phase 5: Final intensive LP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 1: Full LP on parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Full-constraint LP on n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)
    
    t1_deadline = min(deadline, time.monotonic() + 300)
    
    # First, quick FISTA to improve init
    a_fista = run_fista_smooth(parent, 50, 2000, min(t1_deadline, time.monotonic() + 60), lr=0.3)
    tracker.update(a_fista)
    
    # Full LP from best so far
    a_lp = run_lp_optimization(tracker.best, 500, t1_deadline, use_full=True)
    tracker.update(a_lp)
    
    # Round 2 of LP (often catches additional improvement)
    if time.monotonic() < t1_deadline:
        a_lp2 = run_lp_optimization(a_lp, 300, t1_deadline, use_full=True)
        tracker.update(a_lp2)
    
    tracker.report("After Phase 1")
    
    # === Phase 2: Diversification ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Diversification with multiple methods", flush=True)
    print(f"{'='*60}", flush=True)
    
    n_best = tracker.best.size
    t2_deadline = min(deadline - 300, time.monotonic() + 200)
    
    inits = generate_inits(n_best, rng, tracker.best)
    n_inits = min(len(inits), 6)
    time_per_init = max(10, (t2_deadline - time.monotonic()) / max(1, n_inits))
    
    for ii in range(n_inits):
        if time.monotonic() > t2_deadline:
            break
        a0 = inits[ii].copy()
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        
        init_time = min(time_per_init, t2_deadline - time.monotonic())
        if init_time < 5:
            break
        
        # Alternate between methods
        if ii % 3 == 0:
            a_opt = run_fista_smooth(a0, 100, 3000, min(t2_deadline, time.monotonic() + init_time * 0.5), lr=0.3)
            if time.monotonic() < t2_deadline:
                a_opt = run_lp_optimization(a_opt, 30, t2_deadline, use_full=True)
        elif ii % 3 == 1:
            a_opt = run_subgrad_descent(a0, 2000, min(t2_deadline, time.monotonic() + init_time * 0.5), step_init=0.4)
            if time.monotonic() < t2_deadline:
                a_opt = run_lp_optimization(a_opt, 30, t2_deadline, use_full=True)
        else:
            a_opt = run_lp_optimization(a0, 50, min(t2_deadline, time.monotonic() + init_time), use_full=True)
        
        tracker.update(a_opt)
    
    tracker.report("After Phase 2")
    
    # === Phase 3: LP with perturbation restarts (MAIN PHASE) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP + perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)
    
    t3_deadline = min(deadline - 150, time.monotonic() + 500)
    
    n_restarts = 30
    for restart in range(n_restarts):
        if time.monotonic() > t3_deadline:
            break
        
        # Adaptive noise: start larger, shrink over time
        progress = restart / max(1, n_restarts)
        noise_scale = 0.05 * (1.0 - progress * 0.8) + 0.003
        
        a_pert = tracker.best + rng.normal(0, noise_scale, tracker.best.size)
        a_pert = np.maximum(a_pert, 0.0)
        s_p = a_pert.sum()
        if s_p < 1e-10:
            continue
        a_pert = a_pert / s_p
        
        # LP optimize
        est_time_per_iter = 1.0
        max_iters = max(5, int((t3_deadline - time.monotonic()) / max(1.0, est_time_per_iter) / (n_restarts - restart)))
        max_iters = min(max_iters, 100)
        
        a_opt = run_lp_optimization(a_pert, max_iters, t3_deadline, use_full=True)
        tracker.update(a_opt)
        
        if restart % 5 == 0:
            tracker.report(f"restart {restart}/{n_restarts}")
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if time.monotonic() < deadline - 100:
        n_cur = tracker.best.size
        candidates = [1.25, 0.8, 1.5, 0.67]
        for mult in candidates:
            if time.monotonic() > deadline - 100:
                break
            new_n = int(round(n_cur * mult))
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 16) * 16
            if new_n == n_cur or new_n < 128:
                continue
            
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            sub_dl = min(deadline - 100, time.monotonic() + 50)
            a_opt = run_lp_optimization(a_ext, 40, sub_dl, use_full=True)
            tracker.update(a_opt)
    
    tracker.report("After Phase 4")
    
    # === Phase 5: Final intensive polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 10:
        # Round 1: full LP
        a_f = run_lp_optimization(tracker.best, 300, min(deadline - 20, time.monotonic() + 200), use_full=True)
        tracker.update(a_f)
        
        # Round 2: perturb + LP
        if time.monotonic() < deadline - 15:
            for _ in range(5):
                if time.monotonic() > deadline - 15:
                    break
                a_p = tracker.best + rng.normal(0, 0.01, tracker.best.size)
                a_p = np.maximum(a_p, 0.0)
                a_p = a_p / a_p.sum()
                a_f2 = run_lp_optimization(a_p, 50, min(deadline - 10, time.monotonic() + 30), use_full=True)
                tracker.update(a_f2)
    
    tracker.report("FINAL")
    
    # Ensure valid output
    if tracker.best is None:
        tracker.best = np.ones(512) / 512
        tracker.best_score = compute_M(tracker.best) * 2 * 512
    
    print(f"\n{'='*60}", flush=True)
    print(f"RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
