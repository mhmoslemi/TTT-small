"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        return np.ones(n) / n
    lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
    return np.maximum(v - lam, 0.0)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_conv(a, k_set):
    """Compute subgradient of max_{k in k_set} (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    for k in k_set:
        if k == 0:
            g += 2.0 * a
        else:
            # d/da[i] (a*a)[k] = a[i+k] (if valid) + a[i-k] (if valid)
            # a[i+k] valid when 0 <= i+k < n, i.e., 0 <= i < n-k
            # a[i-k] valid when 0 <= i-k < n, i.e., k <= i < n
            if k < n:
                g[:n - k] += a[k:]
                g[k:] += a[:n - k]
    return g


def nesterov_subgrad(a, max_iters, deadline, lr_init=0.1, lr_decay=0.9995,
                      w_set=None, verbose=False):
    """
    Nesterov accelerated subgradient method on the simplex.
    Minimizes 2n * max_k (a*a)[k] subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_J(a)

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # velocity
    t = 1.0
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute subgradient at y
        conv_y = fast_conv(y)
        M = conv_y.max()
        # Get argmax set (within tolerance)
        tol = 1e-3 * M
        k_set = np.where(conv_y >= M - tol)[0]
        # Cap the set size for stability
        if len(k_set) > 20:
            idx_sorted = np.argsort(-conv_y)[:20]
            k_set = idx_sorted

        g = subgradient_conv(y, k_set.astype(int))

        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t - 1) / t_new
        t = t_new

        v = beta * v + lr * g
        lr *= lr_decay

        a_new = y - v
        a_new = project_simplex(a_new)
        y = a_new

        obj = compute_J(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        if verbose and it % 200 == 0:
            print(f"    Subgrad it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj

    """
    Find the LP direction: minimize t such that
    2*(a+g)[k]*(a+g)[k] linearized <= t at current peak, sum(g)=0, g>=-a.
    Simplified: minimize max_k [2*(a*g)[k] + (g*g)[k]] linearized as 2*(a*g)[k]
    subject to sum(g) = 0, -a <= g <= 0 (only decreasing moves)
    OR: find g on simplex that minimizes max_k (a*g)[k].
    """
    n = len(a)
    a = normalize(a.copy())
    conv_a = fast_conv(a)
    M = conv_a.max()

    # LP: minimize t s.t. (a*g)[k] <= M for all k, sum(g) = 1, g >= 0
    # This finds a direction g that keeps cross-convolution bounded.
    rows = []
    cols = []
    vals = []
    b_ub = []

    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
        b_ub.append(M)

    if not rows:
        return a.copy(), compute_J(a)

    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub), bounds=bounds,
                    method='highs', options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
                # Try interpolation
                best_a = a.copy()
                best_obj = compute_J(a)
                for frac in [0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - frac) * a + frac * g
                    a_try = project_simplex(a_try)
                    obj = compute_J(a_try)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_try.copy()
                return best_a, best_obj
    except Exception:
        pass

    return a.copy(), compute_J(a)


def lp_min_max(a, deadline, verbose=False):
    """
    SCA-LP: minimize t s.t. 2*(a*g)[k] - (a*a)[k] <= t, sum(g)=1, g>=0.
    This linearizes (g*g) around current point a.
    """
    n = len(a)
    a = normalize(a.copy())
    conv_a = fast_conv(a)
    M = conv_a.max()

    # Linearize (g*g)[k] ≈ 2*(a*g)[k] - (a*a)[k]
    # So constraint: 2*(a*g)[k] - (a*a)[k] <= t
    rows = []
    cols = []
    vals = []
    b_ub = []

    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        rows.append(k)
        cols.append(n)  # t variable
        vals.append(-1.0)
        b_ub.append(conv_a[k])

    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq_rows = [0] * n
    A_eq_cols = list(range(n))
    A_eq = coo_matrix((np.ones(n), (A_eq_rows, A_eq_cols)), shape=(1, n + 1)).tocsr()
    b_eq = [1.0]
    bounds = [(0, None)] * n + [(None, None)]

    try:
        r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs', options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
                return g
    except Exception:
        pass

    return None


def paired_coord_descent(a, max_sweeps, deadline, verbose=False):
    """
    Paired coordinate descent: at each step, find the argmax lag k*,
    then for each pair (i, i+k*), try to reduce a[i]*a[i+k*] by
    transferring mass to other positions.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_J(a)
    conv = fast_conv(a)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        M = conv.max()
        k_star = int(np.argmax(conv))
        if k_star == 0:
            # Standard CD for diagonal case
            a = _standard_cd_one_sweep(a, conv, best_obj, deadline)
            conv = fast_conv(a)
        elif k_star < n:
            # Paired: reduce products a[i]*a[i+k_star]
            improved = False
            for i in range(n - k_star):
                j = i + k_star
                prod = a[i] * a[j]
                if prod < 1e-10:
                    continue
                # Try moving mass from the larger of a[i], a[j] to a random other position
                if a[i] > a[j]:
                    src = i
                    dst_candidates = [j]  # reduce j helps
                else:
                    src = j
                    dst_candidates = [i]

                # Try reducing src by small amounts and adding to various targets
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = a[src] * frac
                    if delta < 1e-12:
                        continue
                    # Try moving to position (src + k_star) % n or (src - k_star + n) % n
                    for tgt in [(src + k_star) % n, (src - k_star) % n, (src + 1) % n, (src - 1 + n) % n]:
                        if tgt == src:
                            continue
                        a_try = a.copy()
                        a_try[src] -= delta
                        a_try[tgt] += delta
                        if a_try[src] < 0:
                            continue
                        a_try = normalize(a_try)
                        obj = compute_J(a_try)
                        if obj < best_obj - 1e-12:
                            a = a_try
                            conv = fast_conv(a)
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            break

    return best_a, best_obj


def _standard_cd_one_sweep(a, conv, best_obj, deadline):
    """One sweep of standard coordinate descent."""
    n = len(a)
    s = 1.0  # normalized
    for i in range(n):
        if time.monotonic() > deadline:
            break
        old = a[i]
        lo = max(0, i)
        hi = min(2 * n - 2, i + n - 1)
        a_slice = a[lo - i:hi - i + 1].copy()
        kq = 2 * i

        candidates = [0.0, old * 0.5, old * 0.7, old * 1.3, old * 1.5, 1.0 / n]

        best_val = old
        best_local = 2.0 * n * conv.max()

        for x in candidates:
            x = float(max(0.0, x))
            if x < 1e-14:
                x = 0.0
            delta = x - old
            if abs(delta) < 1e-14:
                continue
            conv[lo:hi + 1] += 2.0 * delta * a_slice
            conv[kq] += delta * delta
            new_s = s - old + x
            if new_s < 1e-10:
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                conv[kq] -= delta * delta
                continue
            new_obj = 2.0 * n * conv.max() / (new_s * new_s)
            if new_obj < best_local:
                best_local = new_obj
                best_val = x
            conv[lo:hi + 1] -= 2.0 * delta * a_slice
            conv[kq] -= delta * delta

        if best_val != old:
            delta = best_val - old
            conv[lo:hi + 1] += 2.0 * delta * a_slice
            conv[kq] += delta * delta
            a[i] = best_val
            s = s - old + best_val

    # Renormalize
    if s > 1e-10:
        a /= s
    return a


def fine_coord_descent(a, max_sweeps, deadline, verbose=False):
    """
    Fine-grained coordinate descent with adaptive candidate values
    based on local curvature estimation.
    """
    n = len(a)
    a = normalize(a.copy())
    conv = fast_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max()
    no_improve = 0
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved_this_sweep = False

        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, 0.5 / n]
            else:
                # Adaptive grid around current value
                step = old * 0.1
                candidates = np.arange(old - 3 * step, old + 3 * step + step / 2, step)
                candidates = np.concatenate([np.array([0.0]), candidates, np.array([old * 2])])
                candidates = np.unique(np.clip(candidates, 0, 100))

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            s = 1.0

            best_val = old
            best_local = 2.0 * n * conv.max()

            for x in candidates:
                x = float(max(0.0, x))
                if abs(x - old) < 1e-14:
                    continue
                delta = x - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                conv[kq] += delta * delta
                new_s = s - old + x
                if new_s < 1e-10:
                    conv[lo:hi + 1] -= 2.0 * delta * a_slice
                    conv[kq] -= delta * delta
                    continue
                new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved_this_sweep = True

        current_obj = 2.0 * n * conv.max() / (a.sum() ** 2) if a.sum() > 1e-10 else float('inf')
        a = normalize(a)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            break

    return best_a, best_obj


def sim_annealing(a, max_iters, deadline, t_init=0.1, t_min=0.001, verbose=False):
    """
    Simulated annealing with structured moves:
    - Single coordinate perturbation
    - Pair swap
    - Block shift
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_J(a)
    rng = np.random.default_rng(67890)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max(1, max_iters)
        T = t_init * (t_min / t_init) ** progress

        move_type = rng.integers(0, 3)

        if move_type == 0:
            # Single coordinate
            i = rng.integers(0, n)
            a_try = a.copy()
            a_try[i] *= np.exp(rng.normal(0, 0.3))
            a_try = normalize(a_try)
        elif move_type == 1:
            # Pair swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            a_try = a.copy()
            amt = min(a[i], a[j]) * rng.uniform(0.1, 0.5)
            a_try[i] -= amt
            a_try[j] += amt
            a_try = normalize(a_try)
        else:
            # Block perturbation
            width = rng.integers(max(2, n // 32), max(3, n // 8))
            start = rng.integers(0, max(1, n - width))
            a_try = a.copy()
            block = a_try[start:start + width]
            block *= np.exp(rng.normal(0, 0.15, width))
            a_try[start:start + width] = block
            a_try = normalize(a_try)

        obj_try = compute_J(a_try)
        obj_curr = compute_J(a)

        if obj_try < obj_curr or rng.random() < np.exp(-(obj_try - obj_curr) / max(T, 1e-10)):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 2000:
            break

    return best_a, best_obj


def kkt_adjust(a, deadline, verbose=False):
    """
    KKT-based adjustment: at optimum, for coordinates with a[i] > 0,
    the subgradient should be roughly constant (equal to the Lagrange multiplier).
    Deviations indicate room for improvement.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_J(a)

    conv = fast_conv(a)
    M = conv.max()
    tol = 1e-4 * M
    k_set = np.where(conv >= M - tol)[0]

    for _ in range(50):
        if time.monotonic() > deadline:
            break

        g = subgradient_conv(a, k_set.astype(int))
        # At KKT: g[i] = lambda for all i with a[i] > 0
        # Move: increase a[i] where g[i] < lambda, decrease where g[i] > lambda
        active = a > 1e-8
        if active.sum() == 0:
            break
        lam = g[active].mean()
        diff = lam - g  # positive where we want to increase
        # Apply small adjustment
        a_try = a + 0.05 * diff
        a_try = project_simplex(a_try)
        obj = compute_J(a_try)
        if obj < best_obj:
            best_obj = obj
            best_a = a_try.copy()
            a = a_try
            conv = fast_conv(a)
            M = conv.max()
            tol = 1e-4 * M
            k_set = np.where(conv >= M - tol)[0]
        else:
            # Try smaller step
            a_try = a + 0.01 * diff
            a_try = project_simplex(a_try)
            obj = compute_J(a_try)
            if obj < best_obj:
                best_obj = obj
                best_a = a_try.copy()
                a = a_try
            else:
                break

    return best_a, best_obj


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining subgradient, LP, paired CD,
    fine CD, simulated annealing, and KKT adjustment.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Subgradient + KKT (250s) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Subgradient descent + KKT (n={n_parent})", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase1_end = t_start + 250.0
    a_work = normalize(parent.copy())

    # Run subgradient with different LR settings
    for lr in [0.05, 0.02, 0.01]:
        if time.monotonic() > phase1_end - 30:
            break
        a_sg, obj_sg = nesterov_subgrad(a_work, max_iters=5000,
                                        deadline=min(phase1_end - 10, time.monotonic() + 80),
                                        lr_init=lr, verbose=False)
        update_best(a_sg)
        a_work = best_seq.copy() if best_seq is not None else a_work
        a_work = normalize(a_work)

    # KKT adjustment
    if time.monotonic() < phase1_end - 20:
        a_kkt, obj_kkt = kkt_adjust(a_work, deadline=phase1_end - 5)
        update_best(a_kkt)

    # === PHASE 2: LP-based optimization (350s) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP-based optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase2_end = t_start + 600.0
    a_work = normalize(best_seq.copy()) if best_seq is not None else normalize(parent.copy())

    # Iterative SCA-LP
    for it in range(50):
        if time.monotonic() > phase2_end - 60:
            break
        g = lp_min_max(a_work, deadline=phase2_end)
        if g is None:
            break
        # Try interpolation
        improved = False
        for frac in [0.5, 0.3, 0.7, 1.0, 0.2]:
            a_try = (1 - frac) * a_work + frac * g
            a_try = normalize(a_try)
            obj = compute_J(a_try)
            if obj < compute_J(a_work) - 1e-10:
                a_work = a_try
                update_best(a_work)
                improved = True
                break
        if not improved:
            # Try LP direction
            a_lp, obj_lp = lp_direction(a_work, deadline=phase2_end)
            update_best(a_lp)
            if obj_lp >= compute_J(a_work) - 1e-10:
                if it > 5:
                    break

    # === PHASE 3: Fine coordinate descent (350s) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Fine coordinate descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase3_end = t_start + 950.0
    a_work = normalize(best_seq.copy()) if best_seq is not None else normalize(parent.copy())

    # Multiple passes of fine CD
    for pass_idx in range(3):
        if time.monotonic() > phase3_end - 30:
            break
        t_pass_end = time.monotonic() + min(100, phase3_end - time.monotonic() - 10)
        a_cd, obj_cd = fine_coord_descent(a_work, max_sweeps=20000, deadline=t_pass_end)
        update_best(a_cd)
        a_work = best_seq.copy() if best_seq is not None else a_work
        a_work = normalize(a_work)

    # === PHASE 4: Simulated annealing + paired CD (250s) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Simulated annealing + paired CD", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase4_end = t_start + 1200.0
    a_work = normalize(best_seq.copy()) if best_seq is not None else normalize(parent.copy())

    # Simulated annealing from best
    if time.monotonic() < phase4_end - 60:
        a_sa, obj_sa = sim_annealing(a_work, max_iters=10000,
                                     deadline=min(phase4_end - 60, time.monotonic() + 80),
                                     t_init=0.05, t_min=0.001)
        update_best(a_sa)

    # Paired CD
    if time.monotonic() < phase4_end - 30:
        a_work = normalize(best_seq.copy()) if best_seq is not None else a_work
        a_pc, obj_pc = paired_coord_descent(a_work, max_sweeps=5000,
                                            deadline=min(phase4_end - 10, time.monotonic() + 120))
        update_best(a_pc)

    # === PHASE 5: Final intensive polish (130s) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase5_end = deadline
    a_work = normalize(best_seq.copy()) if best_seq is not None else normalize(parent.copy())

    # Final fine CD
    if time.monotonic() < phase5_end - 40:
        a_cd, obj_cd = fine_coord_descent(a_work, max_sweeps=50000, deadline=phase5_end - 40)
        update_best(a_cd)

    # Final KKT
    if time.monotonic() < phase5_end - 15:
        a_work = normalize(best_seq.copy()) if best_seq is not None else a_work
        a_kkt, obj_kkt = kkt_adjust(a_work, deadline=phase5_end - 5)
        update_best(a_kkt)

    # Final CD on KKT result
    if time.monotonic() < phase5_end - 5:
        a_work = normalize(best_seq.copy()) if best_seq is not None else a_work
        a_cd, obj_cd = fine_coord_descent(a_work, max_sweeps=20000, deadline=phase5_end - 2)
        update_best(a_cd)

    # === Length exploration (if time remains) ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        print(f"\n{'=' * 60}", flush=True)
        print(f"Length exploration (budget {remaining:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        for new_n in [n_parent + 64, n_parent - 64, n_parent + 128, n_parent // 2]:
            if time.monotonic() > deadline - 10:
                break
            if new_n < 64:
                continue
            alloc = min(30, deadline - time.monotonic() - 10)
            if alloc < 15:
                break
            a_new = resample_seq(best_seq, new_n)
            a_new = normalize(a_new)
            a_opt, obj_opt = fine_coord_descent(a_new, max_sweeps=5000,
                                                deadline=time.monotonic() + alloc)
            update_best(a_opt)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
