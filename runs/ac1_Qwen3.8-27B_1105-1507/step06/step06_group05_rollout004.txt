def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

continuation on temperature, LP polishing, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx + s > cssv
    if not np.any(cond):
        return v / n  # fallback
    rho = int(np.max(idx[cond]))
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)


def compute_smoothing_grad(a, tau):
    """
    Compute gradient of tau*log(sum(exp(conv(a,a)/tau))).
    Returns (gradient, conv) tuple.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    
    # Compute weights (softmax of conv/tau)
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500.0, 500.0)
    weights = np.exp(shifted)
    wsum = float(np.sum(weights))
    
    if wsum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= wsum
    
    # Gradient: grad[i] = 2 * sum_j weights[i+j] * a[j]
    # This is 2 * cross_correlation(weights_extended, a) evaluated at [0, n)
    size = 1
    while size < 3 * n:
        size <<= 1
    wf = rfft(weights, size)
    af = rfft(a[::-1], size)
    corr = irfft(wf * af, size)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    
    return grad, conv


def fista_optimize(a, time_budget=60.0, tau_start=None, tau_end=None,
                   step_init=0.05, restart_every=500, verbose=False):
    """
    FISTA optimization with log-sum-exp smoothing and continuation.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start_time = time.monotonic()
    
    if tau_start is None:
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau_start = cmax / (n * 0.05)
    if tau_end is None:
        tau_end = tau_start * 0.01
    
    tau = tau_start
    a_curr = a.copy()
    a_prev = a.copy()
    t_mom = 1.0
    step = step_init
    
    # Continuation loop
    n_tau_levels = 8
    tau_schedule = np.geomspace(tau_start, tau_end, n_tau_levels)
    
    for tau_level in range(n_tau_levels):
        if time.monotonic() - start_time > time_budget:
            break
        tau = tau_schedule[tau_level]
        level_start = time.monotonic()
        level_budget = time_budget - (level_start - start_time)
        level_budget = min(level_budget, time_budget / n_tau_levels * 1.5)
        
        a_curr = best_a.copy()
        a_prev = a_curr.copy()
        t_mom = 1.0
        step = step_init
        
        stagnant = 0
        
        while time.monotonic() - level_start < level_budget:
            grad, conv = compute_smoothing_grad(a_curr, tau)
            
            # FISTA update
            a_new = a_prev - step * grad
            a_new = project_simplex(a_new, 1.0)
            
            # Momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            a_prev = a_new + ((t_mom - 1.0) / t_new) * (a_new - a_curr)
            
            a_curr = a_new
            t_mom = t_new
            
            # Evaluate and update best
            if (int(t_mom) % 50) == 0:
                obj = compute_obj(a_curr)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_curr.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                
                if stagnant > 6:
                    # Restart momentum
                    a_prev = a_curr.copy()
                    t_mom = 1.0
                    step *= 0.7
                    stagnant = 0
                    if step < 0.001:
                        break
                        
    # Final evaluation
    obj = compute_obj(a_curr)
    if obj < best_obj:
        best_a = a_curr.copy()
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_polish(a, max_iters=10, time_budget=60.0):
    """LP-based polishing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start_time = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget - 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            remaining = time_budget - (time.monotonic() - start_time)
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(5, int(remaining * 0.5))})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s < 1e-10:
                break
            g = g / s
        except Exception:
            break
        
        # Try several blend factors
        best_blend = None
        best_blend_obj = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend = a_try
        
        if best_blend is not None:
            a = best_blend
            if best_blend_obj < best_obj:
                best_obj = best_blend_obj
                best_a = a.copy()
        else:
            break
    
    return best_a


def coordinate_descent(a, time_budget=20.0):
    """Coordinate descent polishing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start_time = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start_time < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors
        j_range = range(max(0, k_star - n + 1), min(n, k_star + 1))
        contribs = []
        for j in j_range:
            ki = k_star - j
            if 0 <= ki < n:
                contribs.append((a[j] * a[ki], j, ki))
        contribs.sort(reverse=True)
        
        improved = False
        for _, j1, j2 in contribs[:8]:
            if time.monotonic() - start_time > time_budget:
                break
            if j1 == j2:
                continue
            
            for frac in [0.03, 0.05, 0.1]:
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue
                
                a_try = a.copy()
                delta = frac * a_try[j1]
                if a_try[j1] < delta:
                    continue
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved:
            # Random perturbation
            if rng.random() < 0.3:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

    s = float(np.sum(a))

        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))

def project_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx + s > cssv
    if not np.any(cond):
        return np.full(n, s / n)
    rho = int(np.max(idx[cond]))
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)
