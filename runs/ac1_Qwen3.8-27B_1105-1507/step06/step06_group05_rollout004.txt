import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def subgradient_of_peak(a, k_star):
    """Compute subgradient of f_{k_star}(a) = (a*a)[k_star].
    grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # grad[i] = 2*a[k_star - i] for i where 0 <= k_star - i < n
    # i.e., k_star - n + 1 <= i <= k_star, clipped to [0, n-1]
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    if i_lo <= i_hi:
        # For i in [i_lo, i_hi], k_star - i is in [0, n-1]
        grad[i_lo:i_hi + 1] = 2.0 * a[k_star - i_hi:k_star - i_lo + 1][::-1]
    return grad


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=60):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section_min(f, lo, hi, max_iter=30):
    """Golden section search for minimum."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 0.005:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_alternate_step(a, time_budget=30.0):
    """One LP alternating step with multi-M and golden section line search."""
    n = len(a)
    conv = compute_conv(a)
    M_base = float(np.max(conv))
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    # Try multiple M values
    m_factors = [1.0, 1.02, 0.97, 1.05]
    for mf in m_factors:
        if time.monotonic() > time_budget + (lp_alternate_step._start or 0):
            break
        M = M_base * mf
        g = solve_lp(a, M, time_limit=10)
        if g is None:
            continue
        
        # Golden section for blend
        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)
        
        t_opt = golden_section_min(obj_at_t, 0.0, 1.0, max_iter=20)
        a_try = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a


def projected_subgradient(a, max_iters=500, time_budget=60.0, momentum=0.9,
                          step_init=0.02, k_top=3):
    """
    Projected subgradient method with heavy-ball momentum.
    Uses top-k peak weighting for smoother gradient.
    """
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    step = step_init
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Find top-k peaks within 15% of max
        threshold = cmax * 0.85
        peak_indices = np.where(conv >= threshold)[0]
        
        # Compute weighted subgradient
        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(conv))])
        
        # Weight by how close to max
        weights = np.exp(-(conv[peak_indices] - cmax) * 5.0)
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for pk, w in zip(peak_indices, weights):
            grad += w * subgradient_of_peak(a, int(pk))
        
        # Center gradient (remove component in direction of 1)
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - step * grad_centered
        
        # Update and project
        a_raw = a + v
        a_new = project_simplex(a_raw)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Adaptive step: increase slightly if improving
            step = min(step * 1.01, 0.08)
        else:
            # Step size reduction
            step *= 0.7
            if step < 1e-8:
                step = step_init * 0.1
                v *= 0.5
            # Revert
            a = a  # keep old a
            v = momentum * v  # keep momentum direction
        
        # Decay momentum slightly over time
        momentum = max(0.8, momentum - 0.0002)
    
    return best_a


def peak_shaving(a, time_budget=20.0):
    """
    Coordinate descent that specifically targets the top convolution peaks.
    For each peak, try to reduce it by adjusting contributing variables.
    """
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        cmax = float(conv[k_star])
        
        # Find top contributors to this peak
        # (a*a)[k_star] = sum_j a[j] * a[k_star - j]
        contrib = []
        for j in range(n):
            j2 = k_star - j
            if 0 <= j2 < n:
                contrib.append((a[j] * a[j2], j, j2))
        contrib.sort(reverse=True)
        
        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2 or peak_val < 1e-10:
                continue
            
            # Try moving mass from the larger contributor away
            if a[j1] >= a[j2]:
                src = j1
            else:
                src = j2
            
            # Try several target positions (far from peak contributors)
            for _ in range(5):
                tgt = int(rng.integers(0, n))
                if tgt == src or tgt == j1 or tgt == j2:
                    continue
                
                # Try multiple delta values
                for delta_frac in [0.03, 0.05, 0.08, 0.12]:
                    delta = delta_frac * a[src]
                    if delta <= 0:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    if a_try[src] < 0:
                        continue
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Occasional random perturbation to escape
            if rng.random() < 0.3:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                break  # Stuck, stop
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)

                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")
    
    # =========================================================
    # Phase 5: Final intensive polish (10% time)
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")
    
    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())
        
        for rnd in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.2, 25.0)
            
            # Subgradient with very small steps
            a_pol = projected_subgradient(a_final, max_iters=200, time_budget=alloc * 0.4,
                                          step_init=0.005, momentum=0.85)
            
            # Peak shaving
            if alloc > 8:
                a_pol = peak_shaving(a_pol, time_budget=min(10.0, alloc * 0.3))
            
            # LP if feasible
            if n_final <= 800 and alloc > 12:
                a_pol = lp_alternate_step(a_pol, time_budget=min(8.0, alloc * 0.2))
            
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {rnd}: {best_score:.8f}")
            else:
                a_final = a_pol
                print(f"  Polish {rnd}: {obj_pol:.8f} (no improvement)")
    
    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100
    
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    
    return result
