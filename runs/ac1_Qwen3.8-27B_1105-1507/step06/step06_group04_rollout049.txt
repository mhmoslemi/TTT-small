def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):

        return np.inf
    conv = compute_conv(a)

    return 2.0 * n * max_b / (s * s)

            a = np.maximum(a, 0)

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def compute_gradient_fft(a, weights, fft_size=None):
    """
    Compute gradient of sum_k w[k]*conv(k) w.r.t. a.
    grad[i] = 2 * sum_j a[j] * w[i+j] (with w zero-padded to length 2n-1)
    Uses: ifft(conj(fft(a)) * fft(w_padded)) gives sum_j a[j]*w[m+j]
    """
    n = len(a)
    if fft_size is None:
        fft_size = 1
        while fft_size < 2 * n - 1:
            fft_size <<= 1

    w_padded = np.zeros(fft_size)
    w_padded[:len(weights)] = weights

    A = rfft(a, fft_size)
    W = rfft(w_padded, fft_size)
    corr = irfft(np.conj(A) * W, fft_size)

    grad = 2.0 * corr[:n]
    return grad


def adaptive_temp_subgradient(a, max_iters=1000, time_budget=60.0,
                              tau_init=0.008, tau_min=0.0005,
                              mu=0.92, step_init=0.04):
    """
    Adaptive-temperature subgradient with Nesterov momentum.
    Temperature adapts: if few peaks are active, lower temp to focus;
    if many peaks are active, raise temp to balance.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu_local = mu
    step = step_init
    tau = tau_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        cmax = float(np.max(conv))

        # Adaptive temperature: count peaks within 1% of max
        thresh = cmax * 0.99
        num_active = np.sum(conv >= thresh)
        # If few active peaks, use lower temp (sharper focus)
        # If many active peaks, use higher temp (broader focus)
        adaptive_tau = tau * np.sqrt(max(1, num_active) / max(1, min(n // 10, num_active)))
        adaptive_tau = np.clip(adaptive_tau, tau_min, 0.05)

        # Softmax weights
        shifted = np.clip((conv - cmax) / adaptive_tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via FFT
        grad = compute_gradient_fft(a, weights, fft_size)

        # Center gradient (project onto zero-sum tangent space of simplex)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Nesterov momentum
        v = mu_local * v - step * grad_centered
        a_new = a + v
        a_new = normalize_simplex(a_new)

        obj_new = compute_obj(a_new)
        conv_new = compute_conv(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                # Reduce momentum and step
                mu_local *= 0.95
                step *= 0.8
                v *= 0.3
                stall = 0
            # Still update a (momentum might help in future)
            a = a_new
            conv = conv_new

        # Adaptive step
        if it % 100 == 99:
            if stall == 0:
                step = min(step * 1.03, 0.12)
            else:
                step *= 0.97

        # Decay base temperature slowly
        tau *= 0.9997
        tau = max(tau, tau_min)

    return best_a, best_obj


def sharp_peak_subgradient(a, max_iters=500, time_budget=40.0):
    """
    Sharp-focus subgradient: always focus on the single highest peak.
    Good for the final polish when the max is unique.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu = 0.85
    step = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Focus on top 3 peaks with exp weighting
        cmax = float(np.max(conv))
        top3 = np.argsort(conv)[-3:]
        weights = np.zeros(len(conv))
        for rank, idx in enumerate(top3):
            weights[idx] = np.exp(-0.5 * rank)
        weights /= np.sum(weights)

        grad = compute_gradient_fft(a, weights, fft_size)
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break

        v = mu * v - step * grad_centered
        a_new = a + v
        a_new = normalize_simplex(a_new)

        obj_new = compute_obj(a_new)
        conv_new = compute_conv(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            step = min(step * 1.02, 0.1)
        else:
            step *= 0.9
            v *= 0.5
            a = a_new
            conv = conv_new

    return best_a, best_obj


def peak_surgery(a, time_budget=30.0, num_peaks=3):
    """
    For each of the top peaks, take a step in the exact negative gradient
    direction for that specific peak (not weighted).
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        top_k = np.argsort(conv)[-num_peaks:]

        improved = False
        for k in top_k:
            # Exact gradient for this single peak:
            # grad[i] = 2 * a[k-i] for 0 <= k-i < n
            grad_k = np.zeros(n)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # a[k-i] for i in [lo, hi] means a[k-hi : k-lo+1] reversed
                grad_k[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

            grad_centered = grad_k - np.mean(grad_k)
            gn = np.linalg.norm(grad_centered)
            if gn < 1e-15:
                continue

            # Try different step sizes
            for alpha in [0.02, 0.01, 0.005, 0.001]:
                a_try = a - alpha * grad_centered / gn
                a_try = normalize_simplex(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break

        if not improved and it > 20:
            # Perturb to escape
            rng = np.random.default_rng(it * 7919 + 13)
            a = best_a * np.exp(rng.normal(0, 0.008, n))
            a = normalize_simplex(a)

    return best_a, best_obj


def diverse_starts(n, seed=42):
    """Generate diverse initial sequences of length n."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Triangular
    tri = 1.0 - np.abs(i - center) / center
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # Various Gaussian widths
    for width_frac in [0.15, 0.25, 0.4, 0.6]:
        bell = np.exp(-((i - center) ** 2) / (2 * (width_frac * n) ** 2))
        starts.append(bell / np.sum(bell))

    # Cosine window (Hann-like)
    hann = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1))) if n > 1 else np.ones(n)
    hann = np.maximum(hann, 1e-6)
    starts.append(hann / np.sum(hann))

    # Raised cosine with different beta
    for beta in [0.5, 1.0, 2.0]:
        x = np.abs(i - center) / (n / 2)
        rc = np.cos(np.pi * x * beta / 2) ** 2 for x in x if x < 1/beta else 0
        # Actually let me do this properly
        rc = np.zeros(n)
        for j in range(n):
            xj = abs(j - center) / (n / 2)
            if xj < 1.0 / beta:
                rc[j] = np.cos(np.pi * beta * xj / 2) ** 2
        if np.sum(rc) > 1e-10:
            starts.append(rc / np.sum(rc))

    # Sinusoidal modulation on Gaussian base
    for freq in [2, 3, 5, 7]:
        for amp in [0.3, 0.5]:
            bell = np.exp(-((i - center) ** 2) / (2 * (0.35 * n) ** 2))
            mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n)
            a = bell * mod
            a = np.maximum(a, 0)
            if np.sum(a) > 1e-10:
                starts.append(a / np.sum(a))

    # Random smooth
    for _ in range(4):
        a = np.abs(rng.normal(1, 0.4, n))
        # Smooth
        from scipy.ndimage import uniform_filter1d
        a = uniform_filter1d(a, size=max(3, n // 40))
        starts.append(a / np.sum(a))

    return starts


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            norm_parent = normalize_simplex(parent)
            obj_parent = compute_obj(norm_parent)
            print(f"Parent (n={parent.size}): obj={obj_parent:.8f}")
            return norm_parent.copy(), obj_parent
    except Exception:
        pass
    return None, np.inf


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def polish(a, time_budget=120.0):
    """
    Full polishing pipeline: alternate between adaptive subgradient,
    sharp-peak subgradient, and peak surgery.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    methods = [
        ('adaptive', adaptive_temp_subgradient),
        ('sharp', sharp_peak_subgradient),
        ('surgery', peak_surgery),
    ]
    mi = 0
    total_stall = 0

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        name, func = methods[mi % len(methods)]
        mi += 1

        alloc = min(remaining * 0.25, 45.0)
        if alloc < 8:
            break

        if name == 'adaptive':
            a_new, obj_new = func(best_a, max_iters=int(alloc * 12), time_budget=alloc)
        elif name == 'sharp':
            a_new, obj_new = func(best_a, max_iters=int(alloc * 10), time_budget=alloc)
        else:
            a_new, obj_new = func(best_a, time_budget=alloc)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            total_stall = 0
            print(f"      {name}: {best_obj:.8f}")
        else:
            total_stall += 1
            if total_stall > 6:
                # Perturb and try again
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                scale = 0.01 + 0.005 * (total_stall % 3)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize_simplex(a_pert)
                a_pert, obj_pert = adaptive_temp_subgradient(a_pert,
                                                             max_iters=200, time_budget=8.0)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    total_stall = 0
                    print(f"      perturb+opt: {best_obj:.8f}")
                else:
                    total_stall = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-length diverse exploration + intensive polishing
    + length perturbation + final refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: score={best_score:.8f}, n={n_parent}")

    # Phase 1: Explore lengths with diverse starts
    print("\n=== Phase 1: Exploration ===")

    n_candidates = sorted(set([
        n_parent,
        max(300, n_parent - 100),
        max(300, n_parent - 50),
        min(1300, n_parent + 50),
        min(1300, n_parent + 100),
        min(1500, n_parent + 200),
    ]))

    all_candidates = []

    for n in n_candidates:
        if time.monotonic() > deadline - 350:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        starts = diverse_starts(n, seed=seed + n)
        if best_seq is not None and len(best_seq) == n:
            starts.insert(0, best_seq.copy())

        # Limit starts to fit in time
        max_starts = min(len(starts), max(4, int(remaining // 25)))
        starts = starts[:max_starts]

        alloc_per = min(remaining / len(starts) * 0.7, 20.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 250:
                break
            if alloc_per < 5:
                break

            t1 = min(alloc_per * 0.5, 12.0)
            a_opt, obj_opt = adaptive_temp_subgradient(s,
                                                       max_iters=int(t1 * 10),
                                                       time_budget=t1)

            # Quick peak surgery
            t2 = min(alloc_per * 0.25, 8.0)
            if t2 > 3 and time.monotonic() < deadline - 200:
                a_opt2, obj_opt2 = peak_surgery(a_opt, time_budget=t2, num_peaks=3)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} start{si}: {obj_opt:.8f} ** NEW BEST **")

            all_candidates.append((obj_opt, a_opt.copy()))

    all_candidates.sort(key=lambda x: x[0])
    top_cands = all_candidates[:min(6, len(all_candidates))]
    print(f"Top: {[f'{c[0]:.6f}' for c in top_cands[:4]]}")
    print(f"Best: {best_score:.8f}")

    # Phase 2: Intensive polish of top candidates
    print("\n=== Phase 2: Polishing ===")

    for ci, (obj_c, seq_c) in enumerate(top_cands):
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        alloc = min(remaining * 0.3, 150.0)
        n_c = len(seq_c)
        print(f"  Cand {ci} (n={n_c}, obj={obj_c:.8f}): alloc={alloc:.0f}s")

        a_pol, obj_pol = polish(seq_c.copy(), time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** NEW BEST: {best_score:.8f} **")
        else:
            print(f"  Got {obj_pol:.8f} (no improvement)")

    # Phase 3: Length perturbation of best
    print(f"\n=== Phase 3: Length perturbation (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        deltas = [-100, -50, 50, 100, 150, -150, 200, -200]
        test_lengths = sorted(set([n_best + d for d in deltas
                                   if 250 <= n_best + d <= 1500]))

        for n_test in test_lengths:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            a_res = resample_sequence(best_seq, n_test)
            alloc = min(remaining * 0.2, 50.0)
            a_pol, obj_pol = polish(a_res, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** {best_score:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f}")

    # Phase 4: Final intensive refinement
    print(f"\n=== Phase 4: Final refinement (best={best_score:.8f}) ===")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_f, obj_f = polish(best_seq, time_budget=min(remaining * 0.5, 200.0))
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Polish: {best_score:.8f}")

        # Perturbation sweep
        remaining = deadline - time.monotonic()
        if remaining > 15:
            n_f = len(best_seq)
            rng = np.random.default_rng(seed + 31337)
            for p in range(20):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.003 + 0.002 * (p % 5)
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_f))
                a_pert = normalize_simplex(a_pert)

                if t_a < 4:
                    break

                a_pert, obj_pert = adaptive_temp_subgradient(a_pert,

                                                             time_budget=t_a)
