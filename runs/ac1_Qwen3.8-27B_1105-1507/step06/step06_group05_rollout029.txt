def lse_objective_and_grad(a, T):
    """Compute log-sum-exp objective and gradient for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    
    # Clip for numerical stability
    conv_clipped = np.clip(conv, -1e10, 1e10)
    max_conv = np.max(conv_clipped)
    exp_vals = np.exp((conv_clipped - max_conv) / T)
    log_sum_exp = max_conv / T + np.log(np.sum(exp_vals))
    
    # Objective: 2*n*log_sum_exp (since we normalized sum(a)=1)
    obj = 2.0 * n * log_sum_exp
    
    # Weights
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient: grad[i] = (2/T) * conv(a_rev, w)[n-1+i]
    a_rev = a[::-1]
    L = 1
    while L < n + 2*n - 1 - 1:  # n + (2n-1) - 1 = 3n-2
        L <<= 1
    A = rfft(a_rev, L)
    W = rfft(w, L)
    conv_aw = irfft(A * W, L)
    grad = (2.0 / T) * conv_aw[n-1:2*n-1]
    
    return obj, grad

"""
Multi-scale LSE (log-sum-exp) optimization with L-BFGS, temperature annealing,
diverse initializations, LP polishing, and multi-n exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_obj_grad(a, T):
    """
    Compute the LSE-smoothed objective and its gradient.
    Objective: 2*n*log(sum(exp(conv/T))) / (sum(a))^2
    This is a smooth approximation to 2*n*max(conv)/(sum(a))^2.
    """
    n = len(a)
    conv = compute_conv(a)
    s = float(np.sum(a))
    
    if s < 1e-15:
        return np.inf, np.zeros(n)
    
    # Log-sum-exp for numerical stability
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / T
    # Clip for stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = cmax / T + np.log(np.sum(exp_vals))
    lse = log_sum * s * s  # Unnormalized LSE * (sum)^2 ... 
    
    # Actually let me just compute the objective directly
    # F(a) = 2*n*lse(T, conv) / s^2
    # where lse(T, conv) = log(sum(exp(conv[k]/T)))
    
    obj = 2.0 * n * log_sum / (s * s)
    
    # Weights for gradient
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient of lse wrt a[i] = (1/T) * sum_k w[k] * d(conv[k])/da[i]
    # d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n, else 0
    # So grad_lse[i] = (2/T) * sum_m a[m] * w[i+m]  (m from 0 to n-1, i+m from i to i+n-1)
    
    # conv(a[::-1], w)[n-1 + i] = sum_m a[n-1-m] * w[n-1+i-m]
    # Let j = n-1-m: sum_j a[j] * w[n-1+i-(n-1-j)] = sum_j a[j] * w[i+j]  ✓
    
    a_rev = a[::-1]
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    A = rfft(a_rev, L)
    W = rfft(w, L)
    conv_aw = irfft(A * W, L)
    grad_lse = (2.0 / T) * conv_aw[n - 1:2 * n - 1]
    
    # Full gradient of F(a) = 2*n*lse/s^2:
    # dF/da[i] = (2*n/s^2) * grad_lse[i] - (2*n*lse/s^3) * 1
    #          = (2*n/s^2) * (grad_lse[i] - lse/s)
    
    lse_val = log_sum  # This is lse(T, conv)
    grad = (2.0 * n / (s * s)) * (grad_lse - lse_val / s)
    
    return obj, grad

def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def soft_max_step(a, T, step_size, n_steps=50, time_budget=30.0):

def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_obj_grad(a, T):
    """
    Compute the LSE-smoothed objective and its analytical gradient.
    F(a) = 2*n*log(sum(exp(conv[k]/T))) / (sum(a))^2
    Gradient derived via chain rule, computed efficiently via FFT.
    """
    n = len(a)
    conv = compute_conv(a)
    s = float(np.sum(a))

    if s < 1e-15:
        return np.inf, np.zeros(n)

    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / T, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    log_sum = cmax / T + np.log(np.sum(exp_vals))
    w = exp_vals / np.sum(exp_vals)

    obj = 2.0 * n * log_sum / (s * s)

    # dLSE/da[i] = (2/T) * sum_m a[m] * w[i+m]
    # Computed as conv(a[::-1], w)[n-1 : 2n-1]
    a_rev = a[::-1]
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    A = rfft(a_rev, L)
    W = rfft(w, L)
    conv_aw = irfft(A * W, L)
    dLSE = (2.0 / T) * conv_aw[n - 1:2 * n - 1]

    # dF/da[i] = (2n/s^2) * (dLSE[i] - 2*LSE/s)
    grad = (2.0 * n / (s * s)) * (dLSE - 2.0 * log_sum / s)

    return obj, grad


def lse_obj_grad_logspace(x, T):
    """
    LSE objective and gradient in log-space: a[i] = exp(x[i]).
    da/dx[i] = a[i], so dF/dx[i] = dF/da[i] * a[i].
    """
    a = np.exp(x - np.max(x))  # For numerical stability (scale doesn't matter)
    obj, grad_a = lse_obj_grad(a, T)
    grad_x = grad_a * a
    return obj, grad_x


def lse_optimize(a, T_schedule, max_iter=150, time_budget=60.0, verbose=False, use_logspace=False):
    """
    L-BFGS optimization of LSE-smoothed objective with temperature schedule.
    T_schedule: list of temperatures to try (hot to cold).
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for T in T_schedule:
        if time.monotonic() - start > time_budget:
            break

        if use_logspace:
            x0 = np.log(np.maximum(a, 1e-30))
            fun = lse_obj_grad_logspace
        else:
            x0 = a

            def fun_direct(x):
                x = np.maximum(x, 1e-30)
                return lse_obj_grad(x, T)

            fun = fun_direct

        try:
            res = minimize(fun, x0, method='L-BFGS-B',
                          jac=True,
                          options={'maxiter': max_iter, 'gtol': 1e-8,
                                   'maxls': 80, 'ftol': 1e-12})

            if use_logspace:
                a_new = np.exp(res.x - np.max(res.x))
            else:
                a_new = np.maximum(res.x, 0.0)

            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      T={T:.6f}: obj={obj_new:.8f} **")
            else:
                if verbose:
                    print(f"      T={T:.6f}: {obj_new:.8f} (no impr)")

            a = a_new
        except Exception as e:
            if verbose:
                print(f"      T={T:.6f}: L-BFGS error: {e}")

    return best_a


def lse_optimize_fminbound(a, T_schedule, time_budget=60.0, verbose=False):
    """
    Alternative: use Nelder-Mead or Powell for robustness when L-BFGS fails.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for T in T_schedule:
        if time.monotonic() - start > time_budget:
            break

        def fun_direct(x):
            x = np.maximum(x, 0.0)
            s = float(np.sum(x))
            if s < 1e-15:
                return 1e10
            obj, _ = lse_obj_grad(x, T)
            return obj

        try:
            res = minimize(fun_direct, a, method='Nelder-Mead',
                          options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-10,
                                   'adaptive': True})
            a_new = np.maximum(res.x, 0.0)
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      NM T={T:.6f}: {obj_new:.8f} **")
            a = a_new
        except Exception:
            pass

    return best_a


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_polish(a, max_iters=25, time_budget=80.0, verbose=False):
    """LP-based Frank-Wolfe polishing with adaptive line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=15)
        if g is None:
            break

        # Adaptive line search: try multiple t values
        best_t = 0.0
        best_obj_t = best_obj
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]
        for t in t_candidates:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t

        if best_t > 0:
            a = normalize((1.0 - best_t) * a + best_t * g)
            if best_obj_t < best_obj:
                best_obj = best_obj_t
                best_a = a.copy()
                if verbose:
                    print(f"      LP iter {it}: t={best_t:.2f}, obj={best_obj:.8f} **")
        else:
            if verbose:
                print(f"      LP iter {it}: no improving direction")
            break

    return best_a


def soft_max_gradient(a, T, step_size, n_steps=100, time_budget=30.0, verbose=False):
    """Soft-max gradient descent with adaptive step size."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        _, grad = lse_obj_grad(a, T)

        # Center gradient (scale invariance)
        grad_centered = grad - np.mean(grad)

        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slightly increase step if making good progress
            step_size *= 1.05
        else:
            step_size *= 0.6
            if step_size < 1e-12:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    strategy = rng.integers(0, 8)
    t = np.arange(n, dtype=np.float64) / n
    if strategy == 0:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif strategy == 1:
        # Two Gaussians
        c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
        s1, s2 = rng.uniform(0.05, 0.15), rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / s1) ** 2) + np.exp(-0.5 * ((t - c2) / s2) ** 2)
    elif strategy == 2:
        # Periodic
        freq = rng.uniform(3, 12)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    elif strategy == 3:
        # Power law decay
        a = (t + 0.01) ** (-rng.uniform(0.5, 3.0))
    elif strategy == 4:
        # Sine window
        a = np.sin(np.pi * t) ** rng.uniform(1.0, 4.0)
    elif strategy == 5:
        # Ramp
        a = t ** rng.uniform(0.5, 3.0)
    elif strategy == 6:
        # Checkered
        freq = rng.uniform(5, 20)
        a = 0.5 + 0.5 * np.abs(np.sin(np.pi * freq * t))
    else:
        # Dirichlet
        a = rng.dirichlet(np.ones(n) * rng.uniform(0.1, 1.0)) * n
    return np.maximum(a, 1e-6)


def gradient_check(a, T, verbose=False):
    """Verify analytical gradient against numerical gradient."""
    n = len(a)
    _, grad_analytic = lse_obj_grad(a, T)

    eps = 1e-7
    grad_numeric = np.zeros(n)
    base_obj, _ = lse_obj_grad(a, T)

    for i in range(min(n, 50)):  # Check first 50 for speed
        a_plus = a.copy()
        a_plus[i] += eps
        obj_plus, _ = lse_obj_grad(a_plus, T)

        a_minus = a.copy()
        a_minus[i] -= eps
        obj_minus, _ = lse_obj_grad(a_minus, T)

        grad_numeric[i] = (obj_plus - obj_minus) / (2 * eps)

    # Compare on checked entries
    diff = np.max(np.abs(grad_analytic[:50] - grad_numeric))
    if verbose:
        print(f"    Gradient check: max diff = {diff:.2e}")
    return diff
