def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):
    """
    Solve: min t subject to (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def lse_val(cmax, shifted, w_sum, tau):
    """Compute the log-sum-exp value."""
    return cmax + np.log(w_sum) / tau if w_sum > 0 else cmax


def armijo_line_search(f, a, direction, f0, step, c1=0.1, rho=0.5, max_iter=20):
    """
    Armijo backtracking line search.
    f: objective function taking a sequence
    a: current point
    direction: direction to move (already a valid normalized sequence)
    Returns optimal t in [0, 1].
    """
    t = step
    for _ in range(max_iter):
        a_try = (1.0 - t) * a + t * direction
        a_try = normalize_simplex(a_try)
        f_try = f(a_try)
        if f_try <= f0 - c1 * t * step:  # simplified Armijo
            return t
        t *= rho
    return 0.0

    Log-sum-exp gradient descent with adaptive temperature and backtracking.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    # Adaptive parameters
    tau = 0.01  # initial temperature
    tau_decay = 0.997
    tau_min = 0.0005
    step = 0.15
    mu = 0.7  # momentum
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute LSE gradient
        grad, conv, lse = lse_gradient(a, tau, fft_size)
        
        # Center gradient (stay on simplex)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Momentum update
        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)

        # Evaluate
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slightly increase step on success
            step = min(step * 1.02, 0.3)
        else:
            # Try without momentum
            a_try = a - step * grad_centered / grad_norm
            a_try = normalize_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                step = min(step * 1.01, 0.25)
            else:
                # Reduce step
                step *= 0.85
                v *= 0.5

        # Decay temperature
        tau = max(tau * tau_decay, tau_min)

        # Periodic reheating
        if it % 100 == 99:
            tau = max(tau * 3.0, 0.005)

        # Adaptive momentum
        mu = 0.5 + 0.4 * (it / max(200, max_iters))

    return best_a, best_obj


def peak_targeted_subgradient(a, top_k=5, max_iters=300, time_budget=40.0):
    """
    Subgradient method targeting only the top-K peaks of the convolution.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    step = 0.1
    mu = 0.75
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top K peak indices
        top_idx = np.argsort(conv)[-top_k:]
        weights = np.zeros(len(conv))
        for k in top_idx:
            weights[k] = 1.0 / top_k

        # Compute subgradient: grad[i] = 2 * sum_k w[k] * (a[i+k] + a[i-k])_bounded
        w_full = np.zeros(fft_size)
        w_full[:len(weights)] = weights
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a

        W = rfft(w_full, fft_size)
        A = rfft(a_pad, fft_size)
        conv_wa = irfft(W * A, fft_size)

        grad = 2.0 * conv_wa[:n]
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.02, 0.2)
        else:
            step *= 0.9
            v *= 0.5

        # Slowly increase top_k to focus more broadly
        if it % 50 == 49 and top_k < 20:
            top_k += 1

    return best_a, best_obj


def lp_alternate_enhanced(a, max_iters=40, time_budget=80.0):
    """
    Enhanced LP alternating with adaptive step size and both LP formulations.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    use_minpeak = True
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        if use_minpeak:
            g, t_opt = solve_minpeak_lp(a, time_limit=min(15, int(remaining) - 3))
        else:
            conv = compute_conv(a)
            M = float(np.max(conv))
            g = solve_maxsum_lp(a, M, time_limit=min(15, int(remaining) - 3))

        use_minpeak = not use_minpeak

        if g is None:
            stall += 1
            if stall > 8:
                break
            continue

        # Try multiple step sizes
        cands = [0.3, 0.5, 0.7, 0.8, 1.0, 0.15, 0.2]
        best_t = 0.5
        best_t_obj = best_obj
        best_a_local = best_a.copy()

        for t in cands:
            a_try = (1.0 - t) * a + t * g
            a_try = normalize_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local
            a = best_a_local
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb and try again
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = normalize_simplex(a)
                stall = 0

    return best_a, best_obj


def hann_window(n, center=True):
    """Generate a Hann window of length n."""
    i = np.arange(n)
    if center:
        w = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1))) if n > 1 else np.array([1.0])
    else:
        w = 0.5 * (1 - np.cos(2 * np.pi * i / n))
    return w


def generate_starts(n, rng):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n)
    center = (n - 1) / 2.0

    # Hann window
    w = hann_window(n)
    starts.append(w / np.sum(w))

    # Triangular
    tri = 1.0 - np.abs(i - center) / center
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 5.0) ** 2))
    starts.append(bell / np.sum(bell))

    # Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # Two-sided (high at edges)
    edge = np.exp(-((np.abs(i - center) - center / 2) ** 2) / (n / 10.0))
    edge = np.maximum(edge, 0.01)
    starts.append(edge / np.sum(edge))

    # Sinusoidal modulation on bell
    for freq in [2, 3, 5]:
        mod = 1.0 + 0.5 * np.cos(2 * np.pi * freq * i / n)
        seq = bell_w * mod
        seq = np.maximum(seq, 0)
        s = np.sum(seq)
        if s > 1e-10:
            starts.append(seq / s)

    # Random smooth
    smooth = np.abs(rng.normal(0, 1, n))
    # Apply simple smoothing
    kernel = np.ones(5) / 5
    smooth = np.convolve(smooth, kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    starts.append(smooth / np.sum(smooth))

    return starts


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def polish_full(a, time_budget=100.0):
    """
    Full polishing pipeline alternating between different methods.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)

    methods = ['lse', 'peak', 'lp', 'lse', 'peak']
    mi = 0
    global_stall = 0

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        alloc = min(remaining * 0.3, 40.0)
        method = methods[mi % len(methods)]
        mi += 1

        if method == 'lse':
            a_new, obj_new = lse_gradient_descent(best_a, max_iters=int(alloc * 8), time_budget=alloc)
        elif method == 'peak':
            a_new, obj_new = peak_targeted_subgradient(best_a, top_k=5, max_iters=int(alloc * 6), time_budget=alloc)
        else:  # lp
            if n <= 900:
                a_new, obj_new = lp_alternate_enhanced(best_a, max_iters=20, time_budget=alloc)
            else:
                a_new, obj_new = lse_gradient_descent(best_a, max_iters=int(alloc * 8), time_budget=alloc)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            global_stall = 0
        else:
            global_stall += 1
            if global_stall >= 4:
                # Random restart from best
                rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
                scale = 0.02 + 0.01 * (global_stall % 5)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize_simplex(a_pert)
                # Quick optimize
                a_pert, obj_pert = lse_gradient_descent(a_pert, max_iters=100, time_budget=min(10, remaining * 0.1))
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                global_stall = 0

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining LSE gradient descent,
    peak-targeted subgradient, LP alternating, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    rng = np.random.default_rng(seed)

    # Phase 1: Try the parent with intensive optimization
    if best_seq is not None:
        print("\n=== Phase 1: Intensive optimization of parent ===")
        alloc = min(300, (deadline - time.monotonic()) * 0.4)
        a_pol, obj_pol = polish_full(best_seq.copy(), time_budget=alloc)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Parent polished: {best_score:.8f}")
        else:
            print(f"  Parent polish: {obj_pol:.8f} (no improvement)")

    # Phase 2: Multi-length, multi-start exploration
    print(f"\n=== Phase 2: Multi-length exploration (best: {best_score:.8f}) ===")
    
    n_values = sorted(set([n_parent, n_parent - 100, n_parent + 100, n_parent - 200, n_parent + 200,
                           500, 600, 800, 900]))
    n_values = [x for x in n_values if 300 <= x <= 1200]

    # If we have a good parent, also try resampling it to different lengths
    candidates = []  # (obj, seq, n)
    
    if best_seq is not None:
        for n_test in n_values:
            if time.monotonic() > deadline - 100:
                break
            a_res = normalize_simplex(resample_sequence(best_seq, n_test))
            obj_res = compute_obj(a_res)
            candidates.append((obj_res, a_res, n_test))

    # Generate diverse starts for a few key lengths
    key_lengths = [n_parent, 600, 800]
    key_lengths = sorted(set([x for x in key_lengths if 300 <= x <= 1000]))

    for n in key_lengths:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        starts = generate_starts(n, rng)
        alloc_per = min(remaining / (len(starts) * 2), 15.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 60:
                break
            t_alloc = min(alloc_per, 12.0)
            if t_alloc < 5:
                break

            # Quick LSE optimization
            a_opt, obj_opt = lse_gradient_descent(s, max_iters=int(t_alloc * 5), time_budget=t_alloc)
            
            # Quick peak-targeted
            remaining2 = deadline - time.monotonic()
            if remaining2 > 50:
                a_opt2, obj_opt2 = peak_targeted_subgradient(a_opt, top_k=3,
                                                             max_iters=int(t_alloc * 3),
                                                             time_budget=t_alloc * 0.6)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            candidates.append((obj_opt, a_opt, n))
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {best_score:.8f} ** NEW BEST **")

    # Sort by objective
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5 candidates: {[f'{c[0]:.6f} (n={c[2]})' for c in candidates[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 3: Intensive polish of top candidates
    print(f"\n=== Phase 3: Intensive polish (best: {best_score:.8f}) ===")
    
    top_k = min(4, len(candidates))
    seen_n = set()
    for ci in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        obj_cand, a_cand, n_cand = candidates[ci]
        if obj_cand > best_score + 0.01:
            continue  # Skip if much worse than best
        if n_cand in seen_n:
            continue
        seen_n.add(n_cand)

        alloc = min(remaining * 0.35, 120.0)
        print(f"  Polishing candidate {ci} (n={n_cand}, obj={obj_cand:.8f}) with {alloc:.0f}s")
        
        a_pol, obj_pol = polish_full(a_cand, time_budget=alloc)
        
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {best_score:.8f} (n={n_cand}) **")
        else:
            print(f"  No improvement (got {obj_pol:.8f})")

    # Phase 4: Final intensive round on the absolute best
    print(f"\n=== Phase 4: Final optimization (best: {best_score:.8f}) ===")
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Full polish
            a_final, obj_final = polish_full(best_seq, time_budget=min(remaining * 0.5, 200.0))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: {best_score:.8f}")
            else:
                print(f"  Final polish: {obj_final:.8f}")

            # Try LP on the best if n allows
            remaining = deadline - time.monotonic()
            if remaining > 20 and len(best_seq) <= 900:
                a_lp, obj_lp = lp_alternate_enhanced(best_seq, max_iters=30, time_budget=min(remaining * 0.4, 60.0))
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  LP: {best_score:.8f}")

            # Batch of small perturbations
            remaining = deadline - time.monotonic()
            if remaining > 15:
                n_final = len(best_seq)
                for pidx in range(6):
                    if time.monotonic() > deadline - 10:
                        break
                    scale = 0.01 + 0.005 * pidx
                    a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
                    a_pert = normalize_simplex(a_pert)
                    t_alloc = min(remaining * 0.1, 15.0)
                    if t_alloc < 3:
                        break
                    a_pert, obj_pert = lse_gradient_descent(a_pert, max_iters=int(t_alloc * 8), time_budget=t_alloc)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  Perturb {pidx}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
