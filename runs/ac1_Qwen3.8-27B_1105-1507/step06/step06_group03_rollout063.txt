def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

            return 2.0 * n * conv.max()

        hi = min(2 * n - 2, i + n - 1)
        a_slice = a[lo - i:hi - i + 1].copy()

        valid_kq = kq <= 2 * n - 2

            val = float('inf')

            val = 2.0 * n * conv.max() / (new_s * new_s)

Multi-stage optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
- Fine-grained coordinate descent using Brent's method (continuous 1D optimization)
- Peak-targeted coordinate ordering (focus on coords contributing to max autocorrelation lag)

- Multi-restart with structured perturbations

import numpy as np

from scipy.optimize import minimize_scalar


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

                        conv[kq] -= delta * delta
                    if new_obj < best_val:
                        best_val = new_obj
                        best_x = x

            if abs(best_x - old) > 1e-12:
                delta = best_x - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_x
                s = s - old + best_x

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 300 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f} peak_k={peak_k}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def cd_multi_peak(a, deadline, max_sweeps=5000, num_peaks=5, seed=42, verbose=False):
    """
    Coordinate descent targeting top-K autocorrelation peaks simultaneously.
    Uses coarse ratio-based updates (fast) to handle multiple peaks.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 150

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Find top-K peak lags
        sorted_k = np.argsort(conv)[::-1][:num_peaks]

        # Build combined active set from all top peaks
        active_set = set()
        for k in sorted_k:
            ki = int(k)
            il = max(0, ki - n + 1)
            ih = min(n - 1, ki)
            for ii in range(il, ih + 1):
                active_set.add(ii)

        order = list(active_set)
        rng.shuffle(order)
        remaining = [i for i in range(n) if i not in active_set]
        rng.shuffle(remaining)
        order.extend(remaining)

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            if old > 1e-8:
                ratios = np.array([0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0])
                candidates = ratios * old
            else:
                candidates = np.array([0.0, 1e-5, 1e-4, 1e-3, 5e-4])

            best_val = old
            best_obj_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = float(x)
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = 1e18
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
                if new_obj < best_obj_local:
                    best_obj_local = new_obj
                    best_val = x

            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 300 == 0:
            print(f"    MP-CD sweep {sweep}: obj={best_obj:.10f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def swap_heuristic(a, deadline, max_iters=2000, seed=42, verbose=False):
    """
    Pairwise swap heuristic: try moving mass between pairs of coordinates
    that jointly contribute to the peak autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        peak_k = int(np.argmax(conv))
        # Find top contributors to peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_idx = np.argsort(contribs)[::-1][:20]

        improved = False
        for _ in range(20):
            i = int(rng.choice(top_idx))
            j = int(rng.choice(top_idx))
            if i == j:
                continue

            # Try moving fraction f of (a[i]-a[j]) from larger to smaller
            if a[i] <= a[j]:
                i, j = j, i
            diff = a[i] - a[j]
            if diff < 1e-10:
                continue

            best_f = 0.0
            best_obj_local = best_obj
            for f in [0.1, 0.2, 0.3, 0.5, 0.7]:
                delta = f * diff
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < -1e-10 or a_try[j] < -1e-10:
                    continue
                a_try = np.maximum(a_try, 0)
                ss = a_try.sum()
                if ss < 1e-10:
                    continue
                a_try = a_try / ss
                obj_try = 2.0 * n * fast_conv(a_try).max()
                if obj_try < best_obj_local - 1e-12:
                    best_obj_local = obj_try
                    best_f = f

            if best_f > 0:
                delta = best_f * diff
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = a.sum()
                a = a / s
                s = 1.0
                conv = fast_conv(a)
                best_obj = best_obj_local
                best_a = a.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1

        if not improved:
            no_improve += 5

        if no_improve > 50:
            break

        if verbose and it % 100 == 0:
            print(f"    Swap it {it}: obj={best_obj:.10f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def generate_structured_inits(n, rng, parent=None):
    """Generate diverse structured initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))

    # Symmetric patterns
    for freq in [1, 2, 3, 4, 6, 8]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Triangular
    inits.append(project_simplex(1.0 - np.abs(2 * t - 1)))
    inits.append(project_simplex(np.abs(2 * t - 1)))

    # Bimodal
    for gap in [0.1, 0.2, 0.3]:
        bumps = np.exp(-((t - 0.5 + gap) / 0.1) ** 2) + np.exp(-((t - 0.5 - gap) / 0.1) ** 2)
        inits.append(project_simplex(bumps + 0.1))

    # Zipf-like
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        weights = 1.0 / (np.arange(1, n + 1) ** alpha)
        inits.append(project_simplex(weights))
        inits.append(project_simplex(weights[::-1]))

    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.hanning(ksize)
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-stage optimizer: Brent CD + multi-peak CD + swap heuristic + multi-restart.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_p = len(parent)

    # === Phase 1: Fine Brent CD on parent (250s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Brent CD at n={n_p}", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = t_start + 250.0
    a_work = parent.copy()
    a_work = np.maximum(a_work, 0)
    a_work = a_work / a_work.sum()
    a_opt = cd_brent(a_work, deadline=p1_deadline, max_sweeps=20000, seed=42,
                     brent_maxiter=20, verbose=True)
    update_best(a_opt)
    print(f"  Phase 1 best: J={best_score:.8f}", flush=True)

    # === Phase 2: Multi-peak CD (100s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-peak CD", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = t_start + 350.0
    a_mp = cd_multi_peak(best_seq.copy(), deadline=p2_deadline, max_sweeps=10000,
                         num_peaks=5, seed=77, verbose=True)
    update_best(a_mp)
    print(f"  Phase 2 best: J={best_score:.8f}", flush=True)

    # === Phase 3: Brent CD refinement (150s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Brent CD refinement", flush=True)
    print(f"{'='*60}", flush=True)

    p3_deadline = t_start + 500.0
    a_ref = cd_brent(best_seq.copy(), deadline=p3_deadline, max_sweeps=15000,
                     seed=123, brent_maxiter=30, verbose=True)
    update_best(a_ref)
    print(f"  Phase 3 best: J={best_score:.8f}", flush=True)

    # === Phase 4: Swap heuristic (80s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Swap heuristic", flush=True)
    print(f"{'='*60}", flush=True)

    p4_deadline = t_start + 580.0
    a_sw = swap_heuristic(best_seq.copy(), deadline=p4_deadline, max_iters=3000, seed=55, verbose=True)
    update_best(a_sw)
    # Follow up with Brent CD on swap result
    if time.monotonic() < p4_deadline + 30:
        a_sw2 = cd_brent(best_seq.copy(), deadline=min(p4_deadline + 30, t_start + 620),
                         max_sweeps=5000, seed=88, brent_maxiter=20)
        update_best(a_sw2)
    print(f"  Phase 4 best: J={best_score:.8f}", flush=True)

    # === Phase 5: Multi-restart with perturbations (250s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-restart exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p5_deadline = t_start + 870.0
    num_restarts = 6
    per_restart = (p5_deadline - time.monotonic()) / max(1, num_restarts) - 5

    for r in range(num_restarts):
        remaining = p5_deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_restart, remaining - 5)
        if alloc < 20:
            break

        # Different perturbation strategies
        if r % 3 == 0:
            sigma = 0.01 + 0.02 * r
            a_pert = best_seq + rng.normal(0, sigma, best_seq.size)
            a_pert = project_simplex(a_pert)
        elif r % 3 == 1:
            # Structured perturbation: flip a random block
            block_len = rng.integers(n_p // 10, n_p // 4)
            start = rng.integers(0, max(1, n_p - block_len))
            a_pert = best_seq.copy()
            a_pert[start:start + block_len] *= 0.5
            a_pert = project_simplex(a_pert)
        else:
            # Block swap
            block_len = rng.integers(n_p // 8, n_p // 3)
            s1 = rng.integers(0, max(1, n_p - 2 * block_len))
            s2 = s1 + block_len + rng.integers(1, max(1, n_p - 2 * block_len - block_len))
            a_pert = best_seq.copy()
            tmp = a_pert[s1:s1 + block_len].copy()
            a_pert[s1:s1 + block_len] = a_pert[s2:s2 + block_len]
            a_pert[s2:s2 + block_len] = tmp
            a_pert = project_simplex(a_pert)

        a_r = cd_brent(a_pert, deadline=time.monotonic() + alloc,
                       max_sweeps=5000, seed=1000 + r, brent_maxiter=20)
        update_best(a_r)
        print(f"  Restart {r}: J={compute_J(a_r):.8f}", flush=True)

    print(f"  Phase 5 best: J={best_score:.8f}", flush=True)

    # === Phase 6: Multi-length exploration (150s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p6_deadline = t_start + 1020.0
    n_best = len(best_seq)
    candidate_ns = [n for n in [384, 512, 640, 768, 896, 1024, 1280]
                    if n != n_best and n <= 2048]
    candidate_ns.sort(key=lambda x: abs(x - n_best))

    for new_n in candidate_ns[:4]:
        remaining = p6_deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(70.0, remaining - 10)
        if alloc < 25:
            break
        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()
        a_opt2 = cd_brent(a_ext, deadline=time.monotonic() + alloc,
                          max_sweeps=5000, seed=3000 + new_n, brent_maxiter=20)
        update_best(a_opt2)
        print(f"  n={new_n}: J={compute_J(a_opt2):.8f}", flush=True)

    # === Phase 7: Final intensive polish (remaining time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final intensive polish", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()

        # Long Brent CD with high precision
        t_f1 = time.monotonic() + 0.35 * remaining
        a_f1 = cd_brent(best_seq.copy(), deadline=t_f1, max_sweeps=20000,
                        seed=5555, brent_maxiter=35, verbose=True)
        update_best(a_f1)

        # Multi-peak CD
        if time.monotonic() < deadline - 15:
            t_f2 = time.monotonic() + 0.25 * (deadline - time.monotonic())
            a_f2 = cd_multi_peak(best_seq.copy(), deadline=t_f2, max_sweeps=8000,
                                 num_peaks=3, seed=6666)
            update_best(a_f2)

        # Swap + CD
        if time.monotonic() < deadline - 15:
            t_f3a = time.monotonic() + 0.2 * (deadline - time.monotonic())
            a_f3 = swap_heuristic(best_seq.copy(), deadline=t_f3a, max_iters=1000, seed=8888)
            update_best(a_f3)
            if time.monotonic() < deadline - 10:
                a_f3b = cd_brent(best_seq.copy(), deadline=deadline - 5,
                                  max_sweeps=5000, seed=9999, brent_maxiter=25)
                update_best(a_f3b)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
