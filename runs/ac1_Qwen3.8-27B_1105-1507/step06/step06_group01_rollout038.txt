"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using full-constraint LP-based
mirror descent with Frank-Wolfe alternation, Nesterov acceleration,
spectral initialization, and coordinate descent polish.

Key innovation: Uses the FULL set of 2n-1 LP constraints (not a subset)
to extract accurate dual subgradients, enabling more effective
mirror descent steps than previous approaches.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to probability vector."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def solve_lp_full(a, M):
    """Solve full LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_primal, y_dual) or (None, None) on failure."""
    n = len(a)
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            # Get dual variables for A_ub constraints
            y = np.zeros(2 * n - 1)
            if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                if hasattr(r.ineqlin, 'marginals') and r.ineqlin.marginals is not None:
                    y = np.maximum(r.ineqlin.marginals, 0.0)
            return g, y
    except Exception as e:
        pass
    return None, None


def compute_subgradient(a, y):
    """Compute subgradient of max(a*a) using dual weights y.
    grad[j] = 2 * sum_k y[k] * a[k-j] = 2 * (y * a)[j] (cross-correlation)
    """
    n = len(a)
    s = y.sum()
    if s < 1e-15:
        return np.full(n, 2.0 * a.max())
    y_norm = y / s
    # grad[j] = 2 * sum_{m=0}^{n-1} y_norm[j+m] * a[m]
    grad = np.zeros(n)
    for j in range(n):
        # y_norm has length 2n-1, we need y_norm[j:j+n]
        if j + n <= 2 * n - 1:
            grad[j] = 2.0 * np.dot(y_norm[j:j + n], a)
        else:
            # Handle boundary
            valid = j + n - (2 * n - 1) + 1  # number of valid elements
            if valid > 0:
                grad[j] = 2.0 * np.dot(y_norm[j:2 * n - 1], a[:valid])
    return grad


def run_lp_mirror_descent(a_init, max_iters, deadline, eta_init=0.5):
    """Main optimization: LP-based mirror descent with FW alternation."""
    a = normalize(a_init.copy())
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2 * n * best_M  # since sum(a)=1
    
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Solve LP
        g, y = solve_lp_full(a, M)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g_norm = g / g_sum
        
        # Compute subgradient from dual
        s = y.sum()
        if s < 1e-10:
            # Fallback: use argmax
            k_star = int(np.argmax(b))
            y_fb = np.zeros(2 * n - 1)
            y_fb[k_star] = 1.0
            grad = compute_subgradient(a, y_fb)
        else:
            grad = compute_subgradient(a, y)
        
        # Remove mean (project to tangent of simplex)
        grad_tan = grad - grad.mean()
        
        # Strategy 1: Mirror descent (multiplicative weights)
        eta = eta_init / (1.0 + 0.05 * it)
        log_shift = -eta * grad_tan
        log_shift -= log_shift.max()  # for numerical stability
        a_md = a * np.exp(log_shift)
        a_md = normalize(a_md)
        M_md = fast_conv(a_md).max()
        
        # Strategy 2: Frank-Wolfe step
        gamma = 2.0 / (it + 2)
        a_fw = (1.0 - gamma) * a + gamma * g_norm
        a_fw = normalize(a_fw)
        M_fw = fast_conv(a_fw).max()
        
        # Strategy 3: Blended (average of MD and FW directions)
        a_blend = 0.5 * a_md + 0.5 * a_fw
        a_blend = normalize(a_blend)
        M_blend = fast_conv(a_blend).max()
        
        # Pick the best
        candidates = [(a_md, M_md), (a_fw, M_fw), (a_blend, M_blend)]
        candidates.sort(key=lambda x: x[1])
        a_new, M_new = candidates[0]
        
        # Line search on the best direction
        if M_new < M:
            # Try to improve with line search
            direction = a_new - a
            best_t = 1.0
            best_M_ls = M_new
            for t in [1.0, 0.7, 0.5, 0.3, 0.8, 0.9]:
                a_ls = normalize(a + t * direction)
                M_ls = fast_conv(a_ls).max()
                if M_ls < best_M_ls:
                    best_M_ls = M_ls
                    best_t = t
            if best_t != 1.0:
                a_new = normalize(a + best_t * direction)
                M_new = best_M_ls
        
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            for t in [0.3, 0.1, 0.05]:
                a_try = normalize(a + t * (a_new - a))
                M_try = fast_conv(a_try).max()
                if M_try < best_M - 1e-12:
                    best_M = M_try
                    best_a = a_try.copy()
                    a_new = a_try
                    no_improve = 0
                    break
        
        a = a_new
        
        if no_improve > 50:
            break
    
    return best_a

    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    order = np.arange(n)

            break
        np.random.shuffle(order)
        
        b = fast_conv(a)
        improved = False

            # (a*a)[k] for affected k: k in [i, i+n-1] (clipped to [0, 2n-2])

                    ga = gb
                    fa = fb

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""

    t_new = np.linspace(0, 1, dest_n)

def spectral_init(n, max_iters=50):

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using full-constraint LP-based
mirror descent with Frank-Wolfe alternation, spectral initialization,
and coordinate descent polish.

Key innovation: Uses ALL 2n-1 LP constraints to extract accurate dual
subgradients for mirror descent, combined with Frank-Wolfe steps toward
the LP primal solution. Multi-length search with diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to probability vector (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build the (2n-1) x n sparse convolution matrix A where A[k,j] = a[k-j]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            m = k - j
            if a[m] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_full(a, M):
    """Solve full LP: max sum(g) s.t. A(a)g <= M*1, g >= 0.
    Returns (g_primal, y_dual) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)
    if A.nnz == 0:
        return None, None

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            y = np.zeros(2 * n - 1)
            if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                marg = getattr(r.ineqlin, 'marginals', None)
                if marg is not None:
                    y = np.asarray(marg, dtype=np.float64)
                    y = np.maximum(y, 0.0)
            return g, y
    except Exception:
        pass
    return None, None


def compute_subgradient(a, y):
    """Compute subgradient of max(a*a) using dual weights y (normalized).
    grad[j] = 2 * sum_{m=0}^{n-1} y[j+m] * a[m]
    """
    n = len(a)
    s = y.sum()
    if s < 1e-15:
        k_star = int(np.argmax(fast_conv(a)))
        y = np.zeros(2 * n - 1)
        y[k_star] = 1.0
        s = 1.0
    y_norm = y / s
    grad = np.zeros(n)
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        grad[j] = 2.0 * np.dot(y_norm[j:end], a[:end - j])
    return grad

    a = normalize(a_init)
    n = len(a)

    no_improve = 0
    it = 0

        M = b.max()
        if M < 1e-15:

        if g is None:

            if no_improve > 30:
                break
            it += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            it += 1
            continue

        g_norm = g / g_sum

        # Compute subgradient from dual
        grad = compute_subgradient(a, y)
        grad_tan = grad - grad.mean()

        # Strategy 1: Mirror descent (multiplicative weights / exponential tilting)
        eta = eta_init / (1.0 + 0.03 * it)
        log_shift = -eta * grad_tan
        log_shift -= log_shift.max()

def coordinate_descent_polish(a_init, max_sweeps, deadline):
    """Polish with exact 1D coordinate descent."""
    a = normalize(a_init)
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)

        b = fast_conv(a)
        sweep_changed = False

        for i in order:
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            n_affected = k_hi - k_lo + 1

            c_k = np.zeros(n_affected)
            s_k = np.zeros(n_affected)
            q_k = np.zeros(n_affected)

            x_cur = a[i]
            for idx, k in enumerate(range(k_lo, k_hi + 1)):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        c_k[idx] = b[k] - x_cur * x_cur
                        q_k[idx] = 1.0
                    else:
                        c_k[idx] = b[k] - 2.0 * x_cur * a[j]
                        s_k[idx] = 2.0 * a[j]
                else:
                    c_k[idx] = b[k]

            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0

            x_lo = 0.0
            x_hi = 4.0 * x_cur + 0.05

            def obj_fn(x_test):
                vals = c_k + s_k * x_test + q_k * x_test * x_test
                return max(base_max, vals.max())

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            ga = x_hi - gr * (x_hi - x_lo)
            gb = x_lo + gr * (x_hi - x_lo)
            fa = obj_fn(ga)
            fb = obj_fn(gb)

            for _ in range(25):
                if fa < fb:
                    x_hi = gb
                    gb = ga
                    fb = fa
                    ga = x_hi - gr * (x_hi - x_lo)
                    fa = obj_fn(ga)
                else:
                    x_lo = ga
                    ga = gb
                    fa = fb
                    gb = x_lo + gr * (x_hi - x_lo)
                    fb = obj_fn(gb)

            x_opt = (x_lo + x_hi) / 2.0
            if obj_fn(0.0) <= obj_fn(x_opt):
                x_opt = 0.0

            if abs(x_opt - a[i]) > 1e-13:
                a[i] = x_opt
                sweep_changed = True

        s = a.sum()
        if s > 1e-10:
            a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()

        if not sweep_changed:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def spectral_init(n, max_iters=40):
    """Spectral initialization: iteratively reduce peak convolution."""
    a = np.ones(n) / n
    for _ in range(max_iters):
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        corr = np.zeros(n)
        for j in range(n):
            m = k_star - j
            if 0 <= m < n:
                corr[j] = a[m]
        a = a - 0.15 * (corr - corr.mean())
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        p = normalize(p)
        inits.append(p)
        p_pert = np.maximum(p + rng.normal(0, 0.04, n), 1e-4)
        inits.append(normalize(p_pert))
        p_pert2 = np.maximum(p + rng.normal(0, 0.1, n), 1e-4)
        inits.append(normalize(p_pert2))

    inits.append(np.ones(n) / n)
    inits.append(spectral_init(n, 30))

    # Cosine windows
    for freq in [1, 2]:
        cw = 0.5 + 0.5 * np.cos(np.pi * freq * t)
        cw = np.maximum(cw, 0.01)
        inits.append(normalize(cw))

    # Gaussian
    g = np.exp(-3.0 * (t - 0.5) ** 2) + 0.03
    inits.append(normalize(g))

    # Smooth random
    ksize = max(3, n // 12)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(normalize(smooth))

    # Power law
    pw = (t + 0.05) ** (-0.5)
    inits.append(normalize(pw))

    return inits
