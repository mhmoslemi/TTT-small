    """
    Initialize a non-negative sequence with nearly flat autocorrelation
    using iterative projection (Gerchberg-Saxton).
    """
    rng = np.random.default_rng(42)
    
    # Start with random positive sequence
    a = rng.uniform(0.5, 1.5, n)
    
    # Target: flat power spectrum
    # |A(f)|^2 should be approximately constant
    # A(f) = FFT(a)
    
    for it in range(iters):
        # Compute current power spectrum
        A = np.fft.fft(a, 2*n-1)
        P = np.abs(A)**2
        
        # Target: flat spectrum (proportional to sum(a)^2 / (2n-1))
        # But we need to match the DC component
        P_target = np.full_like(P, np.sum(P) / (2*n-1))
        P_target[0] = P[0]  # preserve DC
        
        # Take square root to get amplitude
        A_amp_target = np.sqrt(P_target)
        
        # Keep current phase, replace amplitude
        phase = np.angle(A)
        A_new = A_amp_target * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.ifft(A_new, 2*n-1)[:n].real
        
        # Project onto non-negative cone
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        if np.sum(a_new) > 0:
            a_new = a_new / np.sum(a_new) * np.sum(a)
        
        a = a_new
    
    return a

max(conv) = max(1, 2t, t^2+2t, 2+2t^2, t^2+2t, 2t, 1)
= max(2+2t^2, t^2+2t) for t > 1

For t > 1: 2+2t^2 vs t^2+2t → 2+2t^2 > t^2+2t iff t^2 - 2t + 2 > 0 iff (t-1)^2 + 1 > 0. Always true!

So for t > 1, max(conv) = 2 + 2t^2.
sum = 2 + 2t
objective = 2*4*(2+2t^2) / (2+2t)^2 = 8*(2+2t^2) / (2(1+t))^2 = 8*(2+2t^2) / (4(1+t)^2) = 2*(2+2t^2)/(1+t)^2 = 4*(1+t^2)/(1+t)^2

f(t) = 4(1+t^2)/(1+t)^2

f'(t) = 4 * [2t(1+t)^2 - (1+t^2)*2(1+t)] / (1+t)^4
= 4 * [2t(1+t) - 2(1+t^2)] / (1+t)^3
= 4 * [2t + 2t^2 - 2 - 2t^2] / (1+t)^3
= 4 * [2t - 2] / (1+t)^3
= 8(t-1) / (1+t)^3

So f'(t) = 0 at t=1, and f'(t) > 0 for t > 1, f'(t) < 0 for t < 1.

This means t=1 is the MINIMUM, and f(1) = 4*2/4 = 2.

So the symmetric sequence [1,1,1,1] (uniform) is actually optimal within the symmetric family [1,t,t,1]! Asymmetric sequences do better.
