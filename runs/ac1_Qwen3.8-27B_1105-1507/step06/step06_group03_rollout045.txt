import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            # Determine affected lags and their current values
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            
            # Generate candidates
            if old < 1e-10:
                max_candidate = 1.0 / max(1, int(0.1 * n))
                candidates = np.concatenate([
                    [0.0],
                    np.linspace(0, max_candidate, 5)[1:]
                ])
            else:
                # More candidates near current value, plus some structural ones
                n_cand = 12
                # Log-spaced around current value
                ratios = np.concatenate([
                    np.linspace(0, 0.5, 3),
                    np.linspace(0.5, 1.5, 5),
                    np.linspace(1.5, 3.0, 3)
                ])
                candidates = np.unique(np.concatenate([[0.0], ratios * old]))
            
            # Also add a candidate that would equalize the max lag
            # Find the critical lag
            critical_k = np.argmax(conv)
            # The contribution of a[i] to lag critical_k:
            j_val = critical_k - i
            if 0 <= j_val < n and j_val != i:
                # Setting a[i] = target would make conv[critical_k] = current_max - old*2*a[j_val] + target*2*a[j_val]
                # We want this to be at most the second-max
                second_max = np.partition(conv, -2)[-2]
                target = (second_max - (conv[critical_k] - 2*old*a[j_val])) / (2*a[j_val])
                if target > 0:
                    candidates = np.unique(np.append(candidates, target))
            
            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)
            
            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                
                a_slice = a[lo - i:hi - i + 1].copy()
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] -= delta * delta
            
            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                a_slice = a[lo - i:hi - i + 1].copy()
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            break
        
        if verbose and sweep % 100 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a

def gs_iteration(a, deadline, n_iters_max=5000, verbose=False):
    """
    Gerchberg-Saxton style iteration to flatten the autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    
    # FFT size (at least 2n-1 for linear convolution)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    target = 1.0 / (2 * n - 1)  # ideal flat value
    
    for it in range(n_iters_max):
        if time.monotonic() > deadline:
            break
        
        # Compute current autocorrelation (circular with zero padding)
        A = np.fft.rfft(a, size)
        B = A * np.conj(A)  # = |A|^2
        b_circ = np.fft.irfft(B, size)  # circular autocorrelation
        
        # b_circ[0] = sum a^2, b_circ[k] = sum a[j]*a[j-k] (circular)
        # The linear autocorrelation is embedded in b_circ
        # We want b_circ to be as flat as possible in the "linear" region
        
        # Modify b_circ to be flatter
        # Use soft thresholding: raise the low values, keep high values
        # Strategy: b_mod = max(b_circ, alpha * target)
        # Then scale to maintain sum = 1
        
        alpha = 0.3 + 0.5 * (it / min(it + 1, 100))  # anneal from 0.3 to 0.8
        
        b_mod = np.maximum(b_circ, alpha * b_circ.max())
        # Ensure sum is preserved
        b_mod = b_mod / b_mod.sum() * b_circ.sum()
        
        # Update A: keep phase, adjust magnitude
        B_mod = np.fft.rfft(b_mod, size)
        # A_new should have |A_new|^2 ~ B_mod
        # But B_mod might not be non-negative, so we need to be careful
        # Use: A_new = A * sqrt(B_mod / B) where B > 0
        with np.errstate(divide='ignore', invalid='ignore'):
            ratio = np.sqrt(np.maximum(B_mod, 0) / np.maximum(B, 1e-30))
        ratio = np.nan_to_num(ratio, nan=1.0, posinf=1.0, neginf=1.0)
        A_new = A * ratio
        
        # IFFT to get new a
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            no_improve += 1
            continue
        a_new = a_new / s_new
        
        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            # Try a damped version
            for damp in [0.5, 0.3, 0.1]:
                a_damp = (1 - damp) * a + damp * a_new
                a_damp = np.maximum(a_damp, 0)
                if a_damp.sum() > 1e-10:
                    a_damp /= a_damp.sum()
                    obj_damp = compute_J(a_damp)
                    if obj_damp < best_obj - 1e-12:
                        best_obj = obj_damp
                        best_a = a_damp.copy()
                        a = a_damp.copy()
                        no_improve = 0
                        break
            else:
                no_improve += 1
        
        if no_improve > 100:
            break
        
        if verbose and it % 50 == 0:
            print(f"    GS it {it}: obj={best_obj:.8f}", flush=True)
    
    return best_a, best_obj

def lp_best_response(a, deadline, max_iters=200, verbose=False):
    """
    LP best-response: solve LP to find g that maximizes sum(g)
    subject to (a*g)[k] <= M for all k, g >= 0.
    Then blend a and g.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        # Build LP matrix: (a*g)[k] = sum_j a[k-j]*g[j] for valid j
        # Constraint: sum_j a[k-j]*g[j] <= M for k = 0,...,2n-2
        # (Note: a[k-j] is only defined for 0<=k-j<n, i.e., k-n+1 <= j <= k, intersect [0,n-1])
        
        rows = []
        cols = []
        vals = []
        
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A_ub = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                       method='highs', options={'time_limit': 30, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() > 1e-10:
                            a_try /= a_try.sum()
                            J_try = compute_J(a_try)
                            if J_try < best_obj - 1e-12:
                                a = a_try
                                best_obj = J_try
                                best_a = a_try.copy()
                                no_improve = 0
                                found = True
                                break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        if no_improve > 20:
            break
        
        if verbose and it % 10 == 0:
            print(f"    LP it {it}: obj={best_obj:.8f}", flush=True)
    
    return best_a, best_obj

def subgradient_descent(a, deadline, max_iters=5000, lr_init=0.1, verbose=False):
    """
    Projected subgradient descent on the simplex.
    Subgradient of max_k conv[k] w.r.t. a[i] = conv[i] + conv[2i-k]... 
    Actually, for the critical lag k*, the subgradient w.r.t. a[i] is:
    a[k*-i] (if 0<=k*-i<n and k*-i != i) + a[i-k*] (if 0<=i-k*<n and i-k* != i)
    + 2*a[i] if 2*i == k*
    More simply: the gradient of (a*a)[k*] w.r.t. a[i] is a[k*-i] + a[i-k*]
    (counting the i term once for each valid position).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    t = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        max_val = conv.max()
        # Subgradient: for the critical lag(s), compute gradient
        # Use a "soft" subgradient: average over top few lags
        sorted_idx = np.argsort(conv)[::-1]
        top_k = min(5, 2*n-1)
        
        grad = np.zeros(n)
        for k in sorted_idx[:top_k]:
            for i in range(n):
                j1 = k - i
                j2 = i - k  # not valid, skip
                if 0 <= j1 < n:
                    grad[i] += a[j1]
                # Also the "other" term: (a*a)[k] includes a[i]*a[k-i] and a[k-i]*a[i]
                # If i != k-i, both terms contribute a[i]*a[k-i] and a[k-i]*a[i] = 2*a[i]*a[k-i]
                # The derivative w.r.t. a[i] is: a[k-i] + (if k-i != i) a[k-i] = 2*a[k-i]
                # Wait, let me be more careful.
                # (a*a)[k] = sum_j a[j] * a[k-j]
                # d/d a[i] of (a*a)[k] = a[k-i] (from j=i term) + (if i != k-i and 0<=i-k... 
                # Actually: (a*a)[k] = sum_{j: valid} a[j] * a[k-j]
                # The terms involving a[i] are:
                #   j = i: a[i] * a[k-i] (if 0 <= k-i < n)
                #   k-j = i, i.e., j = k-i: a[k-i] * a[i] (if 0 <= k-i < n and k-i != i)
                # So d/d a[i] = a[k-i] * (1 if 0<=k-i<n) + a[k-i] * (1 if 0<=k-i<n and k-i!=i)
                # = a[k-i] * (1 + 1{0<=k-i<n and k-i!=i}) if 0<=k-i<n
                # = 2*a[k-i] if 0<=k-i<n and k-i!=i
                # = a[k-i] if 0<=k-i<n and k-i==i (i.e., 2i=k)
                # = 0 otherwise (if k-i not in [0,n))
                # Wait, that's not right either. Let me redo.
                # (a*a)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * a[k-j]
                # d/d a[i]:
                #   If i is in the sum (as j): term is a[i]*a[k-i], derivative is a[k-i]
                #   If k-i is in the sum (as j, meaning j=k-i): term is a[k-i]*a[i], derivative is a[i]... 
                #   Wait no. If j = k-i is in the sum, the term is a[k-i]*a[i], and d/d a[i] of this is a[k-i].
                #   But we also need k-i to be a valid index for a[], i.e., 0 <= k-i < n.
                #   And j = k-i must be in [max(0,k-n+1), min(n-1,k)].
                #   
                # So: d/d a[i] (a*a)[k] = a[k-i] * 1{0<=k-i<n and max(0,k-n+1)<=i<=min(n-1,k)}
                #                           + a[k-i] * 1{0<=k-i<n and max(0,k-n+1)<=k-i<=min(n-1,k) and k-i!=i}
                #
                # The first condition: i is a valid j, which is always true since 0<=i<n.
                #   But also need 0<=k-i<n, which means k-n+1<=i<=k.
                # The second condition: k-i is a valid j (same as first) AND k-i != i.
                #   So it's: 0<=k-i<n AND k-i != i, i.e., 2i != k.
                #
                # Combined: if 0<=k-i<n:
                #   contribution = a[k-i] + (a[k-i] if 2i != k else 0)
                #   = 2*a[k-i] if 2i != k
                #   = a[k-i] if 2i == k
                # else: 0
        
        # Let me just compute it properly using the formula
        grad = np.zeros(n)
        for k in sorted_idx[:top_k]:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    if 2 * i == k:
                        grad[i] += a[j]  # = a[i], only one term
                    else:
                        grad[i] += 2.0 * a[j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Learning rate
        t += 1
        lr = lr_init / (1.0 + 0.5 * np.sqrt(t))
        
        # Update
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
