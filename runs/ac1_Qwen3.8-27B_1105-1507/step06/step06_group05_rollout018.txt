import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz_vec(a, size):
    """Build the convolution matrix as a linear operator for use in LP.
    For the LP, we need A such that (A@g)[k] = sum_j a[max(0,k-n+1):k+1] * g[...]
    Actually, (conv(a,g))[k] = sum_{j=0}^{min(k,n-1)} a[j] * g[k-j] for g of length n
    But we need to be careful about the indices.
    
    conv(a,g)[k] = sum_j a[j] * g[k-j] where g is defined for j=0,...,n-1
    So conv(a,g)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * g[k-j]
    
    For the LP, we need A[k,j] such that (A@g)[k] = conv(a,g)[k]
    A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    """
    n = len(a)
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # This is a Toeplitz matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = a
    return A

def solve_lp(a, M):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0"""
    n = len(a)
    A = build_toeplitz_vec(a, 2*n-1)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        g = np.maximum(res.x, 0.0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build (2n-1) x n matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_full_lp(a, M, time_limit=10):
    """Solve full LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

            s = float(np.sum(g))

    return None

def compute_conv_2(a, b):
    """Compute linear convolution of a and b (different lengths allowed)."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    conv = irfft(fa * fb, size)
    return conv[:na + nb - 1]


def golden_section_min(f, lo, hi, tol=0.005, max_iter=20):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_alternate_accelerated(a, max_iters=60, time_budget=120.0, use_full=False):

    Alternates between full LP and focused LP for speed.

    t_mom = 1.0  # Nesterov momentum parameter

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:

        M = float(np.max(conv))

        else:

        if g is None:
            g = solve_focused_lp(a_curr, M, top_k=5, time_limit=3)
        if g is None:
            break

        # Nesterov lookahead for step size
        theta = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        mom_factor = (t_mom - 1.0) / theta
        t_mom = theta

        # Define objective at blend parameter t
        def obj_at_t(t_val):
            a_try = (1.0 - t_val) * a_curr + t_val * g
            a_try = normalize(a_try)
            return compute_obj(a_try)

        # Find optimal step
        t_opt = golden_section_min(obj_at_t, 0.0, 1.0, tol=0.01, max_iter=12)

        # Apply momentum to step
        t_eff = (1.0 - mom_factor) * t_opt + mom_factor * 0.5
        t_eff = max(0.0, min(1.0, t_eff))

        a_next = normalize((1.0 - t_eff) * a_curr + t_eff * g)
        obj_next = compute_obj(a_next)

        if obj_next < best_obj:
            best_obj = obj_next
            best_a = a_next.copy()
        a_prev = a_curr
        a_curr = a_next

    return best_a


def subgradient_polish(a, max_steps=500, time_budget=30.0):
    """
    Nesterov-accelerated projected subgradient method.
    Minimizes max_k (a*a)[k] subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a_curr = normalize(a.copy().astype(np.float64))
    a_prev = a_curr.copy()
    best_a = a_curr.copy()
    best_obj = compute_obj(a_curr)
    start = time.monotonic()
    t_val = 1.0
    lr_scale = 0.1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        a_look = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        a_look = normalize(a_look)
        t_val = t_next

        # Compute subgradient at a_look
        conv = compute_conv(a_look)
        k_star = int(np.argmax(conv))

        # grad[j] = 2 * a_look[k_star - j] if 0 <= k_star - j < n, else 0
        k_vals = k_star - np.arange(n)
        mask = (k_vals >= 0) & (k_vals < n)
        grad = np.zeros(n)
        grad[mask] = 2.0 * a_look[k_vals[mask]]

        # Adaptive learning rate
        lr = lr_scale / np.sqrt(step + 1.0)

        # Projected gradient step on simplex
        a_new = a_look - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a_curr
        else:
            a_new = a_new / s

        a_prev = a_curr
        a_curr = a_new

        obj = compute_obj(a_curr)
        if obj < best_obj:
            best_obj = obj
            best_a = a_curr.copy()

    return best_a


def spectral_flatten(a, max_iters=40, time_budget=30.0):
    """Iterative spectral flattening via POCS."""
    n = len(a)
    a_curr = normalize(a.copy().astype(np.float64))
    best_a = a_curr.copy()
    best_obj = compute_obj(a_curr)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a_curr, size)
        P = np.abs(A) ** 2
        p_max = np.max(P)
        if p_max < 1e-30:
            break

        # Spectral compression: push toward flat spectrum
        alpha = 0.3 + 0.4 * (it / max_iters)
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a_curr = a_new
        else:
            # Blend to maintain progress
            a_curr = normalize(0.6 * a_curr + 0.4 * a_new)

    return best_a


def peak_deflation(a, time_budget=20.0, n_trials=200):
    """
    Targeted peak deflation: identify top contributors to each peak lag,
    then redistribute mass to reduce those peaks.

        peak_lags = np.argsort(conv)[-3:]

        for k_star in peak_lags:

            if len(j_vals) == 0:

            contributions = a_curr[j_vals] * a_curr[k_star - j_vals]
            top_contrib_idx = j_vals[np.argsort(contributions)[-2:]]

            for j1 in top_contrib_idx:

                if j1 == j2:

                a_try = a_curr.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)

                    best_a = a_try.copy()
                    a_curr = a_try
                    break

        # Occasional random perturbation to escape
        if trial % 30 == 29:

    return best_a

                dst = int(rng.integers(0, n))
                if dst == src:
                    continue
                a_try = a_curr.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                tried += 1

                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a_curr = a_try
                    break

        # Random perturbation to escape local minima
        if iteration % 20 == 0:
            a_curr = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def fourier_perturb(a, rng, strength=0.1):
    """Perturb in Fourier domain for diverse candidates."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = fft(a, size)
    # Random phase perturbation
    phase_noise = rng.normal(0, strength, size)
    A_pert = A * np.exp(1j * phase_noise)
    a_pert = np.real(ifft(A_pert, size))[:n]
    return normalize(a_pert)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy with improved LP alternating,
    cutting-plane acceleration, Fourier restarts, and adaptive n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_best = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_best}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Intensive LP optimization (45% of time)
    # =========================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Intensive LP optimization")
    print(f"{'=' * 60}")

    phase1_budget = (deadline - start_time) * 0.45
    if best_seq is not None:
        a = normalize(best_seq.copy())

        # Sub-phase 1a: Full LP alternating
        alloc_1a = phase1_budget * 0.5
        a_new = lp_alternate_accelerated(a, max_iters=40, time_budget=alloc_1a, use_full=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Full LP: {best_score:.8f}")
        else:
            a = a_new
            print(f"  Full LP: {obj_new:.8f} (no improvement)")

        # Sub-phase 1b: Focused LP alternating (faster, more iterations)
        if time.monotonic() < deadline - (deadline - start_time) * 0.45:
            alloc_1b = phase1_budget * 0.35
            a_new = lp_alternate_accelerated(a, max_iters=80, time_budget=alloc_1b, use_full=False)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Focused LP: {best_score:.8f}")
            else:
                a = a_new
                print(f"  Focused LP: {obj_new:.8f} (no improvement)")

        # Sub-phase 1c: Cutting plane LP
        if time.monotonic() < deadline - (deadline - start_time) * 0.45:
            a = normalize(best_seq.copy())
            remaining_p1 = deadline - (deadline - start_time) * 0.45 - time.monotonic()
            if remaining_p1 > 10:
                a_new = lp_alternate_accelerated(a, max_iters=30, time_budget=min(remaining_p1, 40.0), use_full=True)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  Post-LP: {best_score:.8f}")
                else:
                    print(f"  Post-LP: {obj_new:.8f}")

    # =========================================================
    # Phase 2: Multi-n exploration (20% of time)
    # =========================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Multi-n exploration")
    print(f"{'=' * 60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        # Explore n values around current best
        n_options = []
        for dn in [-100, -75, -50, -30, -20, -10, 0, 10, 20, 30, 50, 75, 100, 150]:
            n_try = n_cur + dn
            if 200 <= n_try <= 1200:
                n_options.append(n_try)
        n_options = sorted(set(n_options))

        for n_try in n_options:
            if time.monotonic() > deadline - (deadline - start_time) * 0.25:
                break
            remaining = deadline - (deadline - start_time) * 0.25 - time.monotonic()
            if remaining < 10:
                break

            if n_try == n_cur:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample_seq(best_seq, n_try))

            # Quick optimization at this n
            alloc = min(remaining * 0.1, 25.0)
            a_try = lp_alternate_accelerated(a_try, max_iters=15, time_budget=alloc * 0.6, use_full=False)
            a_try = subgradient_polish(a_try, max_steps=100, time_budget=alloc * 0.3)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_cur = n_try
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 3: Diversified restarts with annealing (20% of time)
    # =========================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Diversified restarts & annealing")
    print(f"{'=' * 60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)
        restart_count = 0
        phase3_end = deadline - (deadline - start_time) * 0.15

        while time.monotonic() < phase3_end and restart_count < 15:
            remaining = phase3_end - time.monotonic()
            if remaining < 15:
                break
            restart_count += 1
            alloc = min(remaining * 0.15, 40.0)

            # Diverse perturbation strategies
            strategy = restart_count % 5
            if strategy == 0:
                # Fourier perturbation
                a_try = fourier_perturb(best_seq, rng, strength=0.15)
            elif strategy == 1:
                # Gaussian multiplier
                a_try = normalize(best_seq * np.exp(rng.normal(0, 0.1, n_cur)))
            elif strategy == 2:
                # Sinusoidal modulation
                freq = rng.uniform(1.0, 8.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = normalize(best_seq * mod)
            elif strategy == 3:
                # Blend with random Dirichlet
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
                mix = rng.uniform(0.15, 0.4)
                a_try = normalize((1 - mix) * normalize(best_seq) + mix * a_rand)
            else:
                # Block reversal
                a_try = best_seq.copy()
                block_len = int(rng.integers(n_cur // 5, n_cur // 2))
                start_idx = int(rng.integers(0, max(1, n_cur - block_len)))
                a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
                a_try = normalize(a_try)

            # Optimize the perturbed sequence
            a_try = lp_alternate_accelerated(a_try, max_iters=12, time_budget=alloc * 0.5, use_full=False)
            a_try = subgradient_polish(a_try, max_steps=80, time_budget=alloc * 0.3)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart_count} (strat={strategy}): {best_score:.8f} **")
            else:
                print(f"  Restart {restart_count} (strat={strategy}): {obj_try:.8f}")

    # =========================================================
    # Phase 4: Final intensive polish (15% of time)
    # =========================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Final polish")
    print(f"{'=' * 60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = len(a_final)

        for polish_round in range(6):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.2, 35.0)

            # LP polish
            if n_final <= 1000:
                a_pol = lp_alternate_accelerated(a_final, max_iters=15, time_budget=alloc * 0.4, use_full=True)
            else:
                a_pol = a_final.copy()

            # Subgradient polish
            a_pol = subgradient_polish(a_pol, max_steps=200, time_budget=alloc * 0.25)

            # Peak deflation
            a_pol = peak_deflation(a_pol, time_budget=alloc * 0.15)

            # Coordinate descent
            a_pol = coord_descent_polish(a_pol, time_budget=min(alloc * 0.2, 10.0))

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish round {polish_round}: {best_score:.8f}")
            else:
                a_final = a_pol
                print(f"  Polish round {polish_round}: {obj_pol:.8f} (no improvement)")

            # Small perturbation between rounds to escape
            rng = np.random.default_rng(polish_round * 1000 + 42)
            a_final = normalize(a_final * np.exp(rng.normal(0, 0.01, n_final)))

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'=' * 60}")

    return result
