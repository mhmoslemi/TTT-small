def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

    if s < 1e-10:

    conv = compute_conv(a)

    return 2.0 * n * max_b / (s * s)

    if s < 1e-15:

        best_blend_obj = best_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize((1 - alpha) * a + alpha * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_alpha = alpha
        
        a_new = normalize((1 - best_alpha) * a + best_alpha * g)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb to escape
            rng = np.random.default_rng((it * 1000003 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))
    
    return best_a

def softmax_subgradient(a, tau=0.005, lr=0.05, max_steps=200, time_budget=50.0):
    """Smoothed subgradient descent with soft-max weighting."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Soft-max weights over convolution values
        tau_eff = tau + 0.1 * cmax / n
        shifted = (conv - cmax) / tau_eff
        weights = np.exp(np.clip(shifted, -500, 500))
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[j] = sum_k weights[k] * d(conv[k])/d(a[j])
        # d(conv[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n, else 0
        # So grad[j] = 2 * sum_k weights[k] * a[k-j] * I(0 <= k-j < n)
        # = 2 * (weights * a_rev) cross-correlation
        
        # More efficiently: grad[j] = 2 * sum_k weights[k] * a[k-j] * I(0<=k-j<n)
        # = 2 * (conv of weights with a_flip)[n-1 + j]... 
        
        # Actually, let's just compute it directly using the Toeplitz structure:
        # grad[j] = 2 * sum_{k: 0<=k-j<2n-1, 0<=k-j<n} weights[k] * a[k-j]
        # Let m = k-j, so k = j+m, m in [0, n-1]:
        # grad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * a[m] (if j+m < 2n-1)
        
        # This is a cross-correlation of weights with a:
        # grad[j] = 2 * sum_m weights[j+m] * a[m]
        # = 2 * (cross_correlation)(weights, a)[j] (with appropriate offset)
        
        # Using FFT: cross_correlation(weights, a)[j] = sum_m weights[j+m] * a[m]
        # = ifft(fft(weights) * conj(fft(a)))... 
        
        # Actually, let's just use the convolution:
        # sum_m weights[j+m] * a[m] = conv(weights, a_flip)[(2n-2) - j + (n-1)]
        # This is getting complicated. Let me just use a direct computation for small n
        # or FFT for large n.
        
        # Simpler approach: grad = C^T * weights where C is the Toeplitz matrix
        # C[k,j] = a[k-j] if 0<=k-j<n
        # grad[j] = 2 * sum_k C[k,j] * weights[k] = 2 * (C^T * weights)[j]
        
        # C^T * weights: (C^T * w)[j] = sum_k C[k,j] * w[k] = sum_k a[k-j]*w[k] * I(0<=k-j<n)
        # Let m = k-j: = sum_m a[m] * w[j+m] * I(0<=m<n, 0<=j+m<2n-1)
        # = sum_{m=0}^{n-1} a[m] * w[j+m] (with w out of range = 0)
        
        # This is the cross-correlation of w and a:
        # (w ⋆ a)[j] = sum_m w[j+m] * a[m]
        
        # We can compute this as: ifft(fft(w) * conj(fft(a)))[:n]
        # But w has length 2n-1 and a has length n.
        
        # Let's use the fact that:
        # sum_m a[m] * w[j+m] = (a * w_rev)[(2n-2) - j] ... 
        # This is getting too complicated. Let me just do it with FFT properly.
        
        # Cross-correlation: (w ⋆ a)[j] = sum_m w[j+m] * a[m]
        # Let's extend a to length 2n-1 (pad with zeros): a_ext = [a, 0, ..., 0]
        # Then (w ⋆ a_ext)[j] = sum_k w[k] * a_ext[k-j] = sum_k w[k] * a[k-j] * I(0<=k-j<n)
        # Hmm, that's not quite right either.
        
        # Let me just compute it directly:
        # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[j+m] (if j+m < 2n-1)
        
        # For efficiency, use FFT:
        # Define: x[m] = a[m] for m=0..n-1, 0 otherwise (length 2n-1)
        # y[k] = weights[k] for k=0..2n-2
        # Then sum_m x[m] * y[j+m] = (y * x_rev)[(2n-2) + j - (n-1)] ... 
        
        # OK let me just do it the simple way with a loop for clarity,
        # and use FFT if n is large enough.
        
        # Actually, the simplest correct formula:
        # grad[j] = 2 * sum_{k=0}^{2n-2} weights[k] * a[k-j] * I(0 <= k-j < n)
        # = 2 * sum_{k=j}^{j+n-1} weights[k] * a[k-j] * I(k < 2n-1)
        # = 2 * sum_{m=0}^{n-1} weights[j+m] * a[m] * I(j+m < 2n-1)
        
        # This is a "sliding dot product" of a with windows of weights.
        # We can compute all n values at once using FFT:
        # Let's define the cross-correlation: R[j] = sum_{m=0}^{n-1} a[m] * weights[j+m]
        # This is the cross-correlation of a (length n) with weights (length 2n-1)
        # at lag j.
        
        # Using FFT: R = ifft(fft(a, 3n-1) * conj(fft(weights, 3n-1)))  -- not quite
        # Actually: R[j] = sum_m a[m] * weights[j+m]
        # Let's reverse a: a_rev[m] = a[n-1-m]
        # R[j] = sum_m a[n-1-(n-1-m)] * weights[j+m] = sum_{m'} a[m'] * weights[j+n-1-m']
        # where m' = n-1-m, so m = n-1-m'
        # R[j] = sum_{m'=0}^{n-1} a[m'] * weights[j+n-1-m']
        # = (a * weights)[j+n-1] where * is convolution
        
        # So R = conv(a, weights)[n-1 : 2n-1] (taking appropriate slice)
        
        # conv(a, weights) has length n + (2n-1) - 1 = 3n-2
        # We want indices n-1 to 2n-2 (inclusive), which is n values.
        
        if n <= 2000:
            conv_aw = np.convolve(a, weights)
            grad = 2.0 * conv_aw[n-1:2*n-1]
        else:
            # FFT-based
            sz = 1
            while sz < 3 * n:
                sz <<= 1
            fa = rfft(a, sz)
            fw = rfft(weights, sz)
            conv_aw = irfft(fa * fw, sz)
            grad = 2.0 * conv_aw[n-1:2*n-1]
        
        # Project gradient to sum=0 (maintain normalization)
        grad_centered = grad - np.mean(grad)
        
        # Update
        a_new = normalize(a - lr * grad_centered)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Adaptive lr: increase slightly
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
        
        # Anneal temperature
        tau *= 0.995
    
    return best_a

def spectral_polish(a, max_iters=30, time_budget=20.0):
    """Frequency-domain POCS polishing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break
        
        # Compress the power spectrum
        alpha = 0.3 + 0.4 * (it / max_iters)
        P_new = (P / p_max) ** alpha * p_max
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.7 * a + 0.3 * a_new)
    
    return best_a
