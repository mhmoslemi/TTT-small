To maintain sum(a)=1, we project the gradient: `grad_projected[i] = grad[i] - mean(grad)`.
Then: `a_new = a - lr * grad_projected`, project to a>=0, renormalize.

This is essentially a projected gradient method on the simplex.

Let me also think about using **FISTA** (Fast Iterative Shrinkage-Thresholding Algorithm) for acceleration, or at least Nesterov momentum.

Now, let me also think about the **LP approach** more carefully. The previous code's LP formulation is:
- Given a (normalized), find g that minimizes M subject to conv(a,g)[k] <= M, sum(g)=1, g>=0.
- Then blend a <- (1-t)*a + t*g.

At a fixed point, a = g, so conv(a,a)[k] <= M for all k, and the LP constraints are tight at the active set. This means conv(a,a) is as flat as possible.

The issue with the LP approach is that it can get stuck in local minima. The gradient approach might escape these.

3. Use Nesterov momentum.
4. Periodically apply LP polishing.
5. Explore nearby n values.
6. Try different initializations.

Let me also consider: what if we use a **different kind of smoothing** instead of log-sum-exp? For example, we could use the power mean:
`max_approx(a, p) = (sum(a_i^p))^(1/p)` for large p.

This is also smooth for p > 1 and its gradient is easier to compute:
`d/d(a_i) (sum(conv^p))^(1/p) = (p-1)/p * (sum(conv^p))^(1/p-1) * conv[i]^(p-1) * d(conv[i])/d(a_i)`

For p=2 (RMS), this is just `conv[i] / sqrt(sum(conv^2))` times the gradient of conv[i].

Actually, for the purpose of minimizing the max, using a large p (like p=4 or p=8) gives a good approximation to max while being smooth.

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp and power-mean smoothed gradient descent with FFT-accelerated
gradients, Nesterov momentum, LP polishing, and multi-n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def fft_cross_corr(w, a, size):
    """Compute cross-correlation: h[m] = sum_j a[j]*w[j+m] for m=0..len(a)-1."""
    fw = rfft(w, size)
    fa = rfft(a, size)
    return irfft(fw * np.conj(fa), size)[:len(a)]

def smoothed_gradient(a, tau):
    """
    Compute gradient of tau * log-sum-exp(tau * conv(a,a)) / tau = log-sum-exp.
    Actually returns gradient of the LSE-smoothed max:
    d/d(a_i) [(1/tau) * log(sum(exp(tau * conv[a,a])))]
    = sum_k softmax(tau*conv)[k] * d(conv)[k]/d(a_i)
    = sum_k softmax(tau*conv)[k] * 2*a[k-i] (with bounds)
    = 2 * cross_corr(softmax_weights, a)[i]
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    w = np.exp(tau * (conv - max_conv))
    w = w / np.sum(w)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    h = fft_cross_corr(w, a, size)
    return 2.0 * h

def power_mean_gradient(a, p):
    """
    Compute gradient of (sum(conv(a,a)^p))^(1/p).
    = (1/p) * (sum(conv^p))^(1/p - 1) * sum_k p*conv[k]^(p-1) * d(conv[k])/d(a)
    = (sum(conv^p))^(1/p - 1) * sum_k conv[k]^(p-1) * d(conv[k])/d(a)
    = (sum(conv^p))^(1/p - 1) * 2 * cross_corr(conv^(p-1), a)
    """
    n = len(a)
    conv = compute_conv(a)
    conv_p = conv ** p
    sum_p = np.sum(conv_p)
    if sum_p < 1e-30:
        return np.zeros(n)
    scale = sum_p ** (1.0/p - 1.0)
    w = conv ** (p - 1)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    h = fft_cross_corr(w, a, size)
    return 2.0 * scale * h

def lse_gradient_descent(a, time_budget=60.0, initial_tau=10.0, verbose=False):
    """
    Log-sum-exp smoothed gradient descent with Nesterov momentum.
    """
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    v = np.zeros(n)
    lr = 0.01
    tau = initial_tau
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov lookahead
        a_y = a - v  # y = a - v (Nesterov extrapolation)
        y = np.maximum(a_y, 0)
        s_y = np.sum(y)
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y
        
        # Gradient
        grad = smoothed_gradient(y, tau)
        
        # Project gradient (remove mean to maintain sum=1)
        grad_proj = grad - np.mean(grad)
        
        # Update
        v_new = 0.9 * v  # momentum
        a_new = a - lr * grad_proj + v_new
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        v = v_new
        
        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive learning rate: reduce if oscillating
        if it % 100 == 99:
            if verbose and it % 500 == 499:
                print(f"    LSE it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}")
        
        # Increase tau slowly for sharper max approximation
        if it % 200 == 199:
            tau *= 1.2
            lr *= 0.8
        
        # Decrease lr if no improvement
        if it % 100 == 99 and compute_obj(a) > best_obj * 1.01:
            lr *= 0.5
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def lp_alternate_once(a, t=0.5):
    """One LP alternating step."""
    g, M = solve_lp_minpeak(a)
    if g is None:
        return a
    a_new = (1 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated log-sum-exp gradient descent with Nesterov momentum,
power-mean descent, LP alternating polishing, chase-peak heuristic,
multi-n exploration, and structured restarts.

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def _next_power_of_2(m):
    size = 1
    while size < m:
        size <<= 1
    return size


def compute_lse_gradient(a, tau):
    """
    Gradient of LSE-smoothed max(conv(a,a)):
    grad[i] = sum_k softmax(tau*conv)[k] * 2*a[k-i] (with bounds).
    Computed via FFT cross-correlation.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    w = np.exp(tau * (conv - max_conv))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum

    size = _next_power_of_2(2 * n - 1)
    fw = rfft(w, size)
    fa = rfft(a, size)
    h = irfft(fw * np.conj(fa), size)[:n]
    return 2.0 * h


def compute_power_mean_gradient(a, p):
    """
    Gradient of (sum(conv(a,a)^p))^(1/p) w.r.t. a.
    = 2 * (sum(conv^p))^(1/p - 1) * cross_corr(conv^(p-1), a)
    """
    n = len(a)
    conv = compute_conv(a)
    conv_p = conv ** p
    sum_p = float(np.sum(conv_p))
    if sum_p < 1e-300:
        return np.zeros(n)
    scale = sum_p ** (1.0 / p - 1.0)
    w = conv ** (p - 1)

    size = _next_power_of_2(2 * n - 1)
    fw = rfft(w, size)
    fa = rfft(a, size)
    h = irfft(fw * np.conj(fa), size)[:n]
    return 2.0 * scale * h


def lse_gradient_descent(a, time_budget=60.0, initial_tau=20.0, verbose=False):
    """LSE smoothed gradient descent with Nesterov momentum and adaptive params."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    lr = 0.008
    tau = initial_tau
    beta = 0.9
    no_improve = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        y = a - v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y

        # Compute gradient
        grad = compute_lse_gradient(y, tau)

        # Project gradient (remove mean to maintain sum constraint)
        grad_proj = grad - np.mean(grad)

        # Update with momentum
        v_new = beta * v
        a_new = a - lr * grad_proj + v_new
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        v = v_new
        a = a_new

        # Track best
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive scheduling
        if it % 500 == 499:
            if verbose:
                print(f"    LSE it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")
            # Increase tau for sharper max approximation

            # Slowly decrease learning rate
            lr *= 0.92
            # Reset momentum if stuck
            if no_improve > 200:
                v = np.zeros(n)
                no_improve = 0
                lr = max(lr * 0.5, 1e-5)

    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None

    a_new = np.maximum(a_new, 0)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
