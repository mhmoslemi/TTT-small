def compute_smooth_gradient(a, beta=10):
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    w = np.exp(beta * (conv - max_conv))
    w = w / np.sum(w)

    # This is a Hankel matrix

Now, the optimization algorithm:

1. Start with a (normalized, sum=1, a>=0).
2. Compute grad = smooth_gradient(a, beta).
3. Project grad onto the tangent space of the constraint set {sum=1, a>=0}:
   - Subtract mean: grad_tangent = grad - mean(grad)
4. Take a step: a_new = a - alpha * grad_tangent
5. Project to a >= 0: a_new = max(a_new, 0)
6. Renormalize: a_new = a_new / sum(a_new)
7. If objective improved, accept. Otherwise, reduce alpha.
8. Optionally increase beta over time (to sharpen the approximation of max).

This is essentially projected gradient descent on the smoothed objective.

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]

"""
Smoothed gradient descent with FFT-based subgradient, momentum, adaptive beta,
fast subgradient alternating (LP replacement), and periodic LP polish.
Key innovations: O(n log n) gradient via FFT, momentum acceleration,
adaptive softmax for global-to-local transition, multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    na, nb = len(a), len(b)
    total = na + nb - 1
    size = 1
    while size < total:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:total]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=5.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def smooth_gradient_fft(a, beta):
    """
    Compute smoothed subgradient of max(conv(a,a)) w.r.t. a using FFT.
    grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    where w[k] = exp(beta*conv[k]) / sum(exp(beta*conv)).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    w = np.exp(beta * (conv - max_c))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / w_sum

    # r[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Using: r[i] = conv_lin(a, w_rev)[(2n-2)-i] where w_rev = w[::-1]
    w_rev = w[::-1]

    total_len = n + (2 * n - 1) - 1  # = 3n-2
    N = 1
    while N < total_len:
        N <<= 1

    fa = rfft(a, N)
    fw = rfft(w_rev, N)
    conv_result = irfft(fa * fw, N)[:total_len]

    # r[i] = conv_result[(2n-2)-i] for i=0..n-1
    # So r[0] = conv_result[2n-2], r[1] = conv_result[2n-3], ..., r[n-1] = conv_result[n-1]
    r = conv_result[n - 1: 2 * n - 1][::-1]

    grad = 2.0 * r
    return grad, conv, w


def sgd_optimize(a, time_budget=60.0, initial_beta=2.0, final_beta=80.0,
                 momentum=0.85, verbose=False):
    """
    Smoothed gradient descent with momentum and adaptive beta.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    beta = initial_beta
    alpha = 0.5
    stall = 0
    best_stall = 0

    max_iters = 100000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, conv, w = smooth_gradient_fft(a, beta)
        grad = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad
        v = v - np.mean(v)

        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        # Line search
        best_t_obj = np.inf
        best_t = 0.0
        candidates = [alpha, alpha * 0.5, alpha * 0.25, alpha * 2.0, alpha * 0.1]
        for t in candidates:
            if t <= 0:
                continue
            a_try = a - t * v / v_norm
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0:
            a_new = a - best_t * v / v_norm
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                if best_t_obj < best_obj - 1e-14:
                    best_obj = best_t_obj
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                    beta = min(beta * 1.005, final_beta)
                    if best_t_obj < best_stall:

                    stall += 1
                    if stall > 30:
                        # Perturb to escape

def subgradient_alternate(a, time_budget=30.0, verbose=False):
    """
    Fast alternating: optimize g via subgradient (instead of full LP), then blend into a.
    Much faster than LP while still making progress.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    g = a.copy()
    alpha_g = 0.1

    for outer in range(200):
        if time.monotonic() - start > time_budget:
            break

        # Inner: optimize g to minimize max(conv(a, g)) via subgradient
        for inner in range(20):
            if time.monotonic() - start > time_budget:
                break
            conv_ag = compute_conv_cross(a, g)
            k_star = int(np.argmax(conv_ag))
            # Subgradient w.r.t. g: subgrad[j] = a[k_star - j] if 0 <= k_star - j < n
            subgrad = np.zeros(n)
            jmin = max(0, k_star - n + 1)
            jmax = min(n - 1, k_star)
            if jmin <= jmax:
                subgrad[jmin:jmax + 1] = a[k_star - jmax:k_star - jmin + 1]

            g = g - alpha_g * subgrad
            g = np.maximum(g, 0)
            s_g = float(np.sum(g))
            if s_g > 1e-10:
                g = g / s_g
            else:
                g = np.ones(n) / n

        # Blend a toward g
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a


def two_sided_optimize(a, time_budget=30.0):
    """
    Two-sided: optimize a via smoothed gradient AND via subgradient on the 'dual' side.
    The dual side uses the current a to define a linear problem in a new variable.
    """
    n = len(a)

    obj_sgd = compute_obj(a_sgd)
    if obj_sgd < best_obj:
        best_obj = obj_sgd

        obj_alt = compute_obj(a_alt)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([float(np.sum(a) / n)])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    n = len(a)

        improved = False

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Smoothed SGD on parent (main workhorse) ===
    print(f"\n=== Phase 1: Smoothed SGD (n={n_parent}) ===")
    phase1_budget = min(500.0, (deadline - start_time) * 0.40)
    a = best_seq.copy()

    # Run multiple rounds with increasing beta
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(phase1_budget / (4 - round_idx) * (1 + round_idx * 0.3), remaining * 0.6)
        beta_start = 2.0 * (round_idx + 1)
        beta_end = 30.0 * (round_idx + 1)

        a_new = sgd_optimize(a, time_budget=alloc, initial_beta=beta_start,
                             final_beta=beta_end, momentum=0.85, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  SGD Round {round_idx}: obj={best_score:.8f}")
        else:
            print(f"  SGD Round {round_idx}: obj={obj:.8f} (no improvement)")

    # === Phase 2: Subgradient alternating ===
    print(f"\n=== Phase 2: Subgradient Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_alt = subgradient_alternate(a, time_budget=min(remaining * 0.15, 60.0))
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            a = a_alt.copy()
            print(f"  Alt: obj={best_score:.8f} **")
        else:
            print(f"  Alt: obj={obj_alt:.8f}")

    # === Phase 3: LP Polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_lp = lp_polish(a, time_budget=min(remaining * 0.2, 80.0), max_rounds=8)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_parent - 100), max(200, n_parent - 50),
        n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        600, 700, 800, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick SGD
        alloc = min(remaining * 0.2, 40.0)
        a_opt = sgd_optimize(a_try, time_budget=alloc * 0.6, initial_beta=3.0, final_beta=40.0)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            # Polish with LP
            a_opt_lp = lp_polish(a_opt, time_budget=min(remaining * 0.1, 20.0))
            obj_try_lp = compute_obj(a_opt_lp)
            if obj_try_lp < best_score:
                best_score = obj_try_lp
                best_seq = a_opt_lp.copy()
                a = a_opt_lp.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                best_score = obj_try
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Random restarts ===
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    restarts = [
        ("perturb_best", best_seq * np.exp(rng.normal(0, 0.15, n_best))),
        ("perturb_best_small", best_seq * np.exp(rng.normal(0, 0.05, n_best))),
        ("uniform", np.ones(n_best) / n_best),
        ("cosine", (np.cos(np.linspace(0, np.pi, n_best)) ** 2 + 0.3) ),
        ("bimodal", np.exp(-((np.arange(n_best) - n_best // 2) ** 2) / (2 * (n_best // 5) ** 2)) +
                   0.5 * np.exp(-((np.arange(n_best) - n_best // 4) ** 2) / (2 * (n_best // 8) ** 2))),
    ]

    for name, a_init in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_init = np.maximum(a_init, 0)
        s_init = float(np.sum(a_init))
        if s_init < 1e-10:
            continue
        a_init = a_init / s_init

        alloc = min(remaining * 0.25, 45.0)
        a_opt = sgd_optimize(a_init, time_budget=alloc, initial_beta=2.0, final_beta=50.0)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Restart '{name}': obj={best_score:.8f} **")
        else:
            print(f"  Restart '{name}': obj={obj_opt:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        # SGD with high beta
        alloc = min(remaining * 0.3, 60.0)
        a_sgd = sgd_optimize(a_final, time_budget=alloc, initial_beta=20.0, final_beta=200.0,
                             momentum=0.9)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            a_final = a_sgd.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")
        else:
            print(f"  Final SGD: obj={obj_sgd:.8f}")

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_lp = lp_polish(a_final, time_budget=min(remaining * 0.3, 40.0), max_rounds=6)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_final = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
            else:
                print(f"  Final LP: obj={obj_lp:.8f}")

        # One more SGD round
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_last = sgd_optimize(a_final, time_budget=min(remaining * 0.5, 20.0),
                                  initial_beta=50.0, final_beta=500.0, momentum=0.95)
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Final SGD2: obj={best_score:.8f} **")
            else:
                print(f"  Final SGD2: obj={obj_last:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
