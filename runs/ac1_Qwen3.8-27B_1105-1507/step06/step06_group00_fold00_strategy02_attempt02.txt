def propose_candidate(budget_s=1500, seed=None):
    """
    Function to propose a candidate sequence for the autoconvolution minimization task.
    Uses a sequence of length 800 as a base (if provided) and employs dynamic optimization switched between L1 and L2.
    """
    strict_budget_s = budget_s  # To ensure not going over deadline
    parent_len = 800
    two_n = parent_len*2  # This doubled length will appear during optimization steps

    # Candidate generation infrastructure
    best_solution = None
    best_objective = float('inf')
    worst_case = np.asarray([0]*two_n, dtype=np.float64) # But the problem doesn't allow 0 as sum, so fixed to 0.01
    worst_case = worst_case / np.sum(worst_case) # This will be infinity because sum might be 0

    # Array science setup for search
    n_populations = 800
    try:
        # Use the provided parent sequence
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if np.all(parent_arr) and (np.sum(parent_arr) < 0.01):
            raise ValueError
        # Normalize
        total_initial = np.sum(parent_arr)
        if total_initial < 0.01:
            total_initial = 0.01
            parent_arr = parent_arr / (total_initial * 2)
        parent_normalized = parent_arr / total_initial
        best_objective = evaluate_sequence(parent_normalized.astype(float).tolist())
        if np.isfinite(best_objective) and best_objective < best_score:
            best_solution = parent_normalized.copy()
    except Exception as e:
        # Fallback to random start
        print(f"Warning: Could not use parent array. Initialization with exception. {e}")
        pass

    if best_solution is None:
        try:
            # Generate a random initial sequence of length (around) parent_len
            length_init = parent_len # We'll use the parent_len value as base
            # Generate random increases in length slowly? Not for first round.
            best_solution = random_init_for_length(length_init, random_state)
            best_objective = evaluate_sequence(best_solution.astype(float).tolist())
        except Exception as e:
            pass

    returnable_seq = [] 
    # Use the objective calculation to guide divergence (from parent's even/positives)
    time_remaining = budget_s - time.time()

    # If no parent or initial failed, return a best_last fallback (zeros are invalid)
    if time_remaining > 5:  # Given the above random_init_for_length, we need a valid fallback if no parent is used.
        trial_sequence = random_init_for_length(800, random_state, mode=2) # mode 2 gives a smooth one
        trial_obj = evaluate_sequence(trial_sequence.astype(float).tolist())
        if best_objective > trial_obj:
            best_solution = trial_sequence.copy()
            best_objective = trial_obj

    # We allocate each round with a budget proportional to the score's improvement
    # Training in the L2 space (as opposed to the raw overshoot proneness) is more robust but slower.
    # We decide to start with a fast gradient L1 optimization, then switch to Bregman L2 at the third optimization step.
    for iter_idx in range(1, 30): 
        if time_remaining - time.time() < 0: 
            break 
        # Defend against extremely shallow optimization spaces by filtering potential steps
        candidate_gen = AutoConvOptimizer(best_solution, time_remaining)
        score_improve = candidate_gen.search(iter_idx)
        # Update the best_objective with the new one if we found a better candidate
        if score_improve < best_solution:
            best_solution = candidate_gen.current_sequence.copy()
            best_objective = score_improve

    # Final validation
    final_score = evaluate_sequence(best_solution.astype(float).tolist())
    if final_score < best_objective:
        best_objective = final_score

    returnable_seq.append(best_solution.astype(float))
    # We must also ensure that the final sequence doesn't exceed the coefficient cap of 1000 (checked by evaluate_sequence)
    # But the problem confirms that defensive clipping already exists.

    # Return the best sequence found, validated against evaluate_sequence, and keep it within the bounds.
    # Since the problem expects a list of non-negative heights, do:
    valid_result = [float(xc) for xc in best_solution.astype(float) if xc > 0 and xc < 1000]
    return valid_result
