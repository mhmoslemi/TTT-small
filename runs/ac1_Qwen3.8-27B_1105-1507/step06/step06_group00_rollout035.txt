Let me also consider: the problem is to minimize `2n * max(conv(a,a)) / (sum(a))^2`. Since it's scale-invariant, I'll work with `sum(a) = 1`.

For the log-sum-exp approach:
- Objective: `f(a) = tau * log(sum(exp(conv(a,a)/tau)))` (this approximates `max(conv(a,a))` as `tau -> 0`)
- Constraint: `sum(a) = 1, a >= 0`

I'll use the softmax parameterization: `a_i = exp(x_i) / sum(exp(x_j))`.

The gradient of `f` w.r.t. `x`:
`df/dx_i = sum_j (df/da_j) * (da_j/dx_i)`
`da_j/dx_i = a_j * (delta_{ij} - a_i)` (softmax Jacobian)
`df/dx_i = a_i * (df/da_i) - a_i * sum_j (a_j * df/da_j)`
`= a_i * (df/da_i - sum_j a_j * df/da_j)`

Where `df/da_i = 2 * sum_l weights[i+l] * a[l]` (as computed above).

So `sum_j a_j * df/da_j = 2 * sum_j a_j * sum_l weights[j+l] * a[l]`

This can be computed efficiently.

Actually, let me just implement it and make sure it works. For n=800, each function evaluation involves one FFT (O(n log n)) and the gradient is also O(n log n). L-BFGS with maybe 200 iterations should take a few seconds.

```python
"""
Smooth optimization via log-sum-exp with temperature annealing,
combined with LP refinement for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lsq_objective_and_grad(x, tau, n):
    """
    Objective and gradient for log-sum-exp of conv(a,a), where a = softmax(x).
    Returns (f, grad_x).
    """
    # Compute a from x via softmax (with numerical stability)
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    
    # Compute convolution
    conv = compute_conv(a)
    
    # Log-sum-exp
    max_c = float(np.max(conv))
    exp_c = np.exp((conv - max_c) / tau)
    sum_exp = float(np.sum(exp_c))
    f = max_c + tau * np.log(sum_exp)
    
    # Weights (softmax of conv/tau)
    weights = exp_c / sum_exp
    
    # Gradient of f w.r.t. a:
    # df/da_i = 2 * sum_{l=0}^{n-1} weights[i+l] * a[l]
    # Compute via FFT (cross-correlation)
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(weights, size)
    a_rev = np.zeros(size)
    a_rev[:n] = a[::-1]
    AR = rfft(a_rev, size)
    conv_wa = irfft(W * AR, size)
    da = 2.0 * conv_wa[n-1:2*n-1]
    
    # Gradient w.r.t. x (softmax parameterization):
    # df/dx_i = a_i * (da_i - sum_j a_j * da_j)
    mean_da = float(np.sum(a * da))
    dx = a * (da - mean_da)
    
    return f, dx


def lsq_optimize(a_init, tau, maxiter=200, tol=1e-10):
    """Optimize a using log-sum-exp smooth objective with L-BFGS-B."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    x0 = np.log(a_init)
    
    result = minimize(
        lsq_objective_and_grad,
        x0,
        args=(tau, n),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'gtol': tol, 'ftol': 1e-14, 'maxls': 50}
    )
    
    x_opt = result.x
    x_shifted = x_opt - np.max(x_opt)
    a_opt = np.exp(x_shifted)
    a_opt = a_opt / np.sum(a_opt)
    return a_opt, result


def anneal_optimize(a_init, tau_start=0.1, tau_end=0.001, n_temps=15, 
                    maxiter_per_temp=100, time_budget=60):
    """
    Temperature annealing with log-sum-exp objective.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Log-spaced temperatures
    taus = np.geomspace(tau_start, tau_end, n_temps)
    
    for i, tau in enumerate(taus):
        if time.monotonic() - start > time_budget:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        maxiter = max(20, int(maxiter_per_temp * min(1.0, remaining / 10.0)))
        
        # Scale tau relative to current max conv
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        tau_scaled = tau * max_c
        
        a_new, result = lsq_optimize(a, tau_scaled, maxiter=maxiter)
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = best_a
        
        print(f"    Anneal step {i}: tau={tau:.4f}, obj={best_obj:.8f}, "
              f"status={result.status}")
    
    return best_a


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint using FFT-based evaluation
    # We need: for all k, sum_j a[k-j]*g[j] <= M (with appropriate bounds)
    # This is: (A @ g)[k] <= M where A[k,j] = a[k-j] if 0<=k-j<n else 0
    
    # For the LP, we'll use scipy linprog since cvxpy might be slow
    from scipy.optimize import linprog
    
    # Build A matrix (2n-1 x n): A[k,j] = a[k-j] if 0<=k-j<n else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        valid_k = np.arange(j, min(j+n, 2*n-1))
        A[valid_k, j] = a[:len(valid_k)]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def lp_polish(a, n_iters=20, time_budget=30):
    """LP-based polishing step."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)

            s = np.sum(a_try)
            if s < 1e-10:

            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_a = a_try.copy()
                a = a_try
                best_t = t
                break
    
    return best_a


def random_init(n, rng):
    """Generate random non-negative sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n)/n
    elif mode == 2:
        # Smooth (low-pass filtered noise)
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a / np.sum(a)
    elif mode == 3:
        # Gaussian blob
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        x = np.arange(n)
        a = np.exp(-(x-center)**2 / (2*width**2))
        a = a / np.sum(a)
    else:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2*np.pi*np.arange(n)/n)
        a = a / np.sum(a) + 1e-6
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
"""
Smooth optimization via log-sum-exp with temperature annealing and LP polishing
for minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses L-BFGS in softmax parameterization
with adaptive temperature scheduling, multi-n exploration, and diverse restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lsq_obj_grad(x, tau, n):
    """
    Log-sum-exp of conv(a,a) where a = softmax(x), plus gradient w.r.t. x.
    """
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)

    conv = compute_conv(a)
    max_c = float(np.max(conv))
    log_sum_exp = max_c + tau * np.log(float(np.sum(np.exp((conv - max_c) / tau))))

    weights = np.exp((conv - max_c) / tau)
    weights = weights / float(np.sum(weights))

    # da_i = 2 * sum_{l=0}^{n-1} weights[i+l] * a[l] via FFT cross-correlation
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(weights, size)
    a_rev = np.zeros(size)
    a_rev[:n] = a[::-1]
    AR = rfft(a_rev, size)
    conv_result = irfft(W * AR, size)
    da = 2.0 * conv_result[n - 1:2 * n - 1]

    # Gradient w.r.t. x via softmax Jacobian
    mean_da = float(np.dot(a, da))
    dx = a * (da - mean_da)

    return log_sum_exp, dx


def lsq_optimize(a_init, tau, maxiter=200):
    """Run L-BFGS on the log-sum-exp objective in softmax parameterization."""
    n = len(a_init)
    a_safe = np.maximum(a_init, 1e-10)
    a_safe = a_safe / np.sum(a_safe)
    x0 = np.log(a_safe)

    result = minimize(
        lsq_obj_grad,
        x0,
        args=(tau, n),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-14, 'maxls': 50}
    )

    x_opt = result.x
    x_shifted = x_opt - np.max(x_opt)
    a_opt = np.exp(x_shifted)
    a_opt = a_opt / np.sum(a_opt)
    return a_opt, result


def power_sum_obj_grad(x, p, n):
    """
    Lp norm of conv(a,a) where a = softmax(x), plus gradient.
    Uses (sum conv^p)^(1/p) as smooth max approximation.
    """
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)

    conv = np.maximum(compute_conv(a), 1e-30)
    conv_p = conv ** p
    sum_conv_p = float(np.sum(conv_p))
    f = sum_conv_p ** (1.0 / p)

    # df/dconv_k = (1/p) * sum_conv_p^(1/p - 1) * p * conv_k^(p-1)
    #            = sum_conv_p^(1/p-1) * conv_k^(p-1)
    coeff = sum_conv_p ** (1.0 / p - 1.0)
    weighted = coeff * (conv ** (p - 1))

    # da_i = 2 * sum_l weighted[i+l] * a[l]
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(weighted, size)
    a_rev = np.zeros(size)
    a_rev[:n] = a[::-1]
    AR = rfft(a_rev, size)
    conv_result = irfft(W * AR, size)
    da = 2.0 * conv_result[n - 1:2 * n - 1]

    mean_da = float(np.dot(a, da))
    dx = a * (da - mean_da)

    return f, dx


def power_sum_optimize(a_init, p, maxiter=200):
    """Run L-BFGS on the Lp-norm objective in softmax parameterization."""
    n = len(a_init)
    a_safe = np.maximum(a_init, 1e-10)
    a_safe = a_safe / np.sum(a_safe)
    x0 = np.log(a_safe)

    result = minimize(
        power_sum_obj_grad,
        x0,
        args=(p, n),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-14, 'maxls': 50}
    )

    x_opt = result.x
    x_shifted = x_opt - np.max(x_opt)
    a_opt = np.exp(x_shifted)
    a_opt = a_opt / np.sum(a_opt)
    return a_opt, result


def anneal_optimize(a_init, time_budget=60.0, use_power_sum=False):

    Multi-temperature annealing with smooth max approximation.
    Alternates between log-sum-exp and power-sum for robustness.

    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_steps = 20
    conv0 = compute_conv(a)
    scale = float(np.max(conv0))

        if time.monotonic() - start > time_budget:

        if use_power_sum and i >= n_steps // 2:

def solve_lp_minpeak(a, method='highs'):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))

    # Build A (2n-1 x n): A[k,j] = a[k-j] if 0 <= k-j < n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method=method)

    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def lp_polish(a, n_iters=30, time_budget=30.0):
    """LP-based polishing: alternate solving LP and blending."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        improved = False
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

    return best_a


def chase_peak(a, n_steps=200, time_budget=20.0):
    """Subgradient chase-the-peak method."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad -= np.mean(grad)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            step *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.7

    return best_a


def random_init(n, rng):
    """Generate diverse random initialization."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / (n + 1))
        a = a / np.sum(a)
    elif mode == 3:
        # Gaussian
        c = n * rng.uniform(0.3, 0.7)
        w = n * rng.uniform(0.15, 0.4)
        a = np.exp(-(np.arange(n) - c) ** 2 / (2 * w ** 2))
        a = a / np.sum(a)
    elif mode == 4:
        # Bimodal
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.1, 0.25)
        a = np.exp(-(np.arange(n) - c1) ** 2 / (2 * w ** 2)) + \
            np.exp(-(np.arange(n) - c2) ** 2 / (2 * w ** 2))
        a = a / np.sum(a)
    else:
        # Low-pass noise
        freq = rng.uniform(1, 4)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: log-sum-exp smooth optimization with temperature annealing,
    power-sum refinement, LP polishing, chase-peak, multi-n exploration,
    and diverse random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_seq = parent.copy()
                best_score = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=700): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Intensive annealed smooth optimization ===
    print(f"\n=== Phase 1: Annealed Smooth Opt (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a1 = anneal_optimize(best_seq, time_budget=min(remaining * 0.35, 350.0),
                             use_power_sum=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f}")

    # === Phase 2: Chase-peak + LP polish ===
    print(f"\n=== Phase 2: Chase + LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_cp = chase_peak(best_seq, n_steps=300, time_budget=min(remaining * 0.1, 20.0))
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  Chase: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_lp = lp_polish(best_seq, n_iters=25, time_budget=min(remaining * 0.2, 45.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: obj={best_score:.8f} **")
            else:
                print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: Second anneal round at lower temps ===
    print(f"\n=== Phase 3: Low-temp Anneal ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        a3 = anneal_optimize(best_seq, time_budget=min(remaining * 0.3, 200.0),
                             use_power_sum=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(100, n_best - 200), max(100, n_best - 100),
        n_best,
        n_best + 100, n_best + 200, n_best + 400,
        400, 500, 600, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 100 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.15, 35.0)
        a_opt = anneal_optimize(a_try, time_budget=alloc, use_power_sum=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse random restarts ===
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    for r in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        init_type = r % 5
        if init_type == 0:
            a_r = random_init(n_best, rng)
        elif init_type == 1:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.1, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            # Different n with perturbation
            n_r = int(n_best + rng.integers(-150, 250))
            n_r = max(150, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.08, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 3:
            # Symmetric perturbation
            a_r = best_seq.copy()
            a_sym = (a_r + a_r[::-1]) / 2.0
            a_r = a_sym * np.exp(rng.normal(0, 0.05, len(a_sym)))
            a_r = (a_r + a_r[::-1]) / 2.0
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            # Completely new structure
            n_r = int(rng.integers(400, 1200))
            a_r = random_init(n_r, rng)

        alloc = min(remaining * 0.12, 30.0)
        a_opt = anneal_optimize(a_r, time_budget=alloc, use_power_sum=(r % 3 == 0))
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(a_opt)
            print(f"  Restart {r} (type={init_type}, n={n_best}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Deep anneal
        a6 = anneal_optimize(best_seq, time_budget=min(remaining * 0.4, 180.0),
                             use_power_sum=True)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Deep anneal: obj={best_score:.8f} **")

        # Chase peak
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a7 = chase_peak(best_seq, n_steps=500, time_budget=min(remaining * 0.2, 20.0))
            obj7 = compute_obj(a7)
            if obj7 < best_score:
                best_score = obj7
                best_seq = a7.copy()
                print(f"  Final chase: obj={best_score:.8f} **")

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a8 = lp_polish(best_seq, n_iters=40, time_budget=min(remaining * 0.3, 40.0))
            obj8 = compute_obj(a8)
            if obj8 < best_score:
                best_score = obj8
                best_seq = a8.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # One more smooth pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a9 = anneal_optimize(best_seq, time_budget=min(remaining * 0.5, 30.0),
                                 use_power_sum=False)
            obj9 = compute_obj(a9)
            if obj9 < best_score:
                best_score = obj9
                best_seq = a9.copy()
                print(f"  Final smooth: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
