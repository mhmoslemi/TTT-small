- M[i+n-1, i] = a[n-1]

So yes, `M[i:i+n, i] = a`.

Let me verify with a small example: a = [a0, a1], g = [g0, g1].
conv = [a0*g0, a0*g1 + a1*g0, a1*g1]
M @ g:
M[0,0]*g0 = a0*g0 ✓
M[0,1]*g1 = 0 (since k-i = 0-1 = -1 < 0) ✓
M[1,0]*g0 + M[1,1]*g1 = a1*g0 + a0*g1 ✓
M[2,0]*g0 + M[2,1]*g1 = 0 + a1*g1 ✓

Great, so the matrix M has the property that M[i:i+n, i] = a for each i.

Now for the LP:
- Minimize M (scalar)
- Subject to: M_i_vec + sum_j M[k,j]*g[j] <= 0 for each k (i.e., M[k,:] @ g - M <= 0)
- sum(g) = 1
- g >= 0

In scipy linprog format:
- c = [0, 0, ..., 0, 1] (n+1 variables: g[0..n-1], M)
- A_ub[k, 0:n] = M_conv[k, :], A_ub[k, n] = -1
- b_ub = zeros(2n-1)
- A_eq = [[1, 1, ..., 1, 0]]
- b_eq = [1]
- bounds = [(0, None)]*n + [(0, None)]  (M >= 0)

This is correct. Let me code it up.

For efficiency, I'll precompute the convolution matrix structure and only update the values when `a` changes.

Actually, since we rebuild the matrix every iteration (because `a` changes), I'll just build it each time. For n=800, building a 1599x800 matrix takes maybe 1ms in numpy.

```python
"""
LP-alternating optimization with fast HiGHS solver for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: direct scipy linprog (no CVXPY overhead), adaptive line search,
monotone initialization, multi-n exploration, and periodic perturbation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fast_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M, sum(g)=1, g>=0, M>=0.
    Returns g (normalized to sum 1) or None.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    
    # Build convolution matrix M_conv: shape (2n-1, n)
    # M_conv[k, i] = a[k-i] if 0 <= k-i < n else 0
    M_conv = np.zeros((2 * n - 1, n))
    for i in range(n):
        M_conv[i:i + n, i] = a
    
    # LP: min [0,...,0,1] @ [g, M]
    # s.t. [M_conv, -1] @ [g, M] <= 0  (2n-1 constraints)
    #      [1,...,1, 0] @ [g, M] = 1
    #      g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = M_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def alternating_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    step_size = 0.3
    
    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        
        # Try several step sizes
        ts = [step_size * 0.5, step_size, step_size * 1.5, 0.8, 1.0]
        ts = [t for t in ts if 0 < t <= 1.0]
        
        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            # Gradually increase step size
            step_size = min(1.0, step_size * 1.05)
            if verbose and it % 50 == 0:
                print(f"    alt it{it}: obj={best_obj:.10f}")
        else:
            stall += 1
            step_size *= 0.9
            if stall > 30:
                # Perturb
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                step_size = 0.3
                stall = 0
    
    return best_a


def monotone_init(n, mode=0, rng=None):
    """Generate a monotone (or near-monotone) initial sequence."""
    if rng is None:
        rng = np.random.default_rng()
    
    if mode == 0:
        # Linear decreasing
        a = np.linspace(2.0, 0.1, n)
    elif mode == 1:
        # Geometric decreasing
        r = rng.uniform(0.95, 0.999)
        a = r ** np.arange(n)
    elif mode == 2:
        # Quadratic decreasing
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.05
    elif mode == 3:
        # Exponential with noise
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * x) + 0.1
        a += rng.normal(0, 0.02, n)
        a = np.maximum(a, 0)
    elif mode == 4:
        # Power law
        alpha = rng.uniform(0.5, 2.0)
        x = np.arange(1, n + 1)
        a = x ** (-alpha)
    elif mode == 5:
        # Sorted random
        a = np.sort(rng.uniform(0, 1, n))[::-1]
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def chase_peak_a(a, n_steps=200, time_budget=20.0):
    """Chase-the-peak: subgradient method targeting the max convolution bin."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = fast_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of conv[k_star] w.r.t. a[i]:
        # conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/da[i] = a[k-i] + a[i] (if k-i != i) or 2*a[i] (if k-i == i)
        # More carefully: conv[k] = sum_j a[j]*a[k-j]
        # d/da[i]: appears as a[i]*a[k-i] (when j=i) and a[k-i]*a[i] (when j=k-i)
        # If i != k-i: derivative = a[k-i] + a[i]
        # If i == k-i (i.e., 2i = k): derivative = 2*a[i]
        
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if j == i:
                    grad[i] = 2.0 * a[i]
                else:
                    grad[i] = a[j]  # The term a[i]*a[j] contributes a[j] to d/da[i]
                    # Wait, let me be more careful.
                    # conv[k] = sum_{j: valid} a[j] * a[k-j]
                    # The terms involving a[i] are:
                    #   - When j = i: a[i] * a[k-i]
                    #   - When k-j = i, i.e., j = k-i: a[k-i] * a[i]
                    # If i != k-i, these are two separate terms, each contributing a[k-i] to d(conv[k])/da[i]
                    # So total = 2 * a[k-i]... no wait.
                    # d(conv[k])/da[i] = a[k-i] (from j=i term: d(a[i]*a[k-i])/da[i] = a[k-i])
                    #                   + a[k-i] (from j=k-i term: d(a[k-i]*a[i])/da[i] = a[k-i])
                    # = 2*a[k-i] if i != k-i
                    # = 2*a[i] if i == k-i
                    grad[i] = 2.0 * a[j]
            else:
                grad[i] = 0
        
        # Project gradient onto sum=const hyperplane
        grad_centered = grad - np.mean(grad)
        
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step_size = min(1.0, step_size * 1.02)
            else:
                step_size *= 0.9
        else:
            step_size *= 0.5
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

Wait, I realize there might be an issue with the chase_peak gradient. Let me double-check.

`conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]`

The derivative w.r.t. `a[i]`:
- If `i = j` (i.e., the first factor is a[i]): contributes `a[k-i]` if `0 <= k-i < n`
- If `i = k-j` (i.e., the second factor is a[i], so j = k-i): contributes `a[k-i]` if `0 <= k-i < n`

Wait, that's the same condition. So:
- If `0 <= k-i < n` and `i != k-i`: derivative = `a[k-i] + a[k-i] = 2*a[k-i]`
- If `0 <= k-i < n` and `i == k-i` (i.e., `2i = k`): derivative = `2*a[i]`
- If `k-i < 0` or `k-i >= n`: derivative = 0

So in all cases where the derivative is non-zero, it's `2*a[k-i]` (and when `i = k-i`, this is `2*a[i]`).

So the gradient is:
`grad[i] = 2*a[k-i]` if `0 <= k-i < n`, else 0.

This is what I have in the code. Good.

But wait, there's a subtlety. The `conv` array has length `2n-1`, with indices 0 to 2n-2. The maximum could be at any of these positions. When `k = n-1` (the "center"), `k-i` ranges from `n-1` to `-1+... `, so for `i` from 0 to n-1, `k-i` ranges from n-1 to 0, which is always valid. So for k=n-1, all gradient components are non-zero.

For k=0: only `i=0` has `k-i=0` valid, so `grad[0] = 2*a[0]`, rest are 0.
For k=2n-2: only `i=n-1` has `k-i=n-1` valid, so `grad[n-1] = 2*a[n-1]`, rest are 0.

OK this all makes sense.

Now, one concern: the chase_peak method might be slow because it computes the full convolution each iteration. For n=800, the FFT-based convolution takes maybe 0.1ms, so 300 iterations = 30ms. That's fine.

Another concern: the LP solve time. For n=800, the LP has 801 variables and 1599+1=1600 constraints. With HiGHS, this should take about 5-20ms per solve. So in 100 seconds, we can do about 5000-20000 LP solves. That's plenty for convergence.

Let me also think about whether there's a way to speed up the LP. The main bottleneck is probably the dense matrix formulation. For n=800, the A_ub matrix is 1599 x 801 = 1.28M entries = 10MB. That's fine for dense.

For n=2000, it would be 3999 x 2001 = 8M entries = 64MB. Still fine.

Actually, I just realized I should also try a "direct" approach where I parameterize the sequence as monotone and use scipy's minimize with the peak convolution as the objective. But the peak is non-smooth (it's a max), so I'd need to handle subgradients or use a smooth approximation.

The smooth approximation: replace `max_k conv[k]` with `log(sum(exp(conv[k]/T) * T))` (softmax approximation). As T -> 0, this approaches the max. This gives a smooth objective that I can optimize with BFGS or L-BFGS-B.

Let me add this as another method:

```python
def softmax_objective(a_params, T=0.01):
    """Smooth approximation of the peak convolution objective."""
    a = np.maximum(a_params, 0)
    s = np.sum(a)
    if s < 0.01:
        return 1e10
    a = a / s
    n = len(a)
    conv = fast_conv(a)
    # Softmax approximation of max
    conv_scaled = conv / T
    conv_max = np.max(conv_scaled)
```

```python
"""
LP-alternating optimization with fast HiGHS solver, chase-the-peak with dual guidance,
monotone initialization, multi-n exploration, and adaptive perturbations.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fast_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fast_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix M (2n-1 x n) such that M @ g = conv(a, g)."""
    n = len(a)
    M = np.zeros((2 * n - 1, n))
    for i in range(n):
        M[i:i + n, i] = a
    return M


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0, M>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)

    M_mat = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = M_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                dual_ineq = res.ineqlin.marginals if hasattr(res, 'ineqlin') and res.ineqlin is not None else None
                return g_val / s, float(res.x[n]), dual_ineq
    except Exception:
        pass

    return None, None, None


def alternating_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating optimization with adaptive step size and dual-guided chase."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    step_size = 0.4

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp, dual = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search with multiple step sizes
        best_t_obj = best_obj
        best_a_local = best_a.copy()

        ts = sorted(set([step_size * 0.5, step_size, step_size * 1.5, 0.8, 1.0]))
        ts = [t for t in ts if 0 < t <= 1.0]

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            step_size = min(1.0, step_size * 1.08)
            if verbose and it % 100 == 0:
                print(f"      alt it{it}: obj={best_obj:.10f}")
        else:
            stall += 1
            step_size *= 0.92

            # If stalled, try dual-guided chase
            if stall % 10 == 5 and dual is not None:
                conv = fast_conv(a)
                k_star = int(np.argmax(conv))
                grad = np.zeros(n)
                for i in range(n):
                    j = k_star - i
                    if 0 <= j < n:
                        grad[i] = 2.0 * a[j]
                grad_centered = grad - np.mean(grad)
                a_try = a - 0.03 * grad_centered
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                        step_size = 0.4

                a = best_a * np.exp(rng_local.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                step_size = 0.4
                stall = 0

    return best_a


def chase_peak_all(a, n_steps=300, time_budget=25.0):
    """
    Chase-the-peak using all significant convolution peaks (not just the max).
    Uses a weighted subgradient approach.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.04

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = fast_conv(a)
        max_val = float(np.max(conv))

        # Find top peaks (within 90% of max)
        threshold = 0.9 * max_val
        peak_indices = np.where(conv >= threshold)[0]

        # Weighted subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            weight = conv[k] / max_val
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    grad[i] += weight * 2.0 * a[j]

        grad_centered = grad - np.mean(grad)

        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step_size = min(0.5, step_size * 1.03)
            else:
                # Try half step
                a_new2 = a - step_size * 0.5 * grad_centered
                a_new2 = np.maximum(a_new2, 0)
                s_new2 = float(np.sum(a_new2))
                if s_new2 > 1e-10:
                    a_new2 = a_new2 / s_new2
                    obj_new2 = compute_obj(a_new2)
                    if obj_new2 < best_obj:
                        best_obj = obj_new2
                        best_a = a_new2.copy()
                        a = a_new2
                    else:
                        step_size *= 0.85
                else:
                    step_size *= 0.7
        else:
            step_size *= 0.5

    return best_a


def monotone_init(n, mode=0, rng=None):
    """Generate a monotone or structured initial sequence."""
    if rng is None:
        rng = np.random.default_rng()

    if mode == 0:
        a = np.linspace(2.0, 0.1, n)
    elif mode == 1:
        r = rng.uniform(0.97, 0.999)
        a = r ** np.arange(n)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.05
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * x) + 0.08
    elif mode == 4:
        alpha = rng.uniform(0.3, 1.5)
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** (-alpha)
    elif mode == 5:
        a = np.sort(rng.uniform(0.1, 1.0, n))[::-1]
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + x)
    elif mode == 7:
        # S-curve (sigmoid-like)
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(8 * (x - 0.5))) + 0.05
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def pairwise_refine(a, time_budget=15.0):
    """
    Pairwise mass transfer: move mass from peak-contributing positions
    to positions that reduce the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = fast_conv(a)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        max_val = float(conv[k_star])
        improved = False

        # Find positions that contribute to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((i, j, a[i] * a[j] if j != i else a[i] ** 2))

        contributors.sort(key=lambda x: -x[2])

        # Try moving mass from high-contributing to low-contributing positions
        for i, j, contrib in contributors[:20]:
            if time.monotonic() - start > time_budget:
                break
            if i == j:
                continue
            if a[i] <= a[j]:
                continue

            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                new_conv = fast_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    break

        if not improved:
            break

    return best_a

    n_options = [n for n in n_options if 300 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)

        alloc = min(remaining * 0.12, 70.0)
        a_opt = alternating_optimize(a_try, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

        # Try monotone init for this n
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_mono = monotone_init(n_try, mode=rng.integers(0, 8), rng=rng)
            alloc2 = min(remaining * 0.06, 30.0)
            a_opt2 = alternating_optimize(a_mono, time_budget=alloc2)
            obj2 = compute_obj(a_opt2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_opt2.copy()
                n_best = len(best_seq)
                print(f"  ** n={n_try} mono: obj={best_score:.8f} **")

    # === Phase 4: Random restarts ===
    print(f"\n=== Phase 4: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        init_type = r % 4
        if init_type == 0:
            a_r = best_seq * np.exp(rng.normal(0, 0.08, len(best_seq)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 1:
            n_r = n_best + rng.integers(-200, 300)
            n_r = max(300, min(2000, n_r))
            a_r = resample_seq(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            a_r = monotone_init(n_best, mode=rng.integers(0, 8), rng=rng)
        else:
            n_r = n_best
            x = np.linspace(0, 1, n_r)
            # Smooth random perturbation of uniform
            freqs = rng.integers(1, 4, size=2)
            phases = rng.uniform(0, 2 * np.pi, size=2)
            amps = rng.uniform(0.1, 0.4, size=2)
            a_r = 1.0
            for f, p, amp in zip(freqs, phases, amps):
                a_r += amp * np.cos(2 * np.pi * f * x + p)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        alloc = min(remaining * 0.08, 45.0)
        a_opt = alternating_optimize(a_r, time_budget=alloc)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (n={n_best}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.35, 100.0)
        a_opt = alternating_optimize(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cp = chase_peak_all(best_seq, n_steps=500, time_budget=min(remaining * 0.25, 35.0))
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  Final Chase: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_pw = pairwise_refine(best_seq, time_budget=min(remaining * 0.2, 15.0))
        obj_pw = compute_obj(a_pw)
        if obj_pw < best_score:
            best_score = obj_pw
            best_seq = a_pw.copy()
            print(f"  Final Pairwise: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_opt = alternating_optimize(best_seq, time_budget=min(remaining * 0.4, 50.0))
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Final LP2: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
