"""
Combined optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT flat-spectrum iteration, mirror descent with min-norm subgradient,
LP alternating, multi-n exploration, and random restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def fft_flat_spectrum(a, n_iters=200, time_budget=60.0, relaxation=0.5):
    """FFT-based flat spectrum iteration (Gerchberg-Saxton variant)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, L)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        phase = A / mag_safe

        # Flat target: |A| = 1 for all bins
        A_flat = phase
        # Relaxation
        A_new = (1 - relaxation) * A + relaxation * A_flat

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    # Perturb
                    rng = np.random.default_rng(it)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0

    return best_a


def mirror_descent(a, n_iters=2000, time_budget=60.0):
    """Mirror descent (exponential) with multi-peak subgradient."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = np.maximum(a, 1e-15) / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.1
    m_peaks = 5

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top m peaks
        top_idx = np.argsort(conv)[-m_peaks:]
        top_val = conv[top_idx]
        max_val = top_val[-1]

        # Softmax weights
        temp = max(0.001, (max_val - top_val[0]) * 0.5)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        # Weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * a[j]

        # Mirror descent step
        a_new = a * np.exp(-alpha * grad)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                alpha *= 0.98
        else:
            alpha *= 0.5

        # Periodically increase alpha slightly to escape
        if it % 500 == 499:
            alpha = min(alpha * 1.1, 0.5)

    return best_a


def subgradient_nesterov(a, n_iters=2000, time_budget=60.0):
    """Subgradient method with Nesterov momentum and multi-peak."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    beta = 0.85
    m_peaks = 5

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[-m_peaks:]
        top_val = conv_y[top_idx]
        max_val = top_val[-1]

        temp = max(0.001, (max_val - top_val[0]) * 0.5)
        weights = np.exp((top_val - max_val) / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    grad[i] += w * y[j]

        grad = grad - np.mean(grad)

        alpha = 2.0 / (np.linalg.norm(grad) * (it + 10))
        alpha = min(alpha, 0.5)

        v = beta * v + (1 - beta) * grad
        y_new = y - alpha * v

        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new > 1e-10:
            y_new = y_new / s_new
            obj_new = compute_obj(y_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = y_new.copy()
            y = y_new

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Smooth random via random Fourier features
        freqs = rng.uniform(0, 4, 8)
        phases = rng.uniform(0, 2 * np.pi, 8)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        c1, c2 = n // 3, 2 * n // 3
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    else:
        # Random with structure
        a = rng.uniform(0, 1, n)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    return a

        alloc = min(remaining * 0.1, 60.0)
        a_fft = fft_flat_spectrum(best_seq, n_iters=500, time_budget=alloc, relaxation=0.5)
        obj = compute_obj(a_fft)
        print(f"  FFT (relax=0.5): obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_fft.copy()

        # Try different relaxation
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_fft2 = fft_flat_spectrum(best_seq, n_iters=300, time_budget=min(remaining * 0.05, 30.0), relaxation=0.3)
            obj2 = compute_obj(a_fft2)
            print(f"  FFT (relax=0.3): obj={obj2:.8f}")
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_fft2.copy()

    # Phase 2: Mirror descent (exponential weights)
    print(f"\n=== Phase 2: Mirror Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 120.0)
        a_md = mirror_descent(best_seq, n_iters=5000, time_budget=alloc)
        obj = compute_obj(a_md)
        print(f"  Mirror descent: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md.copy()

    # Phase 3: LP alternating (principled but slower)
    print(f"\n=== Phase 3: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Multiple rounds with decreasing budget
        n_rounds = 3
        for rnd in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.2, 80.0)
            a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
            obj = compute_obj(a_lp)
            print(f"  LP round {rnd}: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()

    # Phase 4: More mirror descent (refined)
    print(f"\n=== Phase 4: Refined Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_md2 = mirror_descent(best_seq, n_iters=3000, time_budget=alloc)
        obj = compute_obj(a_md2)
        print(f"  MD round 2: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_md2.copy()

    # Phase 5: Nesterov subgradient
    print(f"\n=== Phase 5: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_ns = subgradient_nesterov(best_seq, n_iters=3000, time_budget=alloc)
        obj = compute_obj(a_ns)
        print(f"  Nesterov: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_ns.copy()

    # Phase 6: Coordinate descent
    print(f"\n=== Phase 6: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.08, 40.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj = compute_obj(a_cd)
        print(f"  Coord descent: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()

    # Phase 7: Multi-n exploration
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200,
        500, 600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if abs(n_try - n_best) < 10:
            continue  # Skip if too close to current best

        a_try = resample_sequence(best_seq, n_try)

        # Quick FFT + mirror descent
        alloc1 = min(remaining * 0.1, 20.0)
        a_opt = fft_flat_spectrum(a_try, n_iters=200, time_budget=alloc1, relaxation=0.5)

        alloc2 = min(remaining * 0.1, 20.0)
        a_opt = mirror_descent(a_opt, n_iters=2000, time_budget=alloc2)

        obj = compute_obj(a_opt)
        status = " **" if obj < best_score else ""
        print(f"  n={n_try}: obj={obj:.8f}{status}")
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # Phase 8: Random restarts
    print(f"\n=== Phase 8: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    restart_budget = min(40.0, remaining * 0.08)

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < restart_budget + 10:
            break

        init_type = r % 5
        if init_type == 0:
            a_r = random_init(n_best, rng)
        elif init_type == 1:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            # Different n
            n_r = int(n_best + rng.integers(-150, 150))
            n_r = max(200, min(1200, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 3:
            # Cosine window
            x = np.linspace(0, np.pi, n_best)
            a_r = np.cos(x) ** 2 + 0.3
            a_r = a_r / np.sum(a_r)
        else:
            # Anti-symmetric perturbation
            a_r = best_seq.copy()
            mid = n_best // 2
            a_r[:mid] *= 1.3
            a_r[mid:] *= 0.8
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        # Optimize
        alloc_fft = min(restart_budget * 0.3, 10.0)
        a_opt = fft_flat_spectrum(a_r, n_iters=200, time_budget=alloc_fft, relaxation=0.5)
        alloc_md = min(restart_budget * 0.5, 15.0)
        a_opt = mirror_descent(a_opt, n_iters=2000, time_budget=alloc_md)

        obj_r = compute_obj(a_opt)
        status = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={obj_r:.8f}{status}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()

    # Phase 9: Final intensive polish
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_final = len(best_seq)
        a_final = best_seq.copy()

        # LP
        alloc = min(remaining * 0.3, 60.0)
        a_lp = lp_alternating(a_final, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        print(f"  Final LP: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            a_final = a_lp.copy()

        # Mirror descent
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.25, 40.0)
            a_md = mirror_descent(a_final, n_iters=3000, time_budget=alloc)
            obj = compute_obj(a_md)
            print(f"  Final MD: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_md.copy()
                a_final = a_md.copy()

        # FFT
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.2, 20.0)
            a_fft = fft_flat_spectrum(a_final, n_iters=300, time_budget=alloc, relaxation=0.4)
            obj = compute_obj(a_fft)
            print(f"  Final FFT: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_fft.copy()
                a_final = a_fft.copy()

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.2, 20.0)
            a_cd = coordinate_descent(a_final, time_budget=alloc)
            obj = compute_obj(a_cd)
            print(f"  Final CD: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.4, 40.0)
            a_last = lp_alternating(best_seq, time_budget=alloc)
            obj = compute_obj(a_last)
            print(f"  Final LP2: obj={obj:.8f}")
            if obj < best_score:
                best_score = obj
                best_seq = a_last.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
