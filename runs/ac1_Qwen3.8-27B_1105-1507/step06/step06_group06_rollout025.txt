def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)

def compute_gradient(a, conv, tau):
    """Compute gradient of log-sum-exp smoothed max of conv w.r.t. a.
    Assumes sum(a) = 1. Returns gradient of L = tau*log(sum(exp(conv[k]/tau)))
    """
    n = len(a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    p = np.exp(shifted)
    p_sum = np.sum(p)
    p = p / p_sum

    # But we need to be careful about the definition
    
    # Actually, let's use the direct formula:
    # dL/da[i] = 2 * sum_{j=0}^{n-1} p[j+i] * a[j] (where j+i must be < 2n-1)

    # This is ifft(fft(p) * conj(fft(a))) with proper zero-padding

Let me also think about the **spectral shaping** more carefully.

The idea: we want |F(a)|^2 (power spectrum) to be as flat as possible, because a flat power spectrum means the autocorrelation has low sidelobes.

But we can't just make |F| flat because we need a to be real and non-negative.

A practical approach:
1. Compute F = fft(a)
2. Compute |F|^2 (power spectrum)
3. Apply a "flattening" operation: |F_new|^2 = |F|^2 * (target / |F|^2)^alpha = |F|^(2-2*alpha) * target^alpha
   For alpha=1, this makes |F| constant (perfectly flat), which gives a delta autocorrelation (impossible for real signals).
   For alpha=0.5, |F_new|^2 = |F|, which compresses the dynamic range.
4. Keep the phase of F, compute F_new = |F_new| * exp(i*angle(F))
5. a_new = ifft(F_new).real
6. Project to non-negative, sum=1

This is similar to what the previous code does but with a different parameterization.

Optimized step function search using FFT-based proximal gradient with Nesterov
momentum, adaptive log-sum-exp temperature, spectral shaping, and LP polishing.
Key improvements: O(n log n) gradient via cross-correlation FFT, adaptive
smoothing, and aggressive multi-length exploration.

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    a_norm = a / sum_a
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient_smoothed(a, conv, tau):
    """Compute gradient of tau*log(sum(exp(conv[k]/tau))) w.r.t. a.
    a should have sum=1. Returns gradient vector of length n.
    """
    n = len(a)
    c = conv
    c_max = float(np.max(c))
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500.0, 500.0)
    p = np.exp(shifted)
    p_sum = float(np.sum(p))
    if p_sum < 1e-300:
        # Fallback: uniform over top peaks
        p = np.zeros(len(c))
        top_k = int(np.argmax(c))
        p[top_k] = 1.0
    else:
        p = p / p_sum
    
    # Gradient = 2 * cross_correlation(p, a)
    # corr[i] = sum_m p[m+i] * a[m]
    # FFT: corr = ifft(conj(fft(p)) * fft(a))
    N_fft = 1
    while N_fft < len(p) + n - 1:
        N_fft <<= 1
    
    fp = fft(p, N_fft)
    fa = fft(a, N_fft)
    corr = ifft(np.conj(fp) * fa, N_fft).real[:n]
    
    grad = 2.0 * corr
    return grad

def compute_smoothed_max(c, tau):
    c_max = float(np.max(c))
    shifted = np.clip((c - c_max) / tau, -500.0, 500.0)
    log_sum = c_max + tau * np.log(np.sum(np.exp(shifted)))
    return log_sum


def spectral_shaping(a, alpha=0.5, time_budget=20.0):
    """
    Flatten the power spectrum to reduce autocorrelation peaks.
    a should have sum=1.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

            break
        
        F = fft(a, N_fft)
        power = np.abs(F) ** 2

        power_norm = power / p_max
        power_compressed = power_norm ** (1 - alpha) * p_max

        # For real signal: F[k] = conj(F[N-k])
        # We modify magnitude, keep phase
        phase = np.angle(F)
        new_mag = np.sqrt(power_compressed)
        F_new = new_mag * np.exp(1j * phase)
        
        # Enforce Hermitian symmetry
        F_new[0] = np.abs(F_new[0])  # DC is real positive

    return A

    a_new = np.interp(x_new, x_old, a)
    return project_nonneg_sum1(a_new)

"""
Optimized step function search using FFT-based proximal gradient with momentum,
adaptive log-sum-exp temperature, spectral shaping with Hermitian symmetry,
LP alternating, and multi-length exploration.
Key improvements over previous: O(n log n) gradient via cross-correlation FFT,
Nesterov acceleration, better spectral method, and aggressive time-adaptive search.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    a_norm = a / sum_a
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient_smoothed(a, conv, tau):
    """
    Compute gradient of tau*log(sum(exp(conv[k]/tau))) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) complexity.
    a should have sum(a)=1.
    """
    n = len(a)
    c = conv
    c_max = float(np.max(c))
    shifted = (c - c_max) / tau
    shifted = np.clip(shifted, -500.0, 500.0)
    p = np.exp(shifted)
    p_sum = float(np.sum(p))
    if p_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / p_sum

    # Gradient = 2 * cross_correlation(p, a)
    # corr[i] = sum_m p[m+i] * a[m]
    # FFT: corr = ifft(conj(fft(p)) * fft(a))
    N_fft = 1
    while N_fft < len(p) + n - 1:
        N_fft <<= 1

    fp = fft(p, N_fft)
    fa = fft(a, N_fft)
    corr = ifft(np.conj(fp) * fa, N_fft).real[:n]

    grad = 2.0 * corr
    # Center for sum-projected optimization
    grad = grad - np.mean(grad)
    return grad


def momentum_optimize(a, max_iters=200, time_budget=30.0,
                      tau_start=0.1, tau_end=0.005,
                      lr_start=0.04, momentum=0.85):
    """
    Momentum-based proximal gradient on smoothed max objective.
    a should have sum(a)=1.
    Returns (best_a, best_obj).
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(max_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (0.997 ** it)

        conv = compute_conv(a)
        grad = compute_gradient_smoothed(a, conv, tau)

        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def nesterov_optimize(a, max_iters=200, time_budget=30.0,
                      tau_start=0.1, tau_end=0.005,
                      lr_start=0.03):
    """
    Nesterov-accelerated proximal gradient on smoothed max objective.
    Returns (best_a, best_obj).
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = a.copy()
    t = 1.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(max_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (0.998 ** it)

        # Nesterov look-ahead
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = a + beta * (a - prev_a)
        y = np.maximum(y, 0)

        # Gradient at look-ahead point
        conv_y = compute_conv(y)
        grad_y = compute_gradient_smoothed(y, conv_y, tau)

        # Update
        a_new = a - lr * grad_y
        a_new = project_nonneg_sum1(a_new)

        prev_a = a
        a = a_new
        t = t_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def spectral_shaping(a, alpha=0.4, time_budget=15.0):
    """
    Flatten the power spectrum to reduce autocorrelation peaks.
    Enforces Hermitian symmetry for real-valued output.
    Returns (best_a, best_obj).
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 2 * n:
        N_fft <<= 1

    for it in range(30):
        if time.monotonic() - start > time_budget:
            break

        F = fft(a, N_fft)
        power = np.abs(F) ** 2
        p_max = float(np.max(power))
        if p_max < 1e-30:
            break

        # Compress power spectrum: reduce dynamic range
        power_norm = power / p_max
        power_compressed = power_norm ** (1.0 - alpha) * p_max

        # Reconstruct with modified magnitude, preserved phase
        phase = np.angle(F)
        new_mag = np.sqrt(power_compressed)
        F_new = new_mag * np.exp(1j * phase)

        # Enforce Hermitian symmetry
        F_new[0] = np.abs(F_new[0])
        if N_fft % 2 == 0:
            F_new[N_fft // 2] = np.abs(F_new[N_fft // 2])
        for k in range(1, N_fft // 2 + 1):
            F_new[N_fft - k] = np.conj(F_new[k])

        a_new = ifft(F_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_alternate(a, max_iters=20, time_budget=30.0):
    """LP alternating method for local refinement."""
    n = len(a)
    if n > 1200:
        return a, compute_obj(a)

    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    A = build_toeplitz(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 8})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s < 1e-10:
                break
            g = g / s
        except Exception:
            break

        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            a_try = (1.0 - best_t) * a + best_t * g
            a = project_nonneg_sum1(a_try)
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
            if it < max_iters - 1:
                A = build_toeplitz(a)
        else:
            stall += 1
            if stall > 5:
                break

    return best_a, best_obj


def chase_peak(a, n_steps=50, base_lr=0.02):
    """Chase the peak using FFT-based subgradient."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for step in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient at the peak: grad[i] = 2*a[k_star - i] for valid i
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        grad_centered = grad - np.mean(grad)

        improved = False
        lr = base_lr / (1.0 + step * 0.02)
        for factor in [1.0, 0.5, 0.25, 0.1]:
            a_try = a - lr * factor * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved and step > 15:
            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_nonneg_sum1(a_new)

                a_pert = a_main * np.exp(rng.normal(0, 0.04, n_main))
                a_pert = project_nonneg_sum1(a_pert)
                a_pert, obj_pert = chase_peak(a_pert, n_steps=20, base_lr=0.02)
                a_pert, obj_pert = momentum_optimize(a_pert, max_iters=30, time_budget=8.0,
                                                     tau_start=0.05, tau_end=0.01, lr_start=0.02)
                if obj_pert < local_obj:
                    local_obj = obj_pert
                    a_main = a_pert.copy()
                    stall = 0
                    if local_obj < best_score:
                        best_score = local_obj
                        best_seq = a_main.copy()
                        print(f"  Loop {loop_count} (perturb): obj={best_score:.8f}")
            stall = 0

    print(f"  Phase 2 final: obj={local_obj:.8f}")

    # === Phase 3: Length fine-tuning ===
    print(f"\n=== Phase 3: Length Fine-tuning ===")
    print(f"Best so far: {best_score:.8f}")
    phase3_budget = min(200.0, (deadline - time.monotonic()) * 0.15)
    phase3_start = time.monotonic()

    n_best = len(best_seq)
    n_tries = sorted(set([n_best + d for d in [-40, -20, -10, 10, 20, 40, 60]]))
    n_tries = [n for n in n_tries if 200 <= n <= 2000 and n != n_best]

    for n_try in n_tries:
        if time.monotonic() - phase3_start > phase3_budget:
            break

        remaining3 = phase3_start + phase3_budget - time.monotonic()
        if remaining3 < 8:
            break

        a_res = resample_sequence(best_seq, n_try)
        a_opt, obj_opt = chase_peak(a_res, n_steps=20, base_lr=0.02)

        rem = phase3_start + phase3_budget - time.monotonic()
        if rem > 5:
            a_opt, obj_opt = nesterov_optimize(
                a_opt, max_iters=50, time_budget=min(rem * 0.3, 15.0),
                tau_start=0.05, tau_end=0.005, lr_start=0.02
            )

        rem = phase3_start + phase3_budget - time.monotonic()
        if rem > 5 and n_try <= 1000:
            a_opt, obj_opt = lp_alternate(a_opt, max_iters=8, time_budget=min(rem * 0.3, 12.0))

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")

    if best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)
        local_final = compute_obj(a_final)

        while time.monotonic() < deadline - 10:
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            # Fine Nesterov
            a_p, obj_p = nesterov_optimize(
                a_final, max_iters=min(60, int(rem)),
                time_budget=min(rem * 0.25, 20.0),
                tau_start=0.003, tau_end=0.001,
                lr_start=0.008
            )
            if obj_p < local_final:
                local_final = obj_p
                a_final = a_p.copy()
                if local_final < best_score:
                    best_score = local_final
                    best_seq = a_final.copy()
                    print(f"  Polish nest: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 5:
                break

            # Chase peak (very fine)
            a_p2, obj_p2 = chase_peak(a_final, n_steps=min(30, int(rem * 2)), base_lr=0.005)
            if obj_p2 < local_final:
                local_final = obj_p2
                a_final = a_p2.copy()
                if local_final < best_score:
                    best_score = local_final
                    best_seq = a_final.copy()
                    print(f"  Polish chase: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 5:
                break

            # LP
            if n_final <= 1000:
                a_p3, obj_p3 = lp_alternate(a_final, max_iters=10,
                                            time_budget=min(rem * 0.2, 15.0))
                if obj_p3 < local_final:
                    local_final = obj_p3
                    a_final = a_p3.copy()
                    if local_final < best_score:
                        best_score = local_final
                        best_seq = a_final.copy()
                        print(f"  Polish LP: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 3:
                break

            # Spectral
            a_p4, obj_p4 = spectral_shaping(a_final, alpha=0.2, time_budget=min(rem * 0.15, 8.0))
            if obj_p4 < local_final:
                local_final = obj_p4
                a_final = a_p4.copy()
                if local_final < best_score:
                    best_score = local_final
                    best_seq = a_final.copy()
                    print(f"  Polish spec: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"\n  evaluate_sequence returned: {val:.8f}")

    if val < np.inf:
        best_score = min(best_score, val)
    else:
        print("  WARNING: fallback to uniform")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
