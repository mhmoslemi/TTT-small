"""
Ramp-initialized optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: a linear ramp already achieves ~4/3 objective, far below the previous
best of 1.5065. We start from ramp/power-ramp initializations and refine using
LP alternating, subgradient descent on the peak, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def ramp_init(n, power=1.0, flat_frac=0.0):
    """Generate a power ramp initialization normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    if flat_frac > 0:
        flat_n = int(n * flat_frac)
        a = np.zeros(n)
        a[:flat_n] = 0.5
        if n - flat_n > 0:
            ramp_i = np.arange(1, n - flat_n + 1, dtype=np.float64)
            a[flat_n:] = ramp_i ** power
    else:
        a = (i + 1.0) ** power
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def build_toeplitz_for_lp(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 3000:
        return None
    A = build_toeplitz_for_lp(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate(a, max_iters=100, time_budget=60.0):
    """LP alternating with adaptive line search."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=20)
        if g is None:
            break

        # Line search
        candidate_ts = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a, best_obj


def subgradient_peak(a, max_iters=500, time_budget=30.0, tau=0.01):
    """Subgradient descent on the smoothed max of the convolution."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.5

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights over peaks
        shifted = (conv - cmax) / tau
        weights = np.exp(np.clip(shifted, -500, 0))
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient: grad[j] = sum_k weights[k] * 2*a[k-j] (if 0<=k-j<n)
        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            if weights[k_idx] < 1e-14:
                continue
            j_start = max(0, k_idx - n + 1)
            j_end = min(n, k_idx + 1)
            if j_end > j_start:
                js = np.arange(j_start, j_end)
                a_idx = k_idx - js
                grad[js] += weights[k_idx] * 2.0 * a[a_idx]

        # Center the gradient (to maintain sum=1 after projection)
        grad_centered = grad - np.mean(grad)

        # Step
        a_new = a - step * grad_centered
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            step *= 0.95
            a = project_nonneg_sum1(a - 0.5 * step * grad_centered)
            obj_a = compute_obj(a)
            if obj_a < best_obj:
                best_obj = obj_a
                best_a = a.copy()

        if it % 100 == 0:
            step = max(0.01, 0.5 * 0.998 ** it)

    return best_a, best_obj


def chase_peak(a, n_steps=100, step_size=0.05):
    """Chase-the-peak: iteratively reduce the maximum convolution peak."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient at the peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
        grad *= 2.0

        grad_centered = grad - np.mean(grad)

        # Try a few step sizes
        improved = False
        for ss in [step_size, step_size * 0.5, step_size * 2.0, step_size * 0.25]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_obj


def multi_start_optimize(a_init, time_budget=60.0):
    """Apply multiple optimization strategies in sequence."""
    a = project_nonneg_sum1(a_init.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    n = len(a)

    # Chase peak first (quick wins)
    if time.monotonic() - start < time_budget * 0.3:
        a_c, obj_c = chase_peak(a, n_steps=50, step_size=0.05)
        if obj_c < best_obj:
            best_obj = obj_c
            best_a = a_c.copy()
            a = a_c.copy()

    # LP alternating
    if time.monotonic() - start < time_budget * 0.7 and n <= 2000:
        remaining = time_budget - (time.monotonic() - start)
        a_l, obj_l = lp_alternate(a, max_iters=30, time_budget=remaining * 0.5)
        if obj_l < best_obj:
            best_obj = obj_l
            best_a = a_l.copy()
            a = a_l.copy()

    # Subgradient refinement
    if time.monotonic() - start < time_budget * 0.9:
        remaining = time_budget - (time.monotonic() - start)
        a_s, obj_s = subgradient_peak(a, max_iters=200, time_budget=remaining * 0.5)
        if obj_s < best_obj:
            best_obj = obj_s
            best_a = a_s.copy()

    # Final chase peak
    if time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        a_f, obj_f = chase_peak(best_a, n_steps=min(50, int(remaining * 3)), step_size=0.03)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f.copy()

    return best_a, best_obj
