import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def solve_maxsum_lp(fixed_seq, M, time_limit=25):

def golden_section_1d(f, lo, hi, tol=1e-10, max_iter=100):
    """Golden section search for 1D minimization on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0

            hi_bound = max(2.0 * old_ai, 4.0 / n)  # reasonable upper bound

"""
Optimized sequence search using fast coordinate descent with exact 1D sub-optimization,
improved subgradient with backtracking line search, LP alternating, and multi-start
diverse initialization. Leverages the convexity of the per-coordinate subproblem
(max of affine + one quadratic) for fast exact coordinate updates. Targets improvement
below 1.5027 by combining multiple complementary optimization strategies.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def compute_conv_and_obj(a):
    """Compute both convolution and objective in one pass."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return compute_conv(a), np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    obj = 2.0 * n * max_b / (s * s)
    return conv, obj


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=25):
    """Solve: min t s.t. (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=25):
    """Solve: max sum(g) s.t. (fixed_seq * g)[k] <= M for all k, g >= 0."""
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section_1d(f, lo, hi, tol=1e-12, max_iter=80):
    """Golden section search for 1D minimization on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def coord_descent(a, max_sweeps=30, time_budget=60.0, momentum=0.0):
    """
    Coordinate descent with exact 1D sub-optimization.
    For each coordinate i, exploits the fact that max_k (a*a)[k] is
    a convex piecewise-affine + quadratic function of a[i].
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    # Precompute indices for D vector
    D_indices = []
    for i in range(n):
        lo = i
        hi = min(i + n, 2 * n - 1)
        D_indices.append((lo, hi))

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            old_ai = a[i]
            a[i] = 0.0

            # Compute conv_rest = (a * a) with a[i] = 0
            fa = rfft(a, fft_size)
            conv_rest = irfft(fa * fa, fft_size)[:2 * n - 1]

            # Build D vector: D[k] = a[k-i] if 0 <= k-i < n
            D = np.zeros(2 * n - 1)
            lo, hi = D_indices[i]
            D[lo:hi] = a[:hi - lo]

            has_quad = (2 * i < 2 * n - 1)
            quad_idx = 2 * i

            # Function to evaluate max(conv_rest + 2*ai*D + ai^2*delta)
            def eval_ai(ai):
                vals = conv_rest + 2.0 * ai * D
                if has_quad:
                    vals[quad_idx] += ai * ai
                return np.max(vals)

            # Upper bound for ai
            hi_bound = max(3.0 * old_ai, 8.0 / n)
            if hi_bound < 1e-10:
                hi_bound = 2.0 / n

            opt_ai = golden_section_1d(eval_ai, 0.0, hi_bound, tol=1e-11, max_iter=50)
            opt_ai = max(0.0, opt_ai)

            if momentum > 0:
                a[i] = (1.0 - momentum) * opt_ai + momentum * old_ai
            else:
                a[i] = opt_ai

        a = normalize_simplex(a)
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def subgradient_accelerated(a, max_iters=2000, time_budget=60.0, n_top=3):
    """
    FFT-accelerated subgradient method with adaptive temperature and
    Nesterov momentum. Chases top-n_top peaks simultaneously.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    v = np.zeros(n)
    mu = 0.8
    step = 0.05
    tau = 0.01

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Use top-n_top peaks with softmax weighting
        sorted_idx = np.argsort(conv)[::-1]
        top_k = sorted_idx[:n_top]
        top_vals = conv[top_k]
        cmax = top_vals[0]
        shifted = np.exp(-(top_vals - cmax) / tau)
        weights_top = shifted / np.sum(shifted)

        # Build sparse weight vector
        weights = np.zeros(2 * n - 1)
        for j, k in enumerate(top_k):
            weights[k] = weights_top[j]

        # Gradient via FFT
        w_ext = np.zeros(fft_size)
        w_ext[:2 * n - 1] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]

        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step and temperature
        step *= 0.999
        tau = max(0.002, tau * 0.998)

    return best_a, best_obj


def lp_alternate(a, max_iters=50, time_budget=100.0):

def parametric_sequence(n, sigma, freq, amp, phase, center_frac=0.5):
    """
    Generate parametric sequence: bell with sinusoidal modulation.
    """
    i = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    bell = np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = bell * mod
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def generate_diverse_starts(n, seed=42):
    """Generate diverse initial sequences for multi-start."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n)
    center = (n - 1) / 2.0

    # 1. Uniform
    starts.append(np.ones(n) / n)

    # 2. Triangular
    tri = 1.0 - np.abs(i - center) / (center + 1e-10)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # 3. Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 8.0) ** 2))
    starts.append(bell / np.sum(bell))

    # 4. Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # 5. Uniform + noise
    noisy = np.abs(np.ones(n) + rng.normal(0, 0.5, n))
    starts.append(noisy / np.sum(noisy))

    # 6-12: Parametric with random parameters
    for _ in range(7):
        sigma = n * (0.1 + 0.4 * rng.random())
        freq = int(1 + rng.integers(0, 10))
        amp = 0.1 + 0.6 * rng.random()
        phase = 2 * np.pi * rng.random()
        center_frac = 0.3 + 0.4 * rng.random()
        starts.append(parametric_sequence(n, sigma, freq, amp, phase, center_frac))

    # 13-15: Mixed structures
    # Two bumps
    b1 = np.exp(-((i - n * 0.3) ** 2) / (2 * (n / 10) ** 2))
    b2 = np.exp(-((i - n * 0.7) ** 2) / (2 * (n / 10) ** 2))
    starts.append((b1 + b2) / np.sum(b1 + b2))

    # Edge concentration
    edge = np.exp(-((i / n) ** 4) * 3) + np.exp(-(((1 - i / n)) ** 4) * 3)
    starts.append(edge / np.sum(edge))

    # Geometric
    geo = 0.99 ** np.abs(i - center)
    starts.append(geo / np.sum(geo))

    # Deduplicate
    unique = [starts[0]]
    for s in starts[1:]:
        if all(np.linalg.norm(s - u) / max(1, np.linalg.norm(u)) > 0.03 for u in unique):
            unique.append(s)

    return unique


def hybrid_polish(a, time_budget=100.0):
    """
    Hybrid polishing: alternate between coordinate descent, subgradient, and LP.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Choose method based on remaining time
        if remaining > 30 and n <= 900:
            # LP step (expensive but powerful)
            a_new, obj_new = lp_alternate(best_a, max_iters=15, time_budget=min(remaining * 0.3, 30.0))
        elif remaining > 10:
            # Coordinate descent (moderate)
            a_new, obj_new = coord_descent(best_a, max_sweeps=5, time_budget=min(remaining * 0.3, 20.0))
        else:
            # Subgradient (fast)
            a_new, obj_new = subgradient_accelerated(best_a, max_iters=int(remaining * 20),
                                                     time_budget=min(remaining * 0.4, 10.0))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5:
                # Perturb and restart
                rng = np.random.default_rng(int(time.monotonic() * 100000) % 100000)
                scale = 0.01 + 0.01 * (stall % 4)

                a_pert = normalize_simplex(a_pert)
                a_pert, obj_pert = coord_descent(a_pert, max_sweeps=3, time_budget=5.0)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                stall = 0

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_res = resample_sequence(best_seq, n_test)
            a_res = normalize_simplex(a_res)
            alloc = min(remaining * 0.2, 30.0)
            a_pol, obj_pol = hybrid_polish(a_res, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** UPDATE to {obj_pol:.8f} **")

    # Phase 6: Final polish
    print(f"\n=== Phase 6: Final polish (best: {best_score:.8f}) ===")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final, obj_final = hybrid_polish(best_seq, time_budget=min(remaining * 0.7, 60.0))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: {best_score:.8f} ** UPDATE **")

        # Batch perturbations
        remaining = deadline - time.monotonic()
        if remaining > 15 and best_seq is not None:
            n_final = len(best_seq)
            rng = np.random.default_rng(seed + 77777)
            for pert in range(8):
                if time.monotonic() > deadline - 5:
                    break
                scale = 0.005 + 0.005 * pert
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
                a_pert = normalize_simplex(a_pert)
                t_alloc = min(remaining * 0.08, 8.0)
                if t_alloc < 2:
                    break
                a_pert, obj_pert = coord_descent(a_pert, max_sweeps=3, time_budget=t_alloc)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {pert} (scale={scale:.3f}): ** {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
