"""
Multi-start optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2
over nonnegative real sequences.  The method combines:
  1. Log-sum-exp / top-k projected gradient descent on the simplex,
     with FFT-based gradients.
  2. Coherent "peak clipping" in the convolution domain and phase-preserved
     square-root recovery.
  3. Occasional linear-programming directions from a linearized max-peak model.
  4. Multi-scale initialization and length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def conv_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    N = next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_obj(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    n = len(a)
    if n == 0 or not np.all(np.isfinite(a)):
        return np.inf
    s = float(np.sum(a))
    if s < 0.01 or not np.isfinite(s):
        return np.inf
    c = conv_fft(a)
    m = float(np.max(c))
    return float(2.0 * n * m / (s * s))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx - (cssv - 1.0)
    pos = np.nonzero(cond > 0.0)[0]
    if pos.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = int(pos[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    res = np.maximum(v - theta, 0.0)
    s = float(np.sum(res))
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if abs(s - 1.0) > 1e-9:
        res = res / s
    return res


def resample_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = len(a)
    if m == 0:
        return np.ones(new_n, dtype=np.float64) / max(1, new_n)
    if m == new_n:
        return project_simplex(a)
    x0 = np.linspace(0.0, 1.0, m)
    x1 = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x1, x0, a)
    return project_simplex(b)

    p = globals().get("height_sequence_1", None)
    if p is None:
        return None
    try:
        arr = np.asarray(p, dtype=np.float64).ravel()
        arr = arr[np.isfinite(arr)]
        if arr.size == 0:
            return None
        arr = np.clip(arr, 0.0, 1000.0)
        return project_simplex(arr)
    except Exception:
        return None


def topk_weights(c, K):
    c = np.asarray(c, dtype=np.float64)
    L = len(c)
    K = max(1, min(int(K), L))
    if K >= L:
        return np.ones(L, dtype=np.float64) / L
    idx = np.argpartition(c, -K)[-K:]
    w = np.zeros(L, dtype=np.float64)
    w[idx] = 1.0 / K
    return w


def softmax_weights(c, tau):
    c = np.asarray(c, dtype=np.float64)
    L = len(c)
    if tau <= 1e-300:
        idx = int(np.argmax(c))
        w = np.zeros(L, dtype=np.float64)
        w[idx] = 1.0
        return w
    cmax = float(np.max(c))
    z = (c - cmax) / tau
    z = np.clip(z, -700.0, 0.0)
    w = np.exp(z)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 1e-300:
        idx = int(np.argmax(c))
        w = np.zeros(L, dtype=np.float64)
        w[idx] = 1.0
    else:
        w /= s
    return w


def weights_to_grad(a, w):
    """
    For weights w over convolution lags, compute
    v_i = sum_j w_{i+j} a_j, which is (1/2) the gradient of w^T conv(a,a).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = len(a)
    N = next_pow2(n + len(w) - 1)
    w_rev = w[::-1]
    F = rfft(w_rev, N) * rfft(a, N)
    conv = irfft(F, N)
    start = len(w) - 1
    return conv[start:start + n]


def pg_lse(a, time_budget, rng, tau_start_frac=0.08, tau_end_frac=0.002):
    """Stabilized projected gradient with adaptive active-set/softmax weights."""
    a = project_simplex(np.asarray(a, dtype=np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    if time_budget <= 0.0:
        return best_a, best_obj

    end = time.monotonic() + float(time_budget)
    n = len(a)
    cur = a.copy()
    cur_obj = best_obj

    step = 1.0
    tau_frac = float(tau_start_frac)
    k_list = (1, 2, 3, 5, 10, 20, 50, 100, 200, 500)
    k_idx = 3
    use_softmax = True
    stall = 0
    no_improve = 0
    mults_big = (1.0, 0.5, 0.2, 2.0)
    mults_normal = (1.0, 0.5, 0.2, 2.0, 0.05, 4.0)

    while time.monotonic() < end - 0.05:
        c = conv_fft(cur)
        s = float(np.sum(cur))
        cmax = float(np.max(c))
        if s < 0.01 or not np.isfinite(cmax):
            cur = best_a.copy()
            cur_obj = best_obj
            c = conv_fft(cur)
            cmax = float(np.max(c))
            s = float(np.sum(cur))
            if s < 0.01 or not np.isfinite(cmax):
                break
        cur_obj = float(2.0 * n * cmax / (s * s))
        if cur_obj < best_obj - 1e-15:
            best_a = cur.copy()
            best_obj = cur_obj
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 350:
            break

        if stall > 2 and rng.random() < 0.55:
            use_softmax = not use_softmax

        if use_softmax:
            tau = max(1e-300, cmax * tau_frac)
            w = softmax_weights(c, tau)
        else:
            K = int(k_list[min(k_idx, len(k_list) - 1)])
            w = topk_weights(c, K)

        g = weights_to_grad(cur, w)
        g -= float(np.mean(g))
        gn = float(np.linalg.norm(g))
        if gn < 1e-18:
            break

        S_cur = float(np.dot(w, c))
        mults = mults_big if n > 1500 else mults_normal
        accepted = False
        chosen = None
        chosen_obj = cur_obj
        chosen_mult = 1.0

        for mult in mults:
            t = step * mult
            cand = project_simplex(cur - t * g)
            cc = conv_fft(cand)
            sc = float(np.sum(cand))
            if sc < 0.01 or not np.isfinite(sc):
                continue
            oc = float(2.0 * n * float(np.max(cc)) / (sc * sc))
            Sc = float(np.dot(w, cc))
            if (Sc < S_cur - 1e-15) or (oc < cur_obj - 1e-14):
                if best_obj < np.inf and oc > best_obj * 1.18:
                    continue
                accepted = True
                chosen = cand
                chosen_obj = oc
                chosen_mult = mult
                break

        if accepted:
            cur = chosen
            cur_obj = chosen_obj
            if cur_obj < best_obj - 1e-15:
                best_a = cur.copy()
                best_obj = cur_obj
                no_improve = 0

            step *= max(0.6, min(1.6, float(chosen_mult)))
            if step > 50.0:
                step = 50.0
            if step < 1e-7:
                step = 1.0
            tau_frac = max(tau_end_frac, tau_frac * 0.992)
            stall = 0
        else:
            step *= 0.5
            stall += 1
            if step < 1e-5:
                sigma = 0.05 / n
                cur = project_simplex(cur + rng.normal(0.0, sigma, n))
                cur_obj = compute_obj(cur)
                if cur_obj < best_obj - 1e-15:
                    best_a = cur.copy()
                    best_obj = cur_obj
                    no_improve = 0
                step = 1.0
                stall = 0
            if stall > 4:
                if use_softmax:
                    tau_frac = min(0.5, tau_frac * 1.8)
                else:
                    k_idx = min(len(k_list) - 1, k_idx + 1)
                stall = 0

        if best_obj < np.inf and cur_obj > best_obj * 1.06 and rng.random() < 0.25:
            cur = best_a.copy()
            cur_obj = best_obj
            step = 1.0
            stall = 0

    return best_a, best_obj


def project_c_upper(c, M):
    """
    Project a nonnegative vector of sum 1 onto {0 <= y <= M, sum(y)=1}.
    This is the correct box/sum projection when lower bound 0 is not active.
    """
    c = np.maximum(np.asarray(c, dtype=np.float64), 0.0)
    s = float(np.sum(c))
    if s <= 1e-300:
        return c
    c = c / s
    L = len(c)
    avg = 1.0 / L
    M = max(float(M), avg * 1.000001)

    y = np.minimum(c, M)
    S = float(np.sum(y))
    if S >= 1.0 - 1e-14:
        y = y / max(1e-300, S)
        return y

    free = y < M - 1e-14
    count = int(np.sum(free))
    while S < 1.0 - 1e-13 and count > 0:
        need = (1.0 - S) / count
        max_delta = float(np.min(M - y[free]))
        if max_delta <= need + 1e-14:
            y[free] += max_delta
            S += max_delta * count
            free = y < M - 1e-14
            count = int(np.sum(free))
        else:
            y[free] += need
            S += need * count
            break

    if S > 1e-300:
        y = y / S
    return y


def coherent_shrink(a, time_budget, rng, shrink_start=0.98):
    """
    Clip the convolution peak, project the desired convolution to the feasible
    set, then recover a by phase-preserved square root (plus random sqrt samples).
    """
    a = project_simplex(np.asarray(a, dtype=np.float64).copy())
    n = len(a)
    L = 2 * n - 1
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    if time_budget <= 0.0:
        return best_a, best_obj

    end = time.monotonic() + float(time_budget)
    N = next_pow2(L)
    avg = 1.0 / L
    cur = a.copy()
    shrink = float(shrink_start)
    stall = 0

    while time.monotonic() < end - 0.2:
        A = rfft(cur, N)
        c = irfft(A * A, N).real[:L]
        c = np.maximum(c, 0.0)
        sc = float(np.sum(cur))
        M = float(np.max(c))
        if sc < 0.01 or not np.isfinite(M) or M <= avg * 1.0005:
            break
        cur_obj = float(2.0 * n * M / (sc * sc))
        if cur_obj < best_obj - 1e-15:
            best_a = cur.copy()
            best_obj = cur_obj

        target = max(avg * 1.000001, M * shrink)
        if target >= M:
            target = max(avg, M * 0.9995)
        ct = project_c_upper(c, target)
        Ct = rfft(ct, N)
        Cold = A * A

        if rng.random() < 0.25:
            An = np.sqrt(Ct.astype(np.complex128))
            m = len(An)
            if m > 1:
                signs = np.where(rng.random(m) < 0.5, -1.0, 1.0)
                signs[0] = 1.0
                if N % 2 == 0:
                    signs[-1] = 1.0
                An = An * signs
            An[0] = np.abs(An[0])
        else:
            An = A.copy()
            scale = float(np.max(np.abs(Cold))) + 1e-300
            mask = np.abs(Cold) > 1e-20 * scale
            ratio = Ct[mask] / Cold[mask]
            An[mask] *= np.sqrt(ratio)
            An[~mask] = np.sqrt(Ct[~mask])
            An[0] = np.abs(An[0])

        if N % 2 == 0:
            An[-1] = np.real(An[-1])

        b = irfft(An, N).real
        cand = project_simplex(b[:n])
        oc = compute_obj(cand)

        if oc < best_obj - 1e-15:
            best_a = cand.copy()
            best_obj = oc
        if oc < cur_obj - 1e-14:
            cur = cand
            stall = 0
            shrink = max(0.90, shrink - 0.005)
        else:
            stall += 1
            shrink = min(0.999, shrink + 0.01)
        if stall > 12:
            break

    return best_a, best_obj


def lp_direction_step(a, rng, K=200, cap_frac=2.0, time_limit=10.0):
    """
    Solve a linearized max-peak LP:
      min m
      s.t. c_k + 2 (A_k a)^T delta <= m   for selected active k
           sum(delta)=0, -a_i <= delta_i <= cap, 0 <= m <= max(c)
    Then line search on a + alpha delta.
    """
    a = project_simplex(np.asarray(a, dtype=np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(a)
    if n > 1600 or time_limit <= 0.5:
        return best_a, best_obj

    c = conv_fft(a)
    M = float(np.max(c))
    L = len(c)
    K = max(1, min(int(K), L))
    if K >= L:
        active = np.arange(L)
    else:
        active = np.argpartition(c, -K)[-K:]

    cap = float(cap_frac) / float(n)
    Arows = np.zeros((len(active), n), dtype=np.float64)
    for ii, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i1 > i0:
            idx = np.arange(i0, i1)
            Arows[ii, i0:i1] = 2.0 * a[k - idx]

    A_ub = np.hstack((Arows, -np.ones((len(active), 1), dtype=np.float64)))
    b_ub = -c[active]
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0], dtype=np.float64)

    lb = np.empty(n + 1, dtype=np.float64)
    ub = np.empty(n + 1, dtype=np.float64)
    lb[:n] = -a
    ub[:n] = cap
    lb[n] = 0.0
    ub[n] = M
    bounds = np.column_stack((lb, ub))

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    time_limit_use = max(1.0, min(30.0, float(time_limit)))

    try:
        res = linprog(
            c_obj, A_ub, b_ub, A_eq, b_eq, bounds,
            method="highs",
            options={"time_limit": time_limit_use, "presolve": True}
        )
        if res.status == 0:
            m = float(res.x[n])
            if m < M * 0.9995:
                delta = np.asarray(res.x[:n], dtype=np.float64)
                for alpha in (1.0, 0.7, 0.3, 0.5, 0.1, 2.0):
                    cand = project_simplex(a + alpha * delta)
                    oc = compute_obj(cand)
                    if oc < best_obj - 1e-15:
                        best_a = cand.copy()
                        best_obj = oc
    except Exception:
        pass

    return best_a, best_obj


def run_combination(a, budget, rng, allow_lp=True):
    """Cycle through PG, coherent shrink, and occasional LP direction."""
    a = project_simplex(np.asarray(a, dtype=np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    if budget <= 0.0:
        return best_a, best_obj

    end = time.monotonic() + float(budget)
    cur = a.copy()
    cur_obj = best_obj
    fails = 0

    while time.monotonic() < end - 1.0:
        rem = end - time.monotonic()
        if rem < 3.0:
            break
        n = len(cur)
        r = rng.random()

        if allow_lp and n <= 1400 and rem > 70.0 and r < 0.10:
            alloc = max(2.0, min(30.0, rem * 0.35))
            seq2, obj2 = lp_direction_step(cur, rng, K=220, cap_frac=2.0, time_limit=alloc)
            cur, cur_obj = seq2, obj2
            if obj2 < best_obj - 1e-15:
                best_a, best_obj = seq2, obj2
                fails = 0
            else:
                fails += 1
        elif rem > 25.0:
            alloc = max(2.0, min(120.0, rem * 0.75))
            tau_st = 0.02 + 0.14 * rng.random()
            seq2, obj2 = pg_lse(cur, alloc, rng, tau_start_frac=tau_st)
            cur, cur_obj = seq2, obj2
            if obj2 < best_obj - 1e-15:
                best_a, best_obj = seq2, obj2
                fails = 0
            else:
                fails += 1
        else:
            if rem > 8.0 and rng.random() < 0.5:
                seq2, obj2 = coherent_shrink(cur, max(1.0, rem - 2.0), rng)
            else:
                seq2, obj2 = pg_lse(cur, max(1.0, rem - 2.0), rng, tau_start_frac=0.05)
            cur, cur_obj = seq2, obj2
            if obj2 < best_obj - 1e-15:
                best_a, best_obj = seq2, obj2
                fails = 0
            else:
                fails += 1

        if fails > 8:
            sigma = 0.05 / max(1, len(best_a))
            cur = project_simplex(best_a + rng.normal(0.0, sigma, len(best_a)))
            cur_obj = compute_obj(cur)
            fails = 0

    return best_a, best_obj


def make_shapes(n, rng, parent, count=10):
    """Construct a diverse set of initial probability vectors of length n."""
    shapes = []
    n = int(max(2, n))

    if parent is not None:
        shapes.append(resample_sequence(parent, n))
        p = resample_sequence(parent, n)
        p = project_simplex(p * np.exp(rng.normal(0.0, 0.15, n)))
        shapes.append(p)

    x = np.linspace(0.0, 1.0, n)

    shapes.append(project_simplex(np.ones(n, dtype=np.float64)))

    shapes.append(project_simplex(np.sin(0.5 * np.pi * x)))
    shapes.append(project_simplex(np.maximum(np.sin(0.5 * np.pi * x), 0.0) ** 0.5))
    shapes.append(project_simplex(np.maximum(np.sin(0.5 * np.pi * x), 0.0) ** 2.0))

    lam = 0.5
    shapes.append(project_simplex(np.exp(-lam * x)))
    lam = 1.0
    shapes.append(project_simplex(np.exp(-lam * x)))
    lam = 2.0
    shapes.append(project_simplex(np.exp(-lam * x)))

    shapes.append(project_simplex((1.0 - x) ** 0.5))
    shapes.append(project_simplex((1.0 - x) ** 2.0))
    shapes.append(project_simplex(x ** 0.5))
    shapes.append(project_simplex(x ** 2.0))

    # Smooth random positive shape.
    for amp in (0.5, 1.0):
        y = rng.normal(size=n)
        N = next_pow2(n)
        F = rfft(y, N)
        cutoff = max(1, int(0.08 * N))
        F[cutoff:] = 0.0
        y = irfft(F, N).real[:n]
        sd = float(np.std(y))
        if sd > 1e-15:
            y = y / sd
        f = np.maximum(1.0 + amp * y, 1e-8)
        shapes.append(project_simplex(f))

    # Low-frequency sine series.
    for _ in range(2):
        f = np.ones(n, dtype=np.float64)
        for h in range(1, 7):
            coef = rng.normal(0.0, 1.0 / (h + 1.0))
            if rng.random() < 0.5:
                f += coef * np.sin(2.0 * np.pi * h * x)
            else:
                f += coef * np.cos(2.0 * np.pi * h * x)
        shapes.append(project_simplex(np.maximum(f, 1e-8)))

    # Two Gaussian bumps.
    for c1, c2, w in ((0.30, 0.70, 0.05), (0.20, 0.80, 0.10)):
        f = np.exp(-((x - c1) ** 2) / (2.0 * w * w)) + np.exp(-((x - c2) ** 2) / (2.0 * w * w))
        shapes.append(project_simplex(f))

    # Logistic ramps.
    for beta, center in ((10.0, 0.30), (10.0, 0.70)):
        f = 1.0 / (1.0 + np.exp(-beta * (x - center)))
        shapes.append(project_simplex(f))

    # Periodic alternating envelopes.
    par = (np.arange(n) % 2).astype(np.float64) * 2.0 - 1.0
    for cval in (0.25, 0.45):
        f = (1.0 + cval * par) * np.maximum(np.sin(0.5 * np.pi * x), 1e-8)
        shapes.append(project_simplex(f))

    out = []
    for sh in shapes:
        out.append(project_simplex(np.asarray(sh, dtype=np.float64).ravel()))
    if len(out) > max(1, int(count)):
        out = out[:max(1, int(count))]
    return out


def pool_add(pool, seq, obj):
    if seq is not None and np.isfinite(obj):
        pool.append((float(obj), np.asarray(seq, dtype=np.float64).copy()))


def pool_key(obj_seq):
    return obj_seq[0]


def propose_candidate(seed=42, budget_s=1500):
    """
    Search routine to be invoked. Returns a list of nonnegative heights.
    """
    t0 = time.monotonic()
    deadline = t0 + max(5.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)

    parent = try_load_parent()
    pool = []

    best_seq = None
    best_obj = np.inf

    if parent is not None:
        pobj = compute_obj(parent)
        if np.isfinite(pobj):
            pool_add(pool, parent, pobj)
            best_seq = parent.copy()
            best_obj = pobj
            print(f"Parent loaded: n={parent.size}, obj={best_obj:.8f}")

    if best_seq is None:
        best_seq = np.ones(512, dtype=np.float64) / 512.0
        best_obj = compute_obj(best_seq)
        pool_add(pool, best_seq, best_obj)
        print(f"Random start: n={best_seq.size}, obj={best_obj:.8f}")

    # ---------------- Phase 1: coarse multi-resolution screening ----------------
    coarse_budget = min(330.0, max(20.0, (deadline - time.monotonic()) * 0.22))
    coarse_start = time.monotonic()
    coarse_end = coarse_start + coarse_budget
    coarse_best = {}

    for n_coarse in (256, 512):
        if time.monotonic() > coarse_end - 5.0:
            break
        shapes = make_shapes(n_coarse, rng, parent, count=9)
        per = max(12.0, (coarse_end - time.monotonic()) / max(1, len(shapes)))
        for sh in shapes:
            if time.monotonic() > coarse_end - 3.0:
                break
            alloc = min(36.0, per * 1.25, deadline - time.monotonic() - 150.0)
            if alloc < 4.0:
                continue
            seq, obj = run_combination(sh, alloc, rng, allow_lp=False)
            pool_add(pool, seq, obj)
            old = coarse_best.get(n_coarse)
            if old is None or obj < old[0]:
                coarse_best[n_coarse] = (obj, seq)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_seq = seq.copy()
                print(f"Coarse n={n_coarse} improved to {best_obj:.8f}")

    print(f"After coarse: best={best_obj:.8f}, n={len(best_seq) if best_seq is not None else 0}, t={time.monotonic()-t0:.1f}s")

    # ---------------- Phase 2: mid-length intensive starts ----------------
    mid_budget = min(650.0, max(30.0, (deadline - time.monotonic()) * 0.35))
    mid_start = time.monotonic()
    mid_end = mid_start + mid_budget

    best_n = int(len(best_seq)) if best_seq is not None else 700
    raw_mid_ns = [
        best_n,
        max(350, best_n - 100),
        min(2500, best_n + 200),
        min(3000, best_n + 500)
    ]
    mid_ns = []
    for nn in raw_mid_ns:
        nn = int(nn)
        if 250 <= nn <= 3200 and nn not in mid_ns:
            mid_ns.append(nn)

    # Keep a few good sequences for resampling, even if shorter ones are not yet globally best.
    pool.sort(key=pool_key)
    source_seqs = []
    if best_seq is not None:
        source_seqs.append(best_seq)
    for obj_s, seq_s in pool:
        if len(source_seqs) >= 5:
            break
        if best_seq is None or len(seq_s) != len(best_seq) or obj_s < best_obj - 0.01:
            source_seqs.append(seq_s)
    for _, seq_c in coarse_best.values():
        if len(source_seqs) < 7:
            source_seqs.append(seq_c)

    for n_mid in mid_ns:
        if time.monotonic() > mid_end - 10.0:
            break
        inits = []
        for src in source_seqs[:3]:
            inits.append(resample_sequence(src, n_mid))

        analytic = make_shapes(n_mid, rng, parent, count=7)
        for idx in (2, 3, len(analytic) - 1):
            if 0 <= idx < len(analytic):
                inits.append(analytic[idx])

        inits = inits[:6]
        per = max(20.0, (mid_end - time.monotonic()) / max(1, len(inits)))
        for sh in inits:
            if time.monotonic() > mid_end - 5.0:
                break
            alloc = min(95.0, per * 1.35, deadline - time.monotonic() - 120.0)
            if alloc < 5.0:
                continue
            seq, obj = run_combination(sh, alloc, rng, allow_lp=(n_mid <= 1500))
            pool_add(pool, seq, obj)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_seq = seq.copy()
                print(f"Mid n={n_mid} improved to {best_obj:.8f}")

    print(f"After mid: best={best_obj:.8f}, n={len(best_seq) if best_seq is not None else 0}, t={time.monotonic()-t0:.1f}s")

    # ---------------- Phase 3: fine multi-length refinement ----------------
    fine_budget = min(700.0, max(40.0, (deadline - time.monotonic()) * 0.45))
    fine_start = time.monotonic()
    fine_end = fine_start + fine_budget

    pool.sort(key=pool_key)
    top_pool = []
    for obj_t, seq_t in pool:
        if len(top_pool) >= 7:
            break
        top_pool.append((obj_t, seq_t))

    fine_tasks = []
    for _, seq_t in top_pool[:5]:
        fine_tasks.append(seq_t)
        n_t = int(len(seq_t))
        if n_t < 2600:
            nn = int(min(3200, n_t + max(60, 0.20 * n_t)))
            fine_tasks.append(resample_sequence(seq_t, nn))
        else:
            nn = int(max(400, n_t - 120))
            fine_tasks.append(resample_sequence(seq_t, nn))

    fine_tasks = fine_tasks[:11]
    if len(fine_tasks) == 0 and best_seq is not None:
        fine_tasks.append(best_seq)

    per = max(35.0, (fine_end - time.monotonic()) / max(1, len(fine_tasks)))
    for sh in fine_tasks:
        if time.monotonic() > fine_end - 5.0:
            break
        alloc = min(190.0, per * 1.45, deadline - time.monotonic() - 70.0)
        if alloc < 8.0:
            continue
        seq, obj = run_combination(sh, alloc, rng, allow_lp=(len(sh) <= 1500))
        pool_add(pool, seq, obj)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_seq = seq.copy()
            print(f"Fine improved to {best_obj:.8f}, n={len(seq)}")

    print(f"After fine: best={best_obj:.8f}, n={len(best_seq) if best_seq is not None else 0}, t={time.monotonic()-t0:.1f}s")

    # ---------------- Phase 4: final polish ----------------
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        final_budget = deadline - time.monotonic() - 5.0
        if final_budget > 40.0:
            first = max(5.0, min(final_budget * 0.62, 350.0))
            seq, obj = run_combination(best_seq, first, rng, allow_lp=(len(best_seq) <= 1500))
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_seq = seq.copy()
                print(f"Final polish improved to {best_obj:.8f}")

            rem = deadline - time.monotonic() - 5.0
            if rem > 20.0:
                n_f = len(best_seq)
                pert = project_simplex(best_seq * np.exp(rng.normal(0.0, 0.04, n_f)))
                seq2, obj2 = run_combination(pert, rem, rng, allow_lp=(n_f <= 1500))
                if obj2 < best_obj - 1e-15:
                    best_obj = obj2
                    best_seq = seq2.copy()
                    print(f"Final perturbed improved to {best_obj:.8f}")

    if best_seq is None:
        best_seq = np.ones(512, dtype=np.float64) / 512.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_obj:
            best_obj = val
    except Exception:
        val = best_obj

    if not np.isfinite(val):
        result = [1.0] * 100
        try:
            val = evaluate_sequence(result)
        except Exception:
            val = 2.0

    print(f"FINAL: n={len(result)}, objective={val:.8f}, elapsed={time.monotonic()-t0:.1f}s")
    return result
