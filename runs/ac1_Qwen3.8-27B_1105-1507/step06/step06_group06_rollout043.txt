Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) adaptive spectral flattening, (2) subgradient with momentum on active set,
(3) LP alternating with warm starts, (4) multiplicative update, (5) multi-length search.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum(a, target_sum=1.0):
    """Project onto {a >= 0, sum(a) = target_sum}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] / new_n if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_nonneg_sum(a_new)


def spectral_flatten(a, max_iters=500, time_budget=30.0, beta_init=0.95, beta_min=0.5):
    """
    Iterative spectral flattening: modify |FFT(a)| to be flatter, 
    keeping phase, then IFFT and project to non-negative.
    """
    n = len(a)
    a = project_nonneg_sum(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    beta = beta_init
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        amp = np.abs(fa)
        amp_max = np.max(amp)
        if amp_max < 1e-15:
            break
        
        # Flatten amplitude
        amp_norm = amp / amp_max
        amp_flat = amp_norm ** beta
        amp_new = amp_flat * amp_max
        
        # Keep phase
        phase = np.angle(fa)
        fa_new = amp_new * np.exp(1j * phase)
        
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Adaptively decrease beta (more aggressive flattening over time)
        beta = max(beta_min, beta * 0.999)
        a = a_new
    
    return best_a, best_obj

def chase_peak_fast(a, k_star, step_size):
    """Take a step to reduce conv[k_star]."""
    n = len(a)
    # Gradient of conv[k_star] w.r.t. a[j] = 2*a[k_star-j] (if 0 <= k_star-j < n)
    grad = np.zeros(n)

        if 0 <= idx < n:

    grad -= np.mean(grad)  # center to maintain sum
    a_new = a - step_size * grad
    return project_nonneg_sum(a_new)

def multiplicative_update(a, max_iters=200, time_budget=20.0):
    """
    Multiplicative update: a[i] *= correction_factor[i]
    where correction_factor is based on how much each element contributes to peaks.
    """
    n = len(a)
    a = project_nonneg_sum(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Compute how much each a[i] contributes to the peak
        # Contribution of a[i] to conv[k] is a[i]*a[k-i]
        # We want to reduce a[i] if it contributes a lot to peaks
        # and increase it if it contributes to valleys
        
        # For each i, compute: sum_k (conv[k] - target) * a[k-i] / (a[i] + eps)
        # This gives the "excess" that a[i] is responsible for
        # target = mean(conv) (to flatten)
        target = np.mean(conv)
        excess = conv - target  # positive where we want to reduce
        
        # Correction for a[i]: proportional to sum_k excess[k] * a[k-i]
        # = correlation of excess and a
        N_fft = 1
        while N_fft < n + len(conv) - 1:
            N_fft <<= 1
        fe = np.fft.fft(excess, N_fft)
        fa_fft = np.fft.fft(a, N_fft)
        corr = np.fft.ifft(np.conj(fa_fft) * fe).real[:n]
        
        # Multiplicative update: a[i] *= exp(-alpha * corr[i] / (a[i] + eps))
        alpha = 0.1 * (1 + it / max_iters)
        corr_safe = corr / (np.abs(corr).max() + 1e-15)
        correction = np.exp(-alpha * corr_safe)
        a_new = a * correction
        a_new = project_nonneg_sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FFT-based multi-peak subgradient with softmax weights,
(2) adaptive spectral flattening with multiple compression strategies,
(3) LP alternating for polishing, (4) multiplicative updates,
(5) multi-length exploration and diverse initializations.
Key improvements: correct gradient computation via FFT, adaptive step sizes,
better time allocation, and more aggressive spectral methods.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum(a, target_sum=1.0):
    """Project onto {a >= 0, sum(a) = target_sum}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] / new_n if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_nonneg_sum(a_new)


def weighted_multipeak_gradient(a, weights):
    """
    Compute grad[j] = 2 * sum_i w[i+j] * a[i] using FFT.
    weights has length 2n-1.
    """
    n = len(a)
    N_fft = 1
    while N_fft < n + len(weights) - 1:
        N_fft <<= 1
    FA = np.fft.fft(a, N_fft)
    FW = np.fft.fft(weights, N_fft)
    grad = 2.0 * np.fft.ifft(np.conj(FA) * FW).real[:n]
    return grad


def multipeak_chase(a, max_iters=2000, time_budget=60.0, tau_frac=0.03):
    """
    Multi-peak subgradient method with softmax weighting.
    Fast: O(n log n) per iteration via FFT gradient.
    """
    n = len(a)
    a = project_nonneg_sum(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        if cmax < 1e-15:
            break

        # Softmax weights over lags
        tau = cmax * tau_frac
        if tau < 1e-15:
            tau = 1e-15
        shifted = np.clip((conv - cmax) / tau, -100.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute weighted gradient via FFT
        grad = weighted_multipeak_gradient(a, weights)
        grad -= np.mean(grad)

        # Adaptive step size
        eta = 0.01 * (1.0 + 2.0 * n / max(n, it + 1))
        a_try = a - eta * grad
        a_try = project_nonneg_sum(a_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try half step
            a_try2 = a - 0.5 * eta * grad
            a_try2 = project_nonneg_sum(a_try2)
            obj_try2 = compute_obj(a_try2)
            if obj_try2 < best_obj:
                best_obj = obj_try2
                best_a = a_try2.copy()
                a = a_try2

    return best_a, best_obj


def chase_single_peak(a, max_iters=500, time_budget=30.0, eta_init=0.02):
    """Fast single-peak chase: O(n) per step."""
    n = len(a)
    a = project_nonneg_sum(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = eta_init
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # O(n) gradient for single peak
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        grad -= np.mean(grad)

        a_try = a - eta * grad
        a_try = project_nonneg_sum(a_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            eta *= 0.95
            a = best_a.copy()

        eta = max(eta * 0.999, 0.001)

    return best_a, best_obj


def spectral_flatten(a, max_iters=500, time_budget=30.0, strategy='power'):
    """
    Iterative spectral flattening with multiple strategies.
    strategy: 'power', 'blend', 'rational', 'log'
    """
    n = len(a)
    a = project_nonneg_sum(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N_fft)
        amp = np.abs(fa)
        amp_max = np.max(amp)
        if amp_max < 1e-15:
            break

        if strategy == 'power':
            # Power compression in amplitude domain
            alpha = 0.7 + 0.25 * (it / max_iters)
            amp_norm = amp / amp_max
            amp_new = (amp_norm ** alpha) * amp_max
        elif strategy == 'blend':
            # Blend toward flat spectrum
            alpha = 0.3 + 0.3 * (it / max_iters)
            amp_mean = np.mean(amp)
            amp_new = (1 - alpha) * amp + alpha * amp_mean
        elif strategy == 'rational':
            # Rational compression
            gamma = amp_max * 0.1
            amp_new = amp * (amp_max + gamma) / (amp + gamma)
        else:
            # Log compression
            alpha = 0.5 + 0.3 * (it / max_iters)
            amp_log = np.log1p(amp / amp_max)
            amp_new = (amp_log ** alpha) * amp_max / (np.log1p(1.0) ** alpha)

        phase = np.angle(fa)
        fa_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def spectral_convolution_flatten(a, max_iters=300, time_budget=30.0):
    """
    Flatten the convolution directly, then extract new a via spectral factorization.
    """
    n = len(a)
    a = project_nonneg_sum(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 2 * n:
        N_fft <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution via FFT
        fa = np.fft.fft(a, N_fft)
        c = np.fft.ifft(fa * fa).real[:2 * n - 1]
        c_max = c.max()
        if c_max < 1e-15:
            break

        # Flatten the convolution
        alpha = 0.6 + 0.35 * (it / max_iters)
        c_norm = c / c_max
        c_flat = c_norm ** alpha * c_max
        c_flat = np.maximum(c_flat, 0)

        # Spectral factorization: find a_new such that FFT(a_new)^2 ~ FFT(c_flat)
        cf = np.fft.fft(c_flat, N_fft)
        # Take square root (complex)
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def multiplicative_update(a, max_iters=300, time_budget=20.0):
    """
    Multiplicative update: adjust a[i] proportionally to its contribution to peaks.
    """
    n = len(a)
    a = project_nonneg_sum(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < n + 2 * n - 1:
        N_fft <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        target = np.mean(conv)
        excess = conv - target

        # Compute contribution of each a[i] to the excess
        # contrib[i] = sum_k excess[k] * a[k-i] (for valid k-i)
        # = 2 * (a ⋆ excess)[i]... actually let's use the same formula
        N2 = 1
        while N2 < n + len(excess) - 1:
            N2 <<= 1
        FA = np.fft.fft(a, N2)
        FE = np.fft.fft(excess, N2)
        contrib = np.fft.ifft(np.conj(FA) * FE).real[:n]

        # Multiplicative correction
        alpha = 0.1 + 0.2 * (it / max_iters)
        max_contrib = np.abs(contrib).max()
        if max_contrib < 1e-15:
            break
        correction = np.exp(-alpha * contrib / max_contrib)
        a_new = a * correction
        a_new = project_nonneg_sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def build_toeplitz_conv(a):
    """Build matrix A where (A @ g)[k] = (conv(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_alternate(a, max_iters=30, time_budget=30.0):
    """LP alternating: fix a, solve for g maximizing sum(g) s.t. conv(a,g) <= M."""
    n = len(a)
    if n > 1500:
        return a, compute_obj(a)

    a = project_nonneg_sum(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    A = build_toeplitz_conv(a)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s < 1e-10:
                break
            g = g / s
        except Exception:
            break

        # Line search
        best_t_obj = best_obj
        best_t_a = best_a
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a
        else:
            break

    return best_a, best_obj


def diversified_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sinusoidal
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Gaussian
    x = np.linspace(-4, 4, n)
    gauss = np.exp(-x ** 2 / 3)
    inits.append(gauss / gauss.sum())

    # Triangular
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(float)
    inits.append(tri / tri.sum())

    # Bimodal
    x = np.linspace(0, 1, n)
    bimodal = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01)
    inits.append(bimodal / bimodal.sum())

    # Random smooth
    freqs = rng.integers(1, 5, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.5, 1.0, size=5)
    sig = np.zeros(n)
    for f, p, a_val in zip(freqs, phases, amps):
        sig += a_val * np.sin(2 * np.pi * f * x + p)
    sig = np.abs(sig) + 0.1
    inits.append(sig / sig.sum())

    # Exponential decay
    exp_dec = np.exp(-np.arange(n) / (n * 0.3))
    inits.append(exp_dec / exp_dec.sum())

    # Cosine raised
    cos_r = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    cos_r = cos_r ** 2 + 0.05
    inits.append(cos_r / cos_r.sum())

    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum(pert))

    return inits


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase approach:
    Phase 1: Diverse initialization + quick screening (10%)
    Phase 2: Intensive optimization with alternating methods (50%)
    Phase 3: Spectral methods + LP polishing (20%)
    Phase 4: Multi-length exploration (10%)
    Phase 5: Final polish (10%)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # === Phase 1: Diverse initialization screening (10%) ===
    print("\n=== Phase 1: Initialization & Screening ===")
    phase1_budget = min(100.0, (deadline - start_time) * 0.10)
    phase1_end = start_time + phase1_budget

    n_options = [n_parent, n_parent + 100, n_parent - 100, n_parent + 200,
                 n_parent + 300, n_parent - 50, n_parent + 50]
    n_options = list(dict.fromkeys([n for n in n_options if 200 <= n <= 2500]))

    candidates = []
    per_n_budget = max(3.0, phase1_budget / max(1, len(n_options)))

    for n_try in n_options:
        if time.monotonic() > phase1_end:
            break
        inits = diversified_initializations(n_try, rng, parent)
        per_init_budget = max(1.0, per_n_budget / max(1, len(inits)))

        for init in inits:
            if time.monotonic() > phase1_end:
                break
            a_q = project_nonneg_sum(init)
            # Quick optimization: single peak chase
            a_q, obj_q = chase_single_peak(a_q, max_iters=50,
                                           time_budget=min(per_init_budget, 5.0),
                                           eta_init=0.02)
            if obj_q < np.inf:
                candidates.append((obj_q, a_q, n_try))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates")
    if candidates:
        print(f"  Best from screening: obj={candidates[0][0]:.8f}, n={candidates[0][2]}")

    for obj_c, a_c, n_c in candidates[:5]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  Updated best: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive optimization (50%) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"Best before: {best_score:.8f}")
    phase2_budget = min(700.0, (deadline - time.monotonic()) * 0.50)
    phase2_end = time.monotonic() + phase2_budget

    top_candidates = [(best_score, best_seq.copy(), len(best_seq))]
    for obj_c, a_c, n_c in candidates[:4]:
        if obj_c < best_score + 0.05:
            top_candidates.append((obj_c, a_c.copy(), n_c))

    for ci, (obj_init, a_init, n_init) in enumerate(top_candidates[:3]):
        if time.monotonic() > phase2_end:
            break
        alloc = min(200.0, (phase2_end - time.monotonic()) / max(1, 3 - ci))
        if alloc < 15:
            break

        a = project_nonneg_sum(a_init)
        local_best = a.copy()
        local_obj = compute_obj(a)
        print(f"  Candidate {ci} (n={n_init}): initial obj={local_obj:.8f}, budget={alloc:.0f}s")

        # Cycle through methods
        methods = ['multipeak', 'spectral_power', 'single_peak', 'spectral_blend',
                   'multipeak', 'conv_flatten', 'single_peak', 'multipeak',
                   'multiplicative', 'spectral_rational']

        for mi, method_name in enumerate(methods):
            if time.monotonic() > phase2_end:
                break
            remaining = phase2_end - time.monotonic()
            if remaining < 10:
                break
            method_budget = min(remaining * 0.3, 30.0)
            if method_budget < 5:
                break

            if method_name == 'multipeak':
                a_new, obj_new = multipeak_chase(a, max_iters=1000,
                                                  time_budget=method_budget,
                                                  tau_frac=0.03)
            elif method_name == 'single_peak':
                a_new, obj_new = chase_single_peak(a, max_iters=500,
                                                    time_budget=method_budget,
                                                    eta_init=0.015)
            elif method_name == 'spectral_power':
                a_new, obj_new = spectral_flatten(a, max_iters=200,
                                                   time_budget=method_budget,
                                                   strategy='power')
            elif method_name == 'spectral_blend':
                a_new, obj_new = spectral_flatten(a, max_iters=200,
                                                   time_budget=method_budget,
                                                   strategy='blend')
            elif method_name == 'spectral_rational':
                a_new, obj_new = spectral_flatten(a, max_iters=200,
                                                   time_budget=method_budget,
                                                   strategy='rational')
            elif method_name == 'conv_flatten':
                a_new, obj_new = spectral_convolution_flatten(a, max_iters=100,
                                                               time_budget=method_budget)
            elif method_name == 'multiplicative':
                a_new, obj_new = multiplicative_update(a, max_iters=200,
                                                        time_budget=method_budget)
            else:
                a_new, obj_new = a, compute_obj(a)

            if obj_new < local_obj:
                local_obj = obj_new
                local_best = a_new.copy()
                a = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"    ** C{ci} M{mi} ({method_name}): obj={best_score:.8f} **")

        print(f"  Candidate {ci} final: obj={local_obj:.8f}")

    # === Phase 3: LP polishing + spectral refinement (20%) ===
    print(f"\n=== Phase 3: LP Polishing ===")
    print(f"Best before: {best_score:.8f}")
    phase3_budget = min(300.0, (deadline - time.monotonic()) * 0.20)
    phase3_end = time.monotonic() + phase3_budget

    a_curr = project_nonneg_sum(best_seq.copy())
    n_curr = len(a_curr)

    # LP if size allows
    if n_curr <= 1200:
        a_lp, obj_lp = lp_alternate(a_curr, max_iters=20,
                                     time_budget=min(phase3_budget * 0.4, 60.0))
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_curr = a_lp.copy()
            print(f"  LP improved: {best_score:.8f}")

    # More spectral methods
    remain3 = phase3_end - time.monotonic()
    if remain3 > 10:
        a_sf, obj_sf = spectral_flatten(a_curr, max_iters=300,
                                         time_budget=min(remain3 * 0.4, 40.0),
                                         strategy='power')
        if obj_sf < best_score:
            best_score = obj_sf
            best_seq = a_sf.copy()
            a_curr = a_sf.copy()
            print(f"  Spectral improved: {best_score:.8f}")

    remain3 = phase3_end - time.monotonic()
    if remain3 > 10:
        a_cf, obj_cf = spectral_convolution_flatten(a_curr, max_iters=100,
                                                     time_budget=min(remain3 * 0.4, 30.0))
        if obj_cf < best_score:
            best_score = obj_cf
            best_seq = a_cf.copy()
            print(f"  Conv-flatten improved: {best_score:.8f}")

    # Multi-peak chase (long run)
    remain3 = phase3_end - time.monotonic()
    if remain3 > 10:
        a_mp, obj_mp = multipeak_chase(a_curr, max_iters=5000,
                                        time_budget=min(remain3, 60.0),
                                        tau_frac=0.02)
        if obj_mp < best_score:
            best_score = obj_mp
            best_seq = a_mp.copy()
            print(f"  Multi-peak improved: {best_score:.8f}")

    # === Phase 4: Multi-length exploration (10%) ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    print(f"Best before: {best_score:.8f}")
    remain4 = deadline - time.monotonic()
    if remain4 > 30:
        phase4_budget = min(150.0, remain4 * 0.15)
        phase4_end = time.monotonic() + phase4_budget

        n_curr = len(best_seq)
        n_tries = [n_curr - 50, n_curr + 50, n_curr + 100, n_curr - 100,
                   n_curr + 150, n_curr + 200]
        n_tries = list(dict.fromkeys([n for n in n_tries if 300 <= n <= 2500]))

        per_n_budget = max(5.0, phase4_budget / max(1, len(n_tries)))

        for n_try in n_tries:
            if time.monotonic() > phase4_end:
                break
            a_res = resample_sequence(best_seq, n_try)

            # Quick optimize
            a_opt, obj_opt = chase_single_peak(a_res, max_iters=100,
                                                time_budget=min(per_n_budget * 0.4, 15.0),
                                                eta_init=0.02)
            remain_n = min(per_n_budget * 0.4, 15.0)
            a_opt2, obj_opt2 = multipeak_chase(a_opt, max_iters=500,
                                                time_budget=remain_n, tau_frac=0.03)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Final polish (remaining) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before: {best_score:.8f}")
    remain5 = deadline - time.monotonic()
    if remain5 > 15 and best_seq is not None:
        a_final = project_nonneg_sum(best_seq.copy())
        n_final = len(a_final)

        # Round 1: multi-peak with small tau
        r1_budget = min(remain5 * 0.25, 30.0)
