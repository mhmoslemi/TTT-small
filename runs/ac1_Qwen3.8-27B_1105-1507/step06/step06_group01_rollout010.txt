import numpy as np

def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Compute smooth max of conv(a,a) and gradient using FFT. O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau

    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # Compute via single convolution: g[i] = (a_rev * p)[i + n - 1]
    a_rev = a[::-1]
    size = 1
    while size < n + (2 * n - 1):
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fp = np.fft.rfft(p, size)
    conv_full = np.fft.irfft(fa * fp, size)
    grad = 2.0 * conv_full[n - 1:2 * n - 1]

    return smooth_val, grad


def run_mirror_descent_fft(a_init, max_iters, deadline, tau_start=50, tau_end=500, step_scale=1.0):
    """Mirror descent (multiplicative weights) with FFT-accelerated smoothed gradient."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac

        _, grad = smooth_max_grad_fft(a, tau)
        grad_tan = grad - grad.mean()

        # 1/sqrt(t) step size (optimal for strongly convex mirror descent)
        step = step_scale / np.sqrt(it + 1) * np.sqrt(2.0 * np.log(n) / tau)
        step = min(step, 2.0)  # cap to avoid instability

        # Multiplicative weights update
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        a = a_new

        if (it + 1) % 10 == 0 or it == max_iters - 1:
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def run_subgradient_descent(a_init, max_iters, deadline, step_scale=0.5, top_m=5):
    """Subgradient method using top-m convolution coefficients for averaged subgradient."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        # Use top-m coefficients for averaged subgradient
        m = min(top_m, len(b))
        top_idx = np.argsort(b)[::-1][:m]
        # Weight by proximity to max
        weights = b[top_idx] - b.min() + 1e-10
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    grad[i] += w * a[j]
        grad *= 2.0
        grad_tan = grad - grad.mean()

        step = step_scale / np.sqrt(it + 1)
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new

        if (it + 1) % 20 == 0:
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def build_active_lp_matrix(n, a, active_indices):
    """Build sparse matrix for LP: A[i,j] = a[k_i - j] if 0 <= k_i - j < n else 0."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, M, threshold_frac=0.95, max_active=200):
    """LP best response using threshold-based active set.
    
    Includes all convolution coefficients >= threshold_frac * max.
    Returns best response g (normalized) or None.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    threshold = threshold_frac * b_max

    # Active set: all coefficients above threshold, capped at max_active
    active_mask = b >= threshold
    active_count = active_mask.sum()
    if active_count > max_active:
        # Use top max_active instead
        top_idx = np.argsort(b)[::-1][:max_active]
        active = top_idx.tolist()
    else:
        active = np.where(active_mask)[0].tolist()

    if len(active) < 2:
        return None

    A = build_active_lp_matrix(n, a, active)
    if A.nnz == 0:
        return None

    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def optimal_mix_t(a, g, t_max=1.0):
    """Find optimal t in [0, t_max] minimizing max_k ((1-t)a + t*g * ((1-t)a + t*g))[k].
    
    Uses ternary search on the convex 1D function.
    """
    n = len(a)

    def max_conv_at_t(t):
        a_t = (1.0 - t) * a + t * g
        b_t = fast_conv(a_t)
        return b_t.max()

    # Ternary search
    lo = 0.0
    hi = t_max
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = max_conv_at_t(m1)
        f2 = max_conv_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    # Check boundaries
    f0 = max_conv_at_t(0.0)
    f1 = max_conv_at_t(t_max)
    fopt = max_conv_at_t(t_opt)
    if f0 <= fopt and f0 <= f1:
        return 0.0
    if f1 <= fopt and f1 <= f0:
        return t_max
    return t_opt


def run_lp_polish(a_init, max_iters, deadline, threshold_frac=0.95, max_active=200):
    """LP mirror descent polish with threshold-based active set."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_best_response(a, best_M, threshold_frac, max_active)
        if g is None:
            no_improve += 1
            if no_improve > 3:
                break
            continue

        improved = False
        # Try optimal t via line search
        t_opt = optimal_mix_t(a, g, t_max=1.0)
        if t_opt > 1e-6:
            a_trial = (1.0 - t_opt) * a + t_opt * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 0.01:
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-12:
                    a = a_trial.copy()
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True

        # Also try fixed small steps
        if not improved:
            for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 0.01:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-12:
                    a = a_trial.copy()
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
            # Tighten threshold as we converge
            threshold_frac = min(threshold_frac + 0.005, 0.999)
        else:
            no_improve += 1
            if no_improve > 10:
                break

    return best_a


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with exact 1D optimization per coordinate."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    rng_cd = np.random.default_rng(seed)
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)

        b = fast_conv(a).copy()
        sweep_improved = False

        for i in order:
            x_cur = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Collect affected convolution terms
            c_vals = []
            s_vals = []
            quad_pos = -1

            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c_vals.append(b[k] - x_cur * x_cur)
                    s_vals.append(0.0)
                    quad_pos = len(c_vals) - 1
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_vals.append(b[k] - 2.0 * x_cur * a[j])
                        s_vals.append(2.0 * a[j])
                    else:
                        c_vals.append(b[k])
                        s_vals.append(0.0)

            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0

            c_arr = np.array(c_vals)
            s_arr = np.array(s_vals)

            def obj_at(x_test):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if quad_pos >= 0:
                    qv = c_arr[quad_pos] + x_test * x_test
                    if qv > m:
                        m = qv
                return m

            # Golden section search
            lo = 0.0
            hi = max(2.0 * x_cur + 0.1, 0.5)
            for _ in range(5):
                if obj_at(hi) > obj_at(hi * 0.5) and hi < 5.0:
                    hi *= 2.0
                else:
                    break

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)

            for _ in range(45):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)

            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0

            x_opt = max(0.0, x_opt)

            if abs(x_opt - a[i]) > 1e-12:
                # Update b values
                diff = x_opt - a[i]
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = c_vals[k - k_lo] + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] = c_vals[k - k_lo] + s_vals[k - k_lo] * x_opt
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-14):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 5
        if no_improve > 15:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.15, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine windows
    for freq in [1, 2, 3, 4]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())

    # Gaussian
    gauss = np.exp(-6.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Spectral init: minimize peak of |DFT|^2
    a_spec = np.ones(n) / n
    for _ in range(50):
        bs = fast_conv(a_spec)
        M_s = bs.max()
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.1 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())

    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_optimization_pipeline(a0, n, deadline, seed):
    """Run the full optimization pipeline on a single initialization."""
    best = np.maximum(a0.copy(), 1e-10)
    best = best / best.sum()
    best_M = fast_conv(best).max()
    t_start = time.monotonic()
    remaining = deadline - t_start
    if remaining < 5:
        return best

    # Stage 1: Mirror descent with FFT gradient (40% of time)
    t1 = t_start + 0.4 * remaining
    a_md = run_mirror_descent_fft(best, 10000, t1, tau_start=30, tau_end=300, step_scale=0.8)
    M_md = fast_conv(a_md).max()
    if M_md < best_M:
        best_M = M_md
        best = a_md.copy()

    # Stage 2: Subgradient with top-m (15% of time)
    t2 = t_start + 0.55 * remaining
    a_sg = run_subgradient_descent(best, 5000, t2, step_scale=0.4, top_m=3)
    M_sg = fast_conv(a_sg).max()
    if M_sg < best_M:
        best_M = M_sg
        best = a_sg.copy()

    # Stage 3: LP polish (30% of time)
    t3 = t_start + 0.85 * remaining
    a_lp = run_lp_polish(best, 50, t3, threshold_frac=0.93, max_active=150)
    M_lp = fast_conv(a_lp).max()
    if M_lp < best_M:
        best_M = M_lp
        best = a_lp.copy()

    # Stage 4: Coordinate descent (10% of time)
    max_sw = max(5, int((deadline - time.monotonic()) / max(0.2, n / 80.0)))
    a_cd = run_coordinate_descent(best, max_sw, deadline, seed=seed)
    M_cd = fast_conv(a_cd).max()
    if M_cd < best_M:
        best_M = M_cd
        best = a_cd.copy()

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining FFT mirror descent, subgradient method,
    LP best-response, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"START: Parent n={n_parent}, J={tracker.best_score:.8f}", flush=True)
    print(f"{'='*60}", flush=True)

    # Phase 1: Optimize parent length
    remaining = deadline - time.monotonic() - 100
    inits = generate_inits(n_parent, rng, parent)
    time_per_init = remaining / max(1, len(inits))

    print(f"\nPhase 1: Optimizing n={n_parent} with {len(inits)} inits", flush=True)
    for ii, a0 in enumerate(inits):
        if time.monotonic() > deadline - 100:
            break
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        sub_dl = min(deadline - 100, time.monotonic() + time_per_init)
        a_opt = run_optimization_pipeline(a0, n_parent, sub_dl, seed=seed + ii * 100)
        tracker.update(a_opt)

    tracker.report("After Phase 1")

    # Phase 2: Try nearby lengths
    print(f"\nPhase 2: Length exploration", flush=True)
    nearby_lengths = []
    for delta in [-64, 64, -128, 128, 256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in nearby_lengths:
            nearby_lengths.append(nl)

    for nl in nearby_lengths[:3]:
        if time.monotonic() > deadline - 50:
            break
        a_ext = resample_seq(tracker.best, nl)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()
        sub_dl = min(deadline - 50, time.monotonic() + 60)
        a_opt = run_optimization_pipeline(a_ext, nl, sub_dl, seed=seed + 500)
        tracker.update(a_opt)

    tracker.report("After Phase 2")

    # Phase 3: Intensive LP polish on best
    print(f"\nPhase 3: Intensive LP polish", flush=True)
    if time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        t_lp1 = min(deadline - 20, time.monotonic() + 80)
        a_lp = run_lp_polish(tracker.best, 30, t_lp1, threshold_frac=0.97, max_active=300)
        tracker.update(a_lp)

        if time.monotonic() < deadline - 15:
            t_lp2 = min(deadline - 10, time.monotonic() + 40)
            a_lp2 = run_lp_polish(a_lp, 20, t_lp2, threshold_frac=0.99, max_active=500)
            tracker.update(a_lp2)

    tracker.report("After Phase 3")

    # Phase 4: Final coordinate descent
    print(f"\nPhase 4: Final coordinate descent", flush=True)
    if time.monotonic() < deadline - 10:
        n_best = tracker.best.size
        max_sw = max(10, int((deadline - 10 - time.monotonic()) / max(0.3, n_best / 60.0)))
        a_cd = run_coordinate_descent(tracker.best, max_sw, deadline - 5, seed=99999)
        tracker.update(a_cd)

    tracker.report("After Phase 4")

    # Phase 5: One more LP pass
    if time.monotonic() < deadline - 5:
        a_final = run_lp_polish(tracker.best, 10, deadline, threshold_frac=0.99, max_active=400)
        tracker.update(a_final)

    tracker.report("FINAL")

    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
