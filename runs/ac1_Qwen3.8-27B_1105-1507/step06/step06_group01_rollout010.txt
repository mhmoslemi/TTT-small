    a = np.maximum(a, 0.0)

    if s < 0.01:

    b = fast_conv(a)

Actually, for the QP, we need to solve:
min t
s.t. a^T Q_k a - t <= 0 for k in K (growing set)
     sum(a) = 1
     a >= 0

Each constraint a^T Q_k a <= t is a quadratic constraint. This is a QCQP (quadratically constrained quadratic program), not a QP. So it's harder.

However, we can use the SOCP reformulation: a^T Q_k a <= t is equivalent to [a; 1]^T [[Q_k, 0.5*e], [0.5*e^T, -1]] [a; 1] <= 0, which is a second-order cone constraint if Q_k is PSD. But Q_k is not necessarily PSD in general... actually for the convolution, Q_k might be PSD.

Let me check: (conv(a,a))[k] = sum_i a[i]*a[k-i]. This can be written as a^T Q_k a where Q_k is a Hankel matrix. For k=0, Q_0 = I (identity), which is PSD. For k>0, Q_k has 1s on the k-th super and sub diagonal. This is not PSD in general (it has negative eigenvalues for k >= 1).

So the constraints are not SOCP-representable in general. We'd need to use a general QCQP solver or a penalty method.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced LP-mirror-descent
approach with adaptive active sets, golden section line search, and multi-phase
optimization. Key improvements over prior: adaptive threshold for LP active set,
proper 1D coordinate descent with correct incremental updates, spectral init,
dual-guided refinement, and aggressive multi-length exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if cond.all():
        theta = (cv[n - 1] - 1.0) / n
    else:
        rho = int(np.argmax(~cond))
        if (~cond).all():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_and_grad(a, tau):
    """Smooth max (log-sum-exp) of conv(a,a) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p[i:i + n], a)
    return smooth_val, grad


def run_fista(a_init, tau_start, tau_end, max_iters, deadline, lr=0.3):
    """FISTA on log-sum-exp smooth approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        _, grad_y = smooth_max_and_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        step = lr / (1.0 + 0.05 * tau * it / 100.0)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1
        if it % 4 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def build_lp_matrix(a, active_indices):
    """Build sparse constraint matrix for LP: (a*g)[k] <= M for k in active."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def solve_lp_best_response(a, M, active_indices, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a, active_indices)
    if A is None or A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-12:
                return g / g_sum
    except Exception:
        pass
    return None


def golden_section_line_search(a, g, best_M, max_iters=30):
    """Golden section search for optimal t in (1-t)*a + t*g."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    # Check if moving in direction g helps at all
    a_test = 0.5 * a + 0.5 * g
    a_test = np.maximum(a_test, 0.0)
    s_test = a_test.sum()
    if s_test < 1e-10:
        return None, best_M
    a_test = a_test / s_test
    M_test = fast_conv(a_test).max()
    if M_test >= best_M:
        # Direction doesn't help; try small step
        a_test2 = 0.1 * a + 0.9 * a  # degenerate
        return None, best_M
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)

    def obj_at(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s_t = a_t.sum()
        if s_t < 1e-10:
            return float('inf')
        a_t = a_t / s_t
        return fast_conv(a_t).max()

    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    best_t = a_pt if fa < fb else b_pt
    best_val = min(fa, fb)
    for _ in range(max_iters):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
        if fa < best_val:
            best_val = fa
            best_t = a_pt
        if fb < best_val:
            best_val = fb
            best_t = b_pt
        if hi - lo < 1e-6:
            break
    if best_val >= best_M:
        return None, best_M
    a_new = (1.0 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return None, best_M
    a_new = a_new / s_new
    return a_new, best_val


def run_lp_mirror_descent_enhanced(a_init, deadline, threshold_start=0.6,
                                    threshold_end=0.95, max_iters=500,
                                    min_active=5, max_active=300):
    """Enhanced LP mirror descent with adaptive active set threshold."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        threshold = threshold_start + (threshold_end - threshold_start) * frac

        b = fast_conv(a)
        M_cur = b.max()
        active = np.where(b >= threshold * M_cur)[0]
        if len(active) < min_active:
            active = np.argsort(b)[::-1][:min_active]
        active = active[:max_active]
        active_list = active.tolist()

        g = solve_lp_best_response(a, M_cur, active_list, time_limit=15)
        if g is None:
            no_improve += 2
            if no_improve > 20:
                break
            continue

        a_new, M_trial = golden_section_line_search(a, g, best_M, max_iters=25)
        if a_new is not None and M_trial < best_M - 1e-14:
            a = a_new
            best_M = M_trial
            best_a = a.copy()
            no_improve = 0
        else:
            # Try direct blend with small step
            for t in [0.3, 0.15, 0.05, 0.02]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_conv(a_trial).max()
                if M_t < best_M - 1e-14:
                    a = a_trial
                    best_M = M_t
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1
                if no_improve > 30:
                    break

    return best_a


def run_subgradient_descent(a_init, deadline, max_iters=20000, step_init=0.5):
    """Subgradient descent with averaging for the max constraint."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    a_avg = a.copy()
    step = step_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        grad_tan = grad - grad.mean()
        a_new = a - step * grad_tan
        a_new = proj_simplex(a_new)
        a_avg = (it * a_avg + a_new) / (it + 1)
        a = a_new
        step = step_init / (1.0 + 0.1 * it)
        if it % 50 == 0:
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_M = M_avg
                best_a = a_avg.copy()
    M_final = fast_conv(a_avg).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a_avg.copy()
    M_direct = fast_conv(a).max()
    if M_direct < best_M:
        best_M = M_direct
        best_a = a.copy()
    return best_a


def run_coordinate_descent_fixed(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with correct incremental b update."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        b = fast_conv(a).copy()
        sweep_improved = False
        for i in order:
            x_cur = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            c_vals = []
            slopes = []
            has_quad = False
            quad_pos = -1
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c = b[k] - x_cur * x_cur
                    c_vals.append(c)
                    slopes.append(0.0)
                    has_quad = True
                    quad_pos = len(c_vals) - 1
                else:
                    j = k - i
                    if 0 <= j < n:
                        c = b[k] - 2.0 * x_cur * a[j]
                        c_vals.append(c)
                        slopes.append(2.0 * a[j])
                    else:
                        c_vals.append(b[k])
                        slopes.append(0.0)
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            c_arr = np.array(c_vals)
            s_arr = np.array(slopes)

            def obj_at(x_test, c_arr=c_arr, s_arr=s_arr, base_max=base_max,
                       has_quad=has_quad, quad_pos=quad_pos):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if has_quad:
                    qv = c_arr[quad_pos] + x_test * x_test
                    if qv > m:
                        m = qv
                return m

            lo = 0.0
            hi = max(2.0 * x_cur, 0.5)
            for _ in range(8):
                if obj_at(hi) > obj_at(hi * 0.5) and hi < 10.0:
                    hi *= 2.0
                else:
                    break
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)
            for _ in range(35):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            x_opt = max(0.0, x_opt)
            if abs(x_opt - x_cur) > 1e-12:
                a[i] = x_opt
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = c_arr[quad_pos] + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] = c_arr[k - k_lo] + slopes[k - k_lo] * x_opt
                        else:
                            b[k] = c_arr[k - k_lo]
                sweep_improved = True
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 3
        if no_improve > 15:
            break
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def spectral_init(n, rng, iterations=50):
    """Initialize using spectral flatness approach."""
    a = np.ones(n) / n
    for _ in range(iterations):
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a[j]
        a = a - 0.05 * (corr - corr.mean())
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.15, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())
    inits.append(np.ones(n) / n)
    # Parabola
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())
    # U-shape
    poly2 = ((2 * t - 1) ** 2) + 0.1
    inits.append(poly2 / poly2.sum())
    # Cosine
    cos_w = 0.5 + 0.3 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    # Spectral
    inits.append(spectral_init(n, rng, iterations=30))
    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_optimization_pipeline(a_init, n, deadline, seed=0):

    best = a_init.copy()
    best_M = fast_conv(best).max()

    # Stage 1: FISTA (30% time)
    t1 = t0 + 0.3 * total_time
    if time.monotonic() < t1:
        a_f = run_fista(best, 30, 200, 2000, t1, lr=0.3)
        M_f = fast_conv(a_f).max()
        if M_f < best_M:
            best_M = M_f
            best = a_f.copy()

    # Stage 2: Subgradient with averaging (20% time)
    t2 = t0 + 0.5 * total_time
    if time.monotonic() < t2:
        a_sg = run_subgradient_descent(best, t2, max_iters=50000, step_init=0.5)
        M_sg = fast_conv(a_sg).max()
        if M_sg < best_M:
            best_M = M_sg
            best = a_sg.copy()

    # Stage 3: LP mirror descent (40% time)
    t3 = t0 + 0.9 * total_time
    if time.monotonic() < t3:
        a_lp = run_lp_mirror_descent_enhanced(best, t3,
                                               threshold_start=0.5,
                                               threshold_end=0.92,
                                               max_iters=300)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()

    # Stage 4: Coordinate descent (10% time)
    if time.monotonic() < deadline:
        max_sw = max(3, int((deadline - time.monotonic()) / max(0.2, n / 80.0)))
        a_cd = run_coordinate_descent_fixed(best, max_sw, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: diverse inits -> FISTA -> subgradient -> LP mirror descent
    -> coordinate descent, across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent.copy())
        print(f"No parent, using uniform n=512", flush=True)

    n_parent = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length FISTA+subgradient exploration", flush=True)
    print(f"{'='*60}", flush=True)

    # Select lengths to try
    lengths = []
    for candidate in [n_parent, 512, 640, 768, 896, 1024, 1280]:
        if 256 <= candidate <= 2048 and candidate not in lengths:
            lengths.append(candidate)
    lengths.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths = lengths[:6]

    reserve_time = 400.0  # reserve for LP polish phase
    available = deadline - time.monotonic() - reserve_time
    available = max(available, 200.0)
    time_per_length = available / len(lengths)

    best_per_length = {}
    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - reserve_time:
            break
        t_start = time.monotonic()
        t_alloc = min(time_per_length, deadline - reserve_time - time.monotonic())
        if t_alloc < 30:
            break
        print(f"\n  n={n}, alloc={t_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None
                               else np.maximum(parent, 0.1))
        time_per_init = t_alloc / max(1, len(inits))
        best_n = None
        best_M_n = float('inf')

        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline - reserve_time:
                break
            sub_dl = min(t_start + t_alloc, deadline - reserve_time)
            if time.monotonic() >= sub_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            # FISTA + subgradient only (skip LP here for speed)
            t_fista = time.monotonic() + max(5, 0.5 * time_per_init)
            a_f = run_fista(a0, 20, 150, 1500, t_fista, lr=0.3)
            t_sg = min(sub_dl, time.monotonic() + max(5, 0.5 * time_per_init))
            a_sg = run_subgradient_descent(a_f, t_sg, max_iters=30000, step_init=0.4)
            M_sg = fast_conv(a_sg).max()
            if M_sg < best_M_n:
                best_M_n = M_sg
                best_n = a_sg.copy()

        if best_n is not None:
            best_per_length[n] = best_n.copy()
            J_val = compute_J(best_n)
            print(f"  n={n}: best M={best_M_n:.10f} J={J_val:.6f}", flush=True)
            tracker.update(best_n)

    tracker.report("After Phase 1")

    # Phase 2: Intensive LP polish on top candidates
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)

    # Pick top 2-3 candidates by M value
    sorted_candidates = sorted(best_per_length.items(), key=lambda x: fast_conv(x[1]).max())
    top_candidates = sorted_candidates[:3]

    lp_deadline = deadline - 100.0
    time_per_lp = max(30, (lp_deadline - time.monotonic()) / max(1, len(top_candidates)))

    for n_cand, a_cand in top_candidates:
        if time.monotonic() > lp_deadline:
            break
        t_lp_end = min(lp_deadline, time.monotonic() + time_per_lp)
        print(f"\n  LP polish n={n_cand}, time={time_per_lp:.0f}s", flush=True)

        # First LP pass with loose threshold
        a_lp1 = run_lp_mirror_descent_enhanced(
            a_cand, min(t_lp_end, time.monotonic() + time_per_lp * 0.6),
            threshold_start=0.4, threshold_end=0.85, max_iters=200)
        tracker.update(a_lp1)

        # Second LP pass with tight threshold
        if time.monotonic() < t_lp_end:
            a_lp2 = run_lp_mirror_descent_enhanced(
                a_lp1, t_lp_end,
                threshold_start=0.8, threshold_end=0.97, max_iters=200)
            tracker.update(a_lp2)

    tracker.report("After Phase 2")

    # Phase 3: High-tau FISTA refinement
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 60:
        t3_end = min(deadline - 40, time.monotonic() + 60)
        for tau in [100, 300, 800]:
            if time.monotonic() > t3_end:
                break
            a_h = run_fista(tracker.best, tau, tau * 2, 3000,
                            min(t3_end, time.monotonic() + 20), lr=0.1)
            tracker.update(a_h)

    tracker.report("After Phase 3")

    # Phase 4: Coordinate descent polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        max_sw = max(5, int((deadline - 15 - time.monotonic()) / max(0.3,
                    tracker.best.size / 60.0)))
        a_cd = run_coordinate_descent_fixed(tracker.best, max_sw, deadline - 10, seed=9999)
        tracker.update(a_cd)

        # Final LP pass
        if time.monotonic() < deadline - 10:
            a_fin = run_lp_mirror_descent_enhanced(
                a_cd, deadline, threshold_start=0.9, threshold_end=0.99,
                max_iters=50, min_active=3, max_active=100)
            tracker.update(a_fin)

    tracker.report("After Phase 4")

    # Phase 5: Try nearby lengths
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_cur = tracker.best.size
        for mult in [1.2, 0.8, 1.5]:
            if time.monotonic() > deadline - 20:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 25)
            a_opt = run_lp_mirror_descent_enhanced(
                a_ext, sub_dl, threshold_start=0.7, threshold_end=0.95, max_iters=50)
            tracker.update(a_opt)

    tracker.report("After Phase 5")

    # Final
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
