def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_grad(a, p):
    """Compute smooth objective and gradient using power-sum approximation of max."""
    n = len(a)
    S = compute_conv(a)
    S = np.maximum(S, 1e-15)
    T = float(np.sum(a))
    if T < 1e-10:
        return np.inf, np.zeros(n)
    
    # M_p = (sum S_k^p)^(1/p), use log for stability
    log_S = np.log(S)
    p_log_S = p * log_S
    max_p_log_S = np.max(p_log_S)
    log_sum = max_p_log_S + np.log(np.sum(np.exp(p_log_S - max_p_log_S)))
    log_M_p = log_sum / p
    M_p = np.exp(log_M_p)
    
    f = M_p / (T * T)
    
    # D[m] = sum_j a[j] * S[j+m]^(p-1)
    S_pm1 = S ** (p - 1)
    a_flip = a[::-1]
    N = 1
    while N < n + len(S_pm1) - 1:
        N <<= 1
    C = irfft(rfft(a_flip, N) * rfft(S_pm1, N), N)[:n + len(S_pm1) - 1]
    D = C[n - 1:n - 1 + n]
    
    M_p_pow = M_p ** (p - 1)
    grad = 2.0 * M_p_pow / (T * T) * D - 2.0 * M_p / (T * T * T)
    
    return f, grad

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, method='cvxpy'):

                except Exception:

            pass

        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(0, None)] * n + [(0, None)]
        res = minimize(lambda x: x[-1], np.zeros(n + 1), method='SLSQP',

    except Exception:

def chase_peak(a, n_steps=100, time_budget=30.0, verbose=False):
    """Chase-the-peak: iteratively reduce the maximum convolution value."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.02
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Gradient of S_{k_star} w.r.t. a[m] = 2*a[k_star - m] (if valid)
        grad = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]
        
        # Project gradient to sum-zero subspace (maintain normalization)
        grad_centered = grad - np.mean(grad)
        
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller step
                a_new2 = a - step_size * 0.3 * grad_centered
                a_new2 = np.maximum(a_new2, 0)
                s_new2 = float(np.sum(a_new2))
                if s_new2 > 1e-10:
                    a_new2 = a_new2 / s_new2
                    obj_new2 = compute_obj(a_new2)
                    if obj_new2 < best_obj - 1e-12:
                        best_obj = obj_new2
                        best_a = a_new2.copy()
                        a = a_new2
                        step_size *= 0.8
                    else:
                        step_size *= 0.7
                else:
                    step_size *= 0.5
        else:
            step_size *= 0.5
    
    return best_a


def coordinate_refine(a, time_budget=30.0, verbose=False):
    """Coordinate descent: move mass from peak-contributing positions."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()

    start = time.monotonic()
    
    for it in range(30):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False
        
        # Find pairs to swap mass
        for i in range(0, n, 2):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 1e-10:
                # Moving mass from i toward j reduces S_{k_star} if a[j] is small
                # dS_{k_star}/(da_i * da_j) involves a[j] and a[i]
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = frac * a[i]
                    if delta > a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
        
        if not improved:
            # Try all coordinates
            for i in range(0, n, 3):
                if time.monotonic() - start > time_budget:
                    break
                # Reduce a[i], increase a[random_other]
                j = (i + n // 2) % n
                if j != i and a[i] > 1e-10:
                    for frac in [0.02, 0.05]:
                        delta = frac * a[i]
                        a[i] -= delta
                        a[j] += delta
                        new_obj = compute_obj(a)
                        if new_obj < best_obj - 1e-12:

                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                            break
            if not improved:
                break
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:

    elif mode == 1:

        return a / np.sum(a)
    elif mode == 2:
        # Cosine window (shifted to break symmetry)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * (x + 0.3))
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif mode == 3:
        # Exponential ramp
        x = np.linspace(0, 1, n)
        a = np.exp(1.5 * x) * (1.0 - x) + 0.1
        return a / np.sum(a)
    elif mode == 4:
        # Bimodal asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-30*(x-0.25)**2) + 0.6*np.exp(-20*(x-0.7)**2) + 0.1

    elif mode == 5:

        a = np.maximum(a, 0.05)
        return a / np.sum(a)
    else:
        # Random smooth
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a / a[-1]
        a = 1.0 + 0.5 * (a - a[::-1])
        a = np.maximum(a, 0.05)
        return a / np.sum(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
