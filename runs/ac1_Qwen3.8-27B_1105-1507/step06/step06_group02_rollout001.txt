import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A@g)_k = sum_j a_{k-j} * g_j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M, prev_x=None):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_conv_matrix(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        opts = {'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                'primal_feasibility_tolerance': 1e-10}
        if prev_x is not None:
            opts['run_time_limit'] = 30.0
        else:
            opts['run_time_limit'] = 30.0
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def smooth_max_grad(a, tau):
    """Compute gradient of log-sum-exp approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, b, b_max


def anderson_mix(history_x, history_Fx, m):
    """Anderson acceleration: find convex combination minimizing residual norm."""
    if m < 2:
        return history_Fx[-1].copy()
    residuals = []
    for i in range(m):
        residuals.append(history_x[i] - history_Fx[i])
    residuals = np.array(residuals)
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m
    alpha = np.clip(alpha, -0.5, 1.5)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0.0)


def resample_seq(src, n_new):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_new:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lp_backtracking(a, deadline, max_iters=500, anderson_window=5):
    """LP mirror descent with backtracking line search and Anderson acceleration."""
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    best_J = 2.0 * n * best_M
    no_imp = 0
    hist_x = []
    hist_Fx = []
    prev_lp_x = None
    t_schedule = [0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            best_J = 2.0 * n * M
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40 and it > 50:
                break

        g = lp_best_response(a, M, prev_x=prev_lp_x)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-12:
            break
        g = g / gs
        prev_lp_x = g.copy()

        # Backtracking line search
        found = False
        for t in t_schedule:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-12:
                continue
            a_new = a_new / s_new
            M_new = float(fast_conv(a_new).max())
            J_new = 2.0 * n * M_new
            if J_new < best_J + 1e-10:
                # Anderson acceleration
                hist_x.append(a.copy())
                hist_Fx.append(a_new.copy())
                if len(hist_x) > anderson_window:
                    hist_x.pop(0)
                    hist_Fx.pop(0)

                if len(hist_x) >= 3:
                    a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
                    sa = a_and.sum()
                    if sa > 1e-12:
                        a_and = a_and / sa
                        M_and = float(fast_conv(a_and).max())
                        if M_and <= M_new * 1.02:
                            a = a_and
                        else:
                            a = a_new
                    else:
                        a = a_new
                else:
                    a = a_new
                found = True
                break
        if not found:
            # Try even smaller step
            t = 0.005
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-12:
                a = a_new / s_new
            break

    return best_a


def run_smooth_refinement(a, deadline, max_iters=2000, tau_start=2.0, tau_end=100.0):
    """Smooth gradient descent with Nesterov momentum for refinement."""
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    momentum = 0.9

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = 0.3 * (1.0 - frac) ** 1.5 + 0.01

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M - 1e-14:
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        a_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-12:
            y = a_new.copy()
        else:
            y = y / sy
        a = a_new

    return best_a


def perturb_sequence(a, rng, strength=0.05):

    n = len(a)
    a_pert = a.copy()
    # Random smooth perturbation
    k = max(3, n // 30)
    noise = rng.normal(0, strength, n)
    kern = np.ones(k) / k

    indices = rng.choice(n, size=n_points, replace=False)
    a_pert[indices] += rng.normal(0, strength * 0.5, n_points)
    return normalize(a_pert)

    # Random smooth
    k = max(3, n // 15)
    kern = np.ones(k) / k
    inits.append(normalize(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01)))

    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(normalize(full))

    return inits


def run_smooth_multi_init(a0, deadline, max_iters=500):
    """Run smooth refinement with early stopping."""
    return run_smooth_refinement(a0, deadline, max_iters=max_iters,
                                  tau_start=1.0, tau_end=150.0)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    best_seq = None
    best_score = float('inf')
    best_n = 0

    def try_update(a):
        nonlocal best_seq, best_score, best_n
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a.copy()
            best_n = a.size
            el = time.monotonic() - t_start
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError, Exception) as e:
        print(f"Parent not available: {e}", flush=True)

    if parent is None:
        parent = np.ones(512)
        try_update(parent)

    # === PHASE 1: Quick multi-length smooth optimization (200s) ===
    p1_deadline = t_start + 200.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length smooth optimization", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [512, 640, 768, 896, 1024, 1152, 1280, 1408]
    best_per_length = {}
    time_per = 180.0 / len(lengths)

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        t0 = time.monotonic()
        inits = make_initializations(n, rng, best_seq if best_seq is not None else parent)
        best_nseq = None
        best_nM = float('inf')

        for idx, a0 in enumerate(inits[:6]):  # Limit inits for speed
            if time.monotonic() > p1_deadline - 10:
                break
            a0 = normalize(a0)
            sub_dl = min(time.monotonic() + time_per * 0.8 / max(1, 6 - idx), p1_deadline - 5)
            a = run_smooth_refinement(a0, sub_dl, max_iters=300,
                                       tau_start=1.0, tau_end=80.0)
            M = float(fast_conv(a).max())
            if M < best_nM:
                best_nM = M
                best_nseq = a.copy()

        if best_nseq is not None:
            J = compute_J(best_nseq)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            try_update(best_nseq)
            best_per_length[n] = best_nseq.copy()

    print(f"Phase 1 done: {time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 2: LP mirror descent on best sequence (main phase) ===
    p2_deadline = t_start + budget_s - 120.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (main optimization)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        # First, smooth refinement to get close to the optimum
        a_work = normalize(best_seq.copy())

        # Quick smooth refinement first
        smooth_dl = min(t_start + 230.0, p2_deadline - 60)
        a_work = run_smooth_refinement(a_work, smooth_dl, max_iters=1000,
                                        tau_start=3.0, tau_end=200.0)
        try_update(a_work)

        # Now LP mirror descent with perturbation cycles
        n_perturb_cycles = 4
        cycle_time = (p2_deadline - time.monotonic()) / n_perturb_cycles
        current = normalize(a_work.copy())

        for cycle in range(n_perturb_cycles):
            if time.monotonic() > p2_deadline:
                break
            cycle_dl = min(time.monotonic() + cycle_time, p2_deadline)
            print(f"\n  LP Cycle {cycle+1}/{n_perturb_cycles} (deadline={cycle_dl - t_start:.0f}s)", flush=True)

            max_it = max(50, int(cycle_time / 0.5))  # Estimate ~0.5s per LP iteration
            a_lp = run_lp_backtracking(current, cycle_dl, max_iters=max_it,
                                        anderson_window=5)
            try_update(a_lp)

            if a_lp is not None:
                current = normalize(a_lp.copy())

            # Perturb for next cycle (except last)
            if cycle < n_perturb_cycles - 1 and time.monotonic() < p2_deadline - 30:
                strength = 0.03 + 0.02 * cycle
                current = perturb_sequence(current, rng, strength=strength)
                # Quick smooth to stabilize after perturbation
                smooth_dl2 = min(time.monotonic() + 20, p2_deadline - 10)
                current = run_smooth_refinement(current, smooth_dl2, max_iters=200,
                                                 tau_start=2.0, tau_end=100.0)
                try_update(current)

    print(f"Phase 2 done: {time.monotonic()-t_start:.0f}s, best J={best_score:.8f}", flush=True)

    # === PHASE 3: Try nearby lengths with LP ===
    p3_deadline = t_start + budget_s - 40.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Nearby length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and time.monotonic() < p3_deadline:
        n_cur = best_seq.size
        for delta in [32, -32, 64, -64, 128, -128, 16, -16]:
            if time.monotonic() > p3_deadline:
                break
            n_try = n_cur + delta
            if n_try < 256 or n_try > 2048:
                continue
            a_try = resample_seq(best_seq, n_try)
            a_try = normalize(a_try)
            # Quick smooth
            a_try = run_smooth_refinement(a_try, min(time.monotonic() + 15, p3_deadline - 5),
                                           max_iters=300, tau_start=3.0, tau_end=100.0)
            try_update(a_try)
            # Brief LP
            if time.monotonic() < p3_deadline - 10:
                a_lp = run_lp_backtracking(a_try, p3_deadline - 5, max_iters=30,
                                            anderson_window=4)
                try_update(a_lp)

    print(f"Phase 3 done: {time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 4: Final intensive LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    final_deadline = t_start + budget_s - 5.0
    if best_seq is not None and time.monotonic() < final_deadline:
        a_final = normalize(best_seq.copy())
        # Very careful LP with small steps
        a_final = run_lp_backtracking(a_final, final_deadline, max_iters=200,
                                       anderson_window=6)
        try_update(a_final)

        # One more smooth pass
        if time.monotonic() < final_deadline - 2:
            a_final = run_smooth_refinement(a_final, final_deadline, max_iters=500,
                                             tau_start=50.0, tau_end=500.0)
            try_update(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = compute_J(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best_seq]
