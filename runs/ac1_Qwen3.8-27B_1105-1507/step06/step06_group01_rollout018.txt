def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def symmetrize(a):

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def coordinate_step(a, i, b, n):
    """Find optimal a[i] >= 0 given all others fixed. Returns new value."""

    if k_lo > 0 or k_hi < 2 * n - 2:

        base_max_unaffected = b[mask].max() if mask.any() else 0.0
    else:
        base_max_unaffected = 0.0

    bases = []
    slopes = []
    has_quad = False
    quad_base = 0.0

        if other == i:

            quad_base = b[k] - x_cur * x_cur
            has_quad = True
        elif 0 <= other < n:

            base = b[k] - 2.0 * x_cur * a[other]
            bases.append(base)
            slopes.append(2.0 * a[other])
        else:
            # Should not happen if k is in valid range
            bases.append(b[k])
            slopes.append(0.0)
    
    if bases:
        base_arr = np.array(bases)
        slope_arr = np.array(slopes)
    else:
        base_arr = np.array([0.0])
        slope_arr = np.array([0.0])

        """Compute max conv value when a[i] = x."""
        # Affected values
        vals = base_arr + slope_arr * x
        m = vals.max()
        if has_quad:
            qv = quad_base + x * x
            if qv > m:
                m = qv
        # Unaffected max
        if m < base_max_unaffected:
            m = base_max_unaffected
        return m
    
    # Golden section search on [0, hi]
    # Find hi: start from current and expand
    hi = max(2.0 * x_cur, 0.5)
    for _ in range(20):
        if obj_at(hi) > obj_at(hi * 0.5):
            hi *= 2.0
        else:
            break
    hi = min(hi, 10.0)  # Cap
    
    lo = 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(50):

            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at(c)

            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
    
    x_opt = (lo + hi) / 2.0
    # Check boundary
    if obj_at(0.0) <= obj_at(x_opt) + 1e-12:
        x_opt = 0.0
    
    return x_opt

                # Just recompute b (fast with FFT for n=768)
                # Actually, for efficiency, update incrementally
                # b[k] += 2*delta*a[k-i] - 2*a[i_old]*delta... 
                # Simpler: just recompute at end of sweep
                
                improved = True
        
        # Recompute b at end of sweep
        b = fast_conv(a)
        a = normalize(a)
        a = symmetrize(a)
        
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not improved:
            no_improve += 2
        if no_improve > 30:
            break
    
    return best_a

def lp_dual_solve(a, M, active_indices):
    """Solve the dual of the LP best response.
    
    Primal: max sum(g) s.t. A g <= M*1, g >= 0
    Dual:   min M * sum(y) s.t. A^T y >= 1, y >= 0
    
    Where A[k, j] = a[k-j] (if 0 <= k-j < n)
    
    We solve the dual and recover the primal from complementary slackness.
    Actually, it's easier to just solve the primal directly with HiGHS.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                    method='highs', options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return r.x / g_sum
    except Exception:
        pass
    return None

def run_lp_mirror(a_init, max_iters, deadline, n_active_start=30):
    """LP mirror descent with adaptive active set."""
    a = symmetrize(np.maximum(a_init.copy(), 0.0))
    a = normalize(a)
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    n_active = n_active_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Active set: indices where b[k] >= (1 - 0.05) * M
        threshold = M * 0.95
        active = np.where(b >= threshold)[0].tolist()
        
        # Also ensure we have at least n_active and at most min(2n-1, 200)
        if len(active) < n_active:
            top_idx = np.argsort(b)[::-1][:n_active]
            active = list(set(active + top_idx.tolist()))
        if len(active) > min(2 * n - 1, 200):
            # Keep only the top ones
            active_sorted = sorted(active, key=lambda k: -b[k])
            active = active_sorted[:min(2 * n - 1, 200)]
        
        g = lp_dual_solve(a, M, active)
        if g is None:
            no_improve += 1
            if no_improve > 5:
                # Try with fewer constraints
                n_active = max(10, n_active // 2)
                active = np.argsort(b)[::-1][:n_active].tolist()
                g = lp_dual_solve(a, M, active)
                if g is None:
                    if no_improve > 10:
                        break
                    continue
        
        g = symmetrize(g)
        g = normalize(g)
        
        # Geometric line search
        improved = False
        for t in [0.3, 0.5, 0.7, 0.2, 0.1, 0.4, 0.8, 0.6, 0.15, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = symmetrize(np.maximum(a_trial, 0.0))
            a_trial = normalize(a_trial)
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break
    
    return best_a

def run_fista(a_init, deadline, tau=100):
    """FISTA on smooth max approximation."""
    a = symmetrize(np.maximum(a_init.copy(), 0.0))
    a = normalize(a)
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    it = 0
    max_iters = 5000
    while it < max_iters and time.monotonic() < deadline:
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient
        grad = np.zeros(n)
        for i in range(n):
            grad[i] = 2.0 * np.dot(p[i:i + n], y)
        
        grad_tan = grad - grad.mean()
        
        lr = 0.3 / (1.0 + 0.01 * it)
        a_new = y - lr * grad_tan
        a_new = proj_simplex(a_new)
        a_new = symmetrize(a_new)
        a_new = normalize(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        y_new = symmetrize(y_new)
        y_new = normalize(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        it += 1
        
        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a

def perturbation(a, rng, strength=0.1):
    """Random perturbation for simulated annealing."""
    n = len(a)
    noise = rng.normal(0, strength, n)
    a_pert = a + noise
    a_pert = np.maximum(a_pert, 0.0)
    a_pert = symmetrize(a_pert)
    a_pert = normalize(a_pert)
    return a_pert

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
