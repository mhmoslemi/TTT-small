import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        return np.ones(n) / n
    rho_idx = int(np.argmax(~cond))
    if not (~cond).any():
        rho_idx = n - 1
    theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def lp_solve(a, active_indices, M, n):
    """Solve the LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        # (a * g)[k] = sum_j a[k-j] * g[j]
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.status == 0 and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g / g.sum(), r.equality_constr_status if hasattr(r, 'equality_constr_status') else None
    except Exception:
        pass

import numpy as np

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        return np.ones(n) / n
    if (~cond).any():
        rho_idx = int(np.argmax(~cond))
    else:
        rho_idx = n - 1
    theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_lp_matrix(n, a, active_indices):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        # (a * g)[k] = sum_j a[k-j] * g[j] where a[m]=0 outside [0,n-1]
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            m = k - j
            if 0 <= m < n and a[m] > 1e-12:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    if not rows:
        return csr_matrix((len(active_indices), n))
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))


def solve_lp(a, active_indices, M, n):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0.
    Returns (g_normalized, dual_values) or (None, None).
    """
    A = build_lp_matrix(n, a, active_indices)
    if A.nnz == 0:
        return None, None
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9})
        if r.status == 0 and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-12:
                # Get dual variables for inequality constraints
                duals = r.ineqlin.marginals if hasattr(r.ineqlin, 'marginals') else None
                return g / gs, duals
    except Exception:
        pass
    return None, None

def frank_wolfe_step(a, b, n, t_iter):
    """Single Frank-Wolfe step with away-steps.
    Returns updated a.
    """
    M = b.max()
    k_star = int(np.argmax(b))
    
    # Subgradient: g[i] = 2 * a[k_star - i] (with zeros out of range)
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]
    
    # Forward direction: minimize <g, v> over simplex => v = e_{argmin g}
    i_fwd = int(np.argmin(g))
    
    # Away step: remove mass from the "heaviest" element
    i_away = int(np.argmax(a))
    
    # Try forward step
    t_fw = 2.0 / (t_iter + 2.0)
    a_fw = a.copy()
    a_fw[i_fwd] += t_fw * (1.0 - a[i_fwd])
    a_fw = np.maximum(a_fw, 0.0)
    a_fw = a_fw / a_fw.sum()
    
    # Try away-forward step
    a_awf = a.copy()
    away_ratio = a[i_away] / max(a.sum(), 1e-15)
    t_awf = min(t_fw, 0.5)
    a_awf[i_fwd] += t_awf
    a_awf[i_away] -= t_awf * (1.0 + (1.0 - a[i_away]) / max(a[i_away], 1e-15) * 0.3)
    a_awf = np.maximum(a_awf, 0.0)
    s_awf = a_awf.sum()
    if s_awf > 1e-10:
        a_awf = a_awf / s_awf
    
    # Pick the better one
    b_fw = fast_conv(a_fw)
    M_fw = b_fw.max()
    
    if a_awf is not None and s_awf > 1e-10:
        b_awf = fast_conv(a_awf)
        M_awf = b_awf.max()
        if M_awf < M_fw:
            return a_awf
    return a_fw


def run_frank_wolfe(a_init, max_iters, deadline):
    """Run Frank-Wolfe with away-steps."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:

def coordinate_1d(a, i, b, n):
    """Optimize a single coordinate a[i] via golden section search."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute bases for affected convolution values
    c_vals = np.zeros(k_hi - k_lo + 1)
    s_vals = np.zeros(k_hi - k_lo + 1)
    quad = False
    
    for idx, k in enumerate(range(k_lo, k_hi + 1)):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            s_vals[idx] = 0.0
            quad = True
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                s_vals[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b[k]
                s_vals[idx] = 0.0
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    def obj(x_test):
        vals = c_vals + s_vals * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad:
            qv = c_vals[np.where(np.abs(s_vals) < 1e-15)[0]] 
            # Find the quadratic term
            for idx2 in range(k_hi - k_lo + 1):
                if abs(s_vals[idx2]) < 1e-15 and k_lo + idx2 == 2 * i:
                    qv2 = c_vals[idx2] + x_test * x_test
                    if qv2 > m:
                        m = qv2
                    break
        return m
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, 0.5)
    
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) <= obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        order = np.random.permutation(n)
        sweep_changed = False
        
        for i in order:
            x_opt = coordinate_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-9:
                delta = x_opt - a[i]
                a[i] = x_opt
                # Update b in-place
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2 * a[i] * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2 * a[j] * delta
                sweep_changed = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_changed:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed parent
        for sigma in [0.02, 0.05, 0.1]:
            p_pert = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_pert / p_pert.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    
    # Gaussian-like
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: LP best-response with adaptive active set,
    Frank-Wolfe, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 1: Primary LP optimization on parent length ===
    n_parent = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP optimization (n={n_parent})", flush=True)
    print(f"{'='*60}", flush=True)
    
    t_phase1_end = time.monotonic() + 0.45 * (deadline - time.monotonic())
    inits = generate_inits(n_parent, rng, tracker.best if tracker.best is not None else parent)
    
    best_for_n = None
    best_M_n = float('inf')
    
    # Allocate time among initializations (more time for parent-based)
    time_per_init = (t_phase1_end - time.monotonic()) / len(inits)
    
    for ii, a0 in enumerate(inits):
        if time.monotonic() > t_phase1_end:
            break
        sub_deadline = min(t_phase1_end, time.monotonic() + time_per_init)
        if time.monotonic() >= sub_deadline and ii > 0:
            break
        
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()
        
        print(f"  Init {ii}/{len(inits)} (parent={ii<4})", flush=True)
        a_opt = run_lp_optimization(a0, sub_deadline, max_active=80, verbose=(ii == 0))
        M_opt = fast_conv(a_opt).max()
        
        if M_opt < best_M_n:
            best_M_n = M_opt
            best_for_n = a_opt.copy()
    
    if best_for_n is not None:
        print(f"  Best for n={n_parent}: M={best_M_n:.10f}", flush=True)
        tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
    
    # === Phase 2: FW + LP alternating on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: FW + LP alternating", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t_phase2_end = time.monotonic() + 0.25 * (deadline - time.monotonic())
        a_cur = tracker.best.copy()
        
        # Alternate between FW (fast) and LP (powerful)
        while time.monotonic() < t_phase2_end:
            # FW block (fast, many iterations)
            fw_deadline = min(t_phase2_end - 5, time.monotonic() + 20)
            a_fw = run_frank_wolfe(a_cur, 50000, fw_deadline)
            M_fw = fast_conv(a_fw).max()
            if M_fw < fast_conv(a_cur).max():
                a_cur = a_fw
                tracker.update(a_cur)
            
            # LP block (slower, fewer iterations)
            lp_deadline = min(t_phase2_end, time.monotonic() + 40)
            if time.monotonic() >= lp_deadline:
                break
            a_lp = run_lp_optimization(a_cur, lp_deadline, max_active=60, verbose=False)
            M_lp = fast_conv(a_lp).max()
            if M_lp < fast_conv(a_cur).max():
                a_cur = a_lp
                tracker.update(a_cur)
            else:
                # No improvement, break
                if time.monotonic() > t_phase2_end * 0.5:
                    break
    
    tracker.report("After Phase 2")
    
    # === Phase 3: Coordinate descent polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t_cd_end = min(deadline - 30, time.monotonic() + 0.15 * (deadline - time.monotonic()))
        n_cd = tracker.best.size
        max_sweeps = max(10, int((t_cd_end - time.monotonic()) / max(0.1, n_cd / 200.0)))
        a_cd = run_coordinate_descent(tracker.best, max_sweeps, t_cd_end)
        tracker.update(a_cd)
        
        # Quick LP after CD
        if time.monotonic() < deadline - 15:
            a_lp2 = run_lp_optimization(a_cd, min(deadline - 10, time.monotonic() + 30),
                                        max_active=50, verbose=False)
            tracker.update(a_lp2)
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Alternative lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Alternative lengths", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_cur = tracker.best.size
        alt_lengths = []
        for mult in [1.25, 0.8, 1.5, 0.67]:
            nl = int(n_cur * mult)
            nl = max(256, min(2048, nl))
            nl = (nl // 16) * 16
            if nl != n_cur and nl not in alt_lengths:
                alt_lengths.append(nl)
        
        for nl in alt_lengths[:3]:
            if time.monotonic() > deadline - 20:
                break
            a_alt = resample_seq(tracker.best, nl)
            a_alt = np.maximum(a_alt, 1e-6)
            a_alt = a_alt / a_alt.sum()
            
            sub_dl = min(deadline - 10, time.monotonic() + 40)
            a_opt = run_lp_optimization(a_alt, sub_dl, max_active=60, verbose=False)
            tracker.update(a_opt)
            print(f"  n={nl}: J={tracker.best_score:.8f}", flush=True)
    
    tracker.report("After Phase 4")
    
    # === Phase 5: Final intensive LP on global best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final intensive LP", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 10:
        a_final = run_lp_optimization(tracker.best, deadline - 5, max_active=100, verbose=True)
        tracker.update(a_final)
        
        # One more CD
        if time.monotonic() < deadline - 3:
            a_cd_final = run_coordinate_descent(a_final, 20, deadline - 1, )
            tracker.update(a_cd_final)
    
    tracker.report("After Phase 5")
    
    # Final report
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
