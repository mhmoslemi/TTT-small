Now, for the **simplex projection** (project onto {x : x >= 0, sum(x) = 1}):

The Euclidean projection of a point `v` onto the probability simplex is:
`P(v)_i = max(v_i - tau, 0)` where `tau` is chosen so that `sum_i max(v_i - tau, 0) = 1`.

Finding `tau` can be done by sorting `v` and using a formula, or by bisection.

The sorting-based method:
1. Sort `v` in descending order: `v_(1) >= v_(2) >= ... >= v_(n)`
2. Find the largest `k` such that `v_(k) - (sum_{j=1}^k v_(j) - (k-1)) / k > 0`
3. Then `tau = (sum_{j=1}^k v_(j) - (k-1)) / k`

This is O(n log n) per projection.

- We minimize `f_tau(a) = logsumexp(b(a)/tau) * tau` subject to `a` on the simplex
- The proximal operator is the simplex projection
- Step size: `L` is the Lipschitz constant of the gradient

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def simplex_projection(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = int(np.max(rho[cond]))
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)

    t1 = np.where(mask1, a[np.minimum(sum_idx, n - 1)], 0.0) @ w

    # t2[i] = sum_k w[k] * a[i-k]
    diff_idx = i_idx - k_idx
    mask2 = (diff_idx >= 0) & (diff_idx < n)
    t2 = np.where(mask2, a[np.minimum(np.abs(diff_idx), n - 1)], 0.0)

    # Need to be more careful with indexing
    # t1[i] = sum_k w[k] * a[i+k] where 0 <= i+k < n
    # t2[i] = sum_k w[k] * a[i-k] where 0 <= i-k < n

    # Redo with proper indexing
    t1 = np.zeros(n)
    t2 = np.zeros(n)
    for i in range(n):
        # t1: valid k where 0 <= i+k < n => k < n-i, k >= 0
        k_max = n - i
        if k_max > 0:
            t1[i] = np.dot(w[:k_max], a[i:i + k_max])
        # t2: valid k where 0 <= i-k < n => k <= i, k > i-n (always true since k >= 0)
        k_max2 = i + 1
        if k_max2 > 0:
            t2[i] = np.dot(w[2 * n - 1 - k_max2:], a[i::-1][:k_max2])

    return t1 + t2


def compute_grad_soft_fast(a, b, tau):
    """Faster gradient computation using FFT-based convolution."""
    n = len(a)
    shifted = np.clip(b / tau, -500.0, 500.0)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    w = exp_shifted / np.sum(exp_shifted)

    # w has length 2n-1, a has length n
    # t2[i] = sum_k w[k] * a[i-k] = (w * a)[i] (convolution)
    # t1[i] = sum_k w[k] * a[i+k] = correlation

    size = 1
    while size < 3 * n:
        size <<= 1

    # t2 = convolve(w, a) evaluated at 0..n-1
    Wf = rfft(w, size)
    Af = rfft(a, size)
    t2_full = irfft(Wf * Af, size)
    t2 = t2_full[:n]  # (w*a)[i] = sum_k w[k]*a[i-k] for i=0..n-1

    # t1[i] = sum_k w[k] * a[i+k]
    # = sum_k w[k] * a[(i+k)]
    # Let's reverse w: wr[k] = w[2n-2-k]
    # Then sum_k w[k]*a[i+k] = sum_k wr[2n-2-k] * a[i+k]
    # = sum_l wr[l] * a[i + 2n-2-l]
    # This is (wr * a)[i + 2n-2]
    wr = w[::-1]
    Wrf = rfft(wr, size)
    t1_full = irfft(Wrf * Af, size)
    t1 = t1_full[2 * n - 2:3 * n - 2]  # take the right slice

    return t1 + t2


def fista_simplex(a_init, max_iters=2000, tau_init=0.005, time_budget=100.0):
    """
    FISTA to minimize max_k b_k(a) (smoothed) subject to a on probability simplex.
    Returns the best sequence found.
    """
    n = len(a_init)
    a_curr = simplex_projection(a_init.copy().astype(np.float64))
    a_prev = a_curr.copy()
    t_k = 1.0
    best_a = a_curr.copy()
    best_obj = compute_obj(a_curr)
    start = time.monotonic()

    # Adaptive step size
    step = 0.5 / (n * tau_init)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        b = compute_conv(a_curr)
        tau = max(tau_init * (0.995 ** it), 0.0001)

        # Compute gradient of smoothed objective
        grad = compute_grad_soft_fast(a_curr, b, tau)

        # FISTA momentum step
        y = a_prev + ((t_k - 1) / (t_k + 1)) * (a_curr - a_prev)

        # Gradient step
        z = y - step * grad

        # Simplx projection (proximal step)
        a_new = simplex_projection(z)

        # Check convergence / update
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a_prev = a_curr.copy()
        a_curr = a_new
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2))

        # Adaptive step: increase if progress, decrease if not
        if obj_new <= best_obj + 1e-12:
            step *= 1.02
        else:
            step *= 0.8

        t_k = t_new
        # Reduce tau slowly
        tau_init = max(tau_init * 0.999, 0.0001)

    return best_a


def lp_peak_minimize(a, time_limit=20):
    """
    Solve LP: minimize t s.t. conv(a,g) <= t, sum(g) = 1, g >= 0.
    Returns g or None.
    """
    n = len(a)
    # Build Toeplitz matrix for conv(a, g)
    m = 2 * n - 1
    A = np.zeros((m, n + 1), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    A[:, n] = -1.0  # coefficient for t

    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t

    b_ub = np.full(m, 1e10)  # we'll handle this differently
    # Actually, let's reformulate:
    # min t
    # s.t. A_toep @ g <= t * 1
    #      sum(g) = 1
    #      g >= 0
    # => A_toep @ g - t * 1 <= 0
    #    sum(g) = 1
    #    g >= 0, t >= 0 (implicit since it bounds non-negative quantities)

    # Variables: [g_0, ..., g_{n-1}, t]
    # Constraints: A_toep @ g - t <= 0 for each row
    #              sum(g) = 1
    #              g >= 0, t free (but will be non-negative at optimum)

    A_toep = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A_toep[k_start:k_end, j] = a[:k_end - k_start]

    # A_ub @ x <= b_ub where x = [g, t]
    A_ub = np.hstack([A_toep, -np.ones((m, 1))])

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate_peak(a, max_iters=30, time_budget=60.0):
    """LP alternating: minimize peak of conv(a,g) to find better g."""
    n = len(a)
    a = simplex_projection(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g = lp_peak_minimize(a, time_limit=max(2, int(remaining * 0.4)))
        if g is None:
            break

        # Try different blend fractions
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = simplex_projection((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t
                best_a = a_try.copy()

        a = best_a
        if best_t == 0.0:
            # No improvement, perturb
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            a = simplex_projection(a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def targeted_peak_reduction(a, time_budget=30.0):
    """
    Targeted moves that specifically reduce the maximum autocorrelation value.
    """
    n = len(a)
    a = simplex_projection(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        b = compute_conv(a)
        max_b = float(np.max(b))
        k_star = int(np.argmax(b))

        # Find top contributors to b[k_star]
        # b[k_star] = sum_j a[j] * a[j + k_star]
        contribs = []
        for j in range(n):
            j2 = j + k_star
            if j2 < n:
                contribs.append((a[j] * a[j2], j, j2))
        contribs.sort(reverse=True)

        improved = False
        for peak_val, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try moving mass from the larger of j1, j2 to a random position
            src = j1 if a[j1] >= a[j2] else j2
            for attempt in range(5):
                dst = int(rng.integers(0, n))
                if dst == j1 or dst == j2:
                    continue
                # Check if dst would contribute to another peak
                b_dst = compute_conv(a)  # reuse
                # Simple heuristic: move a small amount
                frac = rng.uniform(0.02, 0.1)
                delta = frac * a[src]
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                if a_try[src] < 0:
                    continue
                a_try = simplex_projection(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            # Try a different peak
            # Find the second worst b value
            b_sorted = np.sort(b)[::-1]
            if len(b_sorted) > 1 and b_sorted[1] > 0.9 * b_sorted[0]:
                # Multiple peaks are close, try a more global move
                top_k = np.argsort(b)[::-1][:5]
                # Compute "pressure" vector: sum of contributions from top peaks
                pressure = np.zeros(n)
                for k in top_k:
                    for j in range(n):
                        j2 = j + int(k)
                        if j2 < n:
                            pressure[j] += a[j2]
                            pressure[j2] += a[j]
                # Move mass from high-pressure to low-pressure positions
                high_idx = np.argsort(pressure)[::-1][:3]
                low_idx = np.argsort(pressure)[:3]
                for h in high_idx:
                    for l in low_idx:
                        if h == l:
                            continue
                        delta = 0.03 * a[h]
                        a_try = a.copy()
                        a_try[h] -= delta
                        a_try[l] += delta
                        a_try = simplex_projection(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break

            if not improved and it % 10 == 0:
                # Occasional perturbation
                a = simplex_projection(best_a * np.exp(rng.normal(0, 0.02 + 0.01 * (it % 5), n)))

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_max_sum(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    result = np.interp(x_new, x_old, a)
    return result


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search:
    1. FISTA with simplex projection (accelerated, smooth)
    2. LP alternating (peak minimization formulation)
    3. Targeted peak reduction
    4. Multi-n exploration
    5. Final intensive polish with all methods
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # Phase 1: FISTA optimization
    print(f"\n{'='*60}")
    print(f"Phase 1: FISTA with simplex projection (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        for cycle in range(6):
            if time.monotonic() > deadline - 400:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.12, 60.0)
            tau = max(0.003, 0.01 * (0.9 ** cycle))

            a_new = fista_simplex(a, max_iters=3000, tau_init=tau, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  FISTA cycle {cycle} (tau={tau:.5f}): {best_score:.8f} **")
            else:
                a = 0.8 * a + 0.2 * a_new
                print(f"  FISTA cycle {cycle} (tau={tau:.5f}): {obj_new:.8f}")

    # Phase 2: LP peak minimization
    print(f"\n{'='*60}")
    print(f"Phase 2: LP peak minimization (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        n_cur = len(a)
        if n_cur <= 1000:
            for cycle in range(4):
                if time.monotonic() > deadline - 300:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break
                alloc = min(remaining * 0.1, 50.0)

                a_new = lp_alternate_peak(a, max_iters=15, time_budget=alloc)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a = a_new
                    print(f"  LP peak cycle {cycle}: {best_score:.8f} **")
                else:
                    a = 0.7 * a + 0.3 * a_new
                    print(f"  LP peak cycle {cycle}: {obj_new:.8f}")

    # Phase 3: Targeted peak reduction
    print(f"\n{'='*60}")
    print(f"Phase 3: Targeted peak reduction")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        for cycle in range(5):
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.1, 30.0)

            a_new = targeted_peak_reduction(a, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  Targeted cycle {cycle}: {best_score:.8f} **")
            else:
                print(f"  Targeted cycle {cycle}: {obj_new:.8f}")
                # Perturb for next cycle
                rng = np.random.default_rng((cycle * 31 + 7) % (2**31))
                a = simplex_projection(a_new * np.exp(rng.normal(0, 0.02, len(a_new))))

    # Phase 4: FISTA again (should be better with improved starting point)
    print(f"\n{'='*60}")
    print(f"Phase 4: FISTA round 2")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        for cycle in range(4):
            if time.monotonic() > deadline - 150:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.15, 50.0)
            tau = max(0.001, 0.005 * (0.85 ** cycle))

            a_new = fista_simplex(a, max_iters=3000, tau_init=tau, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  FISTA2 cycle {cycle} (tau={tau:.5f}): {best_score:.8f} **")
            else:
                a = 0.85 * a + 0.15 * a_new
                print(f"  FISTA2 cycle {cycle}: {obj_new:.8f}")

    # Phase 5: Multi-n exploration
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else 700
    n_options = []
    for delta in range(-100, 151, 25):
        n_try = n_best + delta
        if 200 <= n_try <= 1500:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.1, 25.0)

        # Quick FISTA
        a_try = fista_simplex(a_try, max_iters=2000, tau_init=0.003, time_budget=alloc)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # Phase 6: Random restarts with diverse perturbations
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)

        for restart in range(10):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.1, 25.0)

            if restart % 5 == 0:
                a_try = best_seq * np.exp(rng.normal(0, 0.06, n_cur))
            elif restart % 5 == 1:
                # Asymmetric shift
                shift = int(rng.integers(-n_cur//4, n_cur//4 + 1))
                a_try = np.roll(best_seq, shift)
            elif restart % 5 == 2:
                # Multiply by random smooth function
                freq = rng.uniform(1, 4)
                phase = rng.uniform(0, 2*np.pi)
                mod = 1.0 + 0.1 * np.cos(2*np.pi*freq*np.arange(n_cur)/n_cur + phase)
                a_try = best_seq * mod
            elif restart % 5 == 3:
                # Blend with asymmetric random
                a_asym = rng.random(n_cur)
                a_asym = np.abs(a_asym - 0.5) + 0.1
                mix = rng.uniform(0.1, 0.3)
                a_try = (1-mix)*best_seq + mix*a_asym
            else:
                # Subsample and resample (slight smoothing)
                sub_n = max(50, n_cur // 2)
                sub_idx = rng.choice(n_cur, sub_n, replace=False)
                sub_idx = np.sort(sub_idx)
                a_sub = best_seq[sub_idx]
                a_try = resample_sequence(a_sub, n_cur)

            a_try = simplex_projection(a_try)
            a_try = fista_simplex(a_try, max_iters=1500, tau_init=0.003, time_budget=alloc)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")

    # Phase 7: Final intensive polish
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = best_seq.copy()
        n_final = len(a_final)

        for polish_round in range(6):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            alloc = min(remaining * 0.2, 30.0)

            # FISTA with very small tau
            a_pol = fista_simplex(a_final, max_iters=2000, tau_init=0.001,
                                  time_budget=alloc * 0.5)

            # Targeted reduction
            a_pol = targeted_peak_reduction(a_pol, time_budget=alloc * 0.3)

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish_round}: {best_score:.8f} **")
            else:
                a_final = 0.9 * a_final + 0.1 * a_pol
                print(f"  Polish {polish_round}: {obj_pol:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
