```python
"""
Hybrid FISTA + LP + coordinate descent optimizer for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses accelerated gradient (FISTA) on log-sum-exp smoothed objective,
warm-started LP refinement, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_obj_and_grad(a, tau):
    """
    Compute smoothed objective and its gradient.
    Uses log-sum-exp smoothing of the max in the objective.
    Objective: 2*n*logsumexp(conv(a,a))/tau / (sum(a))^2
    (The /tau is absorbed: we use logsumexp directly)
    """
    n = len(a)
    s = float(np.sum(a))
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    
    # Log-sum-exp
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_shifted = np.exp(shifted)
    lse = cmax + tau * np.log(float(np.sum(exp_shifted)))
    
    # Weights for gradient
    weights = exp_shifted / float(np.sum(exp_shifted))
    
    # Gradient of lse wrt a:
    # grad_lse[i] = sum_k weights[k] * d(conv_k)/d(a_i)
    # d(conv_k)/d(a_i) = 2*a[k-i] if 0 <= k-i < n, else 0
    # grad_lse[i] = 2 * sum_k weights[k] * a[k-i] * I(0<=k-i<n)
    #             = 2 * sum_j a[j] * weights[i+j] (where j = k-i, so k = i+j)
    
    # Compute via FFT:
    # grad_lse[i] / 2 = sum_{j=0}^{n-1} a[j] * weights[i+j]
    # Define a_flip[j] = a[n-1-j]
    # (a_flip * weights)[n-1+i] = sum_j a_flip[j] * weights[n-1+i-j]
    #   = sum_j a[n-1-j] * weights[n-1+i-j]
    # Let m = n-1-j: = sum_m a[m] * weights[i+m]
    # So grad_lse[i] = 2 * (a_flip * weights)[n-1+i]
    
    a_flip = a[::-1]
    size = 1
    while size < n + 2 * n - 1:
        size <<= 1
    Af = rfft(a_flip, size)
    Wf = rfft(weights, size)
    cross = irfft(Af * Wf, size)
    grad_lse = 2.0 * cross[n - 1:2 * n - 1]
    
    # Full gradient of obj = 2*n*lse / s^2
    # d(obj)/d(a_i) = 2*n*(grad_lse[i]/s^2 - 2*lse/s^3)
    #               = 2*n*(grad_lse[i]*s - 2*lse) / s^3
    grad_obj = 2.0 * n * (grad_lse * s - 2.0 * lse) / (s ** 3)
    
    obj = 2.0 * n * lse / (s * s)
    return obj, grad_obj


def fista_smoothed(a_init, tau=0.01, max_iters=2000, time_budget=60.0, 
                    tau_decay=0.995, step_init=0.5):
    """
    FISTA (accelerated proximal gradient) on the smoothed objective.
    Projection onto {a >= 0, sum(a) = 1} after each step.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    ak = a.copy()  # FISTA momentum variable
    best_a = a.copy()
    best_obj = compute_obj(a)
    t_k = 1.0
    step = step_init
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        obj, grad = smooth_obj_and_grad(a, tau)
        
        # Gradient step
        a_new = a - step * grad
        
        # Project onto {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        ak = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        ak = np.maximum(ak, 0.0)
        sa = float(np.sum(ak))
        if sa < 1e-15:
            ak = a_new.copy()
        else:
            ak = ak / sa
        t_k = t_new
        a = a_new
        
        # Check actual (unsmoothed) objective
        actual_obj = compute_obj(a)
        if actual_obj < best_obj:
            best_obj = actual_obj
            best_a = a.copy()
        
        # Backtracking: if smoothed obj increased too much, reduce step
        obj_new_smooth, _ = smooth_obj_and_grad(a_new, tau)
        if obj_new_smooth > obj + 0.5 * step * step * float(np.dot(grad, grad)):
            step *= 0.7
        
        # Decay temperature
        tau *= tau_decay
        if tau < 1e-6:
            tau = 1e-6
    
    return best_a


def fista_with_restart(a_init, time_budget=120.0):
    """FISTA with multiple restarts at different temperature scales."""
    n = len(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    tau_values = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for tau in tau_values:
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        alloc = min(remaining * 0.3, 25.0)
        
        a_new = fista_smoothed(best_a, tau=tau, max_iters=3000, 
                               time_budget=alloc, tau_decay=0.998, step_init=0.3)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # For large n, this can be slow. Only use for n <= 800.
    if n > 800:
        return None
    
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, max_iters=20, time_budget=60.0):
    """LP-based refinement with line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break
        
        # Line search: find best blend
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            blend = normalize((1 - t) * a + t * g)
            obj_t = compute_obj(blend)
            if obj_t < best_blend_obj:
                best_blend_obj = obj_t
                best_t = t
        
        if best_t > 0:
            a = normalize((1 - best_t) * a + best_t * g)
            if best_blend_obj < best_obj:
                best_obj = best_blend_obj
                best_a = a.copy()
    
    return best_a


def coordinate_polish(a, time_budget=20.0):
    """Coordinate descent polishing: move mass from high-contributor pairs to reduce peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        
        # Find top contributors to peak
        contribs = []
        for j in range(n):
            k2 = k_peak - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)
        
        improved = False
        for _, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                # Reduce a[j1]^2 by spreading
                delta = 0.1 * a[j1]
                # Move to random position far from peak
                targets = [j for j in range(n) if abs(j - j1) > n // 4]
                if not targets:
                    continue
                j_target = int(rng.choice(targets))
                a_try = a.copy()
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
            else:
                # Move mass from larger of j1, j2 to random position
                src = j1 if a[j1] > a[j2] else j2
                delta = 0.05 * a[src]
                j_target = int(rng.integers(0, n))
                if j_target == src or j_target == j1 or j_target == j2:
                    continue
                a_try = a.copy()
                a_try[src] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
        
        if not improved:
            # Random perturbation to escape
            if rng.random() < 0.3:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                break
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(x_new, x_old, a)
    return normalize(new_a)
```

```python
"""
Hybrid FISTA + LP + coordinate descent optimizer for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses accelerated gradient (FISTA) on log-sum-exp smoothed objective with
multi-temperature restarts, warm-started LP refinement, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_obj_and_grad(a, tau):
    """
    Compute smoothed objective (log-sum-exp) and its gradient.
    obj = 2*n*LSE(conv(a,a)) / (sum(a))^2
    where LSE(x) = (1/tau)*log(sum(exp(tau*x)))  [we use the equivalent form]
    """
    n = len(a)
    s = float(np.sum(a))
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    # Log-sum-exp (numerically stable)
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_shifted = np.exp(shifted)
    lse = cmax + tau * np.log(float(np.sum(exp_shifted)))

    # Weights for gradient computation
    weights = exp_shifted / float(np.sum(exp_shifted))

    # Gradient of LSE wrt a:
    # d(LSE)/d(a_i) = sum_k weights[k] * d(conv_k)/d(a_i)
    # d(conv_k)/d(a_i) = 2*a[k-i] if 0 <= k-i < n, else 0
    # So: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j]
    # (where weights is defined on [0, 2n-2])

    # Direct O(n^2) computation (fast for n <= 1200)
    grad_lse = np.zeros(n, dtype=np.float64)
    for i in range(n):
        # sum_j a[j] * weights[i+j] for j=0..n-1 (weights[i+j] valid since i+j <= (n-1)+(n-1) = 2n-2)
        grad_lse[i] = 2.0 * np.dot(a, weights[i:i + n])

    # Full gradient of obj = 2*n*lse / s^2
    # d(obj)/d(a_i) = 2*n * (grad_lse[i]/s^2 - 2*lse/s^3)
    #               = 2*n * (grad_lse[i]*s - 2*lse) / s^3
    grad_obj = 2.0 * n * (grad_lse * s - 2.0 * lse) / (s * s * s)

    obj = 2.0 * n * lse / (s * s)
    return obj, grad_obj


def fista_smoothed(a_init, tau=0.01, max_iters=5000, time_budget=60.0,
                   tau_decay=0.997, step_init=0.5):
    """
    FISTA (accelerated proximal gradient) on the smoothed objective.
    Projection: {a >= 0, sum(a) = 1}.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    ak = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    t_k = 1.0
    step = step_init
    start = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        obj, grad = smooth_obj_and_grad(a, tau)
        grad_norm = float(np.linalg.norm(grad))

        if grad_norm < 1e-12:
            break

        # Normalized gradient step (scale-invariant)
        a_new = a - step * grad

        # Project onto {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Backtracking line search
        obj_new, _ = smooth_obj_and_grad(a_new, tau)
        tries = 0
        while obj_new > obj + 0.5 * step * step * grad_norm * grad_norm and tries < 10:
            step *= 0.6
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-15:
                a_try = a.copy()
            else:
                a_try = a_try / s_try
            obj_new, _ = smooth_obj_and_grad(a_try, tau)
            a_new = a_try
            tries += 1

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        ak = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        ak = np.maximum(ak, 0.0)
        sa = float(np.sum(ak))
        if sa < 1e-15:
            ak = a_new.copy()
        else:
            ak = ak / sa
        t_k = t_new
        a = a_new

        # Evaluate actual objective
        actual_obj = compute_obj(a)
        if actual_obj < best_obj - 1e-12:
            best_obj = actual_obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Decay temperature
        tau *= tau_decay
        if tau < 1e-7:
            tau = 1e-7

        # If stalled for too long, break
        if stall_count > 200:
            break

    return best_a


def fista_multi_scale(a_init, time_budget=120.0):
    """FISTA with multiple temperature scales (coarse to fine) and restarts."""
    n = len(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    tau_values = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    for tau in tau_values:
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        alloc = min(remaining * 0.35, 30.0)

        a_new = fista_smoothed(best_a, tau=tau, max_iters=5000,
                               time_budget=alloc, tau_decay=0.998, step_init=0.4)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"    FISTA tau={tau:.4f}: {best_obj:.8f} **")
        else:
            # Blend to maintain progress
            blended = normalize(0.7 * best_a + 0.3 * a_new)
            obj_blend = compute_obj(blended)
            if obj_blend < best_obj:
                best_obj = obj_blend
                best_a = blended.copy()

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 850:
        return None
    try:
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, max_iters=20, time_budget=90.0):
    """LP-based refinement with blend line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(5, int(remaining * 0.6)))
        if g is None:
            break

        # Golden section line search for blend parameter
        def obj_at_t(t):
            blend = normalize((1.0 - t) * a + t * g)
            return compute_obj(blend)

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = obj_at_t(c_pt)
        fd = obj_at_t(d_pt)
        for _ in range(12):
            if hi - lo < 0.01:
                break
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = obj_at_t(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = obj_at_t(d_pt)

        t_opt = (lo + hi) / 2.0
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Small perturbation to escape
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def coordinate_polish(a, time_budget=25.0, seed=999):
    """Coordinate descent: move mass to reduce convolution peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))

        # Find top contributors to the peak
        contribs = []
        for j in range(n):
            k2 = k_peak - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        improved = False
        for _, j1, j2 in contribs[:10]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                # Spread the diagonal contribution
                for _ in range(3):
                    j_target = int(rng.integers(0, n))
                    if j_target == j1:
                        continue
                    delta = 0.08 * a[j1]
                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
            else:
                # Move mass from the larger contributor
                src = j1 if a[j1] >= a[j2] else j2
                for _ in range(3):
                    j_target = int(rng.integers(0, n))
                    if j_target in (src, j1, j2):
                        continue
                    delta = 0.06 * a[src]
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved:
            if rng.random() < 0.2:
                a = normalize(a * np.exp(rng.normal(0, 0.015, n)))
            else:
                break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(x_new, x_old, a)
    return normalize(new_a)

def spectral_reshape(a, max_iters=80, time_budget=30.0, alpha_init=0.3):
    """
    Spectral equalization: flatten the power spectrum to reduce convolution peaks.
    Alternates between spectral compression and time-domain projection.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    from scipy.fft import fft, ifft

    size = 1
    while size < 2 * n:
        size <<= 1

    alpha = alpha_init
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Compress spectrum toward uniform
        P_norm = P / p_max
        # Use a soft power compression
        P_new = (P_norm ** alpha) * p_max

        # Reconstruct
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        # Project: non-negativity + normalization
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        # Blend to maintain stability
        a_blend = normalize(0.6 * a + 0.4 * a_new)
        obj_blend = compute_obj(a_blend)

        if obj_blend < best_obj:
            best_obj = obj_blend
            best_a = a_blend.copy()
            a = a_blend
        else:
            a = normalize(0.8 * a + 0.2 * a_new)

        alpha *= 1.02  # Gradually increase compression

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase hybrid optimizer:
    Phase 1: FISTA multi-scale (primary fast optimizer)
    Phase 2: LP refinement (exact direction finding)
    Phase 3: Spectral reshaping (different optimization geometry)
    Phase 4: Multi-length exploration
    Phase 5: Final intensive polish (LP + FISTA + coordinate)
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = get_parent()
    if parent is not None:
        best_seq = normalize(parent.copy())
        best_score = parent_obj

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        print(f"Starting from uniform n=700: score={best_score:.6f}")

    n_init = len(best_seq)
    print(f"Initial: n={n_init}, score={best_score:.8f}")
    print(f"Time budget: {budget_s}s, deadline margin: 25s")

    # ============ Phase 1: FISTA multi-scale (35% time) ============
    print(f"\n{'='*60}")
    print(f"Phase 1: FISTA multi-scale (n={n_init})")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 300:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.35, 350.0)
        a_new = fista_multi_scale(best_seq, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Phase 1 done: {best_score:.8f} **")
        else:
            print(f"  Phase 1: {obj_new:.8f} (no improvement)")

    # ============ Phase 2: LP refinement (20% time) ============
    print(f"\n{'='*60}")
    print(f"Phase 2: LP refinement (n={len(best_seq)})")
    print(f"{'='*60}")

    if len(best_seq) <= 850 and time.monotonic() < deadline - 200:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 120.0)
        a_new = lp_refine(best_seq, max_iters=15, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Phase 2 done: {best_score:.8f} **")
        else:
            print(f"  Phase 2: {obj_new:.8f} (no improvement)")

    # ============ Phase 3: Spectral + FISTA (15% time) ============
    print(f"\n{'='*60}")
    print(f"Phase 3: Spectral reshaping + FISTA")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 200:
        remaining = deadline - time.monotonic()

        # Spectral
        alloc_s = min(remaining * 0.08, 30.0)
        if alloc_s > 5:
            a_spec = spectral_reshape(best_seq, max_iters=60, time_budget=alloc_s)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                print(f"  Spectral: {best_score:.8f} **")
            else:
                print(f"  Spectral: {obj_spec:.8f} (no improvement)")

        # FISTA with lower temps
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.12, 60.0)
            a_new = fista_smoothed(best_seq, tau=0.002, max_iters=5000,
                                   time_budget=alloc, tau_decay=0.999, step_init=0.3)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Fine FISTA: {best_score:.8f} **")
            else:
                print(f"  Fine FISTA: {obj_new:.8f}")

    # ============ Phase 4: Multi-length exploration (15% time) ============
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length exploration")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    n_options = sorted(set([
        n_cur - 100, n_cur - 50, n_cur - 25,
        n_cur + 25, n_cur + 50, n_cur + 100, n_cur + 150
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1200 and n != n_cur]

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        a_try = resample(best_seq, n_try)
        alloc = min(remaining * 0.08, 20.0)
        a_try = fista_smoothed(a_try, tau=0.005, max_iters=3000,
                               time_budget=alloc, tau_decay=0.998, step_init=0.3)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ============ Phase 5: Final intensive optimization (15% time) ============
    print(f"\n{'='*60}")
    print(f"Phase 5: Final optimization (n={len(best_seq)})")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()

        # LP if feasible
        if len(best_seq) <= 850:
            alloc_lp = min(remaining * 0.3, 50.0)
            if alloc_lp > 10:
                a_new = lp_refine(best_seq, max_iters=10, time_budget=alloc_lp)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  Final LP: {best_score:.8f} **")

        # FISTA
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc_f = min(remaining * 0.35, 40.0)
            a_new = fista_smoothed(best_seq, tau=0.001, max_iters=5000,
                                   time_budget=alloc_f, tau_decay=0.9995, step_init=0.2)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final FISTA: {best_score:.8f} **")

        # Coordinate polish
        remaining = deadline - time.monotonic()
        if remaining > 8:
            alloc_c = min(remaining * 0.3, 25.0)
            a_new = coordinate_polish(best_seq, time_budget=alloc_c)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Coord polish: {best_score:.8f} **")

    # ============ Phase 6: Tiny perturbations + quick FISTA ============
    print(f"\n{'='*60}")
    print(f"Phase 6: Fine-tuning perturbations")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 5:
        rng = np.random.default_rng(seed)
        n_fin = len(best_seq)
        for i in range(15):
            if time.monotonic() > deadline - 5:
                break
            # Tiny perturbation
            sigma = 0.003 * (1 + i * 0.5)
            a_try = best_seq * np.exp(rng.normal(0, sigma, n_fin))
            a_try = fista_smoothed(a_try, tau=0.0005, max_iters=1000,
                                   time_budget=4.0, tau_decay=0.995, step_init=0.15)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Perturb {i}: {best_score:.8f} **")

    # ============ Return ============
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={min(best_score, val) if val < np.inf else best_score:.8f}")
    print(f"Time used: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return result
```
