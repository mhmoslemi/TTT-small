import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_constraint_matrix(n, a):
    """Build sparse matrix for LP: row k has a(k-j) in column j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(a, M):
    """Solve max sum(g) s.t. A g <= M, g >= 0. Returns (g, lambda)."""
    n = len(a)
    A = build_constraint_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'dual_stall': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            g = np.maximum(r.x, 0)
            # Get dual variables (from the bound constraints)
            # In scipy linprog with method='highs', r.ineqlin.marginals gives duals for A_ub constraints
            lam = np.zeros(2 * n - 1)
            if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                if hasattr(r.ineqlin, 'marginals'):
                    lam = np.maximum(-r.ineqlin.marginals, 0)  # duals for <= constraints
            return g, lam
    except Exception:
        pass
    return None, None

def smooth_grad(a, tau):
    """Compute gradient of smooth max of (a*a)(k)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(len(b)):
        if p[k] < 1e-12:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * p[k] * a[k - j]
    return grad, b, b_max

def run_smooth_descent(a, deadline, max_iters=3000, tau=50.0, lr0=0.3):
    """Nesterov smooth descent with backtracking."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, b, b_max = smooth_grad(y, tau)
        # Nesterov update
        a_new = np.maximum(y - lr0 * grad, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        # Nesterov momentum
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        beta = (t_new - 1) / (t_new + 1)
        y = np.maximum(a_new + beta * (a_new - a), 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        t_k = t_new
        if b_max < best_M:
            best_M = b_max
            best_a = a.copy()
    return best_a

def run_lp_descent(a, deadline, max_iters=100):
    """LP mirror descent with adaptive M and line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
        # Try multiple M values
        best_g = None
        best_sum = 0
        for M_factor in [1.0, 1.05, 1.1, 1.2, 1.5, 2.0]:
            g, lam = solve_lp(a, M * M_factor)
            if g is not None and g.sum() > best_sum:
                best_g = g
                best_sum = g.sum()
        if best_g is None:
            break
        best_g /= best_g.sum()
        # Line search on step size
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
            a_t = np.maximum((1 - t) * a + t * best_g, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-15:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if not improved:
            no_imp += 1
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based mirror descent for initial convergence
2. Direct KKT (Karush-Kuhn-Tucker) active set solving for refinement
3. Multi-length exploration with resampling
4. Key innovation: solving the linear KKT system directly rather than
   relying on iterative descent, giving faster and more accurate convergence
   to the optimal fixed point.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A(n, a):
    """Build constraint matrix for LP: A[k,j] = a(k-j) if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M):
    """Solve max 1^T g s.t. A g <= M, g >= 0."""
    n = len(a)
    A = build_A(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def kkt_solve(a, S, tol=1e-10, max_iters=50):
    """
    Solve KKT conditions for given active set S.
    Find a >= 0, sum(a) = 1, such that:
      - (a*a)(k) are all equal for k in S
      - (a*a)(k) <= (a*a)(k') for k not in S, k' in S
      - For j with a_j > 0: sum_{k in S} a(k-j) = const
      - For j with a_j = 0: sum_{k in S} a(k-j) <= const
    
    Returns (a_opt, converged).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, False
    a = a / s
    
    S_list = sorted(set(S))
    S_set = set(S_list)
    
    for outer in range(max_iters):
        # Determine support J
        J = [j for j in range(n) if a[j] > tol]
        if not J:
            J = list(range(n))
        J_set = set(J)
        J_arr = np.array(J)
        m_j = len(J_arr)
        
        # Build KKT matrix: M[i,k] = 1 if J_arr[i] + J_arr[k] in S_set
        M = np.zeros((m_j, m_j))
        for i in range(m_j):
            j_i = J_arr[i]
            # M[i,k] = 1 if j_i + J_arr[k] in S
            sums = j_i + J_arr
            in_S = np.array([s_val in S_set for s_val in sums])
            M[i, in_S] = 1.0
        
        # System: [M, -1; 1^T, 0] [a_J; c] = [0; 1]
        sys_mat = np.zeros((m_j + 1, m_j + 1))
        sys_mat[:m_j, :m_j] = M
        sys_mat[:m_j, m_j] = -1.0
        sys_mat[m_j, :m_j] = 1.0
        
        rhs = np.zeros(m_j + 1)
        rhs[m_j] = 1.0
        
        # Solve
        try:
            # Add small regularization for stability
            sys_mat_reg = sys_mat.copy()
            sys_mat_reg[m_j, m_j] = 0  # keep as is, the [0] at bottom right
            sol = np.linalg.solve(sys_mat_reg, rhs)
        except np.linalg.LinAlgError:
            try:
                sol, _, _, _ = np.linalg.lstsq(sys_mat, rhs, rcond=1e-12)
            except Exception:
                break
        
        a_candidate = np.zeros(n)
        a_candidate[J_arr] = sol[:m_j]
        c_val = sol[m_j]
        
        # Check non-negativity
        min_val = a_candidate[J_arr].min()
        if min_val < -1e-8:
            # Remove violated indices
            violated = set(J_arr[a_candidate[J_arr] < -1e-8].tolist())
            J = [j for j in J if j not in violated]
            if len(J) < 2
