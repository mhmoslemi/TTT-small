def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):

def gs_flat_conv(a, n_iters=200, beta=0.5):

    a = a / s

    # In frequency domain: |A_k| = 1/sqrt(2n-1) for all k

    best_a = a.copy()
    best_M = fast_conv(a).max()

def smooth_max_gradient(a, tau, sample_lags=None):

    n = len(a)

            # So if j != k-j: d(b_k)/d(a_j) = 2 * a_{k-j}

def run_stochastic_subgrad(a, deadline, max_iters=10000):
    """Stochastic subgradient method with heavy ball momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    rng = np.random.default_rng(99)
    momentum = 0.9
    v = np.zeros(n)
    
    # Start with small lr, increase
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        progress = min(1.0, it / max_iters)
        lr = 0.1 * (1 + progress)  # Actually let's use a decaying lr
        lr = 0.5 / (1 + 0.01 * it)
        
        # Sample a random lag
        k = rng.integers(0, 2 * n - 1)
        # Compute subgradient at this lag
        b = fast_conv(a)
        b_max = float(b.max())
        
        # Use smooth max over a small window around the max
        window = max(1, min(20, (2*n-1)//10))
        top_k = np.argsort(b)[-window:]
        
        tau_s = 10.0
        exp_b = np.exp(tau_s * (b[top_k] - b[top_k].max()))
        p = exp_b / exp_b.sum()
        
        grad = np.zeros(n)
        for idx, k2 in enumerate(top_k):
            lo = max(0, k2 - n + 1)
            hi = min(n - 1, k2)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * p[idx] * a[k2 - j]
        
        # Heavy ball
        v = momentum * v - lr * grad
        a_new = np.maximum(a + v, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def build_A_sparse(n, a):
    """Build sparse matrix for LP: (A @ g)_k = conv(a, g)_k for k=0..2n-2"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def run_lp_iterations(a, deadline, max_iters=50):

    a = np.maximum(a.astype(np.float64), 1e-8)

def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_diverse_inits(n, rng, parent=None):
    """Create diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p * (1 + 0.1 * rng.standard_normal(n)), 0.01))
        inits.append(np.maximum(p + 0.02 * rng.standard_normal(n), 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Power law shapes
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t ** alpha), 0.01))
        inits.append(np.maximum(0.1 + 0.9 * ((1-t) ** alpha), 0.01))
        inits.append(np.maximum(0.1 + 0.9 * np.minimum(t ** alpha, (1-t) ** alpha), 0.01))
    
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            inits.append(np.exp(-0.5 * ((t - center) / width) ** 2) + 0.05)
    
    # Cosine shapes
    for f in [1, 2, 3, 4, 6, 8]:
        for phase in [0, np.pi/2, np.pi]:
            inits.append(np.maximum(0.5 + 0.4 * np.cos(2*np.pi*f*t + phase), 0.01))
    
    # Random smooth (low-pass filtered noise)
    for bandwidth in [3, 5, 10, 20]:
        coeffs = rng.standard_normal(n) + 0.5
        # Simple moving average
        k = max(2, n // bandwidth)
        kern = np.ones(k) / k
        smooth = np.convolve(np.abs(coeffs), kern, 'same')
        inits.append(np.maximum(smooth, 0.01))
    
    # Symmetric random
    half = np.abs(rng.standard_normal(n // 2 + 1)) + 0.3
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(np.maximum(full, 0.01))
    
    # "Barker-like" patterns (all +1 or -1 mapped to positive)
    for pattern_len in [4, 7]:
        pattern = np.ones(pattern_len)
        pattern[1:] = rng.choice([-1, 1], pattern_len - 1)
        # Tile and smooth
        tiled = np.tile(pattern, (n // pattern_len + 1))[:n]
        k = max(2, n // 30)
        kern = np.ones(k) / k
        inits.append(np.maximum(np.convolve(tiled ** 2, kern, 'same'), 0.01))
    
    # Frequency-domain random phase
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.exp(1j * 2 * np.pi * rng.random(size // 2 + 1))
    a_spec = np.fft.irfft(A, size)[:n]
    inits.append(np.maximum(a_spec, 0.01))
    
    # Flat spectrum (target)
    A_flat = np.ones(size // 2 + 1) * np.exp(1j * 2 * np.pi * rng.random(size // 2 + 1))
    a_flat = np.fft.irfft(A_flat, size)[:n]
    inits.append(np.maximum(a_flat, 0.01))
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral flattening for initialization
2. Adaptive smooth gradient descent with Nesterov acceleration and backtracking
3. LP mirror descent with backtracking line search (no Anderson for stability)
4. Multi-length exploration with coarse-to-fine refinement
5. Stochastic subgradient as complementary method
Key improvements: backtracking line search in LP phase, better spectral init,
more aggressive length exploration, and two-level (coarse-fine) optimization.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(a, n_iters=300, beta=0.7):
    """Gerchberg-Saxton: flatten |FFT(a)| to target flat convolution."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(n_iters):
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target flat magnitude
        target_mag = np.sqrt(np.sum(mag ** 2) / len(mag))

        # Adaptive blend: start gentle, become more aggressive
        alpha = min(1.0, beta * (0.3 + 0.7 * it / max(1, n_iters // 2)))
        mag_new = (1 - alpha) * mag + alpha * target_mag

        A_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def smooth_max_gradient_full(a, tau):
    """Full smooth max gradient of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / (exp_b.sum() + 1e-30)
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        w = p[k]
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * w * a[k - j]
    return grad, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=3.0, tau_end=500.0,
                        lr_init=0.3, lr_end=0.001, momentum=0.85):
    """Smooth gradient descent with Nesterov acceleration and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    y = a.copy()
    v = np.zeros(n)

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** (frac ** 0.6)

        grad, b, b_max = smooth_max_gradient_full(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        # Nesterov step with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s

        # Adaptive restart: if objective increased significantly, reset momentum
        M_new = fast_conv(a_new).max()
        if M_new > best_M * 1.5:
            v = np.zeros(n) * 0.5
        a = a_new

    return best_a


def run_smooth_backtrack(a, deadline, max_iters=3000, tau_start=5.0, tau_end=300.0):
    """Smooth gradient with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    t = 0.3
    rho = 0.6
    c1 = 1e-4

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.4)

        grad, b, b_max = smooth_max_gradient_full(a, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        # Gradient direction
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Backtracking line search
        obj_cur = b_max
        for _ in range(20):
            a_try = a - t * grad / g_norm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                t *= rho
                continue
            a_try = a_try / s_try
            M_try = fast_conv(a_try).max()
            if M_try < obj_cur - c1 * t * g_norm:
                break
            t *= rho
        else:
            t *= rho
            a_try = a

        a = a_try
        t = min(t * 1.5, 1.0)  # Try larger step next time

    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix: (A @ g)_k = sum_j a[k-j]*g[j] for k=0..2n-2."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_backtrack(a, deadline, max_iters=100):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    t_prev = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 25:
                break

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Direction
        d = g - a
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break

        # Backtracking line search
        obj_cur = M
        t = t_prev
        improved = False
        for _ in range(15):
            a_try = np.maximum(a + t * d, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                t *= 0.5
                continue
            a_try = a_try / s_try
            M_try = fast_conv(a_try).max()
            # Sufficient decrease condition (Armijo)
            if M_try < obj_cur - 1e-6 * t * d_norm:
                a = a_try
                best_M = min(best_M, M_try)
                best_a = a.copy()
                no_imp = 0
                t_prev = min(t * 1.3, 1.0)
                improved = True
                break
            t *= 0.5

        if not improved:
            # Try smaller fixed steps
            for t_fix in [0.05, 0.01]:
                a_try = np.maximum(a + t_fix * d, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                M_try = fast_conv(a_try).max()
                if M_try < best_M - 1e-14:
                    a = a_try
                    best_M = M_try
                    best_a = a.copy()
                    no_imp = 0
                    t_prev = t_fix
                    break

    return best_a


def run_lp_geometric(a, deadline, max_iters=100):
    """LP mirror descent with geometric step sequence."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        improved = False
        # Try a range of step sizes from large to small
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_t = np.maximum((1 - t) * a + t * g, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                no_imp = 0
                improved = True
                break

        if not improved:

                if M_t < best_M - 1e-15:

                    best_M = M_t
                    best_a = a.copy()
                    improved = True
                    break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p * (1 + 0.05 * rng.standard_normal(n)), 0.01))
        inits.append(np.maximum(p + 0.01 * rng.standard_normal(n), 0.01))
        inits.append(np.maximum(p * (1 + 0.15 * rng.standard_normal(n)), 0.01))

    inits.append(np.ones(n))

    # Power law / polynomial shapes
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.1 + 0.9 * t ** alpha, 0.01))
        inits.append(np.maximum(0.1 + 0.9 * (1 - t) ** alpha, 0.01))

    # Symmetric shapes
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t ** alpha + (1 - t) ** alpha) * 0.5, 0.01))
        inits.append(np.maximum(0.1 + 0.9 * np.maximum(t, 1 - t) ** alpha, 0.01))

    # Gaussian / bump shapes
    for c in [0.25, 0.5, 0.75]:
        for w in [0.1, 0.2, 0.4]:
            inits.append(np.exp(-0.5 * ((t - c) / w) ** 2) + 0.05)

    # Cosine modulation
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t + np.pi / 3), 0.01))

    # Random smooth
    k = max(3, n // 15)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(np.abs(rng.standard_normal(n)) + 0.5, kern, 'same'), 0.01))

    # Symmetric random
    half_n = n // 2 + 1
    half = np.abs(rng.standard_normal(half_n)) * 0.3 + 0.5
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(np.maximum(full, 0.01))

    # Frequency domain: random phase, flat magnitude
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    phase = 2 * np.pi * rng.random(size // 2 + 1)
    A = np.exp(1j * phase)
    a_spec = np.fft.irfft(A, size)[:n]
    inits.append(np.maximum(a_spec, 0.01))

    # GS from random
    a_rand = np.abs(rng.standard_normal(n)) * 0.3 + 0.5
    inits.append(gs_flat_spectral(a_rand, n_iters=150, beta=0.8))

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_flat_spectral(p, n_iters=200, beta=0.85))

    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

        # Estimate LP cost: roughly O(n^2) for building matrix + LP solve
        est_ms = 2.0 * (n_lp / 128.0) ** 2 + 50.0
        print(f"  n={n_lp}, est per-iter={est_ms:.0f}ms", flush=True)

        p3_deadline = min(deadline - 60.0, t_phase3_start + 500.0)
        iters_estimate = int(60.0 / max(0.01, est_ms / 1000.0))
        print(f"  Budget: {p3_deadline - time.monotonic():.0f}s, ~{iters_estimate} iters possible", flush=True)

        # First LP pass: backtracking
        a_lp1 = run_lp_backtrack(best.seq, p3_deadline, max_iters=iters_estimate)
        best.update(a_lp1)

        # Second LP pass: geometric steps (different step selection)
        if time.monotonic() < p3_deadline - 30:
            a_lp2 = run_lp_geometric(a_lp1, min(p3_deadline, time.monotonic() + 120),
                                     max_iters=int(iters_estimate * 0.5))
            best.update(a_lp2)

        # Third pass if time allows
        if time.monotonic() < p3_deadline - 20:
            a_lp3 = run_lp_backtrack(a_lp2 if time.monotonic() < p3_deadline else a_lp1,
                                     p3_deadline, max_iters=int(iters_estimate * 0.3))
            best.update(a_lp3)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic() - t_phase3_start:.1f}s", flush=True)

    # === PHASE 4: Length exploration + final polish ===
    t_phase4_start = time.monotonic()

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration + Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size

        # Try nearby lengths
        for delta in [128, -128, 256, -256, 64, -64, 512]:
            if time.monotonic() > deadline - 30:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 3072:
                continue

            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick smooth optimization
            sub_dl = min(time.monotonic() + 15, deadline - 20)
            a_ext = run_smooth_nesterov(a_ext, sub_dl, max_iters=1500,
                                         tau_start=5.0, tau_end=200.0,
                                         lr_init=0.15, lr_end=0.005,
                                         momentum=0.85)
            best.update(a_ext)

            # Quick LP if time allows
            if time.monotonic() < deadline - 15 and abs(delta) <= 256:
                est_ms_try = 2.0 * (n_try / 128.0) ** 2 + 50.0
                max_it_try = max(5, int(10.0 / max(0.01, est_ms_try / 1000.0)))
                a_lp_try = run_lp_geometric(a_ext, min(time.monotonic() + 15, deadline - 5),
                                             max_iters=max_it_try)
                best.update(a_lp_try)

        # Final LP polish on absolute best
        if best.seq is not None and time.monotonic() < deadline - 5:
            n_final = best.seq.size
            est_ms_final = 2.0 * (n_final / 128.0) ** 2 + 50.0
            max_it_final = max(5, int(20.0 / max(0.01, est_ms_final / 1000.0)))
            a_final = run_lp_backtrack(best.seq, deadline, max_iters=max_it_final)
            best.update(a_final)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic() - t_phase4_start:.1f}s", flush=True)

    # Final report
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    total_time = time.monotonic() - best.t0
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={total_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best.seq]
