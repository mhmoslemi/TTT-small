import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, beta):
    """
    Compute smoothed objective (logsumexp-based) and gradient w.r.t. a.
    Returns (obj, grad) where grad has same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # Gradient of soft_max w.r.t. a[i]
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj


def log_space_smooth_obj_and_grad(x, beta):
    """
    Smoothed objective and gradient in log-space.
    x is log(a), a = exp(x - max(x)) / sum(exp(x - max(x)))
    """
    n = len(x)
    xc = x - np.max(x)
    a = np.exp(xc)
    a = a / np.sum(a)
    
    obj, grad_a = smooth_obj_and_grad(a, beta)
    grad_x = grad_a * a  # chain rule: d(a[i])/d(x[j]) = a[i]*(delta_ij - a[j])
    # Actually: a[i] = exp(x[i]) / sum(exp(x)), so da[i]/dx[j] = a[i]*(delta_ij - a[j])
    # grad_x[j] = sum_i grad_a[i] * da[i]/dx[j] = sum_i grad_a[i] * a[i]*(delta_ij - a[j])
    # = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])
    # But sum_i grad_a[i]*a[i] = 0 for homogeneous functions... let me just use the simple version
    # Actually for the normalized version, the gradient w.r.t. x is:
    # grad_x[j] = a[j] * (grad_a[j] - sum_k grad_a[k]*a[k])
    # The second term is the "mean" correction
    mean_grad = np.sum(grad_a * a)
    grad_x = a * (grad_a - mean_grad)
    
    return obj, grad_x, a

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, res
    except Exception:
        pass
    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def structured_init(n, seed=42):
    """
    Generate structured initial sequences that are known to have good properties.
    """
    rng = np.random.default_rng(seed)
    
    # Option 1: Raised cosine (Hann) window
    x = np.arange(n)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    
    # Option 2: Gaussian window
    center = n / 2.0
    sigma = n / 4.0
    gauss = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    
    # Option 3: Flat top with smooth edges
    edge = max(1, n // 10)
    flat = np.ones(n)
    for i in range(edge):
        t = i / edge
        flat[i] = 0.5 * (1 + np.cos(np.pi * t))
        flat[n-1-i] = flat[i]
    
    # Option 4: Random with noise
    rand = 0.5 + 0.5 * np.cos(2 * np.pi * rng.uniform(0, 2) * x / n)
    rand = np.abs(rand) + 0.1
    
    # Normalize
    options = [hann, gauss, flat, rand]
    chosen = options[rng.integers(0, len(options))]
    chosen = np.maximum(chosen, 1e-10)
    chosen = chosen / np.sum(chosen)
    return chosen

1) L-BFGS as primary smoother optimizer (faster convergence than Adam)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """
    Smoothed objective (logsumexp of beta*conv) and gradient w.r.t. a.
    Uses FFT for efficient gradient.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def lbfgs_smooth(a0, beta, time_budget=60.0, maxiter=300):
    """L-BFGS in log-space on smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / np.sum(a0)

    x0 = np.log(a0)
    x0 -= np.mean(x0)

    start = time.monotonic()
    best_obj = np.inf
    best_x = x0.copy()

    def fun_and_grad(x):
        nonlocal best_obj, best_x
        if time.monotonic() - start > time_budget:
            return np.inf, np.zeros_like(x)
        xc = x - np.max(x)
        a_cur = np.exp(xc)
        a_cur = a_cur / np.sum(a_cur)
        obj, grad_a = smooth_obj_grad(a_cur, beta)
        mean_grad = np.sum(grad_a * a_cur)
        grad_x = a_cur * (grad_a - mean_grad)
        if obj < best_obj:
            best_obj = obj
            best_x = x.copy()
        return obj, grad_x

    try:
        res = minimize(fun_and_grad, x0, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': 5000,
                                'gtol': 1e-9, 'ftol': 1e-14})
    except Exception:
        pass

    xc = best_x - np.max(best_x)
    a_out = np.exp(xc)
    a_out = a_out / np.sum(a_out)
    return a_out

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.12, 50.0)
            a_lp = lp_decreasing_m(best_seq, time_budget=alloc, num_rounds=15)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP-decM {rnd}: {best_score:.8f} ***")
            else:
                print(f"  LP-decM {rnd}: {obj:.8f}")

            # Follow with L-BFGS
            if time.monotonic() < deadline - 60:
                remaining = deadline - time.monotonic()
                a_f = lbfgs_smooth(a_lp, beta=300,
                                   time_budget=min(remaining * 0.08, 25.0),
                                   maxiter=200)
                obj_f = compute_obj(a_f)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  LP+LBFGS {rnd}: {best_score:.8f} ***")

    # === Phase 3: Nesterov subgradient ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Nesterov subgradient")
    print(f"{'='*60}")

    if best_seq is not None:
        for beta_val in [80, 150, 300]:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.10, 30.0)
            a_nes = nesterov_subgrad(best_seq, time_budget=alloc, beta=beta_val)
            obj = compute_obj(a_nes)
            if obj < best_score:
                best_score = obj
                best_seq = a_nes.copy()
                print(f"  Nesterov beta={beta_val}: {best_score:.8f} ***")
            else:
                print(f"  Nesterov beta={beta_val}: {obj:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50, n_best,
            n_best + 50, n_best + 100, n_best + 150, n_best + 250
        ]))
        n_options = [n for n in n_options if 300 <= n <= 1200]

        for n_try in n_options:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
            a_try = a_try / np.sum(a_try)

            alloc = min(remaining * 0.10, 35.0)
            a_opt = adam_log_opt(a_try, max_iters=10000, time_budget=alloc,
                                 beta_start=20.0, beta_end=400.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Alternating LP + L-BFGS (final combined) ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Alternating LP + L-BFGS polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        a = a / np.sum(a)

        for it in range(8):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            # LP step
            if len(a) <= 850:
                a_lp = lp_decreasing_m(a, time_budget=min(remaining * 0.15, 20.0),
                                       num_rounds=5)
            else:
                a_lp = a

            # L-BFGS step
            a_lb = lbfgs_smooth(a_lp, beta=400,
                                time_budget=min(remaining * 0.15, 20.0),
                                maxiter=150)

            obj = compute_obj(a_lb)
            if obj < best_score:
                best_score = obj
                best_seq = a_lb.copy()
                a = a_lb.copy()
                print(f"  Alt iter {it}: {best_score:.8f} ***")
            else:
                print(f"  Alt iter {it}: {obj:.8f}")

    # === Phase 6: Final high-beta polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        a = a / np.sum(a)

        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            a_f1 = lbfgs_smooth(a, beta=800,
                                time_budget=min(remaining * 0.3, 20.0),
                                maxiter=200)
            obj1 = compute_obj(a_f1)
            if obj1 < best_score:
                best_score = obj1
                best_seq = a_f1.copy()
                a = a_f1.copy()
                print(f"  Final LBFGS: {best_score:.8f} ***")

        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_f2 = adam_log_opt(a, max_iters=10000,
                                time_budget=min(remaining * 0.4, 18.0),
                                beta_start=300.0, beta_end=1500.0,
                                lr_start=0.008, lr_decay=0.99999)
            obj2 = compute_obj(a_f2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_f2.copy()
                print(f"  Final Adam: {best_score:.8f} ***")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
