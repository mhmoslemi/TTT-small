import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build matrix A where (A @ g)[k] = conv(a, g)[k], k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp(a, M, time_limit=30):
    """Solve max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_redcost_tola': 1e-7})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def smooth_obj_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=50.0, beta_start=10.0,

    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        a = a / np.sum(a)

        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t_count)

        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if true_obj < best_true_obj:

    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def make_hann(n):
    x = np.arange(n)
    h = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    return h / np.sum(h)


def make_raised_cosine(n, beta_param=1.0):
    """Raised cosine with adjustable rolloff."""
    x = np.linspace(0, 1, n)
    h = 0.5 * (1 + np.cos(2 * np.pi * beta_param * (x - 0.5)))
    h = np.maximum(h, 0)
    s = np.sum(h)
    if s < 1e-10:
        h = np.ones(n) / n
    else:
        h = h / s
    return h

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):

    if n == 0:

    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp(a, M, time_limit=20):
    n = len(a)
    if n > 1100:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_redcost_tola': 1e-8, 'primal_feasibility_tola': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))

    ls = cmax + np.log(e_sum) / beta

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)

    return a_new


def make_hann(n):

        if not improved and outer > 30:
            break
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Aggressive iterative LP-based peak reduction with adaptive M scheduling,
    combined with gradient descent in log-space, peak shaving, coordinate
    descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0
    
    best_seq = None
    best_score = np.inf
    np.random.seed(seed)
    
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")
    
    a = best_seq.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    
    # === Phase 1: Quick Adam to establish good basin (40s) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Quick Adam optimization (n={n_parent})")
    print(f"{'='*60}")
    
    if time.monotonic() < deadline - 150:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.08, 40.0)
        a_new = adam_log_opt(a, max_iters=20000, time_budget=alloc,
                             beta_start=10.0, beta_end=300.0,
                             lr_start=0.03, lr_decay=0.99998)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Adam: {best_score:.8f} ***")
        else:
            print(f"  Adam: {obj:.8f} (no imp)")
    
    # === Phase 2: AGGRESSIVE iterative LP (majority of time) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Aggressive iterative LP (n={n_parent})")
    print(f"{'='*60}")
    
    if n_parent <= 1000 and time.monotonic() < deadline - 150:
        remaining = deadline - time.monotonic()
        # Allocate most time to LP
        alloc_lp = min(remaining * 0.5, 500.0)
        a_lp = lp_iterative_refine(a, time_budget=alloc_lp)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  LP main: {best_score:.8f} ***")
        else:
            print(f"  LP main: {obj:.8f} (no imp)")
    
    # Short Adam between major phases
    if time.monotonic() < deadline - 100 and best_seq is not None:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.05, 20.0)
        a_mid = adam_log_opt(a, max_iters=10000, time_budget=alloc,
                             beta_start=50.0, beta_end=500.0,
                             lr_start=0.01, lr_decay=0.99999)
        obj = compute_obj(a_mid)
        if obj < best_score:
            best_score = obj
            best_seq = a_mid.copy()
            a = a_mid.copy()
            print(f"  Mid-Adam: {best_score:.8f} ***")
    
    # === Phase 3: More LP after Adam ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Second LP pass (n={n_parent})")
    print(f"{'='*60}")
    
    if n_parent <= 1000 and time.monotonic() < deadline - 100:
        remaining = deadline - time.monotonic()
        alloc_lp2 = min(remaining * 0.2, 200.0)
        a_lp2 = lp_iterative_refine(a, time_budget=alloc_lp2)
        obj = compute_obj(a_lp2)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp2.copy()
            a = a_lp2.copy()
            print(f"  LP second: {best_score:.8f} ***")
        else:
            print(f"  LP second: {obj:.8f} (no imp)")
    
    # === Phase 4: Peak shaving ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Peak shaving (n={n_parent})")
    print(f"{'='*60}")
    
    if time.monotonic() < deadline - 80:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.08, 25.0)
        a_shave = peak_shave(a, time_budget=alloc)
        obj = compute_obj(a_shave)
        if obj < best_score:
            best_score = obj
            best_seq = a_shave.copy()
            a = a_shave.copy()
            print(f"  Peak shave: {best_score:.8f} ***")
        else:
            print(f"  Peak shave: {obj:.8f} (no imp)")
    
    # === Phase 5: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Coordinate descent (n={n_parent})")
    print(f"{'='*60}")
    
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.08, 25.0)
        a_cd = coordinate_descent(a, time_budget=alloc, seed=seed)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  Coord descent: {best_score:.8f} ***")
        else:
            print(f"  Coord descent: {obj:.8f} (no imp)")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Multi-n exploration")
    print(f"{'='*60}")
    
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        n_options = []
        for dn in [-50, -100, 50, 100, 150, 200, 300]:
            n_try = n_best + dn
            if 200 <= n_try <= 1200 and n_try not in n_options:
                n_options.append(n_try)
        
        for n_try in n_options:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            
            a_try = resample_sequence(best_seq, n_try)
            s_try = np.sum(a_try)
            if s_try > 1e-10:
                a_try = a_try / s_try
            
            alloc1 = min(remaining * 0.15, 30.0)
            a_opt = adam_log_opt(a_try, max_iters=10000, time_budget=alloc1,
                                 beta_start=10.0, beta_end=300.0,
                                 lr_start=0.03, lr_decay=0.99998)
            
            if n_try <= 900 and time.monotonic() < deadline - 15:
                a_opt = lp_iterative_refine(a_opt, time_budget=10.0)
            
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")
    
    # === Phase 7: Try alternative shapes ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Alternative initial shapes")
    print(f"{'='*60}")
    
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        
        h = make_hann(n_best)
        obj_h = compute_obj(h)
        print(f"  Hann initial obj: {obj_h:.8f}")
        if obj_h < best_score + 0.05:
            a_h = adam_log_opt(h, max_iters=10000, time_budget=15.0,
                               beta_start=10.0, beta_end=300.0,
                               lr_start=0.03, lr_decay=0.99998)
            obj_h = compute_obj(a_h)
            if n_best <= 900:
                a_h = lp_iterative_refine(a_h, time_budget=10.0)
                obj_h = compute_obj(a_h)
            if obj_h < best_score:
                best_score = obj_h
                best_seq = a_h.copy()
                print(f"  Hann optimized: {best_score:.8f} ***")
        
        tri = make_triangle(n_best)
        obj_tri = compute_obj(tri)
        print(f"  Triangle initial obj: {obj_tri:.8f}")
        if obj_tri < best_score + 0.05:
            a_tri = adam_log_opt(tri, max_iters=10000, time_budget=15.0,
                                 beta_start=10.0, beta_end=300.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj_tri = compute_obj(a_tri)
            if obj_tri < best_score:
                best_score = obj_tri
                best_seq = a_tri.copy()
                print(f"  Triangle optimized: {best_score:.8f} ***")
    
    # === Phase 8: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 8: Final intensive polish")
    print(f"{'='*60}")
    
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_final = len(best_seq)
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        
        # Final LP
        if n_final <= 1000 and time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_fl = lp_iterative_refine(a, time_budget=min(remaining * 0.3, 20.0))
            obj = compute_obj(a_fl)
            if obj < best_score:
                best_score = obj
                best_seq = a_fl.copy()
                a = a_fl.copy()
                print(f"  Final LP: {best_score:.8f} ***")
        
        # Final high-beta Adam
        if time.monotonic() < deadline - 3:
            remaining = deadline - time.monotonic()
            a_fa = adam_log_opt(a, max_iters=10000,
                                time_budget=min(remaining * 0.4, 10.0),
                                beta_start=200.0, beta_end=1000.0,
                                lr_start=0.005, lr_decay=0.99999)
            obj = compute_obj(a_fa)
            if obj < best_score:
                best_score = obj
                best_seq = a_fa.copy()
                print(f"  Final Adam: {best_score:.8f} ***")
        
        # Very final coordinate descent
        if time.monotonic() < deadline - 2:
            a_cd2 = coordinate_descent(a, time_budget=5.0, seed=999)
            obj = compute_obj(a_cd2)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd2.copy()
                print(f"  Final CD: {best_score:.8f} ***")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    
    return result
