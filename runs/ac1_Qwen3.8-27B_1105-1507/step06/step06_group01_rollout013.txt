"""
Multi-phase hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FISTA on smooth max, (2) full-constraint LP best-response with geometric
averaging, (3) adaptive top-k subgradient with Langevin exploration, (4) coordinate
descent polish, (5) multi-length search. Key innovation: full LP + geometric (multiplicative)
averaging for the game-theoretic best response, combined with stochastic exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    if not (~cond).any():
        rho = n - 1
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_grad_fft(a, tau):
    """Compute gradient of smooth max (log-sum-exp) of conv(a,a) using FFT.
    
    Returns (smooth_value, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau

    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # This is cross-correlation of p and a at lag i
    # Use FFT for efficiency
    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation: (p * a)[i] = sum_j p[i+j] * a[j]
    # = ifft(FFT(p) * conj(FFT(a)))
    corr = np.fft.irfft(P * np.conj(A), N)[:n]
    grad = 2.0 * corr
    return smooth_val, grad


def topk_subgradient(a, k=10):
    """Compute averaged subgradient over top-k convolution indices.
    
    Returns the averaged subgradient vector.
    """
    n = len(a)
    b = fast_conv(a)
    # Get top-k indices
    if k >= len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argsort(b)[::-1][:k]
    
    # Weight by excess above median
    b_med = np.median(b)
    weights = np.maximum(b[top_idx] - b_med, 1e-10)
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for w, k_idx in zip(weights, top_idx):
        k_idx = int(k_idx)
        # Subgradient at k_idx: s[i] = 2*a[k_idx - i] if 0 <= k_idx - i < n
        lo = max(0, k_idx - n + 1)
        hi = min(n - 1, k_idx)
        for i in range(lo, hi + 1):
            grad[i] += w * 2.0 * a[k_idx - i]
    return grad


def run_fista(a_init, tau_start, tau_end, max_iters, deadline):
    """FISTA on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_grad_fft(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Adaptive step size
        step = 1.5 / (tau * 2.0 + 1.0)
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        if it % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_a = a.copy()
    
    return best_a


def lp_best_response_full(a, M, time_limit=10.0):
    """Solve LP best response using ALL constraints.
    
    max sum(g) s.t. (a*g)[k] <= M for k=0,...,2n-2, g >= 0
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_game_descent(a_init, deadline, n_lp_iters=200, n_sg_iters=200, n_langevin_iters=100):
    """Game descent: alternate between LP best-response, subgradient, and Langevin.
    
    Key innovation: geometric (multiplicative) averaging for LP updates,
    and stochastic exploration via Langevin dynamics.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    rng = np.random.default_rng(12345)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    total_iters = n_lp_iters + n_sg_iters + n_langevin_iters
    it = 0
    
    while it < total_iters and time.monotonic() < deadline:
        # Choose method based on iteration
        if it % 10 < 5:
            # LP best response with geometric averaging
            b = fast_conv(a)
            M_cur = b.max()
            g = lp_best_response_full(a, M_cur, time_limit=8.0)
            if g is not None:
                # Geometric (multiplicative) averaging
                for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
                    a_trial = a ** (1.0 - t) * g ** t
                    a_trial = np.maximum(a_trial, 1e-15)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        no_improve = 0
                        break
                else:
                    no_improve += 1
        elif it % 10 < 8:
            # Top-k subgradient with momentum
            k_val = 5 + (it // 50) * 5  # Increase k over time
            k_val = min(k_val, 100)
            sg = topk_subgradient(a, k=k_val)
            sg_tan = sg - sg.mean()
            
            # Adaptive step
            step = 0.5 / (1.0 + 0.01 * it)
            a_new = proj_simplex(a - step * sg_tan)
            
            M_new = fast_conv(a_new).max()
            if M_new < best_M - 1e-14:
                a = a_new
                best_M = M_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Langevin perturbation (exploration)
            noise_scale = 0.02 / (1.0 + 0.005 * it)
            noise = rng.normal(0, noise_scale, n)
            a_perturbed = proj_simplex(a + noise)
            
            M_pert = fast_conv(a_perturbed).max()
            if M_pert < best_M - 1e-14:
                a = a_perturbed
                best_M = M_pert
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        it += 1
        
        if no_improve > 50:
            # Reset with stronger perturbation
            a = proj_simplex(a + rng.normal(0, 0.05, n))
            no_improve = 0
    
    return best_a


def run_mirror_descent(a_init, max_iters, deadline, step_init=0.3):
    """Mirror descent with KL geometry (multiplicative weights)."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient at active index
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * a[k_star - i]
        
        grad_tan = grad - grad.mean()
        
        # Exponential tilting
        step = step_init / (1.0 + 0.02 * it)
        log_weights = -step * grad_tan
        log_weights -= log_weights.max()
        a_new = a * np.exp(log_weights)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()
    
    return best_a


def coordinate_optimize_1d(a, i, b, n):
    """Find optimal value for a[i] given all others fixed."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.3)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(30):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Run coordinate descent sweeps."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_optimize_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - a[i] * a[i])
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - a[i])
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 3
        if no_improve > 30:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.05, n), 1e-6)
        inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Parabolic (concave)
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())
    
    # U-shaped (convex)
    poly2 = ((2 * t - 1) ** 2) + 0.1
    inits.append(poly2 / poly2.sum())
    
    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Spectral init: try to flatten autocorrelation
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            corr[i] = a_spec[k_star - i]
        a_spec = a_spec - 0.08 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):

    def update(self, a):

        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_combined_optimization(a0, n, deadline, seed):
    """Run multi-stage optimization on a single candidate."""
    best = np.maximum(a0.astype(np.float64), 0.0)
    s = best.sum()
    if s < 1e-10:
        best = np.ones(n) / n
    else:
        best = best / s
    
    best_M = fast_conv(best).max()
    t_start = time.monotonic()
    total_time = max(1, deadline - t_start)
    
    # Stage 1: FISTA (30% of time)
    t1 = t_start + 0.3 * total_time
    if time.monotonic() < t1:
        a_f = run_fista(best, 50, 300, 2000, t1)
        M_f = fast_conv(a_f).max()
        if M_f < best_M:
            best_M = M_f
            best = a_f.copy()
        print(f"    FISTA: M={best_M:.10f}", flush=True)
    
    # Stage 2: Game descent (40% of time)
    t2 = t_start + 0.7 * total_time
    if time.monotonic() < t2:
        remaining = t2 - time.monotonic()
        n_iters = max(50, int(remaining / 0.05))
        a_g = run_game_descent(best, t2, n_lp_iters=n_iters // 10,
                               n_sg_iters=n_iters // 10, n_langevin_iters=n_iters // 10)
        M_g = fast_conv(a_g).max()
        if M_g < best_M:
            best_M = M_g
            best = a_g.copy()
        print(f"    GameDescent: M={best_M:.10f}", flush=True)
    
    # Stage 3: Mirror descent (15% of time)
    t3 = t_start + 0.85 * total_time
    if time.monotonic() < t3:
        a_md = run_mirror_descent(best, 3000, t3, step_init=0.15)
        M_md = fast_conv(a_md).max()
        if M_md < best_M:
            best_M = M_md
            best = a_md.copy()
        print(f"    Mirror: M={best_M:.10f}", flush=True)
    
    # Stage 4: Coordinate descent (15% of time)
    if time.monotonic() < deadline:
        max_sw = max(5, int((deadline - time.monotonic()) / max(0.1, n / 80.0)))
        a_cd = run_coordinate_descent(best, max_sw, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
        print(f"    CD: M={best_M:.10f}", flush=True)
    
    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining FISTA, game descent (LP+geometric averaging),
    mirror descent, and coordinate descent across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 40.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768, J={tracker.best_score:.8f}", flush=True)
    
    n_parent = parent.size
    
    # Determine lengths to try
    lengths_to_try = [n_parent]
    for delta in [-128, -64, 64, 128, 256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths_to_try:
            lengths_to_try.append(nl)
    if 768 not in lengths_to_try:
        lengths_to_try.append(768)
    if 512 not in lengths_to_try:
        lengths_to_try.append(512)
    
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths_to_try = lengths_to_try[:5]
    
    n_lengths = len(lengths_to_try)
    print(f"\nLengths to try: {lengths_to_try}", flush=True)
    
    # === Phase 1: Multi-length FISTA + quick optimization ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length search", flush=True)
    print(f"{'='*60}", flush=True)
    
    time_p1 = min(350, (deadline - time.monotonic()) * 0.3)
    time_per_length = time_p1 / n_lengths
    length_best = {}
    
    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 600:
            break
        t0 = time.monotonic()
        time_alloc = min(time_per_length, (deadline - 600) - time.monotonic())
        if time_alloc < 20:
            break
        
        print(f"\n  n={n}, time={time_alloc:.0f}s", flush=True)
        inits = generate_inits(n, rng, parent)
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        # Use up to 4 inits
        n_inits = min(4, len(inits))
        time_per_init = time_alloc / n_inits
        
        for ii in range(n_inits):
            if time.monotonic() > deadline - 600:
                break
            sub_dl = t0 + (ii + 1) * time_per_init
            sub_dl = min(sub_dl, deadline - 600)
            if time.monotonic() >= sub_dl:
                break
            
            a0 = np.maximum(inits[ii], 1e-6)
            a0 = a0 / a0.sum()
            
            a_opt = run_combined_optimization(a0, n, sub_dl, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best J={J_val:.6f} (elapsed={time.monotonic()-t0:.0f}s)", flush=True)
            tracker.update(best_for_n)
            length_best[n] = best_for_n
    
    tracker.report("After Phase 1")
    
    # === Phase 2: Intensive game descent on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive game descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t2_deadline = min(deadline - 200, time.monotonic() + 400)
        a_gd = run_game_descent(tracker.best, t2_deadline,
                                n_lp_iters=300, n_sg_iters=300, n_langevin_iters=200)
        tracker.update(a_gd)
        print(f"  Game descent done, elapsed={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    tracker.report("After Phase 2")
    
    # === Phase 3: FISTA refinement with high tau ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 100:
        for tau in [300, 800, 1500]:
            if time.monotonic() > deadline - 100:
                break
            t_sub = min(deadline - 100, time.monotonic() + 60)
            a_f = run_fista(tracker.best, tau, tau * 2, 5000, t_sub)
            tracker.update(a_f)
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Coordinate descent polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t4 = min(deadline - 50, time.monotonic() + 150)
        n_best = tracker.best.size
        max_sweeps = max(10, int((t4 - time.monotonic()) / max(0.2, n_best / 60.0)))
        a_cd = run_coordinate_descent(tracker.best, max_sweeps, t4, seed=77777)
        tracker.update(a_cd)
    
    tracker.report("After Phase 4")
    
    # === Phase 5: Final LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t5 = min(deadline - 5, time.monotonic() + 80)
        # Run game descent again with focus on LP
        a_final = run_game_descent(tracker.best, t5,
                                   n_lp_iters=100, n_sg_iters=50, n_langevin_iters=20)
        tracker.update(a_final)
        
        # One more coordinate descent
        if time.monotonic() < deadline - 3:
            a_cd2 = run_coordinate_descent(a_final, 20, deadline - 1, seed=88888)
            tracker.update(a_cd2)
    
    tracker.report("After Phase 5")
    
    # === Phase 6: Try nearby lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_cur = tracker.best.size
        for mult in [1.2, 0.8, 1.5]:
            if time.monotonic() > deadline - 20:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 5, time.monotonic() + 40)
            a_opt = run_combined_optimization(a_ext, new_n, sub_dl, seed=seed + 555)
            tracker.update(a_opt)
    
    tracker.report("After Phase 6")
    
    # Final safety
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
