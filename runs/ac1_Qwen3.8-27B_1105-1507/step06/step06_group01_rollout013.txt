"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Full-constraint LP best-response mirror descent (all 2n-1 constraints)
2. Exact ternary line search for optimal blending
3. Nesterov-accelerated smooth max (log-sum-exp) refinement
4. Active-set QP via CVXPY
5. Coordinate descent polish
6. Multi-length, multi-start search
Key improvement: using ALL convolution constraints in LP (vs top-k in previous),
giving a much tighter best-response direction.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """FFT-based linear cross-convolution a*b."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def full_lp_best_response(a, M, timeout=8.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for ALL k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    conv_len = 2 * n - 1

    # Build sparse A: A[k,j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            idx = k - j
            if a[idx] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])

    A = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': timeout, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-10,
                             'primal_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, 0.0


def ternary_blend(a, g, n_eval_max=50):
    """Find optimal t in [0,1] minimizing max(conv((1-t)a+tg, (1-t)a+tg)).
    Uses ternary search on the 1D convex function.
    """
    ca = fast_conv(a)
    cg = fast_conv(g)
    cag = cross_conv(a, g)

    def obj(t):
        val = (1.0 - t) ** 2 * ca + 2.0 * t * (1.0 - t) * cag + t ** 2 * cg
        return val.max()

    lo = 0.0
    hi = 1.0
    for _ in range(n_eval_max):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def run_lp_mirror_descent(a_init, max_iters, deadline, n_active_full=True):
    """LP mirror descent with full or top-k constraints and exact line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        g, g_sum = full_lp_best_response(a, M, timeout=8.0)
        if g is None:
            no_improve += 3
            if no_improve > 10:
                break
            continue

        # Exact line search
        t_opt = ternary_blend(a, g)
        a_trial = (1.0 - t_opt) * a + t_opt * g
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            no_improve += 1
            continue
        a_trial = a_trial / s_t
        M_trial = fast_conv(a_trial).max()

        if M_trial < best_M - 1e-14:
            best_M = M_trial
            best_a = a_trial.copy()
            a = a_trial.copy()
            no_improve = 0
            if it % 5 == 0:
                print(f"    LP-iter {it}: M={best_M:.10f} t={t_opt:.4f} g_sum={g_sum:.4f}", flush=True)
        else:
            # Try smaller step
            improved = False
            for t_try in [t_opt * 0.5, t_opt * 0.25, t_opt * 0.1, 0.01]:
                a_trial2 = (1.0 - t_try) * a + t_try * g
                a_trial2 = np.maximum(a_trial2, 0.0)
                s_t2 = a_trial2.sum()
                if s_t2 < 1e-10:
                    continue
                a_trial2 = a_trial2 / s_t2
                M_trial2 = fast_conv(a_trial2).max()
                if M_trial2 < best_M - 1e-14:
                    best_M = M_trial2
                    best_a = a_trial2.copy()
                    a = a_trial2.copy()
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
                if no_improve > 15:
                    break

    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) via log-sum-exp and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    smooth_val = b_max + np.log(exp_sum) / tau

    # Gradient: d/d a[i] of (a*a)[k] = 2*a[k-i]
    # grad[i] = 2 * sum_k p[k] * a[k-i]
    grad = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        if k_start < 2 * n - 1:
            grad[i] = 2.0 * np.dot(p[k_start:k_end], a[:k_end - k_start])
    return smooth_val, grad


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def run_fista_smooth(a_init, tau_start, tau_end, max_iters, deadline, lr=0.3):
    """FISTA on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()

    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac

        _, grad_y = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()

        step = lr / (1.0 + 0.01 * tau * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        it += 1

        if it % 10 == 0:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def active_set_qp(a_init, max_iters, deadline, initial_active=None):
    """Active-set QCQP: minimize t s.t. (a*a)[k] <= t for active k, sum(a)=1, a>=0."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    if initial_active is None:
        b = fast_conv(a)
        initial_active = list(np.argsort(b)[::-1][:10])

    active = list(initial_active)

    for iteration in range(max_iters):
        if time.monotonic() > deadline:
            break

        a_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, a_var >= 0]

        for k in active:
            # Build Q_k: Q_k[i,j] = 1 if i+j == k
            # a^T Q_k a = sum_{i+j=k} a[i]*a[j] = (a*a)[k]
            # Use cp.quad_form or equivalent
            # For efficiency, build sparse Q_k
            ii = []
            jj = []
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    ii.append(i)
                    jj.append(j)
            if ii:
                Q = csr_matrix((np.ones(len(ii)), (ii, jj)), shape=(n, n))
                constraints.append(cp.quad_form(a_var, Q) <= t_var)

        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iter=5000, abstol=1e-9, reltol=1e-9)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-8)
            except Exception:
                break

        if a_var.value is None:
            break

        a_new = np.maximum(a_var.value, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new.copy()

        # Check for violated constraints and add to active set
        b_new = fast_conv(a_new)
        max_active_val = max(b_new[k] for k in active) if active else 0
        for k in range(2 * n - 1):
            if k not in active and b_new[k] > max_active_val + 1e-8:
                active.append(k)
                if len(active) > 50:
                    # Remove the least binding constraint
                    active.sort(key=lambda k: -b_new[k])
                    active = active[:40]
        if len(active) > 60:
            active = list(np.argsort(b_new)[::-1][:40])

    return best_a


def coordinate_descent_1d(a, i, b, n):
    """Find optimal value for a[i] given all others."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.5)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Run coordinate descent sweeps."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                delta = x_opt - a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2.0 * x_opt * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * delta
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 2
        if no_improve > 15:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for noise_std in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, noise_std, n), 1e-4)
            inits.append(p_n / p_n.sum())

    inits.append(np.ones(n) / n)

    # Symmetric structures
    for alpha in [2, 4, 8]:
        g = np.exp(-alpha * (t - 0.5) ** 2) + 0.01
        inits.append(g / g.sum())

    # Cosine windows
    for freq in [1, 2, 3]:
        cw = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cw = np.maximum(cw, 0.01)
        inits.append(cw / cw.sum())

    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Triangle
    tri = np.minimum(t, 1 - t) * 4 + 0.1
    inits.append(tri / tri.sum())

    # Random with spectral shaping
    a_spec = rng.uniform(0.5, 1.5, n)
    a_spec /= a_spec.sum()
    for _ in range(10):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.05 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.01)
        a_spec /= a_spec.sum()
    inits.append(a_spec.copy())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)

    lengths = sorted(set([n_parent, 640, 768, 896, 1024]))
    lengths = [l for l in lengths if 256 <= l <= 2048]

    t_phase1_end = time.monotonic() + (deadline - time.monotonic()) * 0.4
    time_per_length = max(30, (t_phase1_end - time.monotonic()) / len(lengths))

    for n in lengths:
        if time.monotonic() > t_phase1_end:
            break
        print(f"\n  Processing n={n}...", flush=True)
        t0 = time.monotonic()
        t_end = min(t_phase1_end, time.monotonic() + time_per_length)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_for_n = float('inf')

        time_per_init = max(5, (t_end - time.monotonic()) / max(1, len(inits)))

        for ii, a0 in enumerate(inits):
            if time.monotonic() > t_end:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            sub_deadline = min(t_end, time.monotonic() + time_per_init)

            # Run LP mirror descent
            max_iters = max(5, int((sub_deadline - time.monotonic()) / 3.0))
            a_opt = run_lp_mirror_descent(a0, max_iters, sub_deadline)
            M_opt = fast_conv(a_opt).max()

            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best J={J_val:.6f} (elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("After Phase 1")

    # === Phase 2: Intensive LP on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive LP on best", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        t_phase2_end = time.monotonic() + (deadline - time.monotonic()) * 0.25
        a_lp = run_lp_mirror_descent(tracker.best, 200, t_phase2_end)
        tracker.update(a_lp)
        print(f"  After LP: J={tracker.best_score:.8f}", flush=True)

    tracker.report("After Phase 2")

    # === Phase 3: FISTA refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        t_phase3_end = time.monotonic() + (deadline - time.monotonic()) * 0.15
        for tau in [100, 300, 600]:
            if time.monotonic() > t_phase3_end:
                break
            a_f = run_fista_smooth(tracker.best, tau, tau * 2, 3000,
                                   min(t_phase3_end, time.monotonic() + 40), lr=0.2)
            tracker.update(a_f)

    tracker.report("After Phase 3")

    # === Phase 4: Coordinate descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        t_phase4_end = min(deadline - 30, time.monotonic() + (deadline - time.monotonic()) * 0.1)
        max_sweeps = max(5, int((t_phase4_end - time.monotonic()) / max(1.0, tracker.best.size / 80.0)))
        a_cd = run_coordinate_descent(tracker.best, max_sweeps, t_phase4_end, seed=99999)
        tracker.update(a_cd)

    tracker.report("After Phase 4")

    # === Phase 5: Final LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        t_phase5_end = deadline - 5
        a_final = run_lp_mirror_descent(tracker.best, 100, t_phase5_end)
        tracker.update(a_final)

    tracker.report("After Phase 5")

    # === Final safety ===
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
