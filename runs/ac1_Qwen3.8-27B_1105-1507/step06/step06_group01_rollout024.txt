def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def fft_smooth_grad_fast(a, b, tau):
    """Optimized smooth max gradient using FFT correlation."""
    n = len(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau

    # Use FFT to compute c[i] = sum_j p[i+j]*a[j]
    # c[i] = sum_j a[j] * p[i+j]
    # This is the cross-correlation of a and p
    # Using: ifft(conj(fft(a, L)) * fft(p, L)) gives circular cross-correlation
    # corr[i] = sum_j a[j] * p[j+i] (for real signals, this is what we want)
    # Wait: the standard FFT cross-correlation:
    # R_ab[i] = sum_j a[j] * conj(b[j-i]) = sum_j a[j+i] * conj(b[j])
    # For real: R_ab[i] = sum_j a[j+i] * b[j]
    # In FFT: R_ab = ifft(fft(a) * conj(fft(b)))
    # So R_ab[i] = sum_j a[j+i] * b[j]
    # We want c[i] = sum_j a[j] * p[i+j] = sum_j p[j+i] * a[j]
    # This is R_pa[i] where R_pa[i] = sum_j p[j+i] * a[j]
    # R_pa = ifft(fft(p) * conj(fft(a)))
    
    L = 1
    while L < 2 * n - 1 + n:  # L >= 3n-1 to avoid wrap
        L <<= 1
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(P * np.conj(A), L)
    c = corr[:n]
    
    return smooth_val, 2.0 * c


def verify_fft_grad(a, b, tau):

def run_subgradient_multi_peak(a_init, deadline, max_iters=5000):
    """Subgradient method using top-K peaks simultaneously."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Use multiple peaks for better gradient information
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Get top-K peaks
        K = min(5, len(b))
        top_k = np.argsort(b)[::-1][:K]
        
        # Weighted subgradient from top peaks
        weights = b[top_k] - b[top_k[-1]] + 1.0  # emphasize higher peaks
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * a[j]
        
        grad_tan = grad - grad.mean()
        
        # Adaptive step size
        step = 1.0 / (np.sqrt(it + 1) * np.sqrt(n))
        step = min(step, 0.5)
        
        a_new = a - step * grad_tan
        a_new = proj_simplex(a_new)
        a = a_new
        
        if it % 5 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def lp_best_response(a, M, active_indices):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = int(k) - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_adaptive(a_init, deadline, max_iters=200):
    """LP mirror descent with adaptive active set."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        
        # Adaptive active set: all indices where b[k] > 0.7 * max(b)
        threshold = 0.7 * best_M
        active = np.where(b >= threshold)[0].tolist()
        
        if len(active) < 5:
            # Use top-10
            active = np.argsort(b)[::-1][:10].tolist()
        elif len(active) > 50:
            # Limit to top 50
            active = np.argsort(b)[::-1][:50].tolist()
        
        g = lp_best_response(a, best_M, active)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_a

        M = b.max()
        
        # Find top peaks
        K = min(10, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:K]
        
        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        for k in top_k:
            k_int = int(k)
            # Contribution of a[i] to b[k] is proportional to a[k-i]
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    pressure[i] += a[j]
        
        # Normalized pressure
        p_mean = pressure.mean()
        p_std = pressure.std() + 1e-10
        pressure_norm = (pressure - p_mean) / p_std
        
        # Multiplicative update (mirror descent with KL geometry)
        eta = 0.3 / (1.0 + 0.01 * it)
        a = a * np.exp(-eta * pressure_norm)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M - 1e-12:
            best_M = M_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Restart with perturbation
            a = best_a + rng_global.normal(0, 0.01, n)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
    
    return best_a


def coordinate_descent_block(a_init, deadline, block_size=16):
    """Block coordinate descent: optimize blocks of coordinates at a time."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    n_blocks = n // block_size
    if n_blocks == 0:
        n_blocks = 1
        block_size = n
    
    order = np.arange(n_blocks)
    
    for sweep in range(10000):
        if time.monotonic() > deadline:
            break
        rng_global.shuffle(order)
        
        b = fast_conv(a)
        improved = False
        
        for bi in order:
            lo = bi * block_size
            hi = min(lo + block_size, n)
            
            # Optimize a[lo:hi] while keeping others fixed
            # Use a few iterations of projected gradient on the block
            a_block = a[lo:hi].copy()
            a_fixed = a.copy()
            
            for _ in range(5):
                b_cur = fast_conv(a_fixed)
                # Gradient wrt a[lo:hi]
                grad_block = np.zeros(hi - lo)
                for i in range(lo, hi):
                    for k in range(max(i, 0), min(2*n-1, i+n)):
                        j = k - i
                        if 0 <= j < n:
                            grad_block[i-lo] += 2.0 * a_fixed[j] * (b_cur[k] == b_cur.max())
                
                a_block = np.maximum(a_block - 0.01 * grad_block, 0.0)
                a_fixed[lo:hi] = a_block
            
            # Renormalize
            a_fixed = np.maximum(a_fixed, 0.0)
            s_f = a_fixed.sum()
            if s_f > 1e-10:
                a_fixed = a_fixed / s_f
            
            M_cur = fast_conv(a_fixed).max()
            if M_cur < best_M - 1e-12:
                a = a_fixed.copy()
                best_M = M_cur
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp (exploit pairing structure)
    ramp = np.linspace(1.0, 0.1, n)
    inits.append(ramp / ramp.sum())
    
    # Reverse ramp
    ramp2 = np.linspace(0.1, 1.0, n)
    inits.append(ramp2 / ramp2.sum())
    
    # Two-sided: high in middle, low at edges
    mid = np.exp(-2.0 * (t - 0.5) ** 2 * 4.0) + 0.05
    inits.append(mid / mid.sum())
    
    # Flat middle, tapered edges
    taper = np.ones(n)
    edge = n // 8
    taper[:edge] = np.linspace(0.3, 1.0, edge)
    taper[-edge:] = np.linspace(1.0, 0.3, edge)
    inits.append(taper / taper.sum())
    
    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.exp(-0.5 * (np.arange(-ksize//2, ksize//2+1))**2 / (ksize//4)**2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # "Anti-peak" init: try to concentrate where convolution contributions cancel
    # Make a[j] * a[n-1-j] small for all j
    # Simple approach: make first half large, second half small
    half = n // 2
    asym = np.ones(n)
    asym[:half] = 2.0
    asym[half:] = 0.5
    inits.append(asym / asym.sum())
    
    # Periodic structure
    for period in [n//4, n//3]:
        if period >= 2:
            per = 1.0 + 0.3 * np.sin(2 * np.pi * t * n / period)
            per = np.maximum(per, 0.01)
            inits.append(per / per.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


# Global RNG for reproducibility
rng_global = np.random.default_rng(12345)


def optimize_single(a0, deadline, n=None):
    """Run multi-method optimization on a single initial point."""
    if n is None:
        n = len(a0)
    
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best = a.copy()
    best_M = fast_conv(a).max()
    
    methods = [
        ("fista", lambda: run_fista_fftfasta(best, deadline)),
        ("peak_shave", lambda: run_peak_shaving(best, deadline)),
        ("lp", lambda: run_lp_adaptive(best, deadline)),
        ("subgrad", lambda: run_subgradient_multi_peak(best, deadline)),
    ]
    
    # Allocate time: try each method for a portion of the deadline
    time_per_method = 0.25 * (deadline - time.monotonic())
    
    for name, method_fn in methods:
        if time.monotonic() > deadline:
            break
        sub_deadline = min(deadline, time.monotonic() + max(5, time_per_method))
        try:
            result = method_fn()
            M_r = fast_conv(result).max()
            if M_r < best_M:
                best_M = M_r
                best = result.copy()
        except Exception as e:
            pass
    
    return best


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-phase search: multi-scale init, diverse optimization, LP polish.
    """
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)
        print(f"No parent, using uniform n=512", flush=True)
    
    n_parent = parent.size
    
    # === Phase 1: Multi-scale warmup ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-scale warmup", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Optimize at smaller scales first, then upscale
    small_n = min(256, n_parent)
    a_small = resample_seq(parent, small_n)
    a_small = np.maximum(a_small, 1e-6)
    a_small = a_small / a_small.sum()
    
    t_small_deadline = min(deadline - 600, time.monotonic() + 60)
    a_small_opt = run_fista_fftfasta(a_small, t_small_deadline)
    
    # Upscale to parent size
    a_scaled = resample_seq(a_small_opt, n_parent)
    a_scaled = np.maximum(a_scaled, 1e-6)
    a_scaled = a_scaled / a_scaled.sum()
    tracker.update(a_scaled)
    
    # === Phase 2: Multi-start at target length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-start optimization (n={n_parent})", flush=True)
    print(f"{'='*60}", flush=True)
    
    inits = generate_inits(n_parent, rng_global, tracker.best)
    
    # Also try a few different lengths
    alt_lengths = [n_parent]
    for delta in [-64, 64, -128, 128, 256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl % 32 == 0:
            alt_lengths.append(nl)
    
    # Time budget for phase 2
    phase2_deadline = min(deadline - 200, time.monotonic() + 0.5 * (deadline - time.monotonic()))
    
    total_inits = len(inits) * len(alt_lengths)
    time_per_init = max(5, (phase2_deadline - time.monotonic()) / max(1, total_inits))
    
    for n_try in alt_lengths:
        if time.monotonic() > phase2_deadline:
            break
        
        if n_try != n_parent:
            inits_try = generate_inits(n_try, rng_global, tracker.best)
        else:
            inits_try = inits
        
        for a0 in inits_try:
            if time.monotonic() > phase2_deadline:
                break
            
            sub_dl = min(phase2_deadline, time.monotonic() + time_per_init)
            if sub_dl - time.monotonic() < 3:
                break
            
            a_opt = optimize_single(a0, sub_dl)
            tracker.update(a_opt)
    
    tracker.report("After Phase 2")
    
    # === Phase 3: Intensive LP on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 60:
        phase3_dl = min(deadline - 50, time.monotonic() + 0.25 * (deadline - time.monotonic()))
        a_lp = run_lp_adaptive(tracker.best, phase3_dl, max_iters=100)
        tracker.update(a_lp)
        tracker.report("After LP")
    
    # === Phase 4: Peak shaving refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Peak shaving refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        phase4_dl = min(deadline - 20, time.monotonic() + 0.15 * (deadline - time.monotonic()))
        a_ps = run_peak_shaving(tracker.best, phase4_dl)
        tracker.update(a_ps)
        tracker.report("After peak shaving")
    
    # === Phase 5: FISTA with high tau ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: High-tau FISTA", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 15:
        phase5_dl = min(deadline - 10, time.monotonic() + 0.1 * (deadline - time.monotonic()))
        a_f = run_fista_fftfasta(tracker.best, phase5_dl, max_iters=20000)
        tracker.update(a_f)
        tracker.report("After FISTA")
    
    # === Phase 6: Final LP + coordinate polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 5:
        a_final = run_lp_adaptive(tracker.best, deadline - 3, max_iters=50)
        tracker.update(a_final)
    
    # === Final check ===
    if tracker.best is None:
        tracker.best = np.ones(512) / 512
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
