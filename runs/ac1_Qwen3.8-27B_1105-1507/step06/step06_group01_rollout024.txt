def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def run_lbfgs_softmax(w_init, tau, max_iters, deadline):
    """L-BFGS-B optimization in softmax parameterization."""
    n = len(w_init)
    
    def objective(w):
        val, _ = smooth_max_conv(w, tau)
        return val
    
    def grad(w):
        _, g = smooth_max_conv(w, tau)
        return g
    
    result = minimize(objective, w_init, jac=grad, method='L-BFGS-B',
                      options={'maxiter': max_iters, 'maxfun': max_iters * 3,
                               'gtol': 1e-8, 'maxls': 50})
    
    a = np.exp(result.x - result.x.max())
    a = a / a.sum()
    return a, result.fun, result

def active_set_qp(a_init, n_active, deadline):
    """Solve QP: min t s.t. conv(a,a)[k] <= t for k in active, sum(a)=1, a>=0."""
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    
    b = fast_conv(a)
    M = b.max()
    top_idx = np.argsort(b)[::-1][:n_active]
    
    # Build constraint functions
    # Variables: [a_0, ..., a_{n-1}, t] of length n+1
    # Objective: minimize t (last variable)
    # Constraints: conv[k] - t <= 0 for k in top_idx
    #              sum(a) = 1 (equality)
    #              a_i >= 0 (bounds)
    
    def obj(x):
        return x[-1]  # minimize t
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    constraints = []
    for k in top_idx:
        def make_ineq(k_val):
            def ineq(x):
                a_x = x[:n]
                # Compute conv[k_val] for this a_x
                # conv[k] = sum_j a[j] * a[k-j] for valid j
                j_min = max(0, k_val - n + 1)
                j_max = min(n - 1, k_val)
                if j_min > j_max:
                    return x[-1]  # constraint is trivially satisfied
                terms = a_x[j_min:j_max+1] * a_x[k_val - j_max:k_val - j_min]
                return x[-1] - terms.sum()
            def ineq_grad(x):
                g = np.zeros(n + 1)
                g[-1] = 1.0
                j_min = max(0, k_val - n + 1)
                j_max = min(n - 1, k_val)
                if j_min <= j_max:
                    for j in range(j_min, j_max + 1):
                        other = k_val - j
                        g[j] -= a_x[other]
                        if j != other:
                            g[other] -= a_x[j]
                return g
            return ineq, ineq_grad
        
        ineq_func, ineq_grad_func = make_ineq(k)
        constraints.append({'type': 'ineq', 'fun': ineq_func, 'jac': ineq_grad_func})
    
    # Equality constraint: sum(a) = 1
    constraints.append({
        'type': 'eq',
        'fun': lambda x: x[:n].sum() - 1.0,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    })
    
    x0 = np.concatenate([a, [M * 1.01]])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-10})
        if result.success:
            a_opt = np.maximum(result.x[:n], 0)
            s = a_opt.sum()
            if s > 1e-10:
                a_opt = a_opt / s
                M_opt = fast_conv(a_opt).max()
                return a_opt, M_opt
    except Exception:
        pass
    
    return None, None

def pair_coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Optimize pairs of coordinates simultaneously using golden section search."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Pick random pairs
        n_pairs = max(50, n // 4)
        i_indices = rng.integers(0, n, size=n_pairs)
        j_indices = rng.integers(0, n, size=n_pairs)
        # Ensure i != j
        same = i_indices == j_indices
        j_indices[same] = (j_indices[same] + 1) % n
        
        sweep_improved = False
        for idx in range(n_pairs):
            i, j = int(i_indices[idx]), int(j_indices[idx])
            if i == j:
                continue
            
            a_i, a_j = a[i], a[j]
            
            # Golden section search over (delta_i, delta_j)
            # We search over a line: a_i -> a_i + t*di, a_j -> a_j + t*dj
            # Choose direction to reduce the max convolution
            b = fast_conv(a)
            k_max = int(np.argmax(b))
            
            # Direction: reduce a at positions that contribute to k_max
            di = -a[(k_max - i) % n] if 0 <= k_max - i < n else 0.0
            dj = -a[(k_max - j) % n] if 0 <= k_max - j < n else 0.0
            
            norm = np.sqrt(di*di + dj*dj)
            if norm < 1e-10:
                continue
            di, dj = di/norm, dj/norm
            
            # Golden section search for t in [t_min, t_max]
            t_min = -max(a_i, a_j) * 0.9
            t_max = 1.0
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            t1 = t_max - gr * (t_max - t_min)
            t2 = t_min + gr * (t_max - t_min)
            
            def eval_at(t):
                a[i] = max(0, a_i + t * di)
                a[j] = max(0, a_j + t * dj)
                b_curr = fast_conv(a)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for noise_level in [0.01, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p + rng.normal(0, noise_level, n), 1e-5)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangular (peaked in middle)
    tri = 1.0 - np.abs(2 * t - 1)
    inits.append(tri / tri.sum())
    
    # Exponential decay
    exp_dec = np.exp(-3 * t)
    inits.append(exp_dec / exp_dec.sum())
    
    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cos_w / cos_w.sum())
    
    # Anti-symmetric bias (large at start, small at end) to reduce center peak
    anti = np.exp(-2 * t) + 0.1 * np.exp(2 * (t - 1))
    inits.append(anti / anti.sum())
    
    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01) / np.maximum(smooth, 0.01).sum())
    
    return inits

def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(w):
    """Numerically stable softmax."""
    w_shift = w - w.max()
    e = np.exp(w_shift)
    return e / e.sum()

            grad_a[i] = 2.0 * tau * np.dot(a[:max_m], p[i:i + max_m])
    
    # Chain rule: a = softmax(w)
    # grad_w[j] = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])
    dot_val = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot_val)
    
    return log_sum, grad_w


def run_lbfgs_softmax(w_init, tau_seq, deadline, max_total_iters=5000):
    """L-BFGS-B with adaptive tau schedule in softmax space."""
    n = len(w_init)
    w = w_init.copy()
    best_a = softmax(w)
    best_M = fast_conv(best_a).max()
    best_w = w.copy()
    total_iters = 0
    
    for tau in tau_seq:
        if time.monotonic() > deadline:
            break
        
        sub_deadline = min(deadline, time.monotonic() + max(5, (deadline - time.monotonic()) * 0.3))
        max_iter = max(100, int((sub_deadline - time.monotonic()) * 5))
        max_iter = min(max_iter, max_total_iters - total_iters)
        if max_iter <= 0:
            break
        
        def objective(w_curr):
            val, _ = lse_topk_conv_grad(w_curr, tau, k_top=8)
            return val
        
        def grad_func(w_curr):
            _, g = lse_topk_conv_grad(w_curr, tau, k_top=8)
            return g
        
        try:
            result = minimize(objective, w, jac=grad_func, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'gtol': 1e-7, 'maxls': 40})
            w = result.x
            total_iters += result.niter
        except Exception:
            break
        
        a_curr = softmax(w)
        M_curr = fast_conv(a_curr).max()
        if M_curr < best_M:
            best_M = M_curr
            best_a = a_curr.copy()
            best_w = w.copy()
    
    return best_a, best_w, best_M


def run_subgradient(a_init, deadline, step_init=0.01, decay=0.9995):
    """Subgradient method with Polyak-type step size for the max-conv objective."""
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    step = step_init
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient: d max(b)/d a[i] = 2*a[k_star - i] if 0 <= k_star-i < n
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Project to tangent space (sum grad = 0 for simplex)
        grad_tan = grad - np.mean(grad)
        
        # Update with step size
        a_new = a - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a = a_new / s_new
        else:
            a = a_new / max(s_new, 1e-10)
        
        step *= decay
        
        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a


def active_set_qp_precise(a_init, n_active_list, deadline):
    """Iterative active set QP with increasing active set sizes."""
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for n_active in n_active_list:
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        top_idx = np.argsort(b)[::-1][:min(n_active, 2 * n - 1)]
        
        # SLSQP: minimize t subject to conv[k] <= t for k in top_idx, sum(a)=1, a>=0
        def obj(x):
            return x[-1]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g
        
        constraints = []
        for k_val in top_idx:
            j_min = max(0, k_val - n + 1)
            j_max = min(n - 1, k_val)
            if j_min > j_max:
                continue
            
            def make_ineq(kv, jm, jx):
                def ineq(x):
                    ax = x[:n]
                    terms = ax[jm:jx + 1] * ax[kv - jx:kv - jm]
                    return x[-1] - terms.sum()
                def ineq_grad(x):
                    g = np.zeros(n + 1)
                    g[-1] = 1.0
                    ax = x[:n]
                    for j in range(jm, jx + 1):
                        other = kv - j
                        g[j] -= ax[other]
                        if j != other:
                            g[other] -= ax[j]
                    return g
                return ineq, ineq_grad
            
            ineq_f, ineq_g = make_ineq(k_val, j_min, j_max)
            constraints.append({'type': 'ineq', 'fun': ineq_f, 'jac': ineq_g})
        
        constraints.append({
            'type': 'eq',
            'fun': lambda x: x[:n].sum() - 1.0,
            'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
        })
        
        x0 = np.concatenate([a, [M * 1.001]])
        bounds = [(0, None)] * n + [(None, None)]
        
        try:
            result = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 300, 'ftol': 1e-12})
            if result.success:
                a_opt = np.maximum(result.x[:n], 0)
                s_opt = a_opt.sum()
                if s_opt > 1e-10:
                    a_opt = a_opt / s_opt
                    M_opt = fast_conv(a_opt).max()
                    if M_opt < best_M - 1e-12:
                        best_M = M_opt
                        best_a = a_opt.copy()
                        a = a_opt.copy()
                        print(f"    QP n_active={n_active}: M={best_M:.12f}", flush=True)
                    else:
                        # No improvement, try next active set size
                        pass
        except Exception as e:
            print(f"    QP error: {e}", flush=True)
    
    return best_a


def pair_cd_refined(a_init, deadline, seed=0):
    """Pair coordinate descent with smart direction selection."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    gr_ratio = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(200):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_max = int(np.argmax(b))
        
        # Second max for tie-breaking
        b_sorted = np.sort(b)[::-1]
        
        n_pairs = max(100, n // 3)
        sweep_improved = False
        
        for _ in range(n_pairs):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                j = (j + 1) % n
            
            a_i, a_j = a[i], a[j]
            
            # Direction: reduce contribution to k_max
            contrib_i = a[(k_max - i) % n] if 0 <= k_max - i < n else 0.0
            contrib_j = a[(k_max - j) % n] if 0 <= k_max - j < n else 0.0
            
            di = -contrib_i
            dj = -contrib_j
            norm = np.sqrt(di * di + dj * dj)
            if norm < 1e-12:
                # Try opposite direction
                di, dj = contrib_i, contrib_j
                norm = np.sqrt(di * di + dj * dj)
                if norm < 1e-12:
                    continue
            di, dj = di / norm, dj / norm
            
            # Golden section on t in [-t_max, t_max]
            t_max = min(a_i, a_j) * 0.95 + 0.01
            t_lo = -t_max
            t_hi = t_max
            
            def eval_t(t_val):
                a[i] = max(0.0, a_i + t_val * di)
                a[j] = max(0.0, a_j + t_val * dj)
                return fast_conv(a).max()
            
            t1 = t_hi - gr_ratio * (t_hi - t_lo)
            t2 = t_lo + gr_ratio * (t_hi - t_lo)
            f1 = eval_t(t1)
            f2 = eval_t(t2)
            
            for _ in range(25):
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_hi - gr_ratio * (t_hi - t_lo)
                    f1 = eval_t(t1)
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_lo + gr_ratio * (t_hi - t_lo)
                    f2 = eval_t(t2)
            
            t_opt = (t_lo + t_hi) / 2.0
            a[i] = max(0.0, a_i + t_opt * di)
            a[j] = max(0.0, a_j + t_opt * dj)
            s_curr = a.sum()
            if s_curr > 1e-10:
                a = a / s_curr
            
            M_new = fast_conv(a).max()
            if M_new < best_M - 1e-13:
                best_M = M_new
                best_a = a.copy()
                sweep_improved = True
        
        if not sweep_improve:
            no_improve += 1
        else:
            no_improve = 0
        if no_improve > 8:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for noise in [0.005, 0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, noise, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Mild modulation to break symmetry
    mod = 1.0 + 0.05 * np.cos(2 * np.pi * t)
    inits.append(mod / mod.sum())
    
    # Two-frequency modulation
    mod2 = 1.0 + 0.03 * np.cos(2 * np.pi * t) + 0.02 * np.cos(4 * np.pi * t)
    inits.append(mod2 / mod2.sum())
    
    # Slight asymmetry
    asym = 1.0 + 0.02 * (t - 0.5)
    inits.append(np.maximum(asym, 0.1) / np.maximum(asym, 0.1).sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS-B in softmax space with adaptive tau (40% time)
    Phase 2: Subgradient with adaptive step (15% time)  
    Phase 3: Active set QP refinement (25% time)
    Phase 4: Pair coordinate descent (20% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent = parent / parent.sum()
            print(f"Parent: n={parent.size}, J={compute_J(parent):.10f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    best_overall = None
    best_score = float('inf')
    
    def track(a, label=""):
        nonlocal best_overall, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = compute_J(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_overall = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s {label}", flush=True)
    
    if parent is not None:
        track(parent, "(parent)")
    
    n_target = 768
    if parent is not None:
        n_target = parent.size
    
    print(f"\nTarget length: {n_target}", flush=True)
    
    # === Phase 1: L-BFGS-B in softmax space ===
    t_p1_end = t_start + 0.38 * (budget_s - 25)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: L-BFGS-B in softmax space (until t={0.38*(budget_s-25):.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    inits = generate_inits(n_target, rng, parent)
    
    # Allocate time across inits
    n_inits = len(inits)
    time_per_init = (t_p1_end - time.monotonic()) / max(1, n_inits)
    
    for ii, a0 in enumerate(inits):
        if time.monotonic() > t_p1_end:
            break
        time_left = t_p1_end - time.monotonic()
        if time_left < 10:
            break
        
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()
        w0 = np.log(a0 + 1e-15)
        
        # Adaptive tau schedule: start sharp, end broad
        tau_seq = [100, 50, 25, 15]
        sub_deadline = min(t_p1_end, time.monotonic() + min(120, time_left * 0.8))
        
        a_opt, _, M_opt = run_lbfgs_softmax(w0, tau_seq, sub_deadline)
        track(a_opt, f"(lbfgs init{ii})")
    
    print(f"  Phase 1 done: J={best_score:.10f}", flush=True)
    
    # === Phase 2: Subgradient refinement ===
    t_p2_end = t_start + 0.53 * (budget_s - 25)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Subgradient method", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_overall is not None:
        variants = generate_inits(n_target, rng, best_overall)
        time_per_var = (t_p2_end - time.monotonic()) / max(1, min(6, len(variants)))
        
        for ii, a0 in enumerate(variants[:6]):
            if time.monotonic() > t_p2_end:
                break
            sub_dl = min(t_p2_end, time.monotonic() + 30)
            a_opt = run_subgradient(a0, sub_dl, step_init=0.005, decay=0.9998)
            track(a_opt, f"(subgrad{ii})")
    
    print(f"  Phase 2 done: J={best_score:.10f}", flush=True)
    
    # === Phase 3: Active set QP ===
    t_p3_end = t_start + 0.78 * (budget_s - 25)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Active set QP (SLSQP)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_overall is not None:
        inits3 = generate_inits(n_target, rng, best_overall)
        time_per_qp = (t_p3_end - time.monotonic()) / max(1, min(5, len(inits3)))
        
        for ii, a0 in enumerate(inits3[:5]):
            if time.monotonic() > t_p3_end:
                break
            sub_dl = min(t_p3_end, time.monotonic() + min(80, time_per_qp))
            a_opt = active_set_qp_precise(a0, [8, 15, 25, 40], sub_dl)
            track(a_opt, f"(qp{ii})")
    
    print(f"  Phase 3 done: J={best_score:.10f}", flush=True)
    
    # === Phase 4: Pair coordinate descent ===
    t_p4_end = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Pair coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_overall is not None:
        inits4 = generate_inits(n_target, rng, best_overall)
        for ii, a0 in enumerate(inits4[:4]):

            sub_dl = min(t_p4_end, time.monotonic() + 30)
            a_opt = pair_cd_refined(a0, sub_dl, seed=seed + 200 + ii)
            track(a_opt, f"(paircd{ii})")
    
    print(f"  Phase 4 done: J={best_score:.10f}", flush=True)
    
    # === Final polish: alternate QP and L-BFGS-B ===
    if best_overall is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL POLISH", flush=True)
        print(f"{'='*60}", flush=True)
        
        a = best_overall.copy()
        for it in range(10):
            if time.monotonic() > deadline - 5:
                break
            # Quick QP
            a_q = active_set_qp_precise(a, [10, 20], min(deadline - 5, time.monotonic() + 20))
            track(a_q, f"(polish_qp{it})")
            a = a_q
            # Quick L-BFGS
            w = np.log(a + 1e-15)
            a_lb, _, _ = run_lbfgs_softmax(w, [30, 15], min(deadline - 5, time.monotonic() + 15))
            track(a_lb, f"(polish_lbfgs{it})")
            if compute_J(a_lb) < compute_J(a):
                a = a_lb
    
    # Final evaluation
    if best_overall is not None:
        final_list = [float(x) for x in best_overall]
        final_score = float(evaluate_sequence(final_list))
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL RESULT: evaluate_sequence={final_score:.10f} compute_J={compute_J(best_overall):.10f} n={best_overall.size}", flush=True)
        print(f"Total time: {time.monotonic()-t_start:.1f}s", flush=True)
        print(f"{'='*60}", flush=True)
        if final_score < best_score:
            best_score = final_score
    else:
        best_overall = np.ones(n_target)
    
    return [float(x) for x in best_overall]
