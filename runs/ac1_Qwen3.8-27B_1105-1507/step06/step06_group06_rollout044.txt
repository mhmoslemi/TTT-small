"""
Focused multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based subgradient with momentum, LP alternating, spectral flattening,
and condensed-explosion perturbations. Adaptive length selection and intensive
polishing on the best candidate.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution cross-product."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def subgradient_fft(a, tau=1.0):

    Compute smoothed subgradient of max(conv(a,a)) using FFT.
    Returns gradient array of shape (n,).
    """

    conv = compute_conv(a)

    shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # grad[i] = 2 * sum_k weights[k] * a[k-i] for 0 <= k-i < n

    while N_fft < len(weights) + n - 1:

    # Cross-correlation: corr[m] = sum_k w[k] * a[k - (m - n + 1)]
    # We want grad[i] = sum_k w[k] * a[k-i] for 0 <= k-i < n

    # corr[m] = sum_k w[k+m] * a[k]... need to be careful with indexing
    # Actually: ifft(FFT(w) * conj(FFT(a))) gives circular cross-correlation
    # corr[m] = sum_k w[(k+m) mod N] * a[k]
    # We want: grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # = corr[i] (with appropriate indexing)
    # Let's just compute it directly for correctness

    for i in range(n):

    # grad[i] = sum_{j=0}^{n-1} W_ext[i+j] * a[j]
    # This is the "diagonal" of the outer product... 
    # Actually this is: grad = first n elements of the auto-convolution trick
    # grad[i] = (W_ext * a)[n-1-i] ... no
    # Let me think: define B of length 2n-1 where B[k] = W_ext[k]
    # grad[i] = sum_{j=0}^{n-1} B[i+j] * a[j]
    # = sum_{k=i}^{i+n-1} B[k] * a[k-i]
    # This is a "sliding dot product"
    # Can compute via: create C of length 2n-1+n-1 = 3n-2
    # C[m] = sum_j B[m-j] * a[j] (convolution of B and reversed a)
    # Actually, let's just use the direct FFT approach
    # grad[i] = sum_j a[j] * W[i+j]
    # Create extended arrays and use FFT
    N = 1
    while N < 2 * n:
        N <<= 1
    # We want the "anti-diagonal" correlation
    # Alternative: pad a to length N, pad W to length N
    # Compute element-wise: for each i, dot(W[i:i+n], a[0:n])
    # This is like a "sliding window dot product"
    # Using FFT: create a_rev = a[::-1], then convolve W with a_rev
    # conv(W, a_rev)[i + n - 1] = sum_j W[i+j] * a_rev[n-1-j] = sum_j W[i+j] * a[n-1-(n-1-j)] = sum_j W[i+j]*a[j]
    # Hmm, that's not quite right either.
    # conv(W, a_rev)[k] = sum_j W[j] * a_rev[k-j] = sum_j W[j] * a[n-1-(k-j)]
    # For k = i + n - 1: sum_j W[j] * a[n-1-i-1-j+n-1]... getting confused.
    # Let me just use a simple FFT-based approach:
    # grad[i] = sum_j a[j] * W[i+j]
    # Define V of length 2n-1: V[k] = W[k]
    # Define U of length n: U[j] = a[j]
    # grad[i] = sum_{j=0}^{n-1} U[j] * V[i+j]
    # This is the cross-correlation at lag i (with V ahead of U)
    # cross_corr[i] = sum_j U[j] * V[i+j]
    # Using FFT: cross_corr = ifft(conj(FFT(U, N)) * FFT(V, N)) ... 
    # Actually: cross_corr[i] = sum_j U[j] * V[i+j]
    # = sum_j U[j] * conj(conj(V[i+j]))
    # Standard formula: ifft(FFT(U) * conj(FFT(V)))[i] = sum_j U[j] * V[j-i]
    # That's the "U ahead" correlation. We want "V ahead".
    # ifft(conj(FFT(U)) * FFT(V))[i] = sum_j conj(U[j]) * V[j+i] (for real: sum_j U[j]*V[j+i])
    # Wait no. Let me just use the definition:
    # R[m] = ifft(conj(FU) * FV)[m] where FU = FFT(U, N), FV = FFT(V, N)
    # R[m] = (1/N) sum_k conj(FU[k]) * FV[k] * exp(2*pi*i*m*k/N)
    # = (1/N) sum_k conj(sum_j U[j]*exp(-2*pi*i*j*k/N)) * (sum_l V[l]*exp(2*pi*i*l*k/N)) * exp(2*pi*i*m*k/N)
    # = sum_j V[j+m] * U[j]  (circular, with mod N)
    # So R[m] = sum_j U[j] * V[(j+m) mod N]
    # For m = i (0 <= i <= n-1) and N >= 2n-1:
    # R[i] = sum_j U[j] * V[(j+i) mod N] = sum_{j=0}^{n-1} a[j] * W[i+j] (since i+j < 2n-1 <= N)
    # Perfect!
    N = 1
    while N < 2 * n:
        N <<= 1
    FU = np.fft.fft(a, N)
    FV = np.fft.fft(weights, N)
    R = np.fft.ifft(np.conj(FU) * FV).real[:n]
    return 2.0 * R


def subgradient_momentum(a, max_iters=200, time_budget=30.0, tau_init=0.3, momentum=0.8):
    """Subgradient optimization with Nesterov momentum."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    step = 0.15

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        tau = tau_init + 0.02 * it
        grad = subgradient_fft_fast(a, tau=tau)
        grad_centered = grad - np.mean(grad)
        v = momentum * v - step * grad_centered
        a = project_nonneg_sum1(a - v)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        step *= 0.997

    return best_a, best_obj


def lp_alternate(a, max_iters=40, time_budget=60.0):
    """LP alternating with adaptive line search."""
    n = len(a)
    if n > 2500:
        return a, compute_obj(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        lp_time = min(remaining * 0.4, 12.0)
        g = solve_lp_max_sum(a, M, time_limit=lp_time)
        if g is None:
            break

        # Line search
        candidate_ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return best_a, best_obj


def spectral_flatten(a, max_iters=50, time_budget=20.0, alpha_start=0.8, alpha_end=0.95):
    """
    Iterative spectral flattening: modify the power spectrum to reduce
    time-domain peaks, then project back.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        progress = it / max(max_iters - 1, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        p_max = np.max(power)
        if p_max < 1e-20:
            break

        # Compress the power spectrum
        p_norm = power / p_max
        p_compressed = p_norm ** alpha
        # Reconstruct
        phase = np.angle(fa)
        magnitude = np.sqrt(p_compressed * p_max)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def spectral_compress_conv(a, max_iters=50, time_budget=20.0, blend=0.3):
    """
    Directly compress the convolution in the frequency domain,
    then extract new sequence via complex square root.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        c = np.fft.ifft(fa * fa).real[:2 * n - 1]
        c = np.maximum(c, 0)
        c_max = np.max(c)
        if c_max < 1e-15:
            break

        # Blend peaks toward median
        median_c = np.median(c[c > c_max * 0.1]) if np.sum(c > c_max * 0.1) > 10 else c_max * 0.3
        c_mod = c.copy()
        peak_mask = c > median_c * 1.3
        if np.any(peak_mask):
            c_mod[peak_mask] = c[peak_mask] * (1 - blend) + median_c * blend

        # Power compression
        c_scaled = c_mod / np.max(c_mod)
        c_mod = c_scaled ** 0.9 * np.max(c_mod)
        c_mod = np.maximum(c_mod, 0)

        # Extract via complex square root
        cf = np.fft.fft(c_mod, N)
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        if not np.all(np.isfinite(a_new)):
            break

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def condensed_explosion(a, rng, frac=0.1):
    """
    Condensed explosion: zero out low-magnitude elements in a random window,
    redistribute their mass to random positions.
    """
    n = len(a)
    a = a.copy()
    wsize = max(10, n // 5)
    wstart = rng.integers(0, n - wsize)
    window = a[wstart:wstart + wsize].copy()
    
    # Find low-magnitude elements in window
    threshold = np.median(window) * 0.5
    low_mask = window < threshold
    n_low = np.sum(low_mask)
    if n_low < 5:
        low_mask = window < np.max(window) * 0.7
        n_low = np.sum(low_mask)
    if n_low < 3:
        return a
    
    mass = np.sum(window[low_mask])
    a[wstart + np.where(low_mask)[0]] = 0
    
    # Redistribute to random positions (preferring high-magnitude areas)
    targets = rng.choice(n, size=min(n_low, 50), replace=False)
    a[targets] += mass / len(targets)
    
    return project_nonneg_sum1(a)


def random_perturb(a, rng, scale=0.05):
    """Multiplicative random perturbation."""
    n = len(a)
    pert = a * np.exp(rng.normal(0, scale, n))
    return project_nonneg_sum1(pert)


def window_noise(a, rng, wsize=None, noise_scale=0.2):
    """Add noise to a random window."""
    n = len(a)
    if wsize is None:
        wsize = max(10, n // 10)
    a = a.copy()
    wstart = rng.integers(0, n - wsize)
    orig = a[wstart:wstart + wsize]
    a[wstart:wstart + wsize] += rng.normal(0, noise_scale * (orig.mean() + 0.01), wsize)
    return project_nonneg_sum1(a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sinusoidal
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())
    
    # Gaussian (centered)
    gauss = np.exp(-((x - 0.5) ** 2) / 0.05)
    inits.append(gauss / gauss.sum())
    
    # U-shaped (edges)
    u_shape = (x - 0.5) ** 2 + 0.1
    inits.append(u_shape / u_shape.sum())
    
    # Bimodal
    bi = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    inits.append(bi / bi.sum())
    
    # W-shaped
    w_shape = np.exp(-((x - 0.2) ** 2) / 0.01) + np.exp(-((x - 0.5) ** 2) / 0.01) + np.exp(-((x - 0.8) ** 2) / 0.01)
    inits.append(w_shape / w_shape.sum())
    
    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())
    
    # Power law
    pw = (x + 0.01) ** 2
    inits.append(pw / pw.sum())
    inits.append(((1 - x) + 0.01) ** 2 + 0.1)
    inits[-1] = inits[-1] / inits[-1].sum()
    
    if parent is not None:
        pr = resample_sequence(parent, n)
        inits.append(pr)
        # Scaled version
        inits.append(project_nonneg_sum1(pr * (1 + 0.1 * np.sin(2 * np.pi * 3 * x))))
    
    return inits


def optimize_sequence(a, time_budget, rng, verbose=False):
    """
    Run a cycle of optimization strategies on a sequence.
    Returns (best_a, best_obj).
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    cycle_num = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        cycle_num += 1
        strat = (cycle_num - 1) % 4
        
        if strat == 0 and n <= 2000:
            # LP alternating (most powerful)
            t_alloc = min(remaining * 0.3, 40.0)
            a_new, obj_new = lp_alternate(a, max_iters=min(30, int(t_alloc)), time_budget=t_alloc)
        elif strat == 1:
            # Spectral flattening
            t_alloc = min(remaining * 0.25, 20.0)
            a_new, obj_new = spectral_flatten(a, max_iters=50, time_budget=t_alloc)
        elif strat == 2:
            # Subgradient with momentum
            t_alloc = min(remaining * 0.2, 25.0)
            a_new, obj_new = subgradient_momentum(a, max_iters=200, time_budget=t_alloc)
        else:
            # Convolution compression + perturbation
            t_alloc = min(remaining * 0.2, 15.0)
            a_new, obj_new = spectral_compress_conv(a, max_iters=40, time_budget=t_alloc)
            # Add perturbation
            a_pert = random_perturb(a_new, rng, scale=0.03)
            a_pert, obj_pert = subgradient_momentum(a_pert, max_iters=50, time_budget=5.0)
            if obj_pert < obj_new:
                a_new, obj_new = a_pert, obj_pert
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose:
                print(f"    cycle {cycle_num} strat {strat}: {best_obj:.8f}")
        else:
            # No improvement - try perturbation
            a_pert = condensed_explosion(a, rng, frac=0.08)
            a_pert, obj_pert = subgradient_momentum(a_pert, max_iters=30, time_budget=8.0)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                a = a_pert
            else:
                a_pert2 = window_noise(a, rng, noise_scale=0.15)
                a_pert2, obj_pert2 = subgradient_momentum(a_pert2, max_iters=30, time_budget=8.0)
                if obj_pert2 < best_obj:
                    best_obj = obj_pert2
                    best_a = a_pert2.copy()
                    a = a_pert2
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization with LP alternating,
    spectral methods, and adaptive perturbations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        # Fallback initialization
        n_init = 700
        x = np.linspace(0, 1, n_init)
        best_seq = np.sin(np.pi * x) ** 2 + 0.1
        best_seq = best_seq / best_seq.sum()
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"Generated initial: n={n_init}, obj={best_score:.8f}")

    print(f"Starting: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: Multi-length initialization screening (8% time) ===
    print("\n=== Phase 1: Multi-length Screening ===")
    phase1_budget = min(100.0, (deadline - start_time) * 0.08)
    
    n_parent = len(best_seq)
    n_options = [n_parent, max(100, n_parent - 100), n_parent + 100,
                 max(200, n_parent + 200), max(300, int(n_parent * 1.5)),
                 max(400, int(n_parent * 2))]
    n_options = list(dict.fromkeys([n for n in n_options if 100 <= n <= 3000]))
    
    candidates = []
    per_n_budget = max(5.0, phase1_budget / len(n_options))
    
    for n_try in n_options:
        if time.monotonic() - start_time > phase1_budget:
            break
        inits = generate_initializations(n_try, rng, parent)
        per_init = max(1.0, per_n_budget / len(inits))
        
        for init in inits:
            if time.monotonic() - start_time > phase1_budget:
                break
            # Quick optimization
            a_q, obj_q = subgradient_momentum(init, max_iters=50, time_budget=per_init)
            if obj_q < np.inf:
                candidates.append((obj_q, a_q, n_try))
    
    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best: {candidates[0][0] if candidates else 'N/A'}")
    
    for obj_c, a_c, n_c in candidates[:3]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  Phase 1 update: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive optimization on best (45% time) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"  Best entering: {best_score:.8f}")
    phase2_budget = min(650.0, (deadline - start_time) * 0.45)
    phase2_start = time.monotonic()
    
    # Main candidate
    a_main, obj_main = optimize_sequence(best_seq, phase2_budget, rng, verbose=True)
    if obj_main < best_score:
        best_score = obj_main
        best_seq = a_main.copy()
        print(f"  Phase 2 main: {best_score:.8f}")

    # Try second-best candidate from screening if different length
    remaining2 = phase2_budget - (time.monotonic() - phase2_start)
    if remaining2 > 30 and len(candidates) > 0:
        best_n = len(best_seq)
        for obj_c, a_c, n_c in candidates[:5]:
            if n_c != best_n and remaining2 > 20:
                t_alloc = min(remaining2 * 0.4, 60.0)
                a_try, obj_try = optimize_sequence(a_c, t_alloc, rng)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  Phase 2 alt (n={n_c}): {best_score:.8f}")
                remaining2 = phase2_budget - (time.monotonic() - phase2_start)
                if remaining2 < 15:
                    break

    # === Phase 3: Resampling and length search (20% time) ===
    print(f"\n=== Phase 3: Length Search ===")
    print(f"  Best entering: {best_score:.8f}")
    phase3_budget = min(280.0, (deadline - time.monotonic()) * 0.20)
    phase3_start = time.monotonic()
    
    n_cur = len(best_seq)
    n_tries = []
    for delta in [-50, -100, 50, 100, 150, 200, -200, 300]:
        nt = n_cur + delta
        if 100 <= nt <= 3000 and nt not in n_tries:
            n_tries.append(nt)
    
    per_n = max(8.0, phase3_budget / max(len(n_tries), 1))
    
    for n_try in n_tries:
        if time.monotonic() - phase3_start > phase3_budget - 5:
            break
        a_res = resample_sequence(best_seq, n_try)
        a_res = project_nonneg_sum1(a_res)
        
        # Quick optimization
        t_alloc = min(per_n, phase3_budget - (time.monotonic() - phase3_start))
        a_opt, obj_opt = optimize_sequence(a_res, t_alloc, rng)
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")

    # === Phase 4: Final aggressive polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Best entering: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    
    if remaining4 > 10 and best_seq is not None:
        a_polish = project_nonneg_sum1(best_seq.copy())
        n_polish = len(a_polish)
        
        # Multiple polish rounds
        round_num = 0
        while time.monotonic() < deadline - 5:
            round_num += 1
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            
            # LP (if size allows)
            if n_polish <= 2000 and rem > 20:
                t_lp = min(rem * 0.3, 50.0)
                a_p, obj_p = lp_alternate(a_polish, max_iters=min(25, int(t_lp)), time_budget=t_lp)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_polish = a_p.copy()
                    print(f"  Polish LP r{round_num}: {best_score:.8f}")
            
            # Spectral
            rem = deadline - time.monotonic()
            if rem > 8:
                a_p2, obj_p2 = spectral_flatten(a_polish, max_iters=40, time_budget=min(rem * 0.2, 15.0))
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_polish = a_p2.copy()
                    print(f"  Polish SPEC r{round_num}: {best_score:.8f}")
            
            # Conv compression
            rem = deadline - time.monotonic()
            if rem > 8:
                a_p3, obj_p3 = spectral_compress_conv(a_polish, max_iters=40, time_budget=min(rem * 0.2, 12.0))
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    a_polish = a_p3.copy()
                    print(f"  Polish COMPRESSION r{round_num}: {best_score:.8f}")
            
            # Small perturbation + quick re-opt
            rem = deadline - time.monotonic()
            if rem > 15:
                a_pert = random_perturb(a_polish, rng, scale=0.02)
                a_pert, obj_pert = subgradient_momentum(a_pert, max_iters=80, time_budget=min(rem * 0.2, 15.0))
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_polish = a_pert.copy()
                    print(f"  Polish PERT r{round_num}: {best_score:.8f}")
            
            # Condensed explosion + re-opt
            rem = deadline - time.monotonic()
            if rem > 20:
                a_exp = condensed_explosion(a_polish, rng, frac=0.05)
                a_exp, obj_exp = subgradient_momentum(a_exp, max_iters=60, time_budget=min(rem * 0.2, 12.0))
                if n_polish <= 1500 and deadline - time.monotonic() > 10:
                    a_exp2, obj_exp2 = lp_alternate(a_exp, max_iters=10, time_budget=min(deadline - time.monotonic() - 3, 15.0))
                    if obj_exp2 < obj_exp:
                        a_exp, obj_exp = a_exp2, obj_exp2
                if obj_exp < best_score:
                    best_score = obj_exp
                    best_seq = a_exp.copy()
                    a_polish = a_exp.copy()
                    print(f"  Polish EXPLODE r{round_num}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    # Ensure all values are valid
    result_arr = np.clip(best_seq.astype(np.float64), 0.0, 1000.0)
    result = [float(x) for x in result_arr]
    
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
