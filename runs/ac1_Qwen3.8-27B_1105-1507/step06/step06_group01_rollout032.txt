import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response_full(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for ALL k, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    # Build constraint matrix
    # (a*g)[k] = sum_j a[k-j]*g[j] for 0<=k-j<n, 0<=j<n
    # For k = 0,...,2n-2
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                            'dual_feasibility_tolerance': 1e-10,
                            'primal_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def lp_best_response_min_peak(a, time_limit=10):
    """Solve LP: minimize M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (M_opt, g_opt) or (None, None)."""
    n = len(a)
    # Variables: g[0..n-1], M
    # We minimize M
    # Constraints: (a*g)[k] - M <= 0 for all k, sum(g) = 1, g >= 0, M free (but >= 0)
    
    # In scipy linprog: minimize c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    # x = [g[0], ..., g[n-1], M]
    # c = [0, ..., 0, 1]
    
    # Build A_ub: (a*g)[k] - M <= 0
    # For each k, row is [a[k-0], a[k-1], ..., a[k-(n-1)], -1] (padding with 0)
    rows = []
    cols = []
    vals = []
    num_k = 2 * n - 1
    for k in range(num_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
        # -1 for M component (column n)
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_k, n + 1))
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    b_ub = np.zeros(num_k)
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]  # M >= 0
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
                return r.x[n], g
    except Exception:
        pass
    return None, None

def run_subgradient(a_init, max_iters, deadline):
    """Subgradient descent on max(conv(a,a)) on the simplex."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    M_0 = best_M
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient: gradient of (a*a)[k_star] w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Project to tangent of simplex
        grad_tan = grad - grad.mean()
        
        # Polyak-like step size
        # f(a) - f* where f* is lower bound
        f_lower = 1.0 / (2 * n - 1)  # theoretical minimum of max when sum=1
        gap = max(M - f_lower, 1e-6)
        grad_norm_sq = np.dot(grad_tan, grad_tan)
        if grad_norm_sq < 1e-15:
            break
        step = gap / grad_norm_sq
        step = min(step, 1.0)  # cap step
        
        a_new = a - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        a = a_new
        
        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0]))
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def fista_smooth(a_init, max_iters, deadline):
    """FISTA on log-sum-exp smooth approximation."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    tau = 50.0
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient
        grad = np.zeros(n)
        for i in range(n):
            gi = np.dot(p[i:i + n], a) if i + n <= len(p) else np.dot(p[i:], a[:len(p)-i])
            grad[i] = 2.0 * gi
        
        grad_tan = grad - grad.mean()
        
        # Step size with 1/L where L ~ 4*tau*max(a) or similar
        step = 0.5 / (1.0 + tau * 0.01 * (it + 1))
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            # Adaptive tau
            tau = min(tau * 1.05, 500.0)
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Blackman window
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    blackman = np.maximum(blackman, 0.01)
    inits.append(blackman / blackman.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * t))
    inits.append(rc / rc.sum())
    
    # Parabolic (concave)
    para = 4 * t * (1 - t) + 0.05
    inits.append(para / para.sum())
    
    # Edge-concentrated
    edge = np.maximum(1.0 - np.abs(2 * t - 1), 0.01)
    inits.append(edge / edge.sum())
    
    # Very edge-concentrated (triangular)
    tri = np.maximum(1.0 - 4 * np.abs(t - 0.5), 0.001)
    inits.append(tri / tri.sum())
    
    # Exponential decay from left
    exp_l = np.exp(-3.0 * t) + 0.01
    inits.append(exp_l / exp_l.sum())
    
    # Exponential decay from right
    exp_r = np.exp(-3.0 * (1 - t)) + 0.01
    inits.append(exp_r / exp_r.sum())
    
    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Spectral init: minimize peak iteratively
    a_spec = np.ones(n) / n
    for _ in range(30):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.05 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    return inits

import numpy as np

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = (a*g)[k] for linear convolution."""
    n = len(a)
    num_k = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_k):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(num_k, n))
    return A

            if g is None:
                no_improve += 1
                if no_improve > 8:
                    break
                continue
            dual_y = None
        
        # Line search: find best blend coefficient
        improved = False
        for t in [0.4, 0.6, 0.3, 0.5, 0.7, 0.2, 0.8, 0.15, 0.25]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-13:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 12:
                break
    
    return best_a


def run_dual_subgradient(a_init, max_iters, deadline):
    """Subgradient descent guided by LP dual variables."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Get dual from LP
        M_opt, g, dual_y = lp_min_peak(a, time_limit=4)
        
        if dual_y is not None and np.sum(dual_y) > 1e-10:
            # Compute gradient from dual: grad[i] = 2 * sum_k dual_y[k] * a[k-i]
            grad = np.zeros(n)
            # Only use significant dual components for efficiency
            top_k_idx = np.argsort(dual_y)[::-1][:min(50, len(dual_y))]
            for k in top_k_idx:
                yk = dual_y[k]
                if yk < 1e-10:
                    break
                for i in range(n):
                    j = int(k) - i
                    if 0 <= j < n:
                        grad[i] += 2.0 * yk * a[j]
        else:
            # Fallback: use argmax subgradient
            b = fast_conv(a)
            k_star = int(np.argmax(b))
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = 2.0 * a[j]
        
        # Project to tangent of simplex
        grad_tan = grad - grad.mean()
        grad_norm_sq = np.dot(grad_tan, grad_tan)
        if grad_norm_sq < 1e-15:
            break
        
        # Adaptive step
        b_curr = fast_conv(a)
        M_curr = b_curr.max()
        f_lower = 1.0 / (2 * n - 1)
        gap = max(M_curr - f_lower, 1e-6)
        step = min(2.0 * gap / grad_norm_sq, 0.5)
        
        a_new = a - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        
        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a


def fista_smooth(a_init, max_iters, deadline):
    """FISTA on log-sum-exp smooth approximation of max(conv)."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    tau = 30.0
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        b_max = b.max()
        # Clip for numerical stability
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient computation (vectorized)
        grad = np.zeros(n)
        for i in range(n):
            hi = min(i + n, len(p))
            if hi > i:
                grad[i] = 2.0 * np.dot(p[i:hi], a[:hi - i])
        
        grad_tan = grad - grad.mean()
        
        # Step with decay
        step = 0.3 / (1.0 + 0.005 * it)
        
        a_new = y - step * grad_tan
        # Project to simplex
        a_new = _proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = _proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        # Increase tau slowly
        if it % 50 == 0 and it > 0:
            tau = min(tau * 1.3, 300.0)
        
        if it % 20 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def _proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0]))
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def coordinate_descent(a_init, max_sweeps, deadline):
    """Coordinate descent with golden section search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    rng_cd = np.random.default_rng(12345)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a).copy()
        order = rng_cd.permutation(n)
        improved_any = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            x_cur = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Precompute for speed
            c_vals = np.zeros(k_hi - k_lo + 1)
            slopes = np.zeros(k_hi - k_lo + 1)
            quad_idx = -1
            for idx, k in enumerate(range(k_lo, k_hi + 1)):
                if k == 2 * i:
                    c_vals[idx] = b[k] - x_cur * x_cur
                    quad_idx = idx
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                        slopes[idx] = 2.0 * a[j]
            
            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, b[:k_lo].max())
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, b[k_hi + 1:].max())
            
            def obj_at(x_test):
                vals = c_vals + slopes * x_test
                m = unaff_max
                vm = vals.max()
                if vm > m:
                    m = vm
                if quad_idx >= 0:
                    qv = c_vals[quad_idx] + x_test * x_test
                    if qv > m:
                        m = qv
                return m
            
            # Golden section
            lo = 0.0
            hi = max(3.0 * max(x_cur, 1e-6), 0.5)
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)
            for _ in range(25):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) <= obj_at(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - x_cur) > 1e-12:
                delta = x_opt - x_cur
                a[i] = x_opt
                for idx, k in enumerate(range(k_lo, k_hi + 1)):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - x_cur * x_cur
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * delta
                improved_any = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
        elif not improved_any:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.01, 0.03, 0.08, 0.15]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-5)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Blackman
    bm = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    inits.append(np.maximum(bm, 0.01) / np.maximum(bm, 0.01).sum())
    
    # Parabolic concave
    para = 4.0 * t * (1 - t) + 0.03
    inits.append(para / para.sum())
    
    # Edge-concentrated (inverted parabola at edges)
    edge = np.maximum(1.0 - 4.0 * np.abs(t - 0.5), 0.005)
    inits.append(edge / edge.sum())
    
    # Strong edge concentration
    strong_edge = np.maximum(np.abs(2 * t - 1) ** 2, 0.001)
    inits.append(strong_edge / strong_edge.sum())
    
    # Exponential from left
    exp_l = np.exp(-4.0 * t) + 0.005
    inits.append(exp_l / exp_l.sum())
    
    # Exponential from right
    exp_r = np.exp(-4.0 * (1 - t)) + 0.005
    inits.append(exp_r / exp_r.sum())
    
    # Random smooth
    ksize = max(5, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.005)
    inits.append(smooth / smooth.sum())
    
    # Spectral init
    a_spec = np.ones(n) / n
    for _ in range(40):
        bs = fast_conv(a_spec)
        k_star = int(np.argmax(bs))
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.08 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def optimize_one(a0, n, deadline, phase_fista=0.15, phase_lp=0.55, phase_dsgd=0.15, phase_cd=0.15):
    """Run full optimization pipeline on a single init."""
    best = np.maximum(a0.astype(np.float64), 1e-8)
    best = best / best.sum()
    best_M = fast_conv(best).max()
    t_start = time.monotonic()
    total_time = max(1.0, deadline - t_start)
    
    # Stage 1: FISTA
    t1 = t_start + phase_fista * total_time
    if time.monotonic() < t1:
        a_f = fista_smooth(best, 5000, t1)
        M_f = fast_conv(a_f).max()
        if M_f < best_M:
            best_M = M_f
            best = a_f
    
    # Stage 2: LP (min-peak)
    t2 = t1 + phase_lp * total_time
    if time.monotonic() < t2:
        a_lp = run_lp_optimize(best, 200, t2, use_min_peak=True)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp
    
    # Stage 3: Dual subgradient
    t3 = t2 + phase_dsgd * total_time
    if time.monotonic() < t3:
        a_sg = run_dual_subgradient(best, 100, t3)
        M_sg = fast_conv(a_sg).max()
        if M_sg < best_M:
            best_M = M_sg
            best = a_sg
    
    # Stage 4: Coordinate descent
    if time.monotonic() < deadline:
        max_sw = max(3, int((deadline - time.monotonic()) / max(0.3, n / 80.0)))
        a_cd = coordinate_descent(best, max_sw, deadline)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd
    
    return best

            if time.monotonic() > deadline - 15:
                break
            new_n = int(n_cur * factor)
            new_n = max(192, min(1536, new_n))
            new_n = (new_n // 32) * 32

          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
