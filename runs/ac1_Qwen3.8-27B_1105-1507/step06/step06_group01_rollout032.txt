import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cumsum_u = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho_arr + 1.0) > cumsum_u
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if rho == 0:
            rho = n - 1
        theta = (cumsum_u[rho - 1] - 1.0) / rho if rho > 0 else 0.0
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad(a, tau):
    """Compute smooth max (log-sum-exp) of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    p = exp_shifted / exp_sum
    smooth_val = b_max + np.log(exp_sum) / tau

    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    # Efficient computation using the structure
    grad = np.zeros(n)
    # For each i, grad[i] = 2 * sum_{k=i}^{i+n-1} p[k] * a[k-i] (clamped to valid range)
    conv_len = 2 * n - 1
    for i in range(n):
        k_lo = max(i, 0)
        k_hi = min(i + n, conv_len)
        if k_lo < k_hi:
            grad[i] = 2.0 * np.dot(p[k_lo:k_hi], a[0:k_hi - k_lo])
    return smooth_val, grad

def run_frank_wolfe(a_init, deadline, n_lags=5, eta0=0.3):

    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    it = 0
    max_iters = 2000000

    while it < max_iters and time.monotonic() < deadline:
        b = fast_conv(a)
        b_max = b.max()

        # Select active lags
        if n_lags >= len(b):
            active = np.arange(len(b))
            weights = b - 0.5 * b_max
            weights = np.maximum(weights, 0.0)
        else:
            top_idx = np.argpartition(b, -n_lags)[-n_lags:]
            active = top_idx
            weights = b[active] - 0.5 * b_max
            weights = np.maximum(weights, 1e-10)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            break
        weights = weights / w_sum

        # Compute subgradient: g[i] = 2 * sum_k w_k * a[k-i]
        g = np.zeros(n)
        for idx, k in enumerate(active):
            w_k = weights[idx]
            # For lag k, subgrad at coord i is 2*a[k-i] if 0<=k-i<n
            k_lo = max(0, k - (n - 1))
            k_hi = min(k, n - 1)
            for i in range(k_lo, k_hi + 1):
                g[i] += w_k * a[k - i]
        g *= 2.0

        # FW step: find vertex with minimum gradient
        i_min = int(np.argmin(g))
        # Adaptive step size (FW with decreasing step)
        step = eta0 / (1.0 + 0.001 * it)
        # Cap step to avoid overshooting
        if a[i_min] < 1e-10:
            step = min(step, 0.5)

        a_new = (1.0 - step) * a + step * np.zeros(n)
        a_new[i_min] = step
        # Actually FW: a_new = (1-t)*a + t*e_{i_min}
        a_new = (1.0 - step) * a
        a_new[i_min] += step
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        it += 1

            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()

    return best_a


def cd_single_coord(a, b, i, n):
    """Optimize coordinate i exactly via golden section search.
    Returns optimal value for a[i] (unnormalized)."""
    x_cur = a[i]

    # Affected lags: k in [i, min(2n-2, i+n-1)] where 0 <= k-i < n
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Build base and slope arrays
    n_affected = 0
    base = np.empty(k_hi - k_lo + 1)
    slope = np.empty(k_hi - k_lo + 1)
    quad_pos = -1
    has_quad = False

    for k in range(k_lo, k_hi + 1):
        j = k - i
        if 0 <= j < n:
            if j == i:  # k = 2i
                base[n_affected] = b[k] - x_cur * x_cur
                slope[n_affected] = 0.0
                quad_pos = n_affected
                has_quad = True
            else:
                base[n_affected] = b[k] - 2.0 * x_cur * a[j]
                slope[n_affected] = 2.0 * a[j]
            n_affected += 1

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    base = base[:n_affected]
    slope = slope[:n_affected]

    def obj_at(x):
        if n_affected == 0:
            return base_max
        vals = base + slope * x
        m = vals.max()
        if has_quad:
            qv = base[quad_pos] + x * x
            if qv > m:
                m = qv
        if base_max > m:
            m = base_max
        return m

    # Golden section search
    lo = 0.0
    hi = max(3.0 * x_cur, 0.1)
    # Expand hi if needed
    for _ in range(5):
        if obj_at(hi) > obj_at(hi * 0.5):
            break
        hi *= 2.0
        if hi > 10.0:
            break

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(30):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_cd(a_init, deadline, max_sweeps=500, focus_top=20):
    """Coordinate descent with focus on high-impact coordinates."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)

        # Determine order: focus on coordinates most affecting the peak
        b_max = b.max()
        peak_k = int(np.argmax(b))
        # Importance of coord i: a[peak_k - i] (if valid)
        importance = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                importance[i] = a[j]
        # Sort by importance (high first)
        order = np.argsort(importance)[::-1]

        sweep_changed = False
        for i in order:
            x_opt = cd_single_coord(a, b, i, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                # Update b in-place
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            b[k] += x_opt * x_opt - old * old
                        else:
                            b[k] += 2.0 * a[j] * (x_opt - old)
                sweep_changed = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M_cur = fast_conv(a).max()
        if M_cur < best_M * (1 - 1e-10):
            best_M = M_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_changed:
            no_improve += 2
        if no_improve > 30:
            break

    return best_a


def build_active_lp(n, a, active_indices, M):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, n_active=20, timeout=10):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= 1 for k in active, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    top_idx = np.argsort(b)[::-1][:min(n_active, 2 * n - 1)]
    active = top_idx.tolist()

    A = build_active_lp(n, a, active, 1.0)
    if A.nnz == 0:
        return None

    c = -np.ones(n)
    b_ub = np.ones(len(active))
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': timeout, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_polish(a_init, deadline, n_active=30, max_iters=200):
    """LP mirror descent for final polish."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    cur_n_active = n_active

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_best_response(a, cur_n_active, timeout=8)
        if g is None:
            cur_n_active = min(cur_n_active * 2, 2 * n - 1)
            no_improve += 1
            if no_improve > 10:
                break
            continue

        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            if cur_n_active > 10:
                cur_n_active = max(10, int(cur_n_active * 0.9))
        else:
            no_improve += 1
            cur_n_active = min(int(cur_n_active * 1.5), 2 * n - 1)
            if no_improve > 20:
                break

    return best_a


def gen_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        from_parent = np.interp(t, np.linspace(0, 1, parent.size), parent)
        from_parent = np.maximum(from_parent, 1e-6)
        inits.append(from_parent / from_parent.sum())
        noisy = np.maximum(from_parent + rng.normal(0, 0.05, n), 1e-6)
        inits.append(noisy / noisy.sum())

    inits.append(np.ones(n) / n)

    # Sine-modulated
    for freq in [1, 2, 3]:
        s = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Gaussian
    g = np.exp(-6.0 * (t - 0.5) ** 2) + 0.05
    inits.append(g / g.sum())

    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm, 0.01)
    inits.append(sm / sm.sum())

    # Random (flat spectrum)
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_full_optimization(a0, n, deadline):
    """Run full optimization pipeline on a single starting point."""
    best = a0.copy()
    T = deadline - time.monotonic()
    if T < 5:
        return best

    # Stage 1: FISTA (35% of time)
    t1 = time.monotonic() + 0.35 * T
    a1 = run_fista(best, t1)
    M1 = fast_conv(a1).max()
    M0 = fast_conv(best).max()
    if M1 < M0:
        best = a1
    print(f"    FISTA: M {M0:.8f} -> {fast_conv(best).max():.8f}", flush=True)

    # Stage 2: Frank-Wolfe (25% of time)
    t2 = time.monotonic() + 0.25 * (deadline - time.monotonic())
    a2 = run_frank_wolfe(best, t2)
    M2 = fast_conv(a2).max()
    M_prev = fast_conv(best).max()
    if M2 < M_prev:
        best = a2
    print(f"    FW: M {M_prev:.8f} -> {fast_conv(best).max():.8f}", flush=True)

    # Stage 3: Coordinate Descent (25% of time)
    t3 = time.monotonic() + 0.25 * (deadline - time.monotonic())
    a3 = run_cd(best, t3)
    M3 = fast_conv(a3).max()
    M_prev = fast_conv(best).max()
    if M3 < M_prev:
        best = a3
    print(f"    CD: M {M_prev:.8f} -> {fast_conv(best).max():.8f}", flush=True)

    # Stage 4: LP Polish (15% of time)
    if time.monotonic() < deadline - 5:
        a4 = run_lp_polish(best, deadline - 2, n_active=25)
        M4 = fast_conv(a4).max()
        M_prev = fast_conv(best).max()
        if M4 < M_prev:
            best = a4
        print(f"    LP: M {M_prev:.8f} -> {fast_conv(best).max():.8f}", flush=True)

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: FISTA + Frank-Wolfe + CD + LP polish,
    with multi-start and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, uniform n=768", flush=True)

    n_parent = parent.size
    if n_parent == 0:
        n_parent = 768

    # Determine lengths to try
    lengths = [n_parent]
    for nl in [512, 768, 1024, 1536]:
        if nl not in lengths and 256 <= nl <= 2048:
            lengths.append(nl)
    lengths.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))

    # Time allocation
    total_time = deadline - time.monotonic()
    # Reserve 40% for parent length, split rest among others
    n_primary = 1
    n_secondary = len(lengths) - 1
    time_primary = 0.45 * total_time
    time_secondary = 0.55 * total_time / max(1, n_secondary)

    print(f"\n{'='*60}", flush=True)
    print(f"Lengths to try: {lengths}", flush=True)
    print(f"Primary n={lengths[0]}: {time_primary:.0f}s")
    print(f"{'='*60}", flush=True)

    # Phase 1: Primary length (most time)
    n = lengths[0]
    t_deadline = time.monotonic() + time_primary
    if n != n_parent and parent is not None:
        a_start = resample_seq(parent, n)
    else:
        a_start = parent.copy()
    a_start = np.maximum(a_start, 1e-8)
    a_start = a_start / a_start.sum()

    # Multiple starts on primary length
    inits = gen_inits(n, rng, tracker.best if tracker.best is not None else parent)
    n_inits = min(len(inits), 4)
    time_per_init = time_primary / n_inits

    best_primary = a_start.copy()
    best_M_primary = fast_conv(a_start).max()

    for ii in range(n_inits):
        if time.monotonic() > t_deadline - 5:
            break
        a0 = inits[ii]
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()
        sub_deadline = time.monotonic() + time_per_init - 2
        a_opt = run_full_optimization(a0, n, sub_deadline)
        M_opt = fast_conv(a_opt).max()
        if M_opt < best_M_primary:
            best_M_primary = M_opt
            best_primary = a_opt.copy()
        tracker.update(a_opt)

    tracker.update(best_primary)
    print(f"\n  Primary n={n}: best J={tracker.best_score:.8f}", flush=True)

    # Phase 2: Secondary lengths (if time remains)
    if time.monotonic() < deadline - 30:
        for nl in lengths[1:]:
            if time.monotonic() > deadline - 30:
                break
            t_sub = time.monotonic() + time_secondary
            if n_parent > 0:
                a_s = resample_seq(tracker.best, nl)
            else:
                a_s = np.ones(nl)
            a_s = np.maximum(a_s, 1e-8)
            a_s = a_s / a_s.sum()

            inits_s = gen_inits(nl, rng, tracker.best)
            n_inits_s = min(len(inits_s), 3)
            time_per = (t_sub - time.monotonic()) / n_inits_s

            for ii in range(n_inits_s):
                if time.monotonic() > t_sub - 5:
                    break
                a0 = inits_s[ii]
                a0 = np.maximum(a0, 1e-8)
                a0 = a0 / a0.sum()
                sub_dl = time.monotonic() + max(5, time_per - 2)
                a_opt = run_full_optimization(a0, nl, sub_dl)
                tracker.update(a_opt)

    tracker.report("After multi-length search")

    # Phase 3: Intensive polish on global best
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Intensive polish on n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)

        n_best = tracker.best.size
        remaining = deadline - 10 - time.monotonic()

        # CD polish
        if remaining > 30:
            t_cd = time.monotonic() + 0.4 * remaining
            a_cd = run_cd(tracker.best, t_cd, max_sweeps=2000, focus_top=30)
            tracker.update(a_cd)

        # LP polish
        if time.monotonic() < deadline - 10:
            t_lp = time.monotonic() + 0.4 * (deadline - 10 - time.monotonic())
            a_lp = run_lp_polish(tracker.best, t_lp, n_active=40, max_iters=300)
            tracker.update(a_lp)

        # Final CD
        if time.monotonic() < deadline - 5:
            a_final = run_cd(tracker.best, deadline - 3, max_sweeps=500, focus_top=50)
            tracker.update(a_final)

    tracker.report("FINAL")

    # Ensure valid output
    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"RESULT: J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
