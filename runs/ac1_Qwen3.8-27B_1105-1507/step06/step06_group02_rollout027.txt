"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral initialization
2. Nesterov-accelerated smooth gradient descent
3. Active-set LP mirror descent with Anderson acceleration

   - Periodic full-constraint audit
4. Multi-length exploration
Key improvement over previous: active-set LP reduces per-iteration cost by 10-50x,
enabling far more iterations and better convergence.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective function value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=200, beta=0.8):
    """Gerchberg-Saxton: iteratively flatten the power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range
        mag_norm = mag / (mag.max() + 1e-30)
        # Power-law compression
        power = 0.3 + 0.7 * (1 - beta)
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.mean()
        # Linear compression
        mag_lin = (1.0 - beta) * mag + beta * mag.mean()
        mag_final = 0.5 * mag_alt + 0.5 * mag_lin
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a

    """Compute smooth approximation of gradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-11)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * p[k] * a[k - j]
    return grad, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.35, lr_end=0.002):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()

    y = a.copy()
    momentum = 0.0
    mu = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        # Nesterov momentum
        prev_a = a.copy()
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        momentum = ((it * mu) / (it + 1 + mu)) 
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:

def solve_active_lp(a, M, active_indices, n):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for k in active_indices, g >= 0."""
    if len(active_indices) == 0:
        return None
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(float(a[k - j]))
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'max_iter': 5000})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def solve_full_lp(a, M, n):

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

def anderson_mix(history_x, history_Fx, m):
    """Anderson mixing: find convex combination minimizing residual norm."""
    if m < 2:
        return np.maximum(history_Fx[-1], 0)
    residuals = []
    for i in range(m):
        residuals.append(history_x[i] - history_Fx[i])
    residuals = np.array(residuals)
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def run_active_lp_descent(a, deadline, max_iters=5000, K_init=50, K_max=500,
                          audit_every=50, t_init=0.4, t_min=0.02):
    """Active-set LP mirror descent with Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    K = K_init
    hist_x = []
    hist_Fx = []
    window = 5
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 80:
                break
        
        # Find active constraints (top-K)
        K_cur = min(K, 2 * n - 1)
        active_idx = np.argsort(b)[-K_cur:]
        
        # Solve active LP
        g = solve_active_lp(a, M, active_idx, n)
        
        if g is None:
            # Try with more constraints
            K = min(K * 2, K_max)
            active_idx = np.argsort(b)[-min(K, 2*n-1):]
            g = solve_active_lp(a, M, active_idx, n)
            if g is None:
                break
        
        # Audit: check if g violates non-active constraints (every audit_every iters)
        if it % audit_every == audit_every - 1:
            b_g = fast_conv(g)
            if b_g.max() > M * 1.5:
                # g is bad, use full LP
                g_full = solve_full_lp(a, M, n)
                if g_full is not None:
                    g = g_full
                    # Increase K for next iterations
                    K = min(K * 2, K_max)
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Step size
        t = max(t_min, t_init / (1 + 0.02 * it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
            M_and = fast_conv(a_and).max()
            M_new = fast_conv(a_new).max()
            if M_and <= M_new * 1.02:
                a = a_and
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

achieving much tighter convergence to the mirror-descent fixed point.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective function value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=200, beta=0.8):
    """Gerchberg-Saxton: iteratively flatten the power spectrum while maintaining non-negativity."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Power-law compression: softer for more iterations
        power = 0.4 + 0.6 * (1.0 - beta)
        mag_comp = (mag ** power)
        mag_comp = mag_comp / (mag_comp.max() + 1e-30)
        # Target: mean level
        target = mag.mean()
        mag_new = (1.0 - beta * 0.7) * mag + beta * 0.7 * target * (mag_comp / mag_comp.mean())
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = float(fast_conv(a_candidate).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad_fft(a, tau):
    """Compute smooth gradient of max(conv(a,a)) using FFT-based cross-correlation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Softmax weights over convolution values
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / (exp_b.sum() + 1e-30)
    # grad(j) = 2 * sum_k p(k) * a(k-j) for valid k-j
    # This is 2 * cross_correlation(p, a)(j)
    # cross(j) = sum_{m=0}^{n-1} p(j+m) * a(m)  [p out of range = 0]
    # Use direct computation (O(n^2) but vectorized with stride tricks for speed)
    # For n up to ~4096, this is fast enough
    size = 1
    while size < 4 * n:
        size <<= 1
    # Use FFT: cross(j) = ifft(FFT(p) .* conj(FFT(a)))(j) ... need to be careful
    # Actually: cross(j) = sum_m p(j+m) a(m) = (p_flipped * a)(j + 2n - 2) ... 
    # Let me just use the correlation form:
    # Define q = p (length 2n-1), then cross(j) = sum_m q(j+m)*a(m)
    # = conv(q, a_rev)(j + n - 1) where a_rev = a[::-1] ... let me verify
    # conv(q, a_rev)(k) = sum_i q(i) a_rev(k-i) = sum_i q(i) a(n-1-k+i)
    # Let j = n-1-k, so k = n-1-j:
    # conv(q, a_rev)(n-1-j) = sum_i q(i) a(j + i - n + 1 + n - 1) ... this is getting confusing
    # Let me just do it directly with a loop but vectorized inner product
    grad = np.zeros(n)
    # p has length 2n-1, a has length n
    # For each j in [0, n-1]: grad[j] = 2 * dot(p[j:j+n], a[0:min(n, 2n-1-j)])
    for j in range(n):
        m_end = min(n, 2 * n - 1 - j)
        if m_end <= 0:
            break
        grad[j] = 2.0 * np.dot(p[j:j + m_end], a[:m_end])
    return grad, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.35, lr_end=0.002):
    """Nesterov-accelerated gradient descent on smooth max approximation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    mu = 0.85
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad_fft(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        # Nesterov momentum
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        # Momentum update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new
    return best_a


def solve_active_lp(a, M, active_indices, n):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for k in active_indices, g >= 0.
    Returns the optimal g or None if infeasible."""
    K = len(active_indices)
    if K == 0:
        return None

    rows = np.zeros(K * n, dtype=np.int32)
    cols = np.zeros(K * n, dtype=np.int32)
    vals = np.zeros(K * n, dtype=np.float64)
    
    pos = 0
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        if count > 0:
            rows[pos:pos + count] = idx
            cols[pos:pos + count] = np.arange(j_min, j_max + 1)
            vals[pos:pos + count] = a[k - j_max:k - j_min + 1][::-1]
            pos += count
    
    if pos == 0:
        return None
    
    A = csr_matrix((vals[:pos], (rows[:pos], cols[:pos])), shape=(K, n))
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'max_iter': 10000})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def solve_full_lp(a, M, n):
    """Solve LP with all 2n-1 constraints (for auditing/final polish)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(float(a[k - j]))
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'max_iter': 50000})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m):
    """Anderson mixing: find combination minimizing residual norm."""
    if m < 2:
        return np.maximum(history_Fx[-1].copy(), 0)
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = float(ones @ G_inv_ones)
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -1.0, 3.0)
    s = float(alpha.sum())
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def run_active_lp_descent(a, deadline, max_iters=5000, K_init=50, K_max=500,
                          audit_every=40, t_init=0.4, t_min=0.01,
                          anderson_window=5):
    """Active-set LP mirror descent with Anderson acceleration and progressive K."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    K = K_init
    hist_x = []
    hist_Fx = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                break
        
        # Find active constraints (top-K by convolution value)
        K_cur = min(K, 2 * n - 1)
        active_idx = np.argsort(b)[-K_cur:]
        
        # Solve active LP
        g = solve_active_lp(a, M, active_idx, n)
        
        if g is None:
            K = min(K * 2, K_max)
            active_idx = np.argsort(b)[-min(K, 2 * n - 1):]
            g = solve_active_lp(a, M, active_idx, n)
            if g is None:
                break
        
        # Periodic audit: check if g satisfies ALL constraints
        if it > 0 and it % audit_every == audit_every - 1:
            b_g = fast_conv(g)
            max_violation = float(b_g.max())
            if max_violation > M * 1.3:
                # Significantly violated - use full LP
                g_full = solve_full_lp(a, M, n)
                if g_full is not None:
                    g = g_full
                K = min(K * 2, K_max)
            elif max_violation > M * 1.05:
                # Mildly violated - grow K
                K = min(int(K * 1.5), K_max)
        
        # Shrink K if stable for many iterations
        if no_imp == 0 and it > 200 and K > K_init * 3:
            K = max(K // 2, K_init)
        
        gs = float(g.sum())
        if gs < 1e-10:
            break
        g = g / gs
        
        # Adaptive step size
        t = max(t_min, t_init / (1.0 + 0.015 * it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = float(a_and.sum())
            if sa > 1e-10:
                a_and = a_and / sa
                M_and = float(fast_conv(a_and).max())
                M_new = float(fast_conv(a_new).max())
                if M_and <= M_new * 1.01:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent):
    """Generate diverse initializations for multi-start."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p * (1.0 + 0.1 * rng.normal(0, 1, n)), 0.01))
    
    inits.append(np.ones(n) / n)
    for alpha in [0.4, 0.8, 1.2, 2.0, 4.0]:
        shape = (t ** alpha + (1 - t) ** alpha) / 2.0
        inits.append(np.maximum(0.05 + 0.95 * shape, 0.01))
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-5 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03)
    
    # Random smooth
    k = max(3, n // 25)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a, label=""):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < self.score:
            old = self.score
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** NEW BEST: J={s:.8f} (was {old:.8f}) n={a.size} t={el:.0f}s {label}", flush=True)
    
    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """Main search function: finds the best non-negative sequence minimizing J."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    best = BestTracker()
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass
    
    if parent is None:
        parent = np.ones(1024)
        best.update(parent, "uniform")
    
    n_main = parent.size
    
    # === PHASE 1: GS + Smooth optimization (80s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: GS + Smooth Optimization (80s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    p1_deadline = time.monotonic() + 80.0
    
    # GS initialization on parent
    a_gs = gs_flat_spectral(parent, n_iters=150, beta=0.8)
    best.update(a_gs, "GS")
    
    # Nesterov smooth from GS
    a_smooth = run_nesterov_smooth(a_gs, min(p1_deadline, time.monotonic() + 40),
                                    max_iters=2000, tau_start=1.0, tau_end=150.0,
                                    lr_init=0.3, lr_end=0.003)
    best.update(a_smooth, "GS+smooth")
    
    # Try a few diverse inits (quick)
    inits = make_initializations(n_main, rng, best.seq if best.seq is not None else parent)
    for idx, a0 in enumerate(inits[1:5]):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        a_try = run_nesterov_smooth(a0, min(time.monotonic() + 8, p1_deadline),
                                     max_iters=400, tau_start=5, tau_end=80,
                                     lr_init=0.2, lr_end=0.01)
        best.update(a_try, f"div_init_{idx}")
    
    best.report("Phase 1 complete")
    
    # === PHASE 2: Progressive Active-Set LP Mirror Descent (~1150s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Progressive Active-Set LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = time.monotonic() + 1150.0
    a_cur = best.seq.copy() if best.seq is not None else a_smooth.copy()
    
    # Pass 1: Fast exploration (K small, large steps) - 350s
    print(f"  Pass 1: K={50}, t={0.5}->{0.1}, 350s", flush=True)
    p1d = time.monotonic() + 350.0
    a_cur = run_active_lp_descent(
        a_cur, min(p1d, p2_deadline),
        max_iters=10000, K_init=40, K_max=200,
        audit_every=30, t_init=0.5, t_min=0.08,
        anderson_window=5)
    best.update(a_cur, "LP_pass1")
    
    # Pass 2: Medium (K larger, smaller steps) - 350s
    if time.monotonic() < p2_deadline - 300:
        print(f"  Pass 2: K={80}, t={0.3}->{0.04}, 350s", flush=True)
        p2d = time.monotonic() + 350.0
        a_cur = run_active_lp_descent(
            best.seq, min(p2d, p2_deadline),
            max_iters=10000, K_init=80, K_max=400,
            audit_every=40, t_init=0.3, t_min=0.04,
            anderson_window=6)
        best.update(a_cur, "LP_pass2")
    
    # Pass 3: Fine (K large, very small steps) - 300s
    if time.monotonic() < p2_deadline - 200:
        print(f"  Pass 3: K={150}, t={0.15}->{0.01}, 300s", flush=True)
        p3d = time.monotonic() + 300.0
        a_cur = run_active_lp_descent(
            best.seq, min(p3d, p2_deadline),
            max_iters=10000, K_init=150, K_max=600,
            audit_every=60, t_init=0.15, t_min=0.01,
            anderson_window=6)
        best.update(a_cur, "LP_pass3")
    
    # Pass 4: Very fine polish
    if time.monotonic() < p2_deadline - 100:
        print(f"  Pass 4: K={300}, t={0.08}->{0.003}, remaining", flush=True)
        a_cur = run_active_lp_descent(
            best.seq, p2_deadline,
            max_iters=5000, K_init=300, K_max=800,
            audit_every=80, t_init=0.08, t_min=0.003,
            anderson_window=8)
        best.update(a_cur, "LP_pass4")
    
    best.report("Phase 2 complete")
    
    # === PHASE 3: Length exploration (150s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Length Exploration (150s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_deadline = time.monotonic() + 150.0
    n_best = best.seq.size
    
    deltas = [128, -128, 256, 512, -256, 384, -384]
    for delta in deltas:
        if time.monotonic() > p3_deadline:
            break
        n_try = n_best + delta
        if n_try < 256 or n_try > 3072:
            continue
        
        a_try = resample_seq(best.seq, n_try)
        a_try = np.maximum(a_try, 1e-6)
        a_try = a_try / a_try.sum()
        
        # Quick GS
        a_try = gs_flat_spectral(a_try, n_iters=50, beta=0.85)
        best.update(a_try, f"len{n_try}_GS")
        
        # LP polish
        if time.monotonic() < p3_deadline - 15:
            a_lp = run_active_lp_descent(
                a_try, min(time.monotonic() + 25, p3_deadline - 5),
                max_iters=1000, K_init=40, K_max=200,
                audit_every=30, t_init=0.4, t_min=0.02,
                anderson_window=5)
            best.update(a_lp, f"len{n_try}_LP")
    
    best.report("Phase 3 complete")
    
    # === PHASE 4: Final deep polish with full LP (remaining time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Deep Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if time.monotonic() < deadline - 10:
        # One more active-set LP with very fine parameters
        a_final = run_active_lp_descent(
            best.seq, deadline,
            max_iters=5000, K_init=200, K_max=1000,
            audit_every=100, t_init=0.1, t_min=0.002,
            anderson_window=8)
        best.update(a_final, "final_active")
    
    # Absolute final: try full LP if time permits (expensive but precise)
    if time.monotonic() < deadline - 5:
        n_f = best.seq.size
        if n_f <= 2048:
            a_f = np.maximum(best.seq, 1e-8)
            a_f = a_f / a_f.sum()
            b_f = fast_conv(a_f)
            M_f = float(b_f.max())
            g_f = solve_full_lp(a_f, M_f, n_f)
            if g_f is not None:
                g_f = g_f / g_f.sum()
                for t_blend in [0.3, 0.2, 0.1, 0.05]:
                    a_blend = np.maximum((1 - t_blend) * a_f + t_blend * g_f, 0)
                    a_blend = a_blend / a_blend.sum()
                    best.update(a_blend, f"fullLP_t{t_blend}")
                    if best.score < 1.503:
                        break
    
    # Fallback
    if best.seq is None:
        best.seq = parent.copy()
        best.score = compute_J(best.seq)
    
    total_time = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={best.score:.8f} n={best.seq.size} time={total_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best.seq]
