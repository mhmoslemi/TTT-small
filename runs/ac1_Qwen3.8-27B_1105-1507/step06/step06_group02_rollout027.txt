import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_avg = mag.mean() + 1e-30
        mag_new = (1.0 - beta) * mag + beta * mag_avg
        power = 0.6
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    shifted -= shifted.max()
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(2 * n - 1)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * p[k] * a[k - j]
    return grad[:n], p, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.3, lr_end=0.002):

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta_nest = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1.0 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        a_new = np.maximum(y - lr * (grad + beta_nest * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_nest * beta_nest)) / 2.0
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y / sy
        a = a_new
    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response_primal(a, M):
    n = len(a)
    rows, cols, vals = build_A_sparse(n, a)
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_vec, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_dual_weights(a, M):
    n = len(a)
    rows, cols, vals = build_A_sparse(n, a)
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    AT = A_mat.T
    c_dual = np.full(2 * n - 1, M)
    b_dual = np.ones(n)
    bounds_dual = [(0, None)] * (2 * n - 1)
    try:
        r = linprog(c_dual, A_ub=-AT, b_ub=-b_dual, bounds=bounds_dual, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            y = np.maximum(r.x, 0)
            sy = y.sum()
            if sy > 1e-15:
                return y / sy
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m):
    if m < 2:
        return history_Fx[-1].copy()
    residuals = []
    for i in range(m):
        residuals.append(history_x[i] - history_Fx[i])
    residuals = np.array(residuals)
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 1.5)
    s_alpha = alpha.sum()
    if abs(s_alpha) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / s_alpha
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def run_lp_mirror_adaptive(a, deadline, max_iters=500, anderson_window=5,
                           max_step=0.5, min_step=0.005, use_dual=False):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    hist_x = []
    hist_Fx = []
    current_step = max_step
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1.0 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        g = lp_best_response_primal(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        # Backtracking line search
        found = False
        t_try = current_step
        for _ in range(10):
            a_t = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s_t = a_t.sum()
            if s_t < 1e-10:
                t_try *= 0.5
                continue
            a_t = a_t / s_t
            M_t = fast_conv(a_t).max()
            if M_t <= M * (1.0 + 1e-10):
                a = a_t
                if M_t < best_M - 1e-14:
                    current_step = min(max_step, t_try * 1.2)
                else:
                    current_step = max(min_step, t_try * 0.7)
                found = True
                break
            t_try *= 0.5
        if not found:
            t_try = min_step
            a_t = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s_t = a_t.sum()
            if s_t < 1e-10:
                break
            a = a_t / s_t
            current_step = min_step
            # Check if we made progress at all
            M_now = fast_conv(a).max()
            if M_now >= M * 1.001:
                a = best_a.copy()
                break
        # Anderson acceleration
        hist_x.append(best_a.copy())
        hist_Fx.append(a.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        if len(hist_x) >= 3 and it % 3 == 0:
            a_and = anderson_mix(hist_x, hist_Fx[len(hist_x) - len(hist_x):], min(len(hist_x), anderson_window))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and = a_and / sa
                M_and = fast_conv(a_and).max()
                M_cur = fast_conv(a).max()
                if M_and <= M_cur * 1.02:
                    a = a_and
    return best_a


def run_lp_mirror_dual(a, deadline, max_iters=300):
    """LP mirror descent using dual weights for subgradient direction."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1.0 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
        # Get primal best response
        g = lp_best_response_primal(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        # Also get dual weights for potential subgradient step
        y_dual = lp_dual_weights(a, M)
        # Try both: LP direction and dual subgradient direction
        candidates = [g]
        if y_dual is not None:
            # Subgradient: grad_i = 2 * sum_k y_k * a_{k-i}
            subgrad = np.zeros(n)
            for k in range(2 * n - 1):
                if y_dual[k] < 1e-12:
                    continue
                for j in range(n):
                    ki = k - j
                    if 0 <= ki < n:
                        subgrad[j] += 2.0 * y_dual[k] * a[ki]
            # Direction: a - step * subgrad, projected
            for step in [0.01, 0.005, 0.001]:
                cand = np.maximum(a - step * subgrad, 0.0)
                sc = cand.sum()
                if sc > 1e-10:
                    candidates.append(cand / sc)
        # Pick best candidate
        best_cand = None
        best_cand_M = float('inf')
        for cand in candidates:
            M_c = fast_conv(cand).max()
            if M_c < best_cand_M:
                best_cand_M = M_c
                best_cand = cand
        if best_cand is None:
            break
        a = best_cand
        if best_cand_M < best_M - 1e-14:
            best_M = best_cand_M
            best_a = a.copy()
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy().astype(np.float64)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.015, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.04, n), 0.001))
        inits.append(np.maximum(p * (1.0 + 0.08 * rng.normal(0, 1, n)), 0.001))
    inits.append(np.ones(n) / n)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        inits.append(np.maximum(0.3 + 0.7 * (t**alpha + (1.0 - t)**alpha) / (2.0 * (0.5**alpha + 0.5**alpha)), 0.01))
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.6 + 0.35 * np.cos(2.0 * np.pi * f * t), 0.01))
    inits.append(np.exp(-3.0 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-6.0 * (t - 0.3)**2) + np.exp(-6.0 * (t - 0.7)**2) + 0.03)
    # Asymmetric linear ramp
    inits.append(np.maximum(0.2 + 0.8 * t, 0.01))
    inits.append(np.maximum(0.2 + 0.8 * (1.0 - t), 0.01))
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.25, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=80, beta=0.85))
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_flat_spectral(p, n_iters=100, beta=0.9))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # Phase 1: Diverse smooth optimization across lengths
    print("PHASE 1: Smooth optimization across lengths", flush=True)
    p1_start = time.monotonic()
    lengths = [1024, 1280, 1536, 1792, 2048]
    best_per_length = {}
    p1_budget = 180.0
    tp = p1_budget / len(lengths)

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        t0 = time.monotonic()
        dl = t0 + tp
        inits = make_diverse_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')
        max_it = max(300, int(4000 * (1024.0 / n)))
        for idx, a0 in enumerate(inits):
            if time.monotonic() > dl - 5:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            sub_dl = dl - 5 - (dl - t0 - (time.monotonic() - t0)) * 0.3
            a = run_nesterov_smooth(a0, min(dl - 2, sub_dl), max_iters=max_it,
                                    tau_start=1.0, tau_end=250.0,
                                    lr_init=0.3, lr_end=0.003)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()
        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()
    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # Phase 2: LP mirror descent with adaptive step on best sequence
    print("PHASE 2: LP Mirror Descent (adaptive)", flush=True)
    p2_start = time.monotonic()
    if best.seq is not None:
        a = best.seq.copy()
        # Run adaptive LP mirror descent
        dl2 = min(time.monotonic() + 500, deadline - 200)
        a = run_lp_mirror_adaptive(a, dl2, max_iters=300, anderson_window=5,
                                   max_step=0.4, min_step=0.005)
        best.update(a)

        # Second pass with smaller steps
        if time.monotonic() < deadline - 150:
            dl3 = min(time.monotonic() + 300, deadline - 100)
            a = run_lp_mirror_adaptive(a, dl3, max_iters=200, anderson_window=5,
                                       max_step=0.2, min_step=0.002)
            best.update(a)
    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # Phase 3: Dual-based refinement
    print("PHASE 3: Dual-based refinement", flush=True)
    p3_start = time.monotonic()
    if best.seq is not None and time.monotonic() < deadline - 100:
        a = best.seq.copy()
        dl4 = min(time.monotonic() + 150, deadline - 60)
        a = run_lp_mirror_dual(a, dl4, max_iters=100)
        best.update(a)
    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # Phase 4: Try other lengths with LP
    print("PHASE 4: Multi-length LP polish", flush=True)
    p4_start = time.monotonic()
    if best_per_length and time.monotonic() < deadline - 50:
        sorted_lengths = sorted(best_per_length.keys(), key=lambda nn: compute_J(best_per_length[nn]))
        for n_try in sorted_lengths[:3]:
            if time.monotonic() > deadline - 30:
                break
            if n_try == best.seq.size:
                continue
            a_c = best_per_length[n_try].copy()
            # Quick smooth polish
            a_c = run_nesterov_smooth(a_c, min(time.monotonic() + 20, deadline - 25),
                                      max_iters=300, tau_start=5, tau_end=100,
                                      lr_init=0.1, lr_end=0.005)
            best.update(a_c)
            # LP polish
            if time.monotonic() < deadline - 20:
                a_lp = run_lp_mirror_adaptive(a_c, min(time.monotonic() + 40, deadline - 15),
                                              max_iters=50, anderson_window=4,
                                              max_step=0.3, min_step=0.01)
                best.update(a_lp)
    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # Phase 5: Perturbation restarts on best
    print("PHASE 5: Perturbation restarts", flush=True)
    p5_start = time.monotonic()
    if best.seq is not None and time.monotonic() < deadline - 30:
        n_best = best.seq.size
        for restart in range(5):
            if time.monotonic() > deadline - 15:
                break
            a_pert = np.maximum(best.seq + rng.normal(0, 0.01, n_best), 1e-6)
            a_pert = a_pert / a_pert.sum()
            a_pert = run_lp_mirror_adaptive(a_pert, min(time.monotonic() + 30, deadline - 10),
                                            max_iters=50, anderson_window=4,
                                            max_step=0.3, min_step=0.01)
            best.update(a_pert)
    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # Phase 6: Final fine-tune LP
    print("PHASE 6: Final fine-tune", flush=True)
    if best.seq is not None and time.monotonic() < deadline - 5:
        a_fin = best.seq.copy()
        a_fin = run_lp_mirror_adaptive(a_fin, deadline - 2, max_iters=30,
                                       anderson_window=3, max_step=0.1, min_step=0.002)
        best.update(a_fin)

    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\nFINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    return [float(x) for x in best.seq]
