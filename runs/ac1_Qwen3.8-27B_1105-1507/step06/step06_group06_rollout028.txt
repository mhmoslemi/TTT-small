**Alternative: Direct subgradient with momentum and annealing**

The subgradient of `max_k conv[k]` at a point where `k*` is the unique max is:
`sg[i] = 2*a[k*-i]` (shifted and truncated)

With the constraint `sum a = 1`, we need to project onto the simplex. The projection onto `{a >= 0, sum a = 1}` is the "Euclidean projection onto the simplex" which can be done in O(n log n).

But the subgradient method converges slowly for max problems. A better approach is the "smoothed max" (log-sum-exp):

`g(a) = tau * log(sum_k exp(conv[k]/tau))`

This is a smooth upper bound on `max_k conv[k]` and converges to it as tau → 0.

The gradient is:
`dg/da[j] = (1/tau) * sum_k exp(conv[k]/tau) * 2*a[k-j] / sum_k exp(conv[k]/tau)`
`= 2 * sum_k w[k] * a[k-j]`

where `w[k] = exp(conv[k]/tau) / sum exp(conv[k']/tau)` is a softmax weight.

This is a weighted average of the subgradients, which is much better than just the max subgradient.

For the constraint `sum a = 1`, we can:
- Use the Lagrangian: `L(a, lambda) = g(a) - lambda * (sum a - 1)`
- Or project onto the simplex after each step.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_benders(a, M, active_set, time_limit=15):
    """
    Solve LP with Benders decomposition: max sum(g) s.t. A*S g <= M, g >= 0.
    Returns (g, new_active_set, violated_indices).
    """
    n = len(a)
    if n > 2500:
        return None, active_set, []
    
    A_full = build_toeplitz(a)
    A_sub = A_full[active_set, :]
    
    c = -np.ones(n)
    b_ub = np.full(len(active_set), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status != 0:
            return None, active_set, []
        g = np.maximum(res.x, 0)
    except Exception:
        return None, active_set, []
    
    if np.sum(g) < 1e-12:
        return None, active_set, []
    
    # Check which constraints are violated
    conv_ag = compute_conv_from_toeplitz(A_full, g)
    violations = np.where(conv_ag > M + 1e-10)[0]
    
    # Update active set: keep violated + tight constraints
    tight = np.where(np.abs(conv_ag - M) < M * 0.01)[0]
    new_active = set(list(active_set) + list(violations) + list(tight[:50]))
    new_active = list(new_active)
    # Limit active set size

        new_active = sorted_by_violation[:200]
    
    return g / np.sum(g), new_active, violations


def compute_conv_from_toeplitz(A_full, g):
    """Compute A_full @ g without building the full conv."""
    return A_full @ g

    M = float(np.max(conv))
    top_k = min(100, 2 * n - 1)
    active_set = list(np.argsort(conv)[-top_k:])
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Re-initialize active set if it's growing too much or not helping
        if it % 10 == 0 or len(active_set) < 20:
            top_k = min(100 + it, 2 * n - 1)
            active_set = list(np.argsort(conv)[-top_k:])
        
        g, new_active, violations = solve_lp_benders(a, M, active_set, time_limit=12)
        active_set = new_active
        
        if g is None:
            stall += 1
            if stall > 5:
                break
            # Try with full constraint set
            if n <= 1000:
                A = build_toeplitz(a)
                c = -np.ones(n)
                b_ub = np.full(2 * n - 1, M)
                bounds = [(0.0, None)] * n
                try:
                    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                                  options={'presolve': True, 'time_limit': 15})
                    if res.status == 0:
                        g = np.maximum(res.x, 0)
                        if np.sum(g) > 1e-12:
                            g = g / np.sum(g)
                        else:
                            g = None
                except Exception:
                    g = None
            if g is None:
                break
            active_set = list(range(2 * n - 1))
        
        # Adaptive line search
        best_t_obj = best_obj
        best_t_val = 0.3
        best_a_local = best_a.copy()
        
        # Try multiple step sizes with focus on aggressive moves
        candidate_ts = np.concatenate([
            np.linspace(0.05, 0.5, 10),
            np.linspace(0.5, 1.0, 5)
        ])
        
        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            stall = 0
        else:
            stall += 1
            if stall > 8:
                # Try a fresh active set
                conv = compute_conv(a)
                active_set = list(np.argsort(conv)[-150:])
                stall = 0
            if stall > 20:
                break
    
    return best_a, best_obj


def fft_smoothed_gradient(a, tau):
    """
    Compute the gradient of the smoothed max objective:
    g(a) = tau * log(sum_k exp(conv[k]/tau))
    grad[j] = 2 * sum_m a[m] * w[m+j]
    where w = softmax(conv/tau)
    """
    n = len(a)
    conv = compute_conv(a)
    shifted = (conv - np.max(conv)) / tau
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= w_sum
    
    # Compute grad[j] = 2 * sum_m a[m] * w[m+j] using FFT
    # This is the cross-correlation of a and w
    N = 1
    while N < n + (2 * n - 1) - 1:
        N <<= 1
    
    fa = np.fft.fft(a, N)
    fw = np.fft.fft(w, N)
    corr = np.fft.ifft(fa * np.conj(fw)).real[:n]
    
    grad = 2.0 * corr
    return grad, conv, w


def smoothed_gradient_opt(a, max_iters=2000, tau_init=0.5, tau_final=0.01, 
                          time_budget=60.0, lr_init=0.05, momentum=0.9):
    """
    Smoothed gradient descent with Nesterov momentum on log-sum-exp objective.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive tau: start large, decrease
        progress = it / max_iters
        tau = tau_init * (tau_final / tau_init) ** progress
        # Learning rate decay
        lr = lr_init * (0.995 ** it)
        
        # Nesterov: compute gradient at lookahead point
        grad_y, conv_y, _ = fft_smoothed_gradient(y, tau)
        
        # Update velocity
        v = momentum * v + lr * grad_y
        
        # Take step from current point (not lookahead)
        y = a - v
        
        # Project
        a_new = project_simplex(y)
        a = a_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def smoothed_gradient_basic(a, max_iters=1000, tau=0.1, time_budget=30.0, lr=0.02):
    """Basic smoothed gradient without momentum (more stable for polish)."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, conv, w = fft_smoothed_gradient(a, tau)
        
        # Center the gradient (tangent space of sum=1)
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step size
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-12:
            break
        step = lr / (1.0 + gnorm * 0.1)
        
        a_new = project_simplex(a - step * grad_centered)
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj


def freq_domain_compress(a, max_iters=200, time_budget=30.0, blend=0.3):
    """
    Frequency domain compression: iteratively flatten the power spectrum
    and project back to non-negative.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        c = np.fft.ifft(fa * fa).real[:2 * n - 1]
        c = np.maximum(c, 0)
        
        c_max = np.max(c)
        if c_max < 1e-15:
            break
        
        # Compress peaks in time domain
        c_norm = c / c_max
        # Apply power compression (reduce dynamic range)
        alpha = 0.6 + 0.3 * (it / max_iters)
        c_compressed = c_norm ** alpha * c_max
        
        # Also reduce the top peak specifically
        top_idx = np.argmax(c_compressed)
        c_compressed[top_idx] *= (1 - blend * 0.5)
        
        # Reconstruct a via complex square root
        cf = np.fft.fft(c_compressed, N)
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a, best_obj


def freq_spectral_equalization(a, max_iters=150, time_budget=25.0):
    """
    Equalize the magnitude spectrum: drive |FFT(a)| toward a constant.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = np.fft.fft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # Compute target magnitude: geometric mean
        log_mag = np.log(mag + 1e-15)
        target_log = np.mean(log_mag)
        
        # Blend toward target
        alpha = 0.3 + 0.4 * (it / max_iters)
        new_log = (1 - alpha) * log_mag + alpha * target_log
        new_mag = np.exp(new_log)
        
        # Reconstruct
        fa_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a, best_obj


def chase_peak(a, n_steps=50, step_size=0.03):
    """Chase-the-peak: iteratively reduce the maximum convolution peak."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
        
        grad_centered = grad - np.mean(grad)
        
        for ss in [step_size, step_size * 0.5, step_size * 2.0]:
            a_try = project_simplex(a - ss * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
    
    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine-modulated (flat spectrum design)
    x = np.linspace(0, 1, n)
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            c = (1.0 + amp * np.cos(2 * np.pi * freq * x)) 
            c = np.maximum(c, 0.01)
            inits.append(c / c.sum())
    
    # Gaussian
    xg = np.linspace(-3, 3, n)
    gauss = np.exp(-xg**2 / 3)
    inits.append(gauss / gauss.sum())
    
    # Sinc-like (main lobe)
    xs = np.linspace(-5, 5, n)
    sinc = np.sinc(xs / 5.0)
    sinc = np.maximum(sinc, 0)
    if sinc.sum() > 0:
        inits.append(sinc / sinc.sum())
    
    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())
    
    # Two-lobe
    lobe1 = np.exp(-((x - 0.3) ** 2) / 0.01)
    lobe2 = np.exp(-((x - 0.7) ** 2) / 0.01)
    two_lobe = lobe1 + lobe2
    inits.append(two_lobe / two_lobe.sum())
    
    # Cosine window
    cos_win = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(cos_win / cos_win.sum())
    
    # Exponential decay
    exp_dec = np.exp(-3 * x)
    inits.append(exp_dec / exp_dec.sum())
    
    if parent is not None:
        # Resampled parent
        inits.append(resample_sequence(parent, n))
        # Perturbed parent
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_simplex(pert))
    
    return inits


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def multi_strategy_optimize(a, n, time_budget, rng):
    """
    Apply multiple optimization strategies in sequence.
    Returns (best_a, best_obj).
    """
    best_a = project_simplex(a.copy())
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    
    strategies = [
        ('lp', lambda: lp_alternate_improved(best_a, max_iters=40, time_budget=time_budget * 0.25)),
        ('smoothed', lambda: smoothed_gradient_opt(best_a, max_iters=500, tau_init=0.3, 
                                                    tau_final=0.02, time_budget=time_budget * 0.2)),
        ('freq_compress', lambda: freq_domain_compress(best_a, max_iters=100, time_budget=time_budget * 0.15)),
        ('chase', lambda: chase_peak(best_a, n_steps=40, step_size=0.025)),
        ('smooth_basic', lambda: smoothed_gradient_basic(best_a, max_iters=300, tau=0.05, 
                                                          time_budget=time_budget * 0.15)),
        ('freq_eq', lambda: freq_spectral_equalization(best_a, max_iters=100, time_budget=time_budget * 0.15)),
    ]
    
    for round_num in range(3):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        for si, (name, func) in enumerate(strategies):
            if time.monotonic() - start > time_budget:
                break
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 3:
                break
            
            # Skip LP for large n
            if name == 'lp' and n > 1500:
                # Use benders version with fewer constraints
                obj_try = compute_obj(best_a)
                continue
            
            t_alloc = min(remaining * 0.4, time_budget * 0.25)
            try:
                a_new, obj_new = func()
            except Exception:
                a_new, obj_new = best_a, best_obj
            
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
    
    return best_a, best_obj

            if rem > 5 and n_try <= 1200:
                a_opt2, obj_opt2 = lp_alternate_improved(a_opt, max_iters=20, 
                                                          time_budget=min(rem * 0.4, 20.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2
            
            rem = phase4_start + phase4_budget - time.monotonic()
            if rem > 3:
                a_opt3, obj_opt3 = smoothed_gradient_basic(a_opt, max_iters=500, tau=0.05,
                                                            time_budget=min(rem * 0.3, 15.0))
                if obj_opt3 < obj_opt:
                    a_opt, obj_opt = a_opt3, obj_opt3
            
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 10 and best_seq is not None:
        a_final = project_simplex(best_seq.copy())
        n_final = len(a_final)
        
        polish_rounds = min(15, int(remaining5 / 8))
        for pr in range(polish_rounds):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            
            # LP polishing
            if n_final <= 1200:
                a_p, obj_p = lp_alternate_improved(a_final, max_iters=min(20, int(rem / 2)),
                                                    time_budget=min(rem * 0.25, 20.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish LP r{pr}: {best_score:.8f}")
            
            # Fine gradient
            rem = deadline - time.monotonic()
            if rem > 3:
                a_p2, obj_p2 = smoothed_gradient_basic(a_final, max_iters=500, tau=0.03,
                                                        time_budget=min(rem * 0.2, 10.0))
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  Polish grad r{pr}: {best_score:.8f}")
            
            # Chase peak
            rem = deadline - time.monotonic()
            if rem > 3:
                a_p3, obj_p3 = chase_peak(a_final, n_steps=20, step_size=0.01)
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    a_final = a_p3.copy()
            
            # Small perturbation + re-optimize
            rem = deadline - time.monotonic()
            if rem > 8:
                a_pert = a_final * np.exp(rng.normal(0, 0.01, n_final))
                a_pert = project_simplex(a_pert)
                a_pert, obj_pert = chase_peak(a_pert, n_steps=20, step_size=0.015)
                if n_final <= 1000:
                    a_pert2, obj_pert2 = lp_alternate_improved(a_pert, max_iters=10,
                                                               time_budget=min(rem * 0.3, 10.0))
                    if obj_pert2 < obj_pert:
                        a_pert, obj_pert = a_pert2, obj_pert2
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_final = a_pert.copy()
                    print(f"  Polish perturb r{pr}: {best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
        print(f"  Validated: {val:.8f}")
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0
        print(f"  FALLBACK to uniform n=100, score=2.0")
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
