def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)

    x_old = np.linspace(0, 1, n)

def build_cross_matrix(a):
    """Build matrix where row k gives coefficients for conv(a, g)[k]"""
    n = len(a)
    # For output k, conv[a,g][k] = sum_j a[k-j] * g[j]
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # This is a Toeplitz-like matrix (actually it's the "cross" Toeplitz)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # For this j, a[k-j] is valid when 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = min(j + n, 2*n-1)
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def build_toeplitz_cross(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def lp_best_response(a, time_limit=10):
    n = len(a)
    if n > 3000:
        return None, None
    A = build_toeplitz_cross(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            M = float(res.x[n])
            s = g.sum()
            if s > 1e-10:
                return g / s, M
    except Exception:
        pass
    return None, None

def lp_alternate(a, max_iters=100, time_budget=60.0, blend_factor=1.0):
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        g, M_lp = lp_best_response(a, time_limit=min(remaining * 0.6, 15))
        if g is None:
            break
        
        # Try different blend factors
        best_local = None
        best_local_obj = best_obj
        for t in [blend_factor, blend_factor * 0.5, 0.3, 0.1, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local = a_try.copy()
        
        if best_local is not None:
            best_obj = best_local_obj
            best_a = best_local.copy()
            a = best_local.copy()
        else:
            # No improvement, try smaller step
            break
    
    return best_a, best_obj

import numpy as np

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_cross_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_best_response(a, time_limit=10):
    n = len(a)
    if n > 2500:
        return None, None
    A = build_cross_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            M = float(res.x[n])
            s = g.sum()
            if s > 1e-10:
                return g / s, M
    except Exception:
        pass
    return None, None


def lp_slack_response(a, slack_weight=0.1, time_limit=10):
    n = len(a)
    if n > 2500:
        return None, None
    A = build_cross_matrix(a)
    n_slack = 2 * n - 1
    n_vars = n + 1 + n_slack
    c = np.zeros(n_vars)
    c[-1] = 1.0
    c[n + 1:] = slack_weight
    A_ub = np.zeros((n_slack, n_vars))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    A_ub[:, n + 1:] = -np.eye(n_slack)
    b_ub = np.zeros(n_slack)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)] + [(0, None)] * n_slack
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def compute_gradient_fft(a, weights):
    n = len(a)
    a_flip = a[::-1]
    N = 1
    while N < n + len(weights) - 1:
        N <<= 1
    result = irfft(rfft(a_flip, N) * rfft(weights, N), N)
    grad = 2.0 * result[n - 1:2 * n - 1]
    return grad


def gradient_descent(a, max_iters=200, time_budget=30.0, tau_init=0.005,
                     step_init=0.01, decay=0.995):
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = step_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        tau = tau_init * (1.0 + 0.5 * it / max_iters)
        shifted = np.clip((conv - c_max) / tau, -400.0, 400.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        grad = compute_gradient_fft(a, w)
        grad_centered = grad - grad.mean()

        a_new = a - step * grad_centered
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        step *= decay

    return best_a, best_obj


def critical_lag_descent(a, max_iters=100, time_budget=30.0, top_k=3):
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.02

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        threshold = c_max * 0.95
        active = np.where(conv >= threshold - 1e-10)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]

        grad_centered = grad - grad.mean()
        a_new = a - step * grad_centered
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        step *= 0.99

    return best_a, best_obj

def generate_initializations(n, rng, parent=None):
    inits = []
    inits.append(np.ones(n) / n)
    x = np.linspace(0, 1, n)
    inits.append(np.exp(-4 * x) / np.exp(-4 * x).sum())
    inits.append(np.exp(-4 * (1 - x)) / np.exp(-4 * (1 - x)).sum())
    inits.append(np.sin(np.pi * x) ** 2 / (np.sin(np.pi * x) ** 2).sum())
    inits.append(np.exp(-((x - 0.3) ** 2) / 0.03) / np.exp(-((x - 0.3) ** 2) / 0.03).sum())
    inits.append(np.exp(-((x - 0.7) ** 2) / 0.03) / np.exp(-((x - 0.7) ** 2) / 0.03).sum())
    inits.append((x ** 3) / (x ** 3).sum())
    inits.append(((1 - x) ** 3) / ((1 - x) ** 3).sum())
    for freq in [2, 3, 5]:
        shape = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
        inits.append(shape / shape.sum())
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())
    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.08, n))
        inits.append(project_nonneg_sum1(pert))
    return inits
