"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with FFT-accelerated log-sum-exp gradient (O(n log n) per iteration)
2. Full LP best-response (all 2n-1 constraints) with HiGHS solver
3. 1D coordinate descent with golden-section search
4. Multi-start, multi-tau continuation, and adaptive restarts
Key improvement: FFT-based gradient computation enables 10-100x more FISTA
iterations within the time budget, leading to faster convergence.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    if not (~cond).any():
        rho = n - 1
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """
    Compute smooth max (log-sum-exp) of conv(a,a) and its gradient.
    Uses FFT for O(n log n) gradient computation.
    Returns (smooth_value, gradient).
    """
    n = len(a)
    # Need L >= 3n-2 for the gradient convolution
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    
    # Compute b = conv(a, a) using FFT
    fa = np.fft.rfft(a, L)
    b = np.fft.irfft(fa * fa, L)[:2 * n - 1]
    
    # Log-sum-exp for numerical stability
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (with a[m]=0 outside [0,n-1])
    # = 2 * (a_flip * p)[n-1+i] where a_flip = a[::-1]
    a_flip = a[::-1]
    faf = np.fft.rfft(a_flip, L)
    fp = np.fft.rfft(p, L)
    conv_result = np.fft.irfft(faf * fp, L)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    
    return smooth_val, grad


def run_fista(a_init, tau_schedule, max_iters, deadline, lr=0.3):
    """
    FISTA (Nesterov accelerated) on smooth max approximation.
    tau_schedule: list of (tau_start, tau_end, n_iters) tuples for continuation.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    it = 0
    
    for tau_s, tau_e, n_seg in tau_schedule:
        if time.monotonic() > deadline:
            break
        for seg_it in range(n_seg):
            if it >= max_iters:
                break
            if time.monotonic() > deadline:
                break
            
            frac = (seg_it + 1) / n_seg if n_seg > 0 else 1.0
            tau = tau_s * (tau_e / tau_s) ** frac
            
            _, grad_y = smooth_max_grad_fft(y, tau)
            
            # Project gradient to tangent space of simplex
            grad_tan = grad_y - grad_y.mean()
            
            # Step size: Lipschitz constant is roughly 4*L where L is the
            # max weight. Use conservative step.
            step = lr / (1.0 + tau * 0.01)
            
            a_new = y - step * grad_tan
            a_new = proj_simplex(a_new)
            
            # Nesterov momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            alpha = (t_mom - 1.0) / t_new
            y_new = a_new + alpha * (a_new - a_prev)
            y_new = proj_simplex(y_new)
            
            a_prev = a
            a = a_new
            y = y_new
            t_mom = t_new
            it += 1
            
            # Track best
            if it % 10 == 0:
                M_cur = fast_conv(a).max()
                if M_cur < best_M:
                    best_M = M_cur
                    best_a = a.copy()
    
    # Final check
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()
    
    return best_a

def build_full_lp_matrix(a):
    """Build the full (2n-1) x n constraint matrix for the LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, n_active=None, time_limit=10):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0.
    If n_active is None, use all constraints.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    
    if n_active is None or n_active >= 2 * n - 1:
        A = build_full_lp_matrix(a)
        b_ub = np.full(2 * n - 1, M)
    else:
        b = fast_conv(a)
        top_idx = np.argsort(b)[::-1][:n_active]
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(top_idx):
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])
        A = csr_matrix((vals, (rows, cols)), shape=(n_active, n))
        b_ub = np.full(n_active, M)
    
    if A.nnz == 0:
        return None
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_iteration(a, M, deadline, n_active=None, time_limit=10):
    """Single LP iteration: find best response and mix."""
    g = lp_best_response(a, M, n_active=n_active, time_limit=time_limit)
    if g is None:
        return a, M, False
    
    # Line search for mixing parameter
    best_t = 0.0
    best_M_trial = M
    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            continue
        a_trial = a_trial / s_t
        M_trial = fast_conv(a_trial).max()
        if M_trial < best_M_trial:
            best_M_trial = M_trial
            best_t = t
    
    if best_t > 0:
        a_new = (1.0 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        return a_new, best_M_trial, True
    return a, M, False


def run_lp_loop(a_init, deadline, n_active=None, max_iters=50, time_limit=10):
    """Run LP iterations until convergence or time limit."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a, M, improved = run_lp_iteration(a, best_M, deadline, n_active, time_limit)
        
        if improved and M < best_M:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                break
    
    return best_a


def coordinate_descent_1d(a, b, n):
    """
    Full coordinate descent sweep. For each coordinate i, find optimal value.
    Uses the fact that the objective as a function of a[i] is convex (max of quadratics).
    """
    a = a.copy()
    b = b.copy()  # current convolution
    
    for i in range(n):
        x_cur = a[i]
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        
        # Unaffected max
        if k_lo > 0 or k_hi < 2 * n - 2:
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[k_lo:k_hi + 1] = False
            base_max = b[mask].max() if mask.any() else 0.0
        else:
            base_max = 0.0
        
        # Precompute constants and slopes
        c_vals = np.zeros(k_hi - k_lo + 1)
        s_vals = np.zeros(k_hi - k_lo + 1)
        for idx, k in enumerate(range(k_lo, k_hi + 1)):
            if k == 2 * i:
                c_vals[idx] = b[k] - x_cur * x_cur
                s_vals[idx] = 0.0
            else:
                j = k - i
                if 0 <= j < n:
                    c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                    s_vals[idx] = 2.0 * a[j]
                else:
                    c_vals[idx] = b[k]
                    s_vals[idx] = 0.0
        
        def obj_at(x):
            """Max convolution value when a[i] = x."""
            vals = c_vals + s_vals * x
            m = base_max
            if vals.size > 0:
                vm = vals.max()
                if vm > m:
                    m = vm
            # Handle the quadratic term at k = 2*i
            if 2 * i >= k_lo and 2 * i <= k_hi:
                idx_q = 2 * i - k_lo
                qv = c_vals[idx_q] + x * x
                if qv > m:
                    m = qv
            return m
        
        # Golden section search for min of convex function on [0, hi]
        lo = 0.0
        hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.5)
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_pt = hi - gr * (hi - lo)
        b_pt = lo + gr * (hi - lo)
        fa = obj_at(a_pt)
        fb = obj_at(b_pt)
        
        for _ in range(30):
            if fa < fb:
                hi = b_pt
                b_pt = a_pt
                fb = fa
                a_pt = hi - gr * (hi - lo)
                fa = obj_at(a_pt)
            else:
                lo = a_pt
                a_pt = b_pt
                fa = fb
                b_pt = lo + gr * (hi - lo)
                fb = obj_at(b_pt)
        
        x_opt = (lo + hi) / 2.0
        if obj_at(0.0) <= obj_at(x_opt):
            x_opt = 0.0
        
        if abs(x_opt - x_cur) > 1e-12:
            # Update b
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    b[k] = c_vals[k - k_lo] + x_opt * x_opt
                else:
                    j = k - i
                    if 0 <= j < n:
                        b[k] = c_vals[k - k_lo] + s_vals[k - k_lo] * x_opt
            a[i] = x_opt
    
    return a


def run_coord_descent(a_init, max_sweeps, deadline):
    """Run coordinate descent."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        a = coordinate_descent_1d(a, b, n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed parent
        for sigma in [0.02, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Parabolic (concave)
    poly = t * (1 - t) + 0.05
    inits.append(poly / poly.sum())
    
    # U-shaped (convex)
    poly2 = (2 * t - 1) ** 2 + 0.05
    inits.append(poly2 / poly2.sum())
    
    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Sawtooth
    saw = (t % 0.25) / 0.25
    inits.append(saw / saw.sum())
    
    # Bent cosine
    bent = 0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t)
    bent = np.maximum(bent, 0.01)
    inits.append(bent / bent.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:

            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s (was {prev:.8f})", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
