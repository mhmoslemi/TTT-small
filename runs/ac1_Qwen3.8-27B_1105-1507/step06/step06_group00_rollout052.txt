def solve_lp(a):
    """Solve: min M s.t. A@g <= M, sum(g)=1, g>=0, M>=0"""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize: M -> c = [0, ..., 0, 1]
    # Constraints: A@g - M <= 0 -> [A, -I] @ [g, M] <= 0
    #              sum(g) = 1 -> [1, ..., 1, 0] @ [g, M] = 1
    #              g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None

        size <<= 1

def solve_lp_fast(a):
    """Solve min-peak LP using HiGHS."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build convolution matrix A (2n-1 x n)
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a_norm
    
    # LP: min M
    # s.t. A@g - M <= 0
    #      sum(g) = 1
    #      g >= 0, M >= 0
    c = np.zeros(n + 1)

    # [A, -I] @ [g, M] <= 0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    
    # A_eq @ x = b_eq
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1),
                  A_eq=A_eq, b_eq=np.array([1.0]),
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None

"""
Optimized LP-alternating + smooth refinement for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses HiGHS LP solver, FFT-based convolution, multi-n exploration, and L-BFGS-B polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp(a, time_limit=None):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.empty((2*n-1, n+1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    opts = {}
    if time_limit is not None:
        opts['time_limit'] = time_limit
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options=opts)
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        s = float(np.sum(g))
        if s > 1e-12:
            return g / s
    return None

def lp_alternate(a, time_budget, max_iters=500, verbose=False):
    """
    LP alternating optimization.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    prev_obj = best_obj
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        # Solve LP
        g = solve_lp(a)
        if g is None:
            break
        
        # Try different t values
        best_t_obj = np.inf
        best_t_a = None
        
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-12:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:

                best_t_a = a_try
        
        if best_t_a is None:

        if best_t_obj < best_obj - 1e-12:

            a = best_t_a

            a = best_t_a  # Still move, might help escape

            if stall_count > 15:

                a = best_a * np.exp(rng.normal(0, 0.05, n))

                if s > 1e-12:

                best_a = a.copy()
                best_obj = compute_obj(a)

def chase_peak(a, time_budget, max_iters=500):
    """
    Chase-the-peak subgradient method.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.02
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: d(max_conv)/da[i] = 2*a[k_star - i] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Center gradient (project off the sum=const direction)
        grad = grad - np.mean(grad)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-12:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step = min(step * 1.1, 0.1)
            else:
                step *= 0.85
                if step < 1e-6:
                    break
        else:
            step *= 0.5
    
    return best_a

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using fast LP alternation (HiGHS),
chase-peak subgradient refinement, smooth L-BFGS-B polish, multi-n exploration,
and structured random restarts.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.empty((2 * n - 1, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        s = float(np.sum(g))
        if s > 1e-12:
            return g / s
    return None

def chase_peak(a, time_budget, max_iters=1000):
    """Chase-the-peak subgradient method with adaptive step."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.015

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient of max_conv w.r.t. a[i] = 2*a[k_star - i]
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * a[k_star - i + 1 - 1]  # a[k_star - i] indexed properly
        # Fix: grad[i] = 2*a[k_star - i] for valid i
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Project: remove mean to stay on sum=const
        grad = grad - np.mean(grad)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-12:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step = min(step * 1.05, 0.08)
            else:
                step *= 0.9
                if step < 1e-7:
                    break
        else:
            step *= 0.5

    return best_a

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

        print(f"Default init (n=800): obj={best_score:.10f}")

    n_parent = len(best_seq)

    # Phase 1: Intensive LP alternation on parent
    print(f"\n=== Phase 1: LP Alternation (n={n_parent}) ===")
    p1_budget = min(450.0, (deadline - start_time) * 0.35)
    a = best_seq.copy()
    for rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p1_budget / (3 - rnd), remaining * 0.7)
        a_new = lp_alternate(a, time_budget=alloc, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP round {rnd}: obj={best_score:.10f}")
        else:
            print(f"  LP round {rnd}: obj={obj_new:.10f} (no improvement)")
            a = a_new

    # Phase 2: Chase-peak refinement
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 2: Chase-Peak ===")
        a_cp = chase_peak(best_seq, time_budget=min(remaining * 0.08, 60.0))
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score - 1e-12:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  Chase-peak: obj={best_score:.10f}")
        else:
            print(f"  Chase-peak: obj={obj_cp:.10f}")

    # Phase 3: Multi-n exploration
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_options = sorted(set([
        max(200, n_parent - 50), max(200, n_parent - 25), n_parent,
        n_parent + 25, n_parent + 50, n_parent + 75, n_parent + 100,
        700, 800, 900, 1000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1200]
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample(best_seq, n_try)

        alloc = min(remaining * 0.15, 80.0)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        # Quick chase-peak on this
        if compute_obj(a_opt) < best_score or abs(n_try - n_parent) <= 50:
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                a_cp2 = chase_peak(a_opt, time_budget=min(remaining2 * 0.1, 20.0))
                obj_try = min(obj_try, compute_obj(a_cp2))
                if compute_obj(a_cp2) < obj_try:
                    a_opt = a_cp2

        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # Phase 4: Smooth L-BFGS-B refinement
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Smooth L-BFGS-B ===")
        a_sm = smooth_refine(best_seq, time_budget=min(remaining * 0.15, 80.0))
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score - 1e-12:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Smooth: obj={best_score:.10f}")
        else:
            print(f"  Smooth: obj={obj_sm:.10f}")

    # Phase 5: Random restarts with structured inits
    print(f"\n=== Phase 5: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    restart_budgets = max(10, (remaining) / 6)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < restart_budgets + 10:
            break

        if r == 0:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            # Different n, resampled + perturbed
            n_r = n_best + int(rng.integers(-50, 50))
            n_r = max(300, min(1100, n_r))
            a_r = resample(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.03, n_r))
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / np.sum(a_r)
        elif r == 2:
            # Bimodal structure
            n_r = n_best
            x = np.arange(n_r) / n_r
            a_r = np.exp(-8.0 * (x - 0.15)**2) + np.exp(-8.0 * (x - 0.85)**2) + 0.3
            a_r = a_r / np.sum(a_r)
        elif r == 3:
            # Cosine window
            n_r = n_best
            x = np.linspace(0, np.pi, n_r)
            a_r = np.cos(x / 2.0) ** 2 + 0.2
            a_r = a_r / np.sum(a_r)
        elif r == 4:
            # Flat with center dip
            n_r = n_best
            a_r = np.ones(n_r)
            center = n_r // 2
            width = n_r // 8
            a_r[center - width:center + width] = 0.5
            a_r = a_r / np.sum(a_r)
        else:
            # Random smooth
            n_r = n_best
            a_r = np.cumsum(rng.uniform(0.5, 1.5, n_r))
            a_r = np.maximum(a_r - a_r[0] + 0.1, 0.01)
            a_r = a_r / np.sum(a_r)

        a_opt = lp_alternate(a_r, time_budget=restart_budgets)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score - 1e-12:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (n={len(a_opt)}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.10f}")

    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP alternate
        alloc = min(remaining * 0.4, 120.0)
        a_fp = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_fp = compute_obj(a_fp)
        if obj_fp < best_score - 1e-12:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final LP: obj={best_score:.10f}")

        # Chase peak
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_cp = chase_peak(best_seq, time_budget=min(remaining * 0.2, 30.0))
            obj_cp = compute_obj(a_cp)
            if obj_cp < best_score - 1e-12:
                best_score = obj_cp
                best_seq = a_cp.copy()
                print(f"  Final CP: obj={best_score:.10f}")

        # Smooth
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sm = smooth_refine(best_seq, time_budget=min(remaining * 0.3, 40.0), beta=150.0)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score - 1e-12:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Final Smooth: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
