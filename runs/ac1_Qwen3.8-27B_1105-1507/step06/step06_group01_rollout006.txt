from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import time

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response_cutting_plane(a, M, n, max_iterations=100, time_limit=30.0):
    """Solve the LP best-response using cutting-plane method.
    
    Finds g >= 0 that maximizes sum(g) subject to (a*g)[k] <= M for ALL k.
    Uses cutting-plane: start with a subset of constraints, add violated ones.
    """
    t_start = time.monotonic()
    
    # Initial active set: top 30 convolution values of a
    b = fast_conv(a)
    top_k = min(50, 2*n-1)
    active_set = set(np.argsort(b)[::-1][:top_k].tolist())
    
    best_g = None
    best_obj = -np.inf
    
    for iteration in range(max_iterations):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Build constraint matrix
        # (a*g)[k] = sum_j a[m] * g[j] where m = k-j, 0 <= m < n
        # For each active k, the constraint is: sum_{j: 0<=k-j<n} a[k-j] * g[j] <= M
        n_active = len(active_set)
        if n_active == 0:
            return None
        
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_set):
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                m = k - j
                if 0 <= m < n:
                    rows.append(idx)
                    cols.append(j)
                    vals.append(a[m])
        
        A = csr_matrix((vals, (rows, cols)), shape=(n_active, n))

        c = -np.ones(n)

                if best_g is not None:
                    return best_g
                continue

                return best_g
            continue
        
        g = np.maximum(r.x, 0.0)

        if obj > best_obj:
            best_obj = obj
            best_g = g.copy()

        b_test = np.convolve(a, g, mode='full')
        violations = np.where(b_test > M + 1e-10)[0]

                    return best_g
                # Try with just the most binding constraints
                active_list = active_list[:max(10, n_act // 2)]
                continue
        except Exception:
            if best_g is not None:
                return best_g
            continue
        
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            return None
        if g_sum > best_obj:
            best_obj = g_sum
            best_g = g.copy()
        
        # Check for violated constraints
        b_test = np.convolve(a, g, mode='full')
        violations = np.where(b_test > M + 1e-9)[0]
        
        if len(violations) == 0:
            return best_g
        
        # Add violated constraints
        for v in violations[:30]:
            if int(v) not in active_list:
                active_list.append(int(v))
        
        # Prevent active set from growing too large
        if len(active_list) > 300:
            # Keep only near-binding constraints
            b_test_act = b_test[active_list]
            slack = M - b_test_act
            # Keep constraints with slack < 0.3*M
            binding_thresh = 0.3 * M
            new_active = [k for k, s in zip(active_list, slack) if s < binding_thresh]
            # Always keep top 20
            top = list(np.argsort(b_test_act)[::-1][:20])
            for t in top:
                if active_list[t] not in new_active:
                    new_active.append(active_list[t])
            active_list = new_active[:200]
    
    return best_g


def mirror_descent_step(a, g, M, n):
    """Adaptive line search for a_new = (1-t)*a + t*g.
    
    Returns (a_new, M_new, t_used).
    """
    s_a = a.sum()
    s_g = g.sum()
    
    best_a = a.copy()
    best_M = M
    best_t = 0.0
    
    # Try a range of t values with golden-section-like refinement
    t_candidates = [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.04, 0.03, 0.02]
    
    # Coarse search
    for t in t_candidates:
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            continue
        a_trial = a_trial / s_t
        b_trial = fast_conv(a_trial)
        M_trial = b_trial.max()
        if M_trial < best_M:
            best_M = M_trial
            best_a = a_trial.copy()
            best_t = t
    
    if best_t == 0.0:
        return a, M, 0.0
    
    # Fine search around best_t
    t_lo = max(0.0, best_t * 0.5)
    t_hi = min(1.0, best_t * 2.0)
    
    for _ in range(10):
        t_mid = (t_lo + t_hi) / 2.0
        t_q1 = t_lo + (t_hi - t_lo) / 3.0
        t_q2 = t_hi - (t_hi - t_lo) / 3.0
        
        a_q1 = np.maximum((1.0 - t_q1) * a + t_q1 * g, 0.0)
        a_q1 = a_q1 / max(a_q1.sum(), 1e-15)
        M_q1 = fast_conv(a_q1).max()
        
        a_q2 = np.maximum((1.0 - t_q2) * a + t_q2 * g, 0.0)
        a_q2 = a_q2 / max(a_q2.sum(), 1e-15)
        M_q2 = fast_conv(a_q2).max()
        
        if M_q1 < M_q2:
            t_hi = t_q2
        else:
            t_lo = t_q1
    
    t_final = (t_lo + t_hi) / 2.0
    a_final = np.maximum((1.0 - t_final) * a + t_final * g, 0.0)
    a_final = a_final / max(a_final.sum(), 1e-15)
    M_final = fast_conv(a_final).max()
    
    if M_final < best_M:
        best_a = a_final
        best_M = M_final
        best_t = t_final
    
    return best_a, best_M, best_t


def run_lp_search(a_init, deadline, verbose=True):
    """Main LP mirror descent search loop."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    a = a / a.sum()
    n = len(a)
    
    b = fast_conv(a)
    M = b.max()
    best_a = a.copy()
    best_M = M
    
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        # Get best response
        g = lp_best_response_cp(a, M, n, max_iters=30, time_limit=10.0)
        if g is None:
            no_improve += 5
            if no_improve > 20:
                break
            continue
        
        g = g / g.sum()
        
        # Mirror descent step with line search
        a_new, M_new, t_used = mirror_descent_step(a, g, M, n)
        
        if M_new < M * (1 - 1e-12):
            a = a_new
            M = M_new
            if M < best_M:
                best_M = M
                best_a = a.copy()
                if verbose and it % 50 == 0:
                    elapsed = time.monotonic() - (deadline - 1500)
                    print(f"    iter {it}: M={M:.12f} t={t_used:.4f}", flush=True)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Try smaller step
                a_new2 = (1.0 - 0.01) * a + 0.01 * g
                a_new2 = np.maximum(a_new2, 0.0)
                a_new2 = a_new2 / a_new2.sum()
                M_new2 = fast_conv(a_new2).max()
                if M_new2 < M:
                    a = a_new2
                    M = M_new2
                    no_improve = 0
                else:
                    break
        
        if time.monotonic() > deadline - 10:
            break
    
    return best_a, best_M


def spectral_init(n, rng):
    """Initialize via spectral flattening."""
    # Start with random positive sequence
    a = rng.uniform(0.5, 1.5, n)
    
    for _ in range(30):
        # Compute aperiodic autocorrelation
        b = fast_conv(a)
        M = b.max()
        avg = b.mean()
        
        # In frequency domain: |F(a)|^2 is related to periodic autocorrelation
        # We want to flatten the spectrum magnitude
        fa = np.fft.rfft(a, 2 * n)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # Flatten: new_mag = (mag * avg/M)^alpha
        alpha = 0.5
        target = np.ones_like(mag) * np.sqrt(avg)
        new_mag = mag * (target / np.maximum(mag, 1e-10)) ** alpha
        new_mag = np.maximum(new_mag, 1e-8)
        
        a = np.fft.irfft(new_mag * np.exp(1j * phase), 2 * n)[:n]
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    
    return a


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0.0)
        if len(p) != n:
            # Resample
            t_old = np.linspace(0, 1, len(p))
            p = np.maximum(np.interp(t, t_old, p), 0.0)
        p = p / max(p.sum(), 1e-10)
        inits.append(p.copy())
        p_noisy = np.maximum(p + rng.normal(0, 0.05, n), 0.01)
        inits.append(p_noisy / p_noisy.sum())
        p_noisy2 = np.maximum(p + rng.normal(0, 0.15, n), 0.01)
        inits.append(p_noisy2 / p_noisy2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Spectral init
    inits.append(spectral_init(n, rng))
    
    # Bimodal (mass at edges)
    bimodal = 0.3 + 1.5 * (np.abs(t - 0.5) ** 2)
    inits.append(bimodal / bimodal.sum())
    
    # Cosine window
    cos_w = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
    cos_w = np.maximum(cos_w, 0.05)
    inits.append(cos_w / cos_w.sum())
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize * 0.5)) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0.5, 1.5, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # "Blazed" - encourage flat autocorrelation
    blazed = np.ones(n)
    # Reduce center more than edges
    center_mask = np.abs(t - 0.5) < 0.2
    blazed[center_mask] *= 0.6
    inits.append(blazed / blazed.sum())
    
    return inits


def coordinate_descent_polish(a, deadline):
    """Coordinate descent polish: optimize one coordinate at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / a.sum()
    n = len(a)
    
    b = fast_conv(a)
    best_a = a.copy()
    best_M = b.max()
    
    max_sweeps = 50
    sweeps = 0
    
    while sweeps < max_sweeps and time.monotonic() < deadline:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            x_cur = a[i]
            # Compute affected convolution values
            # (a*a)[k] changes for k in [i, i + n - 1]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # For each affected k, express (a*a)[k] as function of x = a[i]
            # (a*a)[k] = 2*x*a[k-i] + (constant) for k != 2i
            # (a*a)[k] = x^2 + (constant) for k = 2i
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                base_max = b[mask].max() if mask.any() else 0.0
            else:
                base_max = 0.0
            
            # Build linear/quadratic terms
            c_vals = []
            slopes = []
            quad_at = -1
            for k in range(k_lo, k_hi + 1):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        c = b[k] - x_cur * x_cur
                        slopes.append(0.0)
                        c_vals.append(c)
                        quad_at = k - k_lo
                    else:
                        c = b[k] - 2.0 * x_cur * a[j]
                        c_vals.append(c)
                        slopes.append(2.0 * a[j])
                else:
                    c_vals.append(b[k])
                    slopes.append(0.0)
            
            c_arr = np.array(c_vals) if c_vals else np.array([0.0])
            s_arr = np.array(slopes) if slopes else np.array([0.0])
            
            # Objective: max(base_max, max_j(c_j + s_j * x), x^2 if quad)
            def obj(x_test):
                vals = c_arr + s_arr * x_test
                m = base_max
                if vals.size > 0:
                    vm = vals.max()
                    if vm > m:
                        m = vm
                if quad_at >= 0:
                    qv = c_arr[quad_at] + x_test * x_test
                    if qv > m:
                        m = qv
                return m
            
            # Golden section search on [0, x_hi]
            x_hi = max(2.0 * x_cur, 0.5, np.sqrt(max(b.max(), 0.0)) * 1.5)
            x_lo = 0.0
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = x_hi - gr * (x_hi - x_lo)
            c2 = x_lo + gr * (x_hi - x_lo)
            fc1 = obj(c1)
            fc2 = obj(c2)
            
            for _ in range(30):
                if fc1 < fc2:
                    x_hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = x_hi - gr * (x_hi - x_lo)
                    fc1 = obj(c1)
                else:
                    x_lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = x_lo + gr * (x_hi - x_lo)
                    fc2 = obj(c2)
            
            x_opt = (x_lo + x_hi) / 2.0
            if obj(0.0) <= obj(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - x_cur) > 1e-10:
                # Update b values
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            b[k] = b[k] - x_cur * x_cur + x_opt * x_opt
                        else:
                            b[k] = b[k] - 2.0 * x_cur * a[j] + 2.0 * x_opt * a[j]
                a[i] = x_opt
                improved = True
        
        # Renormalize
        s = a.sum()
        if s > 1e-10:
            a = a / s
            b = fast_conv(a)
            M = b.max()
            if M < best_M * (1 - 1e-10):
                best_M = M
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: diverse inits -> LP mirror descent -> CD polish.
    Uses cutting-plane LP for exact best response.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    deadline = t_global_start + float(budget_s) - 30.0
    
    best_a = None
    best_score = float('inf')
    
    def try_update(a):
        nonlocal best_a, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_global_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        try_update(parent)
    
    n_parent = parent.size
    
    # Determine lengths to try
    lengths = []
    if n_parent > 0:
        lengths.append(n_parent)
    for dl in [-64, 64, 128, -128, 192]:
        nl = n_parent + dl
        if 256 <= nl <= 2048 and nl not in lengths:
            lengths.append(nl)
    if 768 not in lengths:
        lengths.append(768)
    lengths.sort(key=lambda x: abs(x - n_parent))
    lengths = lengths[:5]
    
    print(f"\n{'='*60}", flush=True)
    print(f"Search lengths: {lengths}", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Phase 1: For each length, try multiple initializations with LP search
    # Time allocation: 70% for Phase 1, 15% for Phase 2 (polish), 15% for Phase 3 (exploration)
    t_phase1_end = t_global_start + 0.70 * (budget_s - 30)
    t_phase2_end = t_global_start + 0.85 * (budget_s - 30)
    
    for n in lengths:
        if time.monotonic() > t_phase1_end:
            break
        
        remaining_for_n = t_phase1_end - time.monotonic()
        if remaining_for_n < 30:
            break
        
        inits = diverse_inits(n, rng, best_a if best_a is not None else parent)
        n_inits = len(inits)
        
        # Time per init: allocate based on remaining time and number of inits
        time_per_init = max(20, remaining_for_n / n_inits)
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > t_phase1_end - 10:
                break
            
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()
            
            sub_deadline = min(t_phase1_end - 5, time.monotonic() + time_per_init)
            
            if verbose_check(sub_deadline):
                print(f"  n={n} init {ii+1}/{n_inits}: M_init={fast_conv(a0).max():.10f}", flush=True)
            
            a_opt, M_opt = run_lp_search(a0, sub_deadline, verbose=(ii == 0))
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
            
            # Try update
            try_update(a_opt)
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.10f} J={J_val:.6f}", flush=True)
            try_update(best_for_n)
    
    print(f"\n{'='*60}", flush=True)
    print(f"After Phase 1: J={best_score:.8f} n={best_a.size if best_a is not None else 0}", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Phase 2: Intensive LP polish on the best sequence found
    if best_a is not None and time.monotonic() < t_phase2_end:
        n_best = best_a.size
        print(f"\nPhase 2: LP polish on n={n_best}", flush=True)
        
        # Long LP search from the best
        a_polished, M_polished = run_lp_search(best_a, min(t_phase2_end, time.monotonic() + 120), verbose=True)
        try_update(a_polished)
        
        # Second pass
        if time.monotonic() < t_phase2_end - 20:
            a_polished2, _ = run_lp_search(a_polished, min(t_phase2_end - 5, time.monotonic() + 60), verbose=False)
            try_update(a_polished2)
    
    # Phase 3: Coordinate descent polish
    if best_a is not None and time.monotonic() < deadline - 10:
        print(f"\nPhase 3: Coordinate descent polish", flush=True)
        cd_deadline = min(deadline - 5, time.monotonic() + (deadline - time.monotonic()) * 0.6)
        a_cd = coordinate_descent_polish(best_a, cd_deadline)
        try_update(a_cd)
        
        # After CD, one more LP pass
        if time.monotonic() < deadline - 20:
            a_final, _ = run_lp_search(a_cd, min(deadline - 5, time.monotonic() + 60), verbose=False)
            try_update(a_final)
    
    # Phase 4: Try a few more things if time remains
    if best_a is not None and time.monotonic() < deadline - 30:
        print(f"\nPhase 4: Exploration", flush=True)
        # Try slightly perturbed versions
        for pert_size in [0.02, 0.01, 0.005]:
            if time.monotonic() > deadline - 30:
                break
            a_pert = best_a + rng.normal(0, pert_size, best_a.size)
            a_pert = np.maximum(a_pert, 0.01)
            a_pert = a_pert / a_pert.sum()
            a_opt, _ = run_lp_search(a_pert, min(deadline - 10, time.monotonic() + 30), verbose=False)
            try_update(a_opt)
    
    # Final check
    if best_a is None:
        best_a = np.ones(512)
        best_score = compute_J(best_a)
    
    elapsed = time.monotonic() - t_global_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_a]
