    n = len(a)
    num_vars = n + 1  # g_0,...,g_{n-1}, t
    
    # Objective: minimize t
    c = np.zeros(num_vars)
    c[-1] = 1.0
    
    # Inequality constraints: (Ag)_k - t <= 0 for k=0,...,2n-2
    # A is (2n-1) x n
    # Full constraint matrix: [(2n-1) x n] | [-(2n-1) x 1]
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            m = k - j
            if 0 <= m < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
        # t term
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, num_vars))
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, t free (but >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                bounds=bounds, method='highs', options={'presolve': True})
    
    if r.success:
        g = r.x[:n]
        t_val = r.x[-1]
        return g, t_val
    return None, None

The matrix A has A[k,j] = a[k-j] if 0 <= k-j < n, else 0.

For k=0: only j=0 contributes, A[0,0] = a[0]
For k=1: j=0,1 contribute, A[1,0] = a[1], A[1,1] = a[0]
...
For k=n-1: all j=0,...,n-1 contribute
For k=n: j=1,...,n-1 contribute (a[0] would need k-j=n, which means j=n, out of range)

Wait let me redo this. (Ag)[k] = sum_j A[k,j] * g[j] where A[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0, and 0 <= j <= n-1.

So for a given k and j:
- We need 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k
- And 0 <= j <= n-1
- So j ranges from max(0, k-n+1) to min(n-1, k)

The number of non-zeros in row k is min(k+1, n, 2n-1-k).

Total non-zeros in A: sum_{k=0}^{2n-2} min(k+1, n, 2n-1-k)
For k = 0 to n-1: k+1 terms
For k = n to 2n-2: 2n-1-k terms
Total = n(n+1)/2 + (n-1)n/2 = n^2

So A has n^2 non-zeros, which is about 590K for n=768. Very manageable.

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def build_conv_matrix(a):
    """Build sparse convolution matrix A where b = A @ g gives conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            m = k - j
            if 0 <= m < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_best_response_LP(a):
    """Solve the full LP for best response: min t s.t. (a*g)_k <= t, sum(g)=1, g>=0."""
    n = len(a)
    num_vars = n + 1
    c = np.zeros(num_vars)
    c[-1] = 1.0
    
    A_conv = build_conv_matrix(a)
    # Augment with -1 column for t
    A_aug = A_conv  # we'll handle t separately
    # Actually, let's build the full constraint matrix including t
    # (A_conv @ g) - t <= 0  =>  A_conv @ g + (-1) * t <= 0
    num_constraints = 2 * n - 1
    
    rows = list(A_conv.data is not None and []) # just use the sparse structure
    # Better: build from scratch
    rows = []
    cols = []
    vals = []
    # Non-zeros from A_conv
    coo = A_conv.tocoo()
    for i in range(len(coo.row)):
        rows.append(coo.row[i])
        cols.append(coo.col[i])
        vals.append(coo.data[i])
    # t column
    for k in range(num_constraints):
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_constraints, num_vars))
    b_ub = np.zeros(num_constraints)
    
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                             'primal_feasibility_tolerance': 1e-10,
                             'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
            return g, float(r.x[-1])
    except Exception:
        pass
    return None, None

def spectral_flatten(a, n_iters, deadline, compress_power=0.7):
    """Iterative spectral flattening: compress |FFT| magnitude to reduce peak autocorrelation."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best = a.copy()
    best_M = fast_conv(a).max()
    
    # Use zero-padding for correct linear convolution via FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute FFT of zero-padded a
        A = np.fft.fft(a, size)
        
        # Compress magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        mag_compressed = mag ** compress_power
        # Normalize to maintain same energy (sum of |A|^2 = sum of a^2 * size)
        mag_compressed = mag_compressed * (mag.max() / mag_compressed.max()) if mag_compressed.max() > 0 else mag_compressed
        
        # Reconstruct
        A_new = mag_compressed * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new, size))[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum() if a_new.sum() > 1e-10 else np.ones(n) / n
        
        M = fast_conv(a_new).max()
        if M < best_M:
            best_M = M
            best = a_new.copy()
        
        a = a_new
    
    return best

def lp_fixed_point(a_init, max_iters, deadline, relaxation=None, momentum_beta=0.0):
    """Fixed-point iteration: a <- best_response(a) with optional Nesterov momentum."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best = a.copy()
    best_M = fast_conv(a).max()
    
    a_prev = a.copy()
    t_nest = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        if momentum_beta > 0:
            beta = (1.0 - np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0  # this gives negative...
            # Standard Nesterov: beta_k = (1 - sqrt(1+4t_k^2))/(2t_k) which is negative
            # Actually the standard is: y = x + (t_{k-1}-1)/t_k * (x - x_prev)
            alpha = (t_nest - 1.0) / (t_nest + 1.0)  # simpler momentum
            y = a + alpha * (a - a_prev)
            y = np.maximum(y, 0.0)
            y = y / y.sum() if y.sum() > 1e-10 else a.copy()
        else:
            y = a
            
        # Solve LP for best response
        g, t_val = solve_best_response_LP(y)
        if g is None:
            break
        
        # Update with relaxation
        if relaxation is None:
            # Adaptive: start with 1.0, decrease if oscillating
            t_relax = 1.0 if it < 10 else 0.7
        else:
            t_relax = relaxation
        
        a_new = (1.0 - t_relax) * a + t_relax * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        a_prev = a
        a = a_new
        t_nest += 1.0
        
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best = a.copy()
    
    # Final check
    M = fast_conv(a).max()
    if M < best_M:
        best_M = M
        best = a.copy()
    
    return best

def coordinate_descent_exact(a, i, b, n):

    The objective is f(x) = max(base_max, max_j(c_j + s_j*x), quad_c + x^2)
    where the max is over affected convolution positions.

            # Actually b[k] = sum_j a[j]*a[k-j], and the term with j=i is a[i]*a[k-i]
            # If k-i = i, i.e., k = 2i, then the term is a[i]^2
            quad_c = b[k] - x_cur * x_cur
        else:
            j = k - i

                # If k-j = i and j != i: term is 2*a[j]*a[i] (two symmetric terms)

                    # And if j != k-i (i.e., i != k-i, i.e., 2i != k), there's also a[i]*a[k-i] from the other term
                    # Actually for linear conv: b[k] = sum_{j:max(0,k-n+1)}^{min(n-1,k)} a[j]*a[k-j]
                    # The terms involving a[i] are: a[i]*a[k-i] (j=i) and a[k-i]*a[i] (j=k-i)
                    # These are the same term if i = k-i, i.e., k = 2i
                    # They are different terms if k != 2i
                    if k == 2 * i:
                        # Only one term: a[i]^2
                        pass  # handled above
                    else:
                        # Two terms: a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i]
                        c = b[k] - 2.0 * x_cur * a[j]
                        slopes.append(2.0 * a[j])
                        c_vals.append(c)
                        continue
                else:
                    c_vals.append(b[k])
                    slopes.append(0.0)
                    continue

    # For b[k] = sum_j a[j] * a[k-j] where 0<=j<n and 0<=k-j<n
    # The terms involving a[i] (which we're varying as x):
    #   j = i: a[i] * a[k-i] = x * a[k-i] (if 0 <= k-i < n and k+i < 2n, which is always)
    #   j = k-i: a[k-i] * a[i] = a[k-i] * x (if 0 <= k-i < n)
    # So if i != k-i (i.e., k != 2i): contribution is 2*x*a[k-i] (if 0 <= k-i < n)
    # If i == k-i (i.e., k == 2i): contribution is x^2

    c_vals = []
    slopes = []
    has_quad = False
    quad_c_val = 0.0
    
    for k in range(k_lo, k_hi + 1):
        # Terms involving x = a[i]:
        # Term 1: j=i, contributes x * a[k-i] if 0 <= k-i < n
        # Term 2: j=k-i, contributes a[k-i] * x if 0 <= k-i < n (same condition)
        # If k-i == i (k=2i): both terms are x*a[i] = x^2, but it's actually just one term x^2
        # If k-i != i: two separate terms, total 2*x*a[k-i]
        
        k_minus_i = k - i
        if 0 <= k_minus_i < n:

                # Contribution: x^2
                has_quad = True
                quad_c_val = b[k] - x_cur * x_cur
            else:
                # Contribution: 2*x*a[k-i]

        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = quad_c_val + x_test * x_test
            if qv > m:

    lo = 0.0

    # Ensure hi is large enough
    for _ in range(20):

            hi *= 2.0
        else:
            break
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(60):

            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with exact 1D optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        
        b = fast_conv(a)
        sweep_best_improve = 0.0
        
        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = coordinate_descent_exact(a, i, b, n)
            delta = x_opt - a[i]
            if abs(delta) > 1e-10:
                a[i] = x_opt
                # Update b
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    k_minus_i = k - i
                    if 0 <= k_minus_i < n:
                        if k_minus_i == i:
                            old_contrib = (a[i] - delta) * (a[i] - delta)
                            new_contrib = x_opt * x_opt
                            b[k] += new_contrib - old_contrib
                        else:
                            old_contrib = 2.0 * (a[i] - delta) * a[k_minus_i]
                            new_contrib = 2.0 * x_opt * a[k_minus_i]
                            b[k] += new_contrib - old_contrib
                sweep_best_improve = max(sweep_best_improve, abs(delta))
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if sweep_best_improve < 1e-8:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best

def smooth_max_grad(a, tau):

    b = fast_conv(a)

    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: d/dx_i [sum_k p[k] * b[k]] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    # = 2 * conv(p, a) evaluated appropriately

    # db[k]/da[i] = a[k-i] + a[i-(k)] ... wait
    # db[k]/da[i] = sum of terms where a[i] appears
    # a[i] appears in b[k] when j=i (gives a[k-i]) or k-j=i (gives a[k-i])
    # So db[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if i != k-i, or a[k-i] = a[i] if i = k-i
    # Actually: db[k]/da[i] = a[k-i] (from j=i term) + a[k-i] (from j=k-i term)
    # But these are the same term when i = k-i, so it's just a[i]
    # In general: db[k]/da[i] = 2*a[k-i] if 0<=k-i<n and k!=2i, a[i] if k=2i, 0 otherwise
    
    # For the smooth objective: d/dx_i [tau * logsumexp(b)] = tau * sum_k p[k] * db[k]/dx_i
    # = 2 * tau * sum_k p[k] * a[k-i] (approximately, ignoring the k=2i case)
    # More precisely: = tau * (2 * sum_{k: 0<=k-i<n, k!=2i} p[k]*a[k-i] + sum_{k: k=2i} p[k]*a[i])
    # For large n, the k=2i case is negligible
    
    # Efficient computation using correlation:
    # grad[i] = 2 * sum_k p[k] * a[k-i] = 2 * (p * a)[i] in the "cross-correlation" sense
    # This is: 2 * ifft(fft(p_padded) * conj(fft(a_padded)))... 
    # Actually it's simpler: grad[i] = 2 * sum_{k} p[k] * a[k-i] where a[j]=0 for j not in [0,n-1]
    # This is the cross-correlation of p and a, evaluated at lag i
    
    # Using FFT: cross_corr[i] = sum_k p[k]*a[k-i] = ifft(fft(p) * conj(fft(a))) [need to be careful]
    # Or: cross_corr = ifft(conj(fft(p)) * fft(a)) ... let me think
    # corr[i] = sum_k p[k] * a[k-i]
    # = sum_k p[k] * a[k-i]
    # Let's define: (p * a_flipped)[i] where a_flipped[j] = a[n-1-j]
    # Actually: sum_k p[k]*a[k-i] = (cross-correlation of p and a at lag i)
    # In FFT: corr = ifft(fft(p, N) * conj(fft(a, N))) for circular correlation
    # For linear: zero-pad to size >= 2n-1
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(P * np.conj(A), size)  # This gives circular cross-correlation
    # corr[i] = sum_k p[k] * conj(a[k-i mod size])
    # For linear: corr[i] = sum_k p[k] * a[k-i] for valid k-i
    # Wait, this isn't quite right. Let me use a different approach.
    # sum_k p[k]*a[k-i] for k-i in [0, n-1]
    # = sum_{j=0}^{n-1} p[j+i] * a[j]  (substituting j = k-i)
    # This is a "sliding dot product"
    
    # Using the full correlation via FFT:
    # Let's define c[i] = sum_{k=0}^{2n-2} p[k] * a[k-i] where a[m]=0 for m<0 or m>=n
    # c[i] = sum_{k} p[k] * a[k-i]
    # = ifft(fft(p_zero_padded) * conj(fft(a_zero_padded)))...

    # Cross-correlation: (f ~ g)(n) = ifft(conj(FFT(f)) * FFT(g))
    # (f ~ g)(n) = sum_k f(k) * conj(g(k-n))
    
    # We want: c[i] = sum_k p[k] * a[k-i] = sum_k p[k] * a[k-i]
    # Let m = k-i: c[i] = sum_m p[m+i] * a[m]
    # This is the cross-correlation of p and a at lag i: (p ~ a)(i) = sum_m p[m+i] * a[m]
    # In FFT: (p ~ a)(i) = ifft(conj(FFT(p)) * FFT(a))[i]  ... need to verify
    
    # Actually: cross_correlation(p, a)[i] = sum_k p[k+i] * conj(a[k])
    # In real case: = sum_k p[k+i] * a[k]
    # FFT formula: ifft(conj(fft(p, N)) * fft(a, N))[i] = sum_k p[k-i] * a[k] (for circular)
    # Hmm, the indexing is tricky. Let me just use the direct O(n^2) approach for now,
    # or use scipy.signal.correlate
    
    # For n=768, O(n^2) = ~600K operations per gradient eval, which is fast enough
    grad = np.zeros(n)
    for i in range(n):
        # c[i] = sum_k p[k] * a[k-i] where 0<=k<2n-1, 0<=k-i<n
        # = sum_{j=0}^{n-1} p[j+i] * a[j]
        # p has length 2n-1, so p[j+i] is valid for j+i < 2n-1, i.e., j < 2n-1-i
        # a[j] is valid for 0 <= j < n
        # So j ranges from 0 to min(n-1, 2n-2-i)
        j_hi = min(n - 1, 2 * n - 2 - i)
        if j_hi >= 0:
            grad[i] = 2.0 * np.dot(p[i:i + j_hi + 1], a[:j_hi + 1])
    
    return smooth_val, grad

def run_smooth_descent(a_init, deadline, tau_start=50, tau_end=500, lr=0.3, seed=0):
    """FISTA on smooth (log-sum-exp) approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best = a.copy()
    best_M = fast_conv(a).max()
    
    t_nest = 1.0
    y = a.copy()
    a_prev = a.copy()
    max_iters = 10000
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Lipschitz constant estimate for step size
        step = lr / (tau * (1.0 + 0.01 * it))
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = (t_nest - 1.0) / t_new
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t_nest = t_new
        
        if it % 10 == 0:
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best = a.copy()
    
    M = fast_conv(a).max()
    if M < best_M:
        best_M = M
        best = a.copy()
    
    return best

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        # Resample parent
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            p = np.maximum(np.interp(t, t_old, parent), 0.0)
        else:
            p = np.maximum(parent.astype(np.float64), 0.0)
        if p.sum() > 0:
            inits.append(p / p.sum())
            inits.append(np.maximum(p / p.sum() + rng.normal(0, 0.02, n), 1e-6) / 1)
            inits[-1] = inits[-1] / inits[-1].sum()
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various shaped inits
    inits.append(((t * (1 - t)) + 0.1) / (((t * (1 - t)) + 0.1).sum()))
    inits.append(((2*t - 1)**2 + 0.1) / (((2*t - 1)**2 + 0.1).sum()))
    
    for freq in [1, 2, 3, 4]:
        w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        w = np.maximum(w, 0.01)
        inits.append(w / w.sum())
    
    g = np.exp(-4*(t-0.5)**2) + 0.05
    inits.append(g / g.sum())
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits
