```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: iterative LP refinement with proper 1D line search (quadratic
formulation of the blending objective). Combined with p-norm Adam optimization,
multi-n exploration, and structured multi-start.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_cross_conv(a, b):
    """Compute linear convolution a*b using FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    conv = irfft(fa * fb, size)
    return conv[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for convolution."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=10):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    if n > 2000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def line_search_blend(a, g):
    """
    Find optimal t in [0,1] minimizing max_k conv((1-t)a + t*g, (1-t)a + t*g).
    Uses the quadratic decomposition:
    conv(a_t, a_t)[k] = (1-t)^2*C[k] + 2t(1-t)*B[k] + t^2*A[k]
    where C=conv(a,a), B=conv(a,g), A=conv(g,g).
    """
    n = len(a)
    a = a / np.sum(a)
    g = g / np.sum(g)

    C = compute_conv(a)
    B = compute_cross_conv(a, g)
    A = compute_conv(g)

    # quadratic_k(t) = A[k]*t^2 + (2B[k]-2C[k])*t + C[k]
    quad_A = A  # coeff of t^2
    quad_B = 2.0 * B - 2.0 * C  # coeff of t
    quad_C = C  # constant

    def f_t(t):
        return float(np.max(quad_A * t * t + quad_B * t + quad_C))

    # Use bounded scalar minimization
    res = minimize_scalar(f_t, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 200})
    if res.success:
        return res.x
    return 0.1  # fallback


def lp_iteration(a, num_iters=200, time_budget=300.0, verbose=True):
    """
    Iterative LP refinement: alternate between LP solve and line search.
    """
    start = time.monotonic()
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    best_M = best_obj / (2.0 * n)  # max conv value when sum=1

    stagnant_count = 0

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute current peak
        conv = compute_conv(a)
        M = float(np.max(conv))

        # Check if M hasn't improved
        if M < best_M - 1e-12:
            best_M = M
            best_obj = 2.0 * n * M  # since sum(a)=1
            best_a = a.copy()
            stagnant_count = 0
        else:
            stagnant_count += 1
            if stagnant_count > 30:
                if verbose:
                    print(f"    LP iter {it}: stagnant for {stagnant_count}, stopping")
                break

        # Solve LP
        g = solve_lp(a, M * (1 + 1e-10), time_limit=5)
        if g is None:
            if verbose:
                print(f"    LP iter {it}: LP failed")
            break

        # Check convergence: g proportional to a
        g_diff = np.linalg.norm(g - a) / (np.linalg.norm(a) + 1e-15)
        if g_diff < 1e-6:
            if verbose:
                print(f"    LP iter {it}: converged (g ≈ a)")
            break

        # Line search
        t_opt = line_search_blend(a, g)

        # Apply update
        a_new = (1 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            new_conv = compute_conv(a_new)
            new_M = float(np.max(new_conv))
            if new_M < M:
                a = a_new
                if verbose and it % 20 == 0:
                    print(f"    LP iter {it}: M={M:.10f} -> {new_M:.10f}, t={t_opt:.4f}")
            else:
                # Try smaller t
                for t_try in [t_opt * 0.5, t_opt * 0.25, 0.05, 0.02]:
                    a_try = (1 - t_try) * a + t_try * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        conv_try = compute_conv(a_try)
                        if np.max(conv_try) < M:
                            a = a_try
                            break

    # Final check
    final_obj = compute_obj(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()

    return best_a


def pnorm_obj_grad(a, p):
    """
    Compute p-norm objective and gradient.
    obj = 2n * (sum(conv^p))^(1/p) / (sum(a))^2
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    # Clip to avoid overflow
    conv_pos = np.maximum(conv, 1e-30)
    conv_p = conv_pos ** p
    sum_conv_p = float(np.sum(conv_p))
    obj_p = sum_conv_p ** (1.0 / p)

    obj = 2.0 * n * obj_p / (sum_a * sum_a)

    # Gradient: d(obj_p)/d(a[i]) = sum_conv_p^(1/p-1) * sum_k conv[k]^(p-1) * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with out-of-range = 0)
    # sum_k w[k]*(a[k-i] + a[i-k]) where w[k] = conv[k]^(p-1)
    # = cross_corr(w, a)[i] + conv(w, a)[i]
    # where cross_corr(w,a)[i] = sum_j w[j]*a[j-i] (computed via ifft(FFT(w)*conj(FFT(a))))
    # and conv(w,a)[i] = sum_j w[j]*a[i-j] (computed via ifft(FFT(w)*FFT(a)))

    w = conv_pos ** (p - 1)

    size = 1
    while size < 2 * 2 * n:
        size <<= 1

    W = rfft(w, size)
    A = rfft(a, size)

    cc = irfft(W * np.conj(A), size)  # cross-correlation: sum_j w[j]*a[j-i]
    cv = irfft(W * A, size)  # convolution: sum_j w[j]*a[i-j]

    grad_obj_p = (sum_conv_p ** (1.0 / p - 1.0)) * (cc[:n] + cv[:n])

    # Full gradient with the 1/sum_a^2 factor
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_obj_p - obj_p / sum_a)

    return obj, grad_obj


def adam_pnorm_opt(a, max_iters=20000, time_budget=100.0, p_start=50.0,
                   p_end=50000.0, lr_start=0.05, lr_decay=0.99998,
                   verbose=True):
    """
    Adam optimization in log-space with p-norm objective and p annealing.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start

    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # p annealing (geometric)
        progress = min(it / max(max_iters - 1, 1), 1.0)
        p = p_start * (p_end / p_start) ** progress

        # Compute a from x
        x_c = x - np.max(x)
        a = np.exp(x_c)
        a = a / np.sum(a)

        # Compute objective and gradient
        obj_sm, grad_a = pnorm_obj_grad(a, p)

        # Chain rule for log-space
        grad_x = grad_a * a

        # Adam update
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic centering
        if it % 200 == 0:
            x -= np.mean(x)

        # Check true objective
        if it % 200 == 0 or it == max_iters - 1:
            x_chk = x - np.max(x)
            a_check = np.exp(x_chk)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
                if verbose:
                    print(f"      p-norm Adam it={it}, p={p:.0f}: obj={best_true_obj:.10f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.10f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def diverse_initial_template(n, template_type, rng):
    """Generate diverse initial templates."""
    x = np.linspace(0, 1, n)
    if template_type == 0:  # Hann
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif template_type == 1:  # Raised cosine (wider)
        a = np.maximum(0, np.cos(np.pi * (x - 0.5)))
    elif template_type == 2:  # Plateau with tapered edges
        edge = max(1, n // 10)
        a = np.ones(n)
        for i in range(edge):
            a[i] = 0.5 * (1 - np.cos(np.pi * i / edge))
            a[n - 1 - i] = a[i]
    elif template_type == 3:  # Gaussian
        sigma = n * (0.15 + 0.1 * rng.random())
        a = np.exp(-0.5 * ((x - 0.5) * n / sigma) ** 2)
    elif template_type == 4:  # Triangular
        a = 1 - np.abs(2 * x - 1)
    elif template_type == 5:  # Sine^2
        a = np.sin(np.pi * x) ** 2
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses iterative LP refinement as the primary optimizer,
    combined with p-norm Adam, multi-n exploration, and multi-start.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.10f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase A: Intensive LP iteration on parent ===
    print(f"\n{'='*60}")
    print(f"Phase A: LP iteration (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.35, 500.0)

        a_lp = lp_iteration(a, num_iters=300, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  After LP: {best_score:.10f} ***")
        else:
            print(f"  After LP: {obj_lp:.10f} (no improvement)")

    # === Phase B: p-norm Adam refinement ===
    print(f"\n{'='*60}")
    print(f"Phase B: p-norm Adam refinement")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            a = best_seq.copy()
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

            # Multiple rounds with increasing p
            p_rounds = [
                (50.0, 500.0, 0.05, 0.99998),
                (200.0, 2000.0, 0.03, 0.99999),
                (1000.0, 10000.0, 0.02, 0.99999),
                (5000.0, 50000.0, 0.01, 0.99999),
            ]

            for r_idx, (p0, p1, lr0, lr_dec) in enumerate(p_rounds):
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                alloc = min(remaining * 0.15, 80.0)
                a_opt = adam_pnorm_opt(a, max_iters=15000, time_budget=alloc,
                                       p_start=p0, p_end=p1,
                                       lr_start=lr0, lr_decay=lr_dec,
                                       verbose=(r_idx == 0))
                obj = compute_obj(a_opt)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    a = a_opt.copy()
                    print(f"  p-norm round {r_idx} (p={p0:.0f}-{p1:.0f}): {best_score:.10f} ***")
                else:
                    print(f"  p-norm round {r_idx} (p={p0:.0f}-{p1:.0f}): {obj:.10f}")

    # === Phase C: LP iteration again after p-norm ===
    print(f"\n{'='*60}")
    print(f"Phase C: LP iteration after p-norm")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a = best_seq.copy()
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            alloc = min(remaining * 0.2, 200.0)
            a_lp2 = lp_iteration(a, num_iters=200, time_budget=alloc, verbose=True)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  After LP2: {best_score:.10f} ***")
            else:
                print(f"  After LP2: {obj_lp2:.10f}")

    # === Phase D: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase D: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 150, n_best - 100, n_best - 50,
            n_best + 50, n_best + 100, n_best + 200,
            n_best + 400, n_best + 600
        ]))
        n_options = [n for n in n_options if 100 <= n <= 2000]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

            # Quick LP + p-norm
            alloc_lp = min(remaining * 0.12, 60.0)
            a_opt = lp_iteration(a_try, num_iters=100, time_budget=alloc_lp, verbose=False)

            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc_adam = min(remaining * 0.08, 40.0)
                a_opt = adam_pnorm_opt(a_opt, max_iters=8000, time_budget=alloc_adam,
                                       p_start=50.0, p_end=5000.0,
                                       lr_start=0.03, lr_decay=0.99999, verbose=False)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.10f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.10f}")

    # === Phase E: Multi-start with diverse templates ===
    print(f"\n{'='*60}")
    print(f"Phase E: Multi-start with diverse templates")
    print(f"{'='*60}")

    if best_seq is not None:
        n_base = len(best_seq)

        for tmpl in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            # Use the best n found so far
            n_try = n_base
            a_init = diverse_initial_template(n_try, tmpl, rng)

            # Blend with best sequence for warm start
            if best_seq is not None and len(best_seq) == n_try:
                alpha = 0.3
                a_init = alpha * (best_seq / np.sum(best_seq)) + (1 - alpha) * a_init
                a_init = np.maximum(a_init, 1e-10)
                a_init = a_init / np.sum(a_init)

            alloc_lp = min(remaining * 0.12, 60.0)
            a_opt = lp_iteration(a_init, num_iters=80, time_budget=alloc_lp, verbose=False)

            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc_adam = min(remaining * 0.08, 30.0)
                a_opt = adam_pnorm_opt(a_opt, max_iters=6000, time_budget=alloc_adam,
                                       p_start=100.0, p_end=10000.0,
                                       lr_start=0.03, lr_decay=0.99999, verbose=False)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Template {tmpl}: {best_score:.10f} *** UPDATE")
            else:
                print(f"  Template {tmpl}: {obj:.10f}")

    # === Phase F: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase F: Final intensive polish")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a = best_seq.copy()
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

            # Final LP
            alloc = min(remaining * 0.3, 80.0)
            a_final = lp_iteration(a, num_iters=150, time_budget=alloc, verbose=False)
            obj = compute_obj(a_final)
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()
                a = a_final.copy()
                print(f"  Final LP: {best_score:.10f} ***")

            # Final high-p Adam
            remaining = deadline - time.monotonic()
            if remaining > 15:
                alloc = min(remaining * 0.4, 40.0)
                a_final2 = adam_pnorm_opt(a, max_iters=10000, time_budget=alloc,
                                          p_start=5000.0, p_end=100000.0,
                                          lr_start=0.008, lr_decay=0.99999, verbose=False)
                obj2 = compute_obj(a_final2)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a_final2.copy()
                    print(f"  Final high-p Adam: {best_score:.10f} ***")

            # One more LP after high-p Adam
            remaining = deadline - time.monotonic()
            if remaining > 15:
                alloc = min(remaining * 0.4, 30.0)
                a_final3 = lp_iteration(best_seq, num_iters=100, time_budget=alloc, verbose=False)
                obj3 = compute_obj(a_final3)
                if obj3 < best_score:
                    best_score = obj3
                    best_seq = a_final3.copy()
                    print(f"  Final LP2: {best_score:.10f} ***")

    # === Phase G: Try very large p Adam from scratch on best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a = best_seq.copy()
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            alloc = min(remaining * 0.5, 30.0)
            a_g = adam_pnorm_opt(a, max_iters=8000, time_budget=alloc,
                                 p_start=10000.0, p_end=200000.0,
                                 lr_start=0.005, lr_decay=0.99999, verbose=False)
            obj_g = compute_obj(a_g)
            if obj_g < best_score:
                best_score = obj_g
                best_seq = a_g.copy()
                print(f"  Phase G: {best_score:.10f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
```
