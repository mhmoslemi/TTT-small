Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum a)^2.
Combines LP alternating, Nesterov-accelerated smooth max, and structured restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([1.0])

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_cosine_seq(n):
    """Cosine window sequence."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x) ** 2 + 0.3
    return a / np.sum(a)


def make_triangular_seq(n):
    """Triangular (tent) sequence."""
    mid = n // 2
    a = np.abs(np.arange(n) - mid)
    a = a.max() - a + 1
    return a / np.sum(a)


def make_uniform_seq(n):
    return np.ones(n) / n


def make_bimodal_seq(n):
    """Two-peak sequence."""
    c1 = n // 4
    c2 = 3 * n // 4
    w = n // 8
    x = np.arange(n, dtype=float)
    a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    return a / np.sum(a)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv_a = compute_conv(a)
        max_c = np.max(conv_a)
        exp_c = np.exp(t_param * (conv_a - max_c))
        w = exp_c / np.sum(exp_c)

        idx = np.add.outer(np.arange(n), np.arange(n))
        M = w[idx]
        grad = M @ a
        grad_c = grad - np.mean(grad)

        v = mu * v - lr * grad_c
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.98
            v *= 0.5

        if it % 200 == 199:
            lr = max(base_lr * 0.7 ** (it // 200), 1e-5)

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining LP alternating, Nesterov smooth max,
    peak-chasing subgradient, coordinate refinement, multi-n exploration, and restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Initialization ---
    n0 = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_cosine_seq(n0)
        best_score = compute_obj(best_seq)
        print(f"Default cosine init (n={n0}): obj={best_score:.8f}")

    # --- Phase 1: Intensive LP Alternating (50% of budget) ---
    print(f"\n=== Phase 1: LP Alternating ===")
    phase1_budget = (deadline - start_time) * 0.50
    n_parent = len(best_seq)

    a = best_seq.copy()
    # Multiple rounds with increasing patience
    for round_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(phase1_budget / (5 - round_idx), remaining * 0.55)
        if alloc < 10:
            break

        a_new = lp_alternate(a, time_budget=alloc, verbose=(round_idx == 0))
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Round {round_idx}: obj={best_score:.8f}")
        else:
            print(f"  Round {round_idx}: obj={obj:.8f} (no improvement)")

    # --- Phase 2: Nesterov Smooth Max (10% of budget) ---
    print(f"\n=== Phase 2: Nesterov Smooth Max ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for t_p in [8.0, 15.0, 25.0]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.1, 40.0)
            a_nest = nesterov_smooth_max(a, time_budget=alloc, t_param=t_p)
            obj_n = compute_obj(a_nest)
            if obj_n < best_score:
                best_score = obj_n
                best_seq = a_nest.copy()
                a = a_nest.copy()
                print(f"  Nesterov t={t_p}: obj={best_score:.8f} **")
            else:
                print(f"  Nesterov t={t_p}: obj={obj_n:.8f}")

    # --- Phase 3: Peak Chase + Coordinate Refine (10%) ---
    print(f"\n=== Phase 3: Peak Chase + Coord Refine ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_pc = peak_chase_subgradient(a, time_budget=min(remaining * 0.15, 30.0))
        obj_pc = compute_obj(a_pc)
        if obj_pc < best_score:
            best_score = obj_pc
            best_seq = a_pc.copy()
            a = a_pc.copy()
            print(f"  PeakChase: obj={best_score:.8f} **")
        else:
            print(f"  PeakChase: obj={obj_pc:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_cd = coordinate_refine(a, time_budget=min(remaining * 0.1, 15.0))
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a = a_cd.copy()
                print(f"  CoordRefine: obj={best_score:.8f} **")
            else:
                print(f"  CoordRefine: obj={obj_cd:.8f}")

    # --- Phase 4: Multi-n Exploration (15%) ---
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_options = sorted(set([
            n_parent - 100, n_parent - 50, n_parent - 20,
            n_parent + 20, n_parent + 50, n_parent + 100,
            n_parent + 200,
            600, 700, 900, 1000, 1100
        ]))
        n_options = [n for n in n_options if 300 <= n <= 1200]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if n_try == n_parent:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)

            alloc = min(remaining * 0.15, 35.0)
            a_opt = lp_alternate(a_try, time_budget=alloc)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                n_parent = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # --- Phase 5: Diverse Restarts (10%) ---
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_restarts = max(3, int(remaining // 50))
    restart_budget_each = min(40.0, (remaining - 20) / max(n_restarts, 1))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < restart_budget_each + 10:
            break

        # Choose initialization strategy
        init_type = r % 5
        if init_type == 0:
            a_r = random_init(n_best, rng)
        elif init_type == 1:
            # Multiplicative perturbation of best
            a_r = best_seq * np.exp(rng.normal(0, 0.15, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 2:
            # Different n
            n_r = n_best + int(rng.integers(-150, 150))
            n_r = max(400, min(1100, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.08, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 3:
            # Smooth descent with momentum
            a_r = best_seq.copy()
        else:
            # Structured: raised cosine with random center
            n_r = n_best
            x = np.linspace(0, np.pi, n_r)
            shift = rng.uniform(0, 0.3)
            a_r = np.cos((x + shift) ** 2) + 0.2
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        # Choose optimization method
        method = r % 3
        if method == 0:
            a_opt = lp_alternate(a_r, time_budget=restart_budget_each)
        elif method == 1:
            a_opt = smooth_max_descent(a_r, time_budget=restart_budget_each * 0.7)
            a_opt = lp_alternate(a_opt, time_budget=restart_budget_each * 0.3)
        else:
            a_opt = lp_alternate(a_r, time_budget=restart_budget_each * 0.6)
            a_opt = peak_chase_subgradient(a_opt, time_budget=restart_budget_each * 0.4)

        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (type={init_type}, meth={method}, n={len(a_opt)}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}, meth={method}, n={len(a_opt)}): obj={obj_r:.8f}")

    # --- Phase 6: Final Polish ---
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        # LP alternating
        alloc = min(remaining * 0.4, 60.0)
        a_lp = lp_alternate(a_final, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

        # Peak chase
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_pc = peak_chase_subgradient(a_final, time_budget=min(remaining * 0.25, 25.0))
            obj_pc = compute_obj(a_pc)
            if obj_pc < best_score:
                best_score = obj_pc
                best_seq = a_pc.copy()
                a_final = a_pc.copy()
                print(f"  Final PeakChase: obj={best_score:.8f}")

        # Coordinate refinement
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_cd = coordinate_refine(a_final, time_budget=min(remaining * 0.3, 15.0))
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a_final = a_cd.copy()
                print(f"  Final CoordRefine: obj={best_score:.8f}")

        # One more LP round
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_last = lp_alternate(a_final, time_budget=min(remaining * 0.5, 30.0))
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Final LP2: obj={best_score:.8f}")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
