Peak-minimization search for non-negative step functions using:

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        return a / s
    return np.ones(len(a)) / len(a)


def build_conv_mat(fixed):
    """Build matrix A where (A @ x)[k] = conv(fixed, x)[k]."""
    n = len(fixed)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = fixed[:hi - lo]
    return A


def solve_minpeak_lp(fixed, n_vars=None):
    """
    Solve: min t s.t. conv(fixed, x)[k] <= t for all k, sum(x)=1, x>=0.
    Returns x (normalized) or None.
    """
    n = len(fixed)
    if n_vars is None:
        n_vars = n
    A = build_conv_mat(fixed)
    # Variables: x[0..n-1], t
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))

    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            x = np.maximum(res.x[:n], 0)
            s = np.sum(x)
            if s > 1e-10:
                return x / s
    except Exception:
        pass
    return None

def smooth_obj(a, temp=0.01):
    """Smoothed objective using log-sum-exp."""
    n = len(a)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return 1e10
    a = a / s
    conv = compute_conv(a)
    # log-sum-exp for numerical stability
    cmax = np.max(conv)
    lse = cmax + temp * np.log(np.sum(np.exp((conv - cmax) / temp)))
    return 2.0 * n * lse


def smooth_obj_grad(a, temp=0.01):
    """Gradient of smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return 1e10, np.zeros(n)
    a = a / s
    conv = compute_conv(a)
    cmax = np.max(conv)
    shifted = (conv - cmax) / temp
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
        w_sum = 1.0
    else:
        weights /= w_sum

    # Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a[i]
    # = 2 * sum_k w[k] * a[k-i] (for valid indices)
    # Using FFT: corr[i] = sum_j a[j] * w[i+j]
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W = rfft(w_ext, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    grad = 2.0 * corr[:n]

    # Account for normalization (a -> a/s)
    # d/da [f(a/s)] = (1/s) * grad_f(a/s) - (f(a/s)/s^2) * ones... 
    # Actually since we normalize inside, gradient is:
    # grad_obj = (grad - mean(grad)) / s (projection onto sum=const)
    # But we're already at sum=1, so:
    grad = grad - np.mean(grad)  # project onto tangent of sum=1 hyperplane
    return smooth_obj(a, temp), grad


def lbfgs_polish(a, time_budget=30.0, temp=0.01):
    """Polish using L-BFGS-B with smoothed objective."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Decreasing temperature schedule
    temps = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
    
    for tmp in temps:
        if time.monotonic() - start > time_budget - 2:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        # L-BFGS on the unconstrained problem (with penalty for negatives)
        x0 = a.copy()

        def fun(x):
            x_clip = np.maximum(x, 0)
            s = np.sum(x_clip)
            if s < 1e-10:
                return 1e10
            x_norm = x_clip / s
            return smooth_obj(x_norm, tmp)

        def grad_fun(x):
            x_clip = np.maximum(x, 0)
            s = np.sum(x_clip)
            if s < 1e-10:
                return 1e10, np.zeros(len(x))
            x_norm = x_clip / s
            _, g = smooth_obj_grad(x_norm, tmp)
            # Chain rule through clipping and normalization
            g = g * (x_clip > 0) / s
            return fun(x), g

        try:
            res = minimize(fun, x0, jac=grad_fun, method='L-BFGS-B',
                          options={'maxiter': 200, 'maxfun': 2000,
                                   'gtol': 1e-8})
            if res.success or res.fun < best_obj:
                a_new = normalize(res.x)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
        except Exception:
            pass

    return best_a, best_obj


def parametric_init(n, style='ramp', rng=None):
    """Generate parametric initial sequences."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if style == 'uniform':
        a = np.ones(n)
    elif style == 'ramp_up':
        a = i + 1
    elif style == 'ramp_down':
        a = n - i
    elif style == 'bilinear':
        # High in middle, low at edges (but asymmetric)
        a = np.abs(i - n * 0.3) + 1
        a = 1.0 / a
    elif style == 'gaussian':
        a = np.exp(-((i - center) ** 2) / (2 * (n / 4) ** 2))
    elif style == 'quarter_wave':
        # Like a quarter sine wave
        a = np.sin(np.pi * i / (2 * n)) + 0.1
    elif style == 'exponential':
        a = np.exp(-i / (n / 3))
    elif style == 'two_bump':
        a = np.exp(-((i - n*0.25)**2) / (2*(n/8)**2)) + np.exp(-((i - n*0.75)**2) / (2*(n/8)**2))
    elif style == 'sine_mod':
        a = 1.0 + 0.5 * np.sin(2 * np.pi * i / n)
    elif style == 'noise':
        if rng is None:
            rng = np.random.default_rng(42)
        a = np.abs(rng.normal(1, 0.3, n))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.001)
    return a / np.sum(a)

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        return a / s
    return np.ones(len(a)) / len(a)


def build_conv_mat(fixed):
    """Build matrix A where (A @ x)[k] = conv(fixed, x)[k]."""
    n = len(fixed)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = fixed[:hi - lo]
    return A


def solve_minpeak_lp(fixed):
    """
    Solve: min t s.t. conv(fixed, x)[k] <= t for all k, sum(x)=1, x>=0.
    Returns x (normalized to sum=1) or None.
    """
    n = len(fixed)
    A = build_conv_mat(fixed)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            x = np.maximum(res.x[:n], 0)
            s = np.sum(x)
            if s > 1e-10:
                return x / s
    except Exception:
        pass
    return None


def alternating_lp(a, max_iters=300, time_budget=120.0, verbose=False):
    """
    Alternating min-peak LP with symmetric averaging.
    At each step: fix a, solve LP for x, then a <- (a + x)/2.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        x = solve_minpeak_lp(a)
        if x is None:
            stall += 1
            if stall > 20:
                break
            continue

        a_new = 0.5 * a + 0.5 * x
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 50 == 0:
                print(f"      LP iter {it}: obj={obj_new:.10f}")
        else:
            # Try full step to x
            obj_full = compute_obj(x)
            if obj_full < best_obj - 1e-13:
                best_obj = obj_full
                best_a = x.copy()
                a = x
                stall = 0
            else:
                # Try larger step toward x
                for alpha in [0.7, 0.8, 0.9, 1.0]:
                    a_try = (1 - alpha) * a + alpha * x
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                        break
                else:
                    stall += 1
                    if stall > 25:
                        rng = np.random.default_rng(it * 104729 + 7)
                        scale = 0.005 + 0.005 * min(stall - 25, 10)
                        a = best_a * np.exp(rng.normal(0, scale, n))
                        a = normalize(a)
                        stall = 0

    return best_a, best_obj


def compute_subgradient(a, num_active=None):
    """
    Compute subgradient of max_k conv(a,a)[k] at point a (sum=1).
    If num_active is None, use all lags above 0.99*max.
    Returns gradient vector (length n).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)

    if num_active is not None:
        indices = np.argsort(conv)[-num_active:]
        weights = np.zeros(len(conv))
        weights[indices] = 1.0 / num_active
    else:
        threshold = 0.99 * cmax
        active = conv >= threshold
        if not np.any(active):
            idx = int(np.argmax(conv))
            active = np.zeros(len(conv))
            active[idx] = 1
        weights = active / np.sum(active)

    # grad[i] = 2 * sum_k weights[k] * a[k-i]
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W = rfft(w_ext, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    grad = 2.0 * corr[:n]
    return grad, cmax


def proximal_gradient(a, max_iters=500, time_budget=60.0, verbose=False):
    """
    Proximal gradient method with backtracking line search.
    Minimizes max_k conv(a,a)[k] subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, cmax = compute_subgradient(a)
        # Center gradient (project onto sum=const hyperplane)
        grad_c = grad - np.mean(grad)
        gn = np.linalg.norm(grad_c)
        if gn < 1e-15:
            break

        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(10):
            a_try = a - alpha * grad_c / gn
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14 or obj_try < compute_obj(a) - 1e-10:
                a = a_try
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a.copy()
                improved = True
                break
            alpha *= 0.5

        if improved:
            step = min(step * 1.05, 0.3)
        else:
            step *= 0.7
            if step < 1e-5:
                break

    return best_a, best_obj


def smooth_objective(x, temp):
    """Smoothed objective: 2n * LSE(conv(x,x))/temp / (sum(x))^2."""
    x = np.maximum(x, 0)
    n = len(x)
    s = np.sum(x)
    if s < 1e-10:
        return 1e10
    a = x / s
    conv = compute_conv(a)
    cmax = np.max(conv)
    shifted = np.clip((conv - cmax) / temp, -500, 500)
    lse = cmax + temp * np.log(np.sum(np.exp(shifted)))
    return 2.0 * n * lse / (s * s)


def smooth_objective_grad(x, temp):
    """Gradient of smoothed objective."""
    x = np.maximum(x, 0)
    n = len(x)
    s = np.sum(x)
    if s < 1e-10:
        return 1e10, np.zeros(n)
    a = x / s
    conv = compute_conv(a)
    cmax = np.max(conv)
    shifted = np.clip((conv - cmax) / temp, -500, 500)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
        w_sum = 1.0
    else:
        w /= w_sum

    # d(conv[a,a][k])/da[i] = 2*a[k-i] (for valid k-i)
    # Using FFT: corr[i] = sum_j a[j] * w[i+j]
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_ext = np.zeros(fft_size)
    w_ext[:len(w)] = w
    W = rfft(w_ext, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] = 2 * corr[i]
    grad_a = 2.0 * corr[:n]

    # Chain rule: x -> a = x/s
    # d(f(a))/dx[i] = (1/s) * grad_a[i] - (1/s^2) * sum_j grad_a[j] * x[j]
    grad_x = (1.0 / s) * grad_a - (1.0 / (s * s)) * np.dot(grad_a, x)
    grad_x *= 2.0 * n / (s * s) * (s * s)  # The 2n factor
    # Actually let me redo: f(x) = 2n * LSE(conv(x/s, x/s)) / s^2
    # = 2n * LSE(conv(a,a)) / s^2 where a = x/s
    # dLSE/da[i] = 2 * sum_k w[k] * a[k-i] = grad_a[i]
    # d(conv(a,a))/dx[i] = (1/s) * grad_a[i] - (2*LSE/s^2) * x[i] ... no
    # Let's be more careful.
    # g(a) = 2n * LSE(conv(a,a))  (this is what we minimize, with a = x/s)
    # g'(a)[i] = 2 * sum_k w[k] * a[k-i] * 2n = 2n * grad_a[i]
    # f(x) = g(x/s) / s^2 ... no, f(x) = 2n * LSE(conv(x/s, x/s)) / 1 (since we normalize)
    # Actually f(x) = 2n * LSE(conv(a,a)) where a = x/s, and s = sum(x)
    # df/dx[i] = sum_j (df/da[j]) * (da[j]/dx[i])
    # da[j]/dx[i] = delta(j,i)/s - a[j]/s^2
    # df/da[j] = 2n * 2 * sum_k w[k] * a[k-j] = 2n * grad_a[j]
    # So df/dx[i] = 2n * grad_a[i] / s - 2n * sum_j grad_a[j] * a[j] / s^2
    # = (2n/s) * grad_a[i] - (2n/s^2) * (grad_a . a)

    grad_x = (2.0 * n / s) * grad_a - (2.0 * n / (s * s)) * np.dot(grad_a, a)
    obj = smooth_objective(x, temp)
    return obj, grad_x


def lbfgs_polish(a, time_budget=60.0, verbose=False):
    """Polish using L-BFGS-B with decreasing temperature."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    temps = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]

    for tmp in temps:
        if time.monotonic() - start > time_budget - 2:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        x0 = a.copy()

        def fun(x):
            return smooth_objective(x, tmp)

        def grad_fun(x):
            o, g = smooth_objective_grad(x, tmp)
            return o, g

        try:
            res = minimize(fun, x0, jac=grad_fun, method='L-BFGS-B',
                          options={'maxiter': 100, 'maxfun': 500,
                                   'ftol': 1e-14, 'gtol': 1e-8})
            a_new = normalize(res.x)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"      L-BFGS temp={tmp}: {obj_new:.10f}")
        except Exception:
            pass

    return best_a, best_obj


def parametric_init(n, style, rng=None):
    """Generate parametric initial sequences."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if style == 'uniform':
        a = np.ones(n)
    elif style == 'ramp_up':
        a = i + 1.0
    elif style == 'ramp_down':
        a = n - i
    elif style == 'beta_asym':
        # Beta-like distribution, asymmetric
        a = (i + 1) ** 2 * (n - i)
    elif style == 'gaussian':
        a = np.exp(-((i - center) ** 2) / (2.0 * (n / 5.0) ** 2))
    elif style == 'quarter_sine':
        a = np.sin(np.pi * (i + 1) / (2.0 * n)) + 0.05
    elif style == 'exponential':
        a = np.exp(-2.0 * i / n)
    elif style == 'two_bump':
        a = (np.exp(-((i - 0.3 * n) ** 2) / (2 * (n / 10) ** 2)) +
             np.exp(-((i - 0.7 * n) ** 2) / (2 * (n / 10) ** 2)))
    elif style == 'sine_mod':
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * i / (n / 3.0))
    elif style == 'power_law':
        a = (i + 1) ** (-0.3)
    elif style == 'noise':
        if rng is None:
            rng = np.random.default_rng(0)
        a = np.abs(rng.normal(1.0, 0.4, n))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.001)
    return a / np.sum(a)


def multi_start(n, num_starts=6, time_per_start=20.0, seed=42):
    """Multi-start search with diverse initializations."""
    rng = np.random.default_rng(seed)
    best_a = None
    best_obj = np.inf

    styles = ['ramp_up', 'ramp_down', 'beta_asym', 'gaussian',
              'quarter_sine', 'exponential', 'two_bump', 'sine_mod', 'power_law']

    for si in range(min(num_starts, len(styles))):
        if time_per_start < 3:
            break
        style = styles[si % len(styles)]
        a = parametric_init(n, style, rng)

        t1 = time_per_start * 0.5
        a_lp, obj_lp = alternating_lp(a, max_iters=40, time_budget=t1)

        t2 = time_per_start * 0.3
        a_pg, obj_pg = proximal_gradient(a_lp, max_iters=100, time_budget=t2)

        t3 = time_per_start * 0.2
        a_pol, obj_pol = lbfgs_polish(a_pg, time_budget=t3)

        obj_final = min(obj_lp, obj_pg, obj_pol)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_pol.copy() if obj_pol == obj_final else (a_pg if obj_pg == obj_final else a_lp.copy())

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.10f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining alternating LP, proximal gradient,
    L-BFGS polishing, multi-start, and n-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: score={best_score:.10f}, n={n_parent}")

    # Phase 1: Intensive alternating LP on parent
    print("\n=== Phase 1: Alternating LP on parent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.20, 180.0)
        print(f"  Budget: {alloc:.0f}s")
        a_lp, obj_lp = alternating_lp(best_seq, max_iters=300, time_budget=alloc, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  After LP: {best_score:.10f}")

    # Phase 2: Proximal gradient polish
    print(f"\n=== Phase 2: Proximal gradient (best: {best_score:.10f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.10, 80.0)
        a_pg, obj_pg = proximal_gradient(best_seq, max_iters=500, time_budget=alloc)
        if obj_pg < best_score:
            best_score = obj_pg
            best_seq = a_pg.copy()
            print(f"  After PG: {best_score:.10f}")

    # Phase 3: L-BFGS polish
    print(f"\n=== Phase 3: L-BFGS polish (best: {best_score:.10f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.10, 80.0)
        a_pol, obj_pol = lbfgs_polish(best_seq, time_budget=alloc, verbose=True)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After L-BFGS: {best_score:.10f}")

    # Phase 4: Another LP round (often helps after non-linear methods)
    print(f"\n=== Phase 4: Second LP round (best: {best_score:.10f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 120.0)
        a_lp2, obj_lp2 = alternating_lp(best_seq, max_iters=200, time_budget=alloc)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  After 2nd LP: {best_score:.10f}")

        # Quick PG
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.05, 30.0)
            a_pg2, obj_pg2 = proximal_gradient(best_seq, max_iters=200, time_budget=alloc)
            if obj_pg2 < best_score:
                best_score = obj_pg2
                best_seq = a_pg2.copy()
                print(f"  After 2nd PG: {best_score:.10f}")

    # Phase 5: Multi-start exploration with different n
    print(f"\n=== Phase 5: Multi-start exploration (best: {best_score:.10f}) ===")
    n_candidates = [n_parent, n_parent - 50, n_parent + 50,
                    n_parent - 100, n_parent + 100, 600, 800, 900]
    n_candidates = sorted(set(x for x in n_candidates if 400 <= x <= 1200))
    # Remove already tried
    n_candidates = [x for x in n_candidates if x != n_parent or n_parent not in [n_parent]]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # If we have a good sequence, try resampling to this n
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = normalize(a_init)
        else:
            a_init = np.ones(n_try) / n_try

        t_alloc = min(remaining * 0.15, 40.0)
        if t_alloc < 10:
            break

        # LP on resampled
        a_opt, obj_opt = alternating_lp(a_init, max_iters=50, time_budget=t_alloc * 0.6)
        # PG
        a_opt, obj_opt2 = proximal_gradient(a_opt, max_iters=100, time_budget=t_alloc * 0.25)
        obj_final = min(obj_opt, obj_opt2)

        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_opt if obj_opt <= obj_opt2 else a_opt
            print(f"  n={n_try}: ** NEW BEST {best_score:.10f} **")
        else:
            print(f"  n={n_try}: {obj_final:.10f}")

    # Phase 6: Perturbation-based local search
    print(f"\n=== Phase 6: Perturbation search (best: {best_score:.10f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 31337)
        remaining = deadline - time.monotonic()

        num_pert = min(20, int(remaining / 12))
        for pi in range(num_pert):
            if time.monotonic() > deadline - 30:
                break
            scale = 0.005 + 0.003 * pi
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_pert = normalize(a_pert)

            t_alloc = min((deadline - time.monotonic()) * 0.08, 15.0)
            if t_alloc < 5:
                break

            a_opt, obj_opt = alternating_lp(a_pert, max_iters=30, time_budget=t_alloc * 0.6)
            a_opt, obj_pg = proximal_gradient(a_opt, max_iters=50, time_budget=t
