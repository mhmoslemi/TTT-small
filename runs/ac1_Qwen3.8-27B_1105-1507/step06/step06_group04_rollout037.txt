from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s*s)

def normalize(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-30:
        return a / s
    return np.ones(len(a)) / len(a)

def compute_grad_fft(a, weights, fft_size):
    """Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (with edge effects)"""
    # Using FFT cross-correlation
    A = rfft(a, fft_size)
    W = rfft(weights, fft_size)
    # corr[m] = sum_j a[j] * w[m+j] (linear, with zero-padding)
    corr = irfft(A * np.conj(W), fft_size)
    # grad[i] = 2 * corr[i] for i = 0,...,n-1
    # But we need to be careful about edge effects
    # At the edges, the formula 2*sum_k w[k]*a[k-i] isn't quite right
    # because some terms drop out
    # For simplicity, use 2*corr[i] which is the "interior" gradient
    return 2.0 * corr[:len(a)]

def softmax_weights(conv, tau):
    """Compute softmax weights on the convolution values."""
    cmax = np.max(conv)
    shifted = np.clip((conv - cmax) / tau, -500, 500)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= s

    # Center gradient (project onto sum=const hyperplane)

    logits = -eta * grad_c
    logits -= np.max(logits)  # numerical stability
    weights = np.exp(logits)
    new_a = a * weights
    return normalize(new_a)

def lp_alternate(a, max_iters=50, time_budget=60):
    """LP alternating method."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        tl = min(8, max(3, remaining - 2))
        g, t_opt = lp_minpeak(a, time_limit=tl)
        if g is None:
            stall += 1
            if stall > 5:
                break
            continue
        
        # Line search
        best_t = 1.0
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize((1-t)*a + t*g)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = normalize((1-best_t)*a + best_t*g)
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break
    
    return best_a, best_obj

def fourier_perturb(a, rng, num_modes=3):
    """Perturb a few random Fourier modes."""
    n = len(a)
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    # Randomly select and perturb a few modes
    modes = rng.integers(1, size//2, size=min(num_modes, size//2-1))
    for m in modes:
        magnitude = np.abs(fa[m])
        if magnitude > 1e-10:
            # Phase and amplitude perturbation
            delta_phase = np.exp(1j * rng.normal(0, 0.5))
            delta_amp = rng.normal(0, 0.1)
            fa[m] = fa[m] * (1 + delta_amp) * delta_phase
            fa[size-m] = np.conj(fa[m])
    a_new = irfft(fa, size)[:n]
    return normalize(np.abs(a_new))  # Take absolute value to ensure non-neg
