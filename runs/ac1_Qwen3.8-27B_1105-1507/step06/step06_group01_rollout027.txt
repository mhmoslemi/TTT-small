import cvxpy as cp
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def solve_convex_active(a_init, active_set, n):
    """Solve the convex QP with only active constraints."""
    a = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [sum(a) == 1, a >= 0]
    
    for k in active_set:
        # Build (a*a)_k as a quadratic form
        # (a*a)_k = sum_{i+j=k, 0<=i,j<n} a_i * a_j
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        terms = [a[i] * a[k-i] for i in range(i_min, i_max + 1)]
        constraints.append(cp.sum(terms) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
    except:
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except:
            prob.solve(solver=cp.ECOS, verbose=False)
    
    if a.value is not None:
        return np.maximum(a.value, 0)
    return None

def cutting_plane_solve(a_init, max_iters=100, deadline=None):
    """Cutting-plane method for the convex QP."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    # Start with lag 0 and a few other lags
    active_set = [0, n // 2, n - 1, 2 * n - 2 - n // 2, 2 * n - 2]
    active_set = list(set(k for k in active_set if 0 <= k < 2 * n - 1))
    
    best_a = a.copy()
    best_val = fast_conv(a).max()
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break

        x = solve_convex_active(a, active_set, n)
        if x is None:
            break
        x = x / x.sum()
        
        # Check all constraints
        conv = fast_conv(x)
        violated = [k for k in range(2 * n - 1) if conv[k] > t_val + 1e-10]
        
        if not violated:

            if conv.max() < best_val:
                best_val = conv.max()
                best_a = x.copy()
            break
        
        # Add violated constraints (at most a few per iteration)
        violated.sort(key=lambda k: conv[k], reverse=True)
        active_set = set(active_set)
        for k in violated[:10]:

        active_set = list(active_set)

        if len(active_set) > 200:
            # Keep only the most important ones
            active_set = sorted(active_set, key=lambda k: conv[k], reverse=True)[:200]
        
        a = x
        if conv.max() < best_val:
            best_val = conv.max()
            best_a = x.copy()
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Edge-peaked initializations based on the insight that optimal sequences
   have larger values at the boundaries.
2. Full-constraint LP best-response (all 2n-1 autocorrelation constraints).

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(a, n):
    """Build the sparse LP constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_best_response(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    A = build_lp_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception as e:
        pass
    return None


def run_lp_iteration(a_init, max_iters, deadline, n_active=None, verbose=False):
    """LP mirror descent with full or partial constraints."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Solve LP
        g = solve_lp_best_response(a, best_M, time_limit=8)
        if g is None:
            no_improve += 1
            if no_improve > 3:
                break
            continue
        
        # Line search for mixing
        improved = False
        for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0
    
    return best_a


def mirror_descent_step(a, step):
    """One step of multiplicative weight mirror descent."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    k_star = int(np.argmax(b))
    
    # Subgradient at the active index
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    
    # Center the gradient
    grad_tan = grad - grad.mean()
    
    # Multiplicative update
    log_update = -step * grad_tan
    log_update -= log_update.max()
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    
    return a_new


def run_mirror_descent(a_init, max_iters, deadline, step_init=0.3):
    """Run mirror descent with adaptive step size."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        step = step_init / (1.0 + 0.01 * it)
        a = mirror_descent_step(a, step)
        
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a


def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient using LSE."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    scaled = tau * (b - b_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    exp_b = np.exp(scaled)
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * (p * a_full)[i] where a_full is a padded to length 2n-1
    # Use FFT for efficiency
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    # conv(p, a_pad) but we want sum_k p[k] * a[k-i] for each i
    # This is the cross-correlation of p and a
    # Actually: grad[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * (sum over k of p[k] * a[k-i])
    # For each i, k-i must be in [0, n-1], so k in [i, i+n-1]
    # This is the convolution of p and a, evaluated at specific points
    
    # Let's use FFT: FFT(p) * FFT(a) gives the circular convolution
    # But we need linear. Let's just do it with the direct formula for now.
    # For efficiency, use the fact that grad = 2 * correlate(p, a)
    
    # Use FFT-based correlation
    size = 1
    while size < len(p) + n - 1:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    # Cross-correlation: corr[i] = sum_k p[k+i] * a[k] (for valid i)
    # Actually we want: sum_k p[k] * a[k-i]
    # = sum_k p[k] * a[k-i]
    # Let's define q = flip(a), then sum_k p[k] * a[k-i] = (p * q)[n-1+i]
    # where * denotes circular convolution
    
    # Simpler: just compute directly for small n, or use FFT
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j] for i in [0, n-1]
    # (where p[k] = 0 for k outside [0, 2n-2])
    
    # Use the convolution: conv(p, a)[i+j] includes p[k]*a[i+j-k] for all k
    # We want sum_j p[i+j]*a[j] = sum_k p[k]*a[i+j-k+j]... 
    
    # Let me just use the direct formula with numpy for clarity:
    grad = np.zeros(n)
    for i in range(n):
        # sum_k p[k] * a[k-i] where 0 <= k-i < n, i.e., i <= k <= i+n-1
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo <= k_hi:
            grad[i] = 2.0 * np.dot(p[k_lo:k_hi+1], a[k_lo-i:k_hi-i+1])
    
    return smooth_val, grad


def run_fista(a_init, tau_start, tau_end, max_iters, deadline, lr=0.5):
    """FISTA on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters - 1))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        step = lr / (1.0 + 0.02 * tau)
        a_new = y - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        
        a_prev = a
        a = a_new
        y = y_new

        if (it + 1) % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:

def coordinate_optimize(a, i, b, n):
    """Optimize coordinate i by golden section search on the max autocorrelation."""
    x_cur = a[i]
    if x_cur == 0:
        return 0.0
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute bases and slopes
    c_vals = []
    s_vals = []
    has_quad = False
    quad_idx = -1
    
    for k in range(k_lo, k_hi + 1):
        j = k - i
        if j == i:  # This means k = 2i
            c_vals.append(b[k] - x_cur * x_cur)
            s_vals.append(0.0)
            has_quad = True
            quad_idx = len(c_vals) - 1
        elif 0 <= j < n:
            c_vals.append(b[k] - 2.0 * x_cur * a[j])
            s_vals.append(2.0 * a[j])
        else:
            c_vals.append(b[k])
            s_vals.append(0.0)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(s_vals)
    
    def obj_at(x):
        vals = c_arr + s_arr * x
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_idx] + x * x
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(3.0 * x_cur, 1.0)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi
