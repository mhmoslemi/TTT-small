def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    return fast_conv(a).max()

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    rho = int(np.argmax(~cond)) if (~cond).any() else n - 1
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution (assumes a is normalized to sum=1)."""
    return fast_conv(a).max()


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if (~cond).any():
        rho = int(np.argmax(~cond))
    else:
        rho = n - 1
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_max_grad_fft(a, tau):
    """Smooth max of conv(a,a) via log-sum-exp, with FFT-based gradient.
    Returns (smooth_value, gradient_vector)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / tau

    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # Compute via FFT: cross[i] = conv(p, a_rev)[i+n-1]
    a_rev = a[::-1]
    size = 1
    while size < (2 * n - 1) + (n - 1):
        size <<= 1
    fp = np.fft.rfft(p, size)
    fra = np.fft.rfft(a_rev, size)
    conv_pa = np.fft.irfft(fp * fra, size)
    grad = 2.0 * conv_pa[n - 1:2 * n - 1]

    return smooth_val, grad


def run_fista(a0, deadline, tau_start=100.0, lr=0.3):
    """FISTA on smooth log-sum-exp approximation of max(conv(a,a))."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        tau = tau_start * (1.0 + 0.05 * it / 100.0)
        _, grad_y = smooth_max_grad_fft(y, tau)
        grad_tan = grad_y - grad_y.mean()
        step = lr / (1.0 + 0.02 * it)
        a_new = proj_simplex(y - step * grad_tan)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new
        y_new = proj_simplex(a_new + alpha * (a_new - a_prev))

        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new

        if it % 30 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M * (1 - 1e-9):
                best_M = M_cur
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 30:
                break

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    return best_a


def build_lp_matrix(a, active_indices):
    """Build sparse LP constraint matrix for given active indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, M, n_active, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top n_active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    top_idx = np.argsort(b)[::-1][:min(n_active, 2 * n - 1)]

    A = build_lp_matrix(a, top_idx)
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_mirror(a0, deadline, n_active_start=40, n_active_max=200):
    """LP mirror descent with adaptive active set size."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    n_active = n_active_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = b.max()

        g = lp_best_response(a, M, n_active, time_limit=8)
        if g is None:
            n_active = min(n_active + 20, n_active_max)
            no_improve += 1
            if no_improve > 10:
                break
            continue

        # Try mixing with geometric step sizes
        improved = False
        for t in [0.4, 0.6, 0.3, 0.5, 0.7, 0.8, 0.15, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = normalize(a_trial)
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1.0 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            # Try to shrink active set for speed
            if n_active > 25 and it % 5 == 0:
                n_active = max(25, n_active - 5)
        else:
            no_improve += 1
            n_active = min(n_active + 15, n_active_max)
            if no_improve > 20:
                break

    return best_a


def pairwise_swap_optimize(a, i, j, n):
    """Find best δ for swap a[i]+=δ, a[j]-=δ minimizing max(conv).
    Uses golden section search."""
    b = fast_conv(a)
    M = b.max()

    def eval_delta(d):
        a[i] += d
        a[j] -= d
        val = fast_conv(a).max()
        a[i] -= d
        a[j] += d
        return val

    lo = -a[i] + 1e-12
    hi = a[j] - 1e-12
    if hi <= lo:
        return 0.0

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = eval_delta(c_pt)
    fd = eval_delta(d_pt)

    for _ in range(35):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_delta(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_delta(d_pt)

    return (lo + hi) / 2.0


def run_pairwise_swaps(a0, deadline, n_tries=2000):
    """Iteratively find and apply best pairwise swaps."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)
    no_improve = 0

    for it in range(n_tries):
        if time.monotonic() > deadline:
            break
        # Pick random pair (bias toward high-value coordinates)
        weights = a + 0.01
        weights /= weights.sum()
        i = rng.choice(n, p=weights)
        j = rng.choice(n, p=weights)
        if i == j:
            continue

        delta = pairwise_swap_optimize(a, i, j, n)
        if abs(delta) > 1e-8:
            a[i] += delta
            a[j] -= delta
            a = normalize(a)
            M_new = fast_conv(a).max()
            if M_new < best_M * (1 - 1e-10):
                best_M = M_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            break

    return best_a


def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        # Resample parent to length n
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        p = np.interp(t_new, t_old, parent)
        p = np.maximum(p, 0.0)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed parent
        for std in [0.01, 0.03, 0.05, 0.1]:
            pp = np.maximum(p + rng.normal(0, std, n), 1e-6)
            inits.append(pp / pp.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(cos_w, 0.01) / np.maximum(cos_w, 0.01).sum())

    # Flat-top
    flat = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    inits.append(flat / flat.sum())

    # Random smooth
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: FISTA -> LP mirror descent -> pairwise swaps.
    Focuses time on LP-based local search for fine improvement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)
        tracker.update(parent)
        print(f"No parent, using uniform n=768", flush=True)

    n_parent = parent.size
    n = n_parent  # Work at parent length primarily

    # === Phase 1: Quick FISTA from parent (120s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: FISTA refinement (n={n})", flush=True)
    print(f"{'='*60}", flush=True)

    t1_deadline = time.monotonic() + 120
    a_fista = run_fista(tracker.best if tracker.best is not None else parent,
                        t1_deadline, tau_start=200.0, lr=0.2)
    tracker.update(a_fista)
    tracker.report("After FISTA")

    # === Phase 2: LP Mirror Descent (main workhorse, ~800s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    t2_deadline = time.monotonic() + 800
    a_lp = run_lp_mirror(tracker.best if tracker.best is not None else parent,
                         t2_deadline, n_active_start=40, n_active_max=300)
    tracker.update(a_lp)
    tracker.report("After LP round 1")

    # Second LP pass with tighter parameters
    if time.monotonic() < deadline - 300:
        t2b_deadline = time.monotonic() + 200
        a_lp2 = run_lp_mirror(tracker.best, t2b_deadline, n_active_start=80, n_active_max=500)
        tracker.update(a_lp2)
        tracker.report("After LP round 2")

    # === Phase 3: More LP with different params ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: LP with expanded active set", flush=True)
        print(f"{'='*60}", flush=True)
        t3_deadline = time.monotonic() + 150
        a_lp3 = run_lp_mirror(tracker.best, t3_deadline, n_active_start=100, n_active_max=600)
        tracker.update(a_lp3)
        tracker.report("After LP round 3")

    # === Phase 4: FISTA with high tau for fine tuning ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 4: High-tau FISTA", flush=True)
        print(f"{'='*60}", flush=True)
        t4_deadline = time.monotonic() + 60
        a_f4 = run_fista(tracker.best, t4_deadline, tau_start=500.0, lr=0.1)
        tracker.update(a_f4)
        tracker.report("After high-tau FISTA")

    # Another LP pass
    if time.monotonic() < deadline - 60:
        t4b_deadline = time.monotonic() + 40
        a_lp4 = run_lp_mirror(tracker.best, t4b_deadline, n_active_start=60, n_active_max=400)
        tracker.update(a_lp4)
        tracker.report("After LP round 4")

    # === Phase 5: Pairwise swaps for fine-tuning ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 5: Pairwise swaps", flush=True)
        print(f"{'='*60}", flush=True)
        t5_deadline = time.monotonic() + 60
        a_sw = run_pairwise_swaps(tracker.best, t5_deadline, n_tries=5000)
        tracker.update(a_sw)
        tracker.report("After pairwise swaps")

    # === Phase 6: Final LP polish ===
    if time.monotonic() < deadline - 10:
        t6_deadline = deadline - 5
        a_final = run_lp_mirror(tracker.best, t6_deadline, n_active_start=50, n_active_max=300)
        tracker.update(a_final)

    # === Phase 7: Try slightly different lengths ===
    if time.monotonic() < deadline - 20 and tracker.best is not None:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 7: Length exploration", flush=True)
        print(f"{'='*60}", flush=True)
        n_cur = tracker.best.size
        for mult in [1.1, 0.9, 1.2]:
            if time.monotonic() > deadline - 15:
                break
            new_n = int(n_cur * mult)
            new_n = max(256, min(new_n, 2048))
            if new_n == n_cur:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = normalize(a_res)
            sub_dl = min(deadline - 10, time.monotonic() + 20)
            a_opt = run_lp_mirror(a_res, sub_dl, n_active_start=40, n_active_max=200)
            tracker.update(a_opt)

    tracker.report("FINAL")

    if tracker.best is None:
        tracker.best = np.ones(768)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
