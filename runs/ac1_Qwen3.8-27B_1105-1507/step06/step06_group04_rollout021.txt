    a = a / sum(a)  # normalize

    best_obj = 2 * n * max(conv(a,a))  # with sum=1

        k_star = argmax(r)
        peak = r[k_star]

                g[i] += a[i + k_star]
            if i - k_star >= 0:
                g[i] += a[i - k_star]

import numpy as np

def fft_convolve_self(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_convolve_self(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_cross_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):
    """
    Solve: min t s.t. (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g_normalized, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=20):
    """
    Solve: max sum(g) s.t. (fixed_seq * g)[k] <= M for all k, g >= 0.
    Returns g (normalized) or None.
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def exact_subgradient(a, k_star):
    """Compute exact subgradient of r(k_star) w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    if k_star >= n:
        # For k_star >= n, r(k_star) = sum_{i=k_star-n+1}^{n-1} a[i]*a[i-(k_star-n+1)]
        # Actually for linear conv, r(k) for k >= n uses the "tail"
        # r(k) = sum_{i=0}^{n-1} a[i] * a_pad[k-i] where a_pad is zero-padded
        # For k < n: r(k) = sum_{i=0}^{n-1-k} a[i+k]*a[i]
        # For k >= n: r(k) = sum_{i=k-n+1}^{n-1} a[i]*a[i-(k-n)]  ... let me just use the general form
        # r(k) = sum_i a[i]*a[i+k] where a is zero-padded outside [0,n-1]
        # For k >= 0: sum_{i=max(0,k-n+1)}^{min(n-1, n-1-k+n-1)} ... 
        # Simpler: r(k) for k in [0, 2n-2]:
        #   if k < n: sum_{i=0}^{n-1-k} a[i] * a[i+k]
        #   if k >= n: sum_{i=k-n+1}^{n-1} a[i] * a[i-(k-n)]
        # For k >= n: let j = k - n, then r(k) = sum_{i=j+1}^{n-1} a[i]*a[i-j-1+1] = sum_{i=j+1}^{n-1} a[i]*a[i-j]
        # Wait, let me just handle it directly
        j = k_star - n + 1
        # r(k_star) = sum_{i=j}^{n-1} a[i] * a[i - j]  (where i-j >= 0 means i >= j)
        # Actually: r(k) = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
        # For k >= n: k-i < n means i > k-n, so i from k-n+1 to n-1
        # r(k) = sum_{i=k-n+1}^{n-1} a[i]*a[k-i]
        for i in range(k_star - n + 1, n):
            g[i] += a[k_star - i]  # derivative w.r.t. a[i]: a[k_star - i]
            g[k_star - i] += a[i]  # derivative w.r.t. a[k_star - i]: a[i]
    else:
        for i in range(n - k_star):
            g[i] += a[i + k_star]
            g[i + k_star] += a[i]
    return g


def subgradient_exact(a, max_iters=2000, time_budget=100.0, top_k=1):
    """
    Subgradient method with exact subgradient (averaged over top-k peaks),
    Nesterov momentum, and adaptive step size.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.82
    step = 0.05
    # Best value tracking for Polyak-like step
    f_star = best_obj
    n_steps = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = fft_convolve_self(a)
        # Find top-k peaks
        if top_k == 1:
            k_star = int(np.argmax(conv))
            g = exact_subgradient(a, k_star)
        else:
            threshold = conv[k_star] if 'k_star' in dir() else np.max(conv)
            # Use top-k with weighting
            order = np.argsort(conv)[::-1]
            top_indices = order[:top_k]
            # Weight by proximity to max
            weights = np.exp(-0.5 * (conv[order[:top_k]] - np.max(conv)) / (0.01 * np.max(conv) + 1e-10))
            weights /= np.sum(weights)
            g = np.zeros(n)
            for wi, ki in zip(weights, top_indices):
                g += wi * exact_subgradient(a, int(ki))

        # Center gradient (projection onto sum=const hyperplane)
        g_centered = g - np.mean(g)
        g_norm = np.linalg.norm(g_centered)
        if g_norm < 1e-15:
            break

        # Adaptive step: combine decreasing and Polyak-like
        # Polyak: step = (f - f*) / ||g||^2, but we don't know f*
        # Use: step = step0 / (1 + it*0.01)
        step_current = step / (1.0 + it * 0.005)

        # Nesterov momentum
        v_new = mu * v - step_current * g_centered / g_norm
        a_new = a - v_new
        a_new = normalize_simplex(a_new)

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            v = v_new
        else:
            # Try without momentum
            a_try = a - step_current * g_centered / g_norm
            a_try = normalize_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                v = np.zeros(n)
            else:
                # Backtrack
                v = mu * v
                a = a - 0.5 * step_current * v / (np.linalg.norm(v) + 1e-15)
                a = normalize_simplex(a)
                step *= 0.95

        n_steps += 1

    return best_a, best_obj


def spectral_shaping(a, max_iters=100, time_budget=30.0):
    """
    Iterative spectral shaping: modify the DFT to flatten the power spectrum,
    then project back to non-negative simplex.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 2 * n  # Use circular conv size for frequency domain ops

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute DFT
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2  # Power spectrum (real, non-negative)

        # Flatten: P_new = P^alpha for alpha < 1 (reduces peaks in spectrum)
        alpha = 0.7 + 0.2 * (it / max_iters)  # Start aggressive, become conservative
        P_flat = P ** alpha
        P_flat /= np.sum(P_flat) * len(P_flat)  # Normalize

        # Take square root to get new magnitude
        mag_new = np.sqrt(P_flat + 1e-30)

        # Keep original phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)

        # Inverse DFT
        a_new = irfft(A_new, fft_size)
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            # Fall back
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        # Blend with original for stability
        for blend_t in [0.3, 0.5, 0.7, 1.0]:
            a_blend = (1 - blend_t) * a + blend_t * a_new
            a_blend = normalize_simplex(a_blend)
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_obj:
                best_obj = obj_blend
                best_a = a_blend.copy()
                a = a_blend
                break

    return best_a, best_obj


def fw_peak_reduction(a, max_iters=500, time_budget=50.0):
    """
    Frank-Wolfe style peak reduction: at each step, move mass towards
    the coordinate with smallest subgradient component.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    momentum = 0.0
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = fft_convolve_self(a)
        k_star = int(np.argmax(conv))
        g = exact_subgradient(a, k_star)
        g_centered = g - np.mean(g)

        # FW direction: towards argmin(g)
        j = int(np.argmin(g))
        s = np.zeros(n)
        s[j] = 1.0
        direction = s - a  # FW direction

        # Blend with gradient direction
        g_norm = np.linalg.norm(g_centered)
        if g_norm > 1e-15:
            grad_dir = -g_centered / g_norm
            direction = 0.5 * direction + 0.5 * grad_dir
            direction = normalize_simplex(direction + np.ones(n) / n) - a  # Make it a direction from a

        # Adaptive step
        t = 2.0 / (it + 2)  # FW theoretical step
        t = min(t, 0.3)

        # Try several step sizes
        best_t = 0.0
        best_val = best_obj
        for t_try in [t * 0.25, t * 0.5, t, t * 1.5, t * 2]:
            t_try = min(t_try, 1.0)
            a_try = (1 - t_try) * a + t_try * s
            a_try = normalize_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_val:
                best_val = obj_try
                best_t = t_try

        if best_t > 0:
            a = (1 - best_t) * a + best_t * s
            a = normalize_simplex(a)
            if best_val < best_obj:
                best_obj = best_val
                best_a = a.copy()
        else:
            # Try gradient step
            if g_norm > 1e-15:
                a_try = a - step * g_centered / g_norm
                a_try = normalize_simplex(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
            step *= 0.98

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

    starts.append(np.ones(n) / n)

    # Triangular
    tri = np.maximum(1.0 - np.abs(i - center) / center, 0.01)
    starts.append(tri / np.sum(tri))

    # Gaussian narrow
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 5.0) ** 2))
    starts.append(bell / np.sum(bell))

    # Gaussian wide
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 2.5) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # Cosine envelope
    cos_env = np.cos(np.pi * (i - center) / n) ** 2
    cos_env = np.maximum(cos_env, 0.001)
    starts.append(cos_env / np.sum(cos_env))

    # Parametric with oscillation
    for freq in [2, 3, 5]:
        amp = 0.3
        base = np.exp(-((i - center) ** 2) / (2 * (n / 4.0) ** 2))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n)
        a = np.maximum(base * mod, 0)
        s = np.sum(a)
        if s > 1e-10:
            starts.append(a / s)

    # Random smooth
    for _ in range(3):
        rough = rng.random(n)
        # Smooth with moving average
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        smooth = np.convolve(rough, kernel, mode='same')
        smooth = np.maximum(smooth, 0.001)
        starts.append(smooth / np.sum(smooth))

    return starts


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_pipeline(a, time_budget=100.0, n_max_lp=1000):
    """
    Full polishing pipeline: alternate LP and subgradient methods.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)

    # Phase 1: LP alternating
    remaining = time_budget - (time.monotonic() - start)
    if n <= n_max_lp and remaining > 20:
        alloc = min(remaining * 0.35, 50.0)
        a1, obj1 = lp_alternate_minpeak(best_a, max_iters=30, time_budget=alloc)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            print(f"    LP minpeak: {best_obj:.8f}")

    # Phase 2: Subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = min(remaining * 0.35, 40.0)
        a2, obj2 = subgradient_exact(best_a, max_iters=int(alloc * 15), time_budget=alloc, top_k=1)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"    Subgrad: {best_obj:.8f}")

    # Phase 3: LP maxsum
    remaining = time_budget - (time.monotonic() - start)
    if n <= n_max_lp and remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a3, obj3 = lp_alternate_maxsum(best_a, max_iters=25, time_budget=alloc)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"    LP maxsum: {best_obj:.8f}")

    # Phase 4: Spectral shaping
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.3, 20.0)
        a4, obj4 = spectral_shaping(best_a, max_iters=50, time_budget=alloc)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            print(f"    Spectral: {best_obj:.8f}")

    # Phase 5: FW peak reduction
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.3, 20.0)
        a5, obj5 = fw_peak_reduction(best_a, max_iters=200, time_budget=alloc)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()
            print(f"    FW: {best_obj:.8f}")

    # Phase 6: Final subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.5, 30.0)
        a6, obj6 = subgradient_exact(best_a, max_iters=int(alloc * 20), time_budget=alloc, top_k=3)
        if obj6 < best_obj:
            best_obj = obj6
            best_a = a6.copy()
            print(f"    Final subgrad: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multi-start exploration, LP alternating,
    exact subgradient with momentum, spectral shaping, Frank-Wolfe,
    and length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    print("=== Phase 0: Loading parent ===")
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_best = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f} (n={n_best})")

    # Phase 1: Quick multi-start exploration across lengths
    print("\n=== Phase 1: Multi-start exploration ===")
    n_candidates = sorted(set([
        n_best, n_best - 100, n_best + 100, n_best + 200,
        600, 800, 900, 1000, 1100, 1200
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1400]

    candidates = []  # (obj, seq, n)

    for n in n_candidates:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        starts = generate_diverse_starts(n, seed=seed + n)
        # Include parent resampled if n != n_best
        if best_seq is not None and n != n_best:
            starts.append(normalize_simplex(resample_sequence(best_seq, n)))
        elif best_seq is not None:
            starts.insert(0, normalize_simplex(best_seq.copy()))

        # Quick screen each start
        alloc_per_start = min(remaining / len(starts), 8.0)
        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 250:
                break
            if alloc_per_start < 2:
                break

            # Quick subgradient
            a_opt, obj_opt = subgradient_exact(s, max_iters=int(alloc_per_start * 8),
                                               time_budget=alloc_per_start * 0.6, top_k=1)
            # Quick spectral
            if time.monotonic() < deadline - 200:
                a_opt2, obj_opt2 = spectral_shaping(a_opt, max_iters=20,
                                                    time_budget=alloc_per_start * 0.3)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            candidates.append((obj_opt, a_opt.copy(), n))
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")

    # Sort and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates: {[f'{c[0]:.6f} (n={c[2]})' for c in candidates[:6]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive optimization of top candidates
    print("\n=== Phase 2: Intensive optimization ===")
    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        alloc = min(remaining * 0.3, 80.0)

        print(f"  Cand {ci} (n={n_cand}, obj={obj_cand:.8f}): {alloc:.0f}s budget")
        a_pol, obj_pol = polish_pipeline(a_cand, time_budget=alloc, n_max_lp=1000)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"    ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"    No improvement (got {obj_pol:.8f})")

    # Phase 3: Length exploration with best sequence
    print(f"\n=== Phase 3: Length exploration (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        test_lengths = sorted(set([n_cur - 50, n_cur + 50, n_cur - 100,
                                   n_cur + 100, n_cur + 150, n_cur + 200]))
        test_lengths = [x for x in test_lengths if 400 <= x <= 1500 and x != n_cur]

        for n_test in test_lengths:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_res = normalize_simplex(resample_sequence(best_seq, n_test))
            alloc = min(remaining * 0.2, 30.0)
            a_pol, obj_pol = polish_pipeline(a_res, time_budget=alloc, n_max_lp=1000)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** UPDATE to {obj_pol:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.5, 60.0)
            a_final, obj_final = polish_pipeline(best_seq, time_budget=alloc, n_max_lp=1000)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Polish: {best_score:.8f}")
            else:
                print(f"  Polish: {obj_final:.8f}")

        # Perturbation-based exploration
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_fin = len(best_seq)
            rng = np.random.default_rng(seed + 77777)
            for pi in range(6):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.005 + 0.005 * pi
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_fin))
                a_pert = normalize_simplex(a_pert)
                t_alloc = min(remaining * 0.1, 8.0)
                if t_alloc < 3:
                    break
                a_p, obj_p = subgradient_exact(a_pert, max_iters=int(t_alloc * 15),
                                               time_budget=t_alloc, top_k=2)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Perturb {pi} (s={scale:.4f}): ** {best_score:.8f} **")

    # Phase 5: Final LP on absolute best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_fin = len(best_seq)
        if n_fin <= 1000 and remaining > 20:
            print(f"\n=== Phase 5: Final LP (n={n_fin}) ===")
            a_lp, obj_lp = lp_alternate_minpeak(best_seq, max_iters=25,
                                                time_budget=min(remaining * 0.6, 40.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_lp2, obj_lp2 = lp_alternate_maxsum(best_seq, max_iters=20,
                                                     time_budget=min(remaining * 0.6, 30.0))
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  MaxSum LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
