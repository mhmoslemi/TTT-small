def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

    return np.maximum(np.interp(t_new, t_old, src), 0.0)

At the optimum, we have:

The KKT conditions say:
- There exists t such that (a*a)[k] <= t for all k
- There exist multipliers λ_k >= 0 for the constraints (a*a)[k] <= t, with λ_k > 0 only when (a*a)[k] = t
- There exists a multiplier ν for the equality constraint sum(a) = 1
- There exist multipliers μ_i >= 0 for a_i >= 0
- Stationarity: 2 * sum_k λ_k * (∂(a*a)[k]/∂a[i]) - ν + μ_i = 0
  i.e., 2 * sum_{k: λ_k > 0} λ_k * (a[k-i] + a[i-k] if k != 2i, or 2a[i] if k=2i) - ν + μ_i = 0

For interior points (a[i] > 0, μ_i = 0):
  2 * sum_k λ_k * (a[k-i] + a[i-k]) = ν (with the 2i special case)

This means that at the optimum, the "weighted sum of adjacencies" is constant for all i with a[i] > 0.

This gives us a fixed-point condition that we can iterate on:
1. Given a, compute conv = a*a
2. Set λ_k = 1/max(conv) for k where conv[k] ≈ max(conv), 0 otherwise
   (or use a soft version: λ_k ∝ exp(τ * conv[k]) normalized)
3. Compute the "stationarity residual": r[i] = sum_k λ_k * (a[k-i] + a[i-k])
4. The ideal a satisfies r[i] = constant for all i with a[i] > 0
5. Update a to make r more uniform (e.g., a[i] *= exp(-eta * (r[i] - mean(r))))

This is essentially the "water-filling" or "proportional fitting" approach I mentioned earlier.

Multi-strategy optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                    max_after = 0.0
                max_unaffected = max(max_before, max_after) if (k_min > 0 or k_max < 2 * n - 2) else 0.0

                new_max = max(max_affected, max_unaffected)
                new_s = s + delta
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * new_max / (new_s * new_s)

                if new_obj < best_local_obj:
                    best_local_obj = new_obj
                    best_x = x

            if abs(best_x - old) > 1e-13:
                delta = best_x - old
                a_slice = a[:k_max - i + 1]
                conv[k_min:k_max + 1] += 2.0 * delta * a_slice
                k_self = 2 * i
                if 0 <= k_self < 2 * n - 1 and k_min <= k_self <= k_max:
                    conv[k_self] += delta * delta
                a[i] = best_x
                s = s + delta
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 200 == 0:
            print(f"    CD-ACTIVE sweep {sweep}: obj={best_obj:.8f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def smooth_gradient_descent(a, max_iters, deadline, tau_init=50.0, tau_final=500.0,
                            lr=0.02, verbose=False):
    """
    Smoothed gradient descent on the simplex using log-sum-exp approximation
    of max(conv(a,a)). Uses Nesterov acceleration and adaptive step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal tau
        frac = it / max(1, max_iters)
        tau = tau_init + (tau_final - tau_init) * frac

        # Compute smooth gradient at y
        c = fast_conv(y)
        c_max = c.max()
        # Log-sum-exp weights
        exp_c = np.exp(tau * (c - c_max))
        w = exp_c / exp_c.sum()

        # Gradient of sum_k w[k] * conv[k] w.r.t. a[i]
        # = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
        # = (w * a)[2i] type computation via FFT
        # More precisely: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
        # Let's compute via FFT
        # sum_k w[k]*a[k-i] = (w * a_rev)[n-1+i] where a_rev = a[::-1]
        # Actually: sum_k w[k]*a[k-i] for valid k,i is the cross-correlation
        # Let's use direct FFT approach

        # grad[i] = (w * a)[2i] + (w_rev * a)[2(n-1-i)] ... 
        # Actually simpler: 
        # Let c_w[k] = w[k] * conv[k] -- no, that's not right.
        # 
        # The smooth objective is: phi(a) = (1/tau) * log(sum_k exp(tau * (a*a)[k]))
        # grad phi / grad a[i] = sum_k w[k] * d(a*a)[k] / da[i]
        # d(a*a)[k]/da[i] = a[k-i] + a[i-k] (with bounds, 2a[i] if k=2i)
        # = (w * a)[at position giving sum over k of w[k]*a[k-i]]
        #   + (w_rev * a)[at position giving sum over k of w[k]*a[i-k]]
        
        # First term: T1[i] = sum_k w[k] * a[k-i] for k-i in [0,n-1]
        # = sum_j w[i+j] * a[j] for j in [0, n-1] and i+j in [0, 2n-2]
        # This is like a cross-correlation of w and a
        # T1 = cross_corr(w, a) where (T1)[i] = sum_j w[i+j]*a[j]
        # Using FFT: T1 = ifft(fft(w) * conj(fft(a))) -- not quite
        # Actually T1[i] = sum_j w[i+j] * a[j] = (w * a_flipped)[i + (2n-2)] hmm
        # Let me just compute it directly for clarity, then optimize if needed
        # For n=768, direct computation is O(n^2) = 600K per iteration, which is fine
        
        # Actually, let's use the efficient FFT approach:
        # T1[i] = sum_{j=0}^{n-1} w[i+j] * a[j] where i+j <= 2n-2
        # = cross-correlation of w (length 2n-1) and a (length n)
        # Using FFT:
        size = 1
        while size < 3 * n:
            size <<= 1
        W_fft = np.fft.rfft(w, size)
        A_fft = np.fft.rfft(a, size)
        # T1[i] = sum_j w[i+j]*a[j] = (w cross-corr a)[i]
        # cross_corr(x, y)[k] = sum_j x[k+j]*y[j]
        # = ifft(conj(fft(x)) * fft(y))[k + len(x) - 1] ... 
        # Let's just use: T1 = ifft(fft(w_padded) * conj(fft(a_padded))) 
        # Hmm, this is getting complicated. Let me use a simpler formula.
        # 
        # T1[i] = sum_{k=0}^{2n-2} w[k] * a[k-i] * 1[0<=k-i<=n-1]
        # = sum_{k=i}^{i+n-1} w[k] * a[k-i] (with bounds on k)
        # = sum_{j=0}^{n-1} w[i+j] * a[j] (with bounds)
        # This is the (i+2n-2)-th coefficient of the product of FFT(w) and FFT(a_reversed)?
        # 
        # Let me just compute both terms using convolution:
        # T1[i] = sum_j w[i+j]*a[j] -- this is like (reversed w) convolved with a
        # Let wr = w[::-1] (length 2n-1)
        # Then T1[i] = (wr * a)[(2n-2) - i + 0] ... 
        # 
        # OK let me just do it the simple way with numpy for correctness:
        
        # T1[i] for i in 0..n-1: sum of w[k]*a[k-i] for k in [i, min(i+n-1, 2n-2)]
        # T2[i] for i in 0..n-1: sum of w[k]*a[i-k] for k in [max(0,i-n+1), min(i, 2n-2)]
        # Note T2[i] = T1[2(n-1)-i] by symmetry if w is symmetric, but it's not.
        # Actually T2[i] = sum_k w[k]*a[i-k] = sum_j w[i-j]*a[j] = T1 with w and indices swapped
        # T2[i] = sum_j a[j] * w[i-j] for valid j, i-j
        # = sum_j a[j] * w[i-j]
        # Let j' = i-j, then j = i-j', and T2[i] = sum_{j'} a[i-j'] * w[j']
        # This is the same as (a * w)[i] (convolution of a and w at position i)
        # And T1[i] = sum_j a[j] * w[i+j] = (a * w_rev)[2n-2-i] where w_rev = w[::-1]
        
        # Let me compute:
        # T2 = convolve(a, w, mode='full')[0:n+1] -- gives sum_j a[j]*w[i-j] for i=0..2n-2
        # We want T2[i] for i=0..n-1
        # T1[i] = sum_j a[j]*w[i+j] = (a * w_rev)[(2n-2)-i + 0]... 
        # Let me just compute T1 and T2 via FFT:
        
        # T1: pad w to length 3n, pad a to length 3n
        w_padded = np.zeros(size)
        w_padded[:2*n-1] = w
        a_padded = np.zeros(size)
        a_padded[:n] = a
        
        # T1[i] = sum_j w[i+j]*a[j] = cross-correlation
        # = ifft(conj(fft(w_padded)) * fft(a_padded))
        # Hmm, let me think again.
        # Cross-correlation R[i] = sum_j x[j+i] * y[j]
        # R = ifft(conj(X) * Y) where X=fft(x), Y=fft(y)
        # So T1 = ifft(conj(FFT(w_padded)) * FFT(a_padded))[0:n]
        
        X = np.fft.rfft(w_padded, size)
        Y = np.fft.rfft(a_padded, size)
        T1_full = np.fft.irfft(np.conj(X) * Y, size)[:n]
        
        # T2[i] = sum_j a[j] * w[i-j] = (a * w)[i] (standard convolution)
        # = ifft(FFT(a_padded) * FFT(w_padded))[0:n]
        T2_full = np.fft.irfft(Y * X, size)[:n]
        
        grad = T1_full + T2_full
        
        # Handle the k=2i correction: for each k=2i, the derivative is 2*a[i] not a[i]+a[i]
        # Wait, a[k-i] + a[i-k] when k=2i gives a[i] + a[i] = 2*a[i], which is correct.
        # So no correction needed. Good.
        
        # Project gradient to simplex tangent space
        grad = grad - np.mean(grad)

        # Nesterov step
        t_new = 2.0 / (it + 2.0 + beta)
        y_new = a + (t_new - 1.0) / t_new * (a - y)  # Nesterov extrapolation
        
        # Adaptive step size (simple: fixed with decay)
        step = lr / (1.0 + 0.01 * it)
        
        # Gradient step on y
        y_updated = y_new - step * grad
        
        # Project to simplex
        y_projected = project_simplex(y_updated)
        
        # Update
        a_new = y_projected
        
        # Check improvement
        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Try smaller step
                step *= 0.5
                y_updated = y_new - step * grad
                y_projected = project_simplex(y_updated)
                a_new = y_projected
                obj_new = compute_J(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    no_improve = 0
                else:
                    no_improve = 0  # Reset counter but keep best
        
        y = a_new
        a = a_new

        if verbose and it % 200 == 0:
            print(f"    Smooth-GD it {it}: tau={tau:.0f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def active_set_lp(a, max_iters, deadline, top_k=10, verbose=False):
    """
    Active-set LP: at each iteration, solve LP using only the top-k most active
    convolution constraints. Uses backtracking step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Find top-k active constraints
        if top_k is not None and top_k < 2 * n - 1:
            top_indices = np.argsort(c)[::-1][:top_k]
        else:
            top_indices = np.arange(2 * n - 1)

        # Build LP: minimize t
        # s.t. 2*(a*g)[k] - t <= c[k] for k in top_indices
        #      sum(g) = 1
        #      g >= 0
        # Variables: g[0..n-1], t (index n)
        
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in top_indices:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(len(b_ub))
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(float(c[k]))

        num_constraints = len(b_ub)
        A = coo_matrix((vals, (rows, cols)), shape=(num_constraints, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_eq = coo_matrix((np.ones(n), ([0] * n, list(range(n)))), shape=(1, n + 1)).tocsr()
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Backtracking step size
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

        if verbose and it % 10 == 0:
            print(f"    Active-LP it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def water_filling(a, max_iters, deadline, eta=0.1, verbose=False):
    """
    Multiplicative water-filling: iteratively adjust a[i] to flatten the convolution.
    a[i] *= exp(-eta * responsibility[i]) where responsibility measures how much
    a[i] contributes to the peaks of conv.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_mean = c.sum() / len(c)  # Should be ~1/(2n-1) for normalized a
        c_max = c.max()

        # Compute "excess" at each position
        excess = c - c_mean

        # Responsibility of a[i]: sum over k where excess[k] > 0 of excess[k] * contribution
        # contribution of a[i] to conv[k] (derivative) = a[k-i] + a[i-k]
        # But we want the "responsibility" which is excess[k] * a[k-i] (one-sided)
        # Actually, let's use: r[i] = sum_k excess[k] * (a[k-i] * 1[0<=k-i<n] + a[i-k] * 1[0<=i-k<n])
        # This is the "gradient" of sum_k excess[k]^2 / 2 w.r.t. a[i]
        # = sum_k excess[k] * (a[k-i] + a[i-k])
        # Can compute via FFT

        # Use only positive excess (peaks) for the update
        excess_pos = np.maximum(excess, 0)
        
        size = 1
        while size < 3 * n:
            size <<= 1
        e_padded = np.zeros(size)
        e_padded[:2*n-1] = excess_pos
        a_padded = np.zeros(size)
        a_padded[:n] = a
        
        E_fft = np.fft.rfft(e_padded, size)
        A_fft = np.fft.rfft(a_padded, size)
        
        # T1[i] = sum_k e[k]*a[k-i]
        T1 = np.fft.irfft(np.conj(E_fft) * A_fft, size)[:n]
        # T2[i] = sum_k e[k]*a[i-k]  
        T2 = np.fft.irfft(A_fft * E_fft, size)[:n]
        
        r = T1 + T2
        
        # Normalize r
        r_mean = r.mean()
        r_std = r.std() + 1e-10
        r_norm = (r - r_mean) / r_std

        # Multiplicative update
        eta_use = eta * np.exp(-0.005 * it)  # Decay eta
        a_new = a * np.exp(-eta_use * r_norm)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            # Try smaller step
            a_small = a * np.exp(-eta_use * 0.1 * r_norm)
            a_small = np.maximum(a_small, 1e-10)
            a_small = a_small / a_small.sum()
            obj_small = compute_J(a_small)
            if obj_small < best_obj - 1e-12:
                best_obj = obj_small
                best_a = a_small.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = best_a.copy()  # Don't update if no improvement
        a = a_new if obj_new < best_obj else a

        if no_improve > 50:
            break

        if verbose and it % 50 == 0:
            print(f"    Water-fill it {it}: obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine/sine patterns
    for freq in [1, 2, 3, 4, 5, 7]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Gaussian bumps
    for center in [0.2, 0.5, 0.8]:
        for width in [0.05, 0.1, 0.2, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.05))

    # Power-law
    for alpha in [0.3, 0.7, 1.0, 2.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Bandpass / structured patterns
    for f1, f2 in [(2, 5), (3, 7), (1, 4)]:
        inits.append(project_simplex(1.0 + 0.3 * np.cos(2*np.pi*f1*t) * np.cos(2*np.pi*f2*t)))

    return inits


def solve_single(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: Coordinate descent (35% of time)
    t1 = time.monotonic() + 0.35 * remaining
    a = coord_descent_active(a, max_sweeps=50000, deadline=t1, seed=42)
    a = try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: Smooth gradient descent (25% of time)
    t2 = time.monotonic() + 0.25 * remaining
    a_sm, obj_sm = smooth_gradient_descent(a, max_iters=5000, deadline=t2,
                                           tau_init=30.0, tau_final=300.0, lr=0.03)
    if obj_sm < best_obj:
        best_a = a_sm.copy()
        best_obj = obj_sm
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: Water filling (15% of time)
    t3 = time.monotonic() + 0.15 * remaining
    a_wf, obj_wf = water_filling(a, max_iters=2000, deadline=t3, eta=0.15)
    if obj_wf < best_obj:
        best_a = a_wf.copy()
        best_obj = obj_wf
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Active-set LP (15% of time)
    t4 = time.monotonic() + 0.15 * remaining
    a_lp, obj_lp = active_set_lp(a, max_iters=100, deadline=t4, top_k=10)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 5: Final polish - CD (10% of time)
    t5 = deadline - 2
    a = coord_descent_active(best_a, max_sweeps=50000, deadline=t5, seed=123)
    a = try_update(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses active-peak coordinate descent, smoothed gradient descent, water-filling,
    active-set LP, and multi-n exploration with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Optimize at parent length with diverse inits ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase1_deadline = t_start + min(400.0, 0.35 * (budget_s - 30))
    inits = generate_diverse_inits(n_parent, rng, parent)
    n_init = min(len(inits), 10)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init)

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 30:
            break
        alloc = min(per_init_time * 1.2, remaining_check - 10)
        if alloc < 20:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt, obj_opt = solve_single(a0, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  Init {idx}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Try different lengths ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase2_deadline = t_start + min(800.0, 0.65 * (budget_s - 30))
    # Try lengths around parent and some "nice" numbers
    candidate_ns = set()
    # Near parent
    for offset in [-64, -32, 32, 64, 128]:
        nn = n_parent + offset
        if 128 <= nn <= 4096:
            candidate_ns.add(nn)
    # Nice numbers
    for nn in [256, 384, 512, 640, 768, 1024, 1280, 1536, 2048]:
        if nn != n_parent:
            candidate_ns.add(nn)

    candidate_ns = sorted(candidate_ns, key=lambda x: abs(x - n_parent))[:6]

    for new_n in candidate_ns:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 60:
            break
        alloc = min(120.0, remaining_check - 20)
        if alloc < 40:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_single(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive optimization on best ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive local optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase3_deadline = deadline
    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # Multiple passes with different methods and seeds
            passes = [
                ("CD", lambda a, dl: coord_descent_active(a, 100000, dl, seed=777)),
                ("SmoothGD", lambda a, dl: smooth_gradient_descent(a, 10000, dl, tau_init=20, tau_final=200, lr=0.02)[0]),
                ("WaterFill", lambda a, dl: water_filling(a, 5000, dl, eta=0.1)[0]),
                ("LP", lambda a, dl: active_set_lp(a, 200, dl, top_k=8)[0]),
                ("CD2", lambda a, dl: coord_descent_active(a, 100000, dl, seed=999)),
            ]

            for pname, pfunc in passes:
                remaining = phase3_deadline - time.monotonic()
                if remaining < 20:
                    break
                alloc = remaining * 0.35
                print(f"  {pname} pass (budget {alloc:.0f}s)...", flush=True)
                try:
                    a_result = pfunc(best_seq.copy(), time.monotonic() + alloc)
                    if a_result is not None:
                        update_best(a_result)
                except Exception as e:
                    print(f"    {pname} failed: {e}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
