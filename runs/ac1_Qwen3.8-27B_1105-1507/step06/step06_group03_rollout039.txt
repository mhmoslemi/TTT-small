"""
Improved sequence optimizer using peak-directed coordinate descent,
FISTA with log-sum-exp smoothing, and active-set LP.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def project_simplex(v):

    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    it = 0

    tau_start = 30.0
    tau_end = 200.0

    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        c = fast_conv(y)
        c_max = c.max()
        c_shifted = c - c_max
        exp_c = np.exp(tau * c_shifted)
        exp_sum = exp_c.sum()
        w = exp_c / exp_sum

        # Gradient: grad[i] = 2*tau * sum_{j=0}^{n-1} a[j] * w[i+j]
        # Compute via FFT cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        a_fft = np.fft.rfft(y, N)
        w_fft = np.fft.rfft(w, N)
        cross = np.fft.irfft(a_fft * np.conj(w_fft), N)
        # cross[k] = sum_j y[j] * w[(j-k) mod N]
        # We want G[i] = sum_j y[j] * w[i+j]
        # G[i] = cross[(N-i) % N]
        G = np.array([cross[(N - i) % N] for i in range(n)])

        grad = 2.0 * tau * G
        grad = grad - np.mean(grad)

        # Step size with backtracking
        lr = 1.0 / (tau * 2.0 * n / (n * n) + 0.1)
        lr = min(lr, 0.5 / tau)

        y_new = project_simplex(y - lr * grad)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_norm * t_norm)) / 2.0
        y_next = y_new + (t_norm - 1.0) / t_new * (y_new - y)
        y_next = project_simplex(y_next)
        t_norm = t_new

        obj = compute_J(y_next)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = y_next.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                y = project_simplex(best_a + rng.normal(0, 0.02, n))
                t_norm = 1.0
                no_improve = 0

        y = y_next
        it += 1

    # Final evaluation of y_new path
    obj_final = compute_J(y)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = y.copy()

    return best_a, best_obj


def full_cd(a, max_sweeps, deadline, seed=0):
    """Standard coordinate descent with ratio candidates (as backup)."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)

            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, 1e-2]
            else:
                candidates = [0.0, old * 0.2, old * 0.4, old * 0.6, old * 0.8,
                              old * 1.2, old * 1.5, old * 2.0]

            best_local = 2.0 * n * conv.max() / (s * s)
            best_x = old

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a[lo - i:hi + 1 - i]
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_x = x

                conv[lo:hi + 1] -= 2.0 * delta * a[lo - i:hi + 1 - i]
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] -= delta * delta

            if best_x != old and abs(best_x - old) > 1e-13:
                delta = best_x - old
                conv[lo:hi + 1] += 2.0 * delta * a[lo - i:hi + 1 - i]
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += delta * delta
                a[i] = best_x
                s = s - old + best_x

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 5, 8]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    for center in [0.1, 0.3, 0.5, 0.7, 0.9]:
        for width in [0.05, 0.1, 0.2, 0.3]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.05))

    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Bilinear / tent patterns
    for peak_pos in [0.2, 0.5, 0.8]:
        tent = 1.0 - np.abs(t - peak_pos) / max(peak_pos, 1 - peak_pos)
        inits.append(project_simplex(np.maximum(tent, 0) + 0.05))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: peak-directed CD, active-set LP, FISTA,
    multi-length exploration, and intensive final polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Peak-directed CD at parent length (300s) ===
    phase1_deadline = t_start + 300.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Peak-directed CD at n={n_parent} (budget {phase1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    n_tries = min(len(inits), 8)
    per_try = (phase1_deadline - time.monotonic()) / max(1, n_tries)

    for idx in range(n_tries):
        remaining = phase1_deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_try, remaining - 5)
        if alloc < 15:
            break

        a0 = inits[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        a_opt = peak_cd(a0, max_sweeps=20000, deadline=time.monotonic() + alloc,
                        K=5, seed=seed + idx)
        update_best(a_opt)
        obj = compute_J(a_opt)
        print(f"  Peak CD init {idx}: J={obj:.8f}", flush=True)

    # === PHASE 2: Active-set LP refinement (300s) ===
    phase2_deadline = t_start + 600.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Active-set LP (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase2_deadline - time.monotonic()
        if remaining > 30:
            # Try different K values
            for K_con in [20, 50, 100]:
                rem = phase2_deadline - time.monotonic()
                if rem < 15:
                    break
                alloc = min(rem - 10, 80.0)
                if alloc < 10:
                    break
                print(f"  Active-set LP with K={K_con} (budget {alloc:.0f}s)...", flush=True)
                a_lp, obj_lp = active_set_lp(best_seq, K_constraints=K_con,
                                             deadline=time.monotonic() + alloc,
                                             max_iters=300)
                update_best(a_lp)
                print(f"  LP K={K_con}: J={obj_lp:.8f}", flush=True)

    # === PHASE 3: FISTA optimization (250s) ===
    phase3_deadline = t_start + 850.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: FISTA (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            a_fista, obj_fista = fista_optimize(best_seq, deadline=time.monotonic() + remaining - 10)
            update_best(a_fista)
            print(f"  FISTA: J={obj_fista:.8f}", flush=True)

    # === PHASE 4: Multi-length exploration (200s) ===
    phase4_deadline = t_start + 1050.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-length exploration (budget {phase4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        candidate_ns = [n_parent - 64, n_parent - 32, n_parent + 32, n_parent + 64,
                        n_parent + 128, n_parent - 128]
        candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048 and n != n_parent]
        candidate_ns = sorted(set(candidate_ns))

        for new_n in candidate_ns[:4]:
            remaining = phase4_deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining - 15, 60.0)
            if alloc < 15:
                break

            print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
            a_ext = resample_seq(best_seq, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            a_opt = peak_cd(a_ext, max_sweeps=10000, deadline=time.monotonic() + alloc,
                            K=5, seed=seed + 1000 + new_n)
            update_best(a_opt)
            obj = compute_J(a_opt)
            print(f"  n={new_n}: J={obj:.8f}", flush=True)

    # === PHASE 5: Intensive final polishing (remaining time) ===
    print(f"\n{'=' * 60}", flush=True)
    remaining_total = deadline - time.monotonic()
    print(f"PHASE 5: Final polishing (budget {remaining_total:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining_total > 30:
        a_curr = best_seq.copy()

        # 5a: Long peak-directed CD
        t5a = time.monotonic() + 0.35 * remaining_total
        print(f"  Peak CD pass (budget {t5a - time.monotonic():.0f}s)...", flush=True)
        a_curr = peak_cd(a_curr, max_sweeps=50000, deadline=t5a, K=5, seed=4242)
        update_best(a_curr)
        a_curr = best_seq.copy()

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # 5b: Active-set LP with large K
            t5b = time.monotonic() + 0.25 * remaining
            print(f"  LP K=150 pass (budget {t5b - time.monotonic():.0f}s)...", flush=True)
            a_lp, obj_lp = active_set_lp(a_curr, K_constraints=150,
                                         deadline=t5b, max_iters=500)
            update_best(a_lp)
            a_curr = best_seq.copy()

        remaining = deadline - time.monotonic()
        if remaining > 20:
            # 5c: Standard CD for fine tuning
            t5c = time.monotonic() + 0.25 * remaining
            print(f"  Standard CD pass (budget {t5c - time.monotonic():.0f}s)...", flush=True)
            a_curr = full_cd(a_curr, max_sweeps=30000, deadline=t5c, seed=8888)
            update_best(a_curr)
            a_curr = best_seq.copy()

        remaining = deadline - time.monotonic()
        if remaining > 20:
            # 5d: One more FISTA with different params
            t5d = time.monotonic() + 0.30 * remaining
            print(f"  Final FISTA (budget {t5d - time.monotonic():.0f}s)...", flush=True)
            a_fista2, obj_fista2 = fista_optimize(a_curr, deadline=t5d, max_iters=10000)
            update_best(a_fista2)

        remaining = deadline - time.monotonic()
        if remaining > 10:
            # 5e: Final peak CD
            t5e = deadline - 5
            print(f"  Final peak CD (budget {t5e - time.monotonic():.0f}s)...", flush=True)
            a_final = peak_cd(best_seq, max_sweeps=100000, deadline=t5e, K=7, seed=77777)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
