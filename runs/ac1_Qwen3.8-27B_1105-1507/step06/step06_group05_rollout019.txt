"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value (scale-invariant)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Project onto probability simplex: non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(n, m - j)
        A[j:j + k_end, j] = a[:k_end]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def soft_desc_nesterov(a, time_budget=300.0, tau_init=0.01, lr_init=0.3):
    """
    Nesterov-accelerated soft-subgradient descent on the simplex.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)  # momentum velocity
    tau = tau_init
    lr = lr_init
    momentum = 0.9
    size = 1
    while size < 3 * n:
        size <<= 1

    stagnant = 0
    for step in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        y = a + momentum * v

        # Compute gradient at y (on the simplex)
        y_norm = normalize(y)
        grad, conv, cmax = soft_subgrad_fft(y_norm, tau)

        # Project gradient onto tangent space of simplex
        grad = grad - np.mean(grad)

        # Backtracking line search
        ok = False
        for backtrack in range(8):
            step_size = lr * (0.5 ** backtrack)
            a_try = normalize(y_norm - step_size * grad)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj + 1e-14:
                ok = True
                break
        if not ok:
            # Don't move, just update momentum
            v = momentum * v
            tau *= 0.999
            stagnant += 1
            if stagnant > 200:
                # Perturb to escape
                rng = np.random.default_rng(step * 31 + 7)
                a = normalize(a * np.exp(rng.normal(0, 0.05, n)))
                v = np.zeros(n)
                stagnant = 0
                tau = max(tau, tau_init * 0.5)
                lr = lr_init * 0.5
            continue

        v = a_try - y_norm
        a = a_try
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Cool temperature slowly
        tau = max(tau * 0.9995, 1e-5)
        # Slowly decay learning rate
        if step % 200 == 0 and step > 0:
            lr = max(lr * 0.95, lr_init * 0.01)

    return best_a


def lp_alternate(a, max_iters=40, time_budget=300.0):
    """LP alternating method with golden-section line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(5, int(remaining * 0.5)))
        if g is None:
            break

        # Golden section search for blend parameter
        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)

        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)
        for _ in range(12):
            if hi - lo < 0.01:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at_t(d)

        t_opt = (lo + hi) / 2.0
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Small perturbation to escape
            rng = np.random.default_rng(it * 104729 + 1)
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def peak_targeting_sa(a, time_budget=200.0):
    """
    Simulated annealing with peak-targeting moves.
    Moves are designed to specifically reduce the convolution peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    T = 0.05  # initial temperature
    T_min = 0.001

    while time.monotonic() - start < time_budget:
        if T < T_min:
            break

        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_idx = int(np.argmax(conv))

        # Generate a peak-targeting move
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Reduce a peak contributor
            # Peak at peak_idx: conv[peak_idx] = sum_j a[j]*a[peak_idx-j]
            # Find top contributor
            contribs = []
            for j in range(n):
                k2 = peak_idx - j
                if 0 <= k2 < n:
                    contribs.append((a[j] * a[k2], j, k2))
            if contribs:
                contribs.sort(reverse=True)
                _, j1, j2 = contribs[0]
                # Move mass from the larger of j1,j2 to a random far position
                if a[j1] >= a[j2]:
                    src = j1
                else:
                    src = j2
                dst = int(rng.integers(0, n))
                if dst == src:
                    dst = (dst + 1) % n
                delta = 0.02 * a[src]
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                a_try = normalize(a_try)
            else:
                continue

        elif move_type == 1:
            # Smooth a local region
            center = peak_idx % n
            radius = int(rng.integers(5, max(10, n // 50)))
            lo = max(0, center - radius)
            hi = min(n, center + radius + 1)
            a_try = a.copy()
            region = a_try[lo:hi].copy()
            # Apply Gaussian smoothing
            kernel = np.exp(-0.5 * (np.arange(len(region)) - len(region) // 2) ** 2 / (radius / 3))
            kernel /= kernel.sum()
            a_try[lo:hi] = np.convolve(region, kernel, mode='same')
            a_try = normalize(a_try)

        elif move_type == 2:
            # Random pairwise swap of mass
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 == i2:
                continue
            a_try = a.copy()
            swap = 0.01 * min(a[i1], a[i2])
            a_try[i1] -= swap
            a_try[i2] += swap
            a_try = normalize(a_try)

        else:
            # Gaussian perturbation (scaled by temperature)
            a_try = a * np.exp(rng.normal(0, T * 2, n))
            a_try = normalize(a_try)

        obj_try = compute_obj(a_try)
        delta = obj_try - best_obj

        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

        # Cool down
        T *= 0.998

    return best_a


def coordinate_polish(a, time_budget=60.0):
    """Fine-grained coordinate descent targeting the peak."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = float(conv[peak_idx])

        # Find top contributors to the peak
        contribs = []
        for j in range(n):
            k2 = peak_idx - j
            if 0 <= k2 < n:
                contribs.append((a[j] * a[k2], j, k2))
        contribs.sort(reverse=True)

        improved = False
        for peak_contrib, j1, j2 in contribs[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2 or a[j1] < 1e-10:
                continue

            # Try moving small amount from j1 to a position far from peak
            for attempt in range(3):
                dst = int(rng.integers(0, n))
                if dst == j1 or dst == j2:
                    # Check if dst is actually far from peak
                    if abs(dst - peak_idx) < n // 3 and abs(peak_idx - dst) < n // 3:
                        continue

                for frac in [0.01, 0.03, 0.05, 0.02]:
                    delta = frac * a[j1]
                    a_try = a.copy()
                    a_try[j1] -= delta
                    a_try[dst] += delta
                    if a_try[j1] < 0:
                        continue
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Random perturbation to escape
            if rng.random() < 0.3:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded: n={parent.size}, obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(budget_s=1500):
    """
    Main search function. Multi-phase strategy:
    Phase 1: Nesterov soft-subgradient descent (fast, handles multi-peak)
    Phase 2: LP alternating (smart direction finding)
    Phase 3: Multi-n exploration
    Phase 4: Peak-targeting simulated annealing
    Phase 5: Coordinate descent polish
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_best = len(best_seq) if best_seq is not None else 700
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Soft-subgradient with Nesterov momentum
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Nesterov soft-subgradient (n={n_best})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        alloc = min(400.0, deadline - time.monotonic() - 500)
        if alloc > 30:
            a_new = soft_desc_nesterov(a, time_budget=alloc, tau_init=0.01, lr_init=0.3)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Phase 1 result: {best_score:.8f} (improved!)")
            else:
                print(f"  Phase 1 result: {obj_new:.8f} (no improvement)")
                a_new = normalize(0.7 * normalize(a) + 0.3 * normalize(a_new))
                best_seq = a_new
        else:
            print("  Skipped (insufficient time)")

    # =========================================================
    # Phase 2: LP alternating
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP alternating (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        alloc = min(350.0, deadline - time.monotonic() - 400)
        if alloc > 30:
            a_new = lp_alternate(a, max_iters=35, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Phase 2 result: {best_score:.8f} (improved!)")
            else:
                print(f"  Phase 2 result: {obj_new:.8f} (no improvement)")

    # =========================================================
    # Phase 3: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-n exploration (base n={len(best_seq)})")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    n_options = []
    for delta in [-200, -100, -50, 0, 50, 100, 200, 300]:
        n_try = n_cur + delta
        if 200 <= n_try <= 1200:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_cur:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_sequence(best_seq, n_try))

        # Quick optimization at this n
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 20:
            a_try = soft_desc_nesterov(a_try, time_budget=alloc * 0.6, tau_init=0.01, lr_init=0.2)
            # Brief LP polish
            alloc2 = min(remaining * 0.08, 30.0)
            if alloc2 > 10 and n_try <= 900:
                a_try = lp_alternate(a_try, max_iters=8, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 4: Peak-targeting simulated annealing
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Simulated annealing (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        alloc = min(200.0, deadline - time.monotonic() - 100)
        if alloc > 20:
            a_new = peak_targeting_sa(a, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Phase 4 result: {best_score:.8f} (improved!)")
            else:
                print(f"  Phase 4 result: {obj_new:.8f} (no improvement)")

    # =========================================================
    # Phase 5: Coordinate descent polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Coordinate polish (n={len(best_seq)})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        alloc = min(100.0, deadline - time.monotonic() - 20)
        if alloc > 10:
            a_new = coordinate_polish(a, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Phase 5 result: {best_score:.8f} (improved!)")
            else:
                print(f"  Phase 5 result: {obj_new:.8f} (no improvement)")

    # =========================================================
    # Phase 6: Final combined polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a = normalize(best_seq.copy())
            # Quick soft-subgradient
            alloc1 = min(remaining * 0.4, 60.0)
            if alloc1 > 10:
                a = soft_desc_nesterov(a, time_budget=alloc1, tau_init=0.005, lr_init=0.1)
            # Quick LP
            alloc2 = min(remaining * 0.4, 60.0)
            if alloc2 > 15 and len(a) <= 900:
                a = lp_alternate(a, max_iters=10, time_budget=alloc2)
            # Coordinate
            alloc3 = min(remaining * 0.2, 30.0)
            if alloc3 > 5:
                a = coordinate_polish(a, time_budget=alloc3)

            obj_final = compute_obj(a)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a.copy()
            print(f"  Final polish: {compute_obj(best_seq):.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
        print(f"  evaluate_sequence confirms: {val:.8f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
