"""
Optimized step sequence search using adaptive log-sum-exp (LSE) gradient descent
with spectral diversification, momentum, and LP-based finishing.
Primary method: fast O(n log n) per iteration gradient steps using FFT-based
LSE smooth approximation of the max-convolution objective.
Secondary method: LP-based polishing for final refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Project to probability simplex (non-negative, sum=1)."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_gradient_direction(a, tau):
    """
    Compute the LSE gradient direction for minimizing max(conv(a,a)).
    Returns (grad_centered, conv, cmax) where grad_centered is the direction
    to move (subtract this from a, then normalize).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    if tau < 1e-12:
        tau = 1e-12

    # Softmax weights (numerically stable)
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -60.0, 0.0)
    weights = np.exp(shifted)
    wsum = float(np.sum(weights))
    if wsum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= wsum

    # Compute gradient via FFT:
    # grad[m] = sum_k w[k] * (a[m+k] + a[m-k])
    # = conv(w, a)[m] + conv(w_rev, a)[m + 2n - 2]
    N = 1
    while N < 3 * n:
        N <<= 1

    A = rfft(a, N)
    W = rfft(weights, N)

    # s2[m] = sum_k w[k]*a[m-k] = conv(w, a)[m]
    prod1 = irfft(W * A, N)
    s2 = prod1[:n]

    # s1[m] = sum_k w[k]*a[m+k] = conv(w_rev, a)[m + 2n - 2]
    W_rev = rfft(weights[::-1], N)
    prod2 = irfft(W_rev * A, N)
    s1 = prod2[2 * n - 2: 3 * n - 2]

    grad_raw = s1 + s2
    grad_centered = grad_raw - np.mean(grad_raw)

    return grad_centered, conv, cmax


def spectral_perturb(a, rng, strength=0.1):
    """Add a random spectral perturbation to escape local minima."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = fft(a, size)
    # Add random perturbation to magnitude
    phase = np.angle(A)
    mag = np.abs(A)
    perturbation = 1.0 + strength * rng.normal(0, 1, size)
    mag_new = mag * perturbation
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, size))[:n]
    return normalize(a_new)


def lse_optimize(a, budget, tau_start=None, tau_end=None, max_iters=10000,
                 momentum_decay=0.85, base_step=0.08, diversify_every=200):
    """
    Main LSE gradient optimizer with adaptive temperature, momentum,
    and periodic spectral diversification.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initial cmax for temperature
    conv0 = compute_conv(a)
    cmax0 = float(np.max(conv0))

    if tau_start is None:
        tau_start = cmax0 / 4.0
    if tau_end is None:
        tau_end = cmax0 / (30.0 * n + 1.0)

    # Precompute FFT size
    N = 1
    while N < 3 * n:
        N <<= 1

    step = base_step
    momentum_buf = np.zeros(n)
    rng = np.random.default_rng(int(time.time() * 1000) % (2**31))

    stagnation_counter = 0
    patience = 300

    for it in range(max_iters):
        if time.monotonic() - start > budget - 1:
            break

        # Adaptive temperature: geometric decay
        progress = it / float(max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress
        tau = max(tau, 1e-10)

        # Compute gradient
        grad, conv, cmax = lse_gradient_direction(a, tau)

        # Update momentum
        momentum_buf = momentum_decay * momentum_buf + (1 - momentum_decay) * grad

        # Update
        a_try = a - step * momentum_buf
        a_try = normalize(a_try)

        # Evaluate
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            if obj_try < best_obj - 1e-8:
                stagnation_counter = 0
            else:
                stagnation_counter += 1
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            # Slightly increase step if improving well
            if obj_try < best_obj and step < 0.3:
                step *= 1.01
        else:
            # Rejected step: reduce step size
            step *= 0.7
            a = best_a.copy()
            stagnation_counter += 1

        # Adaptive step floor
        step = max(step, 0.005)

        # Diversification: periodically perturb to escape local minima
        if it > 0 and it % diversify_every == 0:
            if stagnation_counter > patience // 2:
                strength = 0.03 + 0.05 * (stagnation_counter / patience)
                a = spectral_perturb(best_a, rng, strength=min(strength, 0.15))
                momentum_buf *= 0.5
                stagnation_counter = 0
            elif rng.random() < 0.3:
                # Mild random perturbation
                a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))
                momentum_buf *= 0.5

        # Full restart if stuck too long
        if stagnation_counter > patience:
            strength = 0.1
            a = spectral_perturb(best_a, rng, strength)
            step = base_step
            momentum_buf = np.zeros(n)
            stagnation_counter = 0

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def lp_polish(a, budget=30.0):
    """LP-based polishing: solve max sum(g) s.t. conv(a,g) <= max(conv(a,a)), g>=0."""
    n = len(a)
    if n > 1200:
        return a  # Too large for LP
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    M = float(np.max(compute_conv(a)))
    A_toep = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    it = 0
    while time.monotonic() - start < budget - 2:
        remaining = budget - (time.monotonic() - start)
        if remaining < 5:
            break

        try:
            res = linprog(c, A_ub=A_toep, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(3, int(remaining * 0.7)),
                                   'dual_feasibility_tolerance': 1e-9,
                                   'primal_feasibility_tolerance': 1e-9})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g

            # Try different blend points
            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                blend = normalize((1 - t) * a + t * g)
                obj_blend = compute_obj(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0:
                a = normalize((1 - best_t) * a + best_t * g)
                if best_blend_obj < best_obj:
                    best_obj = best_blend_obj
                    best_a = a.copy()
                # Update M and rebuild Toeplitz
                M = float(np.max(compute_conv(a)))
                A_toep = build_toeplitz(a)
                b_ub = np.full(2 * n - 1, M)
            else:
                # Perturb and try again
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))
                M = float(np.max(compute_conv(a)))
                A_toep = build_toeplitz(a)
                b_ub = np.full(2 * n - 1, M)
        except Exception:
            break

        it += 1

    return best_a


def spectral_equalization_step(a, alpha=0.7):
    """One step of spectral equalization (flatten the power spectrum)."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = fft(a, size)
    P = np.abs(A) ** 2
    p_max = float(np.max(P))
    if p_max < 1e-30:
        return a
    P_norm = P / p_max
    P_new = (P_norm ** alpha) * p_max
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, size))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def peak_targeting_gradient(a, top_k=10):
    """
    Subgradient targeting only the top-k peaks of the convolution.
    This is a special case of the LSE gradient with very small tau.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    # Get top-k peak indices
    if top_k >= len(conv):
        peak_idx = np.arange(len(conv))
    else:
        peak_idx = np.argsort(conv)[-top_k:]

    # Equal weights on top peaks
    weights = np.zeros(len(conv))
    weights[peak_idx] = 1.0 / len(peak_idx)

    # Compute gradient (same as lse_gradient_direction but with discrete weights)
    N = 1
    while N < 3 * n:
        N <<= 1
    A = rfft(a, N)
    W = rfft(weights, N)
    prod1 = irfft(W * A, N)
    s2 = prod1[:n]
    W_rev = rfft(weights[::-1], N)
    prod2 = irfft(W_rev * A, N)
    s1 = prod2[2 * n - 2: 3 * n - 2]

    grad = s1 + s2
    grad_centered = grad - np.mean(grad)
    return grad_centered, conv, cmax


def hybrid_optimize(a, budget):
    """
    Hybrid optimizer: alternates between LSE gradient and peak-targeting subgradient,
    with periodic spectral equalization.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv0 = compute_conv(a)
    cmax0 = float(np.max(conv0))
    tau = cmax0 / 5.0
    step = 0.06
    momentum_buf = np.zeros(n)
    rng = np.random.default_rng(int(time.time()) % (2**31))

    it = 0
    stagnation = 0

    while time.monotonic() - start < budget - 1:
        # Choose method based on iteration
        if it < 50:
            # Warm-up: LSE with broad temperature
            grad, conv, cmax = lse_gradient_direction(a, tau)
        elif it % 5 == 0:
            # Every 5th: peak targeting (sharper)
            grad, conv, cmax = peak_targeting_gradient(a, top_k=min(20, n // 10 + 1))
        else:
            # LSE with current temperature
            grad, conv, cmax = lse_gradient_direction(a, tau)

        # Momentum update
        momentum_buf = 0.8 * momentum_buf + 0.2 * grad
        a_try = normalize(a - step * momentum_buf)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            improvement = best_obj - obj_try
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            if improvement < 1e-7:
                stagnation += 1
            else:
                stagnation = 0
            # Decay temperature
            tau *= 0.995
            step = min(step * 1.005, 0.25)
        else:
            step *= 0.65
            a = best_a.copy()
            stagnation += 1

        step = max(step, 0.003)
        tau = max(tau, cmax0 / (100.0 * n))

        # Periodic spectral diversification
        if it > 0 and it % 150 == 0:
            if stagnation > 100:
                a = spectral_perturb(best_a, rng, 0.08)
                momentum_buf *= 0.3
            elif it % 300 == 0:
                a = spectral_equalization_step(best_a, alpha=0.85)
                a = normalize(a)
                momentum_buf *= 0.5
            stagnation = 0

        # Full restart
        if stagnation > 400:
            a = spectral_perturb(best_a, rng, 0.12)
            step = 0.06
            tau = cmax0 / 5.0
            momentum_buf = np.zeros(n)
            stagnation = 0

        it += 1

    return best_a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(budget_s=1500):
    """
    Main search function. Multi-phase strategy:
    Phase 1: Hybrid LSE + peak-targeting gradient (primary, fastest)
    Phase 2: Multi-length exploration
    Phase 3: LP-based polishing at best length
    Phase 4: Final intensive gradient refinement
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    if best_seq is None:
        # Fallback: start from a smoothed random sequence
        n_init = 700
        rng = np.random.default_rng(42)
        best_seq = normalize(rng.dirichlet(np.ones(n_init) * 2.0))
        best_score = compute_obj(best_seq)
        print(f"Initialized random sequence: n={len(best_seq)}, score={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, score={best_score:.8f}")

    # =========================================================
    # Phase 1: Primary hybrid optimization at parent length
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Hybrid LSE optimization (n={n_parent})")
    print(f"{'='*60}")

    a = normalize(best_seq.copy())

    # Run in sub-batches to track progress
    for batch in range(6):
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.14, 120.0)
        if alloc < 10:
            break

        a_new = hybrid_optimize(a, budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Batch {batch}: {best_score:.8f} **")
        else:
            # Blend for diversity
            a = normalize(0.6 * a + 0.4 * a_new)
            print(f"  Batch {batch}: {obj_new:.8f} (best: {best_score:.8f})")

    print(f"  Phase 1 done: best={best_score:.8f}, time={time.monotonic()-start_time:.1f}s")

    # =========================================================
    # Phase 2: Multi-length exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Multi-length exploration")
    print(f"{'='*60}")

    n_best = len(best_seq)
    # Explore nearby lengths
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best - 25,
        n_best, n_best + 25, n_best + 50, n_best + 100, n_best + 150
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]

    for n_try in n_options:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_best:
            a_try = normalize(best_seq.copy())
        else:
            a_try = normalize(resample_seq(best_seq, n_try))

        alloc = min(remaining * 0.10, 50.0)
        if alloc < 5:
            continue

        # Quick optimization at this length
        a_opt = hybrid_optimize(a_try, budget=alloc)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_opt:.8f}")

    print(f"  Phase 2 done: best={best_score:.8f}, n={n_best}, time={time.monotonic()-start_time:.1f}s")

    # =========================================================
    # Phase 3: LP polishing at best length
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: LP polishing (n={n_best})")
    print(f"{'='*60}")

    if n_best <= 1000 and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.25, 80.0)
        if alloc > 10:
            a_lp = lp_polish(normalize(best_seq.copy()), budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP polish: {best_score:.8f} **")
            else:
                print(f"  LP polish: {obj_lp:.8f} (no improvement)")

    # =========================================================
    # Phase 4: Final intensive gradient refinement
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final refinement")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            # Use LSE with very small temperature for fine-tuning
            a_final = normalize(best_seq.copy())
            conv_f = compute_conv(a_final)
            cmax_f = float(np.max(conv_f))

            # Fine-tuned LSE
            a_refined = lse_optimize(
                a_final, budget=min(remaining * 0.6, 60.0),
                tau_start=cmax_f / (20.0 * len(a_final) + 1),
                tau_end=cmax_f / (100.0 * len(a_final) + 1),
                base_step=0.03, diversify_every=300
            )
            obj_ref = compute_obj(a_refined)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_refined.copy()
                print(f"  Final LSE: {best_score:.8f} **")
            else:
                print(f"  Final LSE: {obj_ref:.8f}")

            # One more LP if time allows
            if n_best <= 900 and time.monotonic() < deadline - 5:
                remaining2 = deadline - time.monotonic()
                if remaining2 > 15:
                    a_lp2 = lp_polish(normalize(best_seq.copy()), budget=min(remaining2 - 3, 30.0))
                    obj_lp2 = compute_obj(a_lp2)
                    if obj_lp2 < best_score:
                        best_score = obj_lp2
                        best_seq = a_lp2.copy()
                        print(f"  Final LP: {best_score:.8f} **")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Ensure sum constraint
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate with the actual evaluation function
    try:
        val = evaluate_sequence(result)
        if val < np.inf and val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
