"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA with log-sum-exp smoothing, LP alternating (HiGHS), 
multiplicative updates, multi-n exploration, and random restarts.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} using bisection."""
    n = len(v)
    if n == 0:
        return v
    # Find tau such that sum(max(v - tau, 0)) = target_sum
    lo = float(v.min()) - target_sum
    hi = float(v.max())
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.maximum(v - mid, 0.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    proj = np.maximum(v - tau, 0.0)
    s = float(proj.sum())
    if s > 1e-15:
        proj *= target_sum / s
    return proj

def fista_smoothed(a, time_budget=100.0, verbose=False):
    """
    FISTA with log-sum-exp smoothing of the max-convolution objective.
    Minimizes (1/t)*log(sum exp(t*conv(a,a)[k])) on the probability simplex.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Parameters
    t_start = 10.0
    t_end = 500.0
    base_lr = 0.02
    
    x = a.copy()
    z = a.copy()
    t_k = 1.0
    lr = base_lr
    stall_count = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        t_curr = t_start * np.exp(3.0 * progress)  # Exponential schedule
        t_curr = min(t_curr, t_end)
        
        # Covariates (Nesterov)
        beta = (t_k - 1.0) / (t_k + 0.5 + np.sqrt(1.0 + 2.0 * t_k + t_k * t_k))
        y = x + beta * (x - z)
        y = np.maximum(y, 0.0)
        y_sum = float(np.sum(y))
        if y_sum > 1e-10:
            y = y / y_sum
        else:
            y = x.copy()
        
        # Compute convolution of y
        conv_y = compute_conv(y)
        
        # Softmax weights
        conv_max = float(np.max(conv_y))
        exp_vals = np.exp(t_curr * (conv_y - conv_max))
        w = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: grad[i] = 2 * sum_k w[k] * y[k-i] * 1_{0<=k-i<n}
        # = 2 * cross_corr(w, y)[i]
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        fw = rfft(w, size)
        fy = rfft(y, size)
        cross = irfft(np.conj(fw) * fy, size)
        grad = 2.0 * cross[:n]
        
        # Gradient step with lr scaled by temperature (higher t -> sharper -> need smaller lr)
        lr_eff = lr / (1.0 + 0.1 * progress)
        x_new = y - lr_eff * grad
        x_new = project_simplex(x_new, 1.0)
        
        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                lr *= 0.7
                stall_count = 0
                if lr < 0.002:
                    break
        
        z = x
        x = x_new
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      FISTA it{it}: obj={best_obj:.8f}, t={t_curr:.1f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a

def multiplicative_update(a, time_budget=50.0, verbose=False):
    """
    Multiplicative weight updates for reducing peak convolution.
    a[i] <- a[i] * (normalizer / weighted_avg_contribution[i])
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak = float(np.max(conv))
        
        # Softmax weights (sharp)
        t_curr = 50.0
        conv_norm = conv - peak
        exp_vals = np.exp(t_curr * conv_norm / peak)
        w = exp_vals / float(np.sum(exp_vals))
        
        # Contribution of each a[i]: sum_k w[k] * a[k-i] * 1_{0<=k-i<n}
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        fw = rfft(w, size)
        fa = rfft(a, size)
        cross = irfft(np.conj(fw) * fa, size)
        contrib = np.maximum(cross[:n], 1e-15)
        
        # Multiplicative update
        mean_contrib = float(np.mean(contrib))
        ratio = mean_contrib / contrib
        ratio = np.clip(ratio, 0.5, 2.0)
        a_new = a * np.power(ratio, lr)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                lr *= 0.9
        
        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"      MultUpd it{it}: obj={best_obj:.8f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a

def solve_lp_higths(a, M_target=None):
    """
    Solve LP using HiGHS:
    If M_target is None: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    If M_target is given: max sum(g) s.t. conv(a,g)[k] <= M_target, g>=0
    
    Returns g (normalized to sum 1) or None.
    """
    n = len(a)
    a = a / float(np.sum(a))
    
    # Build A matrix: A[k, j] = a[j] if 0 <= k-j < n... 
    # Actually conv(a,g)[k] = sum_j a[j]*g[k-j] for valid indices
    # = sum_j A[k,j] * g[j]
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # Wait: conv(a,g)[k] = sum_j a[j]*g[k-j]
    # Let i = k-j, so j = k-i:
    # = sum_i a[k-i]*g[i] for valid i, k-i
    # So A[k,i] = a[k-i] if 0 <= k-i < n and 0 <= i < n
    
    # Actually, let's use the standard definition:
    # conv(a,g)[k] = sum_{j} a[j] * g[k-j] where 0<=j<n and 0<=k-j<n
    # = sum_{j} a[j] * g[k-j]
    # For fixed k, the valid j are: max(0, k-(n-1)) <= j <= min(n-1, k)
    # So conv(a,g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j]*g[k-j]
    # Let i = k-j: i ranges from k-min(n-1,k) to k-max(0,k-n+1)
    # = max(0, k-(n-1)) to min(n-1, k)
    # So A[k, i] = a[k-i] for i in valid range
    
    # Simpler: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # But that's not quite right either. Let me just build it directly.
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] includes term a[k-j]*g[j] = a[k-j]*g[j]
        # So A[k, j] = a[k-j] for k = j to j+n-1 (i.e., k-j = 0 to n-1)
        A[j:j+n, j] = a
    
    if M_target is None:
        # min M s.t. A@g <= M, sum(g) = 1, g >= 0
        # Variables: [g (n), M (1)]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
    else:
        # max sum(g) s.t. A@g <= M_target, g >= 0
        c = -np.ones(n)  # minimize -sum(g)
        A_ub = A
        b_ub = np.full(2 * n - 1, M_target)
        A_eq = None
        b_eq = None
        bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0.0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g
    except Exception:
        pass
    
    return None

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_new = np.linspace(0, 1, new_n)
        a_new = a[0] + (a[1] - a[0]) * x_new
        return np.maximum(a_new, 0)
    from scipy.interpolate import PchipInterpolator
    x_old = np.linspace(0, 1, n)
    try:
        interp = PchipInterpolator(x_old, a, nn_bound=0.0)
        x_new = np.linspace(0, 1, new_n)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(np.linspace(0, 1, new_n), x_old, a)
    return np.maximum(a_new, 0.0)

def random_init(n, rng):
    """Generate diverse random non-negative sequences."""
    mode = int(rng.integers(0, 6))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = float(np.sum(a))
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        a = a / float(np.sum(a))
    elif mode == 3:
        # Gaussian-like
        center = n / 2.0
        width = n / (4.0 + rng.uniform(0, 4))
        idx = np.arange(n)
        a = np.exp(-((idx - center) ** 2) / (2 * width ** 2))
        a = a / float(np.sum(a))
    elif mode == 4:
        # Two peaks
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * 0.08
        idx = np.arange(n)
        a = np.exp(-((idx - c1) ** 2) / (2 * w ** 2)) + np.exp(-((idx - c2) ** 2) / (2 * w ** 2))
        a = a / float(np.sum(a))
    else:
        # Smooth random via sine mixture
        freqs = rng.uniform(1, 5, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        idx = np.arange(n) / n
        a = 1.0
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * idx + p)
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        a = a / s if s > 1e-10 else np.ones(n) / n
    return a

def chase_peak(a, time_budget=30.0, verbose=False):

            j = k_star - i
            if 0 <= j < n:

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA with log-sum-exp smoothing, LP alternating (HiGHS), multiplicative updates,
peak-cutting coordinate moves, multi-n exploration, and diverse random restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} using bisection."""
    n = len(v)
    if n == 0:
        return v
    lo = float(v.min()) - target_sum
    hi = float(v.max())
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.maximum(v - mid, 0.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    proj = np.maximum(v - tau, 0.0)
    s = float(proj.sum())
    if s > 1e-15:
        proj *= target_sum / s
    return proj


def fista_smoothed(a, time_budget=100.0, verbose=False):
    """
    FISTA with log-sum-exp smoothing of the max-convolution objective.
    Minimizes (1/t)*log(sum exp(t*conv(a,a)[k])) on the probability simplex.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_start = 15.0
    t_end = 600.0
    base_lr = 0.015

    x = a.copy()
    z = a.copy()
    t_k = 1.0
    lr = base_lr
    stall_count = 0

    it = 0
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        t_curr = t_start * np.exp(3.5 * progress)
        t_curr = min(t_curr, t_end)

        # Nesterov momentum
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_k - 1.0) / t_next
        y = x + beta * (x - z)
        y = np.maximum(y, 0.0)
        y_sum = float(np.sum(y))
        if y_sum > 1e-10:
            y = y / y_sum
        else:
            y = x.copy()

        # Compute convolution
        conv_y = compute_conv(y)

        # Softmax weights (numerically stable)
        conv_max = float(np.max(conv_y))
        scaled = t_curr * (conv_y - conv_max)
        # Clip to avoid overflow
        scaled = np.clip(scaled, -500, 500)
        exp_vals = np.exp(scaled)
        w = exp_vals / float(np.sum(exp_vals))

        # Gradient via cross-correlation
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        fw = rfft(w, size)
        fy = rfft(y, size)
        cross = irfft(np.conj(fw) * fy, size)
        grad = 2.0 * cross[:n]

        # Gradient step
        lr_eff = lr / (1.0 + 0.5 * progress)
        x_new = y - lr_eff * grad
        x_new = project_simplex(x_new, 1.0)

        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 30:
                lr *= 0.75
                stall_count = 0
                if lr < 0.001:
                    # Restart momentum
                    x = best_a.copy()
                    z = best_a.copy()
                    t_k = 1.0
                    lr = base_lr * 0.5

        z = x
        x = x_new
        t_k = t_next

        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      FISTA it{it}: obj={best_obj:.8f}, t={t_curr:.1f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_a

        peak = float(np.max(conv))

        # Softmax weights (adaptive sharpness)

        exp_vals = np.exp(t_curr * conv_norm)
        w = exp_vals / float(np.sum(exp_vals))

        contrib = np.maximum(cross[:n], 1e-15)

        # Multiplicative update

        ratio = mean_contrib / contrib
        ratio = np.clip(ratio, 0.3, 3.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_higths(a, M_target=None):
    """
    Solve LP using HiGHS:
    If M_target is None: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    If M_target is given: max sum(g) s.t. conv(a,g)[k] <= M_target, g>=0
    Returns g (normalized to sum 1) or None.
    """
    n = len(a)
    a = a / float(np.sum(a))

    A = build_conv_matrix(a)

    if M_target is None:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
    else:
        c = -np.ones(n)
        A_ub = A
        b_ub = np.full(2 * n - 1, M_target)
        A_eq = None
        b_eq = None
        bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0.0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g
    except Exception:
        pass

    return None


def lp_alternating(a, time_budget=100.0, verbose=False):

    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()

        return a

            break

        g_lp = solve_lp_higths(a)

        if g_lp is None:

            M_cur = float(np.max(conv_a))
            g_lp = solve_lp_higths(a, M_target=M_cur * 1.001)

        if g_lp is None:
            break

        # Line search with more candidate points
        best_t_obj = best_obj
        best_t_val = 0.0

            a = best_a

def chase_peak(a, time_budget=30.0, verbose=False):

    if s < 1e-10:

    start = time.monotonic()
    step = 0.03
    stall = 0

            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad_c = grad - float(np.mean(grad))
        a_new = a - step * grad_c
        a_new = project_simplex(a_new, 1.0)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new

            step = min(step * 1.01, 0.08)
            stall = 0
        else:
            stall += 1
            step *= 0.8
            if step < 0.002 or stall > 100:
                break

    return best_a


def peak_cut_direct(a, time_budget=30.0, verbose=False):
    """
    Direct peak cutting: identify peak position and move mass to reduce it.
    """
    n = len(a)

    if s < 1e-10:
        return a

        conv = compute_conv(a)

        peak_val = float(conv[k_star])

        # Find positions that contribute most to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Find a high-contribution position and a low-contribution position to swap
        high_idx = int(np.argmax(contribs))
        # Find a position that doesn't contribute to the peak
        non_contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if not (0 <= j < n) or i == j:
                non_contrib[i] = a[i]
        # Actually find positions that contribute least to peak
        low_idx_candidates = [i for i in range(n) if contribs[i] < 0.1 * peak_val / max(n, 1)]

            if it % 10 == 0 and time.monotonic() - start < time_budget:

            break

    return best_a

    try:
        from scipy.interpolate import PchipInterpolator
        x_old = np.linspace(0, 1, n)
        interp = PchipInterpolator(x_old, a, nn_bound=0.0)
        x_new = np.linspace(0, 1, new_n)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, n), a)
    return np.maximum(a_new, 0.0)


def random_init(n, rng):
    """Generate diverse random non-negative sequences."""
    mode = int(rng.integers(0, 7))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = float(np.sum(a))
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.3
        a = a / float(np.sum(a))
    elif mode == 3:
        center = n / 2.0
        width = n / (4.0 + rng.uniform(0, 4))
        idx = np.arange(n)
        a = np.exp(-((idx - center) ** 2) / (2 * width ** 2))
        a = a / float(np.sum(a))
    elif mode == 4:
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * 0.08
        idx = np.arange(n)
        a = np.exp(-((idx - c1) ** 2) / (2 * w ** 2)) + np.exp(-((idx - c2) ** 2) / (2 * w ** 2))
        a = a / float(np.sum(a))
    elif mode == 5:
        freqs = rng.uniform(1, 5, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        idx = np.arange(n) / n
        a = 1.0
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * idx + p)
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        a = a / s if s > 1e-10 else np.ones(n) / n
    else:
        # Wavelet-like: sum of decaying bumps at random positions
        a = np.zeros(n)
        for _ in range(3):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.2)
            idx = np.arange(n)
            a += np.exp(-((idx - center) ** 2) / (2 * width ** 2))
        a = a / float(np.sum(a))
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Hybrid optimization: FISTA + LP alternating + multiplicative + peak cutting +
    multi-n exploration + diverse random restarts + final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: FISTA with log-sum-exp ===
    print(f"\n=== Phase 1: FISTA (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(remaining * 0.18, 220.0)
        a_fista = fista_smoothed(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_fista)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_fista.copy()
            print(f"  FISTA: obj={best_score:.8f} **")
        else:
            print(f"  FISTA: obj={obj_f:.8f}")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(remaining * 0.22, 280.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: Chase Peak + Multiplicative ===
    print(f"\n=== Phase 3: Chase Peak & Multiplicative ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_cp = chase_peak(best_seq, time_budget=min(remaining * 0.08, 40.0))
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  Chase: obj={best_score:.8f} **")
        else:
            print(f"  Chase: obj={obj_cp:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_mu = multiplicative_update(best_seq, time_budget=min(remaining * 0.08, 40.0))
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            print(f"  MultUpd: obj={best_score:.8f} **")
        else:
            print(f"  MultUpd: obj={obj_mu:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_options = sorted(set([
        max(200, n_parent - 150),
        max(200, n_parent - 50),
        n_parent,
        n_parent + 50,
        n_parent + 150,
        n_parent + 300,
        400, 600, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1600]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.12, 55.0)

        # FISTA then LP
        a_opt = fista_smoothed(a_try, time_budget=alloc * 0.5)
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            a_opt2 = lp_alternating(a_opt, time_budget=min(alloc * 0.4, rem2 * 0.4))
            obj2 = compute_obj(a_opt2)
            if obj2 < compute_obj(a_opt):
                a_opt = a_opt2

        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Random Restarts ===
    print(f"\n=== Phase 5: Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    n_restarts = min(12, max(3, int(remaining // 55)))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        init_type = r % 6
        if init_type == 0:
            a_r = random_init(n_best, rng)
        elif init_type == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.06, n_best))
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / float(np.sum(a_r))
        elif init_type == 2:
            n_r = int(n_best + rng.integers(-200, 200))
            n_r = max(200, min(1600, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.04, len(a_r)))
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / float(np.sum(a_r))
        elif init_type == 3:
            # Anti-symmetric perturbation
            a_r = best_seq.copy()
            noise = rng.normal(0, 0.03, n_best)
            noise = noise - noise[::-1]
            a_r = a_r + noise
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / float(np.sum(a_r))
        elif init_type == 4:
            # Raised cosine with random frequency
            n_r = n_best
            x = np.linspace(0, 2 * np.pi * rng.uniform(0.5, 2.0), n_r)
            a_r = 1.0 + 0.4 * np.cos(x)
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / float(np.sum(a_r))
        else:
            # Blend with random
            a_r = 0.7 * best_seq + 0.3 * rng.uniform(0, 1, n_best)
            a_r = a_r / float(np.sum(a_r))

        alloc = min(remaining * 0.13, 70.0)
        a_opt = fista_smoothed(a_r, time_budget=alloc * 0.5)
        rem3 = deadline - time.monotonic()
        if rem3 > 15:
            a_opt = lp_alternating(a_opt, time_budget=min(alloc * 0.4, rem3 * 0.5))
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_r:.8f}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        alloc = min(remaining * 0.3, 100.0)
        a_f = fista_smoothed(a_final, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            a_final = a_f
            best_seq = a_final.copy()
            print(f"  Final FISTA: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 80.0)
            a_l = lp_alternating(a_final, time_budget=alloc)
            obj_l = compute_obj(a_l)
            if obj_l < best_score:
                best_score = obj_l
                a_final = a_l
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_c = chase_peak(a_final, time_budget=min(remaining * 0.3, 25.0))
            obj_c = compute_obj(a_c)
            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                print(f"  Final Chase: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
