class Adam:
    def __init__(self, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = None
        self.v = None
        self.t = 0
    
    def step(self, params, grad):
        self.t += 1
        if self.m is None:
            self.m = np.zeros_like(params)
            self.v = np.zeros_like(params)
        
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad**2

        v_hat = self.v / (1 - self.beta2**self.t)
        
        return params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

"""
Smoothed gradient descent with adaptive log-sum-exp parameter for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses Adam optimizer, multi-n exploration,
and subgradient polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    b = compute_conv(a_n)
    max_b = float(np.max(b))
    return 2.0 * n * max_b


def smooth_gradient(a, t):
    """
    Compute gradient of LSE(conv(a,a), t) w.r.t. a.
    Returns (gradient, lse_value).
    """
    n = len(a)
    b = compute_conv(a)
    
    # Stable softmax
    tb = t * b
    tb_max = float(np.max(tb))
    exp_tb = np.exp(tb - tb_max)
    p = exp_tb / np.sum(exp_tb)
    lse_val = (tb_max + np.log(np.sum(exp_tb))) / t
    
    # Gradient: (2/t) * cross_corr(a, p)
    # cross_corr[i] = sum_j a[j] * p[i+j] for j=0..n-1
    # = h[i+n-1] where h = conv(a[::-1], p)
    a_r = a[::-1]
    size = 1
    while size < n + (2 * n - 1) - 1:
        size <<= 1
    Ar = rfft(a_r, size)
    Pr = rfft(p, size)
    h = irfft(Ar * Pr, size)
    g = h[n - 1: 2 * n - 1]
    
    return (2.0 / t) * g, lse_val


def subgradient_exact(a):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a.
    Returns (subgrad, max_val, argmax_idx).
    """
    n = len(a)
    b = compute_conv(a)
    k_star = int(np.argmax(b))
    
    # Subgradient of b[k*] w.r.t. a[i] = 2 * a[k* - i] (with bounds)
    sg = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            sg[i] = 2.0 * a[j]
    
    return sg, float(b[k_star]), k_star


def project_simplex(v):
    """Exact Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])]) if v[0] > 0 else np.array([1.0])
    
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv[-1] - 1)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)


def project_simplex_fast(v):
    """Fast approximate projection: clip and normalize."""
    v = np.maximum(v, 0)
    s = np.sum(v)
    if s > 1e-15:
        return v / s
    return np.ones(len(v)) / len(v)


def smoothed_gradient_descent(a, time_budget=100.0, t_start=5.0, t_end=500.0,
                                lr=0.01, verbose=False):
    """
    Adam-based gradient descent on the smoothed objective LSE(conv(a,a), t).
    t is annealed from t_start to t_end.
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = project_simplex_fast(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    total_iters = 0
    stagnant = 0
    
    while time.monotonic() - start < time_budget:
        t_step += 1
        # Annealing schedule for t (smoothing parameter)
        progress = min(1.0, t_step / (time_budget * 50))  # crude progress
        # Actually, let's use a time-based schedule
        elapsed = time.monotonic() - start
        anim_progress = min(1.0, elapsed / (time_budget * 0.8))
        # Exponential annealing
        t = t_start * (t_end / t_start) ** anim_progress
        
        # Compute gradient
        grad, lse_val = smooth_gradient(a, t)
        
        # Center the gradient (to maintain sum(a) ≈ 1)
        grad_centered = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex_fast(a_new)
        
        # Check objective
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            stagnant = 0
            if verbose and total_iters % 1000 == 0:
                elapsed = time.monotonic() - start
                print(f"          SGD it{total_iters}: obj={best_obj:.8f}, t={t:.1f}, time={elapsed:.1f}s")
        else:
            stagnant += 1
        
        a = a_new
        total_iters += 1
        
        if stagnant > 5000:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Much more iterations than LP-based approach due to FFT-based gradient computation.

import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] ** 2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    b = compute_conv(a_n)
    max_b = float(np.max(b))
    return 2.0 * n * max_b


def smooth_gradient(a, t):

    Compute gradient of LSE(conv(a,a), t) w.r.t. a.
    Returns (gradient, lse_value).

    b = compute_conv(a)

    tb = t * b
    tb_max = float(np.max(tb))
    exp_tb = np.exp(tb - tb_max)

    lse_val = (tb_max + np.log(np.sum(exp_tb))) / t

    a_r = a[::-1]
    size = 1

    Ar = rfft(a_r, size)
    Pr = rfft(p, size)
    h = irfft(Ar * Pr, size)
    g = h[n - 1: 2 * n - 1]

    return (2.0 / t) * g, lse_val

def project_simplex(v):
    """Exact Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0])
    
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv[-1] - 1)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)


def project_simplex_fast(v):
    """Fast approximate projection: clip and normalize."""
    v = np.maximum(v, 0)
    s = np.sum(v)
    if s > 1e-15:
        return v / s
    return np.ones(len(v)) / len(v)

                stagnant += 1

            if verbose and (t_step - last_print) >= 2000:
                print(f"          SGD it{t_step}: obj={best_obj:.8f}, t={t:.1f}, time={elapsed:.1f}s")
                last_print = t_step

        a = a_new
        total_iters += 1

        # Escape stagnation
        if stagnant > 30:
            rng = np.random.default_rng(total_iters * 7919)
            scale = 0.03 + 0.02 * min((stagnant - 30) // 30, 5)
            a = a * np.exp(rng.normal(0, scale, n))
            a = project_simplex_fast(a)
            stagnant = 0

    return best_a


def subgradient_polish(a, time_budget=20.0, verbose=False):
    """
    Exact subgradient method with projected steps.
    Uses multiple peak subgradients (all bins within threshold of max).
    """
    n = len(a)
    a = project_simplex_fast(np.asarray(a, dtype=np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_step = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        t_step += 1
        eta = 0.05 / np.sqrt(1 + t_step)

        b = compute_conv(a)
        max_b = float(np.max(b))

        # Use weighted subgradient from all "near-max" bins
        threshold = max_b * 0.95
        active = b >= threshold
        if np.sum(active) == 0:
            active = np.zeros(len(b), dtype=bool)
            active[int(np.argmax(b))] = True

        # Weight by proximity to max
        weights = np.maximum(0, b - max_b * 0.9)
        weights = weights / np.sum(weights) if np.sum(weights) > 0 else None

        if weights is not None and np.sum(active) > 1:
            # Multiple subgradient (weighted average)
            sg = np.zeros(n)
            for k in np.where(active)[0]:
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        sg[i] += weights[k] * 2.0 * a[j]
        else:
            k_star = int(np.argmax(b))
            sg = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    sg[i] = 2.0 * a[j]

        a_new = a - eta * sg
        a_new = project_simplex_fast(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new

        if stagnant > 100:
            rng = np.random.default_rng(t_step * 31)
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = project_simplex_fast(a)
            stagnant = 0

    return best_a
