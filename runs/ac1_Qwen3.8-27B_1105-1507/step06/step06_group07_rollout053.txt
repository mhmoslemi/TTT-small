import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A

def lp_solve(a_norm, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0"""
    n = len(a_norm)
    if n > 1000:
        return None
    A = build_toeplitz(a_norm)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except:
        pass
    return None

def lp_refine_iterative(a, time_budget=60, max_iters=20):
    """Iterative LP refinement with line search."""
    start = time.monotonic()
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_cur = a / s
    best_a = a_cur.copy()
    best_obj = compute_obj(a_cur)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a_cur)
        M = np.max(conv)
        
        g = lp_solve(a_cur, M, time_limit=min(20, (time_budget - (time.monotonic()-start)) * 0.3))
        if g is None:
            break
        
        # Line search
        best_t = None
        best_val = best_obj

            a_try = np.maximum(a_try, 0)

            if s_try < 1e-10:

            a_try = a_try / s_try

def adam_log_opt(a, max_iters=20000, time_budget=50.0, beta_start=10.0,
                 beta_end=500.0, lr_start=0.03, lr_decay=0.99998):
    """Adam in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_c = x - np.max(x)
        a_cur = np.exp(x_c)
        a_cur = a_cur / np.sum(a_cur)
        
        conv = compute_conv(a_cur)
        cmax = np.max(conv)
        shifted = beta * (conv - cmax)
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        e_sum = np.sum(e)
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta
        
        # Gradient
        size = 1
        while size < 2 * (2*n-1):
            size <<= 1
        W = rfft(w, size)
        A = rfft(a_cur, size)
        cc = irfft(W * np.conj(A), size)
        grad_ls = 2.0 * cc[:n]
        
        sum_a = np.sum(a_cur)
        grad_a = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
        grad_x = grad_a * a_cur
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 100 == 0:
            x -= np.mean(x)
        
        if it % 200 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

def power_smooth_opt(a, max_iters=20000, time_budget=50.0, p_start=20.0, p_end=500.0,

        a_cur = np.exp(x_c)
        a_cur = a_cur / np.sum(a_cur)

        conv_pos = np.maximum(conv, 1e-30)

        # grad_ls_wrt_a[i] = sum_k w_k * d(conv_k)/d(a_i) where w_k = p * conv_k^{p-1} / (sum conv^p)^{1/p}

def coordinate_refine(a, time_budget=30.0, seed=0):
    """Coordinate descent focusing on peak contributions."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        # Find top peaks
        num_peaks = min(10, len(conv))
        peak_indices = np.argsort(conv)[::-1][:num_peaks]
        
        improved = False
        for p in peak_indices:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            for i in range(i_min, i_max + 1):
                if time.monotonic() - start > time_budget:
                    break
                j = p - i
                if j < 0 or j >= n or j == i:
                    continue
                # Try to reduce contribution to peak p
                # Move mass from the larger of a[i], a[j] to somewhere else
                if best_a[i] > best_a[j]:
                    src, dst = i, j
                else:
                    src, dst = j, i
                for delta_frac in [0.01, 0.02, 0.005]:
                    delta = delta_frac * best_a[src]
                    a_try = best_a.copy()
                    a_try[src] -= delta
                    a_try[dst] += delta
                    if a_try[src] < 0:
                        continue
                    s_try = np.sum(a_try)
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
            if improved:
                break
        
        if not improved:
            # Random perturbation to escape
            idx = rng.integers(0, n)
            delta = 0.001 * np.mean(best_a)
            a_try = best_a.copy()
            a_try[idx] += delta
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
            elif it > 100:
                break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: iterative LP refinement (best-response dynamics).
Secondary: Adam in log-space with logsumexp and power smoothing.
Additional: coordinate descent, multi-n exploration, random restarts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def lp_solve(a_norm, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Return normalized g or None."""
    n = len(a_norm)
    if n > 1000:
        return None
    A = build_toeplitz(a_norm)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine_iterative(a, time_budget=100.0, max_iters=40):
    """Iterative LP refinement with adaptive line search."""
    start = time.monotonic()
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_cur = a / s
    best_a = a_cur.copy()
    best_obj = compute_obj(a_cur)
    print(f"    LP start: obj={best_obj:.8f}", flush=True)

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        conv = compute_conv(a_cur)
        M = float(np.max(conv))

        remaining = time_budget - elapsed
        lp_time = min(remaining * 0.25, 15.0)
        if lp_time < 2:
            break

        g = lp_solve(a_cur, M, time_limit=lp_time)
        if g is None:
            break

        # Adaptive line search
        best_t = None
        best_val = best_obj
        for t in np.linspace(0.05, 1.0, 12):
            a_try = (1 - t) * a_cur + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_val:

def adam_log_opt(a, max_iters=20000, time_budget=50.0, beta_start=10.0,
                 beta_end=500.0, lr_start=0.03, lr_decay=0.99998):
    """Adam in log-space with logsumexp smoothing and temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_start

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute FFT size for gradients
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_c = x - np.max(x)
        a_cur = np.exp(x_c)
        a_cur = a_cur / np.sum(a_cur)

        cmax = float(np.max(conv))

        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        w = e / e_sum

        W = rfft(w, size)
        A = rfft(a_cur, size)
        cc = irfft(W * np.conj(A), size)
        grad_ls = 2.0 * cc[:n]

        sum_a = np.sum(a_cur)
        ls = cmax + np.log(e_sum) / beta
        grad_a = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
        grad_x = grad_a * a_cur

        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

def power_smooth_opt(a, max_iters=20000, time_budget=50.0, p_start=30.0,
                     p_end=800.0, lr_start=0.03, lr_decay=0.99998):
    """Adam in log-space using power smoothing (sum conv^p)^(1/p)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_start

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        p = p_start * (p_end / p_start) ** progress

        x_c = x - np.max(x)
        a_cur = np.exp(x_c)
        a_cur = a_cur / np.sum(a_cur)

        conv = compute_conv(a_cur)
        conv_pos = np.maximum(conv, 1e-30)
        conv_p = conv_pos ** p
        conv_p_sum = np.sum(conv_p)
        ls = conv_p_sum ** (1.0 / p)

        # Gradient: d/d(a_i) of ls = (1/p) * conv_p_sum^(1/p - 1) * p * sum_k(conv_k^(p-1) * d(conv_k)/d(a_i))
        # = conv_p_sum^(-1/p) * sum_k(conv_k^(p-1) * d(conv_k)/d(a_i))
        weights = conv_pos ** (p - 1)
        weights = weights / (conv_p_sum ** (1.0 / p))

        W = rfft(weights, size)
        A = rfft(a_cur, size)
        cc = irfft(W * np.conj(A), size)
        grad_ls = 2.0 * cc[:n]

        sum_a = np.sum(a_cur)
        grad_a = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
        grad_x = grad_a * a_cur

        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 150 == 0:

        if it % 300 == 0 or it == max_iters - 1:
            x_c2 = x - np.max(x)
            a_check = np.exp(x_c2)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()

    return best_a


def peak_shaving(a, time_budget=30.0):
    """Directly try to reduce the highest peaks by adjusting contributing elements."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()

            break

        # Find the maximum peak
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        i_min = max(0, max_idx - (n - 1))
        i_max = min(max_idx, n - 1)

        # Try reducing the largest contributor and moving mass to a random location
        contributors = list(range(i_min, i_max + 1))
        if not contributors:
            break

        # Sort by contribution
        contrib_values = [(i, best_a[i] * best_a[max_idx - i]) for i in contributors]
        contrib_values.sort(key=lambda x: -x[1])

        # Try the top few contributors
        tried = False
        for i, _ in contrib_values[:5]:
            if time.monotonic() - start > time_budget:
                break
            j = max_idx - i
            if j < 0 or j >= n or j == i:
                continue

            for delta_frac in [0.005, 0.01, 0.02, 0.03]:
                delta = delta_frac * best_a[i]
                dest = rng.integers(0, n)
                # Avoid moving to positions that would create new peaks

                    continue

                a_try = best_a.copy()
                a_try[i] -= delta
                a_try[dest] += delta
                if a_try[i] < 0:
                    continue
                s_try = np.sum(a_try)
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    tried = True
                    break
            if tried:
                break

        if not tried and it > 50:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}", flush=True)
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimizer: LP iterative refinement (primary), Adam log-space
    (logsumexp + power smoothing), peak shaving, multi-n exploration,
    coordinate descent, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s", flush=True)

    # === Phase 1: Heavy LP refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: LP iterative refinement (n={n_parent})", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        lp_budget = min(remaining * 0.35, 300)
        a_lp = lp_refine_iterative(best_seq, time_budget=lp_budget, max_iters=50)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP Phase 1: {best_score:.8f} ***", flush=True)
        else:
            print(f"  LP Phase 1: {obj_lp:.8f} (no imp)", flush=True)

    # === Phase 2: Adam logsumexp ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Adam logsumexp (n={len(best_seq) if best_seq is not None else '?'})", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            rounds = [
                (10, 100, 0.04, 0.99997),
                (50, 400, 0.02, 0.99999),
            ]
            for r_idx, (b_s, b_e, lr0, lr_dec) in enumerate(rounds):
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                alloc = min(remaining * 0.15, 80)
                a_new = adam_log_opt(best_seq, max_iters=20000, time_budget=alloc,
                                     beta_start=b_s, beta_end=b_e,
                                     lr_start=lr0, lr_decay=lr_dec)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  Adam {r_idx}: {best_score:.8f} ***", flush=True)
                else:
                    print(f"  Adam {r_idx}: {obj_new:.8f}", flush=True)

    # === Phase 3: More LP refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: LP refinement round 2 (n={len(best_seq) if best_seq is not None else '?'})", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and len(best_seq) <= 900:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            lp_budget = min(remaining * 0.25, 150)
            a_lp2 = lp_refine_iterative(best_seq, time_budget=lp_budget, max_iters=30)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP Phase 3: {best_score:.8f} ***", flush=True)
            else:
                print(f"  LP Phase 3: {obj_lp2:.8f} (no imp)", flush=True)

    # === Phase 4: Power smoothing Adam ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Power smoothing Adam (n={len(best_seq) if best_seq is not None else '?'})", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.15, 80)
            a_ps = power_smooth_opt(best_seq, max_iters=15000, time_budget=alloc,
                                    p_start=30, p_end=800,
                                    lr_start=0.03, lr_decay=0.99998)
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                print(f"  Power smooth: {best_score:.8f} ***", flush=True)
            else:
                print(f"  Power smooth: {obj_ps:.8f}", flush=True)

    # === Phase 5: Peak shaving ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Peak shaving (n={len(best_seq) if best_seq is not None else '?'})", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.1, 40)
            a_ps = peak_shaving(best_seq, time_budget=alloc)
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                print(f"  Peak shaving: {best_score:.8f} ***", flush=True)
            else:
                print(f"  Peak shaving: {obj_ps:.8f}", flush=True)

    # === Phase 6: Multi-n exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 6: Multi-n exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(150, n_best - 100),
            max(150, n_best - 50),
            min(1200, n_best + 50),
            min(1200, n_best + 100),
            min(1200, n_best + 200),
        ]))
        n_options = [n for n in n_options if n != n_best]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

            # LP + Adam combo
            alloc_total = min(remaining * 0.15, 50)
            a_opt = lp_refine_iterative(a_try, time_budget=alloc_total * 0.4, max_iters=5)
            a_opt = adam_log_opt(a_opt, max_iters=10000, time_budget=alloc_total * 0.5,
                                 beta_start=20, beta_end=300,
                                 lr_start=0.02, lr_decay=0.99999)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE", flush=True)
            else:
                print(f"  n={n_try}: {obj:.8f}", flush=True)

    # === Phase 7: Final intensive LP + Adam ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 7: Final polish (n={len(best_seq) if best_seq is not None else '?'})", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20 and len(best_seq) <= 900:
            alloc = min(remaining * 0.3, 60)
            a_fl = lp_refine_iterative(best_seq, time_budget=alloc, max_iters=15)
            obj_fl = compute_obj(a_fl)
            if obj_fl < best_score:
                best_score = obj_fl
                best_seq = a_fl.copy()
                print(f"  Final LP: {best_score:.8f} ***", flush=True)

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_fh = adam_log_opt(best_seq, max_iters=15000,
                                time_budget=min(remaining * 0.4, 30),
                                beta_start=200, beta_end=1500,
                                lr_start=0.005, lr_decay=0.99999)
            obj_fh = compute_obj(a_fh)
            if obj_fh < best_score:
                best_score = obj_fh
                best_seq = a_fh.copy()
                print(f"  Final Adam: {best_score:.8f} ***", flush=True)

    # === Phase 8: Random restarts ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 8: Random restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed)

        for restart in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if restart % 2 == 0:
                scale = 0.02 + 0.01 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            else:
                # Blend with structured sequence
                alpha = rng.uniform(0.1, 0.25)
                # Try a "flat-top" inspired sequence
                kk = np.arange(n_best)
                structured = 1.0 / (1.0 + 0.5 * np.cos(2 * np.pi * kk / n_best))
                structured = structured / np.sum(structured)
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * structured

            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / np.sum(a_pert)

            alloc = min(remaining * 0.2, 25)
            a_opt = adam_log_opt(a_pert, max_iters=10000, time_budget=alloc,
                                 beta_start=15, beta_end=400,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} ***", flush=True)
            else:
                print(f"  Restart {restart}: {obj:.8f}", flush=True)

    # === Final: one more LP pass if time allows ===
    if best_seq is not None and time.monotonic() < deadline - 10 and len(best_seq) <= 800:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_last = lp_refine_iterative(best_seq, time_budget=min(remaining * 0.6, 30), max_iters=5)
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Last LP: {best_score:.8f} ***", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return result
