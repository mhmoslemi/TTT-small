"""
GS-subgradient hybrid for minimizing normalized autocorrelation peak.
Key innovations over previous LP-based approach:
1. Gerchberg-Saxton frequency flattening (drives non-DC spectrum toward flat)
2. FFT-accelerated subgradient with Nesterov momentum and adaptive peak weighting
3. Diverse structural initialization including chirp and block patterns
4. Simulated annealing for escaping local optima
5. Multi-phase refinement: coarse GS -> subgradient -> fine GS -> polish

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    conv = irfft(fa * fa, N)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def gs_flatten(a, n_iters=30, gamma=0.3, keep_dc=True):
    """
    Gerchberg-Saxton style frequency flattening.
    Flattens the non-DC power spectrum while preserving phase.
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = a.astype(np.float64).copy()
    for _ in range(n_iters):
        A = rfft(a, N)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-15)
        if keep_dc:
            mean_pow = np.mean(mag[1:] ** 2)
            new_mag = mag.copy()
            if len(mag) > 1:
                new_mag[1:] = mag[1:] ** (1.0 - 2.0 * gamma) * mean_pow ** gamma
        else:
            mean_pow = np.mean(mag ** 2)
            new_mag = mag ** (1.0 - 2.0 * gamma) * mean_pow ** gamma
        A_new = new_mag * np.exp(1j * np.angle(A))
        a = irfft(A_new, N)[:n]
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
    return a


def gs_flatten_multi(a, stages):
    """
    Multi-stage GS with varying gamma.
    stages: list of (n_iters, gamma) tuples.
    """
    for n_iters, gamma in stages:
        a = gs_flatten(a, n_iters=n_iters, gamma=gamma)
    return a


def subgradient_step(a, step_size, weights, N_fft):
    """
    One subgradient step given precomputed peak weights.
    g[i] = 2 * sum_k weights[k] * a[k-i] (where 0<=k-i<n)
    Computed via FFT cross-correlation.
    """
    W = rfft(weights, N_fft)
    A = rfft(a, N_fft)
    corr = irfft(A * np.conj(W), N_fft)
    grad = 2.0 * corr[:len(a)]
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def get_peak_weights(conv, top_k=3, temperature=0.01):
    """
    Get softmax weights on peaks. Higher temperature = more spread.
    """
    cmax = np.max(conv)
    # Softmax over top peaks with temperature
    shifted = (conv - cmax) / temperature
    shifted = np.clip(shifted, -300.0, 0.0)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= s
    return w


def subgradient_descent(a, n_iters=1000, initial_step=0.03, cooling=0.9995,
                        top_k=3, temperature=0.005, mu=0.9):
    """
    Subgradient descent with Nesterov momentum and adaptive peak weighting.
    """
    n = len(a)
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    best_a = a.copy()
    best_obj = compute_obj(a)
    step = initial_step
    v = np.zeros(n)
    temp = temperature

    for it in range(n_iters):
        conv = compute_conv(a)
        weights = get_peak_weights(conv, top_k=top_k, temperature=temp)
        grad = subgradient_step(a, 1.0, weights, N_fft)
        # Center gradient (remove mean to stay on affine hull)
        grad = grad - np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        # Nesterov momentum
        v = mu * v - step * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s > 1e-15:
            a_new = a_new / s
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        step *= cooling
        temp *= 0.999
    return best_a, best_obj


def annealed_search(a, n_iters=500, initial_temp=0.05, cooling=0.99,
                    step_size=0.03, n_fft=None):
    """
    Simulated annealing: occasionally accept worse solutions.
    """
    n = len(a)
    if n_fft is None:
        n_fft = 1
        while n_fft < 2 * n - 1:
            n_fft <<= 1
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    temp = initial_temp
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        conv = compute_conv(a)
        weights = get_peak_weights(conv, top_k=3, temperature=0.005)
        grad = subgradient_step(a, 1.0, weights, n_fft)
        grad = grad - np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s > 1e-15:
            a_new = a_new / s

        obj_new = compute_obj(a_new)
        delta = obj_new - current_obj

        # Accept if better or with probability exp(-delta/temp)
        if delta < 0 or (temp > 1e-6 and rng.random() < np.exp(-delta / temp)):
            a = a_new
            current_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
        temp *= cooling
    return best_a, best_obj


def generate_chirp(n, beta=0.5, rng=None):
    """Generate a non-negative chirp sequence."""
    i = np.arange(n, dtype=np.float64)
    t = (i - n / 2.0) / (n / 2.0)  # normalized to [-1, 1]
    # Frequency chirp
    freq = t ** 2 * beta
    phase = 2.0 * np.pi * freq
    # Make non-negative: use absolute value of complex exponential
    a = np.abs(np.cos(phase))
    # Add a base to ensure positivity
    a = 0.1 + a
    a = a / np.sum(a)
    return a


def generate_block_pattern(n, n_blocks=3, rng=None):
    """Generate a pattern with separated blocks of energy."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.zeros(n)
    block_size = n // (n_blocks * 2)
    positions = sorted(rng.choice(n, size=n_blocks, replace=False))
    for pos in positions:
        start = max(0, pos)
        end = min(n, pos + block_size)
        a[start:end] = 1.0
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def generate_parametric(n, sigma, freq, amp, phase, center_frac=0.5):
    """Bell shape with sinusoidal modulation."""
    i = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    bell = np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = bell * np.maximum(mod, 0.0)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def propose_candidate(seed=42, time_budget=1500):
    """
    Main search function combining GS frequency flattening, FFT-accelerated
    subgradient descent with Nesterov momentum, simulated annealing, and
    diverse structural initialization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(time_budget) - 30.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # Phase 1: Explore different lengths
    print("\n=== Phase 1: Length exploration ===")
    n_values = [n_parent]
    for dn in [-100, -50, 0, 50, 100, 150, -150, 200]:
        nv = n_parent + dn
        if 400 <= nv <= 1200:
            n_values.append(nv)
    n_values = sorted(set(n_values))

    # Quick screen each length
    length_results = []
    for n in n_values:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        rng = np.random.default_rng(seed + n)
        inits = generate_diverse_inits(n, rng)

        # Quick GS + subgradient on each init
        best_n = None
        best_n_obj = np.inf
        alloc_per = min(remaining / len(inits), 20.0)

        for ii, init in enumerate(inits):
            if time.monotonic() > deadline - 250:
                break
            t_alloc = min(alloc_per, 15.0)
            if t_alloc < 5:
                break

            # Quick GS
            a_gs = gs_flatten(init, n_iters=15, gamma=0.3)
            obj_gs = compute_obj(a_gs)

            # Quick subgradient
            a_sg, obj_sg = subgradient_descent(
                a_gs, n_iters=200, initial_step=0.03,
                cooling=0.9995, top_k=3, temperature=0.005, mu=0.85
            )

            obj_best = min(obj_gs, obj_sg)
            a_best = a_gs if obj_gs < obj_sg else a_sg

            if obj_best < best_n_obj:
                best_n_obj = obj_best
                best_n = a_best.copy()

        if best_n is not None:
            length_results.append((best_n_obj, best_n.copy()))
            if best_n_obj < best_score:
                best_score = best_n_obj
                best_seq = best_n.copy()
                print(f"  n={n}: {best_n_obj:.8f} ** NEW BEST **")
            else:
                print(f"  n={n}: {best_n_obj:.8f}")

    # Sort by objective
    length_results.sort(key=lambda x: x[0])
    print(f"\nTop length results: {[f'{c[0]:.6f} (n={len(c[1])})' for c in length_results[:5]]}")

    # Phase 2: Intensive refinement of top candidates
    print(f"\n=== Phase 2: Intensive refinement (current best: {best_score:.8f}) ===")

    top_k = min(4, len(length_results))
    for ci in range(top_k):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = length_results[ci][1].copy()
        obj_cand = length_results[ci][0]
        n_cand = len(a_cand)
        alloc = min(remaining * 0.35, 150.0)

        print(f"  Refining candidate {ci} (n={n_cand}, obj={obj_cand:.8f}) with {alloc:.0f}s")

        a_ref, obj_ref = refine_sequence(a_cand, time_budget=alloc * 0.6, seed=seed + ci)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    After refine: {obj_ref:.8f} ** UPDATE **")

        remaining = deadline - time.monotonic()
        if remaining > 30 and obj_ref < best_score + 0.01:
            a_agg, obj_agg = refine_aggressive(a_ref, time_budget=min(remaining * 0.3, 60.0), seed=seed + 100 + ci)
            if obj_agg < best_score:
                best_score = obj_agg
                best_seq = a_agg.copy()
                print(f"    After aggressive: {obj_agg:.8f} ** UPDATE **")

    # Phase 3: Cross-length transfer
    print(f"\n=== Phase 3: Cross-length transfer (best: {best_score:.8f}, n={len(best_seq)}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Resample to nearby lengths and refine
        test_lengths = [n_best - 20, n_best + 20, n_best - 50, n_best + 50,
                        n_best - 100, n_best + 100]
        test_lengths = [x for x in test_lengths if 400 <= x <= 1200 and x != n_best]
        test_lengths = sorted(set(test_lengths))

        for n_test in test_lengths:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_test)
            a_resampled = np.interp(x_new, x_old, best_seq)
            a_resampled = np.maximum(a_resampled, 0)
            a_resampled = a_resampled / np.sum(a_resampled)

            alloc = min(remaining * 0.2, 30.0)
            a_ref, obj_ref = refine_sequence(a_resampled, time_budget=alloc, seed=seed + 200 + n_test)

            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  n={n_test}: ** UPDATE to {best_score:.8f} **")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_final, obj_final = refine_aggressive(best_seq, time_budget=min(remaining * 0.5, 90.0), seed=seed + 500)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Aggressive polish: {best_score:.8f} ** UPDATE **")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                # One more round of GS with very fine gamma
                for gamma in [0.1, 0.15, 0.2, 0.25]:
                    if time.monotonic() > deadline - 5:
                        break
                    a_try = gs_flatten(best_seq, n_iters=20, gamma=gamma)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        print(f"  GS gamma={gamma}: {best_score:.8f} ** UPDATE **")

        # Final subgradient with very small steps
        remaining = deadline - time.monotonic()
        if remaining > 10 and best_seq is not None:
            n_final = len(best_seq)
            N_fft = 1
            while N_fft < 2 * n_final - 1:
                N_fft <<= 1
            a_final, obj_final = subgradient_descent(
                best_seq, n_iters=int(remaining * 20), initial_step=0.005,
                cooling=0.9999, top_k=5, temperature=0.001, mu=0.95
            )
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final SG: {best_score:.8f} ** UPDATE **")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
