"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:
1. Adam-optimizer-based subgradient descent with adaptive per-coordinate learning rates
2. Multi-scale (coarse-to-fine) optimization
3. Better spectral preconditioning with iterative amplitude modification
4. Parabolic interpolation line search
5. LP-guided updates with tighter convergence
6. Targeted pairwise coordinate descent for final polish

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, .)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=30):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def spectral_shaping(a, max_iters=100, time_budget=60.0):
    """
    Iterative spectral shaping to flatten the power spectrum.
    Uses amplitude modification in frequency domain with better convergence.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_mean = float(np.mean(P))
        if p_mean < 1e-30:
            break
        p_max = float(np.max(P))
        p_min = float(np.min(P))

        # Iterative amplitude compression with adaptive strength
        # Start strong, ease off
        progress = it / float(max_iters)
        alpha = 0.3 + 0.4 * (1.0 - progress)  # 0.7 -> 0.3

        # Compress dynamic range
        P_norm = P / p_mean
        P_new = P_norm ** alpha
        P_new = P_new * p_mean  # Keep same total power

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]

        # Project
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:

        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend to maintain progress
            blend = 0.3 + 0.4 * progress
            a = (1 - blend) * a + blend * a_new
            a = normalize(a)

    return best_a


def adam_subgradient(a, max_steps=300, time_budget=80.0, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """
    Adam-optimizer subgradient descent.
    Uses soft-max weighting over convolution peaks for smooth subgradient.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam state
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0

    # FFT size for gradient computation
    size = 1
    while size < 3 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        t += 1

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive temperature: start broad, narrow down
        tau = cmax / (n * (0.1 + 0.5 * step / max_steps))
        tau = max(tau, cmax / (10 * n))

        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute subgradient via FFT
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Center the gradient (remove mean to preserve sum)
        grad_centered = grad - np.mean(grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Adaptive learning rate that decreases over time
        lr_t = lr / (1.0 + 0.01 * step)
        a_try = a - lr_t * m_hat / (np.sqrt(v_hat) + epsilon)

        # Project
        a_try = np.maximum(a_try, 0.0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            break
        a_try = a_try / s

        obj_new = compute_obj(a_try)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_try.copy()
            a = a_try
        else:
            # If no improvement, try a smaller step
            a_try = a - 0.5 * lr_t * m_hat / (np.sqrt(v_hat) + epsilon)
            a_try = np.maximum(a_try, 0.0)
            s = float(np.sum(a_try))
            if s > 1e-10:
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try

    return best_a


def lp_alternate_improved(a, max_iters=40, time_budget=90.0):
    """
    Improved LP alternating with parabolic line search.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(5, int(remaining * 0.5)))
        if g is None:
            break

        # Parabolic interpolation line search
        # Evaluate at 3 points
        t1 = 0.3
        t2 = 0.6
        t3 = 0.9
        f1 = compute_obj(normalize((1 - t1) * a + t1 * g))
        f2 = compute_obj(normalize((1 - t2) * a + t2 * g))
        f3 = compute_obj(normalize((1 - t3) * a + t3 * g))

        # Fit parabola
        if f1 == f2 or f2 == f3:
            t_opt = 0.5 if f2 <= f1 else t3
        else:
            denom = (f1 - 2*f2 + f3)
            if abs(denom) > 1e-15:
                t_opt = 0.5 + 0.5 * (f1 - f3) / (2.0 * denom)
            else:
                t_opt = t2
            t_opt = np.clip(t_opt, 0.05, 0.95)

        # Refine with a few more evaluations around t_opt
        for _ in range(3):
            delta = 0.1 * (t_opt - 0.5 + 0.5)
            t_lo = max(0.01, t_opt - delta)
            t_hi = min(0.99, t_opt + delta)
            f_lo = compute_obj(normalize((1 - t_lo) * a + t_lo * g))
            f_hi = compute_obj(normalize((1 - t_hi) * a + t_hi * g))
            if f_lo < f2:
                t_opt = t_lo
            elif f_hi < f2:
                t_opt = t_hi
            else:
                break

        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb to escape
            rng = np.random.default_rng((it * 104729 + 42) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def multi_scale_optimize(a, time_budget=120.0):
    """
    Coarse-to-fine optimization: downsample, optimize, upsample, refine.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Downsample factors
    factors = [4, 2]
    current = a

    for factor in factors:
        if time.monotonic() > time_budget - 30:
            break
        n_coarse = n // factor
        if n_coarse < 30:
            continue

        # Downsample by averaging blocks
        n_blocks = n // factor
        a_coarse = np.zeros(n_blocks)
        for i in range(n_blocks):
            a_coarse[i] = np.mean(a[i*factor:(i+1)*factor])
        a_coarse = normalize(a_coarse)

        # Optimize at coarse level
        remaining = time_budget - (time.monotonic() - start)
        alloc = min(remaining * 0.4, 30.0)
        a_coarse = adam_subgradient(a_coarse, max_steps=100, time_budget=alloc)

        # Upsample
        a_up = np.zeros(n)
        for i in range(n_blocks):
            a_up[i*factor:(i+1)*factor] = a_coarse[i]
        a_up = normalize(a_up)

        # Refine at full resolution
        remaining = time_budget - (time.monotonic() - start)
        alloc = min(remaining * 0.3, 30.0)
        # Blend with original
        a_blend = 0.5 * normalize(a) + 0.5 * a_up
        a_blend = adam_subgradient(a_blend, max_steps=100, time_budget=alloc)

        obj = compute_obj(a_blend)
        if obj < best_obj:
            best_obj = obj
            best_a = a_blend.copy()
            current = a_blend

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """
    Targeted coordinate descent: move mass from high-contributing positions
    to reduce the peak convolution value.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        if not contrib:
            break

        improved = False
        for peak_val, j1, j2 in contrib[:8]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try several target positions
            for _ in range(5):
                j_target = int(rng.integers(0, n))
                if j_target in (j1, j2):
                    continue

                for frac in [0.02, 0.05, 0.1, 0.2]:
                    a_try = a.copy()
                    delta = frac * a_try[j1]
                    if a_try[j1] - delta < 0:
                        continue
                    a_try[j1] -= delta
                    a_try[j_target] += delta
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)

                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved:
            # Random perturbation
            if rng.random() < 0.3:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                # Block perturbation
                block = int(rng.integers(n//10, n//3))
                start_idx = int(rng.integers(0, n - block))
                a_try = a.copy()
                a_try[start_idx:start_idx+block] *= 0.8
                rest = np.delete(np.arange(n), range(start_idx, start_idx+block))
                if len(rest) > 0:
                    a_try[rest] += 0.8 * a[start_idx:start_idx+block].sum() / len(rest)
                a_try = normalize(a_try)
                if compute_obj(a_try) < compute_obj(a):
                    a = a_try

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
