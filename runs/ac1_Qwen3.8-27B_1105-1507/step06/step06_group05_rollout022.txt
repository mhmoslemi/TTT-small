"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative smooth gradient descent, LP-based peak reduction,
length exploration, and diverse restarts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_grad_step(a, tau, eta, momentum_buf, beta=0.7):
    """One step of multiplicative smooth gradient descent."""
    n = len(a)
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    
    shifted = np.clip((conv - c_max) / tau, -500.0, 500.0)
    p = np.exp(shifted)
    p_sum = float(np.sum(p))
    if p_sum < 1e-300:
        idx = int(np.argmax(conv))
        p = np.zeros(len(conv))
        p[idx] = 1.0
    else:
        p /= p_sum
    
    # Gradient via cross-correlation
    g = np.correlate(p, a, 'valid')
    g = g - np.mean(g)
    
    # Momentum
    momentum_buf[:] = beta * momentum_buf + (1.0 - beta) * g
    
    # Multiplicative update
    a_new = a * np.exp(-eta * momentum_buf)
    a_new = normalize(a_new)
    return a_new


def multiplicative_gradient_descent(a, time_budget=60.0, tau0=0.01, eta0=0.15):
    """Full multiplicative smooth gradient descent with adaptive parameters."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = tau0
    eta = eta0
    momentum_buf = np.zeros(n)
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        a_new = smooth_grad_step(a, tau, eta, momentum_buf)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 20:
                eta *= 0.9
                if eta < 1e-4:
                    # Perturb to escape
                    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                    a = normalize(best_a * np.exp(rng.normal(0, 0.05, n)))
                    momentum_buf[:] = 0
                    stall_count = 0
                    eta = eta0 * 0.5
            else:
                # Accept anyway with small probability (simulated annealing)
                if np.random.random() < 0.05 and obj_new < best_obj * 1.01:
                    a = a_new
        
        tau *= 0.998
        if tau < 0.0005:
            tau = 0.0005
    
    return best_a


def build_toeplitz_fast(a):
    """Build the convolution matrix efficiently."""
    n = len(a)
    m = 2 * n - 1
    # Use scipy or manual construction
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A


def solve_lp_fast(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_fast(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10,
                               'mip_rel_gap': 1e-6})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_cycle(a, max_iters=15, time_budget=60.0):
    """LP-based optimization cycle."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_fast(a, M, time_limit=max(3, int(remaining * 0.4)))
        if g is None:
            break
        
        # Golden section for blend parameter t
        def obj_at_t(t):
            return compute_obj(normalize((1.0 - t) * a + t * g))
        
        lo, hi = 0.0, 1.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        fc1 = obj_at_t(c1)
        fc2 = obj_at_t(c2)
        for _ in range(10):
            if hi - lo < 0.02:
                break
            if fc1 < fc2:
                hi = c2; c2 = c1; fc2 = fc1
                c1 = hi - gr * (hi - lo)
                fc1 = obj_at_t(c1)
