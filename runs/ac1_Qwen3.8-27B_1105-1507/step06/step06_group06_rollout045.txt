    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def multi_peak_gradient(a, tau=0.1, top_k=None):
    """
    Compute the gradient of the softmax-smoothed max of conv(a,a).
    Uses FFT cross-correlation for efficiency.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    
    # Compute weights
    if top_k is not None and top_k < len(conv):
        # Use top-K peaks with uniform weight
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        w = np.zeros(len(conv))
        w[sorted_idx] = 1.0 / top_k
    else:
        # Softmax weights
        shifted = (conv - c_max) / tau
        shifted = np.clip(shifted, -50.0, 0.0)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum
    
    # Compute gradient via FFT cross-correlation
    # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    
    fw = fft(w, N)
    fa = fft(a, N)
    corr = ifft(fw * np.conj(fa)).real[:n]
    
    grad = 2.0 * corr
    return grad, w, conv


def softmax_step(a, b, step_size, tau=0.1, top_k=None):
    """
    One step of gradient descent in softmax parameterization.
    a = softmax(b) / sum(softmax(b))
    Returns updated b and a.
    """
    n = len(a)
    grad, w, conv = multi_peak_gradient(a, tau=tau, top_k=top_k)
    
    # Gradient in b-space: grad_b[j] = a[j] * (grad[j] - sum_i grad[i]*a[i])
    weighted_mean = np.dot(grad, a)
    grad_b = a * (grad - weighted_mean)
    
    # Step
    b_new = b - step_size * grad_b
    
    # Numerical stability: clip b
    b_new = np.clip(b_new, -50.0, 50.0)
    
    # Compute new a
    b_shifted = b_new - np.max(b_new)
    exp_b = np.exp(b_shifted)
    a_new = exp_b / np.sum(exp_b)
    
    return a_new, b_new, grad_b, conv


def multi_peak_gradient_descent(a_init, max_iters=5000, time_budget=60.0,
                                  step_init=0.5, tau_init=0.05, top_k_init=5):
    """
    Main gradient descent loop with adaptive parameters and momentum.
    """
    n = len(a_init)
    a = project_nonneg_sum1(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Initialize b from a
    b = np.log(np.maximum(a, 1e-12))
    
    step = step_init
    tau = tau_init
    top_k = top_k_init
    momentum = 0.7
    v_b = np.zeros(n)
    
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        a_new, b_new, grad_b, conv = softmax_step(a, b, step, tau=tau, top_k=top_k)
        
        # Check objective
        obj = 2.0 * n * float(np.max(conv))  # sum=1, so this is the objective
        
        # Actually compute properly
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            b = b_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Don't update a/b, but still update momentum
            # Actually let's accept even if slightly worse for momentum
            if obj < best_obj * 1.001:
                a = a_new.copy()
                b = b_new.copy()
        
        # Adaptive step size
        if no_improve_count > 20:
            step *= 0.85
            tau *= 1.1  # broaden the peak focus
            top_k = min(n, top_k + 2)
            no_improve_count = 0
        
        # Periodically reset momentum and add perturbation
        if it % 200 == 199:
            v_b *= 0.5
            # Small random perturbation
            pert = np.random.normal(0, 0.01, n)
            a = a * np.exp(pert)
            a = project_nonneg_sum1(a)
            b = np.log(np.maximum(a, 1e-12))
            step = min(step * 1.2, step_init)
        
        # Gradually increase tau (focus on more peaks as we converge)
        progress = it / max_iters
        tau = tau_init + 0.15 * progress
        top_k = int(top_k_init + 10 * progress)
        
        # Decrease step over time
        step = step_init * (0.9995 ** it)
        step = max(step, 0.01)
    
    return best_a, best_obj


def spectral_shaping(a, max_iters=100, time_budget=20.0, alpha=0.3):
    """
    Shape the spectrum to reduce the self-convolution peak.
    Modify |FFT(a)| to reduce FFT(a)^2 peaks in time domain.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = fft(a, N)
        
        # The self-convolution in frequency is fa^2
        # We want ifft(fa^2) to have low peak
        # Strategy: modify fa to make fa^2 more "delta-like" at frequency 0
        
        # Approach: compress the magnitude spectrum
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # The DC component should be dominant
        dc = mag[0]
        if dc < 1e-15:
            break
        
        # Normalize other components relative to DC
        mag_norm = mag.copy()
        mag_norm[0] = 1.0
        mag_norm[1:] /= dc
        
        # Compress: pull non-DC magnitudes toward a target value
        target = 0.3 + 0.2 * (it / max_iters)  # gradually increase allowed spread
        compressed = np.where(mag_norm > target,
                              target + (mag_norm - target) * (1 - alpha),
                              mag_norm)
        
        # Reconstruct
        mag_new = compressed * dc
        mag_new[0] = dc  # Keep DC fixed (preserves sum)
        fa_new = mag_new * np.exp(1j * phase)
        
        a_new = ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def condensed_explosion(a, rng, frac_zero=0.2, n_redistribute=5):
    """
    Condensed explosion: zero out a fraction of low-magnitude elements
    and redistribute their mass to high-magnitude elements.
    """
    n = len(a)
    a = a.copy()
    n_zero = max(5, int(n * frac_zero))
    
    # Find the smallest elements
    sorted_idx = np.argsort(a)
    zero_idx = sorted_idx[:n_zero]
    keep_idx = sorted_idx[n_zero:]
    
    # Remove mass from zero_idx
    mass_removed = a[zero_idx].sum()
    a[zero_idx] = 0.0
    
    # Redistribute to top elements with random weighting
    n_target = min(n_redistribute, len(keep_idx))
    target_idx = keep_idx[:n_target]
    weights = rng.dirichlet(np.ones(n_target) * 2.0)
    a[target_idx] += mass_removed * weights
    
    a = project_nonneg_sum1(a)
    return a


def lp_alternate(a, max_iters=30, time_budget=30.0):
    """LP alternating method for moderate n."""
    n = len(a)
    if n > 1200:
        return a, compute_obj(a)
    
    a = project_nonneg_sum1(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Build constraint matrix
        # (a*g)[k] = sum_j a[j] * g[k-j] <= M for all k
        # This is a Toeplitz-like system
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            A[k_start:k_end, j] = a[:k_end - k_start]
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                s = float(np.sum(g))
                if s > 1e-10:
                    g = g / s
                    # Line search
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                        a_try = (1 - t) * a + t * g
                        a_try = project_nonneg_sum1(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            break
        except Exception:
            pass
    
    return best_a, best_obj


def chase_peak_fast(a, n_steps=30, step_size=0.05):
    """Fast single-peak chasing using direct gradient."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for _ in range(n_steps):
        fa = rfft(a, N)
        conv = irfft(fa * fa, N)[:2 * n - 1]
        k_star = int(np.argmax(conv))
        
        # Gradient: grad[i] = 2*a[k_star - i] for valid index
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        if i_end > i_start:
            indices = np.arange(i_start, i_end)
            a_indices = k_star - indices
            grad[indices] = 2.0 * a[a_indices]
        
        # Center
        grad -= np.mean(grad)
        
        # Try step
        a_try = a - step_size * grad
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
    
    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    inits.append(np.exp(-x**2 / 3) / np.exp(-x**2 / 3).sum())
    
    # Triangular
    tri = np.minimum(np.arange(1, n + 1), n - np.arange(1, n + 1) + 1)
    inits.append(tri / tri.sum())
    
    # Sinusoidal
    x = np.linspace(0, 1, n)
    inits.append((np.sin(np.pi * x)**2 + 0.05) / (np.sin(np.pi * x)**2 + 0.05).sum())
    
    # Random
    inits.append((rng.random(n) + 0.01) / (rng.random(n) + 0.01).sum())
    
    # Two-peak
    p = np.zeros(n)
    p[n//4 - 5:n//4 + 5] = 1.0
    p[3*n//4 - 5:3*n//4 + 5] = 1.0
    inits.append(p / p.sum())
    
    # Bimodal
    x = np.linspace(0, 1, n)
    bim = np.exp(-((x-0.25)**2)/0.01) + np.exp(-((x-0.75)**2)/0.01) + 0.01
    inits.append(bim / bim.sum())
    
    # Spiky
    spiky = np.zeros(n)
    n_spikes = max(5, n // 20)
    spike_pos = rng.choice(n, size=n_spikes, replace=False)
    spiky[spike_pos] = 1.0
    inits.append(spiky / max(spiky.sum(), 1e-10))
    
    if parent is not None:
        inits.append(resample_sequence(parent, n))
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(pert))
    
    return inits

                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 6: Final intensive polish (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining6 = deadline - time.monotonic()
    
    if remaining6 > 20 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)
        
        # Multiple rounds of best strategy
        for pr in range(min(20, int(remaining6 / 10))):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            
            # Gradient with tuned params
            a_p, obj_p = multi_peak_gradient_descent(
                a_final, max_iters=200, time_budget=min(rem * 0.35, 30.0),
                step_init=0.3 + 0.1 * (pr % 3),
                tau_init=0.04 + 0.03 * (pr % 3),
                top_k_init=3 + pr
            )
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish r{pr}: {best_score:.8f}")
            
            # Chase peak
            remain = deadline - time.monotonic()
            if remain > 5:
                a_c, obj_c = chase_peak_fast(a_final, n_steps=30, step_size=0.02)
                if obj_c < best_score:
                    best_score = obj_c
                    best_seq = a_c.copy()
                    a_final = a_c.copy()
            
            # Small perturbation
            remain = deadline - time.monotonic()
            if remain > 10:
                a_pert = a_final * np.exp(rng.normal(0, 0.02, n_final))
                a_pert = project_nonneg_sum1(a_pert)
                a_pp, obj_pp = multi_peak_gradient_descent(
                    a_pert, max_iters=100, time_budget=min(remain * 0.3, 15.0),
                    step_init=0.4, tau_init=0.06, top_k_init=5
                )
                if obj_pp < best_score:
                    best_score = obj_pp
                    best_seq = a_pp.copy()
                    a_final = a_pp.copy()
                    print(f"  Polish pert r{pr}: {best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
