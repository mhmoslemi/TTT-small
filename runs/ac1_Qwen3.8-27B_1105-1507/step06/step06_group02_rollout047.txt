    """Find best t in [0,1] for a(t) = (1-t)*a + t*g (normalized)."""
    # Golden section search
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5) - 1) / 2
    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)
    f1 = eval_J_interp(a, g, t1)
    f2 = eval_J_interp(a, g, t2)
    for _ in range(30):
        if time.monotonic() > deadline:
            break
        if f1 < f2:
            hi = t2
            t2 = t1
            f2 = f1
            t1 = hi - gr * (hi - lo)
            f1 = eval_J_interp(a, g, t1)
        else:
            lo = t1
            t1 = t2
            f1 = f2
            t2 = lo + gr * (hi - lo)
            f2 = eval_J_interp(a, g, t2)
    t_best = (lo + hi) / 2
    a_new = np.maximum((1-t_best)*a + t_best*g, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

        tau = 1.0 * (150.0) ** (frac ** 0.5)
        lr = 0.3 * (0.003 / 0.3) ** frac
        b = fast_conv(y)
        b_max = float(b.max())
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / (exp_b.sum() + 1e-30)
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * p[k] * seg
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=10000):
    """LP mirror descent with line-search step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_imp = 0
    it_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Check convergence: if g is close to a, we're at a fixed point
        if np.abs(g - a).max() < 1e-8:
            break

        # Line search: try several step sizes, pick best
        best_t = 0.3
        best_M_t = float('inf')
        for t in [0.5, 0.4, 0.35, 0.3, 0.25, 0.2, 0.15, 0.1]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t

        a = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s
        it_count += 1

        J = compute_J(a)
        if J < best_J * (1 - 1e-9):
            best_J = J
            best_a = a.copy()
            no_imp = 0
            if it_count % 10 == 0:
                print(f"    LP iter {it_count}: J={J:.8f}, t={best_t:.2f}, M={best_M_t:.8f}", flush=True)
        else:
            no_imp += 1
            if no_imp > 25:
                break

    return best_a


def lp_mirror_fixed(a, deadline, t=0.3, max_iters=10000):
    """LP mirror descent with fixed step size (faster per iteration)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_imp = 0
    it_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        if np.abs(g - a).max() < 1e-8:
            break

        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
        it_count += 1

        J = compute_J(a)
        if J < best_J * (1 - 1e-9):
            best_J = J
            best_a = a.copy()
            no_imp = 0
            if it_count % 20 == 0:
                print(f"    LP-fixed iter {it_count}: J={J:.8f}, t={t}", flush=True)
        else:
            no_imp += 1
            if no_imp > 40:
                break

    return best_a


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a, label=""):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s {label}", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent, "parent")
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent, "ones")

    # === PHASE 1: Quick smooth initialization on multiple lengths (180s) ===
    p1_deadline = time.monotonic() + 180
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Init (180s)", flush=True)
    print(f"{'='*50}", flush=True)

    best_per_length = {}
    n_parent = parent.size
    lengths_to_try = sorted(set([
        n_parent,
        max(256, n_parent // 2),
        min(4096, n_parent * 2),
        1024, 1536, 2048
    ]))
    lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 4096]
    print(f"Lengths to try: {lengths_to_try}", flush=True)

    for n in lengths_to_try:
        if time.monotonic() > p1_deadline:
            break
        t0 = time.monotonic()
        a0 = resample_seq(parent, n)
        a0 = np.maximum(a0, 1e-6)
        a0 /= a0.sum()
        # Give less time to longer sequences
        allot = 25.0 * (1024.0 / n)
        a = smooth_nesterov(a0, min(time.monotonic() + allot, p1_deadline), max_iters=3000)
        J = compute_J(a)
        best.update(a, f"smooth n={n}")
        best_per_length[n] = a.copy()
        print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)

    best.report("Phase 1")

    # === PHASE 2: LP mirror descent on best (650s) ===
    p2_deadline = min(time.monotonic() + 650, deadline - 150)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent on best", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        print(f"  Starting n={n}, J={best.score:.8f}", flush=True)

        # Line search LP (converges faster per iteration, but more expensive)
        dl1 = min(time.monotonic() + 200, p2_deadline - 300)
        a = lp_mirror_descent(best.seq, dl1, max_iters=5000)
        best.update(a, "LP linesearch")

        # Fixed step LP with moderate t (more iterations)
        if time.monotonic() < p2_deadline - 100:
            a = lp_mirror_fixed(a, min(time.monotonic() + 250, p2_deadline - 50), t=0.3, max_iters=10000)
            best.update(a, "LP t=0.3")

        # Fixed step LP with smaller t (finer convergence)
        if time.monotonic() < p2_deadline:
            a = lp_mirror_fixed(a, p2_deadline, t=0.15, max_iters=10000)
            best.update(a, "LP t=0.15")

    best.report("Phase 2")

    # === PHASE 3: Second/third best length with LP (250s) ===
    p3_deadline = min(time.monotonic() + 250, deadline - 80)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Other lengths LP", flush=True)
    print(f"{'='*50}", flush=True)

    if best_per_length and time.monotonic() < p3_deadline:
        n_best = best.seq.size if best.seq is not None else 0
        other_lengths = [(n, a) for n, a in best_per_length.items() if n != n_best]
        other_lengths.sort(key=lambda x: compute_J(x[1]))
        for n_try, a_try in other_lengths[:2]:
            if time.monotonic() > p3_deadline - 40:
                break
            print(f"  Trying n={n_try}...", flush=True)
            a_try = np.maximum(a_try, 1e-6)
            a_try /= a_try.sum()
            # Quick smooth polish
            a_try = smooth_nesterov(a_try, min(time.monotonic() + 15, p3_deadline - 80), max_iters=1000)
            best.update(a_try, f"smooth2 n={n_try}")
            # LP refinement
            if time.monotonic() < p3_deadline - 20:
                a_try = lp_mirror_fixed(a_try, p3_deadline - 10, t=0.25, max_iters=5000)
                best.update(a_try, f"LP n={n_try}")

    best.report("Phase 3")

    # === PHASE 4: Final polishing with decreasing steps (120s) ===
    p4_deadline = deadline
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < p4_deadline - 5:
        for t in [0.2, 0.1, 0.05, 0.03]:
            if time.monotonic() > p4_deadline - 3:
                break
            remaining = p4_deadline - time.monotonic() - 3
            if remaining < 5:
                break
            print(f"  Polish with t={t}, remaining={remaining:.0f}s", flush=True)
            a = lp_mirror_fixed(best.seq, time.monotonic() + remaining, t=t, max_iters=10000)
            best.update(a, f"polish t={t}")

    best.report("Phase 4")

    # === PHASE 5: Try slight perturbations to escape potential local minima ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*50}", flush=True)
        print(f"PHASE 5: Perturbation search", flush=True)
        print(f"{'='*50}", flush=True)
        n_best = best.seq.size
        for scale in [0.01, 0.02, 0.05]:
            if time.monotonic() > deadline - 5:
                break
            a_pert = np.maximum(best.seq + rng.normal(0, scale, n_best), 1e-6)
            a_pert /= a_pert.sum()
            a_pert = lp_mirror_fixed(a_pert, min(time.monotonic() + 10, deadline - 3), t=0.3, max_iters=2000)
            best.update(a_pert, f"perturb scale={scale}")

    best.report("Final")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
