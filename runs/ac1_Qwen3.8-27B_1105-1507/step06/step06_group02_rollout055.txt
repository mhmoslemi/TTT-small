"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation for smoothed max-convolution objective
2. Nesterov-accelerated smooth descent with adaptive p-norm parameter
3. LP mirror descent with backtracking line search and Anderson acceleration
4. Multi-length exploration with diverse spectral initializations
5. Final polish phase with exact line search
Key improvements: O(n log n) gradients, adaptive line search, longer sequences.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def grad_pnorm(a, p, tau=0.0):
    """Gradient of (sum b_k^p)^(1/p) for large p (approximates max)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Use log-sum-exp style with effective temperature
    if tau > 0:
        exp_b = np.exp(tau * (b - b_max))
        w = exp_b / exp_b.sum()
    else:
        # p-norm gradient: w_k = b_k^(p-1) / sum(b_j^(p-1))
        b_pos = np.maximum(b, 1e-30)
        b_pow = b_pos ** (p - 1)
        w = b_pow / b_pow.sum()
    # grad[m] = 2 * sum_l a[l] * w[l+m] (cross-correlation)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    cross_corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    grad = 2.0 * cross_corr
    return grad, b, b_max, w


def nesterov_smooth(a, deadline, max_iters=3000, p_start=50, p_end=500,
                     lr_init=0.25, lr_end=0.001, momentum=0.9):
    """Nesterov-accelerated gradient descent with p-norm objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    v = np.zeros(n)

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        p_val = p_start * (p_end / p_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max, w = grad_pnorm(y, p_val)

        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = y.copy()

        v = momentum * v - lr * grad
        y_new = y + v
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-10:
            v = 0.0
            y_new = y.copy()
        else:
            y_new /= sy
        y = y_new

    return best_a


def build_conv_matrix_sparse(n, a):
    """Build sparse convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M_target=None):
    """Solve LP: min t s.t. (a*g)_k <= t, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix_sparse(n, a)
    # Variables: g_0,...,g_{n-1}, t (n+1 vars)
    # Constraints: (Ag)_k - t <= 0 for k=0,...,2n-2
    #              sum(g) = 1
    #              g >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t

    # Inequality: A @ g - t <= 0
    # As (A | -I_col) @ [g; t] <= 0
    A_ineq = np.hstack([A.toarray(), -np.eye(2 * n - 1, 1)])

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        r = linprog(c, A_ub=A_ineq, b_ub=np.zeros(2 * n - 1),
                    A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                    method='highs', options={'presolve': True,
                                              'dual_feasibility_tolerance': 1e-9,
                                              'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            if g.sum() > 1e-15:
                g /= g.sum()
                return g, r.x[-1]
    except Exception:
        pass
    return None, None


def anderson_mix(hist_x, hist_Fx, m, clip_range=(-1.0, 2.0)):
    """Anderson mixing: find alpha to minimize ||sum alpha_i (x_i - F(x_i))||^2."""
    if m < 2:
        return hist_Fx[-1].copy()
    residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        alpha = alpha / alpha.sum()
        alpha = np.clip(alpha, clip_range[0], clip_range[1])
        s = alpha.sum()
        if s <= 1e-15:
            alpha = ones / m
        else:
            alpha /= s
    except np.linalg.LinAlgError:
        alpha = ones / m
    mixed = np.zeros(len(hist_x[0]))
    for i in range(m):
        mixed += alpha[i] * hist_Fx[i]
    return np.maximum(mixed, 0.0)


def lp_mirror_descent(a, deadline, max_iters=200, anderson_window=5,

    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_imp = 0

    hist_x = []
    hist_Fx = []

    for it in range(max_iters):

            break

        g, t_opt = lp_best_response(a)
        if g is None:
            break

        # Try different step sizes with line search
        best_t = t_initial
        best_val = float('inf')
        best_a_t = None

        if use_line_search:
            t_candidates = [t_initial, t_initial * 0.7, t_initial * 0.5,
                           t_initial * 0.3, t_initial * 0.2, t_initial * 0.1,
                           t_min]
            for t in t_candidates:
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0.0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t /= s
                J_t = compute_J(a_t)
                if J_t < best_val:
                    best_val = J_t
                    best_t = t
                    best_a_t = a_t.copy()
                # Early stop if we found a good step
                if J_t < best_J * 0.99:
                    break
        else:
            t = max(t_min, t_initial / (1 + 0.03 * it))
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s > 1e-10:
                a_t /= s
                best_a_t = a_t
                best_val = compute_J(a_t)

        if best_a_t is None:
            break

        # Anderson acceleration
        if len(hist_x) >= 3:
            a_anderson = anderson_mix(hist_x, hist_Fx, len(hist_x))
            sa = a_anderson.sum()
            if sa > 1e-10:
                a_anderson /= sa
                J_and = compute_J(a_anderson)
                if J_and <= best_val * 1.02:
                    best_a_t = a_anderson
                    best_val = J_and

        if best_val < best_J:
            best_J = best_val
            best_a = best_a_t.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 25:
                # Decay step size
                t_initial = max(t_min, t_initial * 0.7)

                    break

        hist_x.append(a.copy())
        hist_Fx.append(best_a_t.copy())

            hist_Fx.pop(0)

        a = best_a_t
        # Decay initial step size slowly
        t_initial = max(t_min, t_initial * 0.98)

def gs_flat_spectral(init, n_iters=200, beta=0.85):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        # Compress dynamic range
        power = 0.4 + 0.4 * beta
        mag_new = (mag_norm ** power) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 1e-10)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    inits[-1] = inits[-1] / inits[-1].sum()

    # Symmetric random

    # Prolate-like: Time-limited approximation of sinc
    sinc = np.sinc(t - 0.5)

class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

    n_len = len(lengths)

    for idx_n, n in enumerate(lengths):
        t0 = time.monotonic()
        remaining = p1_deadline - time.monotonic()
        if remaining < 20:
            break
        allot = min(time_per_len * 1.2, remaining * 0.6)
        dl = time.monotonic() + allot

        # Choose parent for this length
        par = parent
        if best.seq is not None:
            par = best.seq
        inits = make_initializations(n, rng, par)

        best_n_a = None
        best_n_J = float('inf')

        # Run smooth optimization on each init (limit number of inits for time)
        max_inits = min(len(inits), max(3, int(8 * (300.0 / n))))
        for idx_a, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 /= a0.sum()

            # Adaptive iterations based on remaining time
            sub_dl = time.monotonic() + (dl - time.monotonic()) / (max_inits - idx_a)
            max_it = max(200, int(5000 * (512.0 / n)))

            a = nesterov_smooth(a0, sub_dl, max_iters=max_it,
                                p_start=30, p_end=300,
                                lr_init=0.2, lr_end=0.001, momentum=0.9)

            J = compute_J(a)
            if J < best_n_J:
                best_n_J = J
                best_n_a = a.copy()

        if best_n_a is not None:
            J = compute_J(best_n_a)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n_a)
            best_per_length[n] = best_n_a.copy()

    best.report("Phase 1")

    # === PHASE 2: LP Mirror Descent on best sequences ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print("PHASE 2: LP Mirror Descent + Line Search", flush=True)
    print(f"{'='*50}", flush=True)

    # Sort lengths by J
    sorted_lens = sorted(best_per_length.keys(),
                         key=lambda x: compute_J(best_per_length[x]))

    # Spend most time on the best 2-3 lengths
    top_lens = sorted_lens[:3]
    p2_deadline = deadline - 150  # Reserve 150s for final phase

    for idx_n, n in enumerate(top_lens):
        remaining = p2_deadline - time.monotonic()
        if remaining < 30:
            break

        a_start = best_per_length[n].copy()
        # If we have a better global best at this length, use it
        if best.seq is not None and best.seq.size == n:
            if compute_J(best.seq) < compute_J(a_start):
                a_start = best.seq.copy()

        # Estimate LP time
        est_lp_s = 0.05 * (n / 512.0) ** 1.5
        n_iters = max(20, int(min(remaining * 0.8, 600) / est_lp_s))
        n_iters = min(n_iters, 300)  # Cap iterations

        print(f"  n={n}, est_lp={est_lp_s:.3f}s, iters={n_iters}, remaining={remaining:.0f}s", flush=True)

        # First pass: larger steps
        dl1 = time.monotonic() + min(remaining * 0.5, 300)
        a = lp_mirror_descent(a_start, dl1, max_iters=n_iters,
                              anderson_window=5, t_initial=0.4, t_min=0.01,
                              use_line_search=True)
        best.update(a)

        # Second pass: smaller steps for refinement
        if time.monotonic() < p2_deadline - 30:
            remaining2 = p2_deadline - time.monotonic()
            dl2 = time.monotonic() + min(remaining2 * 0.7, 200)
            a2 = lp_mirror_descent(a, dl2, max_iters=int(n_iters * 0.6),
                                   anderson_window=6, t_initial=0.15, t_min=0.003,
                                   use_line_search=True)
            best.update(a2)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Final polish ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print("PHASE 3: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        # Polish the global best
        a_final = lp_mirror_descent(best.seq, min(time.monotonic() + 40, deadline - 10),
                                    max_iters=80, anderson_window=5,
                                    t_initial=0.1, t_min=0.002,
                                    use_line_search=True)
        best.update(a_final)

        # Try resampling to nearby lengths
        n_best = best.seq.size
        for delta in [32, -32, 64, -64, 128, -128]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 2560:
                continue
            a_res = resample_seq(best.seq, n_try)
            a_res = np.maximum(a_res, 1e-8)
            a_res /= a_res.sum()
            # Quick smooth
            a_res = nesterov_smooth(a_res, min(time.monotonic() + 8, deadline - 5),
                                    max_iters=300, p_start=100, p_end=300,
                                    lr_init=0.1, lr_end=0.005, momentum=0.9)
            best.update(a_res)
            # Quick LP polish
            if time.monotonic() < deadline - 5:
                a_lp = lp_mirror_descent(a_res, deadline - 3, max_iters=15,
                                         anderson_window=4, t_initial=0.2,
                                         t_min=0.01, use_line_search=True)
                best.update(a_lp)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: One more LP pass with very small steps ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        a_ultra = lp_mirror_descent(best.seq, deadline, max_iters=50,
                                    anderson_window=5, t_initial=0.05, t_min=0.001,
                                    use_line_search=True)
        best.update(a_ultra)
        best.report("Phase 4")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(512) / 512.0
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
