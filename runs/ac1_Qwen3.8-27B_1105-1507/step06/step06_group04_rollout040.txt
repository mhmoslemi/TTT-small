    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    x0 = np.zeros(n + 1)
    if prev_g is not None:
        x0[:n] = prev_g
        x0[n] = float(np.max(M @ prev_g))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                t_opt = float(res.x[n])
                # Extract dual for inequality constraints
                y = np.abs(res.ineqlin.carrier) if hasattr(res.ineqlin, 'carrier') else None
                return g, t_opt, y
    except Exception:
        pass
    return None, None, None


def solve_adaptive_peak_lp(a, M, top_k_frac=0.3, prev_g=None, time_limit=20):
    """
    Solve LP with only the top-K*m constraints (adaptive constraint set).
    This reduces LP size and focuses on the most critical peaks.
    """
    n = len(a)
    # Compute current convolution to identify active set
    conv = M @ normalize(a)
    threshold = np.percentile(conv, (1 - top_k_frac) * 100)
    active_mask = conv >= threshold
    num_active = np.sum(active_mask)
    
    if num_active < max(10, 2 * n // 10):
        active_mask = conv >= np.percentile(conv, 50)
    
    if num_active == 2 * n - 1:
        return solve_peak_lp(a, M, prev_g, time_limit)
    
    M_sub = M[active_mask, :]
    n_sub = len(active_mask)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([M_sub, -np.ones((n_sub, 1))])
    b_ub = np.zeros(n_sub)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def lse_subgradient(a, tau):
    """
    Compute subgradient of smoothed max(conv(a,a))/sum(a)^2 using log-sum-exp.
    Returns (gradient, current_obj).
    """
    n = len(a)
    s = np.sum(a)
    conv = compute_conv(a)
    
    c_max = np.max(conv)
    shifted = (conv - c_max) / tau
    shifted = np.clip(shifted, -500.0, 500.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum
    
    # Gradient of LSE part: grad[i] = (1/tau) * sum_k w[k] * dC[k]/da[i]
    # dC[k]/da[i] = 2*a[k-i] if 0<=k-i<n (general case, handles k=2i too)
    # So sum_k w[k]*dC[k]/da[i] = 2 * sum_k w[k]*a[k-i]
    # = 2 * (cross-correlation of w and a) at lag i
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    W = rfft(w_padded, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    
    lse_grad = 2.0 * corr[:n] / tau
    
    # Gradient of -2*log(S): -2/S
    full_grad = lse_grad - 2.0 / s * np.ones(n)
    
    obj = compute_obj(a)
    return full_grad, obj


def lse_descent(a, max_iters=3000, time_budget=60.0, tau_init=0.003, tau_decay=0.9997):
    """
    Gradient descent on log-sum-exp approximation with momentum and backtracking.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    mu = 0.88
    tau = tau_init
    lr = 0.08
    consecutive_worse = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, obj = lse_subgradient(a, tau)
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            consecutive_worse = 0
        
        grad_c = grad - np.mean(grad)
        gn = np.linalg.norm(grad_c)
        if gn < 1e-15:
            break
        
        # Momentum update
        v = mu * v - lr * grad_c / gn
        a_new = a + v
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < obj:
            a = a_new
            lr = min(lr * 1.01, 0.15)
        else:
            # Backtrack
            for _ in range(4):
                lr *= 0.5
                v = v * 0.5
                a_new = a + v
                a_new = normalize(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    a = a_new
                    best_obj = obj_new
                    best_a = a.new.copy() if hasattr(a, 'new') else a.copy()
                    break
            else:
                consecutive_worse += 1
                if consecutive_worse > 20:
                    # Perturb to escape
                    rng = np.random.default_rng(it * 104729 + 7)
                    a = best_a * np.exp(rng.normal(0, 0.03, n))
                    a = normalize(a)
                    v = np.zeros(n)
                    consecutive_worse = 0
        
        tau *= tau_decay
    
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    return best_a, best_obj


def lp_alternating_improved(a, M, max_iters=100, time_budget=200.0):
    """
    Improved LP alternating with warm starts, adaptive constraints, and better line search.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_g = None
    last_good_t = 0.5
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        lp_time = min(15.0, remaining * 0.3)
        if lp_time < 3:
            break
        
        # Solve LP (adaptive constraint set for speed)
        if it % 3 == 0 and n > 400:
            g, _ = solve_adaptive_peak_lp(a, M, top_k_frac=0.2, prev_g=prev_g,
                                          time_limit=lp_time)
        else:
            g, _, _ = solve_peak_lp(a, M, prev_g=prev_g, time_limit=lp_time)
        
        if g is None:
            stall += 1
            if stall > 15:
                break
            continue
        
        prev_g = g
        stall = 0
        
        # Multi-scale line search
        t_cands = [last_good_t * 0.5, last_good_t * 0.75, last_good_t,
                   last_good_t * 1.25, last_good_t * 1.5,
                   0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        t_cands = sorted(set([max(0.01, min(1.0, t)) for t in t_cands]))
        
        best_t = last_good_t
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        
        for t in t_cands:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local
            a = best_a_local
            last_good_t = best_t
        else:
            # Try a small perturbation
            rng = np.random.default_rng(it * 31337 + 11)
            pert_scale = 0.01 + 0.01 * min(stall, 10)
            a_pert = best_a * np.exp(rng.normal(0, pert_scale, n))
            a_pert = normalize(a_pert)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert
                a = a_pert
            stall += 1
            if stall > 20:
                break
    
    return best_a, best_obj


def spectral_init(n, seed=42):
    """
    Generate initial sequence using spectral shaping.
    Try to make the autocorrelation as flat as possible.
    """
    rng = np.random.default_rng(seed)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    # Start with random phases
    best_a = np.ones(n) / n
    best_obj = compute_obj(best_a)
    
    for attempt in range(20):
        # Random magnitude perturbation
        A = np.fft.rfft(np.ones(n), fft_size)
        # Perturb magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Apply random smoothing to magnitude
        smooth_sigma = 0.1 + 0.5 * rng.random()
        kernel = np.exp(-0.5 * (np.arange(fft_size // 2) / (fft_size // 2 * smooth_sigma)) ** 2)
        new_mag = np.abs(np.fft.irfft(mag * kernel, fft_size)[:fft_size // 2])
        # Actually let's work in the time domain differently
        
        # Alternative: multiply by a random smooth window
        t = np.arange(n) / n
        window = 1.0 + rng.uniform(-0.3, 0.3) * np.cos(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 2 * np.pi))
        window = np.maximum(window, 0.1)
        a = np.ones(n) * window
        a = normalize(a)
        
        # Apply a few spectral iterations
        for _ in range(5):
            A = rfft(a, fft_size)
            mag = np.abs(A)
            ph = np.angle(A)
            # Flatten magnitude
            mean_mag = np.mean(mag)
            mag = mag - 0.3 * (mag - mean_mag)
            mag = np.maximum(mag, 0.01)
            a = irfft(mag * np.exp(1j * ph), fft_size)[:n]
            a = normalize(a)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def multi_peak_chase(a, num_peaks=5, max_iters=500, time_budget=30.0):
    """
    Chase top-K peaks simultaneously using weighted subgradient.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.04
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Weight top K peaks
        top_k = np.argsort(conv)[-num_peaks:]
        c_max = conv[top_k[0]]
        weights = np.full(len(conv), 0.001)
        for k in top_k:
            weights[k] = 1.0 / num_peaks
        
        # Compute gradient
        w_padded = np.zeros(fft_size)
        w_padded[:2 * n - 1] = weights
        W = rfft(w_padded, fft_size)
        A = rfft(a, fft_size)
        corr = irfft(A * np.conj(W), fft_size)
        grad = 2.0 * corr[:n] - 2.0 * np.mean(corr[:n])
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        a_new = normalize(a - step * grad / gn)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.02, 0.1)
        else:
            step *= 0.9
            if step < 0.001:
                # Perturb
                rng = np.random.default_rng(it * 7919 + 3)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = normalize(a)
                step = 0.04
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def polish_sequence(a, time_budget=120.0):
    """
    Full polishing pipeline alternating between LP, LSE descent, and peak chasing.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute convolution matrix (expensive but done once)
    M = build_conv_matrix(a)
    
    phase = 0
    stall = 0
    
    while time.monotonic() - start < time_budget - 3:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break
        
        alloc = min(remaining * 0.2, 40.0)
        
        if phase % 4 == 0 and n <= 900:
            # LP-based
            a_new, obj_new = lp_alternating_improved(best_a, M, max_iters=20,
                                                      time_budget=min(alloc, 30.0))
        elif phase % 4 == 1:
            # LSE descent
            a_new, obj_new = lse_descent(best_a, max_iters=int(alloc * 20),
                                          time_budget=min(alloc, 25.0),
                                          tau_init=0.002, tau_decay=0.9995)
        elif phase % 4 == 2:
            # Multi-peak chase
            a_new, obj_new = multi_peak_chase(best_a, num_peaks=5,
                                               max_iters=int(alloc * 8),
                                               time_budget=min(alloc, 20.0))
        else:
            # LSE with different params
            a_new, obj_new = lse_descent(best_a, max_iters=int(alloc * 15),
                                          time_budget=min(alloc, 25.0),
                                          tau_init=0.01, tau_decay=0.999)
        
        phase += 1
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            # Rebuild M since a changed
            M = build_conv_matrix(best_a)
            print(f"    Polish phase {phase}: {best_obj:.8f} (improved)")
        else:
            stall += 1
            if stall > 3:
                # Structured perturbation
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
                # Try different perturbation strategies
                strategy = stall % 4
                if strategy == 0:
                    # Multiplicative noise
                    a_pert = best_a * np.exp(rng.normal(0, 0.03, n))
                elif strategy == 1:
                    # Smooth perturbation
                    noise = rng.normal(0, 0.02, n)
                    # Smooth it
                    for _ in range(3):
                        noise = 0.5 * (noise + np.roll(noise, 1))
                    a_pert = best_a * np.exp(noise)
                elif strategy == 2:
                    # End-bias perturbation
                    t = np.arange(n) / n
                    bias = 1.0 + 0.05 * np.cos(2 * np.pi * t)
                    a_pert = best_a * bias
                else:
                    # Center-bias perturbation
                    t = np.arange(n) / n
                    bias = 1.0 + 0.05 * np.sin(np.pi * t)
                    a_pert = best_a * bias
                
                a_pert = normalize(a_pert)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert
                    M = build_conv_matrix(best_a)
                stall = 0
    
    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded: n={parent.size}, obj={obj:.8f}")
            return parent.copy(), obj
    except Exception as e:
        print(f"Could not load parent: {e}")
    return None, np.inf


def try_single_n(n, seed, time_budget):
    """Try optimizing at a specific length n."""
    start = time.monotonic()
    
    # Generate multiple starts
    starts = []
    rng = np.random.default_rng(seed)
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Spectral init
    a_spec, _ = spectral_init(n, seed=seed)
    starts.append(a_spec)
    
    # Bell-shaped
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 4.0) ** 2))
    starts.append(bell / np.sum(bell))
    
    # Edge-concentrated
    edge = np.exp(-np.minimum(i, n - 1 - i) / (n / 8.0))
    starts.append(edge / np.sum(edge))
    
    # Noisy uniform
    noisy = np.abs(np.ones(n) + rng.normal(0, 0.2, n))
    starts.append(noisy / np.sum(noisy))
    
    # Quick screen
    best_a = None
    best_obj = np.inf
    per_start_budget = time_budget / len(starts)
    
    for si, s in enumerate(starts):
        if time.monotonic() - start > time_budget - 5:
            break
        t_alloc = min(per_start_budget, time_budget - (time.monotonic() - start) - 3)
        if t_alloc < 5:
            break
        
        a_opt, obj_opt = lse_descent(s, max_iters=int(t_alloc * 15),
                                      time_budget=t_alloc * 0.6,
                                      tau_init=0.005, tau_decay=0.9995)
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_opt2, obj_opt2 = multi_peak_chase(a_opt, num_peaks=4,
                                                 max_iters=int(remaining * 3),
                                                 time_budget=remaining * 0.4)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2
        
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase approach:
    1. Load parent and evaluate
    2. Quick exploration across n values
    3. Intensive polish on best candidates
    4. Final refinement
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_seq = parent.copy()
        best_score = parent_obj
    
    if best_seq is None:
        n0 = 700
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
    
    n_parent = len(best_seq)
    print(f"Initial: n={n_parent}, score={best_score:.8f}")
    
    # Phase 1: Quick exploration of nearby n values
    print("\n=== Phase 1: n-exploration ===")
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent - 20,
        n_parent, n_parent + 20, n_parent + 50, n_parent + 100
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 1200]
    
    time_per_n = min(60.0, (deadline - time.monotonic()) / max(len(n_candidates), 1) - 5)
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        alloc = min(time_per_n, remaining * 0.3)
        if alloc < 10:
            continue
        
        print(f"  Trying n={n_try} (budget={alloc:.0f}s)...")
        
        if n_try == n_parent and parent is not None:
            # Start from parent for this n
            a_start = normalize(best_seq.copy())
            a_opt, obj_opt = polish_sequence(a_start, time_budget=alloc * 0.4)
            # Also try fresh starts
            a_fresh, obj_fresh = try_single_n(n_try, seed + n_try, alloc * 0.4)
            if obj_fresh < obj_opt:
                a_opt, obj_opt = a_fresh, obj_fresh
        else:
            a_opt, obj_opt = try_single_n(n_try, seed + n_try, alloc)
        
        if a_opt is not None and obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"    ** NEW BEST: n={n_try}, score={best_score:.8f} **")
        elif a_opt is not None:
            print(f"    Best at n={n_try}: {obj_opt:.8f}")
    
    # Phase 2: Intensive polish on the best
    print(f"\n=== Phase 2: Intensive polish (n={n_parent}, score={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 300.0)
        print(f"  Polishing with {alloc:.0f}s budget...")
        a_pol, obj_pol = polish_sequence(best_seq, time_budget=alloc)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** IMPROVED: {best_score:.8f} **")
        else:
            print(f"  No improvement from polish: {obj_pol:.8f}")
    
    # Phase 3: Try resampling to slightly different lengths
    print(f"\n=== Phase 3: Length refinement (score={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        n_best = len(best_seq)
        for dn in [-30, -15, 15, 30, -60, 60]:
            if time.monotonic() > deadline - 50:
                break
            remaining2 = deadline - time.monotonic()
            if remaining2 < 15:
                break
            
            n_new = n_best + dn
            if n_new < 200 or n_new > 1500:
                continue
            
            a_res = resample_sequence(best_seq, n_new)
            a_res = normalize(a_res)
            obj_res = compute_obj(a_res)
            
            if obj_res < best_score:
                print(f"  n={n_new}: resampled obj={obj_res:.8f}, polishing...")
                alloc = min(remaining2 * 0.3, 60.0)
                a_pol, obj_pol = polish_sequence(a_res, time_budget=alloc)
                if obj_pol < best_score:
                    best_score = obj_pol
                    best_seq = a_pol.copy()
                    n_best = n_new
                    print(f"  ** NEW BEST: n={n_new}, score={best_score:.8f} **")
            else:
                print(f"  n={n_new}: {obj_res:.8f} (worse than {best_score:.8f})")
    
    # Phase 4: Final intensive LP polish
    print(f"\n=== Phase 4: Final LP polish (score={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_seq is not None:
        n_final = len(best_seq)
        if n_final <= 900:
            M = build_conv_matrix(normalize(best_seq))
            alloc = min(remaining * 0.5, 120.0)
            a_final, obj_final = lp_alternating_improved(best_seq, M,
                                                          max_iters=50, time_budget=alloc)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  ** LP IMPROVED: {best_score:.8f} **")
            else:
                print(f"  LP result: {obj_final:.8f}")
        
        # One more LSE pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_lse, obj_lse = lse_descent(best_seq, max_iters=int(remaining * 20),
                                          time_budget=remaining * 0.6,
                                          tau_init=0.001, tau_decay=0.9998)
            if obj_lse < best_score:
                best_score = obj_lse
                best_seq = a_lse.copy()
                print(f"  ** LSE IMPROVED: {best_score:.8f} **")
    
    # Phase 5: Final multi-peak chase
    remaining = deadline - time.monotonic()
    if remaining > 10 and best_seq is not None:
        a_chase, obj_chase = multi_peak_chase(best_seq, num_peaks=8,
                                               max_iters=int(remaining * 5),
                                               time_budget=remaining * 0.7)
        if obj_chase < best_score:
            best_score = obj_chase
            best_seq = a_chase.copy()
            print(f"  ** CHASE IMPROVED: {best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
