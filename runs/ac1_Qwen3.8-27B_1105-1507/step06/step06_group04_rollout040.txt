"""
Peak-reduction search using alternating LP (min-peak formulation) with
multiple restarts, spectral initialization, and adaptive length exploration.
Focuses on the LP as the primary optimization engine for fast convergence.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build sparse representation info for the Toeplitz convolution matrix.
    Returns a scipy sparse matrix where (A @ g)[k] = (convolve(a, g))[k]."""
    from scipy.sparse import lil_matrix
    n = len(a)
    A = lil_matrix((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A.tocsr()


def solve_minpeak_lp(fixed_seq, time_limit=10):
    """
    Solve: min t subject to (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    # Build the constraint matrix
    # (A @ g)[k] = sum_j a[j] * g[k-j] (where valid) = (convolve(a, g))[k]
    # We need: A @ g - t <= 0 for all k
    # Variables: g[0..n-1], t (total n+1 variables)
    
    from scipy.sparse import hstack, csc_matrix, eye as sp_eye
    
    A_conv = build_toeplitz_convolve_mat(fixed_seq)  # (2n-1) x n
    
    # A_ub: (2n-1) x (n+1), with columns [A_conv | -1]
    neg_one = csc_matrix((2 * n - 1, 1), dtype=np.float64)
    # Fill with -1
    data = np.full(2 * n - 1, -1.0)
    rows = np.arange(2 * n - 1)
    cols = np.zeros(2 * n - 1, dtype=int)
    neg_one = csc_matrix((data, (rows, cols)), shape=(2 * n - 1, 1))
    
    A_ub = hstack([A_conv, neg_one], format='csc')
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    bounds = [(0.0, None)] * n + [(None, None)]  # g >= 0, t free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception as e:
        pass
    return None, None


def lp_alternate(a, max_iters=500, time_budget=300.0, verbose=False):
    """
    Pure alternating LP method with full updates.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()

    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:

        # Solve LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        tl = min(15.0, remaining * 0.3)
        
        g, t_opt = solve_minpeak_lp(a, time_limit=tl)
        if g is None:
            stall += 1
            if stall > 5:
                break
            continue
        
        # Full update
        a_new = g
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 50 == 0:

        elif obj_new < best_obj + 1e-10:

            stall = 0

def generate_starts(n, seed=42):
    """Generate diverse starting sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Gaussian (various widths)
    for sigma_frac in [0.15, 0.25, 0.35, 0.5]:
        bell = np.exp(-((i - center) ** 2) / (2 * (n * sigma_frac) ** 2))
        starts.append(bell / np.sum(bell))
    
    # Triangular
    tri = 1.0 - np.abs(i - center) / center
    tri = np.maximum(tri, 0.0)
    if tri.sum() > 0:
        starts.append(tri / tri.sum())
    
    # Cosine-modulated
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            phase = 2 * np.pi * rng.random()
            mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
            a = np.maximum(mod, 0)
            s = a.sum()
            if s > 0:
                starts.append(a / s)
    
    # Random
    for _ in range(5):
        a = np.abs(rng.normal(1, 0.5, n))
        starts.append(a / a.sum())
    
    return starts


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Peak-reduction search using alternating min-peak LP with diverse multi-start,
symmetric initialization, adaptive restarts, and length exploration.
The LP is the primary engine: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix, hstack
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_conv_matrix(a):
    """Build sparse Toeplitz matrix A where (A@g)[k] = (convolve(a,g))[k]."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        if hi > lo:
            A[lo:hi, j] = a[:hi - lo]
    return A.tocsr()


def solve_minpeak_lp(a, time_limit=10.0):
    """
    Solve: min t s.t. (a*g)[k] <= t for k=0..2n-2, sum(g)=1, g>=0.
    Returns (g_normalized, t_opt) or (None, None).
    """
    n = len(a)
    A_conv = build_conv_matrix(a)

    neg_one_data = np.full(2 * n - 1, -1.0)
    neg_one = csc_matrix((neg_one_data, (np.arange(2 * n - 1), np.zeros(2 * n - 1, dtype=int))),
                         shape=(2 * n - 1, 1))

    A_ub = hstack([A_conv, neg_one], format='csc')
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternate(a, max_iters=2000, time_budget=300.0, verbose=False):
    """
    Alternating min-peak LP with full updates and adaptive blending.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_obj = best_obj

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        tl = min(10.0, remaining * 0.2)

        g, t_opt = solve_minpeak_lp(a, time_limit=tl)
        if g is None:
            stall += 1
            if stall > 8:
                break
            continue

        obj_new = compute_obj(g)

        if obj_new < best_obj - 1e-13:
            a = g
            best_a = g.copy()
            best_obj = obj_new
            stall = 0
            if verbose and (it % 100 == 0 or obj_new < last_obj - 0.001):
                print(f"    LP[{it}]: obj={obj_new:.8f}")
            last_obj = best_obj
        else:
            # Try blends
            found = False
            for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
                a_try = normalize_simplex((1.0 - t) * best_a + t * g)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    a = a_try
                    best_a = a_try.copy()
                    best_obj = obj_try
                    stall = 0
                    found = True
                    break

                stall += 1
                if stall > 8:
                    # Perturb to escape

                    scale = 0.04 + 0.01 * min(stall - 8, 20)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = normalize_simplex(a)
                    stall = 0
                    if verbose:
                        print(f"    Perturb at iter {it}, scale={scale:.3f}")

    return best_a, best_obj


def generate_starts(n, seed=42):
    """Generate diverse starting sequences of length n."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    starts.append(np.ones(n) / n)

    # Symmetric Gaussian bells
    for sf in [0.12, 0.2, 0.3, 0.45]:
        bell = np.exp(-((i - center) ** 2) / (2.0 * (n * sf) ** 2))
        starts.append(bell / bell.sum())

    # Symmetric triangular
    tri = 1.0 - np.abs(i - center) / center
    tri = np.maximum(tri, 0.0)
    if tri.sum() > 0:
        starts.append(tri / tri.sum())

    # Cosine-modulated (symmetric)
    for freq in [1, 2, 3, 4]:
        mod = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * (i - center) / n)
        a = np.maximum(mod, 0.0)
        s = a.sum()
        if s > 0:
            starts.append(a / s)

    # Single peak (delta-like)
    for pos_frac in [0.3, 0.5, 0.7]:
        a = np.zeros(n)
        pos = int(pos_frac * n)
        width = max(1, n // 20)
        a[max(0, pos - width):min(n, pos + width + 1)] = 1.0
        a = a / a.sum()
        starts.append(a)

    # Two peaks
    for gap in [0.2, 0.35]:
        a = np.zeros(n)
        p1 = int(n * (0.5 - gap))
        p2 = int(n * (0.5 + gap))
        width = max(1, n // 15)
        a[max(0, p1 - width):min(n, p1 + width + 1)] = 1.0
        a[max(0, p2 - width):min(n, p2 + width + 1)] = 1.0
        s = a.sum()
        if s > 0:
            starts.append(a / s)

    # Random (smoothed)
    for _ in range(3):
        raw = rng.random(n)
        # Smooth with a small kernel
        kernel = np.ones(5) / 5.0
        a = np.convolve(raw, kernel, mode='same')
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            starts.append(a / s)

    # Symmetric random
    for _ in range(2):
        half_len = n // 2 + 1
        half = np.abs(rng.normal(1.0, 0.3, half_len))
        a = np.concatenate([half, half[-2::-1]])[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            starts.append(a / s)

    return starts


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def symmetric_lp_alternate(a, max_iters=1000, time_budget=200.0, verbose=False):

    start = time.monotonic()
    stall = 0

    def symmetrize(h):
        full = np.concatenate([h, h[-2::-1]])[:n]
        return normalize_simplex(full)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

            if stall > 6:
                break
            continue

        # Project g onto symmetric subspace
        g_sym = 0.5 * (g_full + g_full[::-1])
        g_sym = normalize_simplex(g_sym)
        g_half = g_sym[:half_n].copy()
        g_half = normalize_simplex(g_half)

        obj_new = compute_obj(symmetrize(g_half))

        if obj_new < best_obj - 1e-13:
            a_half = g_half
            best_half = g_half.copy()

            if verbose and it % 50 == 0:

        else:
            found = False
            for t in [0.3, 0.2, 0.1]:
                blend = normalize_simplex((1-t)*best_half + t*g_half)
                obj_b = compute_obj(symmetrize(blend))
                if obj_b < best_obj - 1e-14:
                    a_half = blend
                    best_half = blend.copy()
                    best_obj = obj_b
                    stall = 0

            if not found:

                    rng = np.random.default_rng((it * 104729 + 13) % (2**31))

    return symmetrize(best_half), best_obj

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
