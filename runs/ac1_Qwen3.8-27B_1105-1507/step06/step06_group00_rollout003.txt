Enhanced optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines log-sum-exp smoothed projected gradient (FISTA), LP alternating,

import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

If `|DFT(a)_k|^2 = C` for all k, then by Parseval: `n * sum_i a_i^2 = n * C`, so `C = sum_i a_i^2`. And the inverse DFT of a constant is a delta function, which means `a * a (circular) = C * delta`. But we want the LINEAR convolution to be flat, not the circular one.

For large n, the difference between linear and circular convolution is small (the edge effects are O(1/n)). So approximately, we want the DFT power spectrum to be flat.

Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothed gradient, LP alternating, simulated annealing,
and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def lse_gradient(a, tau):
    """Compute log-sum-exp smoothed objective value and gradient."""
    n = len(a)
    conv = compute_conv(a)
    
    max_c = float(np.max(conv))
    exp_shifted = np.exp((conv - max_c) / tau)
    sum_exp = float(np.sum(exp_shifted))
    
    lse_val = max_c + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    
    # Gradient via cross-correlation using FFT
    size = 1
    while size < 2 * n:
        size <<= 1
    W = np.fft.fft(weights, size)
    A = np.fft.fft(a, size)
    grad = np.fft.ifft(W * np.conj(A)).real[:n]
    
    return lse_val, grad


def project_simplex(x):
    """Project x onto the probability simplex {p: sum(p)=1, p>=0}."""
    n = len(x)
    if np.all(x <= 0):
        return np.ones(n) / n
    # Sort in decreasing order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(x - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def lse_optimize(a, tau_schedule, n_iters_per_tau, time_budget):
    """
    Optimize using log-sum-exp smoothed projected gradient.
    tau_schedule: list of temperatures (decreasing)
    Returns best a found.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_schedule:
        # Adaptive step size: start with 1/L where L is the Lipschitz constant
        # For the LSE objective, L ≈ 2 * max(a) / tau (rough bound)
        step = tau / (2.0 * float(np.max(a)) + 1e-10)
        step = min(step, 0.5)
        
        y = a.copy()  # FISTA momentum variable
        t_mom = 1.0
        
        for it in range(n_iters_per_tau):
            if time.monotonic() - start > time_budget:
                return best_a
            
            # Compute gradient at y
            lse_val, grad = lse_gradient(y, tau)
            
            # Gradient step
            z = y - step * grad
            
            # Project onto simplex
            x = project_simplex(z)
            
            # Check objective (use the true objective, not the smoothed one)
            obj = compute_obj(x)
            if obj < best_obj:
                best_obj = obj
                best_a = x.copy()
            
            # FISTA momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2)) / 2.0
            y = x + (t_mom - 1.0) / t_new * (x - y)
            t_mom = t_new
            
            # Adaptive step size (Barbera-Borwein style)
            # If objective increased, reduce step
            # If objective decreased significantly, increase step
            if it > 0 and it % 10 == 0:
                current_obj = compute_obj(x)
                if current_obj > best_obj * 1.01:
                    step *= 0.7
                elif current_obj < best_obj * 0.99:
                    step = min(step * 1.2, tau)
        
        a = best_a.copy()
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
