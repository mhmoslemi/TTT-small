import numpy as np
from scipy.fft import rfft, irfft
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n+m-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = fft_convolve(a_norm, a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def smooth_max_gradient(a, beta):
    """Compute gradient of soft-max of conv(a,a) weighted by beta."""
    n = len(a)
    conv = fft_convolve(a, a)  # length 2n-1
    # Softmax weights
    max_conv = np.max(conv)
    exp_conv = np.exp(beta * (conv - max_conv))
    w = exp_conv / np.sum(exp_conv)
    
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is a cross-correlation, computable via FFT
    # sum_j w[i+j] * a[j] for i in [0, 2n-2]
    # Let's compute via convolution:
    # Define b[j] = a[n-1-j] (reverse a)
    # Then sum_j w[i+j] * a[j] = (w * b)[i + n - 1]
    # where * denotes full convolution
    b = a[::-1]
    wc = fft_convolve(w, b)  # length (2n-1) + n - 1 = 3n-2
    # We want positions i + n - 1 for i in [0, n-1]
    # So we want wc[n-1 : n-1+n] = wc[n-1 : 2n-1]
    grad = 2.0 * wc[n-1:2*n-1]
    
    return grad, conv, w

def mirror_descent_step(a, grad, eta):
    """Multiplicative weight update."""
    log_a = np.log(a + 1e-300) - eta * grad
    # softmax for numerical stability
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    s = np.sum(a_new)
    if s > 1e-300:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

"""
Hybrid FFT-based mirror descent + LP alternating optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses log-sum-exp smooth max with beta annealing,
multiplicative weight updates via FFT, and final LP polish for precision.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = fft_convolve(a_norm, a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def compute_conv(a):
    """Full auto-convolution."""
    return fft_convolve(a, a)


def smooth_max_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv_k(a))).
    Returns (grad, conv, smooth_val).
    """
    n = len(a)
    conv = fft_convolve(a, a)  # length 2n-1
    max_c = np.max(conv)
    exp_c = np.exp(beta * (conv - max_c))
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    smooth_val = (max_c + np.log(sum_exp)) / beta

    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = 2 * (w * a_rev)[i + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    wc = fft_convolve(w, a_rev)  # length (2n-1) + n - 1 = 3n-2
    grad = 2.0 * wc[n - 1:2 * n - 1]

    return grad, conv, smooth_val

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak_fast(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS for speed.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Use scipy linprog with HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    # Fallback to cvxpy
    try:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.HIGHS, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    return None, None

def subgradient_refine(a, time_budget=30.0):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        lp_budget = min(remaining * 0.25, 120.0)
        a_lp = lp_alternating_fast(a_current, time_budget=lp_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_current = a_lp.copy()
            print(f"  LP result: obj={best_score:.8f}")
        else:
            print(f"  LP result: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: More mirror descent rounds ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Additional MD Rounds ===")
        for round_i in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.15, 50.0)
            a_md2 = mirror_descent_optimize(a_current, time_budget=alloc)
            obj_md2 = compute_obj(a_md2)
            if obj_md2 < best_score:
                best_score = obj_md2
                best_seq = a_md2.copy()
                a_current = a_md2.copy()
                print(f"  MD round {round_i}: obj={best_score:.8f}")
            else:
                print(f"  MD round {round_i}: obj={obj_md2:.8f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Multi-n Exploration ===")
        n_options = sorted(set([
            n_parent - 50, n_parent - 25, n_parent,
            n_parent + 25, n_parent + 50, n_parent + 100,
            n_parent + 200
        ]))
        n_options = [n for n in n_options if 300 <= n <= 1500]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if n_try == n_parent:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)

            alloc = min(remaining * 0.2, 40.0)
            a_opt = mirror_descent_optimize(a_try, time_budget=alloc)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                n_parent = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Random Restarts ===")
        n_best = len(best_seq)
        
        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            init_type = r % 3
            if init_type == 0:
                # Perturbed best
                a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
                a_r = np.maximum(a_r, 0)
                s = np.sum(a_r)
                if s > 0:
                    a_r = a_r / s
            elif init_type == 1:
                # Cosine window
                x = np.linspace(0, np.pi, n_best)
                a_r = np.cos(x) ** 2 + 0.3
                a_r = a_r / np.sum(a_r)
            else:
                # Different n with perturbation
                n_r = n_best + int(rng.integers(-100, 100))
                n_r = max(400, min(1200, n_r))
                a_r = resample_sequence(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
                a_r = np.maximum(a_r, 0)
                s = np.sum(a_r)
                if s > 0:
                    a_r = a_r / s

            alloc = min(remaining * 0.15, 35.0)
            a_opt = mirror_descent_optimize(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                n_parent = len(a_opt)
                print(f"  Restart {r} (type={init_type}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (type={init_type}): obj={obj_r:.8f}")

    # === Phase 7: Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 7: Final LP Polish ===")
        lp_alloc = min(remaining * 0.4, 60.0)
        a_final = lp_alternating_fast(best_seq, time_budget=lp_alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    # === Phase 8: One last mirror descent + subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        md_alloc = min(remaining * 0.4, 30.0)
        a_last = mirror_descent_optimize(best_seq, time_budget=md_alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final MD: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            sg_alloc = min(remaining * 0.4, 15.0)
            a_sg2 = subgradient_refine(best_seq, time_budget=sg_alloc)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  Final SG: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
