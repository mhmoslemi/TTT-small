```python
import numpy as np

def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def soft_max_conv(a, tau):
    """Compute soft-max of convolution with temperature tau."""
    conv = compute_conv_fft(a)
    cmax = np.max(conv)
    shifted = (conv - cmax) / tau
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    log_sum = cmax + tau * np.log(np.sum(np.exp(shifted)))
    return log_sum, conv


def softmax_params_to_a(x):
    """Convert softmax parameters to probability vector."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x)


def compute_corr_weights(W, a, fft_size=None):
    """
    Compute corr[i] = sum_k W[k] * a[k-i] for i = 0..n-1.
    This is the cross-correlation of W (length 2n-1) and a (length n) at lags 0..n-1.
    """
    n = len(a)
    if fft_size is None:
        fft_size = 1
        while fft_size < 3 * n - 1:
            fft_size <<= 1
    
    W_f = rfft(W, fft_size)
    a_f = rfft(a, fft_size)
    corr = irfft(W_f * np.conj(a_f), fft_size)
    return corr[:n]


def soft_max_gradient_x(x, tau, fft_size=None):
    """
    Compute gradient of soft_max(conv(softmax(x), softmax(x)), tau) with respect to x.
    Returns (value, gradient).
    """
    n = len(x)
    a = softmax_params_to_a(x)
    conv = compute_conv_fft(a)
    cmax = np.max(conv)
    
    # Compute weights
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500, 0)
    W = np.exp(shifted)
    W_sum = np.sum(W)
    W = W / W_sum
    
    # Soft-max value
    S = cmax + tau * np.log(W_sum)
    
    # Compute correlation
    corr = compute_corr_weights(W, a, fft_size)
    
    # Compute gradient: dS/dx[i] = 2*a[i]*(corr[i] - cw)
    cw = np.sum(a * corr)
    grad = 2.0 * a * (corr - cw)
    
    return S, grad


def softmax_lbfgs(x0, tau, max_iter=200, time_budget=30.0, tol=1e-10):
    """Run L-BFGS on the softmax-parameterized soft-max objective."""

        val, grad = soft_max_gradient_x(x, tau, fft_size)

    res = minimize(fun_and_grad, x0, method='L-BFGS-B',
                   options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8,
                            'maxls': 50, 'disp': False})
    
    return res.x

def build_toeplitz(a):
    """Build Toeplitz matrix for LP."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def spectral_flatten(a, max_iters=40, time_budget=30.0):
    """Frequency-domain flattening via phase-preserving magnitude processing."""

    a = normalize(a.copy())

        a_new = np.real(ifft(A_new, size))[:n]

        if obj_new < best_obj:

def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
```

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses softmax-parameterized L-BFGS with annealed soft-max objective as the main optimizer,
combined with LP polishing, spectral flattening, coordinate descent, multi-n exploration,
and diverse random restarts.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


# Global state for the objective wrapper (to avoid closures)
_opt_tau = 1.0
_opt_fft_size = 2048
_opt_start_time = 0.0
_opt_time_budget = 0.0


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_params_to_a(x):
    """Convert log-space parameters to probability vector via softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    total = np.sum(exp_x)
    if total < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / total


def compute_corr_weights(W, a, fft_size):
    """
    Compute corr[i] = sum_j W[i+j] * a[j] for i = 0..n-1 via FFT cross-correlation.
    W has length 2n-1, a has length n.
    """
    n = len(a)
    W_f = rfft(W, fft_size)
    a_f = rfft(a, fft_size)
    corr = irfft(W_f * np.conj(a_f), fft_size)
    return corr[:n]


def soft_max_gradient_x(x, tau, fft_size):
    """
    Compute value and gradient of soft_max(conv(softmax(x), softmax(x)), tau) w.r.t. x.
    Returns (value, gradient_vector).
    """
    n = len(x)
    a = softmax_params_to_a(x)
    conv = compute_conv_fft(a)
    cmax = float(np.max(conv))

    # Compute normalized softmax weights over conv peaks
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500.0, 0.0)
    W = np.exp(shifted)
    W_sum = float(np.sum(W))
    if W_sum < 1e-300:
        W = np.zeros_like(conv)
        W[int(np.argmax(conv))] = 1.0
    else:
        W /= W_sum

    # Soft-max value
    S = cmax + tau * np.log(max(W_sum, 1e-300))

    # Compute cross-correlation corr[i] = sum_j W[i+j] * a[j]
    corr = compute_corr_weights(W, a, fft_size)

    # Gradient: dS/dx[i] = 2*a[i]*(corr[i] - cw)
    cw = float(np.sum(a * corr))
    grad = 2.0 * a * (corr - cw)

    return S, grad


def _obj_grad_wrapper(x):
    """Top-level wrapper for scipy minimize (avoids closures)."""
    if time.monotonic() - _opt_start_time > _opt_time_budget:
        return 1e10, np.zeros_like(x)
    return soft_max_gradient_x(x, _opt_tau, _opt_fft_size)


def softmax_lbfgs(x0, tau, max_iter=200, time_budget=30.0):
    """Run L-BFGS on the softmax-parameterized soft-max objective."""
    n = len(x0)
    fft_size = 1
    while fft_size < 2 * n:
        fft_size <<= 1

    # Set global state
    global _opt_tau, _opt_fft_size, _opt_start_time, _opt_time_budget
    _opt_tau = tau
    _opt_fft_size = fft_size
    _opt_start_time = time.monotonic()
    _opt_time_budget = time_budget

    try:
        res = minimize(_obj_grad_wrapper, x0, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-9,
                                'maxls': 50, 'disp': False})
        return res.x
    except Exception:
        return x0.copy()

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_polish(a, max_iters=20, time_budget=60.0):
    """LP-based polishing: alternate LP direction finding with line search."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = compute_conv_fft(a)
        M = float(np.max(conv))

        A_mat = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True,
                                   'time_limit': max(2, int(remaining * 0.4)),
                                   'dual_feasibility_tolerance': 1e-10,
                                   'primal_feasibility_tolerance': 1e-10})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g
        except Exception:
            break

        # Line search for blend parameter
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize((1.0 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0 and best_t_obj < best_obj:
            a = normalize((1.0 - best_t) * a + best_t * g)
            best_obj = best_t_obj
            best_a = a.copy()
        else:
            rng = np.random.default_rng((it * 7919 + 12345) % (2**31 - 1))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03 + 0.01 * (it % 3), n)))

    return best_a

                for _ in range(2):
                    tgt = int(rng.integers(0, n))
                    if tgt != j1 and tgt != j2:
                        a_try[tgt] += delta
                        break
                else:
                    # Fallback: spread to center
                    mid = n // 2
                    if mid != j1 and mid != j2:
                        a_try[mid] += delta

                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)

                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            if rng.random() < 0.15:
                a = normalize(a * np.exp(rng.normal(0, 0.02, n)))
            else:
                break

    return best_a


def spectral_flatten(a, max_iters=40, time_budget=30.0):
    """Frequency-domain magnitude flattening with phase preservation."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        alpha = 0.3 + 0.5 * (it / float(max_iters))
        P_new = (P / p_max) ** alpha * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def sparsify_and_reopt(a, frac_keep, time_budget=20.0):
    """Zero out the smallest fraction of elements, then re-optimize."""
    n = len(a)
    a = normalize(a.copy())
    threshold_idx = int(n * (1.0 - frac_keep))
    if threshold_idx < n // 4:
        threshold_idx = n // 4

    sorted_idx = np.argsort(a)
    a[sorted_idx[:threshold_idx]] = 0.0
    a = normalize(a)

    conv = compute_conv_fft(a)
    cmax = float(np.max(conv))
    tau_sched = [cmax * 0.3, cmax * 0.1, cmax / 500]
    a = run_annealing(a, tau_sched, max_iter_per_tau=40, time_budget=time_budget)
    return a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining softmax L-BFGS with annealed
    soft-max, LP polishing, spectral methods, multi-n exploration, restarts,
    and sparsification.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    n_base = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_base}, score={best_score:.8f}, deadline in {budget_s - 15:.0f}s")

    # ============================================================
    # Phase 1: Primary annealing (softmax L-BFGS)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Primary annealing (n={n_base})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))

        # Geometric tau schedule from broad to focused
        tau_schedule = []
        tau = cmax * 0.8
        while tau > cmax / 200:
            tau_schedule.append(tau)
            tau *= 0.65
        tau_schedule.append(cmax / 500)
        tau_schedule.append(cmax / 2000)

        alloc = min(280.0, (deadline - start_time) * 0.25)
        a_new = run_annealing(a, tau_schedule, max_iter_per_tau=100, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Primary annealing: {best_score:.8f}")
        else:
            print(f"  Primary annealing: {obj_new:.8f} (no improvement)")

    # ============================================================
    # Phase 2: LP polishing rounds
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP polishing")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        n_a = len(a)

        for lp_round in range(6):
            if time.monotonic() > deadline - 500:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(40.0, remaining * 0.06)

            if n_a <= 1000:
                a_lp = lp_polish(a, max_iters=12, time_budget=alloc)
            else:
                a_lp = a

            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a = a_lp
                print(f"  LP round {lp_round}: {best_score:.8f}")
            else:
                print(f"  LP round {lp_round}: {obj_lp:.8f} (best={best_score:.8f})")
                rng = np.random.default_rng(lp_round * 137 + 42)
                a = normalize(a_lp * np.exp(rng.normal(0, 0.025, len(a_lp))))

    # ============================================================
    # Phase 3: Second annealing pass (finer tau)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Second annealing + spectral")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))

        if time.monotonic() < deadline - 350:
            tau_fine = list(np.geomspace(cmax * 0.15, cmax / 3000, 12))
            a_new = run_annealing(a, tau_fine, max_iter_per_tau=80, time_budget=120.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Second annealing: {best_score:.8f}")
            else:
                print(f"  Second annealing: {obj_new:.8f}")

        if time.monotonic() < deadline - 300:
            a_spec = spectral_flatten(normalize(best_seq.copy()), max_iters=35, time_budget=25.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                print(f"  Spectral: {best_score:.8f}")
            else:
                print(f"  Spectral: {obj_spec:.8f}")

        # Follow-up LP after spectral
        if time.monotonic() < deadline - 280:
            a_lp = lp_polish(normalize(best_seq.copy()), max_iters=10, time_budget=35.0)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Post-spectral LP: {best_score:.8f}")

    # ============================================================
    # Phase 4: Multi-n exploration
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")

    n_cur = len(best_seq) if best_seq is not None else n_base
    n_options = sorted(set([
        max(150, n_cur - 200), max(150, n_cur - 100), max(150, n_cur - 50),
        max(150, n_cur - 25),
        n_cur,
        n_cur + 25, n_cur + 50, n_cur + 100, n_cur + 150, n_cur + 200, n_cur + 300
    ]))
    n_options = [n for n in n_options if 150 <= n <= 1500]

    for n_try in n_options:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if best_seq is not None and n_try != len(best_seq):
            a_try = normalize(resample_sequence(best_seq, n_try))
        elif best_seq is not None:
            a_try = normalize(best_seq.copy())
        else:
            a_try = np.ones(n_try) / n_try

        conv = compute_conv_fft(a_try)
        cmax = float(np.max(conv))
        tau_sched = [cmax * 0.6, cmax * 0.25, cmax * 0.08, cmax / 500, cmax / 2000]

        alloc = min(35.0, remaining * 0.12)
        a_try = run_annealing(a_try, tau_sched, max_iter_per_tau=60, time_budget=alloc)

        if n_try <= 900 and remaining > 50:
            a_try = lp_polish(a_try, max_iters=5, time_budget=12.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # ============================================================
    # Phase 5: Sparsification attempts
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Sparsification")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 100:
        for frac in [0.7, 0.5, 0.3]:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_sparse = sparsify_and_reopt(best_seq, frac, time_budget=min(25, remaining * 0.3))
            obj_sparse = compute_obj(a_sparse)
            if obj_sparse < best_score:
                best_score = obj_sparse
                best_seq = a_sparse.copy()
                print(f"  Sparsify keep={frac}: {best_score:.8f} **")
            else:
                print(f"  Sparsify keep={frac}: {obj_sparse:.8f}")

    # ============================================================
    # Phase 6: Random restarts with diverse strategies
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        rng = np.random.default_rng(seed)

        for restart in range(12):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(25.0, remaining * 0.08)

            # Diverse perturbation strategies
            r = restart % 6
            if r == 0:
                a_try = best_seq * np.exp(rng.normal(0, 0.05, n_final))
            elif r == 1:
                freq = rng.uniform(0.5, 5.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.12 * np.sin(2 * np.pi * freq * np.arange(n_final) / n_final + phase)
                a_try = best_seq * mod
            elif r == 2:
                a_rand = rng.dirichlet(np.ones(n_final) * 0.3)
                mix = rng.uniform(0.15, 0.4)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_rand
            elif r == 3:
                a_try = best_seq.copy()
                blen = int(rng.integers(n_final // 5, n_final // 2))
                bstart = int(rng.integers(0, n_final - blen))
                a_try[bstart:bstart + blen] = a_try[bstart:bstart + blen][::-1]
            elif r == 4:
                # Random frequency modulation
                freqs = rng.uniform(1, 8, size=3)
                mods = np.ones(n_final)
                for f in freqs:
                    mods += rng.uniform(0.03, 0.08) * np.sin(2 * np.pi * f * np.arange(n_final) / n_final)
                a_try = best_seq * mods
            else:
                # Random spikes
                a_try = best_seq.copy()
                n_spikes = int(rng.integers(5, 20))
                spike_targets = rng.choice(n_final, size=n_spikes, replace=False)
                for st in spike_targets:
                    a_try[st] *= rng.uniform(1.5, 4.0)

            a_try = normalize(a_try)

            conv = compute_conv_fft(a_try)
            cmax = float(np.max(conv))
            tau_sched = [cmax * 0.4, cmax * 0.1, cmax / 500]
            a_try = run_annealing(a_try, tau_sched, max_iter_per_tau=50, time_budget=alloc)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")

    # ============================================================
    # Phase 7: Final intensive polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = len(a_final)

        for polish in range(6):
            if time.monotonic() > deadline - 3:
                break
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            alloc = min(remaining * 0.4, 22.0)
            if alloc < 3:
                break

            # Fine annealing
            conv = compute_conv_fft(a_final)
            cmax = float(np.max(conv))
            tau_very_fine = list(np.geomspace(cmax / 50, cmax / 10000, 6))
            a_pol = run_annealing(a_final, tau_very_fine, max_iter_per_tau=60, time_budget=alloc * 0.35)

            # LP polish
            if n_final <= 1000 and alloc > 8:
                a_pol = lp_polish(a_pol, max_iters=8, time_budget=alloc * 0.35)

            # Coordinate descent
            if alloc > 4:
                a_pol = coordinate_descent_peak(a_pol, time_budget=min(6, alloc * 0.2))

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {polish}: {best_score:.8f}")
            else:
                a_final = a_pol
                print(f"  Polish {polish}: {obj_pol:.8f} (best={best_score:.8f})")

    # ============================================================
    # Final validation and return
    # ============================================================
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
```
