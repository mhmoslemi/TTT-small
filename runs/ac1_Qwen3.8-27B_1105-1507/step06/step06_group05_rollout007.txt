"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses IRLS peak suppression, tight LP alternating, exact 2-variable coordinate descent,
simulated annealing for escape, and multi-n exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_max_sum(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def irls_peak_suppression(a, max_iters=200, time_budget=60.0):
    """
    Iterative Reweighted Least Squares: iteratively identify peaks in convolution,
    compute gradient toward reducing them, and update with multiplicative weights.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    eta = 0.05  # initial step size
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Find top peaks
        n_peaks = max(1, min(n // 2, int(0.1 * n)))
        peak_vals = np.sort(conv)[::-1]
        threshold = peak_vals[min(n_peaks, len(peak_vals) - 1)]
        
        # Build weight vector: higher weight for higher peaks
        # Use exponential weighting
        alpha = 5.0 + it * 0.1  # increasing sharpness
        weights = np.exp(alpha * (conv - cmax) / max(cmax, 1e-10))
        weights = np.maximum(weights, 1e-10)
        weights /= np.sum(weights)
        
        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        # Multiplicative update (mirror descent with KL divergence)
        # a_new[i] = a[i] * exp(-eta * grad[i])
        # Then renormalize
        log_update = -eta * (grad - np.mean(grad)) / max(np.max(np.abs(grad)), 1e-10)
        log_update = np.clip(log_update, -2.0, 2.0)
        a_new = a * np.exp(log_update)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.05, 0.5)
        else:
            eta *= 0.7
            if eta < 1e-5:
                break
    
    return best_a

def exact_2var_descent(a, time_budget=30.0):
    """
    For each convolution peak, find the pair (i, j) with i+j = peak_idx that
    contributes most, and optimize the mass transfer between them exactly.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find contributors to peak at k_star
        # conv[k_star] = sum_j a[j] * a[k_star - j] where 0 <= j < n and 0 <= k_star - j < n
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        
        if j_min >= j_max:
            break
        
        # Compute contributions
        js = np.arange(j_min, j_max + 1)
        contributions = a[js] * a[k_star - js]
        
        # Pick the top contributor
        top_idx = int(np.argmax(contributions))
        j1 = js[top_idx]
        j2 = k_star - j1
        
        if j1 == j2:
            # Can't transfer to self, pick second best
            contributions[top_idx] = -np.inf
            if contributions.max() < 0:
                break
            top_idx = int(np.argmax(contributions))
            j1 = js[top_idx]
            j2 = k_star - j1
        
        # Optimize: move delta from j1 to j2 (or vice versa)
        # conv[k_star] changes by: -a[j2]*delta + a[j1]*(1) ... actually complex
        # Let's just try a small grid
        a_orig = a.copy()
        a1 = a1_val = a[j1]
        a2 = a[j2]
        
        best_delta = 0
        best_local_obj = compute_obj(a)
        
        for delta_frac in np.linspace(-0.3, 0.3, 13):
            delta = delta_frac * max(a1, a2)
            a_try = a.copy()
            a_try[j1] = max(0, a1 - delta)
            a_try[j2] = a2 + delta
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_delta = delta
        
        if best_delta != 0:
            a[j1] = max(0, a1 - best_delta)
            a[j2] = a2 + best_delta
            a = normalize(a)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        else:
            # Try random perturbation
            idx = int(rng.integers(0, n))
            a_try = a.copy()
            a_try[idx] *= 1.0 + rng.normal(0, 0.1)
            a_try = np.maximum(a_try, 0)
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj * 1.01:  # accept slightly worse (annealing)
                a = a_try
    
    return best_a

def lp_tight_alternate(a, max_iters=50, time_budget=120.0):
    """
    LP alternating with tight M: solve LP with M slightly below current peak
    to force aggressive redistribution. Multiple M values tried.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try multiple M values: current peak, and slightly below
        m_values = [M, M * 0.99, M * 0.98]
        
        best_g = None
        for m_val in m_values:
            g = solve_lp_max_sum(a, m_val, time_limit=max(5, int(remaining * 0.3)))
            if g is not None:
                best_g = g
                break
        
        if best_g is None:
            # Fallback: use subgradient direction
            k_star = int(np.argmax(conv))
            grad = np.zeros(n)
            for j in range(n):
                k2 = k_star - j
                if 0 <= k2 < n:
                    grad[j] = a[k2]
            grad_centered = grad - np.mean(grad)
            a_try = normalize(a - 0.01 * grad_centered)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
            continue
        
        # Try multiple blend factors
        best_blend_obj = np.inf
        best_blend_a = None
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = normalize((1 - t) * a + t * best_g)
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try
        
        if best_blend_a is not None and best_blend_obj < best_obj:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            a = best_blend_a
    
    return best_a

def simulated_annealing_polish(a, time_budget=30.0):
    """
    Simulated annealing: accept worse solutions with probability exp(-delta/T).
    Use small random perturbations.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    T = 0.005  # initial temperature
    T_min = 1e-5
    
    while time.monotonic() - start < time_budget:
        # Generate perturbation
        scale = 0.02 + 0.03 * (T / 0.005)
        a_try = a * np.exp(rng.normal(0, scale, n))
        a_try = normalize(a_try)
        
        obj_try = compute_obj(a_try)
        delta = obj_try - best_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            a = a_try
            
        T *= 0.995
        if T < T_min:
            T = T_min
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_conv(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0..2n-2."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def golden_section_search(f, lo, hi, max_iter=20):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 0.01:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0

def irls_peak_suppression(a, max_iters=300, time_budget=60.0):
    """
    IRLS: iteratively find dominant peaks, compute weighted gradient,
    apply multiplicative (mirror descent) update.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 3 * n:
        size <<= 1
    
    eta = 0.08
    sharpness = 3.0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        cmin_nonzero = float(np.max(conv[conv > 1e-10])) if np.any(conv > 1e-10) else cmax
        
        # Exponential weighting of peaks
        sharpness = 3.0 + it * 0.05
        weights = np.exp(sharpness * (conv - cmax) / max(cmax * 0.5, 1e-10))
        weights = np.maximum(weights, 1e-10)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient via cross-correlation
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        # Mirror descent (multiplicative update)
        grad_centered = grad - np.mean(grad)
        scale = np.max(np.abs(grad_centered))
        if scale < 1e-15:
            break
        log_update = np.clip(-eta * grad_centered / scale, -1.5, 1.5)
        a_new = a * np.exp(log_update)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new

            eta = min(eta * 1.03, 0.8)

            if eta < 1e-5:

def subgradient_with_momentum(a, max_steps=500, time_budget=30.0):
    """
    Subgradient descent with momentum on the multiplicative (log) scale.
    Targets the current argmax of convolution.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.05
    
    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient at argmax
        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        for j in range(j_min, j_max + 1):
            grad[j] = a[k_star - j]
        
        # Center gradient
        grad_centered = grad - np.mean(grad)
        momentum = beta * momentum + (1 - beta) * grad_centered
        
        # Multiplicative update on log scale
        scale = np.max(np.abs(momentum))
        if scale < 1e-15:

            best_obj = obj_new

            a = a_new
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.9
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

    if best_seq is not None:
        a = normalize(best_seq.copy())
        
        for cycle in range(8):
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.22, 100.0)
            
            a_new = lp_dual_alternate(a, max_iters=40, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  LP cycle {cycle}: {best_score:.8f} **")
            else:
                a = a_new
                print(f"  LP cycle {cycle}: {obj_new:.8f}")
    
    # Phase 2: IRLS + subgradient with momentum
    print(f"\n{'='*60}")
    print(f"Phase 2: IRLS + subgradient momentum")
    print(f"{'='*60}")
    
    if best_seq is not None:
        a = normalize(best_seq.copy())
        
        if time.monotonic() < deadline - 150:
            a_irls = irls_peak_suppression(a, max_iters=300, time_budget=50.0)
            obj_irls = compute_obj(a_irls)
            if obj_irls < best_score:
                best_score = obj_irls
                best_seq = a_irls.copy()
                a = a_irls
                print(f"  IRLS: {best_score:.8f} **")
            else:
                print(f"  IRLS: {obj_irls:.8f}")
        
        if time.monotonic() < deadline - 130:
            a_sub = subgradient_with_momentum(a, max_steps=500, time_budget=35.0)
            obj_sub = compute_obj(a_sub)
            if obj_sub < best_score:
                best_score = obj_sub
                best_seq = a_sub.copy()
                a = a_sub
                print(f"  Subgrad momentum: {best_score:.8f} **")
            else:
                print(f"  Subgrad momentum: {obj_sub:.8f}")
        
        # Follow up with LP
        if time.monotonic() < deadline - 100:
            a_new = lp_dual_alternate(a, max_iters=30, time_budget=60.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Post-Phase2 LP: {best_score:.8f} **")
            else:
                print(f"  Post-Phase2 LP: {obj_new:.8f}")
    
    # Phase 3: Exact 2-variable coordinate descent
    print(f"\n{'='*60}")
    print(f"Phase 3: 2-variable coordinate descent")
    print(f"{'='*60}")
    
    if best_seq is not None:
        a = normalize(best_seq.copy())
        a_2v = exact_2var_descent(a, time_budget=40.0)
        obj_2v = compute_obj(a_2v)
        if obj_2v < best_score:
            best_score = obj_2v
            best_seq = a_2v.copy()
            print(f"  2-var: {best_score:.8f} **")
        else:
            print(f"  2-var: {obj_2v:.8f}")
    
    # Phase 4: Multi-n exploration
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-n exploration")
    print(f"{'='*60}")
    
    n_best = len(best_seq) if best_seq is not None else 700
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best - 25,
        n_best, n_best + 25, n_best + 50, n_best + 100,
        n_best + 150, n_best + 200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1100]
    
    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        if best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = np.ones(n_try) / n_try
        
        alloc = min(remaining * 0.1, 35.0)
        
        # IRLS at this n
        if alloc > 10:
            a_try = irls_peak_suppression(a_try, max_iters=100, time_budget=alloc * 0.4)
        
        # LP if n manageable
        if n_try <= 900 and alloc > 15:
            a_try = lp_dual_alternate(a_try, max_iters=15, time_budget=alloc * 0.45)
        
        # 2-var polish
        if alloc > 20:
            a_try = exact_2var_descent(a_try, time_budget=min(10, alloc * 0.15))
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")
    
    # Phase 5: Random restarts
    print(f"\n{'='*60}")
    print(f"Phase 5: Random restarts")
    print(f"{'='*60}")
    
    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(seed)
        
        for restart in range(8):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.12, 30.0)
            
            if restart % 4 == 0:
                a_try = normalize(best_seq * np.exp(rng.normal(0, 0.06, n_cur)))
            elif restart % 4 == 1:
                freq = rng.uniform(0.5, 6.0)
                x = np.arange(n_cur) / n_cur
                mod = 1.0 + 0.12 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
                a_try = normalize(best_seq * mod)
            elif restart % 4 == 2:
                a_rand = rng.dirichlet(np.ones(n_cur) * 0.3)
                mix = rng.uniform(0.15, 0.4)
                a_try = normalize((1 - mix) * normalize(best_seq) + mix * a_rand)
            else:
                a_try = best_seq.copy()
                blen = int(rng.integers(max(1, n_cur // 5), max(2, n_cur // 2)))
                s_idx = int(rng.integers(0, max(1, n_cur - blen)))
                a_try[s_idx:s_idx + blen] = a_try[s_idx:s_idx + blen][::-1]
                a_try = normalize(a_try)
            
            if n_cur <= 900 and alloc > 15:
                a_try = lp_dual_alternate(a_try, max_iters=10, time_budget=alloc * 0.4)
            if alloc > 10:
                a_try = irls_peak_suppression(a_try, max_iters=80, time_budget=alloc * 0.35)
            
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart}: {best_score:.8f} **")
            else:
                print(f"  Restart {restart}: {obj_try:.8f}")
    
    # Phase 6: Final intensive polish
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")
    
    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        
        for rnd in range(4):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.3, 30.0)
            
            if len(a_final) <= 900 and alloc > 15:
                a_pol = lp_dual_alternate(a_final, max_iters=20, time_budget=alloc * 0.45)
            else:
                a_pol = a_final.copy()
            
            if alloc > 10:
                a_pol = irls_peak_suppression(a_pol, max_iters=150, time_budget=alloc * 0.3)
            
            if alloc > 15:
                a_pol = exact_2var_descent(a_pol, time_budget=min(10, alloc * 0.2))
            
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {rnd}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {rnd}: {obj_pol:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100
    
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    
    return result
