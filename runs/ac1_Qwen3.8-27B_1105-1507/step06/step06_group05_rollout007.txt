def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):

    cmax = float(np.max(conv))

        return 0.0
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    return cmax + tau * float(np.log(np.sum(np.exp(shifted))))


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

                      options={'presolve': True, 'time_limit': time_limit})

            g = np.maximum(res.x, 0.0)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key strategies: Adam optimizer with log-sum-exp annealing, LP alternating with
warm starts, systematic multi-n exploration, population-based diversification,
and intensive final polishing with alternating methods.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def adam_lse_optimize(a, max_steps=500, time_budget=60.0, tau_init=0.01, tau_final=0.001,
                      lr_init=0.05, seed_offset=0):
    """Adam optimizer on log-sum-exp approximation of max(conv)."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    step_count = 0
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    no_improve_count = 0

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for _ in range(max_steps):
        if time.monotonic() - start > time_budget - 1:
            break

        progress = min(step_count / max_steps, 1.0)
        tau = tau_init * (tau_final / tau_init) ** progress

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        if cmax < 1e-30:
            break

        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
            w_sum = 1.0
        else:
            weights /= w_sum

        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)

        step_count += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad_centered
        v_buf = beta2 * v_buf + (1 - beta2) * grad_centered ** 2
        m_hat = m_buf / (1 - beta1 ** step_count)
        v_hat = v_buf / (1 - beta2 ** step_count)

        a_try = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_try = np.maximum(a_try, 0.0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = a.copy()
        else:
            a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count % 10 == 0:
                lr *= 0.85
            if no_improve_count > 100:
                lr = lr_init * 0.3
                m_buf *= 0.5
                v_buf *= 0.5
                no_improve_count = 0

    return best_a


def lp_alternate(a, max_iters=30, time_budget=60.0, blend_candidates=(0.3, 0.5, 0.7, 0.9, 1.0)):
    """LP alternating with multiple blend factor trials."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        lp_time = max(3, int(remaining * 0.45))
        g = solve_lp_max_sum(a, M, time_limit=lp_time)
        if g is None:
            break

        best_t = 1.0
        best_try_obj = compute_obj(normalize(g))
        for t in blend_candidates:
            a_try = normalize((1 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_try_obj:
                best_try_obj = obj_try
                best_t = t

        a_new = normalize((1 - best_t) * a + best_t * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            rng = np.random.default_rng((it * 6271 + 41) % (2 ** 31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.02 + 0.01 * (it % 3), n)))

    return best_a


def subgradient_averaging(a, n_steps=200, time_budget=30.0):
    """Polyak average of subgradient iterates for max(conv) minimization."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    avg = np.zeros(n)
    start = time.monotonic()
    t0 = 1.0

    for t in range(1, n_steps + 1):
        if time.monotonic() - start > time_budget - 1:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient at the peak position
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            k1 = k_star - i
            if 0 <= k1 < n:
                val += a[k1]
            k2 = i - k_star
            if 0 <= k2 < n:
                val += a[k2]
            grad[i] = val

        grad_centered = grad - np.mean(grad)
        step_size = t0 / (t + t0)

        a_try = a - step_size * grad_centered
        a_try = np.maximum(a_try, 0.0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a = a_try / s_try
        avg += a

    avg /= min(n_steps, t)
    return normalize(avg)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def diversify(a, method, rng):
    """Apply a diversification method to sequence a."""
    n = len(a)
    if method == 0:
        return a * np.exp(rng.normal(0, 0.06, n))
    elif method == 1:
        freq = rng.uniform(1.0, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.15 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        return a * mod
    elif method == 2:
        a_rand = rng.dirichlet(np.ones(n) * 0.3)
        mix = rng.uniform(0.1, 0.4)
        return (1 - mix) * normalize(a) + mix * a_rand
    elif method == 3:
        a_try = a.copy()
        block_len = max(5, int(rng.integers(n // 5, n // 2)))
        start_idx = int(rng.integers(0, max(1, n - block_len)))
        a_try[start_idx:start_idx + block_len] = a_try[start_idx:start_idx + block_len][::-1]
        return a_try
    elif method == 4:
        # Frequency domain phase perturbation
        size = 1
        while size < 2 * n:
            size <<= 1
        A = fft(a, size)
        phase_noise = rng.normal(0, 0.08, size)
        A = A * np.exp(1j * phase_noise)
        a_try = np.real(ifft(A, size))[:n]
        return a_try
    else:
        return a * np.exp(rng.normal(0, 0.04, n))
