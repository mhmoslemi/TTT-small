```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov smooth-max gradient descent for fast multi-length initialization
2. Constraint-pruned LP mirror descent with adaptive step sizes and active-set management
3. Periodic active-set updates and random perturbation for escape from plateaus
4. Multi-length exploration, resampling, and final fine-grained polish
Key improvements over baseline: smarter LP constraint selection, cached sparse matrices,
more aggressive step-size search, and structured time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[js] += 2.0 * p[k] * a[k - js]
    return grad, b, b_max


def run_nesterov(a_init, deadline, max_iters=2000, tau_start=1.0, tau_end=300.0,
                 lr_init=0.5, lr_end=0.001):
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    a_prev = a.copy()
    t_val = 1.0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = (it + 1) / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = (t_val - 1.0) / t_new
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1.0 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a_prev = a
        a = a_new
        t_val = t_new
    return best_a


def build_lp_sparse(n, a, lags):
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(lags), n))
    return A


class LPRefiner:
    def __init__(self, n, top_frac=0.12):
        self.n = n
        self.top_frac = top_frac
        self.n_constraints = max(30, int((2 * n - 1) * top_frac))
        self.current_lags = None
        self.current_A = None
        self.iter_count = 0

    def update_active_set(self, a):
        b = fast_conv(a)
        n = len(a)
        top_indices = np.argsort(b)[::-1][:self.n_constraints]
        for mandatory in [0, n - 1, 2 * n - 2]:
            if mandatory not in top_indices:
                top_indices = np.append(top_indices, mandatory)
        self.current_lags = sorted(set(top_indices.tolist()))
        self.current_A = build_lp_sparse(n, a, self.current_lags)
        self.iter_count = 0

    def solve_lp(self, M):
        if self.current_A is None:
            return None
        n = self.n
        c = -np.ones(n)
        b_ub = np.full(len(self.current_lags), M)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=self.current_A, b_ub=b_ub, bounds=bounds,
                        method='highs',
                        options={'presolve': True,
                                 'dual_feasibility_tolerance': 1e-9,
                                 'primal_feasibility_tolerance': 1e-9})
            if r.success and r.x is not None and r.x.sum() > 1e-15:
                return np.maximum(r.x, 0)
        except Exception:
            pass
        return None


def lp_mirror_descent(a_init, deadline, max_iters=5000, top_frac=0.12,
                      update_every=15, steps_grid=None):
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    refiner = LPRefiner(n, top_frac)
    refiner.update_active_set(a)
    if steps_grid is None:
        steps_grid = [0.4, 0.3, 0.25, 0.2, 0.15, 0.12, 0.1, 0.08, 0.06,
                      0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003]
    rng = np.random.default_rng(12345)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = refiner.solve_lp(M)
        if g is None:
            refiner.update_active_set(a)
            g = refiner.solve_lp(M)
            if g is None:
                break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        improved = False
        for step in steps_grid:
            a_t = np.maximum((1.0 - step) * a + step * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = float(fast_conv(a_t).max())
            if M_t < best_M * (1.0 - 1e-10):
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                # Try perturbation
                pert = np.maximum(a + rng.normal(0, 0.005, n), 1e-6)
                pert = pert / pert.sum()
                M_pert = float(fast_conv(pert).max())
                if M_pert < best_M:
                    a = pert
                    best_a = a.copy()
                    best_M = M_pert
                    no_imp = 0
                    refiner.update_active_set(a)
                else:
                    a = pert
                    no_imp = 0
                    refiner.update_active_set(a)
                    if no_imp > 60:
                        break
        refiner.iter_count += 1
        if refiner.iter_count >= update_every:
            refiner.update_active_set(a)
    return best_a, best_M


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p * (1.0 + 0.1 * rng.normal(0, 1, n)), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.06, n), 0.01))
    inits.append(np.ones(n))
    for alpha in [0.5, 1.0, 1.5, 2.5]:
        inits.append(np.maximum(0.2 + 0.8 * (t**alpha + (1.0 - t)**alpha) / 2.0, 0.01))
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4.0 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-6.0 * (t - 0.3)**2) + np.exp(-6.0 * (t - 0.7)**2) + 0.03)
    k = max(3, n // 25)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def gs_init(a, n_iters=100, beta=0.85):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        power = 0.5 + 0.3 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 1e-8)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate = a_candidate / sc
        M_c = float(fast_conv(a_candidate).max())
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # === PHASE 1: Diverse initialization with smooth optimization ===
    p1_deadline = time.monotonic() + 220.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Optimization (220s)", flush=True)
    print(f"{'='*50}", flush=True)

    base_n = parent.size
    lengths = sorted(set([384, 512, 640, 768, 896, 1024, 1152, 1280,
                          1408, 1536, 1792, 2048,
                          max(256, base_n - 128), base_n, min(2048, base_n + 128)]))
    lengths = [n for n in lengths if 256 <= n <= 2048]

    best_per_length = {}
    n_lengths = len(lengths)
    time_per_length = 220.0 / n_lengths

    for li, n in enumerate(lengths):
        if time.monotonic() > p1_deadline - 20:
            break
        t0 = time.monotonic()
        remaining = p1_deadline - time.monotonic()
        allot = min(time_per_length * 1.5, remaining * 0.4)
        sub_dl = time.monotonic() + allot

        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        # Add GS initializations
        a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
        inits.append(gs_init(a_rand, n_iters=80, beta=0.85))
        if best.seq is not None:
            inits.append(gs_init(resample_seq(best.seq, n), n_iters=80, beta=0.9))

        best_n = None
        best_M = float('inf')
        # Run on top few initializations
        n_run = min(6, len(inits))
        for idx in range(n_run):
            if time.monotonic() > sub_dl:
                break
            a0 = np.maximum(inits[idx].astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            frac_remaining = (sub_dl - time.monotonic()) / max(1e-6, sub_dl - t0)
            iters = max(200, int(1500 * frac_remaining))
            a = run_nesterov(a0, sub_dl, max_iters=iters,
                             tau_start=1.0, tau_end=200.0,
                             lr_init=0.4, lr_end=0.003)
            M = float(fast_conv(a).max())
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 done: {time.monotonic() - (deadline - 1470):.0f}s elapsed", flush=True)

    # === PHASE 2: Main LP refinement ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (main refinement)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        print(f"  Refining n={n}, J={best.score:.6f}", flush=True)
        lp_budget = min(750.0, deadline - 300 - time.monotonic())
        lp_dl = time.monotonic() + lp_budget
        print(f"  LP budget: {lp_budget:.0f}s", flush=True)

        a, M = lp_mirror_descent(best.seq, lp_dl, max_iters=5000,
                                 top_frac=0.12, update_every=15)
        best.update(a)
        print(f"  After LP pass 1: M={M:.8f}", flush=True)

        if time.monotonic() < deadline - 200:
            a2, M2 = lp_mirror_descent(a, deadline - 150, max_iters=3000,
                                       top_frac=0.10, update_every=10)
            best.update(a2)
            print(f"  After LP pass 2: M={M2:.8f}", flush=True)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Multi-length LP exploration ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*50}", flush=True)

    if best_per_length and time.monotonic() < deadline - 100:
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda x: compute_J(best_per_length[x]))
        tested = set()
        if best.seq is not None:
            tested.add(best.seq.size)
        count = 0
        for n_try in sorted_lengths:
            if time.monotonic() > deadline - 80:
                break
            if n_try in tested:
                continue
            if count >= 3:
                break
            count += 1
            tested.add(n_try)
            a_cand = resample_seq(best.seq, n_try)
            a_cand = np.maximum(a_cand, 1e-6)
            a_cand = a_cand / a_cand.sum()
            a_cand = run_nesterov(a_cand, min(time.monotonic() + 15, deadline - 70),
                                  max_iters=400, tau_start=3, tau_end=80,
                                  lr_init=0.2, lr_end=0.01)
            best.update(a_cand)
            if time.monotonic() < deadline - 70:
                a_lp, _ = lp_mirror_descent(a_cand, min(time.monotonic() + 60, deadline - 60),
                                            max_iters=1000, top_frac=0.12)
                best.update(a_lp)
                print(f"  n={n_try}: J after LP = {compute_J(a_lp):.6f}", flush=True)

    best.report("Phase 3")

    # === PHASE 4: Resampling and final polish ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 15:
        # Try nearby lengths
        n_best = best.seq.size
        deltas = [32, -32, 64, -64, 128, -128, 16, -16]
        for delta in deltas:
            if time.monotonic() > deadline - 15:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 2500:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext = run_nesterov(a_ext, min(time.monotonic() + 8, deadline - 10),
                                 max_iters=200, tau_start=5, tau_end=60,
                                 lr_init=0.15, lr_end=0.01)
            best.update(a_ext)
            if time.monotonic() < deadline - 8:
                a_lp, _ = lp_mirror_descent(a_ext, deadline - 5,
                                            max_iters=500, top_frac=0.12)
                best.update(a_lp)

    best.report("Phase 4")

    # === PHASE 5: Ultra-fine final LP ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        print(f"\n  Ultra-fine final LP...", flush=True)
        fine_steps = [0.05, 0.04, 0.03, 0.025, 0.02, 0.015, 0.01, 0.008,
                      0.006, 0.005, 0.004, 0.003, 0.002, 0.001]
        a_final, _ = lp_mirror_descent(best.seq, deadline, max_iters=2000,
                                       top_frac=0.10, update_every=8,
                                       steps_grid=fine_steps)
        best.update(a_final)
        best.report("Phase 5")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
```
