def fast_conv(a):
    """FFT-based full convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_grad(a, tau):

    Returns (soft_max_value, gradient_wrt_a, full_conv).

    # Actually: conv(a,a)[k] = sum_{i+j=k} a[i]*a[j]

    # More carefully: d/d(a[i]) sum_{p+q=k} a[p]*a[q] = sum_{q: i+q=k} a[q] + sum_{p: p+i=k} a[p]
    # = 2 * sum_{q: i+q=k} a[q] - a[i]*[2i=k]

    n = len(a)
    for i in range(n):
        # sum over q such that i+q = k, for each k weighted by p[k]
        # = sum_k p[k] * (sum over q with i+q=k of a[q])
        # = sum_q a[q] * p[i+q]  (where p[i+q] is 0 if i+q >= 2n-1)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with homotopy on softmax temperature (smoothed max)
2. LP mirror descent with backtracking line search
3. Multi-M LP relaxation for escaping local minima
4. L-BFGS-B with adaptive temperature
5. Diverse initialization and length exploration

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def eval_J_fast(a):
    """Fast evaluation for internal use (assumes a is numpy array, non-negative)."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * len(a) * b.max() / (s * s)


def softmax_objective_and_grad(a, tau):
    """
    Returns (soft_max_value, gradient) where soft_max approximates max(conv(a,a)).
    grad is the gradient of soft_max wrt a.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    p = exp_shifted / Z  # softmax weights, length 2n-1
    
    soft_max_val = b_max + np.log(Z) / tau
    
    # grad[m] = 2 * sum_{q=0}^{n-1} a[q] * p[m+q]
    # Compute via FFT: g = conv(a_reversed, p)[n-1 : 2n-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa_rev = np.fft.rfft(a[::-1], size)
    fp = np.fft.rfft(p, size)
    prod = np.fft.irfft(fa_rev * fp, size)
    g = prod[n-1:2*n-1]
    grad = 2.0 * g
    
    return soft_max_val, grad


def full_objective_and_grad(a, tau):
    """
    Returns (J_smooth, grad_J) for J_smooth = 2*n*soft_max(conv(a,a))/(sum(a))^2
    """
    n = len(a)
    sm, grad_sm = softmax_objective_and_grad(a, tau)
    s = a.sum()
    J = 2.0 * n * sm / (s * s)
    # dJ/da = 2n * [grad_sm * s^2 - sm * 2s * 1] / s^4
    #       = 2n * [grad_sm / s - 2*sm / s^2]
    #       = (2n/s) * [grad_sm - 2*sm/s]
    grad_J = (2.0 * n / s) * (grad_sm - 2.0 * sm / s)
    return J, grad_J


def run_adam(a, deadline, max_iters=20000, tau_start=5.0, tau_end=5000.0,
             lr_init=0.01, lr_end=0.0001, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on the smoothed objective with homotopy on tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = eval_J_fast(a)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        # Homotopy: increase tau over time
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        J, grad = full_objective_and_grad(a, tau)
        
        # Update moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        # Track best
        J_exact = eval_J_fast(a)
        if J_exact < best_J:
            best_J = J_exact
            best_a = a.copy()
    
    return best_a

                       bounds=[(0, None)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8,
                                'maxfun': max_iter * 2})
        if res.success or res.fun < eval_J_fast(a):
            a_res = np.maximum(res.x, 0)
            s = a_res.sum()
            if s > 1e-10:
                a_res /= s
            if eval_J_fast(a_res) < eval_J_fast(a):
                return a_res
    except Exception as e:
        pass
    return a


def build_A_values(n, a):
    """Build sparse matrix for LP: (a*g)[k] = sum_j A[k,j]*g[j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    rows, cols, vals = build_A_values(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_backtrack(a, deadline, max_iters=300, t_init=0.5, min_t=0.005):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)

            best_a = a.copy()
            no_imp = 0
        else:

            if no_imp > 50:
                break
        
        # Get LP best response
        g = lp_best_response(a, M)

        if gs < 1e-10:

        a_current_M = M

def resample_seq(src, n_target):
    """Resample a sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_target:
        return src.copy()
    if src.size == 0:
        return np.ones(n_target)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_target)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def make_structured_init(n, rng, style='flat'):
    """Create structured initial sequences."""
    t = np.linspace(0, 1, n, endpoint=False)
    if style == 'flat':
        return np.ones(n)
    elif style == 'bimodal':
        sigma = max(1.0, n / 20.0)
        x = np.linspace(0, 1, n)
        return np.exp(-8*(x-0.25)**2) + np.exp(-8*(x-0.75)**2) + 0.01
    elif style == 'smooth_random':
        k = max(5, n // 16)
        kern = np.ones(k) / k
        return np.maximum(np.convolve(rng.uniform(0.2, 1.0, n), kern, 'same'), 0.01)
    elif style == 'cosine':
        f = rng.integers(1, 4)
        return np.maximum(0.5 + 0.45 * np.cos(2*np.pi*f*t + rng.uniform(0, 2*np.pi)), 0.01)
    elif style == 'power':
        alpha = rng.uniform(0.3, 3.0)
        return np.maximum(0.1 + 0.9 * t**alpha, 0.01)
    elif style == 'antisymmetric':
        half = np.maximum(rng.uniform(0.3, 1.0, n//2 + 1), 0.01)
        full = np.concatenate([half, half[-2::-1]])[:n]
        return full
    elif style == 'sidon_like':
        # Try to create a sequence with low autocorrelation peak
        # Based on "Sidon" idea: spread energy evenly
        base = np.ones(n) / n
        # Add structured noise
        freq = rng.permutation(n)[:n//4]
        amp = 0.1 / (np.sqrt(freq.astype(float) + 1))
        for i, f in enumerate(freq):
            base += amp[i] * np.cos(2*np.pi*f*t + rng.uniform(0, 2*np.pi))
        return np.maximum(base, 0.001)
    else:
        return np.ones(n)


def gs_spectral(a, n_iters=200, beta=0.85):
    """Gerchberg-Saxton: flatten the power spectrum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress magnitude
        mag_power = mag ** 0.7
        mag_new = (1.0 - beta) * mag + beta * (mag_power / (mag_power.max() + 1e-30)) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 1e-10)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a

    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value J(a) = 2n*max(conv(a,a))/(sum a)^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def eval_J_fast(a):
    """Fast internal evaluation (a assumed non-negative numpy array)."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * len(a) * b.max() / (s * s)


def softmax_obj_grad(a, tau):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient via FFT.
    Returns (soft_max_value, gradient_vector).
    Gradient: grad[m] = 2 * sum_{q=0}^{n-1} a[q] * p[m+q]
    Computed as cross-correlation via FFT.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    p = exp_shifted / Z
    soft_max_val = b_max + np.log(Z) / tau
    # grad[m] = 2 * sum_q a[q] * p[m+q] = 2 * (a_rev * p)[n-1+m]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa_rev = np.fft.rfft(a[::-1], size)
    fp = np.fft.rfft(p, size)
    prod = np.fft.irfft(fa_rev * fp, size)
    g = prod[n - 1:2 * n - 1]
    grad = 2.0 * g
    return soft_max_val, grad


def J_and_grad(a, tau):
    """Full objective J_smooth = 2n*soft_max/(sum)^2 and its gradient."""
    n = len(a)
    sm, grad_sm = softmax_obj_grad(a, tau)
    s = a.sum()
    J = 2.0 * n * sm / (s * s)
    grad_J = (2.0 * n / s) * (grad_sm - 2.0 * sm / s)
    return J, grad_J


def run_adam(a, deadline, max_iters=20000, tau_start=5.0, tau_end=5000.0,
             lr_init=0.015, lr_end=0.0003, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam on smoothed J with homotopy schedule on tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = eval_J_fast(a)
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        J, grad = J_and_grad(a, tau)
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
        m_hat = m_vec / (1.0 - beta1 ** it)
        v_hat = v_vec / (1.0 - beta2 ** it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        if it % 200 == 0:
            J_exact = eval_J_fast(a)
            if J_exact < best_J:
                best_J = J_exact
                best_a = a.copy()
    J_exact = eval_J_fast(a)
    if J_exact < best_J:
        best_a = a.copy()
    return best_a


def run_lbfgs(a, deadline, tau=200.0, max_iter=300):
    """L-BFGS-B on the smoothed objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    def obj_func(x):
        x_pos = np.maximum(x, 1e-14)
        s = x_pos.sum()
        if s < 1e-14:
            return 1e10
        sm, _ = softmax_obj_grad(x_pos, tau)
        return 2.0 * n * sm / (s * s)

    def obj_grad(x):
        x_pos = np.maximum(x, 1e-14)
        s = x_pos.sum()
        if s < 1e-14:
            return 1e10, np.zeros(n)
        sm, grad_sm = softmax_obj_grad(x_pos, tau)
        J = 2.0 * n * sm / (s * s)
        grad = (2.0 * n / s) * (grad_sm - 2.0 * sm / s)
        return J, grad

    try:
        res = minimize(obj_grad, a, jac=True, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-14,
                                'gtol': 1e-8, 'maxfun': max_iter * 3})
        if res.fun < 1e9:
            a_res = np.maximum(res.x, 0)
            s = a_res.sum()
            if s > 1e-10:
                a_res = a_res / s
                if eval_J_fast(a_res) < eval_J_fast(a):
                    return a_res
    except Exception:
        pass
    return a


def build_lp_matrix(n, a):
    """Build sparse matrix A where (a*g)[k] = (A g)_k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def run_lp_backtrack(a, deadline, max_iters=300, t_init=0.4):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 60:
                break
        g = lp_best_response(a, M)
        if g is None:
            g = lp_best_response(a, M * 1.005)
            if g is None:
                break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        improved = False
        t = t_init
        for _ in range(25):
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < M * (1 - 1e-12):
                a = a_trial
                improved = True
                break
            t *= 0.5
        if not improved:
            # Multi-M escape
            for M_mult in [1.01, 1.03, 1.08]:
                g_alt = lp_best_response(a, M * M_mult)
                if g_alt is None:
                    continue
                gs_alt = g_alt.sum()
                if gs_alt < 1e-10:
                    continue
                g_alt = g_alt / gs_alt
                for t2 in [0.3, 0.2, 0.1, 0.05]:
                    a_trial = np.maximum((1.0 - t2) * a + t2 * g_alt, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial = a_trial / s
                    if fast_conv(a_trial).max() < M * (1 - 1e-12):
                        a = a_trial
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            no_imp += 1
    return best_a


def run_lp_fista(a, deadline, max_iters=200):
    """LP with Fista-like acceleration (momentum on the iterate)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    no_imp = 0
    t_init = 0.35
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(y)
        M = float(b.max())
        g = lp_best_response(y, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        t = t_init / (1.0 + 0.03 * it)
        a_new = np.maximum((1.0 - t) * y + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        t_k_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + ((t_k - 1.0) / (t_k_new - 1.0)) * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-10:
            y_new = a_new
        else:
            y_new = y_new / sy
        M_new = fast_conv(a_new).max()
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        a = a_new
        y = y_new
        t_k = t_k_new
    return best_a
