```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize_simplex(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a
```

```python
def compute_gradient_weights(a, weights):
    """Compute gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (valid indices only)"""
    n = len(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W = rfft(w_ext, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    # corr[i] = sum_j a[j] * w[i+j] (for valid i+j < 2n-1)
    return 2.0 * corr[:n]
```

```python
    best_a = a.copy()
    best_obj = compute_obj(a)

    v = np.zeros(n)

    total_iters = 500  # Will be cut by time budget
    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        frac = it / total_iters
        tau = tau_init * (tau_final / tau_init) ** frac
        
        # Compute convolution and weights
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500, 500)
        weights = np.exp(shifted)

        grad = compute_gradient_weights(a, weights)
```

```python
def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

def gradient_from_weights(a, weights):
    """
    Compute grad[i] = 2 * sum_k weights[k] * a[k-i] for valid indices.
    Uses FFT for efficiency.
    """
    n = len(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_ext = np.zeros(fft_size)
    w_ext[:len(weights)] = weights
    W = rfft(w_ext, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    return 2.0 * corr[:n]

def build_toeplitz(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A

def solve_minpeak_lp(fixed_seq, time_limit=20):
    """
    Solve: min t s.t. (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.

    n = len(fixed_seq)
    if n > 1200:
        return None, None
    A = build_toeplitz(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:

    except Exception:

    return None, None

def parametric_init(n, sigma_frac, freq, amp, phase, edge_decay):
    """Generate a parametric initial sequence."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n * sigma_frac
    bell = np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    edge = 1.0 / (1.0 + edge_decay * ((i - center) / (n/2))**6)
    a = bell * mod * edge
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
```
