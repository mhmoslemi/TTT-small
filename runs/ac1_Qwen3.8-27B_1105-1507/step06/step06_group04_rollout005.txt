from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def top_k_subgradient(a, K):
    """Compute top-K subgradient of max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    # Get top K peak positions
    top_k = np.argpartition(conv, -K)[-K:]
    w = np.zeros(2*n-1)
    w[top_k] = 1.0 / K
    
    # Compute gradient: g[i] = 2 * sum_p w[p] * a[p-i]

    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    grad = 2.0 * corr[:n]
    return grad, conv, top_k

    a = simplex_project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = alpha_init
    stall = 0
    tau = 0.01
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        if use_smooth and it < 500:
            grad, conv = smooth_max_gradient(a, tau)
            tau *= 0.995
        else:
            K = min(2 * n - 1, max(1, (it // 200) + 1))
            K = min(K, 15)
            grad, conv, top_k = top_k_gradient(a, K)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        accepted = False
        for bt in range(15):
            a_new = simplex_project(a - alpha * grad_centered / grad_norm)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                a = a_new
                best_obj = obj_new
                best_a = a.copy()
                stall = 0
                accepted = True
                break
            alpha *= 0.75
        if not accepted:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.03 + 0.02 * (stall % 5)
                a = simplex_project(best_a * np.exp(rng.normal(0, scale, n)))
                alpha = alpha_init
                stall = 0
    return best_a, best_obj


def remez_pairwise_swap(a, num_rounds=200, time_budget=10.0):
    n = len(a)
    a = simplex_project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    for rnd in range(num_rounds):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        # Find a good pair (i, j) to swap
        # i should be where a[i] is large and contributes to the peak
        # j should be where a[j] is small
        contrib = np.zeros(n)
        for i in range(n):
            p = k_star - i
            if 0 <= p < 2 * n - 1:
                contrib[i] = abs(a[p]) if p < n else 0.0
        sorted_by_a = np.argsort(a)[::-1]
        best_delta_obj = best_obj
        best_new_a = a.copy()
        for i_idx in range(min(10, n)):
            i = sorted_by_a[i_idx]
            if a[i] < 1e-8:
                break
            for j_idx in range(min(10, n)):
                j = sorted_by_a[n - 1 - j_idx]
                if j == i or a[j] >= a[i]:
                    continue
                # Try several delta values
                max_delta = a[i] * 0.5
                for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
                    delta = min(max_delta, a[i] * frac)
                    if delta < 1e-10:
                        continue
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    a_try = simplex_project(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_delta_obj:
                        best_delta_obj = obj_try
                        best_new_a = a_try.copy()
        if best_obj - best_delta_obj > 1e-14:
            a = best_new_a.copy()
            best_obj = best_delta_obj
            best_a = a.copy()
    return best_a, best_obj


def generate_diverse_starts(n, seed=42):
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    starts.append(np.ones(n) / n)
    tri = 1.0 - np.abs(i - center) / (center + 1)
    tri = np.maximum(tri, 0.01)
    starts.append(simplex_project(tri))
    for sigma_frac in [0.2, 0.35, 0.5, 0.7]:
        bell = np.exp(-((i - center) ** 2) / (2 * (n * sigma_frac) ** 2))
        starts.append(simplex_project(bell))
    for _ in range(6):
        sigma = n * (0.15 + 0.4 * rng.random())
        freq = 1 + int(rng.integers(0, 6))
        amp = 0.1 + 0.4 * rng.random()
        phase = 2 * np.pi * rng.random()
        mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
        bell = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        a = bell * mod
        a = np.maximum(a, 0)
        starts.append(simplex_project(a))
    noisy = np.abs(np.ones(n) + rng.normal(0, 0.3, n))
    starts.append(simplex_project(noisy))
    # Edge-heavy
    edge = np.abs(i - center) / center
    a = 0.3 + 0.7 * edge
    starts.append(simplex_project(a))
    return starts


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return simplex_project(a_new)


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return simplex_project(parent), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget):
    n = len(a)
    start = time.monotonic()
    best_a = simplex_project(a.copy())
    best_obj = compute_obj(best_a)
    # Phase A: subgradient
    t1 = time_budget * 0.4
    a1, obj1 = subgradient_optimize(best_a, t1, use_smooth=True)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
    # Phase B: LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15 and n <= 900:
        t2 = min(remaining * 0.35, 60.0)
        a2, obj2 = lp_alternate(best_a, t2, lp_time=12)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
    # Phase C: more subgradient (no smooth)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t3 = min(remaining * 0.3, 30.0)
        a3, obj3 = subgradient_optimize(best_a, t3, use_smooth=False, alpha_init=0.05)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
    # Phase D: Remez pairwise
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a4, obj4 = remez_pairwise_swap(best_a, num_rounds=100, time_budget=min(remaining, 10.0))
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # Phase 1: Multi-start exploration
    print("\n=== Phase 1: Multi-start exploration ===")
    n_values = sorted(set([n_parent, n_parent - 50, n_parent + 50,
                           n_parent - 100, n_parent + 100]))
    n_values = [x for x in n_values if 400 <= x <= 1100]

    candidates = []
    for n in n_values:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n == n_parent and best_seq is not None:
            starts = [simplex_project(best_seq.copy())]
            starts.extend(generate_diverse_starts(n, seed=seed))
        else:
            starts = generate_diverse_starts(n, seed=seed + n)

        alloc_per = min(remaining / max(len(starts), 1), 25.0)
        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 200:
                break
            t_alloc = min(alloc_per, 20.0)
            if t_alloc < 5:
                break
            a_opt, obj_opt = subgradient_optimize(s, t_alloc, use_smooth=True, alpha_init=0.12)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")
            candidates.append((obj_opt, a_opt.copy()))
            print(f"  n={n}, start {si}: {obj_opt:.6f} (best={best_score:.6f})")

    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive polishing of top candidates
    print("\n=== Phase 2: Intensive polishing ===")
    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_cand = candidates[ci][1].copy()
        n_cand = len(a_cand)
        alloc = min(remaining * 0.3, 120.0)
        print(f"  Cand {ci} (n={n_cand}, obj={candidates[ci][0]:.8f}): {alloc:.0f}s")
        a_pol, obj_pol = polish_sequence(a_cand, alloc)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"  No improvement ({obj_pol:.8f})")

    # Phase 3: Length exploration
    print(f"\n=== Phase 3: Length exploration (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = sorted(set([x for x in [n_best - 50, n_best + 50,
                                                n_best - 100, n_best + 100,
                                                n_best + 150, n_best - 150]
                                   if 400 <= x <= 1200]))
        for n_test in test_lengths:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_rs = resample_sequence(best_seq, n_test)
            alloc = min(remaining * 0.2, 40.0)
            a_pol, obj_pol = polish_sequence(a_rs, alloc)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** {best_score:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.6f}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_fin, obj_fin = polish_sequence(best_seq, min(remaining * 0.6, 120.0))
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Final: {best_score:.8f}")
        # Perturbation attempts
        remaining = deadline - time.monotonic()
        if remaining > 15 and best_seq is not None:
            n_fin = len(best_seq)
            rng = np.random.default_rng(seed + 77777)
            for pidx in range(8):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.015 + 0.01 * pidx
                a_pert = simplex_project(best_seq * np.exp(rng.normal(0, scale, n_fin)))
                t_alloc = min(remaining * 0.1, 12.0)
                if t_alloc < 3:
                    break
                a_pert, obj_pert = subgradient_optimize(a_pert, t_alloc, use_smooth=False, alpha_init=0.05)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {pidx}: ** {best_score:.8f} **")

    # Phase 5: Final LP if time permits
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        if n_best <= 800 and remaining > 25:
            print(f"\n=== Phase 5: Final LP (n={n_best}) ===")
            a_lp, obj_lp = lp_alternate(best_seq, min(remaining * 0.5, 50.0), lp_time=10)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: {best_score:.8f}")
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_md, obj_md = remez_pairwise_swap(best_seq, num_rounds=200, time_budget=min(remaining2, 15.0))
                if obj_md < best_score:
                    best_score = obj_md
                    best_seq = a_md.copy()
                    print(f"  Remez: {best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
