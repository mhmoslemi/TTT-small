def fast_conv(a):

def compute_J(a):

    a = np.maximum(a, 0)

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

        idx = (size - i) % size
        grad1[i] = corr[idx]
    grad2 = conv_wa[:n]
    
    grad = grad1 + grad2
    return grad, weights, c

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        return np.ones(n) / n
    lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
    return np.maximum(v - lam, 0.0)

def arcsine_init(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt(np.maximum((i + 0.5) * (n - i - 0.5), 1e-10))
    return vals / vals.sum()

def smooth_objective_and_grad(a, tau):
    """Compute smooth approximation of max(conv) and its gradient."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    shifted = c - c_max
    exp_c = np.exp(tau * shifted)
    Z = exp_c.sum()
    w = exp_c / Z
    
    # Smooth max ≈ c_max + log(Z)/tau
    smooth_max = c_max + np.log(Z) / tau
    
    # Gradient: w[k] * d(conv[k])/da[i]
    # d(conv[k])/da[i] = a[k-i] + a[i-k] (with boundary: a[j]=0 for j<0 or j>=n)
    grad = np.zeros(n)
    for i in range(n):
        # term1: sum_k w[k] * a[k-i] (valid when 0<=k-i<n, 0<=k<2n-1)
        # k ranges from i to i+n-1 (intersected with [0, 2n-2])
        k_lo = max(0, i)
        k_hi = min(2*n-2, i+n-1)
        if k_lo <= k_hi:
            grad[i] += np.dot(w[k_lo:k_hi+1], a[k_lo-i:k_hi-i+1])
        # term2: sum_k w[k] * a[i-k] (valid when 0<=i-k<n, 0<=k<2n-1)
        # k ranges from i-n+1 to i (intersected with [0, 2n-2])
        k_lo2 = max(0, i-n+1)
        k_hi2 = min(2*n-2, i)
        if k_lo2 <= k_hi2:
            grad[i] += np.dot(w[k_lo2:k_hi2+1], a[i-k_hi2:i-k_lo2+1])
    
    return smooth_max, grad, w, c

def projected_gradient_descent(a, max_iters, deadline, tau_init=50.0, tau_end=500.0,
                                 lr_init=0.1, verbose=False):
    """Projected gradient descent on the simplex with smooth max."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_init + (tau_end - tau_init) * frac
        lr = lr_init / (1.0 + 0.05 * it)
        
        smooth_max, grad, w, c = smooth_objective_and_grad(a, tau)
        
        # Project gradient onto simplex tangent space
        grad_proj = grad - np.mean(grad)
        
        # Update
        a_new = a - lr * grad_proj
        a_new = project_simplex(a_new)
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtracking: try smaller step
            if no_improve > 10:
                for factor in [0.5, 0.25, 0.1]:
                    a_try = a - lr * factor * grad_proj
                    a_try = project_simplex(a_try)
                    J_try = compute_J(a_try)
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a_try.copy()
                        a = a_try
                        no_improve = 0
                        break
                else:
                    # Reset
                    a = best_a + np.random.default_rng(it).normal(0, 0.01, n)
                    a = project_simplex(a)
                    no_improve = 0
        
        a = a_new if J_new <= best_J else a  # Keep a moving forward
        
        if verbose and it % 100 == 0:
            print(f"    PGD it {it}: tau={tau:.0f} J={best_J:.8f}", flush=True)
        
        if no_improve > 50:
            break
    
    return best_a, best_J

def multiplicative_updates(a, max_iters, deadline, target_ratio=1.0, verbose=False):
    """
    Multiplicative updates inspired by "water filling":
    Positions that contribute to high-convolution regions get decreased.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_max = c.max()
        # Target: all conv coefficients should be ≈ S^2/(2n-1) = 1/(2n-1) (since sum=1)
        target = 1.0 / (2 * n - 1)
        
        # Compute "pressure" at each position
        # pressure[i] = sum_k (c[k] - target) * |d(conv[k])/da[i]| / (2*n*a[i])
        # Simplified: pressure[i] = sum_k max(0, c[k] - target) * (a[k-i] + a[i-k]) / a[i]
        
        # Use smooth version: weight = c[k] / c_max (normalized)
        w = c / c.max()
        
        pressure = np.zeros(n)
        for i in range(n):
            # Contribution of a[i] to each c[k]
            # d(conv[k])/da[i] = a[k-i] + a[i-k]
            k_lo = max(0, i)
            k_hi = min(2*n-2, i+n-1)
            if k_lo <= k_hi:
                contrib1 = np.dot(w[k_lo:k_hi+1], a[k_lo-i:k_hi-i+1])
            else:
                contrib1 = 0
            k_lo2 = max(0, i-n+1)
            k_hi2 = min(2*n-2, i)
            if k_lo2 <= k_hi2:
                contrib2 = np.dot(w[k_lo2:k_hi2+1], a[i-k_hi2:i-k_lo2+1])
            else:
                contrib2 = 0
            pressure[i] = contrib1 + contrib2
        
        # Multiplicative update: a[i] *= exp(-eta * pressure[i])
        eta = 0.1
        a_new = a * np.exp(-eta * pressure)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.5  # Reduce step size
            if no_improve > 30:
                break
        
        if verbose and it % 50 == 0:
            print(f"    MU it {it}: eta={eta:.4f} J={best_J:.8f}", flush=True)
    
    return best_a, best_J

"""
Minimizer for J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Arcsine distribution initialization (U-shaped: large at edges, small center)
2. Projected gradient descent with smooth-max and backtracking
3. Exact coordinate descent via golden section search
4. Multiplicative water-filling updates
5. Symmetry exploitation and multi-length exploration
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        return np.ones(n) / n
    lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
    return np.maximum(v - lam, 0.0)


def arcsine_init(n):
    """Discrete arcsine distribution: 1/sqrt((i+0.5)*(n-i-0.5))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt(np.maximum((i + 0.5) * (n - i - 0.5), 1e-10))
    return vals / vals.sum()


def u_shaped_init(n, power=1.0):
    """U-shaped: (t*(1-t))^(-power/2)."""
    i = np.arange(n, dtype=np.float64)
    t = i / max(n - 1, 1)
    x = np.maximum(t * (1.0 - t), 1e-8)
    vals = x ** (-power / 2.0)
    return vals / vals.sum()


def exp_edge_init(n, alpha=2.0):
    """Exponential edge enhancement: exp(-alpha * sin^2(pi*t))."""
    i = np.arange(n, dtype=np.float64)
    t = i / max(n - 1, 1)
    vals = np.exp(-alpha * np.sin(np.pi * t) ** 2)
    return vals / vals.sum()


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_objective_and_grad(a, tau):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Returns (smooth_max, gradient, weights, full_conv).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    shifted = c - c_max
    exp_c = np.exp(tau * shifted)
    Z = exp_c.sum()
    w = exp_c / Z
    smooth_max = c_max + np.log(Z) / tau

    # Gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # with a[j] = 0 for j < 0 or j >= n
    grad = np.zeros(n)
    for i in range(n):
        # Term 1: sum_k w[k] * a[k-i], valid when 0 <= k-i < n
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo <= k_hi:
            grad[i] += np.dot(w[k_lo:k_hi + 1], a[k_lo - i:k_hi - i + 1])
        # Term 2: sum_k w[k] * a[i-k], valid when 0 <= i-k < n
        k_lo2 = max(0, i - n + 1)
        k_hi2 = min(2 * n - 2, i)
        if k_lo2 <= k_hi2:
            grad[i] += np.dot(w[k_lo2:k_hi2 + 1], a[i - k_hi2:i - k_lo2 + 1])

    return smooth_max, grad, w, c


def pgd_optimize(a, max_iters, deadline, tau_init=50.0, tau_end=500.0,
                 lr_init=0.05, verbose=False):
    """Projected gradient descent with smooth max and backtracking."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    rng = np.random.default_rng(42)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_init + (tau_end - tau_init) * frac
        lr = lr_init / (1.0 + 0.02 * it)

        smooth_max, grad, w, c = smooth_objective_and_grad(a, tau)

        # Project gradient to simplex tangent space
        grad_proj = grad - np.mean(grad)

        # Try the step
        a_new = a - lr * grad_proj
        a_new = project_simplex(a_new)
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Backtracking line search
            found = False
            for factor in [0.5, 0.25, 0.1, 0.05]:
                a_try = a - lr * factor * grad_proj
                a_try = project_simplex(a_try)
                J_try = compute_J(a_try)
                if J_try < best_J - 1e-12:
                    best_J = J_try
                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0
                    found = True
                    break
            if not found and no_improve > 30:
                # Perturb and restart
                a = best_a + rng.normal(0, 0.005, n)
                a = project_simplex(a)
                no_improve = 0

        if verbose and it % 200 == 0:
            print(f"      PGD it {it}: tau={tau:.0f} J={best_J:.8f}", flush=True)

        if no_improve > 80:
            break

    return best_a, best_J


def multiplicative_updates(a, max_iters, deadline, eta_init=0.1, verbose=False):
    """
    Multiplicative water-filling: positions contributing to high-conv
    regions are reduced multiplicatively.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    eta = eta_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_max = c.max()
        # Normalized weights emphasizing peaks
        w = c / c_max

        # Compute pressure at each position
        pressure = np.zeros(n)
        for i in range(n):
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            if k_lo <= k_hi:
                pressure[i] += np.dot(w[k_lo:k_hi + 1], a[k_lo - i:k_hi - i + 1])
            k_lo2 = max(0, i - n + 1)
            k_hi2 = min(2 * n - 2, i)
            if k_lo2 <= k_hi2:
                pressure[i] += np.dot(w[k_lo2:k_hi2 + 1], a[i - k_hi2:i - k_lo2 + 1])

        # Multiplicative update
        a_new = a * np.exp(-eta * pressure)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Gradually increase eta
            eta = min(eta * 1.05, 1.0)
        else:
            no_improve += 1
            eta *= 0.7
            if no_improve > 50:
                break

        if verbose and it % 100 == 0:
            print(f"      MU it {it}: eta={eta:.4f} J={best_J:.8f}", flush=True)

    return best_a, best_J


def coord_descent_golden(a, max_sweeps, deadline, seed=42, verbose=False):
    """
    Coordinate descent with golden section search for exact 1D minimization.
    Works with unnormalized a (J is scale-invariant).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    golden = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            old = a[i]
            current_max = conv.max()

            # Search range for a[i]
            lo = 0.0
            hi = min(2.0 * old + 0.05, 0.5) if old > 1e-8 else 0.1

            if old < 1e-10:
                # Try a few small values
                candidates = [0.0, 0.001, 0.005, 0.01, 0.02, 0.05]
                best_x = old
                best_f = 2.0 * n * current_max
                for x in candidates:
                    delta = x - old
                    if abs(delta) < 1e-15:
                        continue
                    k_lo = max(0, i)
                    k_hi = min(2 * n - 2, i + n - 1)
                    conv[k_lo:k_hi + 1] += 2.0 * delta * a[k_lo - i:k_hi - i + 1]
                    k2 = 2 * i
                    if k2 <= 2 * n - 2:
                        conv[k2] += delta * delta
                    f = 2.0 * n * conv.max()
                    conv[k_lo:k_hi + 1] -= 2.0 * delta * a[k_lo - i:k_hi - i + 1]
                    if k2 <= 2 * n - 2:
                        conv[k2] -= delta * delta
                    if f < best_f:
                        best_f = f
                        best_x = x
            else:
                # Golden section search
                a_pt = hi - golden * (hi - lo)
                b_pt = lo + golden * (hi - lo)

                def eval_at(x):
                    delta = x - old
                    if abs(delta) < 1e-15:
                        return 2.0 * n * conv.max()
                    k_lo = max(0, i)
                    k_hi = min(2 * n - 2, i + n - 1)
                    conv[k_lo:k_hi + 1] += 2.0 * delta * a[k_lo - i:k_hi - i + 1]
                    k2 = 2 * i
                    if k2 <= 2 * n - 2:
                        conv[k2] += delta * delta
                    f = 2.0 * n * conv.max()
                    conv[k_lo:k_hi + 1] -= 2.0 * delta * a[k_lo - i:k_hi - i + 1]
                    if k2 <= 2 * n - 2:
                        conv[k2] -= delta * delta
                    return f

                fa = eval_at(a_pt)
                fb = eval_at(b_pt)

                for _ in range(35):
                    if fa < fb:
                        hi = b_pt
                        b_pt = a_pt
                        fb = fa
                        a_pt = hi - golden * (hi - lo)
                        fa = eval_at(a_pt)
                    else:
                        lo = a_pt
                        a_pt = b_pt
                        fa = fb
                        b_pt = lo + golden * (hi - lo)
                        fb = eval_at(b_pt)

                best_x = (lo + hi) / 2.0
                best_f = eval_at(best_x)
                # Also check 0
                f0 = eval_at(0.0)
                if f0 < best_f:
                    best_x = 0.0
                    best_f = f0

            current_f = 2.0 * n * conv.max()
            if best_f < current_f - 1e-12:
                delta = best_x - old
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                conv[k_lo:k_hi + 1] += 2.0 * delta * a[k_lo - i:k_hi - i + 1]
                k2 = 2 * i
                if k2 <= 2 * n - 2:
                    conv[k2] += delta * delta
                a[i] = best_x
                improved = True

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            break

        if verbose and sweep % 100 == 0:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f} J={best_obj:.8f}", flush=True)

    a = best_a
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    return a


def solve_one(a0, budget, verbose=False):
    """
    Full optimization pipeline for a single initial sequence:
    1. PGD (30%)
    2. Multiplicative updates (20%)
    3. Coordinate descent with golden section (50%)
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def check_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a_new
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = budget - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: PGD
    t1 = time.monotonic() + 0.30 * remaining
    a_pgd, obj_pgd = pgd_optimize(a, max_iters=10000, deadline=t1,
                                   tau_init=30, tau_end=300, lr_init=0.03)
    check_update(a_pgd)
    a = best_a.copy()

    remaining = budget - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: Multiplicative updates
    t2 = time.monotonic() + 0.20 * remaining
    a_mu, obj_mu = multiplicative_updates(a, max_iters=5000, deadline=t2, eta_init=0.05)
    check_update(a_mu)
    a = best_a.copy()

    remaining = budget - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: Coordinate descent (the workhorse)
    t3 = budget - 2
    a_cd = coord_descent_golden(a, max_sweeps=50000, deadline=t3, seed=777)
    check_update(a_cd)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    - Phase 1: Diverse initializations (arcsine, U-shaped, exp-edge) at multiple lengths
    - Phase 2: Intensive local optimization on the best candidate
    - Phase 3: Final polish with long coordinate descent
    
    Key innovations over previous approach:
    - Arcsine distribution initialization (theoretically motivated)
    - Golden section coordinate descent (exact 1D vs fixed candidates)
    - Multiplicative water-filling updates
    - PGD with backtracking instead of Adam
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                update_best(parent)
                print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Generate initializations for a given length
    def get_inits(n, parent_seq):
        inits = []
        # Arcsine (theoretically optimal shape)
        inits.append(arcsine_init(n))
        # U-shaped with various powers
        for pw in [0.5, 0.8, 1.0, 1.5]:
            inits.append(u_shaped_init(n, power=pw))
        # Exponential edge
        for alpha in [1.0, 2.0, 3.0, 4.0]:
            inits.append(exp_edge_init(n, alpha=alpha))
        # Parent resampled
        if parent_seq is not None:
            pr = resample_seq(parent_seq, n)
            if pr.sum() > 1e-10:
                inits.append(pr / pr.sum())
            # Perturbed parent
            for sigma in [0.02, 0.05, 0.1]:
                pert = pr + rng.normal(0, sigma, n)
                pert = project_simplex(pert)
                inits.append(pert)
        # Uniform (baseline)
        inits.append(np.ones(n) / n)
        return inits

    # === PHASE 1: Multi-length, multi-init exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Exploration (budget 600s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_parent = parent.size if parent is not None else 768
    candidate_ns = sorted(set([384, 512, 640, 768, 1024, 1280]),
                          key=lambda x: abs(x - n_parent))

    phase1_deadline = t_start + 600.0
    per_n_budget = (phase1_deadline - t_start) / len(candidate_ns)

    for new_n in candidate_ns:
        remaining = phase1_deadline - time.monotonic()
        if remaining < 50:
            break
        n_alloc = min(per_n_budget, remaining - 10)
        if n_alloc < 30:
            break

        print(f"\n  === n={new_n} (budget {n_alloc:.0f}s) ===", flush=True)
        inits = get_inits(new_n, parent)

        # Select top diverse initializations (max 4 per length)
        selected_indices = [0, 1, 3, 5] if len(inits) > 5 else list(range(min(4, len(inits))))
        per_init = n_alloc / max(1, len(selected_indices))

        for idx in selected_indices:
            if idx >= len(inits):
                continue
            rem = phase1_deadline - time.monotonic()
            if rem < 15:
                break
            alloc = min(per_init, rem - 5)
            if alloc < 10:
                continue

            a0 = inits[idx]
            a0 = np.maximum(a0, 0)
            if a0.sum() < 1e-10:
                continue
            a0 = a0 / a0.sum()

            a_opt, obj_opt = solve_one(a0, time.monotonic() + alloc, verbose=False)
            update_best(a_opt)
            print(f"    Init {idx}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Intensive optimization on best ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Intensive optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase2_deadline = t_start + 1200.0

    if best_seq is not None:
        remaining = phase2_deadline - time.monotonic()
        if remaining > 60:
            # Multiple optimization passes from the best
            a_curr = best_seq.copy()

            # Pass 1: Long PGD with high tau (fine-tuning)
            t_p1 = time.monotonic() + 0.25 * remaining
            print(f"  PGD fine-tune ({t_p1 - time.monotonic():.0f}s)...", flush=True)
            a_pgd, _ = pgd_optimize(a_curr, max_iters=30000, deadline=t_p1,
                                     tau_init=200, tau_end=2000, lr_init=0.01)
            update_best(a_pgd)
            a_curr = best_seq.copy()

            remaining = phase2_deadline - time.monotonic()
            if remaining > 30:
                # Pass 2: Multiplicative updates
                t_p2 = time.monotonic() + 0.25 * remaining
                print(f"  MU ({t_p2 - time.monotonic():.0f}s)...", flush=True)
                a_mu, _ = multiplicative_updates(a_curr, max_iters=20000,
                                                  deadline=t_p2, eta_init=0.03)
                update_best(a_mu)
                a_curr = best_seq.copy()

            remaining = phase2_deadline - time.monotonic()
            if remaining > 30:
                # Pass 3: Very long coordinate descent
                t_p3 = time.monotonic() + 0.50 * remaining
                print(f"  Long CD ({t_p3 - time.monotonic():.0f}s)...", flush=True)
                a_cd = coord_descent_golden(a_curr, max_sweeps=100000,
                                             deadline=t_p3, seed=31415)
                update_best(a_cd)

    # === PHASE 3: Final polish ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Final polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Final coordinate descent on the absolute best
            print(f"  Final CD ({remaining - 5:.0f}s)...", flush=True)
            a_final = coord_descent_golden(best_seq.copy(), max_sweeps=50000,
                                            deadline=deadline - 5, seed=27182)
            update_best(a_final)

            # Try one more PGD from the CD result
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_pgd2, _ = pgd_optimize(best_seq.copy(), max_iters=10000,
                                          deadline=deadline - 3,
                                          tau_init=500, tau_end=5000, lr_init=0.005)
                update_best(a_pgd2)

    # Fallback
    if best_seq is None:
        best_seq = arcsine_init(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL RESULT: J={best_score:.8f} n={best_seq.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
