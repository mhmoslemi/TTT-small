        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(a, time_limit=8):
    """
    Solve: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0.
    Returns (g, t, active_mask) or (None, None, None).
    """
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
            t = float(res.x[n])
            conv_ag = A @ g
            tol = 1e-5 * max(1.0, abs(t))
            active = np.abs(conv_ag - t) < tol
            if not np.any(active):
                active = np.zeros(2 * n - 1, dtype=bool)
                active[np.argmax(conv_ag)] = True
            return g, t, active
    except Exception:
        pass
    return None, None, None


def solve_maxsum_lp(a, M, time_limit=8):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None.
    """
    n = len(a)
    A = build_toeplitz(a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def smooth_max_subgradient(a, tau):
    """
    Compute smooth max subgradient.
    Returns (gradient, cmax, weights).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # grad[i] = 2 * sum_j a[j] * weights[i+j] (for valid indices)
    # = 2 * ifft(FFT(a) * conj(FFT(weights)))[i]
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    FA = rfft(a, fft_size)
    FW = rfft(weights, fft_size)
    corr = irfft(FA * np.conj(FW), fft_size)
    grad = 2.0 * corr[:n]
    return grad, cmax, weights


def lp_gradient_direction(a, g, active):
    """
    Compute gradient direction from LP active set.
    grad[i] = sum_k active_weight[k] * g[k-i] (for valid k-i)
    """
    n = len(a)
    active_float = active.astype(np.float64)
    s = np.sum(active_float)
    if s > 0:
        active_float /= s
    else:
        active_float = np.zeros(2 * n - 1)
        active_float[np.argmax(compute_conv(a))] = 1.0

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    FA = rfft(active_float, fft_size)
    FG = rfft(g, fft_size)
    # grad[i] = sum_j g[j] * active[i+j] = ifft(conj(FFT(g)) * FFT(active))[i]
    corr = irfft(np.conj(FG) * FA, fft_size)
    return corr[:n]


def adaptive_subgradient_v2(a, max_iters=2000, time_budget=60.0):
    """
    Adaptive subgradient with:
    - Feedback-based temperature (based on peak concentration)
    - Two-phase step size (explore then exploit)
    - Nesterov momentum with adaptive coefficient
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.7
    step = 0.1
    tau = 0.02

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Two-phase: first 40% explore (large steps), rest exploit
        phase_frac = it / max(max_iters, 1)
        if phase_frac < 0.4:
            step = 0.08 + 0.04 * (1 - phase_frac / 0.4)
            mu = 0.6 + 0.2 * (phase_frac / 0.4)
        else:
            step = 0.04 * (1 - (phase_frac - 0.4) / 0.6) + 0.005
            mu = 0.8 + 0.1 * ((phase_frac - 0.4) / 0.6)

        grad, cmax, weights = smooth_max_subgradient(a, tau)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Adaptive temperature: if peaks are concentrated, decrease tau
        conv = compute_conv(a)
        cmax_now = float(np.max(conv))
        # Count "near-peak" values (within 5% of max)
        near_peaks = np.sum(conv > 0.95 * cmax_now)
        if near_peaks > 10:
            tau *= 0.98  # More concentrated, cool faster
        elif near_peaks < 3:
            tau *= 1.01  # Spread out, warm slightly
        tau = max(0.0005, min(0.05, tau))

    return best_a, best_obj


def lp_guided_search(a, max_iters=100, time_budget=80.0):
    """
    LP-guided optimization: alternate between LP solve and gradient step.
    Use LP to get informed direction, then take a step.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.75
    step = 0.15
    stall = 0
    last_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Solve LP every iteration (it's fast for n~700)
        lp_time = min(5.0, remaining * 0.3)
        g, t_opt, active = solve_minpeak_lp(a, time_limit=lp_time)

        if g is not None:
            # Compute LP-based gradient
            grad = lp_gradient_direction(a, g, active)
            grad_centered = grad - np.mean(grad)
            gn = np.linalg.norm(grad_centered)

            if gn > 1e-15:
                v = mu * v - step * grad_centered / gn
                a_new = a - v
                a_new = normalize_simplex(a_new)
                obj_new = compute_obj(a_new)

                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    # Try blending with g directly
                    blend_t = 0.3
                    a_blend = (1 - blend_t) * a + blend_t * g
                    a_blend = normalize_simplex(a_blend)
                    obj_blend = compute_obj(a_blend)
                    if obj_blend < best_obj - 1e-13:
                        best_obj = obj_blend
                        best_a = a_blend.copy()
                        a = a_blend
                        stall = 0
                    else:
                        stall += 1
                        if stall > 8:
                            # Perturb and reset
                            rng = np.random.default_rng(it * 7919 + 3)
                            scale = 0.03 + 0.02 * min(stall - 8, 10)
                            a = best_a * np.exp(rng.normal(0, scale, n))
                            a = normalize_simplex(a)
                            v = np.zeros(n)
                            stall = 0
                            step = min(step * 0.9, 0.05)
            else:
                stall += 1
        else:
            # LP failed, fallback to smooth max
            grad, cmax, weights = smooth_max_subgradient(a, 0.01)
            grad_centered = grad - np.mean(grad)
            gn = np.linalg.norm(grad_centered)
            if gn > 1e-15:
                a_new = a - 0.05 * grad_centered / gn
                a_new = normalize_simplex(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new

        step *= 0.998

    return best_a, best_obj


def combined_optimize(a, time_budget=60.0):
    """
    Combined optimization: alternate between LP-guided and smooth subgradient.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    mode = 0  # 0 = LP, 1 = subgradient
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        alloc = min(remaining * 0.3, 30.0)

        if mode == 0 and n <= 900:
            a_new, obj_new = lp_guided_search(a, max_iters=30, time_budget=alloc)
        else:
            a_new, obj_new = adaptive_subgradient_v2(a, max_iters=int(alloc * 15), time_budget=alloc)

        mode = 1 - mode

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 3:
                # Perturbation
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                scale = 0.02 + 0.01 * (stall % 5)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize_simplex(a_pert)
                a = a_pert
                stall = 0

    return best_a, best_obj


def generate_structured_starts(n, seed=42):
    """
    Generate diverse structured starts:
    - Uniform, triangular, Gaussian
    - Golay-inspired (from complementary sequence structure)
    - CAZAC-inspired (flat spectrum target)
    - Random with different sparsity levels
    """
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # 1. Uniform
    starts.append(np.ones(n) / n)

    # 2. Triangular
    tri = 1.0 - np.abs(i - center) / (center + 1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # 3. Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 8.0) ** 2))
    starts.append(bell / np.sum(bell))

    # 4. Gaussian (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # 5. Sinusoidal modulation of uniform
    for freq in [2, 3, 5]:
        mod = 1.0 + 0.3 * np.cos(2 * np.pi * freq * i / n)
        mod = np.maximum(mod, 0.05)
        starts.append(mod / np.sum(mod))

    # 6. Golay-inspired: use m-sequence like structure (mapped to non-negative)
    # Generate a pseudo-random binary sequence, map to [0.5, 1.0]
    mseq = 1.0 + 0.5 * np.sign(np.sin(2 * np.pi * 0.618 * i + rng.random() * 2 * np.pi))
    starts.append(mseq / np.sum(mseq))

    # 7. Power-law decay from center
    plaw = np.exp(-np.abs(i - center) / (n * 0.3))
    starts.append(plaw / np.sum(plaw))

    # 8. Two-peak structure
    two_peak = np.exp(-((i - n * 0.25) ** 2) / (2 * (n * 0.1) ** 2)) + \
               np.exp(-((i - n * 0.75) ** 2) / (2 * (n * 0.1) ** 2))
    starts.append(two_peak / np.sum(two_peak))

    # 9-15: Random with varying smoothness
    for smoothness in [0.1, 0.3, 0.5, 0.7, 0.9, 0.95, 0.99]:
        raw = np.abs(rng.normal(0, 1, n))
        if smoothness > 0.5:
            # Low-pass filter via FFT
            fft_size = next_pow2(n)
            F = rfft(raw, fft_size)
            cutoff = int(fft_size * (1 - smoothness))
            F[cutoff:] = 0
            raw = np.maximum(irfft(F, fft_size)[:n], 0.01)
        starts.append(raw / np.sum(raw))

    # Deduplicate
    unique = [starts[0]]
    for s in starts[1:]:
        if all(np.linalg.norm(s - u) > 0.02 for u in unique):
            unique.append(s)

    return unique


def next_pow2(x):
    """Next power of 2 >= x."""
    return 1 << (x - 1).bit_length()


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500.0):
    """
    Main search function combining LP-guided optimization, adaptive subgradient,
    structured multi-start, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # Phase 1: Quick optimization of parent
    print("\n=== Phase 1: Optimize parent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.15, 120.0)
        a_opt, obj_opt = combined_optimize(best_seq, time_budget=t_alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After quick opt: {best_score:.8f}")

    # Phase 2: Multi-start exploration with diverse n
    print(f"\n=== Phase 2: Multi-start exploration (best: {best_score:.8f}) ===")

    n_values = [n_parent]
    for dn in [-100, -50, 50, 100, -150, 150, -200, 200, -300, 300]:
        nv = n_parent + dn
        if 400 <= nv <= 1200:
            n_values.append(nv)
    n_values = sorted(set(n_values))

    candidates = []  # (obj, seq)

    for n in n_values:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Generate starts
        if n == n_parent and best_seq is not None:
            starts = [normalize_simplex(best_seq.copy())]
            starts.extend(generate_structured_starts(n, seed=seed))
        else:
            starts = generate_structured_starts(n, seed=seed + n * 31)

        # Screen each start
        alloc_per_start = min(remaining / (len(starts) + 1), 20.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(alloc_per_start, 15.0)
            if t_alloc < 3:
                break

            # Quick subgradient
            a_opt, obj_opt = adaptive_subgradient_v2(
                s, max_iters=int(t_alloc * 12), time_budget=t_alloc * 0.5)

            # If promising, do a bit more
            if obj_opt < best_score + 0.01 and time.monotonic() < deadline - 100:
                remaining2 = deadline - time.monotonic()
                if remaining2 > 30:
                    a_opt2, obj_opt2 = combined_optimize(
                        a_opt, time_budget=min(remaining2 * 0.2, 20.0))
                    if obj_opt2 < obj_opt:
                        a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")

            candidates.append((obj_opt, a_opt.copy(), n))

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 3: Intensive optimization of top candidates
    print(f"\n=== Phase 3: Intensive optimization (best: {best_score:.8f}) ===")

    top_k = min(6, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        obj_cand, a_cand, n_cand = candidates[ci]
        if obj_cand > best_score + 0.005:
            break  # Not worth optimizing

        alloc = min(remaining * 0.25, 100.0)
        print(f"  Cand {ci} (n={n_cand}, obj={obj_cand:.8f}): {alloc:.0f}s budget")

        a_pol, obj_pol = combined_optimize(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")

    # Phase 4: Length exploration
    print(f"\n=== Phase 4: Length exploration (best: {best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = [n_best - 50, n_best + 50, n_best - 100, n_best + 100,
                        n_best + 200, n_best - 200]
        test_lengths = [x for x in test_lengths if 400 <= x <= 1300]
        test_lengths = sorted(set(test_lengths))

        for n_test in test_lengths:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_res = resample_sequence(best_seq, n_test)
            a_res = normalize_simplex(a_res)
            alloc = min(remaining * 0.15, 40.0)
            a_pol, obj_pol = combined_optimize(a_res, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** {obj_pol:.8f} **")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final polish (best: {best_score:.8f}) ===")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_final, obj_final = combined_optimize(best_seq, time_budget=remaining * 0.6)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  After polish: {best_score:.8f}")

        # Batch of small perturbations
        remaining = deadline - time.monotonic()
        if remaining > 15 and best_seq is not None:
            n_f = len(best_seq)
            rng = np.random.default_rng(seed + 77777)
            for pi in range(8):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.005 + 0.005 * pi
                a_p = best_seq * np.exp(rng.normal(0, scale, n_f))
                a_p = normalize_simplex(a_p)
                t_a = min(remaining * 0.1, 8.0)
                if t_a < 2:
                    break
                a_p, obj_p = adaptive_subgradient_v2(a_p, max_iters=int(t_a * 15), time_budget=t_a)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Perturb {pi}: ** {best_score:.8f} **")

    # Phase 6: Final LP polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_f = len(best_seq)
        if n_f <= 900 and remaining > 20:
            print(f"\n=== Phase 6: Final LP (n={n_f}) ===")
            a_lp, obj_lp = lp_guided_search(best_seq, max_iters=30, time_budget=min(remaining * 0.5, 50.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
