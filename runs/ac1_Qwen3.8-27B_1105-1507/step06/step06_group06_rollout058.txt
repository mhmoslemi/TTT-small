"""
Optimizes non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft peak-chasing with Nesterov momentum, adaptive softmax temperature,
frequency-domain smoothing, multi-length exploration, and perturbation-based
restarts to escape local minima. Key innovation: adaptive tau scheduling and
dual-gradient (time + frequency domain) updates.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def soft_peak_gradient(a, conv, tau):
    """
    Compute softmax-weighted gradient of max(conv) w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i in [0, n).
    Uses FFT for efficiency.
    """
    n = len(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    w_sum = np.sum(exp_vals)
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w = exp_vals / w_sum

    # grad[i] = 2 * sum_k w[k] * a[k-i] where 0 <= k-i < n
    # This is 2 * cross-correlation of w and a
    # corr[m] = sum_k w[k] * a[k - (m - n + 1)] ... complicated
    # Let's think: we want grad[i] = sum_k w[k] * a[k-i] for max(0, k-n+1) <= i <= min(n-1, k)
    # Equivalently: for each k, add w[k] * a[k-i] to grad[i] for i in [max(0,k-n+1), min(n-1,k)]
    # 
    # Using FFT: let's pad w to length 2n-1 and a to length 2n-1 (or just n)
    # sum_k w[k] * a[k-i] = (w * a_flipped)[n-1+i] where a_flipped = a[::-1]
    # Wait: (w * a_flipped)[m] = sum_j w[j] * a_flipped[m-j] = sum_j w[j] * a[n-1-m+j]
    # We want sum_k w[k] * a[k-i], so set n-1-m+j = k-i => j = k - (k-i) = i... no.
    # 
    # Let me define: b[i] = a[n-1-i] (reversed a)
    # (w * b)[m] = sum_j w[j] * b[m-j] = sum_j w[j] * a[n-1-(m-j)] = sum_j w[j] * a[n-1-m+j]
    # We want: sum_j w[j] * a[j-i]
    # Set n-1-m+j = j-i => n-1-m = -i => m = n-1+i
    # So sum_j w[j] * a[j-i] = (w * b)[n-1+i] where b = a[::-1]
    
    N_fft = 1
    while N_fft < len(conv) + n - 1:
        N_fft <<= 1
    
    fw = rfft(w, N_fft)
    fb = rfft(a[::-1], N_fft)
    corr = irfft(fw * fb, N_fft)
    # grad[i] = 2 * corr[n-1+i] for i = 0, ..., n-1
    grad = 2.0 * corr[n - 1:2 * n - 1]
    
    return grad


def soft_peak_optimize(a, max_iters=500, time_budget=60.0, tau_init=0.001, tau_end=0.01,
                       lr_init=0.1, momentum=0.9, name=""):
    """
    Soft peak chasing with Nesterov momentum and adaptive tau.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    lr = lr_init
    stall_count = 0
    prev_obj = best_obj

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive tau: start small (focus on peak), grow to smooth out
        progress = it / max(max_iters - 1, 1)
        tau = tau_init * (tau_end / tau_init) ** progress

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Scale tau relative to the dynamic range of conv
        conv_range = float(np.max(conv) - np.min(conv))
        if conv_range > 1e-15:
            tau = max(tau, conv_range * 0.01)
            tau = min(tau, conv_range * 2.0)

        grad = soft_peak_gradient(a, conv, tau)
        
        # Center gradient (maintain sum constraint)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov step
        v = momentum * v - lr * grad_centered
        a_new = a + v  # Note: we're maximizing negative objective, so +v
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            a = a_new  # Still move, but don't update best
            stall_count += 1
            lr *= 0.95
            if stall_count > 30:
                # Reset momentum and try a perturbation
                v = np.zeros(n)
                lr = lr_init * 0.5
                a = best_a * np.exp(np.random.default_rng(it).normal(0, 0.05, n))
                a = project_nonneg_sum1(a)
                stall_count = 0

        # Adaptive lr increase if making steady progress
        if stall_count == 0 and it > 10:
            lr = min(lr * 1.02, lr_init)

    return best_a, best_obj


def freq_domain_smooth(a, max_iters=100, time_budget=20.0, alpha=0.5):
    """
    Frequency domain smoothing: iteratively compress the power spectrum
    to flatten the autocorrelation.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = fft(a, N)
        power = np.abs(fa) ** 2

        # Compress power spectrum: reduce peaks
        ps_max = np.max(power)
        if ps_max < 1e-20:
            break

        # Log compression
        ps_norm = power / ps_max
        ps_floor = np.max(ps_norm[ps_norm > 0]) * 1e-3 if np.any(ps_norm > 0) else 1e-10
        ps_clipped = np.maximum(ps_norm, ps_floor)
        # Apply power compression
        ps_compressed = ps_clipped ** alpha

        # Reconstruct with modified magnitude
        phase = np.angle(fa)
        magnitude_old = np.abs(fa)
        magnitude_new = magnitude_old * (ps_compressed / ps_clipped) ** 0.5
        fa_new = magnitude_new * np.exp(1j * phase)

        a_new = ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def freq_root_optimize(a, max_iters=200, time_budget=25.0, blend=0.3, power=0.7):
    """
    Frequency-domain error reduction via complex square root:
    Modify the autocorrelation in frequency domain, then take sqrt to get new a.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 4 * n:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = fft(a, N)
        # The autocorrelation periodic version
        auto = ifft(fa * fa).real

        # Focus on the linear convolution part
        # Periodic auto has the linear conv in the middle
        # For N >= 4n, the linear conv is in indices [0, 2n-2] and [N-2n+2, N-1]

        # Work with the periodic autocorrelation
        c = np.real(ifft(fa * fa))
        
        c_max = np.max(c)
        if c_max < 1e-15:
            break

        # Compress peaks in the autocorrelation
        c_mod = c.copy()
        threshold = c_max * (1.0 - blend * 0.3)
        peak_mask = c > threshold
        if np.any(peak_mask):
            c_mod[peak_mask] = c[peak_mask] * (1 - blend) + threshold * blend

        # Additional power compression
        c_mod = np.maximum(c_mod, 0)
        if np.max(c_mod) > 0:
            c_scaled = c_mod / np.max(c_mod)
            c_mod = c_scaled ** power * np.max(c_mod)

        cf = fft(c_mod, N)
        # Take complex square root
        fa_new = np.sqrt(cf + 0j)
        a_new = ifft(fa_new).real[:n]
        
        # Ensure it's the "right" branch (positive mean)
        if np.mean(a_new) < 0:
            a_new = -a_new

        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj

    ss_idx = 0

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Find top-k peaks
        n_peaks = min(5, len(conv))
        peak_threshold = cmax * 0.9
        peak_mask = conv > peak_threshold
        if not np.any(peak_mask):
            peak_mask = np.zeros(len(conv), dtype=bool)
            peak_mask[int(np.argmax(conv))] = True

        # Weighted gradient
        w = np.zeros(len(conv))
        w[peak_mask] = conv[peak_mask]
        w_sum = w.sum()
        if w_sum < 1e-30:
            w[int(np.argmax(conv))] = 1.0
            w_sum = 1.0
        w /= w_sum

        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_end > i_start:
                indices = np.arange(i_start, i_end)
                a_indices = k - indices
                grad[indices] += w[k] * 2.0 * a[a_indices]

        grad_centered = grad - np.mean(grad)

        # Try different step sizes
        improved = False
        for ss in step_sizes[ss_idx:]:
            a_try = a - ss * grad_centered
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                ss_idx = 0  # Reset to largest step
                break
        if not improved:
            ss_idx = min(ss_idx + 1, len(step_sizes) - 1)
            if ss_idx == len(step_sizes) - 1:
                # Stuck, try perturbation
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = project_nonneg_sum1(a)

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_minimal_peak(a, time_limit=15):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    from scipy.optimize import linprog
    n = len(a)
    if n > 1500:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def lp_alternate_improved(a, max_iters=30, time_budget=40.0):
    """Improved LP alternating with better line search."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
        from scipy.optimize import linprog
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s < 1e-10:
                break
            g = g / s
        except Exception:
            break

        # Line search with golden section
        best_t = 0.5
        best_t_obj = np.inf
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a_try = (1.0 - best_t) * a + best_t * g
            best_a = project_nonneg_sum1(a_try).copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return best_a, best_obj


def generate_analytical_inits(n, rng):
    """Generate diverse analytical initializations."""
    inits = []
    x = np.linspace(0, 1, n)

    # Power law (increasing) - key for reducing center peak
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a = (x + 0.01) ** alpha
        inits.append(a / a.sum())

    # Exponential
    for beta in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(beta * x)
        inits.append(a / a.sum())

    # Cosine window (reduces edges)
    a = (1 + np.cos(np.pi * (x - 0.5) * 2)) ** 0.5 + 0.1
    inits.append(a / a.sum())

    # Skewed cosine: high in middle-left, low at right edge
    a = np.exp(-2 * (x - 0.4) ** 2) + 0.05
    inits.append(a / a.sum())

    # Two-scale: coarse ramp with fine oscillation
    for freq in [10, 30, 100]:
        base = x ** 0.5
        osc = 0.1 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(base + osc, 0.01)
        inits.append(a / a.sum())

    # "Barker-like": alternating high/low with envelope
    envelope = x ** 0.3
    pattern = np.ones(n)
    pattern[::2] = 0.3
    a = envelope * pattern
    inits.append(a / a.sum())

    # Random
    a = rng.random(n) + 0.01
    inits.append(a / a.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular (peak in middle)
    tri = 1 - np.abs(2 * x - 1)
    tri = np.maximum(tri, 0.05)
    inits.append(tri / tri.sum())

    return inits

def dual_domain_optimize(a, max_iters=300, time_budget=60.0):
    """
    Alternating between time-domain soft peak chasing and frequency-domain smoothing.
    Every few steps, switch to frequency domain for a smooth update.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    lr = 0.05
    tau = 0.002
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        
        # Every 20 steps, do a frequency domain smoothing step
        if it % 20 == 19:
            a_fd, obj_fd = freq_domain_smooth(a, max_iters=10, time_budget=2.0, alpha=0.6)
            if obj_fd < best_obj:
                best_obj = obj_fd
                best_a = a_fd.copy()
                a = a_fd.copy()
            continue

        # Time domain: soft peak chasing
        tau = 0.001 * (10.0) ** progress  # Increase tau over time
        conv = compute_conv(a)
        conv_range = float(np.max(conv) - np.min(conv))
        tau = max(tau, conv_range * 0.005)
        tau = min(tau, conv_range * 1.0)

        grad = soft_peak_gradient(a, conv, tau)
        grad_centered = grad - np.mean(grad)
        
        v = 0.9 * v - lr * grad_centered
        a_new = project_nonneg_sum1(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.1)
        else:
            a = a_new
            lr *= 0.97

    return best_a, best_obj


def perturbation_restart(a, n_perturbs=5, rng=None):
    """Try several perturbed versions and keep the best after quick optimization."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

    for p in range(n_perturbs):
        # Different perturbation styles
        style = p % 4
        if style == 0:
            # Multiplicative noise
            a_pert = a * np.exp(rng.normal(0, 0.1, n))
        elif style == 1:
            # Window swap
            wsize = max(5, n // 20)
            w1 = rng.integers(0, n - wsize)
            w2 = rng.integers(0, n - wsize)
            a_pert = a.copy()
            if abs(w1 - w2) > wsize:
                a_pert[w1:w1+wsize], a_pert[w2:w2+wsize] = a_pert[w2:w2+wsize].copy(), a_pert[w1:w1+wsize].copy()
        elif style == 2:
            # Local inversion
            wsize = max(10, n // 10)
            wstart = rng.integers(0, n - wsize)
            a_pert = a.copy()
            a_pert[wstart:wstart+wsize] = a_pert[wstart+wsize-1:wstart-1:-1]
        else:
            # Spike removal and redistribution
            n_remove = max(5, n // 20)
            idx = rng.choice(n, n_remove, replace=False)
            a_pert = a.copy()
            mass = a_pert[idx].sum()
            a_pert[idx] = 0
            fill_idx = rng.choice(n, n_remove, replace=False)
            a_pert[fill_idx] += mass / n_remove

        a_pert = project_nonneg_sum1(a_pert)
        
        # Quick optimization
        a_opt, obj_opt = soft_peak_optimize(a_pert, max_iters=50, time_budget=5.0,
                                            tau_init=0.001, tau_end=0.01,
                                            lr_init=0.05, momentum=0.9)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining soft peak chasing,
    frequency domain methods, LP alternating, and perturbation restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf
    n_best = 700

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent = project_nonneg_sum1(parent)
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()
            n_best = parent.size

    if best_seq is None:
        # Start with power law
        n_init = 700
        x = np.linspace(0, 1, n_init)
        best_seq = (x + 0.01) ** 0.5
        best_seq = project_nonneg_sum1(best_seq)
        best_score = compute_obj(best_seq)
        n_best = n_init
        print(f"Initialized power law (n={n_best}): obj={best_score:.8f}")

    # === Phase 1: Quick screening of initializations (8% budget) ===
    print("\n=== Phase 1: Initialization Screening ===")
    p1_budget = min(100.0, (deadline - start_time) * 0.08)
    p1_start = time.monotonic()

    # Try parent at current length + analytical inits at several lengths
    n_options = [n_best, max(500, n_best - 100), min(2000, n_best + 100),
                 max(400, n_best - 200), min(2500, n_best + 200)]
    n_options = list(dict.fromkeys([n for n in n_options if 300 <= n <= 3000]))

    candidates = []

    # Parent-based candidates
    for n_try in n_options:
        if time.monotonic() - p1_start > p1_budget:
            break
        a_res = project_nonneg_sum1(resample_sequence(best_seq, n_try))
        a_opt, obj_opt = soft_peak_optimize(a_res, max_iters=50, time_budget=3.0,
                                            tau_init=0.001, tau_end=0.01, lr_init=0.05)
        candidates.append((obj_opt, a_opt, n_try))

    # Analytical inits at the best length
    inits = generate_analytical_inits(n_best, rng)
    per_init_budget = max(1.0, (p1_budget - (time.monotonic() - p1_start)) / len(inits))

    for init in inits:
        if time.monotonic() - p1_start > p1_budget:
            break
        a_quick, obj_quick = soft_peak_optimize(init, max_iters=int(per_init_budget * 10),
                                                time_budget=per_init_budget,
                                                tau_init=0.001, tau_end=0.01, lr_init=0.05)
        candidates.append((obj_quick, a_quick, n_best))

    candidates.sort(key=lambda x: x[0])
    print(f"  Screened {len(candidates)} candidates, best: {candidates[0][0]:.8f}")

    for obj_c, a_c, n_c in candidates[:3]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            n_best = n_c
            print(f"  New best from screening: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive dual-domain optimization (45% budget) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"Best so far: {best_score:.8f}, n={n_best}")
    p2_budget = min(650.0, (deadline - time.monotonic()) * 0.45)
    p2_start = time.monotonic()

    a_work = project_nonneg_sum1(best_seq.copy())
    local_best = a_work.copy()
    local_obj = compute_obj(a_work)

    # Main optimization loop: alternate strategies
    n_strategies = 5
    stall_total = 0

    for cycle in range(200):
        if time.monotonic() - p2_start > p2_budget:
            break
        remaining = p2_start + p2_budget - time.monotonic()
        if remaining < 5:
            break

        strat = cycle % n_strategies
        alloc = min(remaining * 0.15, 30.0)

        if strat == 0:
            # Soft peak chasing (main workhorse)
            a_new, obj_new = soft_peak_optimize(a_work, max_iters=100, time_budget=alloc,
                                                tau_init=0.0005, tau_end=0.02,
                                                lr_init=0.08, momentum=0.92)
        elif strat == 1:
            # Dual domain
            a_new, obj_new = dual_domain_optimize(a_work, max_iters=80, time_budget=alloc)
        elif strat == 2:
            # Frequency root
            a_new, obj_new = freq_root_optimize(a_work, max_iters=50, time_budget=min(alloc, 15.0),
                                                blend=0.2 + 0.1 * (cycle // n_strategies % 3),
                                                power=0.6 + 0.05 * (cycle // n_strategies % 4))
        elif strat == 3:
            # LP alternate (only for moderate n)
            if n_best <= 1000:
                a_new, obj_new = lp_alternate_improved(a_work, max_iters=10,
                                                       time_budget=min(alloc, 20.0))
            else:
                a_new, obj_new = freq_domain_smooth(a_work, max_iters=30,
                                                    time_budget=min(alloc, 15.0),
                                                    alpha=0.4 + 0.1 * (cycle % 3))
        else:
            # Fine peak chase
            a_new, obj_new = peak_chase_fine(a_work, n_steps=50, time_budget=min(alloc, 20.0))

        if obj_new < local_obj:
            improvement = local_obj - obj_new
            local_obj = obj_new
            local_best = a_new.copy()
            a_work = a_new.copy()
            stall_total = 0
            if local_obj < best_score:
                best_score = local_obj
                best_seq = local_best.copy()
                print(f"  ** cycle {cycle} strat {strat}: obj={best_score:.8f} (improve {improvement:.2e}) **")
        else:
            stall_total += 1
            if stall_total > 15:
                # Perturbation restart
                a_work, obj_pert = perturbation_restart(a_work, n_perturbs=3, rng=rng)
                if obj_pert < local_obj:
                    local_obj = obj_pert
                    local_best = a_work.copy()
                    if local_obj < best_score:
                        best_score = local_obj
                        best_seq = local_best.copy()
                        print(f"  ** perturbation restart: obj={best_score:.8f} **")
                stall_total = 0

    print(f"  Phase 2 done: local_obj={local_obj:.8f}")

    # === Phase 3: Multi-length fine-tuning (15% budget) ===
    print(f"\n=== Phase 3: Multi-length Tuning ===")
    print(f"Best so far: {best_score:.8f}")
    p3_budget = min(200.0, (deadline - time.monotonic()) * 0.15)
    p3_start = time.monotonic()

    n_current = len(best_seq)
    n_tries = [n_current - 20, n_current + 20, n_current - 50, n_current + 50,
               n_current - 80, n_current + 80, n_current + 120]
    n_tries = [n for n in n_tries if 300 <= n <= 3000]
    n_tries = list(dict.fromkeys(n_tries))

    per_n_budget = max(5.0, p3_budget / len(n_tries))

    for n_try in n_tries:
        if time.monotonic() - p3_start > p3_budget:
            break
        remaining = p3_start + p3_budget - time.monotonic()
        if remaining < 5:
            break

        a_res = project_nonneg_sum1(resample_sequence(best_seq, n_try))
        
        # Optimize at this length
        a_opt, obj_opt = soft_peak_optimize(a_res, max_iters=80,
                                            time_budget=min(per_n_budget * 0.6, 15.0),
                                            tau_init=0.0005, tau_end=0.01, lr_init=0.05)
        if n_try <= 1000:
            a_opt2, obj_opt2 = lp_alternate_improved(a_opt, max_iters=8,
                                                     time_budget=min(per_n_budget * 0.3, 10.0))
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2
        a_opt3, obj_opt3 = peak_chase_fine(a_opt, n_steps=30,
                                           time_budget=min(per_n_budget * 0.2, 8.0))
        if obj_opt3 < obj_opt:
            a_opt, obj_opt = a_opt3, obj_opt3

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f} (no improvement)")

    # === Phase 4: Final intensive polish (remaining budget) ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining = deadline - time.monotonic()
    print(f"Remaining time: {remaining:.1f}s")

    if remaining > 15 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)

        # Round 1: Fine peak chase
        if remaining > 20:
            a_p, obj_p = peak_chase_fine(a_final, n_steps=min(200, int(remaining * 2)),
                                         time_budget=min(remaining * 0.25, 40.0))
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish fine chase: {best_score:.8f}")

        # Round 2: LP if feasible
        remaining = deadline - time.monotonic()
        if remaining > 20 and n_final <= 1000:
            a_p, obj_p = lp_alternate_improved(a_final, max_iters=min(30, int(remaining)),
                                               time_budget=min(remaining * 0.3, 40.0))
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish LP: {best_score:.8f}")

        # Round 3: Frequency root with varied parameters
        remaining = deadline - time.monotonic()
        if remaining > 15:
            for blend in [0.2, 0.4, 0.6]:
                if remaining < 5:
                    break
                a_p, obj_p = freq_root_optimize(a_final, max_iters=50,
                                                time_budget=min(remaining * 0.3, 15.0),
                                                blend=blend, power=0.7)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish freq_root (blend={blend}): {best_score:.8f}")
                remaining = deadline - time.monotonic()

        # Round 4: Very fine peak chase
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_p, obj_p = peak_chase_fine(a_final, n_steps=min(300, int(remaining * 3)),
                                         time_budget=remaining * 0.8)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Polish final chase: {best_score:.8f}")

        # Round 5: Tiny perturbations + re-optimize
        remaining = deadline - time.monotonic()
        if remaining > 10:
            for _ in range(3):
                if remaining < 5:
                    break
                a_pert = a_final * np.exp(rng.normal(0, 0.01, n_final))
                a_pert = project_nonneg_sum1(a_pert)
                a_pert, obj_pert = soft_peak_optimize(a_pert, max_iters=50,
                                                      time_budget=min(remaining * 0.3, 10.0),
                                                      tau_init=0.0003, tau_end=0.005,
                                                      lr_init=0.03, momentum=0.95)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Polish perturbation: {best_score:.8f}")
                remaining = deadline - time.monotonic()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
