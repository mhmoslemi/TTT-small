def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient_conv(a, weights):
    """
    Compute the gradient of sum_k weights[k] * conv(a,a)[k] with respect to a.
    weights is a vector of length 2n-1.
    Returns vector of length n.
    """
    n = len(a)
    size = 1
    while size < 2 * len(weights):
        size <<= 1
    W = rfft(weights, size)
    A = rfft(a, size)
    # grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) (with bounds)
    # This is 2 * (weights * a) (circular convolution, taking first n elements)
    # Actually: sum_k w[k] * a[k-i] = (w * a)[i] (circular)
    # And sum_k w[k] * a[i-k] = (w * a)[-i] = (a * w)[i] (by symmetry of convolution)
    # Since w and a are real, (w*a)[i] = (a*w)[i], so grad = 2 * (w*a)[i]
    cc = irfft(W * A, size)
    return 2.0 * cc[:n]

def project_simplex_nonneg(v, target_sum=1.0):
    """
    Project v onto {x >= 0, sum(x) = target_sum}.
    Uses bisection on the threshold.
    """
    n = len(v)
    # We want to find t such that sum_i max(v[i] - t, 0) = target_sum
    v_sorted = np.sort(v)[::-1]
    
    # Find the water level by scanning
    cumsum = np.cumsum(v_sorted)
    # max(v[i] - t, 0) = target_sum
    # For the top k elements: sum(v[i] for i in top k) - k*t = target_sum
    # t = (sum_top_k - target_sum) / k
    # We need the largest k such that v_sorted[k-1] > t
    
    # Binary search for k
    lo, hi = 1, n
    best_k = 1
    while lo <= hi:
        mid = (lo + hi) // 2
        t = (cumsum[mid-1] - target_sum) / mid
        if t < v_sorted[mid-1]:
            best_k = mid
            lo = mid + 1
        else:
            hi = mid - 1
    
    t = (cumsum[best_k-1] - target_sum) / best_k
    result = np.maximum(v - t, 0)
    # Ensure exact sum
    # Due to floating point, adjust
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        # Distribute the difference to the non-zero elements
        mask = result > 0
        result[mask] += diff / mask.sum()
        result = np.maximum(result, 0)
    return result

Wait, one more important consideration: the `project_simplex_nonneg` function. I need to make sure it's correct and efficient.

The projection onto {x >= 0, sum(x) = 1} is:
- Find t such that sum_i max(v[i] - t, 0) = 1
- Return max(v - t, 0)

The standard algorithm (Chen & Dodziuk, 2001):
1. Sort v in descending order: v_(1) >= v_(2) >= ... >= v_(n)
2. Find the largest rho such that v_(rho) - (sum_{j=1}^{rho} v_(j) - 1) / rho > 0
3. t = (sum_{j=1}^{rho} v_(j) - 1) / rho
4. return max(v - t, 0)

This is O(n log n) due to the sorting. For n=700, this is fast.

Let me implement this:

def project_simplex_nonneg(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    condition = u - (cv - target_sum) / rho > 0
    rho_star = np.max(rho[condition])
    theta = (cv[rho_star - 1] - target_sum) / rho_star
    return np.maximum(v - theta, 0)

Multi-phase optimizer: Nesterov-accelerated projected gradient on smoothed
logsumexp objective, LP fixed-point iteration, exact subgradient polish,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a * a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def conv_grad(a, w):
    """
    Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a.
    w has length 2n-1, a has length n. Returns gradient of length n.
    Result: 2 * (cross_correlation(w, a))[:n]
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    return 2.0 * cc[:n]


def project_simplex_nonneg(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    condition = u - (cv - target_sum) / rho > 0
    if not np.any(condition):
        return np.full(n, target_sum / n)
    rho_star = int(np.max(rho[condition]))
    theta = (cv[rho_star - 1] - target_sum) / rho_star
    return np.maximum(v - theta, 0)

    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    grad = conv_grad(a, w)
    return ls, grad, w

        # Compute smoothed max and gradient at look-ahead point v
        ls, grad, _ = smooth_max_and_grad(v, beta)

        # Gradient of 2n * ls w.r.t. a
        grad_f = 2.0 * n * grad

        # Gradient step + project
        u = v - lr * grad_f
        a_new = project_simplex_nonneg(u, 1.0)

        # Nesterov momentum update
        if it == 0:
            v = a_new.copy()
        else:
            t = it + 1
            mu = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            v = a_new + ((mu - 1.0) / mu) * (a_new - a_prev)
            # Keep v in the feasible set (non-negative, normalized)
            v = np.maximum(v, 0.0)
            sv = np.sum(v)
            if sv > 1e-15:
                v = v / sv
            else:
                v = a_new.copy()

        a_prev = a_new

        # Periodic objective check
        if it % 200 == 0 or it == max_iters - 1:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

    return best_a


def nesterov_exact_subgrad(a_init, max_iters=20000, time_budget=100.0,
                            lr_start=0.02, lr_decay=0.99997,
                            tolerance_frac=0.999):
    """
    Nesterov-accelerated projected subgradient on the exact max objective.
    Uses uniform weights over near-active set.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    v = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        lr = lr_start * (lr_decay ** it)

        # Compute exact max and active set at look-ahead point
        conv_v = compute_conv(v)
        max_val = float(np.max(conv_v))
        threshold = max_val * tolerance_frac
        active = conv_v >= threshold
        if not np.any(active):
            active = np.zeros(len(conv_v), dtype=bool)
            active[np.argmax(conv_v)] = True

        # Uniform weights over active set
        w = np.zeros(len(conv_v))
        w[active] = 1.0 / float(active.sum())

        # Subgradient of 2n * max_conv
        grad = 2.0 * n * conv_grad(v, w)

        # Step + project
        u = v - lr * grad
        a_new = project_simplex_nonneg(u, 1.0)

        # Nesterov momentum
        if it == 0:
            v = a_new.copy()
        else:
            t = it + 1
            mu = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            v = a_new + ((mu - 1.0) / mu) * (a_new - a_prev)
            v = np.maximum(v, 0.0)
            sv = np.sum(v)
            if sv > 1e-15:
                v = v / sv
            else:
                v = a_new.copy()

        a_prev = a_new

        if it % 100 == 0 or it == max_iters - 1:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

    return best_a


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_fixed_point(a_init, max_iters=20, time_budget=60.0, blend=0.5):
    """
    LP fixed-point iteration:
    1. Compute M = max(conv(a,a))
    2. Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
    3. a <- (1-blend)*a + blend*g
    4. Repeat
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=min(10, (time_budget - (time.monotonic() - start)) * 0.3))
        if g is None:
            break

        # Try different blend ratios
        improved = False
        for t in [blend, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try the current blend anyway
            a_try = (1.0 - blend) * a + blend * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            a = a_try

    return best_a


def coordinate_descent(a_init, time_budget=30.0, seed=42):
    """Fine-tune via coordinate descent with focal peak targeting."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Focus on top peaks
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for p in peak_indices:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            for i in range(i_min, i_max + 1):
                if time.monotonic() - start > time_budget:
                    break
                j = p - i
                if j < 0 or j >= n:
                    continue
                # Try moving a small amount from the larger of a[i], a[j]
                # toward reducing the peak
                base = max(a[i], a[j])
                if base < 1e-10:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    for direction in [1, -1]:
                        delta = direction * frac * base
                        a_try = a.copy()
                        if i == j:
                            continue
                        a_try[i] -= delta
                        a_try[j] += delta
                        if a_try[i] < -1e-12 or a_try[j] < -1e-12:
                            continue
                        a_try = np.maximum(a_try, 0)
                        s_try = np.sum(a_try)
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
            if time.monotonic() - start > time_budget:
                break

        if not improved and it > 100:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf
