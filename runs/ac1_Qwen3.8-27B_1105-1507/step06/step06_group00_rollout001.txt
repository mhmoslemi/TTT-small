"""

Key improvements: faster LP solving, adaptive line search, multiple n exploration,
structured restarts, peak chasing, and smoothness-regularized variants.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        # So for fixed j, A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        # But k goes from 0 to 2n-2, so k-j goes from -j to 2n-2-j
        # We need 0 <= k-j < n, i.e., j <= k <= j+n-1
        # And k-j < n means k < j+n, which is k <= j+n-1
        # Also k <= 2n-2
        k_start = j
        k_end = min(j + n, 2 * n - 1)  # exclusive
        # a[k-j] for k in [k_start, k_end)
        # k-j ranges from 0 to k_end-1-j
        A[k_start:k_end, j] = a[:k_end - j]
    return A


def solve_lp_minpeak_clarabel(a):
    """Solve min-peak LP using CLARABEL. Returns (g, M) or (None, None)."""
    n = len(a)
    A_mat = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(M.value)
    except Exception:
        pass

    # Fallback to ECOS
    try:
        prob.solve(solver=cp.ECOS)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(M.value)
    except Exception:
        pass

    # Fallback to SCS
    try:
        prob.solve(solver=cp.SCS)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(M.value)
    except Exception:
        pass

    return None, None

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def lp_alternate_step(a, use_scipy=False):
    """One step of LP alternating. Returns (g, M)."""
    if use_scipy:
        return solve_lp_minpeak_scipy(a)
    return solve_lp_minpeak_clarabel(a)


def line_search(a, g, n_candidates=8):
    """
    Find best t in (0, 1] for a_new = (1-t)*a + t*g.
    Returns (best_t, best_a, best_obj).
    """
    best_t = 0.0
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Log-spaced and linear candidates
    ts = np.unique(np.concatenate([
        np.linspace(0.05, 1.0, n_candidates),
        np.array([0.1, 0.25, 0.5, 0.75, 1.0])
    ]))

    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()

    return best_t, best_a, best_obj


def alt_optimize(a, time_budget=60.0, verbose=False, use_scipy=False):
    """
    LP alternating optimization with adaptive step size and restart.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_step = 0.5
    max_stall = 15

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = lp_alternate_step(a, use_scipy)
        if g_lp is None:
            break

        # Line search
        t_best, a_new, obj_new = line_search(a, g_lp, n_candidates=10)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            last_step = t_best
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"    alt it{it}: obj={best_obj:.8f}, t={t_best:.3f}")
        else:
            stall_count += 1
            if stall_count >= max_stall:
                # Perturb
                rng = np.random.default_rng(it * 7919 + n)
                scale = 0.01 * (1 + stall_count // max_stall)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                max_stall = min(max_stall + 5, 40)

    return best_a

"""
LP-alternating optimization with multiple improvements for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Uses efficient LP solving, adaptive line search, peak chasing,
n-exploration, and structured restarts.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    n_out = len(a) + len(b) - 1
    size = 1
    while size < n_out:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n_out]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fft_convolve(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_direct(a, solver='auto'):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    if solver in ('auto', 'hiGHS'):
        try:
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(0, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(res.x[n])
        except Exception:
            pass

    if solver in ('auto', 'clarabel'):
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=cp.CLARABEL)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g
