"""

backtracking line search, LP polishing, and multi-length exploration.
Starts from parent sequence and focuses iterative refinement.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def fft_cross_corr(w, a):
    """Compute cross-correlation: corr[i] = sum_j a[j] * w[i+j] for i=0..n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    corr = irfft(A * np.conj(W), size)
    return corr[:n]


def build_toeplitz_convolve_mat(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=30):
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=30):
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def smoothed_subgradient_step(a, tau):
    """One step of smoothed subgradient. Returns (grad, weights)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    grad = 2.0 * fft_cross_corr(weights, a)
    return grad, weights, conv


def backtracking_line_search(a, direction, f_current, c0=1.0, rho=0.7, sigma=0.1,
                             max_iter=30):
    """Backtracking line search. Returns (t, a_new, f_new)."""
    n = len(a)
    c = c0
    for _ in range(max_iter):
        a_try = (1.0 - c) * a + c * direction
        a_try = normalize_simplex(a_try)
        f_try = compute_obj(a_try)
        if f_try <= f_current - sigma * c * _dir_magnitude(a, direction):
            return c, a_try, f_try
        c *= rho
    return c, a_try, f_try


def _dir_magnitude(a, direction):
    """Approximate magnitude of direction in the objective landscape."""
    diff = direction - a
    return max(np.linalg.norm(diff), 1e-10)


def optimize_subgradient(a, time_budget=60.0, initial_tau=0.005, tau_decay=0.9997,
                         initial_step=0.3, step_decay=0.999):
    """Smoothed subgradient with backtracking and adaptive temperature."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = initial_tau
    step = initial_step
    v = np.zeros(n)
    momentum = 0.75

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        grad, weights, conv = smoothed_subgradient_step(a, tau)
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break

        # Update momentum
        v = momentum * v - grad_centered / gn
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm

        # Direction: move against the gradient (to reduce peaks)
        direction = a - step * v

        # Backtracking line search
        c, a_new, f_new = backtracking_line_search(a, direction, best_obj,
                                                    c0=1.0, rho=0.6, sigma=0.01,
                                                    max_iter=15)

        if f_new < best_obj:
            best_obj = f_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = normalize_simplex(a - 0.1 * step * v)
            f_try = compute_obj(a_try)
            if f_try < best_obj:
                best_obj = f_try
                best_a = a_try.copy()
                a = a_try
            else:
                step *= 0.95

        # Adapt parameters
        tau *= tau_decay
        if it % 100 == 99:
            step = max(0.05, step * 0.98)
            momentum = min(0.9, momentum + 0.001)

    return best_a, best_obj

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    print("\n=== Phase 2: LP-focused optimization ===")
    remaining = deadline - start_time
    if remaining > 60 and len(best_seq) <= 900:
        alloc2 = min(remaining * 0.2, 200.0)
        a_lp, obj_lp = lp_alternate_optimize(best_seq, time_budget=alloc2, use_minpeak=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"Phase 2 (min-peak LP): obj={best_score:.8f} ** IMPROVEMENT **")
        else:
            print(f"Phase 2 (min-peak LP): {obj_lp:.8f} (no improvement)")

        remaining = deadline - start_time
        if remaining > 30:
            alloc2b = min(remaining * 0.15, 100.0)
            a_lp2, obj_lp2 = lp_alternate_optimize(best_seq, time_budget=alloc2b, use_minpeak=False)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"Phase 2b (max-sum LP): obj={best_score:.8f} ** IMPROVEMENT **")
            else:
                print(f"Phase 2b (max-sum LP): {obj_lp2:.8f} (no improvement)")

    # Phase 3: Try nearby lengths
    print("\n=== Phase 3: Length exploration ===")
    n_best = len(best_seq)
    test_lengths = sorted(set([
        n_best - 50, n_best + 50, n_best - 100, n_best + 100,
        n_best - 150, n_best + 150, n_best - 200, n_best + 200
    ]))
    test_lengths = [x for x in test_lengths if 300 <= x <= 1200]

    for n_test in test_lengths:
        remaining = deadline - start_time
        if remaining < 40:
            break
        if n_test == n_best:
            continue

        a_resampled = normalize_simplex(resample_sequence(best_seq, n_test))
        obj_init = compute_obj(a_resampled)

        alloc = min(remaining * 0.12, 50.0)
        a_pol, obj_pol = hybrid_polish(a_resampled, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  n={n_test}: obj={best_score:.8f} ** NEW BEST **")
        else:
            print(f"  n={n_test}: {obj_pol:.8f} (init={obj_init:.8f}, no improvement)")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best={best_score:.8f}) ===")
    remaining = deadline - start_time
    if remaining > 30:
        alloc4 = min(remaining * 0.5, 200.0)
        a_final, obj_final = hybrid_polish(best_seq, time_budget=alloc4)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"Phase 4: obj={best_score:.8f} ** IMPROVEMENT **")
        else:
            print(f"Phase 4: {obj_final:.8f} (no improvement)")

    # Phase 5: Targeted perturbations
    print(f"\n=== Phase 5: Perturbation search (best={best_score:.8f}) ===")
    remaining = deadline - start_time
    if remaining > 20:
        n_final = len(best_seq)
        rng = np.random.default_rng(seed + 12345)

        for pidx in range(min(8, int(remaining / 10))):
            if time.monotonic() > deadline - 10:
                break
            # Different perturbation strategies
            if pidx % 3 == 0:
                # Small multiplicative noise
                scale = 0.005 + 0.005 * pidx
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
            elif pidx % 3 == 1:
                # Reshape: multiply by a gentle curve
                i = np.arange(n_final)
                curve = 1.0 + 0.1 * np.sin(2 * np.pi * (pidx + 1) * i / n_final)
                a_pert = best_seq * curve
            else:
                # Local modification: modify a random block
                block_size = max(10, n_final // 20)
                start_idx = rng.integers(0, n_final - block_size)
                a_pert = best_seq.copy()
                a_pert[start_idx:start_idx + block_size] *= (0.8 + 0.4 * rng.random())
            
            a_pert = normalize_simplex(a_pert)
            t_alloc = min(remaining * 0.08, 15.0)
            if t_alloc < 5:
                break

            a_pert, obj_pert = optimize_subgradient(a_pert, time_budget=t_alloc,
                                                     initial_tau=0.002, initial_step=0.2)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {pidx}: obj={best_score:.8f} ** IMPROVEMENT **")

    # Phase 6: One final LP shot if time permits
    remaining = deadline - start_time
    if remaining > 20 and len(best_seq) <= 900:
        print(f"\n=== Phase 6: Final LP (best={best_score:.8f}) ===")
        alloc6 = min(remaining * 0.6, 60.0)
        a_final_lp, obj_final_lp = lp_alternate_optimize(best_seq, time_budget=alloc6, use_minpeak=True)
        if obj_final_lp < best_score:
            best_score = obj_final_lp
            best_seq = a_final_lp.copy()
            print(f"Phase 6: obj={best_score:.8f} ** IMPROVEMENT **")
        else:
            print(f"Phase 6: {obj_final_lp:.8f} (no improvement)")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
        print(f"Final validated score: {best_score:.8f}")
    elif not np.isfinite(val):
        print("WARNING: Final validation failed, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
