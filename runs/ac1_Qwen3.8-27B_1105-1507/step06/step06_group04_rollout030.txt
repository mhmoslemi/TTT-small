"""
Coordinate descent with incremental convolution updates, combined with
adaptive subgradient refinement, multi-start diverse search, and length
exploration. The key innovation is exact 1D coordinate optimization via
bounded scalar minimization, which provides more precise local improvements
than subgradient methods, enabling escape from the 1.506547 plateau.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def coord_descent(a_init, time_budget=120.0, seed=42, max_cycles=500):
    """
    Exact coordinate descent with incremental convolution updates.
    Each coordinate is optimized by solving a 1D bounded minimization problem.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    best_obj = 2.0 * n * max_conv
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    order = np.arange(n)
    start_time = time.monotonic()
    no_improve_cycles = 0
    prev_cycle_best = best_obj

    for cycle in range(max_cycles):
        if time.monotonic() - start_time > time_budget - 3:
            break

        rng.shuffle(order)
        cycle_improved = False

        for i in order:
            if time.monotonic() - start_time > time_budget - 3:
                break

            x_old = a[i]
            if x_old < 1e-15:
                # Coordinate is ~0, skip (or try small perturbation)
                continue

            # Precompute B, A for the 1D subproblem
            # new_conv[m] = B[m] + A[m]*x + Q[m]*x^2
            # where Q is 1 only at m=2i (if valid)
            B = conv.copy()
            A = np.zeros(2 * n - 1)
            has_quad = (2 * i < 2 * n - 1)

            # For j in [0, min(n, 2n-1-i)): m = i+j
            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:
                js = np.arange(j_max)
                ms = i + js
                a_vals = a[js]

                # Mask for j != i
                mask = (js != i)
                if np.any(mask):
                    ms_aff = ms[mask]
                    a_aff = a_vals[mask]
                    B[ms_aff] -= 2.0 * x_old * a_aff
                    A[ms_aff] = 2.0 * a_aff

                # Handle j == i (m = 2i)
                if i < j_max:
                    B[2 * i] -= x_old * x_old

            # Bounds for x
            x_max = max(3.0 * x_old + 0.01, 0.1)
            x_min = 0.0

            # Define 1D objective
            denom_base = 1.0 - x_old  # sum of all other elements

            def f_1d(x):
                vals = B + A * x
                if has_quad:
                    vals[2 * i] += x * x
                denom = denom_base + x
                if denom < 1e-15:
                    return 1e10
                return np.max(vals) / (denom * denom)

            # Check if current value is already near-optimal
            f_current = f_1d(x_old)

            # Try a few quick candidate values before full optimization
            candidates = [x_old * 0.5, x_old * 0.8, x_old * 1.0, x_old * 1.2, x_old * 1.5,
                          x_old * 0.7, x_old * 1.3]
            candidates = [c for c in candidates if x_min <= c <= x_max]
            if not candidates:
                candidates = [x_old]

            best_cand_val = f_current
            best_cand_x = x_old
            for c in candidates:
                fc = f_1d(c)
                if fc < best_cand_val - 1e-14:
                    best_cand_val = fc
                    best_cand_x = c

            # Full bounded minimization around the best candidate
            # Use a narrow range around the best candidate for speed
            tol = max(0.05 * x_old, 0.01)
            lo = max(x_min, best_cand_x - tol)
            hi = min(x_max, best_cand_x + tol)

            res = minimize_scalar(f_1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-9, 'maxiter': 40})
            x_new = float(res.x)

            # Also try the wider range if the narrow search hit a boundary
            if x_new <= lo + 1e-8 or x_new >= hi - 1e-8:
                res2 = minimize_scalar(f_1d, bounds=(x_min, x_max), method='bounded',
                                       options={'xatol': 1e-8, 'maxiter': 30})
                x_new2 = float(res2.x)
                if f_1d(x_new2) < f_1d(x_new):
                    x_new = x_new2

            delta = x_new - x_old
            if abs(delta) < 1e-14:
                continue

            # Update a[i]
            a[i] = x_new

            # Update convolution incrementally
            # For m where 0<=m-i<n:
            #   if m != 2i: conv[m] += 2*delta*a[m-i] (a[m-i] unchanged since m-i != i)
            #   if m == 2i: conv[m] += delta*(x_old + x_new)
            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:
                js = np.arange(j_max)
                ms = i + js
                a_vals = a[js].copy()  # a hasn't changed at these indices (except i)

                mask = (js != i)
                if np.any(mask):
                    ms_aff = ms[mask]
                    conv[ms_aff] += 2.0 * delta * a_vals[mask]

                if i < j_max:
                    conv[2 * i] += delta * (x_old + x_new)

            # Renormalize
            current_sum = 1.0 + delta
            if current_sum < 1e-15:
                a = np.ones(n) / n
                conv = compute_conv(a)
                s = 1.0
                continue

            scale = 1.0 / current_sum
            a = a * scale
            conv = conv * (scale * scale)
            s = 1.0

            # Check improvement
            new_max = float(np.max(conv))
            obj = 2.0 * n * new_max
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                cycle_improved = True

        # End for i in order

        if cycle_improved:
            no_improve_cycles = 0
        else:
            no_improve_cycles += 1
            if no_improve_cycles > 15:
                # Perturbation to escape local optimum
                rng_pert = np.random.default_rng(seed + cycle)
                scale_pert = 0.02 + 0.01 * (no_improve_cycles % 5)
                a_pert = best_a * np.exp(rng_pert.normal(0, scale_pert, n))
                a_pert = normalize_simplex(a_pert)
                obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    conv = compute_conv(a)
                else:
                    # Keep best, reset
                    a = best_a.copy()
                    conv = compute_conv(a)
                no_improve_cycles = 0

    return best_a, best_obj


def subgradient_fft(a_init, max_iters=3000, time_budget=30.0, seed=42):
    """
    FFT-accelerated subgradient with adaptive temperature and momentum.
    """
    n = len(a_init)
    a = normalize_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start_time = time.monotonic()

    v = np.zeros(n)
    mu = 0.8
    step = 0.05
    tau = 0.01

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        mean_conv = np.mean(conv)

        # Adaptive temperature: decrease as we get closer to flat
        spread = cmax - mean_conv
        tau = max(0.003, spread / (2 * n) * 0.5)

        shifted = np.clip((conv - cmax) / tau, -400.0, 400.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via FFT
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step size
        step *= 0.9995
        if obj > best_obj + 1e-6:
            step *= 0.9
            a = best_a.copy()
            v = v * 0.5

    return best_a, best_obj


def chase_top_peaks(a_init, num_peaks=5, time_budget=20.0, seed=42):
    """
    Target the top-K peaks with weighted subgradient and line search.
    """
    n = len(a_init)
    a = normalize_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start_time = time.monotonic()
    step = 0.02

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(500):
        if time.monotonic() - start_time > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        mean_c = np.mean(conv)

        # Weight peaks by their excess
        excess = np.maximum(conv - mean_c, 0)
        total_excess = np.sum(excess)
        if total_excess < 1e-15:
            break
        weights = excess / total_excess

        # Also boost the top peaks
        top_k = np.argsort(conv)[-num_peaks:]
        for k in top_k:
            weights[k] += 0.5
        weights /= np.sum(weights)

        # Compute gradient
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break

        # Line search
        best_t = step
        best_obj_t = best_obj
        best_a_t = a.copy()
        for t in [step, step * 0.5, step * 0.25, step * 2, step * 0.75]:
            a_try = a - t * grad_centered / gn
            a_try = normalize_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
                best_a_t = a_try.copy()

        a = best_a_t
        if best_obj_t < best_obj - 1e-13:
            best_obj = best_obj_t
            best_a = a.copy()
        else:
            step *= 0.8

    return best_a, best_obj


def parametric_init(n, mode=0, rng=None):
    """Generate diverse parametric initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if mode == 0:  # Uniform
        a = np.ones(n)
    elif mode == 1:  # Bell
        sigma = n * (0.2 + 0.3 * rng.random())
        a = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
    elif mode == 2:  # Modulated bell
        sigma = n * 0.3
        freq = int(1 + rng.integers(0, 10))
        amp = 0.3 + 0.5 * rng.random()
        phase = 2 * np.pi * rng.random()
        bell = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = bell * np.maximum(mod, 0.1)
    elif mode == 3:  # Sum of 2-3 Gaussians
        num_gauss = 2 + rng.integers(0, 2)
        a = np.zeros(n)
        for _ in range(num_gauss):
            mu = n * rng.random()
            sig = n * (0.05 + 0.15 * rng.random())
            amp = 0.5 + rng.random()
            a += amp * np.exp(-((i - mu) ** 2) / (2 * sig ** 2))
    elif mode == 4:  # Pure modulation (no envelope)
        freqs = sorted(rng.integers(1, 12, size=2 + rng.integers(0, 3)).tolist())
        a = np.ones(n)
        for f in freqs:
            amp = 0.2 + 0.3 * rng.random()
            phase = 2 * np.pi * rng.random()
            a = a * (1.0 + amp * np.cos(2 * np.pi * f * i / n + phase))
        a = np.maximum(a, 0.01)
    elif mode == 5:  # Linear ramp + modulation
        ramp = np.linspace(0.3, 1.0, n)
        freq = int(1 + rng.integers(0, 6))
        amp = 0.2 + 0.4 * rng.random()
        phase = 2 * np.pi * rng.random()
        a = ramp * (1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase))
        a = np.maximum(a, 0.01)
    else:  # Random smooth (filtered noise)
        a = rng.random(n)
        # Smooth with a simple moving average
        k = max(3, n // 100)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def combined_optimize(a_init, time_budget=60.0, seed=42):

    """

    if cd_budget > 5:
        a_cd, obj_cd = coord_descent(a, time_budget=cd_budget, seed=seed)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            stall = 0
        else:
            stall += 1

    remaining = time_budget - (time.monotonic() - start_time)
    if remaining < 10:
        return best_a, best_obj

    # Phase B: Subgradient

    if sg_budget > 3:

            best_obj = obj_sg
            best_a = a_sg.copy()
            stall = 0

    remaining = time_budget - (time.monotonic() - start_time)
    if remaining < 10:
        return best_a, best_obj

    # Phase C: Peak chasing

    if pc_budget > 3:

        if obj_pc < best_obj:

            best_a = a_pc.copy()
            stall = 0

    remaining = time_budget - (time.monotonic() - start_time)

    try:

        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):

        seen_n.add(n_cand)

        alloc = min(remaining * 0.4, 180.0)

              f"intensive with {alloc:.0f}s")

                                     seed=seed + 1000 + ci)
        if obj_cd < obj_cand:

                                           seed=seed + 2000 + ci)
            if obj_sg < obj_cand:

            if remaining3 > 20:

                                               seed=seed + 3000 + ci)

            if remaining2 > 15 and obj_pol < best_score + 0.001:

                                             seed=seed + 6000 + n_test)

            if obj_pol < best_score:

            else:

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")

    if best_seq is not None:

        if remaining > 30:
            # Long coordinate descent
            cd_budget = min(remaining * 0.5, 120.0)
            a_final, obj_final = coord_descent(best_seq, time_budget=cd_budget,
                                               seed=seed + 9000)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Coord descent: {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_sg, obj_sg = subgradient_fft(best_seq, time_budget=min(remaining * 0.3, 30),
                                               seed=seed + 9500)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    print(f"  Subgradient: {best_score:.8f}")

                # One more coord descent

                if remaining > 15:

                                                 seed=seed + 9800)
                    if obj_cd < best_score:
                        best_score = obj_cd
                        best_seq = a_cd.copy()
                        print(f"  Final coord: {best_score:.8f}")

            # Try a few perturbations
            remaining = deadline - time.monotonic()
            if remaining > 15:
                n_final = len(best_seq)

                    scale_p = 0.015 + 0.01 * pert_idx
                    a_p = best_seq * np.exp(rng_pert.normal(0, scale_p, n_final))
                    a_p = normalize_simplex(a_p)

                    a_p, obj_p = coord_descent(a_p, time_budget=t_alloc,
                                               seed=seed + 80000 + pert_idx)

                        best_seq = a_p.copy()
                        print(f"  Perturb {pert_idx} (s={scale_p:.3f}): {best_score:.8f}")

    # Phase 5: Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif np.isinf(val):

        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result

```python
"""
Coordinate descent with incremental convolution updates, combined with
adaptive subgradient refinement, multi-start diverse search, and length
exploration. The key innovation is exact 1D coordinate optimization via
bounded scalar minimization, which provides more precise local improvements
than subgradient methods, enabling escape from the 1.506547 plateau.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def coord_descent(a_init, time_budget=120.0, seed=42, max_cycles=500):
    """
    Exact coordinate descent with incremental convolution updates.
    Each coordinate is optimized by solving a 1D bounded minimization problem.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    best_obj = 2.0 * n * max_conv
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    order = np.arange(n)
    start_time = time.monotonic()
    no_improve_cycles = 0

    for cycle in range(max_cycles):
        if time.monotonic() - start_time > time_budget - 3:
            break

        rng.shuffle(order)
        cycle_improved = False

        for i in order:
            if time.monotonic() - start_time > time_budget - 3:
                break

            x_old = a[i]
            if x_old < 1e-15:
                continue

            B = conv.copy()
            A = np.zeros(2 * n - 1)
            has_quad = (2 * i < 2 * n - 1)

            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:
                js = np.arange(j_max)
                ms = i + js
                a_vals = a[js]

                mask = (js != i)
                if np.any(mask):
                    ms_aff = ms[mask]
                    a_aff = a_vals[mask]
                    B[ms_aff] -= 2.0 * x_old * a_aff
                    A[ms_aff] = 2.0 * a_aff

                if i < j_max:
                    B[2 * i] -= x_old * x_old

            x_max = max(3.0 * x_old + 0.01, 0.1)
            x_min = 0.0
            denom_base = 1.0 - x_old

            def f_1d(x):
                vals = B + A * x
                if has_quad:
                    vals[2 * i] += x * x
                denom = denom_base + x
                if denom < 1e-15:
                    return 1e10
                return np.max(vals) / (denom * denom)

            # Quick candidate check
            candidates = [x_old * 0.5, x_old * 0.7, x_old * 0.85,
                          x_old * 1.0, x_old * 1.15, x_old * 1.3, x_old * 1.5]
            candidates = [c for c in candidates if x_min <= c <= x_max]
            if not candidates:
                candidates = [x_old]

            best_cand_val = f_1d(x_old)
            best_cand_x = x_old
            for c in candidates:
                fc = f_1d(c)
                if fc < best_cand_val - 1e-14:
                    best_cand_val = fc
                    best_cand_x = c

            # Narrow bounded minimization
            tol = max(0.08 * x_old, 0.005)
            lo = max(x_min, best_cand_x - tol)
            hi = min(x_max, best_cand_x + tol)

            res = minimize_scalar(f_1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-9, 'maxiter': 35})
            x_new = float(res.x)

            if x_new <= lo + 1e-8 or x_new >= hi - 1e-8:
                res2 = minimize_scalar(f_1d, bounds=(x_min, x_max), method='bounded',
                                       options={'xatol': 1e-8, 'maxiter': 25})
                x_new2 = float(res2.x)
                if f_1d(x_new2) < f_1d(x_new):
                    x_new = x_new2

            delta = x_new - x_old
            if abs(delta) < 1e-14:
                continue

            a[i] = x_new

            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:
                js = np.arange(j_max)
                ms = i + js
                a_vals = a[js].copy()
                mask = (js != i)
                if np.any(mask):
                    ms_aff = ms[mask]
                    conv[ms_aff] += 2.0 * delta * a_vals[mask]
                if i < j_max:
                    conv[2 * i] += delta * (x_old + x_new)

            current_sum = 1.0 + delta
            if current_sum < 1e-15:
                a = np.ones(n) / n
                conv = compute_conv(a)
                continue

            scale = 1.0 / current_sum
            a = a * scale
            conv = conv * (scale * scale)

            new_max = float(np.max(conv))
            obj = 2.0 * n * new_max
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                cycle_improved = True

        if cycle_improved:
            no_improve_cycles = 0
        else:
            no_improve_cycles += 1
            if no_improve_cycles > 12:
                rng_pert = np.random.default_rng(seed + cycle * 7)
                scale_pert = 0.015 + 0.01 * (no_improve_cycles % 6)
                a_pert = best_a * np.exp(rng_pert.normal(0, scale_pert, n))
                a_pert = normalize_simplex(a_pert)
                obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    conv = compute_conv(a)
                else:
                    a = best_a.copy()
                    conv = compute_conv(a)
                no_improve_cycles = 0

    return best_a, best_obj


def subgradient_fft(a_init, max_iters=3000, time_budget=30.0, seed=42):
    """FFT-accelerated subgradient with adaptive temperature and momentum."""
    n = len(a_init)
    a = normalize_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start_time = time.monotonic()

    v = np.zeros(n)
    mu = 0.8
    step = 0.05
    tau = 0.01

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        mean_conv = float(np.mean(conv))
        spread = cmax - mean_conv
        tau = max(0.003, spread / (2 * n) * 0.5)

        shifted = np.clip((conv - cmax) / tau, -400.0, 400.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-
