"""
Multi-phase optimizer using FFT-based gradient descent with Anderson acceleration,
homotopy (log-sum-exp smoothing), and coordinate descent polishing.
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.sparse import coo_matrix, csr_matrix
from scipy.optimize import linprog


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        return np.ones(n) / n
    lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_max_grad(a, tau):
    """
    Compute gradient of log-sum-exp(tau * conv(a,a)) / tau.
    Returns (smooth_max_value, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    c_shifted = c - c.max()
    exp_c = np.exp(tau * c_shifted)
    w = exp_c / exp_c.sum()
    smooth_val = c.max() + np.log(exp_c.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (with zero extension)
    # = 2 * (w * a_flipped)[i + n - 1]
    a_flipped = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a_flipped, size)
    conv_wa = np.fft.irfft(W * A, size)
    
    grad = 2.0 * conv_wa[n - 1: 2 * n - 1]
    return smooth_val, grad


def active_set_grad(a, K):
    """
    Compute gradient using top-K active constraints.
    Returns gradient vector.
    """
    n = len(a)
    c = fast_conv(a)
    # Select top K
    if K < len(c):
        top_indices = np.argsort(c)[::-1][:K]
    else:
        top_indices = np.arange(len(c))
    
    w = np.zeros(2 * n - 1)
    w[top_indices] = 1.0 / K  # Normalize
    
    a_flipped = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a_flipped, size)
    conv_wa = np.fft.irfft(W * A, size)
    
    grad = 2.0 * conv_wa[n - 1: 2 * n - 1]
    return grad, c.max()


def anderson_step(a, grad, memory, memory_grads, m_max=5):
    """
    Anderson acceleration step.
    Returns the accelerated iterate.
    """
    n = len(a)
    if len(memory) < 2:
        return a - grad * 0.01  # Default step
    
    m = min(len(memory), m_max)
    # Use last m iterates
    A_mem = np.stack(memory[-m:])  # shape (m, n)
    G_mem = np.stack(memory_grads[-m:])  # shape (m, n)
    
    # Solve: min_alpha sum_j alpha_j * ||G(a_j) - a_j||^2
    # s_j = G(a_j) - a_j (the fixed-point residual)
    S = G_mem - A_mem  # shape (m, n)
    
    # Gramm-Schmidt or least squares
    # min ||sum alpha_j * S_j||^2 s.t. sum alpha_j = 1
    # This is a quadratic program with equality constraint
    
    # Using the formula: alpha = B^+ e where B_{ij} = <s_i, s_j>
    B = S @ S.T  # shape (m, m)
    e = np.ones(m)
    
    # Solve B * alpha = e (constrained to sum = 1)
    # Use Lagrange: B*alpha + lambda*e = 0, e^T*alpha = 1
    # Augmented system:
    M = np.zeros((m + 1, m + 1))
    M[:m, :m] = B
    M[:m, m] = e
    M[m, :m] = e
    rhs = np.zeros(m + 1)
    rhs[m] = 1.0
    
    try:
        sol = np.linalg.solve(M, rhs)
        alpha = sol[:m]
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m
    
    # Ensure alpha is a probability vector
    alpha = np.maximum(alpha, 0)
    alpha_sum = alpha.sum()
    if alpha_sum < 1e-10:
        alpha = np.ones(m) / m
    else:
        alpha /= alpha_sum
    
    # Compute accelerated point
    a_acc = sum(alpha[j] * A_mem[j] for j in range(m))
    return a_acc


def gradient_descent_anderson(a, max_iters, deadline, K_init=5, K_max=None,
                              tau_init=10.0, tau_max=50.0, lr_init=0.05,
                              m_max=5, verbose=False):
    """
    Gradient descent with Anderson acceleration and homotopy.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    
    if K_max is None:
        K_max = min(2 * n - 1, max(K_init * 10, n))
    
    memory = [a.copy()]
    memory_grads = []
    
    no_improve = 0
    max_no_improve = 200
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Interpolate tau and K
        frac = min(1.0, it / (max_iters * 0.7))
        tau = tau_init + (tau_max - tau_init) * frac
        K = int(K_init + (K_max - K_init) * frac)
        K = max(1, K)
        
        # Compute gradient
        if it % 5 == 0 or K <= 10:
            # Use active set gradient (cheaper, more focused)
            grad, cmax = active_set_grad(a, K)
        else:
            # Use smooth gradient (smoother, more global)
            smooth_val, grad = smooth_max_grad(a, tau)
        
        # Project to simplex tangent space
        grad = grad - np.mean(grad)
        
        # Line search (backtracking)
        lr = lr_init / (1.0 + 0.05 * it ** 0.3)
        found_step = False
        for attempt in range(10):
            a_new = a - lr * grad
            a_new = project_simplex(a_new)
            obj_new = compute_J(a_new)
            if obj_new < best_obj + 1e-10:
                found_step = True
                break
            lr *= 0.5
        
        if not found_step:
            no_improve += 1
            if no_improve > max_no_improve:
                break
            # Try a smaller fixed step
            a_new = project_simplex(a - 0.001 * grad)
        else:
            no_improve = 0
        
        # Anderson acceleration
        memory.append(a_new.copy())
        memory_grads.append(grad.copy())
        if len(memory) > m_max + 1:
            memory = memory[-(m_max + 1):]
            memory_grads = memory_grads[-(m_max + 1):]
        
        # Apply Anderson
        if len(memory) >= 3:
            a_and = anderson_step(a_new, grad, memory, memory_grads, m_max)
            a_and = project_simplex(a_and)
            obj_and = compute_J(a_and)
            if obj_and < obj_new:
                a_new = a_and
        
        a = a_new
        obj = compute_J(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and it % 200 == 0:
            print(f"    GD it {it}: tau={tau:.1f} K={K} obj={best_obj:.8f}", flush=True)
    
    return best_a, best_obj


def coord_descent_fast(a, max_sweeps, deadline, random_order=False, seed=0):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 200
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        improved = False
        
        for i in order:
            old = a[i]
            
            # Determine candidate values
            if old < 1e-12:
                candidates = np.array([0.0, 1e-4, 1e-3, 5e-3, 1e-2])
            else:
                ratios = np.array([0.0, 0.2, 0.5, 0.7, 1.0, 1.3, 1.5, 1.8])
                candidates = np.unique(np.concatenate([[0.0], ratios * old]))
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)
            
            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def lp_step(a, deadline, time_per_lp=10):
    """
    Single LP step: maximize sum(g) s.t. (a*g)[k] <= M, g >= 0.
    """
    n = len(a)
    if time.monotonic() > deadline:
        return a, compute_J(a)
    
    c = fast_conv(a)
    M = c.max()
    
    # For large n, the LP has 2n-1 constraints and n variables
    # This can be expensive. Use only a subset of constraints.
    if n > 500:
        # Use only top 100 constraints + random 100
        top_k = 100
        top_idx = np.argsort(c)[::-1][:top_k]
        rand_idx = np.random.default_rng(42).choice(2*n-1, size=min(100, 2*n-1), replace=False)
        sel_idx = np.unique(np.concatenate([top_idx, rand_idx]))
    else:
        sel_idx = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    b_ub_list = []
    
    for k in sel_idx:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(len(b_ub_list))
            cols.append(j)
            vals.append(a[k - j])
        b_ub_list.append(M)
    
    A = coo_matrix((vals, (rows, cols)), shape=(len(b_ub_list), n)).tocsr()
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub_list), bounds=bounds,
                    method='highs', options={'time_limit': time_per_lp, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
                return g
    except Exception:
        pass
    
    return a, compute_J(a)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and len(parent) > 0:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps

"""
Multi-phase optimizer using FFT-based gradient descent with momentum,
adaptive active-set tracking, coordinate descent polishing, and diverse
initializations. Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. Verified O(n log n) gradient via single FFT convolution
2. Heavy-ball momentum with backtracking line search
3. Adaptive active-set size (start focused, widen over time)
4. Window-function initializations (Blackman, Nuttall, etc.)
5. Sparsification attempts for final polish
"""

import numpy as np
import time
from scipy.sparse import coo_matrix, csr_matrix
from scipy.optimize import linprog


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        return np.ones(n) / n
    lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def compute_gradient_active(a, K):
    """
    Compute gradient of sum of top-K convolution values.
    grad[i] = 2 * sum_{k in topK} a[k-i] (with zero extension).
    Uses FFT: grad = 2 * (w * a_flipped)[n-1 : 2n-1]
    """
    n = len(a)
    c = fast_conv(a)
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / K
    a_flipped = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a_flipped, size)
    conv_result = np.fft.irfft(W * A, size)
    grad = 2.0 * conv_result[n - 1: 2 * n - 1]
    return grad, c


def compute_gradient_smooth(a, tau):
    """
    Compute gradient of (1/tau)*log(sum(exp(tau*c))) where c=conv(a,a).
    This is a smooth approximation to max(c).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    w = exp_c / exp_c.sum()
    a_flipped = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a_flipped, size)
    conv_result = np.fft.irfft(W * A, size)
    grad = 2.0 * conv_result[n - 1: 2 * n - 1]
    return grad, c


def gradient_descent_momentum(a, max_iters, deadline, K_init=5, K_max=None,
                              tau_init=10.0, tau_max=80.0,
                              lr_init=0.05, momentum=0.9,

    Projected gradient descent with heavy-ball momentum and backtracking.
    Uses adaptive active set / smooth approximation.

    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    if K_max is None:
        K_max = min(2 * n - 1, max(K_init * 5, n))
    velocity = np.zeros(n)
    no_improve = 0
    max_no_improve = 300
    prev_grad = None
    use_smooth = False

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters * 0.6))
        tau = tau_init + (tau_max - tau_init) * frac
        K = max(1, int(K_init + (K_max - K_init) * frac))
        # Switch to smooth when K is large relative to n
        use_smooth = K > n // 3

        if use_smooth:
            grad, c = compute_gradient_smooth(a, tau)
        else:
            grad, c = compute_gradient_active(a, K)

        # Project to simplex tangent (zero mean)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 1
        else:
            # Momentum
            velocity = momentum * velocity + (1 - momentum) * grad
            direction = -velocity
            # Backtracking line search
            lr = lr_init / (1.0 + 0.02 * it ** 0.4)
            found = False
            for _ in range(12):
                a_try = a + lr * direction
                a_try = project_simplex(a_try)
                obj_try = compute_J(a_try)
                if obj_try < best_obj + 1e-12:
                    a = a_try
                    best_a = a_try.copy()
                    best_obj = obj_try
                    no_improve = 0
                    found = True
                    break
                lr *= 0.5
            if not found:
                no_improve += 1
                # Very small step
                a = project_simplex(a + 0.001 * direction)
                obj_now = compute_J(a)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
                    no_improve = 0

        if no_improve > max_no_improve:
            break
        # Periodic perturbation to escape stagnation
        if no_improve > 100 and no_improve % 50 == 0:
            a = best_a + np.random.default_rng(it).normal(0, 0.01, n)
            a = project_simplex(a)
            velocity = np.zeros(n)
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    GD it {it}: K={K} tau={tau:.1f} obj={best_obj:.8f}", flush=True)

    return best_a, best_obj


def coord_descent_fast(a, max_sweeps, deadline, random_order=False, seed=0,
                       n_candidates=8):
    """Fast coordinate descent with incremental convolution."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 150
    ratios = np.array([0.0, 0.25, 0.5, 0.7, 1.0, 1.2, 1.4, 1.7, 2.0])

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        improved = False
        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = np.array([0.0, 1e-4, 5e-4, 2e-3, 1e-2, 5e-2])
            else:
                candidates = np.unique(np.concatenate([[0.0], ratios[:n_candidates] * old]))
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)

            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)
            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True
        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def lp_mirror_step(a, deadline, time_per_lp=8):
    """LP best-response: max sum(g) s.t. (a*g)[k] <= M, g>=0."""
    n = len(a)
    if time.monotonic() > deadline:
        return None
    c = fast_conv(a)
    M = c.max()
    # Subset constraints for large n
    if n > 400:
        n_cons = min(2 * n - 1, 300)
        top_k = min(150, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        n_rand = n_cons - top_k
        rand_idx = np.random.default_rng(7).choice(2 * n - 1, size=min(n_rand, 2 * n - 1), replace=False)
        sel = np.unique(np.concatenate([top_idx, rand_idx]))[:n_cons]
    else:
        sel = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    b_ub_list = []
    for k in sel:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(len(b_ub_list))
            cols.append(j)
            vals.append(a[k - j])
        b_ub_list.append(M)
    if len(rows) == 0:
        return None
    A = coo_matrix((vals, (rows, cols)), shape=(len(b_ub_list), n)).tocsr()
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub_list), bounds=bounds,
                    method='highs', options={'time_limit': time_per_lp, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-10:
                g /= g.sum()
                return g
    except Exception:
        pass
    return None


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences including window functions."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None and len(parent) > 0:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.12, n)))
    # Uniform
    inits.append(np.ones(n) / n)
    # Standard windows
    # Blackman
    win = np.blackman(n).astype(np.float64)
    inits.append(project_simplex(win + 0.01))
    # Hann
    win = np.hanning(n).astype(np.float64)
    inits.append(project_simplex(win + 0.01))
    # Hamming
    win = np.hamming(n).astype(np.float64)
    inits.append(project_simplex(win + 0.01))
    # Nuttall (approximate)
    a0, a1, a2, a3 = 0.36358, 0.48917, 0.13659, 0.01064
    win = a0 - a1 * np.cos(2 * np.pi * t) + a2 * np.cos(4 * np.pi * t) - a3 * np.cos(6 * np.pi * t)
    inits.append(project_simplex(win + 0.01))
    # Cosine patterns
    for freq in [1, 2, 3, 4, 5, 7]:
        for amp in [0.15, 0.3, 0.5, 0.7]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
    # U-shape (edge concentration)
    for alpha in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(project_simplex(np.abs(t - 0.5) ** alpha + 0.01))
    # Dual peaks
    for c1, c2 in [(0.2, 0.8), (0.1, 0.9), (0.3, 0.7)]:
        win = np.exp(-10 * (t - c1) ** 2) + np.exp(-10 * (t - c2) ** 2)
        inits.append(project_simplex(win + 0.05))
    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))
    # Triangular (peak in middle)
    inits.append(project_simplex(1.0 - 2.0 * np.abs(t - 0.5) + 0.01))
    return inits


def solve_combined(a, deadline, verbose=False):
    """Combined pipeline: GD+momentum, CD, LP, final CD."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_obj
    # Stage 1: GD with momentum (40%)
    t1 = time.monotonic() + 0.40 * remaining
    a_gd, obj_gd = gradient_descent_momentum(
        a, max_iters=8000, deadline=t1,
        K_init=3, K_max=min(2 * n - 1, n),
        tau_init=5.0, tau_max=50.0, lr_init=0.04, momentum=0.85
    )
    if obj_gd < best_obj:
        best_a = a_gd.copy()
        best_obj = obj_gd
    a = best_a.copy()
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj
    # Stage 2: CD polish (35%)
    t2 = time.monotonic() + 0.35 * remaining
    a_cd = coord_descent_fast(a, max_sweeps=20000, deadline=t2, random_order=True, seed=42)
    obj_cd = compute_J(a_cd)
    if obj_cd < best_obj:
        best_a = a_cd
        best_obj = obj_cd
    a = best_a.copy()
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj
    # Stage 3: LP step + blend (15%)
    if remaining > 20:
        t3 = time.monotonic() + 0.15 * remaining
        g = lp_mirror_step(a, t3, time_per_lp=min(8, max(2, int(t3 - time.monotonic()) - 1)))
        if g is not None:
            found = False
            for frac in [0.3, 0.2, 0.15, 0.1, 0.05]:
                a_try = (1 - frac) * a + frac * g
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_J(a_try)
                if obj_try < best_obj:
                    best_a = a_try
                    best_obj = obj_try
                    a = a_try
                    found = True
                    break
        # Continue with CD if time
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_cd2 = coord_descent_fast(a, max_sweeps=10000,
                                       deadline=deadline - 3, random_order=True, seed=99)
            obj_cd2 = compute_J(a_cd2)
            if obj_cd2 < best_obj:
                best_a = a_cd2
                best_obj = obj_cd2
    return best_a, best_obj


def sparsify(a, n_zero_targets, deadline):
    """Try setting smallest elements to zero and re-optimizing."""
    n = len(a)
    a = np.maximum(a.copy(), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    for nz in n_zero_targets:
        if time.monotonic() > deadline:
            break
        if nz >= n:
            continue
        a_try = a.copy()
        smallest_idx = np.argsort(a_try)[:nz]
        a_try[smallest_idx] = 0.0
        if a_try.sum() < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        # Quick CD to recover
        a_try = coord_descent_fast(a_try, max_sweeps=2000,
                                   deadline=min(deadline, time.monotonic() + 15),
                                   random_order=True, seed=55)
        obj_try = compute_J(a_try)
        if obj_try < best_obj:
            best_a = a_try
            best_obj = obj_try
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses FFT gradient + momentum, coordinate descent, LP, multi-length,
    diverse inits, and sparsification.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Optimize at parent length ===
    phase1_deadline = t_start + 400.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Combined optimization at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    init_objs = [(compute_J(init), init) for init in inits]
    init_objs.sort(key=lambda x: x[0])

    n_init = min(len(init_objs), 10)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_init) - 3

    for idx in range(n_init):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 25:
            break
        alloc = min(per_init_time, remaining_check - 3)
        if alloc < 15:
            break
        a0 = init_objs[idx][1]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()
        a_opt, obj_opt = solve_combined(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  Init {idx} (J0={init_objs[idx][0]:.6f}): J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration ===
    phase2_deadline = t_start + 900.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = [256, 320, 384, 448, 512, 576, 640, 704, 768, 832, 896, 960, 1024, 1152, 1280]
    candidate_ns = [n for n in candidate_ns if n != n_parent and n <= 1280]
    candidate_ns.sort(key=lambda n: abs(n - n_parent))

    for new_n in candidate_ns[:7]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 45:
            break
        alloc = min(75.0, remaining_check - 15)
        if alloc < 25:
            break
        print(f"  Try n={new_n} ({alloc:.0f}s)", flush=True)
        base = best_seq if best_seq is not None else parent
        a_ext = resample_seq(base, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()
        a_opt, obj_opt = solve_combined(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive optimization on best ===
    phase3_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive optimization ({phase3_deadline - time.monotonic():.0f}s left)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        if remaining > 40:
            # Multiple diverse passes
            n_best = len(best_seq)

            # Pass A: Long CD (sequential order for thoroughness)
            tA = time.monotonic() + 0.18 * remaining
            print(f"  CD seq ({tA - time.monotonic():.0f}s)", flush=True)
            aA = coord_descent_fast(best_seq, max_sweeps=50000, deadline=tA,
                                     random_order=False, seed=1)
            update_best(aA)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass B: GD with very focused active set
                tB = time.monotonic() + 0.22 * remaining
                print(f"  GD focused ({tB - time.monotonic():.0f}s)", flush=True)
                aB, objB = gradient_descent_momentum(
                    best_seq, max_iters=15000, deadline=tB,
                    K_init=1, K_max=15, tau_init=1.0, tau_max=8.0,
                    lr_init=0.008, momentum=0.8
                )
                update_best(aB)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass C: GD with wide active set + smooth
                tC = time.monotonic() + 0.22 * remaining
                print(f"  GD wide+smooth ({tC - time.monotonic():.0f}s)", flush=True)
                aC, objC = gradient_descent_momentum(
                    best_seq, max_iters=15000, deadline=tC,
                    K_init=20, K_max=2 * n_best - 1, tau_init=20.0, tau_max=100.0,
                    lr_init=0.03, momentum=0.9
                )
                update_best(aC)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass D: CD random on best
                tD = time.monotonic() + 0.18 * remaining
                print(f"  CD rand ({tD - time.monotonic():.0f}s)", flush=True)
                aD = coord_descent_fast(best_seq, max_sweeps=50000, deadline=tD,
                                         random_order=True, seed=777)
                update_best(aD)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass E: LP + blend + CD
                tE = time.monotonic() + 0.12 * remaining
                print(f"  LP ({tE - time.monotonic():.0f}s)", flush=True)
                g = lp_mirror_step(best_seq, tE, time_per_lp=min(10, max(3, int(tE - time.monotonic()) - 2)))
                if g is not None:
                    for frac in [0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02]:
                        a_try = (1 - frac) * best_seq + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        obj_try = compute_J(a_try)
                        if obj_try < best_score:
                            update_best(a_try)
                            break
                # Final CD on whatever is best now
                remaining = phase3_deadline - time.monotonic()
                if remaining > 15:
                    tF = deadline - 5
                    print(f"  Final CD ({tF - time.monotonic():.0f}s)", flush=True)
                    aF = coord_descent_fast(best_seq, max_sweeps=50000, deadline=tF,
                                             random_order=True, seed=4242)
                    update_best(aF)

            # Sparsification attempt
            remaining = phase3_deadline - time.monotonic()
            if remaining > 20 and best_seq is not None:
                n_best = len(best_seq)
                zero_targets = [int(n_best * 0.05), int(n_best * 0.1), int(n_best * 0.15)]
                zero_targets = [z for z in zero_targets if 0 < z < n_best]
                print(f"  Sparsify ({remaining:.0f}s)", flush=True)
                aS, objS = sparsify(best_seq, zero_targets, deadline - 5)
                update_best(aS)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
