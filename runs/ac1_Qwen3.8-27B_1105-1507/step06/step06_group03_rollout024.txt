    """Project onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1, dtype=np.float64)
    cond = u - cssv/ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)

        return np.full(new_n, src[0] if src.size == 1 else 0.5)
    t_old = np.linspace(0.0, 1.0, src.size)
    t_new = np.linspace(0.0, 1.0, new_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def cd_fast(a, deadline, seed=0, verbose=False, max_no_improve=1200):
    """
    Fast coordinate descent with incremental convolution updates.
    Tracks sum separately for correct objective computation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = 1.0
    else:
        a = a / s
        s = 1.0
    conv = fft_self_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    sweep = 0

    # Candidate ratios (multiplicative perturbation around current value)
    ratios = np.array([0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.2, 3.0])

    while time.monotonic() < deadline:
        if sweep % 2 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            old = a[i]
            lo = i
            hi = i + n - 1
            kq = 2 * i

            if old < 1e-10:
                cands = np.array([0.0, 0.5 / n, 1.0 / n, 1.5 / n, 2.0 / n])
            else:
                cands = old * ratios

            best_val = old
            cur_max = conv.max()
            best_local = 2.0 * n * cur_max / (s * s)

            for x in cands:
                x = float(x)
                delta = x - old
                if abs(delta) < 1e-16:
                    continue
                s_new = s + delta
                if s_new < 1e-12:
                    continue

                # Apply incremental update
                conv[lo:hi + 1] += 2.0 * delta * a
                conv[kq] += delta * delta

                local_obj = 2.0 * n * conv.max() / (s_new * s_new)

                if local_obj < best_local:
                    best_local = local_obj
                    best_val = x

                # Revert
                conv[lo:hi + 1] -= 2.0 * delta * a
                conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-15:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a
                conv[kq] += delta * delta
                a[i] = best_val
                s = s + delta

        sweep += 1

        # Periodic renormalization to prevent drift
        if sweep % 200 == 0:
            s_curr = a.sum()
            if s_curr > 1e-12 and abs(s_curr - 1.0) > 1e-6:
                a = a / s_curr
                conv = fft_self_conv(a)
                s = 1.0

        cur_obj = 2.0 * n * conv.max() / (s * s)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 200 == 0:
            print(f"        CD sweep {sweep}: obj={best_obj:.10f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0.0)
    s_final = a.sum()
    if s_final > 1e-12:
        a = a / s_final
    return a


def cd_fine(a, deadline, seed=0, verbose=False):
    """
    Fine-tuning coordinate descent with very small perturbations.
    Best used when the sequence is already close to optimal.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = 1.0
    else:
        a = a / s
        s = 1.0
    conv = fft_self_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    sweep = 0
    max_no_improve = 2000

    # Fine ratios - small perturbations around 1.0
    ratios = np.array([0.0, 1.0 - 0.01, 1.0 - 0.005, 1.0 - 0.002, 1.0 - 0.001,
                       1.0, 1.0 + 0.001, 1.0 + 0.002, 1.0 + 0.005, 1.0 + 0.01,
                       1.0 + 0.02, 1.0 + 0.05])

    while time.monotonic() < deadline:
        order = rng.permutation(n)

        for i in order:
            old = a[i]
            lo = i
            hi = i + n - 1
            kq = 2 * i

            if old < 1e-12:
                continue

            cands = old * ratios

            best_val = old
            cur_max = conv.max()
            best_local = 2.0 * n * cur_max / (s * s)

            for x in cands:
                x = float(x)
                delta = x - old
                if abs(delta) < 1e-16:
                    continue
                s_new = s + delta
                if s_new < 1e-12:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a
                conv[kq] += delta * delta

                local_obj = 2.0 * n * conv.max() / (s_new * s_new)

                if local_obj < best_local:
                    best_local = local_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a
                conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-15:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a
                conv[kq] += delta * delta
                a[i] = best_val
                s = s + delta

        sweep += 1

        if sweep % 500 == 0:
            s_curr = a.sum()
            if s_curr > 1e-12 and abs(s_curr - 1.0) > 1e-6:
                a = a / s_curr
                conv = fft_self_conv(a)
                s = 1.0

        cur_obj = 2.0 * n * conv.max() / (s * s)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 500 == 0:
            print(f"        Fine CD sweep {sweep}: obj={best_obj:.12f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0.0)
    s_final = a.sum()
    if s_final > 1e-12:
        a = a / s_final
    return a


def sca_lp_step(a, deadline, max_iters=50, verbose=False):
    """
    SCA-LP: At each step, solve LP to find direction that keeps
    convolution below current max, take a step, renormalize.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = obj_value(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 30:
            break

        c = fft_self_conv(a)
        M = c.max()

        # LP: minimize t s.t. 2*(a*g)[k] - c[k] <= t, sum(g)=1, g>=0
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            t_lp = min(20.0, max(5.0, (deadline - time.monotonic()) * 0.3))
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': t_lp})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    found = False
                    for frac in [0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        J_try = obj_value(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose:
            print(f"        SCA-LP it {it}: obj={best_obj:.10f}", flush=True)

        if no_improve > 10:
            break

    return best_a, best_obj


def smooth_gradient_step(a, deadline, max_iters=200, tau=50.0, lr=0.02, verbose=False):
    """
    Gradient descent with log-sum-exp smoothing of the max.
    Works on the simplex (sum=1 normalization).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = obj_value(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fft_self_conv(a)
        c_shift = c - c.max()
        exp_c = np.exp(tau * c_shift)
        w = exp_c / exp_c.sum()

        # Gradient of smooth max: grad_i = sum_k w[k] * d(conv[k])/da[i]
        # d(conv[k])/da[i] = a[k-i] (if 0<=k-i<n) + a[i-(k-n+1-i)] ... 
        # Actually: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/da[i] = a[k-i] (from j=i) + a[i-k+n] ... no
        # d(conv[k])/da[i] = sum over j where j=i or k-j=i
        # = a[k-i] (if 0<=k-i<n) + a[k-i] (if k-i != i and 0<=k-i<n)
        # = 2*a[k-i] if 0<=k-i<n and k!=2i, or a[k-i] + (a[i]+a[k-i]) if k=2i
        # Simpler: use FFT
        # grad = ifft(w_fft * a_rev_fft) appropriately extracted

        a_rev = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        W_fft = np.fft.rfft(w, size)
        Arev_fft = np.fft.rfft(a_rev, size)
        conv_w = np.fft.irfft(W_fft * Arev_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            # d(conv[k])/da[i] = a[k-i] for k-i in [0,n-1] plus a[k-i] for the other term
            # Using the convolution: (w * a_rev)[n-1+i] gives sum_k w[k]*a_rev[k-(n-1-i)]
            # a_rev[j] = a[n-1-j], so a_rev[k-n+1-i] = a[n-1-k+n-1-i] = a[2n-2-k-i]
            # Hmm this is getting complicated. Let me use the direct formula.
            idx = n - 1 + i
            if 0 <= idx < len(conv_w):
                grad[i] = conv_w[idx]

        # Center gradient for simplex projection
        grad = grad - np.mean(grad)

        a_new = a - lr * grad
        a_new = proj_simplex(a_new)

        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

        a = a_new
        # Anneal learning rate
        lr = max(0.001, 0.02 * (0.995 ** it))

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Strategy: intensive coordinate descent (fast + fine), occasional LP moves,
    multi-length exploration, and smooth gradient polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_score = float(evaluate_sequence([float(x) for x in parent]))
            if np.isfinite(update_score):
                best_seq = parent.copy()
                best_score = update_score
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError, ValueError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        try_update(parent)

    n_parent = len(parent)
    print(f"\n{'=' * 60}", flush=True)
    print(f"Starting: n={n_parent}, J={best_score:.8f}", flush=True)
    print(f"Total budget: {budget_s:.0f}s", flush=True)
    print(f"{'=' * 60}\n", flush=True)

    # === PHASE 1: Intensive coordinate descent (55% of time) ===
    phase1_end = t_start + 0.55 * (budget_s - 30)
    print(f"PHASE 1: Coordinate Descent (deadline t={phase1_end - t_start:.0f}s)", flush=True)

    a_curr = parent.copy()
    a_curr = np.maximum(a_curr, 0)
    a_curr = a_curr / a_curr.sum()

    # Sub-phase 1a: General CD
    t_1a = min(phase1_end - 60, t_start + 0.35 * (budget_s - 30))
    a_curr = cd_fast(a_curr, deadline=t_1a, seed=42, verbose=True)
    try_update(a_curr)

    remaining = phase1_end - time.monotonic()
    if remaining > 30:
        # Sub-phase 1b: Fine CD
        t_1b = time.monotonic() + remaining
        a_curr = cd_fine(a_curr, deadline=t_1b, seed=123, verbose=True)
        try_update(a_curr)

    # === PHASE 2: LP-based global moves (20% of time) ===
    phase2_end = t_start + 0.75 * (budget_s - 30)
    print(f"\nPHASE 2: SCA-LP (deadline t={phase2_end - t_start:.0f}s)", flush=True)

    a_base = best_seq.copy() if best_seq is not None else a_curr
    a_base = np.maximum(a_base, 1e-10)
    a_base = a_base / a_base.sum()

    a_lp, obj_lp = sca_lp_step(a_base, deadline=phase2_end - 60, max_iters=30, verbose=True)
    try_update(a_lp)

    # Follow up with CD on LP result
    remaining = phase2_end - time.monotonic()
    if remaining > 20:
        a_curr = cd_fast(a_lp, deadline=time.monotonic() + remaining - 5, seed=777)
        try_update(a_curr)

    # === PHASE 3: Multi-length exploration (15% of time) ===
    phase3_end = t_start + 0.90 * (budget_s - 30)
    print(f"\nPHASE 3: Multi-Length (deadline t={phase3_end - t_start:.0f}s)", flush=True)

    candidate_lengths = [n_parent - 64, n_parent, n_parent + 64, n_parent + 128]
    candidate_lengths = sorted(set([n for n in candidate_lengths if n >= 64 and n <= 4096]))

    per_len_time = (phase3_end - time.monotonic()) / max(1, len(candidate_lengths)) - 10

    for new_n in candidate_lengths:
        remaining = phase3_end - time.monotonic()
        if remaining < 40:
            break
        alloc = min(per_len_time, remaining - 10)
        if alloc < 25:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt = cd_fast(a_ext, deadline=time.monotonic() + alloc * 0.7, seed=new_n)
        a_opt = cd_fine(a_opt, deadline=time.monotonic() + alloc * 0.25, seed=new_n + 1)
        try_update(a_opt)
        print(f"  n={new_n}: done", flush=True)

    # === PHASE 4: Final intensive optimization (10% of time) ===
    phase4_end = deadline
    print(f"\nPHASE 4: Final Polish (deadline t={phase4_end - t_start:.0f}s)", flush=True)

    if best_seq is not None:
        a_curr = best_seq.copy()
        a_curr = np.maximum(a_curr, 0)
        a_curr = a_curr / a_curr.sum()

        remaining = phase4_end - time.monotonic()
        if remaining > 60:
            # Long fine CD
            t_final = time.monotonic() + remaining * 0.7
            print(f"  Final fine CD (budget {t_final - time.monotonic():.0f}s)...", flush=True)
            a_curr = cd_fine(a_curr, deadline=t_final, seed=999, verbose=True)
            try_update(a_curr)

            remaining = phase4_end - time.monotonic()
            if remaining > 20:
                # One more general CD
                a_curr = cd_fast(best_seq, deadline=time.monotonic() + remaining * 0.8,
                                 seed=2024, max_no_improve=500)
                try_update(a_curr)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        try_update(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.10f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
