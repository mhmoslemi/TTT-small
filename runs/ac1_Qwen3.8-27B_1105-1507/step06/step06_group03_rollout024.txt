    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coord_descent_bisect(a, deadline, seed=42, max_sweeps=50000):
    """
    Coordinate descent with bisection line search for exact coordinate optimum.
    For each coordinate i, we find the optimal value by noting that
    the objective as a function of a_i is:
    J(x) = 2n * max_k (c_k + 2*x*a_k + x^2 * delta_k) / (s + x - a_i)^2
    where c_k is the current conv, a_k is the relevant slice, delta_k is 1 at k=2i.
    
    Since the max makes this piecewise, we use a simpler approach:
    try a set of candidate values and pick the best.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    a = a / s
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in range(n):
            i = order[i]
            old = a[i]

            lo_idx = max(0, i)
            hi_idx = min(2 * n - 2, i + n - 1)
            a_slice = a[lo_idx - i:hi_idx - i + 1].copy()
            kq = 2 * i

            # Try candidate values
            if old < 1e-6:
                candidates = [0.0, 1e-5, 1e-4, 5e-4, 1e-3, 5e-3]
            else:
                candidates = np.array([0.0, 0.1*old, 0.3*old, 0.5*old, 0.7*old,
                                       0.9*old, 1.1*old, 1.3*old, 1.7*old, 2.0*old])
                # Add some absolute values
                candidates = np.concatenate([candidates, [1e-4, 5e-4, 1e-3]])
                candidates = np.unique(candidates)
                candidates = candidates[(candidates >= 0)]

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = float(x)
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                conv_save_lo = conv[lo_idx:hi_idx + 1].copy()
                conv[lo_idx:hi_idx + 1] += 2.0 * delta * a_slice
                if 0 <= kq < len(conv):
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    conv[lo_idx:hi_idx + 1] = conv_save_lo
                    continue
                new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo_idx:hi_idx + 1] = conv_save_lo

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo_idx:hi_idx + 1] += 2.0 * delta * a_slice
                if 0 <= kq < len(conv):
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    return a, compute_J(a)

def sca_lp_optimize(a, deadline, max_iters=500):
    """
    SCA-LP optimization. At each step, linearize the auto-correlation
    constraints and solve an LP to find a better direction.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # LP: minimize t
        # s.t. 2*(a * g)[k] - c[k] <= t  for k=0,...,2n-2
        #      sum(g) = 1
        #      g >= 0
        # Variables: g[0..n-1], t (index n)

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        A_ub = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]

        A_eq = coo_matrix((np.ones(n), (np.zeros(n), np.arange(n))), shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    # Line search
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try = a_try / a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj

def smooth_adam_opt(a, deadline, tau_start=20, tau_end=200, lr=0.05, max_iters=10000):
    """Adam with log-sum-exp smoothing on the simplex."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_count = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters * 0.7))
        tau = tau_start * (tau_end / tau_start) ** frac

        c = fast_conv(a)
        c_max = c.max()
        c_shifted = c - c_max
        exp_c = np.exp(np.clip(tau * c_shifted, -500, 500))
        w = exp_c / exp_c.sum()

        # Gradient
        a_rev = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w[::-1], size)
        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        grad = grad - np.mean(grad)

        t_count += 1
        step_lr = lr / (1.0 + 0.05 * t_count ** 0.4)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - step_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Restart
                rng = np.random.default_rng(it + 12345)
                a = project_simplex(best_a + rng.normal(0, 0.01, n))
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

    return best_a, best_obj

def multi_restart_opt(a, deadline, n_restarts=None):
    """Multiple restarts with small perturbations from the best sequence."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_J(a)
    rng = np.random.default_rng(999)

    if n_restarts is None:
        n_restarts = 20

    time_per_restart = (deadline - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > deadline - 10:
            break

        # Generate perturbation
        sigma = 0.02 / (1 + 0.3 * r)
        a_pert = project_simplex(best_a + rng.normal(0, sigma, n))

        # Quick coordinate descent
        rd = time.monotonic() + min(time_per_restart * 0.6, 30)
        a_opt, obj_opt = coord_descent_bisect(a_pert, deadline=rd, seed=1000 + r)

        if obj_opt < best_obj:
            best_a = a_opt.copy()
            best_obj = obj_opt

        # Quick LP step
        if time.monotonic() < deadline - 10:
            rl = time.monotonic() + min(time_per_restart * 0.4, 20)
            a_lp, obj_lp = sca_lp_optimize(a_opt, deadline=rl, max_iters=20)
            if obj_lp < best_obj:
                best_a = a_lp.copy()
                best_obj = obj_lp

    return best_a, best_obj

"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k_idx = np.where(cond)[0]
    if len(k_idx) == 0:
        return np.ones(n) / n
    lam = cssv[k_idx[-1]] / (k_idx[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent(a, deadline, seed=42, max_sweeps=80000):
    """Coordinate descent with adaptive candidate set and incremental conv updates."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    a = a / s
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for idx in range(n):
            i = order[idx]
            old = a[i]

            lo_idx = max(0, i)
            hi_idx = min(2 * n - 2, i + n - 1)
            a_slice = a[lo_idx - i:hi_idx - i + 1].copy()
            kq = 2 * i

            if old < 1e-6:
                candidates = [0.0, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3]
            else:
                # Adaptive: more candidates near current value
                ratios = [0.0, 0.1, 0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 2.0, 3.0]
                candidates = [r * old for r in ratios]
                # Add absolute scale candidates
                median_val = np.median(a[a > 0]) if np.any(a > 0) else 1e-3
                candidates.extend([median_val * r for r in [0.5, 1.0, 2.0, 4.0]])
                candidates = sorted(set(max(0.0, c) for c in candidates))

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = float(x)
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                # Save and modify conv
                conv[lo_idx:hi_idx + 1] += 2.0 * delta * a_slice
                if 0 <= kq < len(conv):
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    conv[lo_idx:hi_idx + 1] -= 2.0 * delta * a_slice
                    if 0 <= kq < len(conv):
                        conv[kq] -= delta * delta
                    continue

                new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo_idx:hi_idx + 1] -= 2.0 * delta * a_slice
                if 0 <= kq < len(conv):
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo_idx:hi_idx + 1] += 2.0 * delta * a_slice
                if 0 <= kq < len(conv):
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    return a, compute_J(a)


def sca_lp_optimize(a, deadline, max_iters=300, penalty_lambda=0.0):
    """
    SCA-LP with optional quadratic penalty.
    Minimizes t + lambda * linearized sum(a_i^2)
    subject to linearized auto-correlation constraints.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    it = 0

    while it < max_iters:
        if time.monotonic() > deadline:
            break
        it += 1

        c = fast_conv(a)
        M = c.max()

        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        # Objective: minimize t + 2*lambda * sum(a_curr_i * g_i)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        if penalty_lambda > 0:
            c_obj[:n] += 2.0 * penalty_lambda * a

        A_ub = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        bounds = [(0, None)] * n + [(None, None)]

        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    found = False
                    for frac in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
                        a_try = (1.0 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try = a_try / a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 40:
            break

    return best_a, best_obj


def smooth_adam_opt(a, deadline, tau_start=20.0, tau_end=300.0, lr=0.05, max_iters=15000):
    """Adam optimizer with log-sum-exp smoothing on the probability simplex."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_count = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, int(max_iters * 0.7)))
        tau = tau_start * (tau_end / tau_start) ** frac

        c = fast_conv(a)
        c_max = c.max()
        c_shifted = c - c_max
        exp_c = np.exp(np.clip(tau * c_shifted, -500, 500))
        w = exp_c / exp_c.sum()

        # Gradient via FFT
        a_rev = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w[::-1], size)
        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        grad = grad - np.mean(grad)

        t_count += 1
        step_lr = lr / (1.0 + 0.05 * t_count ** 0.4)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)

        a_new = a - step_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                a = project_simplex(best_a + rng.normal(0, 0.01, n))
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

    return best_a, best_obj
