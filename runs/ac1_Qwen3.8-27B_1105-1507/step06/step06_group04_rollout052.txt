```python
"""
Coordinate descent with golden section search for peak convolution minimization.
Explores multiple sequence lengths and uses periodic perturbations to escape local optima.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def rfft(x, n=None):
    from scipy.fft import rfft as _rfft
    return _rfft(x, n)


def irfft(x, n=None):
    from scipy.fft import irfft as _irfft
    return _irfft(x, n)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def coord_decent_sweep(a, rng, verbose=False):
    """One sweep of coordinate descent. Returns updated a."""
    n = len(a)
    conv = compute_conv(a)
    S_total = np.sum(a)

        old = a[i]
        if old < 1e-15:
            # Try to increase from zero
            x_try = 1.0 / (2 * n)  # small positive value
            S = S_total - old  # = S_total since old ≈ 0
            # Evaluate
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)
            j_idx = np.arange(k_min, k_max + 1) - i
            new_vals = conv[k_min:k_max+1].copy()
            new_vals += 2.0 * x_try * a[j_idx]
            if k_min <= 2*i <= k_max:
                new_vals[2*i - k_min] -= 2.0 * x_try * a[i]
                new_vals[2*i - k_min] += x_try * x_try
```

```python
Coordinate descent with golden section search for peak convolution minimization.
Uses incremental FFT-based convolution updates, periodic perturbations, and
multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def eval_coord(x, a, conv, S, i, n, old):
    """
    Evaluate objective when a[i] = x, given conv computed with a[i] = old.
    S = sum of all a[j] for j != i.
    """
    k_min = i
    k_max = min(i + n - 1, 2 * n - 2)
    j_idx = np.arange(k_max - k_min + 1)  # = 0, 1, ..., k_max-k_min
    
    seg = conv[k_min:k_max + 1].copy()
    seg += 2.0 * (x - old) * a[j_idx]

        pos = 2 * i - k_min
        seg[pos] += (x - old) ** 2

    return 2.0 * n * max_val / (sum_val * sum_val)
```

```python
"""
Coordinate descent with golden section search for peak convolution minimization.
Key innovation: exact 1D optimization per coordinate using the structure that
changing a[i] only affects conv[k] for k in [i, i+n-1] via a simple linear+quadratic
update. This makes each coordinate evaluation O(n) and a full sweep O(n^2), enabling
thousands of sweeps within the time budget. Combined with adaptive perturbations
and multi-length exploration to escape local optima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def eval_coord(x, a, conv, S, i, n, old):
    """
    Evaluate objective when a[i] = x (others unchanged).
    Uses the fact that only conv[i:i+n] is affected.
    S = sum of a[j] for j != i.
    """
    seg = conv[i:i + n].copy()
    seg += 2.0 * (x - old) * a
    seg[i] += (x - old) ** 2
    max_val = np.max(seg)
    sum_val = S + x
    if sum_val <= 0:
        return np.inf
    return 2.0 * n * max_val / (sum_val * sum_val)


def coord_sweep(a, conv, S_total, rng, n_sweeps=1, time_limit=None, start_time=None):
    """
    Perform n_sweeps of cyclic coordinate descent.
    Returns (a, conv, S_total, current_obj).

            break

            old = a[i]
            S = S_total - old

            if old < 1e-18:

                best_x = 0.0

                for x_try in [1.0 / (3 * n), 1.0 / (2 * n), 1.0 / n, 2.0 / n]:
                    f_try = eval_coord(x_try, a, conv, S, i, n, 0.0)
                    if f_try < best_f:
                        best_f = f_try
                        best_x = x_try
                if best_x > 0:
                    a[i] = best_x
                    conv[i:i + n] += 2.0 * best_x * a
                    conv[i + i] += best_x * best_x  # (x-0)^2 = x^2
                    S_total += best_x
                continue

            # Adaptive search range
            width = max(3.0 * mean_val, old * 0.5, 2.0 / n)

            x_hi = old + width

                    best_grid_x = gx

            idx = int(np.argmin([eval_coord(gx, a, conv, S, i, n, old) for gx in grid]))
            if idx > 0 and idx < len(grid) - 1:
                gs_lo = grid[max(0, idx - 1)]
                gs_hi = grid[min(len(grid) - 1, idx + 1)]
            else:
                gs_lo = x_lo
                gs_hi = x_hi

                    fc = fd

def perturb_sequence(a, rng, strength=0.1):
    """Perturb with multiplicative log-normal noise."""
    n = len(a)
    noise = rng.normal(0, strength, n)
    a_new = a * np.exp(noise)
    return normalize_simplex(a_new)


def perturb_block(a, rng, block_size=10, strength=0.3):
    """Perturb a random block of coordinates."""
    n = len(a)
    a_new = a.copy()
    start = rng.integers(0, max(1, n - block_size))
    block = slice(start, start + block_size)
    a_new[block] *= np.exp(rng.normal(0, strength, block_size))
    return normalize_simplex(a_new)


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize_simplex(a_new)
```
