"""

import numpy as np
from scipy.fft import rfft, irfft
import time

try:
    import cvxpy as cp

except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def cross_correlation(a, w):
    """
    Compute corr[m] = sum_j a[j] * w[j+m] for m=0..len(a)-1.
    Uses FFT for efficiency. w has length >= len(a).
    """
    n = len(a)
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w, size)
    return irfft(np.conj(fa) * fw, size)[:n]


def gradient_qnorm(a, q):
    """
    Compute (sub)gradient of (sum conv[k]^q)^(1/q) w.r.t. a.
    Returns centered gradient (preserves sum).
    """
    n = len(a)
    conv = compute_conv(a)
    conv = np.maximum(conv, 0)
    max_c = float(np.max(conv))
    if max_c < 1e-30:
        return np.zeros(n)

    if q == np.inf:
        k_star = int(np.argmax(conv))
        weights = np.zeros(2 * n - 1)
        weights[k_star] = 1.0
    else:
        weights = conv ** (q - 1)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-30:
        return np.zeros(n)
    weights = weights / w_sum

    corr = cross_correlation(a, weights)
    grad = 2.0 * corr
    grad = grad - np.mean(grad)
    return grad


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_weights) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    if not HAS_CVXPY:
        return None, None, None

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    ineq = A @ g <= M
    eq = cp.sum(g) == 1
    prob = cp.Problem(cp.Minimize(M), [ineq, eq])

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    g_val = None
    M_val = None
    y_val = None

    for solver in solvers:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                M_val = float(M.value)
                try:
                    y_raw = ineq.dual_value
                    if y_raw is not None:
                        y_val = np.maximum(y_raw, 0)
                        ys = float(np.sum(y_val))
                        if ys > 1e-15:
                            y_val = y_val / ys
                        else:
                            y_val = None
                except Exception:
                    y_val = None
                break
        except Exception:
            continue

    if g_val is None:
        # Fallback: scipy linprog
        try:
            from scipy.optimize import linprog
            c_vec = np.zeros(n + 1)
            c_vec[-1] = 1.0
            A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
            b_eq = np.array([1.0])
            bounds = [(0, None)] * (n + 1)
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'time_limit': time_limit * 1000})
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                M_val = float(res.x[n])
                y_val = None
        except Exception:
            pass

    if g_val is None:
        return None, None, None

    s = float(np.sum(g_val))
    if s < 1e-10:
        return None, None, None
    g_val = g_val / s
    return g_val, M_val, y_val


def smoothed_gradient_descent(a, time_budget=120.0, verbose=False):
    """
    Smoothed gradient descent with adaptive q-norm and learning rate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Schedule: start smooth, get sharper
    q_schedule = [2, 4, 8, 16, 32, 64, 128, 256, 512, np.inf]
    lr = 0.05

    for q in q_schedule:
        if time.monotonic() - start > time_budget:
            break
        phase_budget = min(15.0, (time_budget - (time.monotonic() - start)) * 0.3)
        phase_start = time.monotonic()
        lr_local = lr
        stall = 0

        for it in range(2000):
            if time.monotonic() - phase_start > phase_budget:
                break

            grad = gradient_qnorm(a, q)
            a_new = a - lr_local * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                lr_local *= 1.02
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    lr_local *= 0.7
                    if lr_local < 1e-8:
                        break
                a = best_a.copy()

        lr = lr_local
        if verbose:
            print(f"      q={q}: obj={best_obj:.8f}")

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def gaussian_init(n, rng):
    """Initialize with a discrete Gaussian shape."""
    center = n / 2.0
    sigma = n / (2.0 * rng.uniform(1.5, 3.5))
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a + 0.01
    a = a / np.sum(a)
    return a


def polynomial_init(n, rng):
    """Initialize with a polynomial shape."""
    x = np.linspace(0, 1, n)
    c0 = rng.uniform(0.5, 1.5)
    c1 = rng.uniform(-0.5, 0.5)
    c2 = rng.uniform(-1.0, 0.0)
    a = c0 + c1 * x + c2 * (x - 0.5) ** 2
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining smoothed gradient descent,
    LP alternating, dual-guided steps, symmetry, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: Smoothed gradient descent from parent
    print("\n=== Phase 1: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.2)
        a1 = smoothed_gradient_descent(best_seq, time_budget=alloc, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  After SGD: obj={best_score:.8f}")
        else:
            print(f"  After SGD: obj={obj1:.8f} (no improvement)")

    # Phase 2: LP alternating
    print("\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(200.0, remaining * 0.25)
        a2 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  After LP-alt: obj={best_score:.8f}")
        else:
            print(f"  After LP-alt: obj={obj2:.8f} (no improvement)")

    # Phase 3: Symmetric refinement
    print("\n=== Phase 3: Symmetric Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.15)
        a3 = symmetric_refine(best_seq, time_budget=alloc)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  After Sym: obj={best_score:.8f}")
        else:
            print(f"  After Sym: obj={obj3:.8f} (no improvement)")

    # Phase 4: Multi-n exploration
    print("\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        n_parent - 100, n_parent - 50, n_parent - 25, n_parent,
        n_parent + 25, n_parent + 50, n_parent + 100, n_parent + 200,
        500, 600, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(40.0, remaining * 0.2)
        # Quick optimization
        a_opt = smoothed_gradient_descent(a_try, time_budget=alloc * 0.5)
        obj_q = compute_obj(a_opt)
        if obj_q < best_score:
            # Follow up with LP
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_opt2 = lp_alternating(a_opt, time_budget=min(30.0, remaining2 * 0.4))
                obj_q = compute_obj(a_opt2)
        if obj_q < best_score:
            best_score = obj_q
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_q:.8f}")

    # Phase 5: Restart with different initializations
    print("\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        init_type = r % 5
        if init_type == 0:
            a_r = gaussian_init(n_best, rng)
        elif init_type == 1:
            a_r = polynomial_init(n_best, rng)
        elif init_type == 2:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 3:
            # Different n with gaussian
            n_r = max(400, min(1200, n_best + int(rng.integers(-200, 200))))
            a_r = gaussian_init(n_r, rng)
        else:
            # Cosine window
            x = np.linspace(0, np.pi, n_best)
            a_r = (np.cos(x) ** 2 + 0.3)
            a_r = a_r / np.sum(a_r)

        alloc = min(30.0, remaining * 0.12)
        a_opt = smoothed_gradient_descent(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            # Follow up with LP
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_opt2 = lp_alternating(a_opt, time_budget=min(25.0, remaining2 * 0.4))
                obj_r = compute_obj(a_opt2)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (type={init_type}, n={len(a_opt)}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_r:.8f}")

    # Phase 6: Final intensive polish
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_final = len(best_seq)
        a_final = best_seq.copy()

        # Multiple rounds of LP alternating
        for round_idx in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.3, 60.0)
            a_new = lp_alternating(a_final, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_final = a_new.copy()
                print(f"  Polish round {round_idx}: obj={best_score:.8f}")
            else:
                print(f"  Polish round {round_idx}: obj={obj_new:.8f} (no improvement)")

        # Symmetric polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sym = symmetric_refine(a_final, time_budget=min(30.0, remaining * 0.3))
            obj_sym = compute_obj(a_sym)
            if obj_sym < best_score:
                best_score = obj_sym
                best_seq = a_sym.copy()
                print(f"  Final Sym: obj={best_score:.8f}")

        # One more SGD pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sg = smoothed_gradient_descent(best_seq, time_budget=min(20.0, remaining * 0.4))
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Final SGD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
