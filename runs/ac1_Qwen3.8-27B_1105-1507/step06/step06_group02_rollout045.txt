import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute the objective J(a)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M):
    """Solve the LP: max sum(g) s.t. (a*g)(k) <= M, g >= 0."""
    n = len(a)
    # Build sparse convolution matrix
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0), r
    except Exception:
        pass
    return None, None

def anderson_mix(history, m, max_alpha=2.0):
    """Anderson mixing: find best extrapolation from history."""
    # history is list of (x, Fx) pairs
    if len(history) < m:
        m = len(history)
    if m < 2:
        return history[-1][1]
    
    # Last m points
    xs = np.array([history[-m+i][0] for i in range(m)])
    Fxs = np.array([history[-m+i][1] for i in range(m)])
    
    # Residuals
    Rs = xs - Fxs  # (m, n)
    
    # Build Gram matrix
    G = Rs @ Rs.T  # (m, m)
    G += 1e-14 * np.eye(m)
    
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except:
        alpha = np.ones(m) / m
    
    # Clip and normalize
    alpha = np.clip(alpha, -max_alpha, max_alpha)
    s = alpha.sum()
    if abs(s) < 1e-15:
        return np.maximum(Fxs[-1], 0)
    alpha /= s
    
    mixed = alpha @ Fxs
    return np.maximum(mixed, 0)

import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0), r
    except Exception:
        pass
    return None, None


def anderson_mix(history, max_alpha=3.0):
    m = len(history)
    if m < 2:
        return np.maximum(history[-1][1], 0)
    xs = np.array([h[0] for h in history[-m:]])
    Fxs = np.array([h[1] for h in history[-m:]])
    Rs = xs - Fxs
    G = Rs @ Rs.T
    G += 1e-14 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = float(ones @ G_inv_ones)
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m
    alpha = np.clip(alpha, -max_alpha, max_alpha)
    s = alpha.sum()
    if abs(s) < 1e-15:
        return np.maximum(Fxs[-1], 0)
    alpha /= s
    mixed = alpha @ Fxs
    return np.maximum(mixed, 0)


def nesterov_smooth(a, deadline, max_iters=5000, tau_start=0.5, tau_end=500.0,
                     lr_init=0.5, lr_end=0.002):

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    momentum = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        b_y = fast_conv(y)
        b_max = float(b_y.max())
        shifted = np.clip(tau * (b_y - b_max), -80, 0)
        exp_b = np.exp(shifted)
        p = exp_b / exp_b.sum()

        grad = 2.0 * np.correlate(p, y, mode='valid')

        a_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)

        sy = y.sum()
        if sy < 1e-10:

        else:

        a = a_new

    return best_a

def resample_seq(src, n_target):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_target:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_target)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gs_flat(a, n_iters=100, beta=0.85):
    """Gerchberg-Saxton spectral flattening."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_max = mag.max()
        if mag_max < 1e-15:
            break
        mag_norm = mag / mag_max
        mag_new = (1.0 - beta) * mag_norm + beta * (mag_norm.mean() + 1e-30)
        mag_new = mag_new / (mag_new.max() + 1e-30)
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_cand = np.maximum(a_new, 0)
        sc = a_cand.sum()
        if sc < 1e-15:
            break
        a_cand = a_cand / sc
        M_c = fast_conv(a_cand).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_cand.copy()
        a = a_cand
    return best_a

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent loaded: n={parent.size}, J={best.score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512)
        best.update(parent)

    # === PHASE 1: Nesterov smooth descent on parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Nesterov Smooth Descent", flush=True)
    print(f"{'='*60}", flush=True)

    n_main = parent.size
    p1_deadline = time.monotonic() + min(250.0, (deadline - time.monotonic()) * 0.2)
    a_smooth = nesterov_smooth(parent, p1_deadline, max_iters=8000,
                               tau_start=0.5, tau_end=400.0,
                               lr_init=0.4, lr_end=0.002)
    best.update(a_smooth)
    print(f"Phase 1 done: {time.monotonic() - best.t0:.0f}s total", flush=True)

    # === PHASE 2: Main LP-Anderson on best sequence ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP-Anderson Fixed-Point (n={best.seq.size})", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = min(time.monotonic() + 900.0, deadline - 200.0)
    a_lp = run_lp_anderson(best.seq, p2_deadline, max_iters=10000,
                           anderson_window=12, t_base=0.4, min_t=0.002,
                           verbose=True)
    best.update(a_lp)

    # Second pass with smaller step
    if time.monotonic() < deadline - 180:
        p2b_deadline = min(time.monotonic() + 400.0, deadline - 150.0)
        a_lp2 = run_lp_anderson(a_lp, p2b_deadline, max_iters=5000,
                                anderson_window=10, t_base=0.2, min_t=0.001,
                                verbose=True)
        best.update(a_lp2)

    # === PHASE 3: Plain LP backtracking for final polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Plain LP Backtracking Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 60:
        a_plain = run_lp_plain(best.seq, deadline - 40, max_iters=300)
        best.update(a_plain)

    # === PHASE 4: Try nearby lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        deltas = [0, -64, 64, -128, 128, -32, 32, 256, -256]
        for d in deltas:
            if time.monotonic() > deadline - 40:
                break
            n_try = n_best + d
            if n_try < 256 or n_try > 2560:
                continue
            if time.monotonic() > deadline - 30:
                break
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            # Quick Nesterov
            sub_dl = min(time.monotonic() + 20.0, deadline - 25)
            a_ext = nesterov_smooth(a_ext, sub_dl, max_iters=500,
                                    tau_start=5.0, tau_end=100.0,
                                    lr_init=0.2, lr_end=0.01)
            best.update(a_ext)
            # Quick LP
            if time.monotonic() < deadline - 15:
                a_ext_lp = run_lp_anderson(a_ext, min(time.monotonic() + 30.0, deadline - 10),
                                           max_iters=100, anderson_window=8,
                                           t_base=0.3, min_t=0.005, verbose=False)
                best.update(a_ext_lp)

    # === PHASE 5: Final Anderson pass on global best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        a_final = run_lp_anderson(best.seq, deadline - 2, max_iters=1000,
                                  anderson_window=12, t_base=0.15, min_t=0.001,
                                  verbose=True)
        best.update(a_final)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} total_time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
