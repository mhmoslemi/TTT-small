```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_A_for_constraints(a, k_indices):
    """Build matrix A where A[idx, i] = a[k_indices[idx] - i] if valid."""
    n = len(a)
    K = len(k_indices)
    A = np.zeros((K, n))
    for idx, k in enumerate(k_indices):
        # conv[k] = sum_i a[i] * a[k-i]
        # The coefficient of g_i in constraint k is a[k-i] (the fixed part)
        # Actually we need: sum_i A[idx,i] * g_i = sum_i a[k-i] * g_i
        # where a[k-i] is the fixed value (not g)
        valid = (k - np.arange(n))
        mask = (valid >= 0) & (valid < n)
        A[idx, mask] = a[valid[mask]]
    return A

def solve_lp_active(a, K=50, max_constraints=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for active k, sum(g)=1, g>=0.
    Uses only top K convolution positions as constraints.
    """
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    
    if max_constraints is None:
        max_constraints = min(K, 2 * n - 1)
    
    # Get top K indices (where constraints are most likely binding)
    top_idx = np.argsort(conv)[-max_constraints:][::-1]
    
    A = build_A_for_constraints(a, top_idx)
    
    # Variables: g (n vars) + M (1 var)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((max_constraints, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(max_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def alternating_lp_active(a, time_budget=100.0, K=50, verbose=False):

        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)

    direction = np.zeros(n)
    step_size = 0.3
    stall = 0

        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_active(a, K=K)
        if g_lp is None:
            break
        
        # Compute direction
        new_dir = g_lp - a
        direction = momentum * direction + (1 - momentum) * new_dir
        
        # Line search
        improved = False
        for t in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2]:

            a_try = np.maximum(a_try, 0)

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

def peak_shave(a, n_iters=200, time_budget=30.0):
    """
    Direct peak shaving: iteratively reduce the maximum convolution bin
    by adjusting coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find the pair (i, k_star-i) that contributes most to the peak
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Reduce the largest contributor
        i_max = int(np.argmax(contributions))
        j_max = k_star - i_max
        if j_max < 0 or j_max >= n or j_max == i_max:
            # Try to move mass elsewhere
            i_max = int(np.argmax(a))
        
        # Move some mass from i_max to a "good" position
        # Good position: one where it contributes less to the peak
        # and doesn't create a new peak
        
        # Find position with minimum contribution to ALL top conv bins
        top_ks = np.argsort(conv)[-10:]
        
        best_target = -1
        best_score = np.inf
        for t_pos in range(0, n, max(1, n // 100)):
            score = 0
            for k in top_ks:
                j = k - t_pos
                if 0 <= j < n:
                    score += a[j]
            if score < best_score:
                best_score = score
                best_target = t_pos
        
        if best_target < 0 or best_target == i_max:
            continue
        
        for frac in [0.01, 0.02, 0.05, 0.1]:
            delta = frac * a[i_max]
            a[i_max] -= delta
            a[best_target] += delta
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s < 1e-10:
                a[i_max] += delta
                a[best_target] -= delta
                continue
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                break
            else:
                a[i_max] += delta
                a[best_target] -= delta
    
    return best_a

def coordinate_descent(a, time_budget=30.0):
    """
    Fine-grained coordinate descent: try small adjustments to pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    n_rounds = 0
    while time.monotonic() - start < time_budget:
        n_rounds += 1
        # Pick a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try moving mass from higher to lower
        if a[i] < a[j]:
            i, j = j, i
        
        for frac in [0.01, 0.03, 0.1]:
            delta = frac * a[i]
            a_old_i = a[i]
            a_old_j = a[j]
            a[i] = a_old_i - delta
            a[j] = a_old_j + delta
            
            obj = compute_obj(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                conv = compute_conv(a)
                k_star = int(np.argmax(conv))
                break
            else:
                a[i] = a_old_i
                a[j] = a_old_j
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp(a, K=50):
    """LP with top-K active constraints. Returns normalized g or None."""
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    K = min(K, 2 * n - 1)
    top_idx = np.argsort(conv)[-K:][::-1]
    
    # Build A matrix: A[idx, i] = a[top_idx[idx] - i] if valid
    A = np.zeros((K, n))
    for idx in range(K):
        k = top_idx[idx]
        i_vals = np.arange(n)
        j_vals = k - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        A[idx, mask] = a[j_vals[mask]]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None
```

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines active-constraint LP alternating, projected subgradient descent,
peak shaving, coordinate descent, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_active(a, K=50, random_frac=0.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top-K (or random) k,
    sum(g)=1, g>=0. Returns normalized g or None.
    """
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    total_k = 2 * n - 1
    K = min(K, total_k)

    if random_frac > 0 and np.random.random() < random_frac:
        idx = np.random.choice(total_k, size=K, replace=False)
    else:
        idx = np.argsort(conv)[-K:][::-1]

    # Build A matrix efficiently
    A = np.zeros((K, n))
    for r in range(K):
        k = idx[r]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            A[r, i_start:i_end + 1] = a[k - i_end:k - i_start + 1][::-1]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None

    n = len(a)

    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    direction = np.zeros(n)
    eta = 0.5
    stall = 0
    rng = np.random.default_rng(n + 7)

        if time.monotonic() - start > time_budget:
            break

                sg[i] += a[j2]

                v = np.zeros(n)
                stall = 0

def coord_descent(a, time_budget=30.0, verbose=False):
    """
    Randomized coordinate descent: try small mass transfers between pairs.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99999)

    prob = a / a.sum()

    while time.monotonic() - start < time_budget:
        i = int(rng.choice(n, p=prob))
        j = int(rng.choice(n, p=prob))
        if i == j:
            continue

        if a[i] < a[j]:
            i, j = j, i

        for frac in [0.005, 0.02, 0.05]:
            delta = frac * a[i]
            old_i, old_j = a[i], a[j]
            a[i] = old_i - delta
            a[j] = old_j + delta
            a = np.maximum(a, 0)
            s2 = float(np.sum(a))
            if s2 < 1e-10:
                a[i] = old_i
                a[j] = old_j
                continue
            a = a / s2
            obj = compute_obj(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                prob = a / a.sum()
                break
            else:
                a[i] = old_i
                a[j] = old_j
                a = a / np.sum(a)

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:

    xo = np.linspace(0, 1, n)

    return np.maximum(np.interp(xn, xo, a), 0)
