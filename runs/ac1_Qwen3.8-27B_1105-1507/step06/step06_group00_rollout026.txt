```python
import numpy as np
import time
from scipy.fft import rfft, irfft

# Helper functions

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))

def compute_gradient(a, w):
    """Compute s = 2 * cross_correlation(w, a) for the objective."""
    n = len(a)
    a_flip = a[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a_flip, size)
    s_full = irfft(fw * fa, size)[:3*n-2]
    s = 2.0 * s_full[n-1 : 2*n-1]
    return s

def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def fast_gradient(a, time_budget=100.0, verbose=False):
    """
    Fast gradient method with momentum for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    momentum = np.zeros(n)
    alpha = 0.1
    beta = 10.0  # softmax sharpness
    gamma = 0.9  # momentum coefficient
    
    no_improve = 0
    for it in range(1000000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution
        conv = compute_conv(a)
        
        # Compute weights (soft-max emphasis on hot bins)
        # Use a shifted softmax to emphasize peaks
        conv_shifted = conv - conv.mean()
        w = np.exp(beta * conv_shifted / (conv.max() - conv.min() + 1e-10))
        w = w / w.sum()
        
        # Compute gradient
        s = compute_gradient(a, w)
        s = s - s.mean()  # center to maintain sum=1
        
        # Momentum update
        momentum = gamma * momentum - alpha * s
        a_new = a + momentum
        
        # Project
        a_new = project(a_new)
        
        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            if obj_new < best_obj * (1 - 0.001):
                no_improve = 0
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Adapt step size
            alpha = min(alpha * 1.01, 0.5)
        else:
            no_improve += 1
            if no_improve > 100:
                # Reduce step size or restart momentum
                alpha *= 0.9
                momentum *= 0.5
                no_improve = 0
            if no_improve > 500:
                # Full restart of momentum
                momentum = np.zeros(n)
                alpha = max(alpha * 0.5, 0.01)
                no_improve = 0
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Fast Grad it{it}: obj={best_obj:.8f}, alpha={alpha:.4f}, time={elapsed:.1f}s")
    
    return best_a, best_obj
```

But wait, I'm not sure the harmonic update is a good approximation to the LP. Let me think about it more carefully.

The LP objective is: min M s.t. (A@g)[k] <= M for all k, sum(g) = 1, g >= 0.

At the optimum, the active constraints are those k where (A@g)[k] = M. The dual variables lambda[k] > 0 for active k.

The KKT conditions give: for each i with g[i] > 0, sum_k lambda[k] * A[k,i] = 1 (the "reduced cost" is zero).

So the LP solution has the property that the "field" h[i] = sum_k lambda[k] * A[k,i] is constant (= 1) on the support of g, and > 1 outside the support.

The harmonic update g[i] = 1/(1+field[i]) is a smooth approximation to this: it gives larger g[i] where the field is small (and smaller where the field is large). In the limit of a "sharp" approximating function, it would give g[i] = 1/field[i] on the support (which is like the LP solution).

So the harmonic update is a reasonable heuristic. Let me use it.

Actually, I realize there's an even simpler and possibly better approach. The LP solution g* minimizes max_k (A@g)[k]. By duality, this is equivalent to finding a probability distribution lambda on the bins that "spreads out" the load as evenly as possible.

A iterative method for this is the "multiplicative weights update":
- Initialize g uniformly
- Repeat:
  - Compute v[k] = (A@g)[k] / M (normalized violations, M = max_k (A@g)[k])
  - Update: g[i] = g[i] * (1 - eta * sum_k v[k] * A[k,i])
  - Renormalize g to sum to 1

This is the " egalitarian" or "proportional fair" allocation. It should converge to a solution that equalizes the load across bins, which is related to (but not exactly) the LP optimum.

Actually, the MWU for "minimize the maximum" LP is:
- Track M (the current maximum)
- For each i, compute the "cost" c[i] = sum_k (v[k] * A[k,i]) where v[k] = A[k,:]·g / M
- Update: g[i] *= (1 - eta * c[i])

This is essentially the same as the harmonic update with a different weighting.

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast harmonic-alternating (FFT-based) as main workhorse,
LP refinement for polish, and multi-n exploration with random restarts.
Key innovation: replaces expensive LP calls with fast FFT-based field
computation and harmonic updates, enabling millions of cheap iterations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def project_onto_simplex(a):
    """Project a onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_field(a, w):
    """
    Compute field[i] = sum_k w[k] * a[k-i] (cross-correlation).
    Uses FFT for efficiency.
    """
    n = len(a)
    a_flip = a[::-1].copy()
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a_flip, size)
    s_full = irfft(fw * fa, size)
    # Extract the relevant part: field[i] = s_full[i + n - 1] for i in [0, n)
    field = s_full[n - 1: 2 * n - 1]
    return field


def harmonic_update(a, mode=0, beta=5.0):
    """
    Compute a 'best response' direction using harmonic update.
    mode=0: w = conv(a,a) normalized
    mode=1: w = exp(beta*(conv - mean)/std) normalized  
    mode=2: w = max(0, conv - mean)^2 normalized
    mode=3: w = uniform
    """
    n = len(a)
    conv = compute_conv(a)
    
    if mode == 0:
        w = conv / (conv.sum() + 1e-15)
    elif mode == 1:
        c_mean = conv.mean()
        c_std = conv.std() + 1e-10
        w = np.exp(beta * (conv - c_mean) / c_std)
        w = w / (w.sum() + 1e-15)
    elif mode == 2:
        c_mean = conv.mean()
        w = np.maximum(conv - c_mean, 0) ** 2
        ws = w.sum()
        w = w / ws if ws > 1e-15 else np.ones(2*n-1) / (2*n-1)
    else:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    
    field = compute_field(a, w)
    # Harmonic: g[i] ~ 1/(1 + field[i])
    g = 1.0 / (1.0 + np.maximum(field, 1e-10))
    g = project_onto_simplex(g)
    return g


def fast_alternate(a, time_budget=100.0, verbose=False, mode=0, beta=5.0):
    """
    Fast alternating method using harmonic updates.
    Alternates: g = harmonic(a), then a = harmonic(g), with blending.
    """
    n = len(a)
    a = project_onto_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t = 0.4
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        
        # Compute g from a
        g = harmonic_update(a, mode=mode, beta=beta)
        
        # Blend
        a_new = (1.0 - t) * a + t * g
        a_new = project_onto_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slowly increase t
            t = min(t + 0.005, 0.9)
        else:
            no_improve += 1
            if no_improve > 50:
                # Try reverse: a from g
                a_rev = harmonic_update(g, mode=mode, beta=beta)
                a_new2 = (1.0 - t) * a + t * a_rev
                a_new2 = project_onto_simplex(a_new2)
                obj_new2 = compute_obj(a_new2)
                if obj_new2 < best_obj - 1e-12:
                    best_obj = obj_new2
                    best_a = a_new2.copy()
                    a = a_new2
                    no_improve = 0
                else:
                    t *= 0.9
                    if t < 0.05:
                        # Perturb
                        rng = np.random.default_rng(it % (2**32))
                        a = best_a * np.exp(rng.normal(0, 0.05, n))
                        a = project_onto_simplex(a)
                        t = 0.3
                        no_improve = 0
            elif no_improve > 200:
                # Larger perturbation
                rng = np.random.default_rng(it % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = project_onto_simplex(a)
                t = 0.4
                no_improve = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      FastAlt it{it}: obj={best_obj:.8f}, t={t:.3f}, time={elapsed:.1f}s")
    
    # Final check with direct a update
    a_final = harmonic_update(best_a, mode=mode, beta=beta)
    for tt in [0.3, 0.5, 0.7, 1.0]:
        a_try = project_onto_simplex((1-tt)*best_a + tt*a_final)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try
    
    return best_a, best_obj


def gradient_step(a, time_budget=100.0, verbose=False):
    """
    Subgradient method with momentum targeting top peaks.
    """
    n = len(a)
    a = project_onto_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    alpha = 0.05
    gamma = 0.85
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        
        conv = compute_conv(a)
        
        # Weight: emphasize top bins
        c_thresh = np.percentile(conv, 50)
        w = np.maximum(conv - c_thresh, 0)
        ws = w.sum()
        if ws < 1e-15:
            w = conv / conv.sum()
        else:
            w = w / ws
        
        # Gradient: s[i] = 2 * sum_k w[k] * a[k-i]
        s = compute_field(a, w) * 2.0
        s = s - s.mean()  # center to maintain sum=1
        
        # Momentum update
        momentum = gamma * momentum - alpha * s
        a_new = project_onto_simplex(a + momentum)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if no_improve == 0 and it % 100 == 0:
                alpha = min(alpha * 1.02, 0.3)
        else:
            no_improve += 1
            if no_improve > 50:
                alpha *= 0.95
                momentum *= 0.5
            if no_improve > 300:
                # Perturb
                rng = np.random.default_rng(it % (2**32))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = project_onto_simplex(a)
                momentum = np.zeros(n)
                alpha = 0.03
                no_improve = 0
    
    return best_a, best_obj


def build_conv_matrix_sparse(n, a):
    """For small n, build the convolution matrix A where (A@g)[k] = conv(a,g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(2 * n - 1):
            i2 = k - j
            if 0 <= i2 < n:
                A[k, j] = a[i2]
    return A


def lp_minpeak_simple(a, time_limit=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    if n > 2000:
        return None, None  # Too large for scipy
    
    a = project_onto_simplex(a.copy())
    
    # Build A matrix: (A@g)[k] = conv(a, g)[k]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(2 * n - 1):
            i2 = k - j
            if 0 <= i2 < n:
                A[k, j] = a[i2]
    
    # LP: min M
    # Variables: g[0..n-1], M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub @ [g, M] <= b_ub:  A@g - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq @ [g, M] = b_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    from scipy.optimize import linprog
    start = time.monotonic()
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and time.monotonic() - start < time_limit:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    
    return None, None


def lp_alternate(a, time_budget=100.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = project_onto_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t = 0.5
    stall = 0
    
    while time.monotonic() - start < time_budget:

        g_lp, M_lp = lp_minpeak_simple(a, time_limit=min(remaining * 0.5, 15.0))
        if g_lp is None:

        # Line search

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_onto_simplex(a_new)


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        # Smooth
        freqs = rng.integers(1, 5, 3)
        phases = rng.uniform(0, 2*np.pi, 3)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2*np.pi*f*x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Bimodal
        centers = [n*0.3, n*0.7]
        widths = [n*0.1, n*0.15]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c)**2) / (2*w**2))
        a = a / a.sum()
    else:
        # Windowed
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / a.sum()
    return project_onto_simplex(a)

            a_try = resample_sequence(best_seq, n_try)
        
        # Quick fast alternate
        alloc = min(remaining * 0.15, 60.0)
        a_new, obj_new = fast_alternate(a_try, time_budget=alloc, mode=1, beta=5.0)
        
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_new:.8f}")
    
    # ============ PHASE 5: Random restarts with fast method ============
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_curr = len(best_seq)
    
    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        init_type = r % 5
        if init_type == 0:
            a_r = random_init(n_curr, rng)
        elif init_type == 1:
            # Perturbed best
            scale = 0.05 + 0.02 * (r % 3)
            a_r = best_seq * np.exp(rng.normal(0, scale, n_curr))
            a_r = project_onto_simplex(a_r)
        elif init_type == 2:
            # Different n
            n_r = max(300, min(1500, n_curr + rng.integers(-150, 150)))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = project_onto_simplex(a_r)
        elif init_type == 3:
            # Cosine window
            x = np.linspace(0, np.pi, n_curr)
            a_r = np.cos(x) ** 2 + 0.3
            a_r = project_onto_simplex(a_r)
        else:
            # Concentrated
            center = n_curr // 2
            width = n_curr // 6
            a_r = np.exp(-((np.arange(n_curr) - center) ** 2) / (2 * width ** 2))
            a_r = a_r + 0.1 * rng.uniform(0, 1, n_curr)
            a_r = project_onto_simplex(a_r)
        
        alloc = min(remaining * 0.1, 40.0)
        a_new, obj_new = fast_alternate(a_r, time_budget=alloc, mode=1, beta=5.0)
        
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Restart {r} (type={init_type}, n={len(a_new)}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (type={init_type}): obj={obj_new:.8f}")
    
    # ============ PHASE 6: Final intensive polish ============
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        # More fast alternating
        alloc = min(remaining * 0.3, 100.0)
        a_new, obj_new = fast_alternate(best_seq, time_budget=alloc, mode=1, beta=8.0)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Polish fast: obj={best_score:.8f} **")
        
        # Gradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.2, 50.0)
            a_new, obj_new = gradient_step(best_seq, time_budget=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Polish grad: obj={best_score:.8f} **")
        
        # LP if size allows
        remaining = deadline - time.monotonic()
        if remaining > 20 and len(best_seq) <= 1200:
            alloc = min(remaining * 0.3, 60.0)
            a_new, obj_new = lp_alternate(best_seq, time_budget=alloc, verbose=True)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Polish LP: obj={best_score:.8f} **")
    
    # ============ Final validation ============
    if best_seq is None:
        best_seq = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.array(result))
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
```
