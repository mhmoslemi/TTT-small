import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution: (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(n, m - j)
        A[j:j + k_end, j] = a[:k_end]
    return A


def solve_lp(a, M, time_limit=60):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(n, m - j)
        A[j:j + k_end, j] = a[:k_end]
    return A


def solve_lp(a, M, time_limit=60):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def soft_gradient(a, time_budget=60.0, max_steps=300):
    """
    Soft-max subgradient descent with FFT-accelerated gradient computation.
    Uses adaptive temperature and step size with backtracking.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    step_size = 0.04
    tau = 0.005

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive temperature: narrow as we converge
        tau_eff = max(tau, cmax / (5.0 * n))
        shifted = np.clip((conv - cmax) / tau_eff, -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            w_full = np.zeros(size)
            w_full[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
            w_full = np.zeros(size)
            w_full[:len(weights)] = weights

        # Gradient via FFT cross-correlation
        a_full = np.zeros(size)
        a_full[:n] = a

        Wf = rfft(w_full, size)
        Af = rfft(a_full, size)
        cross = irfft(Wf * np.conj(Af), size)
        grad = 2.0 * cross[:n]

        # Center gradient to preserve sum (approximately)
        grad_centered = grad - np.mean(grad)

        # Backtracking line search
        found_step = False
        for trial in range(5):
            a_try = a - step_size * grad_centered
            a_try = np.maximum(a_try, 0.0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                step_size *= 0.5
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)

            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                step_size = min(step_size * 1.05, 0.15)
                found_step = True
                break
            else:
                step_size *= 0.5

            if step_size < 1e-9:
                break
            # Small perturbation to escape
            rng = np.random.default_rng((step * 61 + 7) % (2**31))
            a = normalize(a * np.exp(rng.normal(0, 0.01, n)))

        tau *= 0.995

def spectral_flatten(a, max_iters=80, time_budget=50.0):
    """

            break

        A = np.fft.fft(a, size)
        mag = np.abs(A)
        max_mag = float(np.max(mag))
        if max_mag < 1e-30:
            break

        # Progressive power compression
        alpha = 0.6 + 0.35 * (it / float(max(iters_for_alpha := max_iters), 1))

        mag_new = max_mag * (mag_norm ** alpha)

        else:

            blend = 0.2 + 0.3 * (it / float(max_iters))
            a = normalize(blend * a_new + (1.0 - blend) * a)

def multiplicative_update(a, max_iters=100, time_budget=40.0):
    """
    Multiplicative update inspired by NMF: adjust each a[i] proportionally
    to how much it contributes to peak vs off-peak convolution values.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # For each i, compute "goodness" = weighted average of (M - conv[k])
        # over positions k where a[i] contributes
        # contrib[i] = sum_k (M - conv[k]) * a[k-i] / sum_k a[k-i]
        # Simplified: use the gradient of (sum of squared deviations from mean)

        # Compute: for each i, which k does it contribute to?
        # a[i] contributes to conv[k] for k = i, i+1, ..., i+n-1 (where a[k-i] > 0)
        # Weight by how "bad" those positions are

        # Use FFT for efficiency
        # "badness" of position k: (cmax - conv[k]) is good (low peak)
        # We want to increase a[i] if it contributes to "good" (low) positions
        # and decrease it if it contributes to "bad" (high) positions

        # Define potential: Phi[i] = sum_k (conv[k] - mean_conv) * a[k-i]
        # This is the correlation of (conv - mean) with a, which is the gradient
        mean_conv = float(np.mean(conv))
        residual = conv - mean_conv  # negative where low, positive where high

        # Gradient of sum_k residual[k] * conv[k] w.r.t. a[i]
        # = 2 * sum_k residual[k] * a[k-i] (weighted by residual)
        # Actually simpler: just use the standard gradient but weighted

        # Compute weighted gradient
        w_full = np.zeros(2 * n - 1)
        w_full[:] = residual

        size = 1
        while size < 3 * n:
            size <<= 1

        w_pad = np.zeros(size)
        w_pad[:len(w_full)] = w_full
        a_pad = np.zeros(size)
        a_pad[:n] = a

        Wf = rfft(w_pad, size)
        Af = rfft(a_pad, size)
        cross = irfft(Wf * np.conj(Af), size)
        grad = 2.0 * cross[:n]

        # Multiplicative update: a[i] *= exp(-lr * grad[i])
        lr = 0.1 * (1.0 + 0.5 * (it / float(max_iters)))
        a_new = a * np.exp(-lr * (grad - np.mean(grad)))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = normalize(0.5 * a + 0.5 * a_new)

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.10, 100.0)

            a_new = lp_optimize(a, max_iters=60, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score - 1e-10:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  LP cycle {cycle}: {best_score:.8f} (improved)")
            else:
                a = a_new
                print(f"  LP cycle {cycle}: {obj_new:.8f} (stalled)")
                # Escalate perturbation
                if cycle >= 2:
                    rng = np.random.default_rng(cycle * 7919 + 42)
                    sigma = 0.05 + 0.03 * cycle
                    a = normalize(a * np.exp(rng.normal(0, sigma, n_parent)))

    # ============ Phase 2: Diverse optimization methods ============
    print(f"\n{'='*50}")
    print(f"Phase 2: Gradient + Spectral + Multiplicative")
    print(f"{'='*50}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        # Soft gradient
        if time.monotonic() < deadline - 250:
            a_grad = soft_gradient(a, time_budget=50.0, max_steps=200)
            obj_grad = compute_obj(a_grad)
            if obj_grad < best_score:
                best_score = obj_grad
                best_seq = a_grad.copy()
                a = a_grad
                print(f"  Soft gradient: {best_score:.8f} **")
            else:
                print(f"  Soft gradient: {obj_grad:.8f}")
                a = normalize(0.6 * normalize(a) + 0.4 * normalize(a_grad))

        # Spectral flatten
        if time.monotonic() < deadline - 200:
            a_spec = spectral_flatten(a, max_iters=60, time_budget=40.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                a = a_spec
                print(f"  Spectral: {best_score:.8f} **")
            else:
                print(f"  Spectral: {obj_spec:.8f}")
                a = normalize(0.7 * normalize(a) + 0.3 * normalize(a_spec))

        # Multiplicative update
        if time.monotonic() < deadline - 150:
            a_mult = multiplicative_update(a, max_iters=80, time_budget=35.0)
            obj_mult = compute_obj(a_mult)
            if obj_mult < best_score:
                best_score = obj_mult
                best_seq = a_mult.copy()
                a = a_mult
                print(f"  Multiplicative: {best_score:.8f} **")
            else:
                print(f"  Multiplicative: {obj_mult:.8f}")

        # LP polish after Phase 2
        if time.monotonic() < deadline - 100:
            a_lp = lp_optimize(a, max_iters=40, time_budget=60.0)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a = a_lp
                print(f"  Post-Phase2 LP: {best_score:.8f} **")
            else:
                print(f"  Post-Phase2 LP: {obj_lp:.8f}")

    # ============ Phase 3: Multi-n exploration ============
    print(f"\n{'='*50}")
    print(f"Phase 3: Multi-n exploration")
    print(f"{'='*50}")

    if best_seq is not None:
        n_cur = len(best_seq)
        # Explore a range of n values
        n_options = set()
        for delta in range(-300, 400, 25):
            n_try = n_cur + delta
            if 200 <= n_try <= 2000:
                n_options.add(n_try)
        # Also try some specific values
        for n_sp in [300, 400, 500, 600, 700, 800, 900, 1000, 1100, 1200]:
            if 200 <= n_sp <= 2000:
                n_options.add(n_sp)
        n_options = sorted(n_options)

        # Prioritize n values close to current best
        n_options.sort(key=lambda x: abs(x - n_cur))

        for n_try in n_options[:20]:  # Try at most 20 different n values
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if n_try == n_cur:
                a_try = normalize(best_seq.copy())
            else:
                a_try = normalize(resample(best_seq, n_try))

            alloc = min(remaining * 0.06, 25.0)
            # Quick optimization at this n
            a_try = lp_optimize(a_try, max_iters=15, time_budget=alloc * 0.5)
            a_try = soft_gradient(a_try, time_budget=alloc * 0.3, max_steps=40)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_cur = n_try
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # ============ Phase 4: Random restarts with diverse strategies ============
    print(f"\n{'='*50}")
    print(f"Phase 4: Random restarts")
    print(f"{'='*50}")

    if best_seq is not None:
        n_final = len(best_seq)
        rng = np.random.default_rng(12345)

        for restart in range(12):
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.07, 35.0)

            strategy = restart % 6
            if strategy == 0:
                # Gaussian multiplicative
                a_try = best_seq * np.exp(rng.normal(0, 0.08, n_final))
            elif strategy == 1:
                # Sinusoidal at random frequency
                freq = rng.uniform(1.0, 10.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.15 * np.sin(2 * np.pi * freq * np.arange(n_final) / n_final + phase)
                a_try = best_seq * mod
            elif strategy == 2:
                # Blend with Dirichlet
                a_rand = rng.dirichlet(np.ones(n_final) * 0.3)
                a_try = 0.65 * normalize(best_seq) + 0.35 * a_rand
            elif strategy == 3:
                # Block operations: reverse + shift
                a_try = best_seq.copy()
                blen = int(rng.integers(n_final // 6, n_final // 3))
                sidx = int(rng.integers(0, n_final - blen))
                a_try[sidx:sidx + blen] = a_try[sidx:sidx + blen][::-1]
            elif strategy == 4:
                # Two-frequency amplitude modulation
                f1 = rng.uniform(0.5, 4.0)
                f2 = rng.uniform(5.0, 15.0)
                mod = 1.0 + 0.1 * np.sin(2 * np.pi * f1 * np.arange(n_final) / n_final) \
                          + 0.08 * np.cos(2 * np.pi * f2 * np.arange(n_final) / n_final)
                a_try = best_seq * mod
            else:
                # Peak targeting: reduce middle, boost edges or vice versa
                weights = np.ones(n_final)
                center = n_final // 2
                spread = n_final // 4
                for i in range(n_final):
                    dist = abs(i - center) / float(spread)
                    weights[i] = 1.0 + 0.2 * np.cos(np.pi * min(dist, 1.0))
                a_try = best_seq * weights * np.exp(rng.normal(0, 0.03, n_final))

            a_try = normalize(a_try)
            a_try = lp_optimize(a_try, max_iters=20, time_budget=alloc * 0.5)
            a_try = soft_gradient(a_try, time_budget=alloc * 0.3, max_steps=40)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Restart {restart} (strat {strategy}): {best_score:.8f} **")
            else:
                print(f"  Restart {restart} (strat {strategy}): {obj_try:.8f}")

    # ============ Phase 5: Final intensive polish ============
    print(f"\n{'='*50}")
    print(f"Phase 5: Final polish")
    print(f"{'='*50}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = len(a_final)

        for rnd in range(8):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            alloc = min(remaining * 0.12, 30.0)

            # Cycle through methods
            if rnd % 3 == 0:
                a_pol = lp_optimize(a_final, max_iters=25, time_budget=alloc * 0.6)
            elif rnd % 3 == 1:
                a_pol = soft_gradient(a_final, time_budget=alloc * 0.6, max_steps=50)
            else:
                a_pol = multiplicative_update(a_final, max_iters=40, time_budget=alloc * 0.5)

            # Light spectral polish
            a_pol = spectral_flatten(a_pol, max_iters=10, time_budget=alloc * 0.2)

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {rnd}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {rnd}: {obj_pol:.8f}")

    # ============ Final validation ============
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 200

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = min(best_score, val)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
