Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternation with momentum-enhanced gradient, peak-pair balancing,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses dual LP formulations (max-sum and min-peak), Nesterov momentum gradient,
peak-pair balancing, simulated annealing, and multi-n exploration.
Key innovations over previous: min-peak LP, momentum, peak-pair balancing,
sharper gradient targeting, and simulated annealing for exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_min_peak(a, time_limit=15):
    """LP: min t s.t. conv(a,g) <= t, sum(g)=1, g>=0. Returns (g, t) or None."""
    n = len(a)
    A = build_toeplitz(a)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Minimize t => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Inequality: A*g - t <= 0 for each row => [A | -I] * [g; t] <= 0
    A_ineq = np.zeros((2 * n - 1, n + 1))
    A_ineq[:, :n] = A
    A_ineq[np.arange(2 * n - 1), n] = -1.0
    b_ineq = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None


def sharp_subgradient(a, K=8):
    """Compute subgradient targeting top-K peaks of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    # Get top-K peak positions
    top_k = np.argpartition(conv, -K)[-K:] if K < len(conv) else np.arange(len(conv))
    grad = np.zeros(n)
    for k in top_k:
        k = int(k)
        for i in range(n):
            v1 = k - i
            v2 = i - k
            if 0 <= v1 < n:
                grad[i] += a[v1]
            if 0 <= v2 < n:
                grad[i] += a[v2]
    return grad / float(len(top_k))


def soft_gradient(a, tau=0.003):
    """Compute soft gradient (log-sum-exp weighting) of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    if cmax < 1e-30:
        return np.zeros(n)
    shifted = np.clip((conv - cmax) / tau, -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = float(np.sum(weights))
    if wsum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
        wsum = 1.0
    else:
        weights /= wsum
    # Compute gradient via FFT
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    w_ext = np.zeros(2 * n - 1)
    w_ext[:] = weights
    a_flip = a[::-1]
    Wf = rfft(w_ext, size)
    Af = rfft(a_flip, size)
    corr = irfft(Wf * Af, size)
    return 2.0 * corr[n - 1:2 * n - 1]


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gradient_descent_momentum(a, max_steps=300, time_budget=40.0, K=8,
                              lr_init=0.03, beta=0.7, tau_soft=0.003):
    """Nesterov momentum gradient descent with sharp subgradient."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        a_look = a - beta * v
        a_look = np.maximum(a_look, 0.0)
        s = float(np.sum(a_look))
        if s > 1e-15:
            a_look = a_look / s
        else:
            a_look = a.copy()

        # Compute gradient at lookahead point
        if step % 3 == 0:
            grad = sharp_subgradient(a_look, K=K)
        else:
            grad = soft_gradient(a_look, tau=tau_soft)

        # Center gradient (remove mean to preserve sum)
        grad_c = grad - np.mean(grad)

        # Update momentum
        v = beta * v + (1.0 - beta) * grad_c
        a_new = a - lr * v
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slightly increase lr on improvement
            lr = min(lr * 1.02, lr_init * 2.0)
        else:
            a = a_new  # Still move (momentum)
            lr *= 0.7

        # Decay lr slowly
        lr = max(lr * 0.999, lr_init * 0.3)

    return best_a


def peak_pair_balancing(a, time_budget=20.0, top_K=5):
    """Direct peak-pair balancing: for each top peak, balance contributing pairs."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for _ in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        top_peaks = np.argpartition(conv, -top_K)[-top_K:]

        improved = False
        for k in top_peaks:
            k = int(k)
            # Find contributing pairs
            pairs = []
            for j in range(n):
                k2 = k - j
                if 0 <= k2 < n and j != k2:
                    pairs.append((a[j] * a[k2], j, k2))
            pairs.sort(reverse=True)

            for prod, j1, j2 in pairs[:3]:
                if j1 >= j2:
                    continue
                # Try to balance: move mass from larger to smaller
                if a[j1] > a[j2]:
                    src, dst = j1, j2
                else:
                    src, dst = j2, j1
                delta = 0.1 * (a[src] - a[dst])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                a_try = project_nonneg_sum1(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            # Random perturbation to escape
            a = normalize(a * np.exp(rng.normal(0, 0.02, n)))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def spectral_flatten(a, max_iters=40, time_budget=30.0):
    """Spectral equalization with adaptive compression."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    from scipy.fft import fft, ifft

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Adaptive compression: more aggressive early, less later
        alpha = 0.3 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a

            else:
                print(f"  Momentum gradient: {obj2:.8f}")
            a = a2 if obj2 < compute_obj(a1) else a1

        # Round 3: LP min-peak
        if time.monotonic() < deadline - 200:
            a3 = lp_alternation_enhanced(a, max_iters=20, time_budget=allocated * 0.25,
                                         use_min_peak=True)
            obj3 = compute_obj(a3)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  LP min-peak: {best_score:.8f} **")
            else:
                print(f"  LP min-peak: {obj3:.8f}")
            a = a3

        # Round 4: Peak-pair balancing
        if time.monotonic() < deadline - 200:
            a4 = peak_pair_balancing(a, time_budget=min(30, allocated * 0.15))
            obj4 = compute_obj(a4)
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"  Peak-pair: {best_score:.8f} **")
            else:
                print(f"  Peak-pair: {obj4:.8f}")
            a = a4

    # =========================================================
    # Phase 2: Spectral + combined optimization
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Spectral + combined (n={len(best_seq) if best_seq is not None else n_base})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())

        if time.monotonic() < deadline - 150:
            a_spec = spectral_flatten(a, max_iters=40, time_budget=35.0)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                print(f"  Spectral: {best_score:.8f} **")
            else:
                print(f"  Spectral: {obj_spec:.8f}")
                a = normalize(0.7 * a + 0.3 * a_spec)

        # Follow-up LP
        if time.monotonic() < deadline - 120:
            a_new = lp_alternation_enhanced(a, max_iters=20, time_budget=40.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Post-spectral LP: {best_score:.8f} **")
            else:
                print(f"  Post-spectral LP: {obj_new:.8f}")

    # =========================================================
    # Phase 3: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        n_options = sorted(set([
            max(200, n_cur - 100), max(200, n_cur - 50),
            n_cur,
            n_cur + 50, n_cur + 100, n_cur + 200
        ]))
        n_options = [n for n in n_options if 200 <= n <= 1200]

        for n_try in n_options:
            if time.monotonic() > deadline - 60:
                break
            if n_try == n_cur:
                continue

            a_try = normalize(resample_sequence(best_seq, n_try))
            alloc = min(30, (deadline - time.monotonic()) * 0.1)

            # Quick optimization
            if n_try <= 900 and alloc > 10:
                a_try = lp_alternation_enhanced(a_try, max_iters=10, time_budget=alloc * 0.5)
            a_try = gradient_descent_momentum(a_try, max_steps=80, time_budget=alloc * 0.4,
                                              K=6, lr_init=0.02)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # =========================================================
    # Phase 4: Simulated annealing with diverse moves
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Simulated annealing")
    print(f"{'='*60}")

    if best_seq is not None:
        n_sa = len(best_seq)
        a_sa = normalize(best_seq.copy())
        obj_sa = best_score
        T_init = 0.005
        T = T_init
        cooling = 0.95
        rng = np.random.default_rng(seed + 100)

        sa_start = time.monotonic()
        sa_budget = min(120, (deadline - start_time) * 0.12)
        n_iter = 0

        while time.monotonic() - sa_start < sa_budget and T > 1e-6:
            n_iter += 1
            remaining = sa_budget - (time.monotonic() - sa_start)
            if remaining < 5:
                break

            # Choose move type
            move_type = rng.integers(0, 4)

            if move_type == 0:
                # Gradient step
                a_new = gradient_descent_momentum(a_sa, max_steps=30, time_budget=min(8, remaining * 0.3),
                                                  K=5, lr_init=0.02, beta=0.6)
            elif move_type == 1:
                # LP step (only for small n)
                if n_sa <= 800:
                    a_new = lp_alternation_enhanced(a_sa, max_iters=5, time_budget=min(10, remaining * 0.3))
                else:
                    a_new = gradient_descent_momentum(a_sa, max_steps=30, time_budget=min(8, remaining * 0.3),
                                                      K=5, lr_init=0.02)
            elif move_type == 2:
                # Peak-pair balancing
                a_new = peak_pair_balancing(a_sa, time_budget=min(8, remaining * 0.3), top_K=3)
            else:
                # Random perturbation + gradient
                sigma = 0.05 + 0.05 * T / T_init
                a_pert = normalize(a_sa * np.exp(rng.normal(0, sigma, n_sa)))
                a_new = gradient_descent_momentum(a_pert, max_steps=20, time_budget=min(5, remaining * 0.2),
                                                  K=5, lr_init=0.02)

            obj_new = compute_obj(a_new)
            delta = obj_new - obj_sa

            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                a_sa = a_new
                obj_sa = obj_new
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  SA iter {n_iter}: {best_score:.8f} **")

            T *= cooling

        print(f"  SA done: {n_iter} iters, best={best_score:.8f}")

    # =========================================================
    # Phase 5: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a_final = normalize(best_seq.copy())

        polish_rounds = 5
        for r in range(polish_rounds):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.3, 40.0)

            # LP
            if n_final <= 900 and alloc > 15:
                a_pol = lp_alternation_enhanced(a_final, max_iters=12, time_budget=alloc * 0.4)
            else:
                a_pol = a_final.copy()

            # Gradient
            if time.monotonic() < deadline - 10:
                a_pol = gradient_descent_momentum(a_pol, max_steps=60, time_budget=alloc * 0.3,
                                                  K=5, lr_init=0.01, beta=0.5)

            # Peak-pair
            if time.monotonic() < deadline - 10:
                a_pol = peak_pair_balancing(a_pol, time_budget=min(10, alloc * 0.2), top_K=3)

            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Polish {r}: {best_score:.8f} **")
            else:
                a_final = a_pol
                print(f"  Polish {r}: {obj_pol:.8f}")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
