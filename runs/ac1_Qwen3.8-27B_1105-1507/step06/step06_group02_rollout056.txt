import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_cross(a, b):
    n1, n2 = len(a), len(b)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n1 + n2 - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def solve_lp_min_M(a, time_limit=20):
    n = len(a)
    rows, cols, vals = build_conv_sparse(n, a)
    n_rows = 2 * n - 1
    n_cols = n + 1
    all_rows = rows + list(range(n_rows))
    all_cols = cols + [n] * n_rows
    all_vals = vals + [-1.0] * n_rows
    A_ub = csr_matrix((all_vals, (all_rows, all_cols)), shape=(n_rows, n_cols))
    c = np.zeros(n_cols)
    c[n] = 1.0
    A_eq = np.zeros((1, n_cols))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0, None)] * n + [(0, None)]
    try:
        r = linprog(c, A_ub=A_ub, b_ub=np.zeros(n_rows),
                    A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                    method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            M_star = float(r.x[n])
            g_star = np.maximum(r.x[:n], 0)
            gs = g_star.sum()
            if gs > 1e-10:
                g_star /= gs
                return M_star, g_star, r
    except Exception:
        pass
    return None, None, None


def solve_lp_max_sum(a, M, time_limit=20):
    n = len(a)
    rows, cols, vals = build_conv_sparse(n, a)
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                    method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None and r.x.sum() > 1e-10:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_fixed_point(a, deadline, max_iters=500, verbose=True, perturb_every=40,
                   perturb_size=0.02):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = compute_J(best_a)
    t = 0.4
    no_imp = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            best_J = compute_J(best_a)
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP iter {it}: M={M:.9f} J={best_J:.8f} t={t:.3f}", flush=True)

        g = solve_lp_min_M(a)
        if g[1] is None:
            g_alt = solve_lp_max_sum(a, M)
            if g_alt is None:
                break
            g_star = g_alt
            gs = g_star.sum()
            if gs < 1e-10:
                break
            g_star = g_star / gs
        else:
            g_star = g[1]

        best_t = t
        best_M_trial = M
        trial_ts = sorted(set([min(1.0, t * 1.3), t, max(0.05, t * 0.7),
                               max(0.02, t * 0.4), max(0.01, t * 0.2)]))
        for t_trial in trial_ts:
            a_trial = (1.0 - t_trial) * a + t_trial * g_star
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_t = t_trial

        a = (1.0 - best_t) * a + best_t * g_star
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s
        t = best_t

        M_now = fast_conv(a).max()
        if M_now < best_M * (1 - 1e-12):
            no_imp = 0
        else:
            no_imp += 1
            t = max(0.02, t * 0.8)
            if no_imp > 30:
                if verbose:
                    print(f"    LP stopped at iter {it}: converged (no imp {no_imp})", flush=True)
                break

        if perturb_every > 0 and it % perturb_every == perturb_every - 1 and no_imp < 15:
            a_pert = a + perturb_size * rng.normal(0, 1, n)
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert /= a_pert.sum()
            M_pert = fast_conv(a_pert).max()
            if M_pert < best_M:
                a = a_pert
                best_M = M_pert
                best_a = a.copy()
                best_J = compute_J(best_a)
                t = 0.3

    return best_a


def softmax_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted - shifted.max())
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * p[k] * a[k - j]
    return grad, p, b, b_max


def nesterov_smooth(a, deadline, max_iters=3000, tau=50.0, lr_init=0.3, lr_end=0.005):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        lr = lr_init * (lr_end / lr_init) ** (frac ** 0.7)
        tau_cur = 5.0 * (tau / 5.0) ** (frac ** 0.5)
        grad, p, b, b_max = softmax_max_grad(y, tau_cur)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad / y.sum() + beta * (y - a) / y.sum()), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def gs_flat(a, n_iters=200, beta=0.85):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        mag_new = (1.0 - beta) * mag + beta * (mag.mean() + 1e-30)
        power = 0.3 + 0.4 * beta
        mag_alt = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_alt
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.005))
        inits.append(gs_flat(p, n_iters=100, beta=0.9))
    inits.append(np.ones(n) / n)
    # Ramp (linearly increasing)
    inits.append(np.maximum(t + 0.1, 0.01))
    # Anti-ramp
    inits.append(np.maximum(1.0 - t + 0.1, 0.01))
    # Step (half high, half low)
    inits.append(np.where(t < 0.5, 1.0, 0.3))
    inits.append(np.where(t < 0.3, 0.3, 1.0))
    # S-curve
    inits.append(1.0 / (1.0 + np.exp(-10 * (t - 0.5))))
    inits.append(1.0 / (1.0 + np.exp(-5 * (t - 0.3))))
    # Gaussian bumps
    inits.append(np.exp(-20 * (t - 0.3) ** 2) + 0.2 * np.exp(-20 * (t - 0.7) ** 2) + 0.05)
    inits.append(np.exp(-50 * (t - 0.2) ** 2) + 0.3 * np.exp(-50 * (t - 0.8) ** 2) + 0.05)
    # Cosine windows
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.45 * np.cos(2 * np.pi * f * t - np.pi), 0.01))
    # Power law
    for alpha in [0.5, 1.5, 3.0]:
        inits.append(np.maximum(t ** alpha + 0.1, 0.01))
        inits.append(np.maximum((1 - t) ** alpha + 0.1, 0.01))
    # Random smooth
    k = max(5, n // 15)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.4, n), 0.01)
    inits.append(gs_flat(a_rand, n_iters=80, beta=0.85))
    # Triangular
    tri = np.minimum(t, 1 - t) * 2
    inits.append(np.maximum(tri + 0.1, 0.01))
    # V-shaped (anti-triangle)
    inits.append(np.maximum(1.0 - tri * 0.8 + 0.2, 0.01))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):

            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def lbfgs_smooth_optimize(a, deadline, tau=100.0, max_iter=200):
    """Use L-BFGS on the smoothed objective to polish."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()

    def objective(x):
        if time.monotonic() > deadline:
            return 1e10
        x_pos = np.maximum(x, 1e-10)
        x_pos = x_pos / x_pos.sum()
        b = fast_conv(x_pos)
        b_max = float(b.max())
        # Smooth max approximation
        shifted = tau * b
        shifted = shifted - shifted.max()
        log_sum_exp = np.log(np.sum(np.exp(shifted))) + shifted.max() / tau
        return 2.0 * n * log_sum_exp

    def gradient(x):
        if time.monotonic() > deadline:
            return np.zeros(n)
        x_pos = np.maximum(x, 1e-10)
        x_pos = x_pos / x_pos.sum()
        b = fast_conv(x_pos)
        shifted = tau * b
        shifted = shifted - shifted.max()
        p = np.exp(shifted)
        p = p / p.sum()
        grad = np.zeros(n)
        active = np.where(p > 1e-12)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * p[k] * x_pos[k - j]
        # Account for normalization
        grad = grad - x_pos * (grad @ x_pos) / x_pos.sum()
        # Scale by 2n / S^2
        return 2.0 * n * grad

    try:
        res = minimize(objective, a, jac=gradient, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'gtol': 1e-12,
                                'maxls': 50, 'maxfun': 5000})
        if res.success and time.monotonic() < deadline:
            a_opt = np.maximum(res.x, 0)
            s = a_opt.sum()
            if s > 1e-10:
                a_opt /= s
                return a_opt
    except Exception:

    return a
