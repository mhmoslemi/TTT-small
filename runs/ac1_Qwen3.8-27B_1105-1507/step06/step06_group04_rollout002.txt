Combined log-sum-exp smoothed optimization with Adam, alternating min-peak LP,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):
    """
    Solve: min t s.t. (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None

                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:

def smoothed_gradient(a, tau):
    """
    Compute gradient of smoothed objective: 2n * LSE(tau*conv(a,a)) / (sum a)^2.
    Returns (gradient, smoothed_max_value).
    """
    n = len(a)
    s = float(np.sum(a))
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(np.clip(tau * (conv - cmax), -100, 100))
    w_sum = np.sum(exp_vals)
    weights = exp_vals / w_sum
    L = cmax + (1.0 / tau) * np.log(w_sum)

    # dL/da_i = 2 * sum_j a[j] * weights[i+j]
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    w_ext = np.zeros(fft_size)
    w_ext[:2 * n - 1] = weights
    W_ext = rfft(w_ext, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(A * np.conj(W_ext), fft_size)
    dL = 2.0 * corr[:n]

    grad = (2.0 * n / (s * s)) * (dL - 2.0 * L * a / s)
    return grad, L

def parametric_sequence(n, sigma, freq, amp, phase):
    """Bell shape with sinusoidal modulation."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    bell = np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = bell * mod
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def generate_diverse_starts(n, num_starts=12, seed=42):
    """Generate diverse initial sequences."""

    starts.append(np.ones(n) / n)

    tri = 1.0 - np.abs(i - center) / center
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    bell = np.exp(-((i - center) ** 2) / (2 * (n / 5.0) ** 2))
    starts.append(bell / np.sum(bell))

    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 2.5) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    noisy = np.abs(np.ones(n) + rng.normal(0, 0.3, n))

            unique.append(s)
    return unique


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15 and n <= 800:
        t3 = min(remaining * 0.3, 30.0)
        a3, obj3 = minpeak_lp_alternate(best_a, max_iters=30, time_budget=t3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    # Phase 4: Peak chase
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t4 = min(remaining * 0.4, 15.0)
        a4, obj4 = chase_peaks(best_a, num_peaks=5, num_steps=int(t4 * 10), time_budget=t4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    # Phase 5: Final subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t5 = remaining * 0.8
        a5, obj5 = subgradient_fft(best_a, max_iters=2000, time_budget=t5,
                                    tau_init=0.005, tau_final=0.001)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start diverse exploration with LSE-Adam optimization,
    alternating LP, and multi-scale polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: best={best_score:.8f}, n={n_parent}")

    # Phase 1: Multi-start exploration
    print("\n=== Phase 1: Multi-start exploration ===")

    n_values = sorted(set([n_parent, n_parent - 50, n_parent + 50,
                           n_parent - 100, n_parent + 100]))
    n_values = [nv for nv in n_values if 200 <= nv <= 1200]

    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 250:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n == n_parent and best_seq is not None:
            starts = [normalize_simplex(best_seq.copy())]
            starts.extend(generate_diverse_starts(n, num_starts=8, seed=seed))
        else:
            starts = generate_diverse_starts(n, num_starts=8, seed=seed + n * 7)

        print(f"  n={n}: {len(starts)} starts, {remaining:.0f}s remaining")

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 200:
                break
            t_remaining = deadline - time.monotonic()
            if t_remaining < 15:
                break

            alloc = min(t_remaining / max(len(starts) - si, 1), 20.0)
            if alloc < 5:
                break

            # Quick LSE-Adam
            a_opt, obj_opt = lse_adam_optimize(s, max_iters=1500, time_budget=alloc * 0.5,
                                                tau_init=3.0, tau_final=40.0, lr_init=0.005)
            # Quick subgradient
            t_rem2 = deadline - time.monotonic()
            if t_rem2 > 100:
                a_opt2, obj_opt2 = subgradient_fft(a_opt, max_iters=1000,
                                                    time_budget=min(alloc * 0.3, 10.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"    start {si}: {obj_opt:.8f} ** NEW BEST **")
            candidates.append((obj_opt, a_opt.copy()))

    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive polishing of top candidates
    print("\n=== Phase 2: Intensive polishing ===")

    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]
        alloc = min(remaining * 0.25, 100.0)
        print(f"  Cand {ci} (n={len(a_cand)}, obj={obj_cand:.8f}): {alloc:.0f}s")

        a_pol, obj_pol = polish_pipeline(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"  No improvement ({obj_pol:.8f})")

    # Phase 3: Length exploration
    print(f"\n=== Phase 3: Length exploration (best: {best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = sorted(set([n_best - 30, n_best + 30, n_best - 60, n_best + 60,
                                    n_best - 100, n_best + 100]))
        test_lengths = [x for x in test_lengths if 300 <= x <= 1000]

        for n_test in test_lengths:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_res = normalize_simplex(resample_sequence(best_seq, n_test))
            alloc = min(remaining * 0.2, 30.0)
            a_pol, obj_pol = polish_pipeline(a_res, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** {best_score:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f}")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_final, obj_final = polish_pipeline(best_seq, time_budget=remaining * 0.5)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Polish: {best_score:.8f}")

            # Perturbation round
            remaining = deadline - time.monotonic()
            if remaining > 20:
                n_f = len(best_seq)
                rng = np.random.default_rng(seed + 77777)
                for pi in range(8):
                    if time.monotonic() > deadline - 10:
                        break
                    scale = 0.005 + 0.01 * pi
                    a_p = best_seq * np.exp(rng.normal(0, scale, n_f))
                    a_p = normalize_simplex(a_p)
                    t_a = min(remaining * 0.1, 15.0)
                    if t_a < 3:
                        break
                    a_p, obj_p = lse_adam_optimize(a_p, max_iters=1000,
                                                    time_budget=t_a * 0.6,
                                                    tau_init=5.0, tau_final=50.0,
                                                    lr_init=0.003)
                    if time.monotonic() < deadline - 15:
                        a_p2, obj_p2 = subgradient_fft(a_p, max_iters=500,
                                                        time_budget=min(5.0, deadline - time.monotonic() - 10))
                        if obj_p2 < obj_p:
                            a_p, obj_p = a_p2, obj_p2
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        print(f"  Perturb {pi}: ** {best_score:.8f} **")

    # Phase 5: Final LP polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_b = len(best_seq)
        if n_b <= 700 and remaining > 20:
            print(f"\n=== Phase 5: Final LP (n={n_b}) ===")
            a_lp, obj_lp = minpeak_lp_alternate(best_seq, max_iters=20,
                                                 time_budget=min(remaining * 0.5, 40.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_lp2, obj_lp2 = maxsum_lp_alternate(best_seq, max_iters=15,
                                                      time_budget=min(remaining * 0.6, 25.0))
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  MaxSum LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
