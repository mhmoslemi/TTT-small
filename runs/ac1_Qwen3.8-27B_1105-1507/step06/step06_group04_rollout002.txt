"""
Optimized step function search using LP alternating, subgradient refinement,
coordinate descent, and multi-start exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: better LP line search, top-K peak subgradient, coordinate descent polish,
adaptive continuation, and diverse parametric starts.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_conv_matrix(u):
    """Build Toeplitz matrix A where (A @ v)[k] = (convolve(u, v))[k]."""
    n = len(u)
    # Range of k: 0 to 2n-2
    # For each k, (u*v)[k] = sum_j u[j] * v[k-j] where valid indices
    # A[k, j] = u[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k, intersected with [0, n-1]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # u[j] * v[k-j] contributes to position k = j + m for m = 0..n-1
        lo = j
        hi = j + n
        A[lo:hi, j] = u[:hi - lo]
    return A


def solve_minpeak_lp(u, time_limit=10):
    """
    Solve: min t s.t. (u * v)[k] <= t for all k, sum(v)=1, v>=0.
    Returns (v, t_opt) or (None, None).
    """
    n = len(u)
    A = build_conv_matrix(u)
    # Variables: v[0..n-1], t
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            v = np.maximum(res.x[:n], 0)
            s = np.sum(v)
            if s > 1e-10:
                v = v / s
                return v, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(u, M, time_limit=10):
    """
    Solve: max sum(v) s.t. (u * v)[k] <= M for all k, v >= 0.
    Returns v normalized to sum=1, or None.

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

def subgradient_topk(a, max_iters=2000, time_budget=60.0, num_peaks=3):
    """
    Subgradient method targeting top-K peaks with Nesterov momentum.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.8
    step = 0.05
    tau = 0.01

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Soft weights over peaks
        shifted = np.clip((conv - cmax) / tau, -400.0, 400.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via FFT cross-correlation
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W = rfft(w_ext, fft_size)
        A = rfft(a, fft_size)
        corr = irfft(A * np.conj(W), fft_size)
        grad = 2.0 * corr[:n]

        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        step *= 0.9995
        tau *= 0.999

    return best_a, best_obj


def coordinate_descent(a, max_iters=5000, time_budget=60.0):
    """
    Coordinate descent: optimize one coordinate at a time.
    For each coordinate i, find optimal a[i] given others fixed.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute: for each i, which conv positions does a[i] affect?
    # conv[k] contains a[i]*a[k-i] if 0 <= k-i < n
    # So a[i] affects conv[k] for k in [i, i+n-1] (clamped to [0, 2n-2])

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            # Compute current conv
            conv = compute_conv(a)
            cmax = float(np.max(conv))

            # Try to reduce cmax by adjusting a[i]
            # Binary search for optimal a[i]
            # We want to minimize max(conv) after changing a[i]
            # The conv values that change are those at k where 0<=k-i<n
            
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Current values of affected conv
            # conv[k] = sum_j a[j]*a[k-j], the terms with a[i] are:
            # a[i]*a[k-i] (when j=i) + a[k-i]*a[i] (when j=k-i, if k-i != i)
            # = a[i] * a[k-i] + (a[i]*a[k-i] if k-i != i else 0)
            # Let's compute the "base" conv without a[i] contribution
            
            old_aval = a[i]
            
            # For each k in [k_lo, k_hi], the change in conv[k] when a[i] changes to x:
            # delta_k = x*a[k-i] + (x*a[k-i] if k-i != i else 0) - old_aval*a[k-i] - (old_aval*a[k-i] if k-i != i else 0)
            # = (x - old_aval) * a[k-i] * (1 + (1 if k-i != i else 0))
            
            # So conv_new[k] = conv[k] + (x - old_aval) * coeff[k]
            # where coeff[k] = a[k-i] * (1 + (1 if k-i != i else 0))
            
            # We want to find x >= 0 that minimizes max_k conv_new[k]
            # subject to sum(a_new) = 1 (so other elements scale)
            
            # This is complex with the normalization. Let's simplify:
            # Just try a few values of x and pick the best after normalization
            
            # Candidates for x: slightly less, slightly more, much less
            sum_rest = float(np.sum(a) - old_aval)
            if sum_rest < 1e-15:
                continue
            
            best_x = old_aval
            best_obj_local = best_obj
            
            for frac in [0.0, 0.2, 0.5, 0.8, 1.2, 1.5, 2.0]:
                x_try = max(0.0, old_aval * frac)
                a_try = a.copy()
                a_try[i] = x_try
                a_try = normalize_simplex(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_local:
                    best_obj_local = obj_try
                    best_x = x_try
            
            if best_obj_local < best_obj - 1e-14:
                a[i] = best_x
                a = normalize_simplex(a)
                best_obj = best_obj_local
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Try random coordinate restart
            rng = np.random.default_rng(it * 104729 + 7)
            i = int(rng.integers(0, n))
            a[i] *= (0.5 + rng.random())
            a = normalize_simplex(a)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            else:
                a = best_a.copy()

    return best_a, best_obj


def parametric_start(n, rng):
    """Generate a diverse parametric starting sequence."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    choice = rng.integers(0, 6)
    
    if choice == 0:
        # Gaussian
        sigma = n * (0.1 + 0.4 * rng.random())
        a = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
    elif choice == 1:
        # Triangular
        a = 1.0 - np.abs(i - center) / (center + 1)
    elif choice == 2:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    elif choice == 3:
        # Modulated: bell * cos
        sigma = n * 0.3
        bell = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        freq = 1 + int(rng.integers(0, 5))
        a = bell * (1 + 0.5 * np.cos(2 * np.pi * freq * i / n))
    elif choice == 4:
        # Two-hump
        s1 = n * 0.25
        s2 = n * 0.25
        a = np.exp(-((i - center) ** 2) / (2 * s1 ** 2)) + np.exp(-((i - center) ** 2) / (2 * s2 ** 2))
        a = np.exp(-((i - center) ** 2) / (2 * (n*0.15) ** 2)) + 0.5 * np.exp(-((i - center*1.5) ** 2) / (2 * (n*0.1) ** 2))
    else:
        # Smooth random (low-pass filtered)
        a = np.abs(rng.normal(0, 1, n))
        # Simple moving average smoothing
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def refine_sequence(a, time_budget=60.0):
    """
    Multi-method refinement: LP alternating + subgradient + coordinate descent.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)

    # Method 1: LP alternating
    if n <= 1000:
        alloc = min(time_budget * 0.35, 30.0)
        a1, obj1 = lp_alternate(best_a, max_iters=30, time_budget=alloc)
        if obj1 < best_obj - 1e-13:
            best_obj = obj1
            best_a = a1.copy()

    # Method 2: Subgradient top-K
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.3, 20.0)
        a2, obj2 = subgradient_topk(best_a, max_iters=int(alloc * 30), time_budget=alloc, num_peaks=3)
        if obj2 < best_obj - 1e-13:
            best_obj = obj2
            best_a = a2.copy()

    # Method 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.3, 20.0)
        a3, obj3 = coordinate_descent(best_a, max_iters=int(alloc * 50), time_budget=alloc)
        if obj3 < best_obj - 1e-13:
            best_obj = obj3
            best_a = a3.copy()

    # Method 4: Another round of LP if time allows
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15 and n <= 800:
        alloc = min(remaining * 0.4, 20.0)
        a4, obj4 = lp_alternate(best_a, max_iters=20, time_budget=alloc)
        if obj4 < best_obj - 1e-13:
            best_obj = obj4
            best_a = a4.copy()

    return best_a, best_obj


def resample_to_length(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(budget_s=1500):
    """
    Main search function combining LP alternating, subgradient, coordinate descent,
    multi-start exploration, and length sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        parent_norm = normalize_simplex(parent.copy())
        pobj = compute_obj(parent_norm)
        print(f"Parent normalized: n={len(parent_norm)}, obj={pobj:.8f}")
        if pobj < best_score:
            best_score = pobj
            best_seq = parent_norm.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)

    n_parent = len(best_seq)
    print(f"Starting best: {best_score:.8f}, n={n_parent}")

    # === Phase 1: Intensive LP alternating on parent ===
    print("\n=== Phase 1: LP alternating on parent ===")
    t1_start = time.monotonic()
    remaining = deadline - time.monotonic()
    alloc1 = min(remaining * 0.2, 180.0)
    a1, obj1 = lp_alternate(best_seq, max_iters=80, time_budget=alloc1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  Phase 1 result: {best_score:.8f}")
    else:
        print(f"  Phase 1: no improvement ({obj1:.8f})")

    # === Phase 2: Multi-start exploration ===
    print("\n=== Phase 2: Multi-start exploration ===")
    rng = np.random.default_rng(12345)
    
    n_trials = [n_parent, n_parent - 100, n_parent + 100, 500, 800, 1000, 1200]
    n_trials = sorted(set([x for x in n_trials if 300 <= x <= 1500]))
    
    candidates = []
    candidates.append((best_score, best_seq.copy()))

    for n_t in n_trials:
        if time.monotonic() > deadline - 300:
            break
        remaining2 = deadline - time.monotonic()
        if remaining2 < 60:
            break

        num_starts = 4
        alloc_per_start = min(remaining2 / (num_starts * 2), 15.0)
        if alloc_per_start < 5:
            num_starts = max(2, int(remaining2 / 30))

        for si in range(num_starts):
            if time.monotonic() > deadline - 200:
                break
            
            if si == 0 and n_t == n_parent and best_seq is not None:
                s_init = best_seq.copy()
            else:
                s_init = parametric_start(n_t, rng)

            # Quick refine
            t_alloc = min(alloc_per_start, 12.0)
            s_init, s_obj = subgradient_topk(s_init, max_iters=int(t_alloc * 15), time_budget=t_alloc * 0.5, num_peaks=3)
            
            if s_obj < best_score:
                best_score = s_obj
                best_seq = s_init.copy()
                print(f"  n={n_t}, start {si}: {s_obj:.8f} ** NEW BEST **")
            
            candidates.append((s_obj, s_init.copy()))

    candidates.sort(key=lambda x: x[0])
    print(f"  Top 3 candidates: {[f'{c[0]:.6f}' for c in candidates[:3]]}")

    # === Phase 3: Intensive refinement of top candidates ===
    print("\n=== Phase 3: Intensive refinement ===")
    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 200:
            break
        remaining3 = deadline - time.monotonic()
        if remaining3 < 60:
            break

        a_c = candidates[ci][1].copy()
        alloc3 = min(remaining3 * 0.25, 80.0)
        print(f"  Refining candidate {ci} (n={len(a_c)}, obj={candidates[ci][0]:.8f}), budget={alloc3:.0f}s")
        
        a_r, obj_r = refine_sequence(a_c, time_budget=alloc3)
        
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** UPDATE: {best_score:.8f} **")
        else:
            print(f"  No improvement ({obj_r:.8f})")

    # === Phase 4: Length exploration from best ===
    print(f"\n=== Phase 4: Length exploration (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = sorted(set([x for x in [n_best-50, n_best+50, n_best-100, n_best+100, 
                                                 n_best-200, n_best+200, 600, 800, 1000]
                                   if 300 <= x <= 1500]))

        for n_test in test_lengths:
            if time.monotonic() > deadline - 150:
                break
            remaining4 = deadline - time.monotonic()
            if remaining4 < 30:
                break

            a_res = resample_to_length(best_seq, n_test)
            a_res = normalize_simplex(a_res)
            
            alloc4 = min(remaining4 * 0.2, 30.0)
            a_pol, obj_pol = refine_sequence(a_res, time_budget=alloc4)
            
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** {best_score:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final polish (best: {best_score:.8f}) ===")
    if best_seq is not None:
        remaining5 = deadline - time.monotonic()
        if remaining5 > 30:
            alloc5 = min(remaining5 * 0.5, 120.0)
            a_fin, obj_fin = refine_sequence(best_seq, time_budget=alloc5)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Final polish: {best_score:.8f}")
            else:
                print(f"  Final polish: {obj_fin:.8f}")

        # Coordinate descent final
        remaining5 = deadline - time.monotonic()
        if remaining5 > 20 and best_seq is not None:
            a_cd, obj_cd = coordinate_descent(best_seq, max_iters=int(remaining5 * 30), time_budget=min(remaining5*0.3, 30.0))
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Coord descent: {best_score:.8f}")

        # Final perturbations
        remaining5 = deadline - time.monotonic()
        if remaining5 > 15 and best_seq is not None:
            n_fin = len(best_seq)
            rng2 = np.random.default_rng(99999)
            for pi in range(8):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.01 + 0.005 * pi
                a_p = best_seq * np.exp(rng2.normal(0, scale, n_fin))
                a_p = normalize_simplex(a_p)
                t_alloc = min(remaining5 * 0.1, 10.0)
                if t_alloc < 3:
                    break
                a_p, obj_p = subgradient_topk(a_p, max_iters=int(t_alloc * 20), time_budget=t_alloc, num_peaks=2)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Perturb {pi}: ** {best_score:.8f} **")

    # === Phase 6: One final LP if time and size allow ===
    if best_seq is not None and len(best_seq) <= 900 and time.monotonic() < deadline - 20:
        remaining6 = deadline - time.monotonic()
        print(f"\n=== Phase 6: Final LP (n={len(best_seq)}, {remaining6:.0f}s left) ===")
        a_lp, obj_lp = lp_alternate(best_seq, max_iters=30, time_budget=min(remaining6 * 0.7, 50.0))
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP final: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
