   def compute_conv(a):
       """Compute linear convolution a*a using FFT."""
       n = len(a)
       if n == 0:
           return np.array([0.0])
       size = 1
       while size < 2 * n - 1:
           size <<= 1
       fa = rfft(a, size)
       conv = irfft(fa * fa, size)
       return conv[:2 * n - 1]
   
   def compute_obj(a):
       """Compute objective."""
       n = len(a)
       if n == 0:
           return np.inf
       conv = compute_conv(a)
       max_b = float(np.max(conv))
       sum_a = float(np.sum(a))
       if sum_a < 0.01:
           return np.inf
       return 2.0 * n * max_b / (sum_a * sum_a)
   
   def project_nonneg_sum1(a):
       """Project onto {a >= 0, sum(a) = 1}."""
       a = np.maximum(a, 0)
       s = np.sum(a)
       if s < 1e-15:
           return np.ones(len(a)) / len(a)
       return a / s

   def multi_peak_descent(a, time_budget=60, max_iters=500):
       """
       Multi-peak gradient descent with Armijo line search.
       Targets the top K peaks simultaneously.
       """
       n = len(a)
       a = project_nonneg_sum1(a.copy().astype(np.float64))
       best_a = a.copy()
       best_obj = compute_obj(a)
       start = time.monotonic()
       
       K = 3  # number of peaks to target
       base_step = 0.05
       
       for it in range(max_iters):
           if time.monotonic() - start > time_budget:
               break
           
           conv = compute_conv(a)
           cmax = float(np.max(conv))
           
           # Find top K peaks
           peak_indices = np.argpartition(conv, -K)[-K:]
           peak_indices = peak_indices[np.argsort(conv[peak_indices][::-1])]
           
           # Weighted combination of gradients
           weights = np.zeros(2*n-1)
           for k in peak_indices:
               # Weight proportional to how much this peak exceeds the median
               weights[k] = max(0, conv[k] - np.median(conv))
           if weights.sum() < 1e-15:
               weights[peak_indices[0]] = 1.0
           weights /= weights.sum()
           
           # Compute gradient: grad[i] = sum_k weights[k] * 2*a[k-i]
           grad = np.zeros(n)
           for k in peak_indices:
               w = weights[k]
               if w < 1e-12:
                   continue
               # grad[i] += 2*w*a[k-i] for valid i
               i_start = max(0, k - n + 1)
               i_end = min(n, k + 1)
               if i_end > i_start:
                   indices = np.arange(i_start, i_end)
                   a_idx = k - indices
                   grad[indices] += 2.0 * w * a[a_idx]
           
           # Center the gradient (to preserve sum)
           grad -= np.mean(grad)
           
           # Armijo line search
           best_step = base_step
           best_obj_try = np.inf
           for step_scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
               step = base_step * step_scale
               a_try = project_nonneg_sum1(a - step * grad)
               obj_try = compute_obj(a_try)
               if obj_try < best_obj_try:
                   best_obj_try = obj_try
                   best_step = step
           
           if best_obj_try < best_obj - 1e-12:
               best_obj = best_obj_try
               best_a = a_try.copy()
               a = a_try.copy()
               base_step = min(base_step * 1.02, 0.1)
           else:
               base_step *= 0.7
               if base_step < 1e-4:
                   break
       
       return best_a, best_obj

       Iteratively reweighted QP for peak reduction.
       At each step, solve a QP that minimizes the sum of the top-K convolution values.
       """
       n = len(a)

       best_a = a.copy()
       best_obj = compute_obj(a)

               break

           top_k = np.argpartition(conv, -K)[-K:]

"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_nonneg_sum1(a_new)


def build_toeplitz_conv(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]. Shape: (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def fista_peak_qp(a_init, top_k_indices, max_iters=300, time_budget=30.0):
    """
    Solve min a^T B a s.t. sum(a)=1, a>=0 using FISTA.
    B[i,j] = 1 if i+j in top_k_indices, else 0.
    """
    n = len(a_init)
    x = a_init.copy()
    y = a_init.copy()
    t = 1.0
    K = len(top_k_indices)
    L = 2.0 * K + 1  # Lipschitz constant upper bound
    step = 1.0 / L
    start = time.monotonic()
    top_k_set = set(top_k_indices.tolist()) if hasattr(top_k_indices, 'tolist') else set(top_k_indices)

    for k in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute B @ y: (B@y)[i] = sum_j B[i,j]*y[j] = sum_j y[j] if i+j in top_k
        # Equivalently: (B@y)[i] = sum_{j: i+j in top_k} y[j]
        # For each i, valid j are those with i+j in top_k, i.e., j in {k-i for k in top_k}
        grad = np.zeros(n)
        for ki in top_k_indices:
            # grad[i] += y[ki - i] for valid i
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_end > i_start:
                grad[i_start:i_end] += y[ki - np.arange(i_start, i_end)]

        # Gradient of a^T B a is 2*B@a (B is symmetric)
        grad = 2.0 * grad

        # FISTA update
        x_new = project_nonneg_sum1(y - step * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        x = x_new
        t = t_new

    return x

def spectral_shaping(a, max_iters=200, time_budget=30.0, alpha_init=0.5):
    """
    Spectral shaping: iteratively flatten the power spectrum to reduce convolution peaks.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        pmax = np.max(power)
        if pmax < 1e-15:
            break

        # Gradually reduce compression
        alpha = alpha_init + (1.0 - alpha_init) * (it / max_iters)
        power_norm = power / pmax
        power_compressed = np.maximum(power_norm, 1e-12) ** alpha

        # Reconstruct
        phase = np.angle(fa)
        magnitude = np.sqrt(power_compressed) * np.sqrt(pmax)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def chase_peak_adaptive(a, max_iters=200, time_budget=30.0):
    """
    Adaptive peak chasing: target the single maximum peak with adaptive step size.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.02

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Gradient: grad[i] = 2*a[k_star - i] for valid i
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        if i_end > i_start:
            grad[i_start:i_end] = 2.0 * a[k_star - np.arange(i_start, i_end)]

        grad -= np.mean(grad)

        # Adaptive line search
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.125]:
            s = step * scale
            a_try = project_nonneg_sum1(a - s * grad)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                step = min(step * 1.01, 0.05)
                improved = True

            if step < 1e-5:

    return best_a, best_obj


def dual_lp_direction(a, time_budget=20.0):
    """
    Find direction d that minimizes max_k (a*d)[k] subject to sum(d)=0, |d|<=1.
    This gives the most "peak-reducing" direction.
    """
    n = len(a)
    if n > 2000:
        return None

    start = time.monotonic()
    conv = compute_conv(a)
    M = float(np.max(conv))

    # We want to find d that minimizes t such that:
    # (a*d)[k] <= t for all k
    # sum(d) = 0
    # -1 <= d[i] <= 1
    #
    # LP: min t
    # s.t. A_d @ d <= t * 1  (where A_d is the Toeplitz of a)
    #      sum(d) = 0
    #      -1 <= d <= 1
    
    A = build_toeplitz_conv(a)
    
    # Variables: d[0..n-1], t
    # (A @ d)[k] - t <= 0 for all k
    # sum(d) = 0
    # -1 <= d[i] <= 1, t >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # (A@d)[k] - t <= 0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-1.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': int(time_budget)})
        if res.status == 0 and res.x is not None:
            d = res.x[:n]
            if np.linalg.norm(d) > 1e-10:
                return d
    except Exception:
        pass
    return None


def dual_lp_step(a, max_iters=30, time_budget=40.0):
    """Use dual LP to find peak-reducing direction and apply it."""
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        d = dual_lp_direction(a, time_budget=min(10.0, time_budget - (time.monotonic() - start)))
        if d is None:
            break

        # Scale d to be a valid direction (maintain non-negativity)
        # a + alpha * d >= 0 => alpha <= -a[i]/d[i] for d[i] < 0
        neg_mask = d < 0
        if np.any(neg_mask):
            alpha_max = np.min(-a[neg_mask] / d[neg_mask]) * 0.99
        else:
            alpha_max = 1.0
        alpha_max = min(alpha_max, 0.5)

        if alpha_max < 1e-6:
            break

        # Try a few alpha values
        improved = False
        for alpha_frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            alpha = alpha_max * alpha_frac
            a_try = project_nonneg_sum1(a + alpha * d)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_obj


def parametric_search(n, rng, time_budget=30.0):
    """
    Search over parametric families for good initial sequences.
    """
    best_a = None
    best_obj = np.inf
    start = time.monotonic()

    x = np.linspace(0, 1, n)

    # Various parametric forms
    candidates = []

    # Gaussian with varying width
    for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(g)

    # Sine power
    for p in [1.0, 1.5, 2.0, 3.0, 4.0]:
        s = np.sin(np.pi * x) ** p
        candidates.append(s)

    # Cosine modulated
    for freq in [1, 2, 3, 4]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            c = np.ones(n) * (1 - amp) + amp * np.cos(2 * np.pi * freq * x)
            c = np.maximum(c, 0.01)
            candidates.append(c)

    # Two-sided decay
    for rate in [0.5, 1.0, 2.0, 3.0, 5.0]:
        center = n // 2
        dist = np.abs(np.arange(n) - center)
        d = np.exp(-rate * dist / n)
        candidates.append(d)

    # Random
    for _ in range(5):
        r = rng.random(n) + 0.1
        candidates.append(r)

    # Evaluate all
    for c in candidates:
        if time.monotonic() - start > time_budget:
            break
        a_c = project_nonneg_sum1(c)
        # Quick refinement
        a_c, obj_c = chase_peak_adaptive(a_c, max_iters=30, time_budget=3.0)
        if obj_c < best_obj:
            best_obj = obj_c
            best_a = a_c.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple strategies:
    1. Load parent and quick screening
    2. Multi-peak FISTA optimization
    3. LP alternating
    4. Spectral shaping
    5. Multi-length exploration
    6. Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()
    else:
        # Default initialization
        n_default = 700
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n={n_parent}")

    # === Phase 1: Multi-peak FISTA (25% time) ===
    print("\n=== Phase 1: Multi-peak FISTA ===")
    phase1_time = min(300.0, (deadline - start_time) * 0.25)
    a1, obj1 = multi_peak_fista(best_seq, max_iters=80, time_budget=phase1_time, K=3, inner_iters=80)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  FISTA K=3: {best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        a1b, obj1b = multi_peak_fista(best_seq, max_iters=60,
                                      time_budget=min(60.0, remaining * 0.3), K=5, inner_iters=60)
        if obj1b < best_score:
            best_score = obj1b
            best_seq = a1b.copy()
            print(f"  FISTA K=5: {best_score:.8f}")

    # === Phase 2: LP Alternating (25% time) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    print(f"Best: {best_score:.8f}")
    phase2_time = min(350.0, (deadline - time.monotonic()) * 0.25)
    a2, obj2 = lp_alternate(best_seq, max_iters=30, time_budget=phase2_time)
    if obj2 < best_score:
        best_score = obj2
        best_seq = a2.copy()
        print(f"  LP: {best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Dual LP
        a2b, obj2b = dual_lp_step(best_seq, max_iters=20, time_budget=min(100.0, remaining * 0.5))
        if obj2b < best_score:
            best_score = obj2b
            best_seq = a2b.copy()
            print(f"  Dual LP: {best_score:.8f}")

    # === Phase 3: Spectral shaping + refinement (20% time) ===
    print(f"\n=== Phase 3: Spectral Shaping ===")
    print(f"Best: {best_score:.8f}")
    phase3_time = min(200.0, (deadline - time.monotonic()) * 0.20)
    a3, obj3 = spectral_shaping(best_seq, max_iters=200, time_budget=min(60.0, phase3_time * 0.5),
                                alpha_init=0.4)
    if obj3 < best_score:
        best_score = obj3
        best_seq = a3.copy()
        print(f"  Spectral: {best_score:.8f}")

    # Chase peak after spectral
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a3b, obj3b = chase_peak_adaptive(best_seq, max_iters=100,
                                         time_budget=min(40.0, remaining * 0.4))
        if obj3b < best_score:
            best_score = obj3b
            best_seq = a3b.copy()
            print(f"  Chase: {best_score:.8f}")

    # === Phase 4: Multi-length exploration (20% time) ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    print(f"Best: {best_score:.8f}")
    phase4_time = min(250.0, (deadline - time.monotonic()) * 0.20)
    phase4_start = time.monotonic()

    n_current = len(best_seq)
    n_tries = []
    for delta in [-30, -20, -10, 10, 20, 30, 50, -50]:
        n_try = n_current + delta
        if 50 <= n_try <= 2000 and n_try not in n_tries:
            n_tries.append(n_try)

    per_n_time = max(10.0, phase4_time / len(n_tries))

    for n_try in n_tries:
        if time.monotonic() > phase4_start + phase4_time:
            break

        a_resampled = resample_sequence(best_seq, n_try)

        # Quick optimization chain
        a_opt, obj_opt = chase_peak_adaptive(a_resampled, max_iters=50, time_budget=per_n_time * 0.3)
        if n_try <= 1500:
            a_opt2, obj_opt2 = multi_peak_fista(a_opt, max_iters=20,
                                                time_budget=per_n_time * 0.4, K=3, inner_iters=40)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2
        a_opt3, obj_opt3 = chase_peak_adaptive(a_opt, max_iters=30, time_budget=per_n_time * 0.2)
        if obj_opt3 < obj_opt:
            a_opt, obj_opt = a_opt3, obj_opt3

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    print(f"Time remaining: {remaining5:.1f}s")

    if remaining5 > 20 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)
        polish_start = time.monotonic()

        # Round 1: Multi-peak FISTA
        if n_final <= 1500:
            a_p, obj_p = multi_peak_fista(a_final, max_iters=50,
                                          time_budget=min(60.0, remaining5 * 0.3), K=3, inner_iters=80)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  Polish FISTA: {best_score:.8f}")

        # Round 2: LP
        remain = deadline - time.monotonic()
        if remain > 30 and n_final <= 1500:
            a_p2, obj_p2 = lp_alternate(a_final, max_iters=20, time_budget=min(60.0, remain * 0.5))
            if obj_p2 < best_score:
                best_score = obj_p2
                best_seq = a_p2.copy()
                a_final = a_p2.copy()
                print(f"  Polish LP: {best_score:.8f}")

        # Round 3: Chase peak with very small steps
        remain = deadline - time.monotonic()
        if remain > 15:
            a_p3, obj_p3 = chase_peak_adaptive(a_final, max_iters=200, time_budget=min(40.0, remain * 0.6))
            if obj_p3 < best_score:
                best_score = obj_p3
                best_seq = a_p3.copy()

        # Round 4: Small perturbation + re-optimize
        remain = deadline - time.monotonic()
        if remain > 30:
            for perturb_round in range(3):
                remain = deadline - time.monotonic()
                if remain < 15:
                    break
                a_pert = a_final * np.exp(rng.normal(0, 0.01, n_final))
                a_pert = project_nonneg_sum1(a_pert)
                a_pert, obj_pert = chase_peak_adaptive(a_pert, max_iters=50,
                                                       time_budget=min(20.0, remain * 0.4))
                if n_final <= 1200:
                    a_pert2, obj_pert2 = multi_peak_fista(a_pert, max_iters=15,
                                                          time_budget=min(15.0, remain * 0.3), K=3, inner_iters=50)
                    if obj_pert2 < obj_pert:
                        a_pert, obj_pert = a_pert2, obj_pert2
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_final = a_pert.copy()
                    print(f"  Perturb {perturb_round}: {best_score:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0
        print("WARNING: evaluate returned inf, using fallback")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
