def compute_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

                step_size = s_norm / s_y
                step_size = np.clip(step_size, 0.01, 1.0)
        
        a_prev = a.copy()
        g_prev = grad.copy()
        
        # Take step
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
                # Reduce step size on stall
                step_size *= 0.9
                # Don't update a if it got worse
                a_new = a
            
            a = a_new
            
            # Averaging (Polyak)
            count += 1
            avg_a = (count * avg_a + a) / (count + 1)
            
            # Periodically check the average
            if count % 50 == 0:
                avg_obj = compute_obj(avg_a)
                if avg_obj < best_obj:
                    best_obj = avg_obj
                    best_a = avg_a.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      SGD it{it}: obj={best_obj:.8f}, step={step_size:.4f}, t={elapsed:.1f}s")
    
    return best_a


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix efficiently
    # A[k, j] = a_norm[j] if k >= j, a_norm[k-j] if k < j ... 
    # Actually (a*g)[k] = sum_j a_norm[j] * g[k-j] for valid indices
    # So A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    # But that's wrong. (a*g)[k] = sum_i a[i] * g[k-i]
    # So for fixed k, (a*g)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * g[k-i]
    # Let j = k-i, so i = k-j:
    # (a*g)[k] = sum_{j: 0<=k-j<n, 0<=j<n} a[k-j] * g[j]
    # = sum_j A[k,j] * g[j] where A[k,j] = a[k-j] if 0<=k-j<n and 0<=j<n
    
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a_norm[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search
        best_t_obj = best_obj
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and it % 10 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(it * 7 + n)
                scale = 0.03
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent: adjust pairs to reduce peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Find the top contributors to the peak
        # grad[j] = 2 * a[k_star - j] for valid j
        j_min = max(0, k_star - (n - 1))
        j_max = min(n - 1, k_star)
        
        # Try to move mass from high-contributing positions to low-contributing ones
        candidates = []
        for j in range(j_min, j_max + 1):
            contrib = a[j] * a[k_star - j]
            candidates.append((contrib, j, k_star - j))
        
        candidates.sort(reverse=True)
        
        for contrib, j, idx in candidates[:20]:
            if time.monotonic() - start > time_budget:
                break
            if j == idx:
                continue
            if a[j] <= 1e-6:
                continue
            
            # Try reducing a[j] and increasing a[idx] (or another position)
            for frac in [0.01, 0.05, 0.1, 0.2]:
                delta = frac * a[j]
                a[j] -= delta
                a[idx] += delta
                if a[j] < 0:
                    a[j] += delta
                    a[idx] -= delta
                    continue
                s_new = float(np.sum(a))
                a_try = a / s_new
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
                else:
                    a[j] += delta
                    a[idx] -= delta
                    break

        if not improved:
            # Try random pairs
            rng = np.random.default_rng(it)
            for _ in range(10):
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 == i2:
                    continue
                for frac in [0.05, 0.1]:
                    if a[i1] < 1e-6:
                        continue
                    delta = frac * a[i1]
                    a[i1] -= delta
                    a[i2] += delta
                    s_new = float(np.sum(a))
                    a_try = a / s_new
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                    else:
                        a[i1] += delta
                        a[i2] -= delta
                        break
            if not improved:
                break

    return best_a


def multi_peak_subgradient(a, time_budget=60.0, verbose=False):
    """
    Subgradient method that considers multiple peaks (not just the max).
    Uses a weighted combination of subgradients from top-k peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.1
    avg_a = a.copy()
    count = 0
    stall = 0

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Consider top-k peaks
        k = min(5, len(conv))
        top_k = np.argsort(conv)[-k:][::-1]
        
        # Weight by how close each peak is to the max
        weights = np.exp(-(conv[top_k] - max_val) * n)  # Soft attention
        weights = weights / weights.sum()
        
        # Combined subgradient
        combined_grad = np.zeros(n)
        for w, k_idx in zip(weights, top_k):
            combined_grad += w * subgradient(a, int(k_idx))
        
        grad_centered = combined_grad - np.mean(combined_grad)
        grad_norm = np.linalg.norm(grad_centered) + 1e-12
        
        # Polyak-type step: step = (f(a) - f_lower) / ||grad||
        # Lower bound: 2n * (sum a)^2 / (2n-1) / n = 2/(2n-1) ... use 1.0 as lower bound
        f_lower = 1.0
        polyak_step = (best_obj / (2.0 * n) - f_lower / (2.0 * n)) / grad_norm
        step_size = np.clip(step_size, 0.001, 1.0)
        
        a_new = a - step_size * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
                step_size = min(step_size * 1.05, 1.0)
            else:
                stall += 1
                step_size *= 0.95
                a_new = a
            
            a = a_new
            count += 1
            avg_a = (count * avg_a + a) / (count + 1)
            
            if count % 100 == 0:
                avg_obj = compute_obj(avg_a)
                if avg_obj < best_obj:
                    best_obj = avg_obj
                    best_a = avg_a.copy()
        
        if stall > 50:
            # Perturb
            rng = np.random.default_rng(it)
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            best_a = a.copy()
            best_obj = compute_obj(a)
            avg_a = a.copy()
            count = 0
            stall = 0
            step_size = 0.1

        if verbose and it % 200 == 0:
            print(f"      MPSG it{it}: obj={best_obj:.8f}, step={step_size:.4f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Hybrid optimization: subgradient descent + LP alternating + coordinate descent.
    Exploits non-symmetric structure of optimal sequences.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Subgradient descent (main workhorse) ===
    print(f"\n=== Phase 1: Subgradient Descent (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc1 = min(remaining * 0.3, 300.0)
    
    a_sgd = subgradient_descent(best_seq.copy(), time_budget=alloc1, verbose=True)
    obj_sgd = compute_obj(a_sgd)
    if obj_sgd < best_score:
        best_score = obj_sgd
        best_seq = a_sgd.copy()
        print(f"  SGD result: obj={best_score:.8f}")
    else:
        print(f"  SGD: obj={obj_sgd:.8f} (no improvement)")

    # === Phase 2: Multi-peak subgradient ===
    print(f"\n=== Phase 2: Multi-Peak Subgradient ===")
    remaining = deadline - time.monotonic()
    alloc2 = min(remaining * 0.2, 200.0)
    
    a_mpsg = multi_peak_subgradient(best_seq.copy(), time_budget=alloc2, verbose=True)
    obj_mpsg = compute_obj(a_mpsg)
    if obj_mpsg < best_score:
        best_score = obj_mpsg
        best_seq = a_mpsg.copy()
        print(f"  MPSG result: obj={best_score:.8f}")
    else:
        print(f"  MPSG: obj={obj_mpsg:.8f} (no improvement)")

    # === Phase 3: LP Alternating ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    alloc3 = min(remaining * 0.25, 200.0)
    
    n_lp = len(best_seq)
    # Use a smaller n for LP if n is too large (LP is slow for large n)
    if n_lp > 600:
        # Resample to 600 for LP
        n_lp_use = 600
        x_old = np.linspace(0, 1, n_lp)
        x_new = np.linspace(0, 1, n_lp_use)
        a_lp = np.interp(x_new, x_old, best_seq)
        a_lp = np.maximum(a_lp, 0)
        a_lp = a_lp / np.sum(a_lp)
    else:
        n_lp_use = n_lp
        a_lp = best_seq.copy()
    
    a_lp_opt = lp_alternating(a_lp, time_budget=alloc3, verbose=True)
    obj_lp = compute_obj(a_lp_opt)
    
    # Resample back to original n
    if n_lp_use != n_lp:
        x_new = np.linspace(0, 1, n_lp)
        a_lp_opt = np.interp(x_new, np.linspace(0, 1, n_lp_use), a_lp_opt)
        a_lp_opt = np.maximum(a_lp_opt, 0)
        a_lp_opt = a_lp_opt / np.sum(a_lp_opt)
        obj_lp = compute_obj(a_lp_opt)
    
    if obj_lp < best_score:
        best_score = obj_lp
        best_seq = a_lp_opt.copy()
        n_best = len(best_seq)
        print(f"  LP result: obj={best_score:.8f}")
    else:
        print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    alloc4 = min(remaining * 0.15, 100.0)
    
    a_cd = coordinate_descent(best_seq.copy(), time_budget=alloc4)
    obj_cd = compute_obj(a_cd)
    if obj_cd < best_score:
        best_score = obj_cd
        best_seq = a_cd.copy()
        print(f"  CD result: obj={best_score:.8f}")
    else:
        print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 5: n-sweep with resampling ===
    print(f"\n=== Phase 5: n-Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    n_options = []
    for delta in [-100, -50, -25, 0, 25, 50, 100, 200]:
        n_try = n_best + delta

            n_options.append(n_try)
    # Also try some specific values
    for n_val in [400, 500, 600, 800, 1000, 1200, 1500]:

            n_options.append(n_val)
    n_options = sorted(set(n_options))
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_best:
            continue
        
        alloc = min(remaining * 0.1, 40.0)
        if alloc < 10:
            break
        
        # Resample
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_seq)
        a_try = np.maximum(a_try, 0)
        a_try = a_try / np.sum(a_try)
        
        # Quick SGD
        a_opt = subgradient_descent(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = len(best_seq)

        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    print(f"\n=== Phase 6: Final Polish ===")

        alloc = min(remaining * 0.3, 100.0)
        a_final = subgradient_descent(best_seq.copy(), time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final SGD: obj={best_score:.8f}")
        
        # CD
        remaining = deadline - time.monotonic()

        if remaining > 15:

            obj_final = compute_obj(a_final)
