"""
Subgradient + LP hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based convolution, Barzilai-Borwein step size, softmax-active-lag weighting,
momentum, periodic LP escape, multi-n exploration, and diverse random restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def normalize(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full(len(a), 1.0 / len(a))
    return a / s


def subgradient_step(a, alpha, momentum_vec, beta, num_active=50):
    """
    One subgradient step with softmax-weighted active lags.
    Returns (new_a, new_momentum, s_g, s_y) for BB step size.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weighting of lags near the max
    k_star = int(np.argmax(conv))
    max_val = conv[k_star]
    
    # Get top num_active lags
    if num_active >= len(conv):
        weights = np.exp(-conv / max_val)  # normalized by max
        weights /= np.sum(weights)
        active_indices = np.arange(len(conv))
    else:
        # Get indices of top lags
        top_k = np.argsort(conv)[-num_active:]
        top_k_vals = conv[top_k]
        # Softmax weights among top k
        w = np.exp(-(top_k_vals - top_k_vals.max()) / (top_k_vals.max() * 0.1 + 1e-10))
        w /= np.sum(w)
        active_indices = top_k
    
    # Compute weighted subgradient
    sg = np.zeros(n)
    for k_idx in range(len(active_indices)):
        k = active_indices[k_idx]
        w = weights[k_idx] if num_active >= len(conv) else w[k_idx]
        # Subgradient at lag k: sg[i] += w * a[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            sg[i] += w * a[k - i]
    
    # Normalize subgradient
    sg_norm = np.linalg.norm(sg)
    if sg_norm > 1e-15:
        sg = sg / sg_norm
    
    # Momentum update
    new_momentum = beta * momentum_vec + (1 - beta) * sg
    
    # Step
    a_new = a - alpha * new_momentum
    a_new = normalize(a_new)
    
    return a_new, new_momentum, sg


def subgradient_optimize(a, time_budget=60.0, verbose=False):
    """
    Subgradient optimization with Barzilai-Borwein step size and momentum.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Initialize
    momentum = np.zeros(n)
    beta = 0.9  # momentum parameter
    alpha = 0.01  # initial step size
    num_active = min(100, n)  # number of active lags
    
    # BB state
    s_prev = None
    y_prev = None
    obj_prev = best_obj
    a_prev = a.copy()
    
    stall = 0
    max_stall = 30
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient step
        a_new, momentum, sg = subgradient_step(a, alpha, momentum, beta, num_active)
        obj_new = compute_obj(a_new)
        
        # Barzilai-Borwein step size update
        if s_prev is not None and y_prev is not None:
            sy = float(np.dot(s_prev, y_prev))
            ss = float(np.dot(s_prev, s_prev))
            yy = float(np.dot(y_prev, y_prev))
            if ss > 1e-15 and abs(sy) > 1e-15:
                if sy > 0:
                    alpha_bb = ss / sy
                else:
                    alpha_bb = ss / yy if yy > 1e-15 else alpha
                alpha = np.clip(alpha_bb, 1e-4, 0.1)
        
        s_prev = a_new - a
        y_prev = sg  # approximate y as the subgradient
        a = a_new
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Reset with perturbation
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.05, n))
                a = normalize(a)
                momentum = np.zeros(n)
                alpha = 0.01
                stall = 0
                max_stall = min(max_stall + 10, 60)
        
        if verbose and it % 100 == 0:
            print(f"          subgrad it{it}: obj={best_obj:.8f}, alpha={alpha:.6f}")
    
    return best_a
