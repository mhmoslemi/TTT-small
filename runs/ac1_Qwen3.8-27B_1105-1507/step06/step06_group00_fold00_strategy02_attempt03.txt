def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients and returns the objective value.
    The objective is: 2 * n * max(conv(a, a)) / (sum(a))**2.
    Returns np.inf if invalid (e.g., negative values, NaN, or too small sum)."""
    if not isinstance(sequence, list):

            return np.inf

            return np.inf

            return np.inf
