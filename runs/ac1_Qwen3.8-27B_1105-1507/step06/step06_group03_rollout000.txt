def spectral_deflation(a, k_star, strength, n_fft):
    """Reduce the autocorrelation peak at lag k_star by modifying the spectrum."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    
    A = np.fft.rfft(a, size)
    power = np.abs(A)**2
    
    # Compute the contribution of each frequency bin to position k_star
    # IFFT(|A|^2)[k] = (1/size) * sum_m |A[m]|^2 * exp(2*pi*i*m*k/size)
    # For real signals: = (1/size) * (|A[0]|^2 + 2*sum_{m=1}^{size/2-1} |A[m]|^2*cos(2*pi*m*k/size) + |A[size/2]|^2*(-1)^k)
    
    m = np.arange(size // 2 + 1)
    cos_vals = np.cos(2 * np.pi * m * k_star / size)
    
    # Reduce power at bins that are "in phase" with position k_star
    mask = cos_vals > 0.5  # Only reduce strongly contributing bins
    power[mask] -= strength * power[mask] * cos_vals[mask]
    power = np.maximum(power, 0)
    
    # Reconstruct with modified power (keep phase)
    A_new = A * np.sqrt(np.maximum(power, 0)) / (np.abs(A) + 1e-30)
    a_new = np.fft.irfft(A_new, size)[:n]
    
    return np.maximum(a_new, 0)

import numpy as np

from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    for k in top_k:
        cos_vals = np.cos(2 * np.pi * m * k / size)
        # Reduce power where cos is positive (constructive interference at lag k)
        reduce_amount = strength * np.maximum(cos_vals, 0)
        power = power - reduce_amount * power
        power = np.maximum(power, 1e-30)
    
    # Reconstruct
    A_new = np.sqrt(power) * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, size)[:n]
    
    return np.maximum(a_new, 0)
