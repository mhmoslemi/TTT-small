import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J_normalized(p):
    """Compute J = 2n * max(p*p) for normalized p (sum=1)."""
    conv = fast_conv(p)
    return 2.0 * len(p) * conv.max()

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def active_set_lp_step(p, n_active=50, time_limit=5):
    """
    Find a direction q that maximizes sum(q) subject to:
    (p*q)[k] <= (p*p)[k] for the top n_active peaks
    q >= 0
    """
    n = len(p)
    conv = fast_conv(p)
    
    # Get active set: top n_active peaks
    k_active = np.argsort(conv)[::-1][:n_active]
    M_vals = conv[k_active]
    
    # Build LP: max sum(q) s.t. (p*q)[k] <= M_vals[j] for j, q >= 0
    rows = []
    cols = []
    vals = []
    b_ub = []
    
    for idx, k in enumerate(k_active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(p[k - j])
        b_ub.append(M_vals[idx])
    
    A = coo_matrix((vals, (rows, cols)), shape=(n_active, n)).tocsr()
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            q = np.maximum(r.x, 0)
            if q.sum() > 1e-10:
                return q / q.sum()
    except Exception:
        pass
    return None

def subgradient_descent(p, max_iters, deadline, n_active=50, lr=0.1, verbose=False):
    """
    Subgradient descent with LP-based direction finding.
    """
    n = len(p)
    p = np.maximum(p.astype(np.float64), 0)
    p = p / p.sum()
    best_p = p.copy()
    best_J = compute_J_normalized(p)
    
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # LP direction
        q = active_set_lp_step(p, n_active=n_active, time_limit=3)
        if q is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        # Line search
        found = False
        for frac in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            p_try = (1 - frac) * p + frac * q
            p_try = np.maximum(p_try, 0)
            if p_try.sum() < 1e-10:
                continue
            p_try = p_try / p_try.sum()
            J_try = compute_J_normalized(p_try)
            if J_try < best_J:
                p = p_try
                best_J = J_try
                best_p = p_try.copy()
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_p, best_J

def entropy_mirror_descent(p, max_iters, deadline, lr=0.1, verbose=False):
    """
    Mirror descent with entropy kernel. Update: p_new = p * exp(-lr * grad) / sum(p * exp(-lr * grad))
    where grad[i] is the subgradient of max_k (p*p)[k] w.r.t. p[i].
    """
    n = len(p)
    p = np.maximum(p.astype(np.float64), 1e-15)
    p = p / p.sum()
    best_p = p.copy()
    best_J = compute_J_normalized(p)
    
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(p)
        M = conv.max()
        
        # Subgradient: for each i, grad[i] = sum_{k: conv[k]=M} (p[k-i] + p[i-k])
        # Use top 10% as "active"
        threshold = M * 0.95
        active_mask = conv >= threshold
        active_k = np.where(active_mask)[0]
        
        # Compute gradient efficiently using FFT
        # grad[i] = sum_{k in active} (p[k-i] + p[i-k])
        # = sum_k active[k] * p[k-i] + sum_k active[k] * p[i-k]
        # First term: (active * p_rev)[n-1+i]
        # Second term: (active_rev * p_rev)[3n-3-i]
        
        p_rev = p[::-1]
        active = active_mask.astype(np.float64)
        active_rev = active[::-1]
        
        size = 1
        while size < 3 * n:
            size <<= 1
        
        P = np.fft.rfft(p_rev, size)
        A = np.fft.rfft(active, size)
        Ar = np.fft.rfft(active_rev, size)
        
        c1 = np.fft.irfft(A * P, size)
        c2 = np.fft.irfft(Ar * P, size)
        
        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < size:
                grad[i] += c1[idx1]
            if 0 <= idx2 < size:
                grad[i] += c2[idx2]
        
        # Mirror descent update
        lr_eff = lr / (1 + 0.1 * it ** 0.3)
        log_weights = -lr_eff * grad
        log_weights -= log_weights.max()  # numerical stability
        weights = np.exp(log_weights)
        p_new = p * weights
        p_new_sum = p_new.sum()
        if p_new_sum < 1e-10:
            p_new = p.copy()
        else:
            p_new = p_new / p_new_sum
        
        J_new = compute_J_normalized(p_new)
        if J_new < best_J:
            best_J = J_new
            best_p = p_new.copy()
            p = p_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Perturb and restart
                p = best_p + np.random.default_rng(it).normal(0, 0.01, n)
                p = project_simplex(p)
                no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    EM it {it}: J={best_J:.8f}", flush=True)
    
    return best_p, best_J

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
