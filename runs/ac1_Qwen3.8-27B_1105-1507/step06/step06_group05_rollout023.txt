def compute_conv(a):

        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)

def compute_obj(a):

    s = float(np.sum(a))

        return np.inf

    s = float(np.sum(a))

    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lse_gradient_optimize(a, tau_start=5.0, tau_end=300.0, max_iter=100, time_budget=60.0):

    a_padded = np.zeros(size)

    lr = 0.05

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        # Anneal temperature
        frac = it / max(max_iter - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights
        shifted = tau * (conv - cmax)
        np.clip(shifted, -500, 500, out=shifted)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(2 * n - 1)
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # cross_corr[j] = sum_k W[k] * a[k-j]

        W_padded[:2 * n - 1] = weights
        a_padded[:] = 0
        a_padded[:n] = a

        Af = rfft(a_padded, size)
        corr = irfft(Wf * np.conj(Af), size)
        grad = 2.0 * corr[:n]

        # Center gradient to maintain sum constraint approximately
        grad_centered = grad - np.mean(grad)

        # Backtracking line search with projected gradient
        obj_cur = best_obj
        lr_try = lr
        found = False
        for _ in range(15):
            a_try = a - lr_try * grad_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))

                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

                found = True
                break
            lr_try *= 0.5

        if found:
            lr = min(lr_try * 1.1, 0.5)
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
        else:
            # Try mirror descent step as alternative

                best_a = a_md.copy()
                a = a_md
            else:

def u_shaped_init(n, p=2.0, sharpness=1.0):

def two_block_init(n, frac=1.0 / 3.0, smooth=0.3):

        else:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

            a = normalize(best_seq.copy())
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.12, 50.0)
            a_lse = lse_gradient_optimize(a, tau_start=5.0, tau_end=500.0,
                                          max_iter=80, time_budget=alloc)
            obj = compute_obj(a_lse)
            if obj < best_score:
                best_score = obj
                best_seq = a_lse.copy()
                print(f"  LSE: {best_score:.8f} **")
            else:
                # Blend and retry
                a_blend = normalize(0.6 * normalize(best_seq) + 0.4 * normalize(a_lse))
                a_blend = lp_alternate(a_blend, max_iters=10, time_budget=30.0)
                obj = compute_obj(a_blend)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_blend.copy()
                    print(f"  Blend+LP: {best_score:.8f} **")
                else:
                    print(f"  LSE: {obj:.8f} (no improvement)")

    # =========================================================
    # Phase 3: Multi-n exploration
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-n exploration")
    print(f"{'='*60}")

    n_best = len(best_seq) if best_seq is not None else n_base
    n_options = sorted(set([
        n_best - 150, n_best - 100, n_best - 50, n_best - 25,
        n_best, n_best + 25, n_best + 50, n_best + 100,
        n_best + 150, n_best + 200, n_best + 300
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.08, 35.0)

        if best_seq is not None and len(best_seq) != n_try:
            a_try = normalize(resample_sequence(best_seq, n_try))
        elif best_seq is not None:
            a_try = normalize(best_seq.copy())
        else:
            a_try = two_block_init(n_try, frac=1.0 / 3.0)

        # Use combined optimization
        a_try = polyanalyze_and_optimize(a_try, time_budget=alloc)
        obj = compute_obj(a_try)

        if obj < best_score:
            best_score = obj
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # =========================================================
    # Phase 4: Random restarts with diverse strategies
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_cur = len(best_seq)
        rng = np.random.default_rng(42)

        for restart in range(10):
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.08, 30.0)

            strategy = restart % 5
            if strategy == 0:
                # Gaussian noise
                a_try = best_seq * np.exp(rng.normal(0, 0.08, n_cur))
            elif strategy == 1:
                # Sinusoidal modulation
                freq = rng.uniform(0.5, 5.0)
                phase = rng.uniform(0, 2 * np.pi)
                mod = 1.0 + 0.15 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
                a_try = best_seq * mod
            elif strategy == 2:
                # Blend with new structured init
                a_struct = two_block_init(n_cur, frac=rng.uniform(0.25, 0.4))
                mix = rng.uniform(0.3, 0.6)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_struct
            elif strategy == 3:
                # Blend with U-shaped
                a_u = u_shaped_init(n_cur, p=rng.uniform(1.5, 4.0))
                mix = rng.uniform(0.2, 0.5)
                a_try = (1 - mix) * normalize(best_seq) + mix * a_u
            else:
                # Perturb specific regions (edges vs center)
                a_try = best_seq.copy()
                region = rng.choice(['left', 'right', 'center'])
                if region == 'left':
                    a_try[:n_cur // 3] *= np.exp(rng.normal(0, 0.1, n_cur // 3))
                elif region == 'right':
                    a_try[2 * n_cur // 3:] *= np.exp(rng.normal(0, 0.1, n_cur // 3))
                else:
                    a_try[n_cur // 3:2 * n_cur // 3] *= np.exp(rng.normal(0, 0.1, n_cur // 3))

            a_try = normalize(a_try)

            # Optimize
            if time.monotonic() < deadline - 30:
                a_try = lp_alternate(a_try, max_iters=10, time_budget=alloc * 0.5)
                a_try = lse_gradient_optimize(a_try, tau_start=10.0, tau_end=300.0,
                                              max_iter=30, time_budget=alloc * 0.5)

            obj = compute_obj(a_try)
            if obj < best_score:
                best_score = obj
                best_seq = a_try.copy()
                print(f"  Restart {restart} (strat {strategy}): {best_score:.8f} **")
            else:
                print(f"  Restart {restart} (strat {strategy}): {obj:.8f}")

    # =========================================================
    # Phase 5: Final intensive polish
    # =========================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        a = normalize(best_seq.copy())
        n_final = len(a)

        for round_ in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.2, 40.0)

            # LP
            if n_final <= 1000:
                a = lp_alternate(a, max_iters=20, time_budget=alloc * 0.5)

            # LSE with very sharp tau
            a = lse_gradient_optimize(a, tau_start=50.0, tau_end=1000.0,
                                      max_iter=50, time_budget=alloc * 0.35)

            # Subgradient polish
            a = subgradient_optimize(a, max_iter=30, time_budget=alloc * 0.15)

            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Polish {round_}: {best_score:.8f} **")
            else:
                print(f"  Polish {round_}: {obj:.8f}")
                # Small perturbation to escape
                rng = np.random.default_rng(round_ * 31 + 7)
                a = normalize(best_seq * np.exp(rng.normal(0, 0.02, n_final)))

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = two_block_init(700, frac=1.0 / 3.0)

    # Scale to reasonable range (avoid very small values that might cause issues)
    result_arr = np.asarray(best_seq, dtype=np.float64)
    result_arr = np.clip(result_arr, 0.0, 1000.0)

    # Ensure sum constraint
    s = float(np.sum(result_arr))
    if s < 0.01:
        result_arr = np.ones(100)

    result = [float(x) for x in result_arr]

    # Validate with actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
