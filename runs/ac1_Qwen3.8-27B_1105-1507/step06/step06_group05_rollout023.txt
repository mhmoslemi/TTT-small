```python
    size = 1

    fa = rfft(a, size)

def compute_obj(a):

    s = float(np.sum(a))

    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Use binary search for the threshold
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.sum(v_sorted > 0)
    if rho == 0:
        return np.ones(n) / n
    theta = (css[rho-1] - 1.0) / rho
    # Refine with the actual v
    projected = np.maximum(v - theta, 0.0)
    s = float(np.sum(projected))
    if s < 1e-15:
        return np.ones(n) / n
    return projected / s
```

```python
def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    # Find rho: largest index where u[rho] - (css[rho] - 1) / (rho+1) > 0
    cond = u - (css - 1.0) / np.arange(1, n+1)
    rho = np.sum(cond > 0)
    if rho == 0:
        return np.ones(n) / n
    theta = (css[rho-1] - 1.0) / rho
    projected = np.maximum(v - theta, 0.0)
    s = float(np.sum(projected))
    if s < 1e-15:
        return np.ones(n) / n
    return projected / s
```

```python
def adam_smooth_optimize(a_init, tau_start, tau_end, max_iter, lr, time_budget):
    """
    Adam optimization of the log-sum-exp smoothed autocorrelation peak.
    a is maintained on the simplex (non-negative, sum=1).
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    best_a = a.copy()
    best_smooth_obj = np.inf
    start = time.monotonic()
    
    # Precompute FFT size
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    for t in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive temperature
        tau = tau_start * (tau_end / tau_start) ** (t / max(1, max_iter-1))
        
        # Compute smoothed objective and gradient
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        shifted = tau * (conv - c_max)
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w = w / w_sum
        
        smooth_obj = 2.0 * n * (c_max + np.log(w_sum) / tau)
        
        # Normalize w for gradient (the gradient formula uses p_k = exp(τ*conv_k)/sum, 
        # which is exactly w after normalization)
        # grad_i = 2*n * (D_i + D_{2n-2-i}) where D = cross_corr(a, w)
        
        a_pad = np.zeros(size)
        a_pad[:n] = a
        w_pad = np.zeros(size)
        w_pad[:2*n-1] = w
        
        Af = rfft(a_pad, size)
        Wf = rfft(w_pad, size)
        D = irfft(np.conj(Af) * Wf, size)
        
        D_fwd = D[:n]
        D_bwd = D[n-1:2*n-1][::-1]
        grad = 2.0 * n * (D_fwd + D_bwd)
        
        # Center the gradient (remove component along the all-ones direction
        # since we're optimizing on the simplex)
        grad = grad - np.mean(grad)
        
        # Adam update
        t1 = t + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t1)
        v_hat = v / (1 - beta2**t1)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Evaluate
        if smooth_obj < best_smooth_obj:
            best_smooth_obj = smooth_obj
            best_a = a.copy()
        
        # Also check actual objective periodically
        actual_obj = compute_obj(a_new)
        if actual_obj < compute_obj(best_a):
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a
```

- ∂a_i/∂b_j = a_i * (δ_ij - a_j)
- ∂f/∂b_j = sum_i (∂f/∂a_i) * a_i * (δ_ij - a_i) ... wait
- ∂f/∂b_j = sum_i (∂f/∂a_i) * (∂a_i/∂b_j) = sum_i (∂f/∂a_i) * a_i * (δ_ij - a_j)
  = a_j * (∂f/∂a_j - sum_i (∂f/∂a_i) * a_i)

So the gradient in b-space is: `a * (grad_a - dot(a, grad_a))`

This is nice because:
1. No need for projection
2. The step in b-space is unconstrained
3. Adam works naturally in this space
4. Non-negativity is automatic

```python
def adam_log_param_optimize(a_init, tau_start, tau_end, max_iter, lr, time_budget):
    n = len(a_init)
    a_init = normalize(a_init.copy().astype(np.float64))
    # Initialize b such that softmax(b) = a_init
    b = np.log(a_init + 1e-30)  # This gives softmax(b) ≈ a_init (not exact due to the +1e-30)
    # Better: b = log(a_init), then softmax(b) = a_init / sum(a_init) = a_init (since sum = 1)
    # Actually softmax(log(a)) = a / sum(a) = a when sum(a) = 1. 
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    for t in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Current a from b
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a = a / a.sum()
        
        # Adaptive temperature
        progress = t / max(1, max_iter - 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute grad_a (gradient of smooth objective w.r.t. a)
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        shifted = tau * (conv - c_max)
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w = w / w_sum
        
        # Cross-correlation
        a_pad = np.zeros(size)
        a_pad[:n] = a
        w_pad = np.zeros(size)
        w_pad[:2*n-1] = w
        
        Af = rfft(a_pad, size)
        Wf = rfft(w_pad, size)
        D = irfft(np.conj(Af) * Wf, size)
        
        D_fwd = D[:n]
        D_bwd = D[n-1:2*n-1][::-1]
        grad_a = 2.0 * n * (D_fwd + D_bwd)
        
        # Convert to b-space gradient
        grad_b = a * (grad_a - np.dot(a, grad_a))
        
        # Adam update
        t1 = t + 1
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * grad_b**2
        m_hat = m / (1 - beta1**t1)
        v_hat = v / (1 - beta2**t1)
        
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate actual objective periodically
        if t % 10 == 0 or t == max_iter - 1:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a
```
