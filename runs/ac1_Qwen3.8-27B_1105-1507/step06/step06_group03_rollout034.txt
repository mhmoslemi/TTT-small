def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(idx)
            cols.append(n)  # t variable at index n
            vals.append(-1.0)
            b_ub.append(c[k])

        A_ub = coo_matrix((vals, (rows, cols)), shape=(n_constraints, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]
        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), list(range(n)))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        t_lp_start = time.monotonic()
        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': time_per_lp, 'presolve': True})
        except Exception:
            no_improve += 1
            continue

        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
                # Line search
                found = False
                for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                    a_try = (1 - frac) * a + frac * g
                    a_try = np.maximum(a_try, 0)
                    if a_try.sum() < 1e-10:
                        continue
                    a_try = a_try / a_try.sum()
                    J_try = compute_J(a_try)
                    if J_try < best_obj - 1e-12:
                        a = a_try
                        best_obj = J_try
                        best_a = a_try.copy()
                        no_improve = 0
                        found = True
                        break
                if not found:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj


def smooth_adam(a, max_iters, deadline, tau_init=15.0, tau_max=150.0, lr_init=0.04):
    """Adam with log-sum-exp smoothing, working on the simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    no_improve = 0
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init + (tau_max - tau_init) * (frac ** 0.7)

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
        # Use FFT for efficiency
        size = 1
        while size < 3 * n:
            size <<= 1

        a_padded = np.zeros(size)
        a_padded[:n] = a
        w_padded = np.zeros(size)
        w_padded[:2 * n - 1] = w

        A_fft = np.fft.rfft(a_padded)
        W_fft = np.fft.rfft(w_padded)

        conv_wa = np.fft.irfft(W_fft * A_fft, size)

        # grad[i] = (w * a)[n-1+i] + (w * a_flipped)[...]
        # Actually: d(
        # conv[k])/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # grad[i] = sum_k w[k]*(a[k-i] + a[i-k])
        #          = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
        # First term: (w * a)_rev[i] where (w*a)[m] = sum_k w[k]*a[m-k], take m=i
        # Actually let's just do it directly with the FFT conv result
        # (w * a)[k] = sum_j w[j] * a[k-j]
        # We want sum_k w[k] * a[k-i] = (w * a_rev)[i] where a_rev[j] = a[-j mod n]... 
        # This is getting complex. Let me just compute it directly for moderate n.
        
        grad = np.zeros(n)
        for i in range(n):
            # sum_k w[k] * a[k-i] for valid k-i (0 <= k-i <= n-1)
            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            if k_min <= k_max:
                grad[i] += np.sum(w[k_min:k_max+1] * a[0:k_max-k_min+1])
            # sum_k w[k] * a[i-k] for valid i-k (0 <= i-k <= n-1)
            k_min2 = max(0, i - (n - 1))
            k_max2 = min(2 * n - 2, i - 1)
            if k_min2 <= k_max2:
                grad[i] += np.sum(w[k_min2:k_max2+1] * a[i-k_max2:i-k_min2+1][::-1])

        # Project to simplex tangent
        grad = grad - np.mean(grad)

        t_count += 1
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                a = best_a + rng.normal(0, 0.02, n)
                a = project_simplex(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

    return best_a, best_obj


def lp_mirror_descent(a, max_iters, deadline, top_k=50):
    """LP mirror descent with reduced constraints."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        k_indices = np.argsort(c)[::-1][:top_k]

        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(k_indices):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])

        A = coo_matrix((vals, (rows, cols)), shape=(len(k_indices), n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(len(k_indices), M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.15, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try = a_try / a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 25:
            break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.10, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.20, n)))

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 5, 8]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t)))

    for center in [0.2, 0.35, 0.5, 0.65, 0.8]:
        for width in [0.05, 0.1, 0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.01))

    for alpha in [0.3, 0.5, 1.0, 2.0, 3.0, 5.0]:
        inits.append(project_simplex((t + 0.01) ** alpha))
        inits.append(project_simplex(((1 - t) + 0.01) ** alpha))

    # Bell-shaped (high in middle, low at edges)
    inits.append(project_simplex(np.sin(np.pi * t) + 0.01))
    inits.append(project_simplex(np.sin(np.pi * t) ** 2 + 0.01))
    inits.append(project_simplex(np.sin(np.pi * t) ** 4 + 0.01))

    # W-shaped
    inits.append(project_simplex(1.0 + 0.5 * np.cos(4 * np.pi * t)))

    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    return inits


def solve_sequence(a, deadline, verbose=False):
    """Multi-stage optimization for a single sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return best_obj

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 1: Fast CD (25%)
    t1 = time.monotonic() + 0.25 * remaining
    a = fast_cd(a, max_sweeps=5000, deadline=t1, seed=42)
    try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 2: Golden section CD (25%)
    t2 = time.monotonic() + 0.25 * remaining
    a = golden_section_cd(a, max_sweeps=1000, deadline=t2, gs_iters=8, seed=77)
    try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 3: Reduced SCA-LP (25%)
    t3 = time.monotonic() + 0.25 * remaining
    a_lp, obj_lp = reduced_sca_lp(a, max_iters=100, deadline=t3, top_k=30)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_obj

    # Stage 4: Smooth Adam (25%)
    t4 = deadline - 5
    a_adam, obj_adam = smooth_adam(a, max_iters=3000, deadline=t4,
                                    tau_init=10.0, tau_max=100.0, lr_init=0.03)
    if obj_adam < best_obj:
        best_a = a_adam.copy()
        best_obj = obj_adam

    # Final fast CD polish
    if time.monotonic() < deadline - 10:
        a_final = fast_cd(best_a, max_sweeps=3000, deadline=deadline - 2, seed=999)
        try_update(a_final)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses: fast CD, golden section CD, reduced SCA-LP, Adam with log-sum-exp,
    multi-length exploration, and perturbation-based restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Intensive optimization at parent length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Optimization at n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_deadline = t_start + 450.0

    # First, quick CD on parent
    a1 = fast_cd(parent, max_sweeps=3000, deadline=min(t_start + 60, phase1_deadline), seed=1)
    update_best(a1)

    # Then golden section CD
    remaining = phase1_deadline - time.monotonic()
    if remaining > 30:
        a2 = golden_section_cd(a1, max_sweeps=500, deadline=t_start + 200, gs_iters=10, seed=2)
        update_best(a2)

    # Then SCA-LP
    remaining = phase1_deadline - time.monotonic()
    if remaining > 30:
        a3, obj3 = reduced_sca_lp(best_seq, max_iters=80, deadline=phase1_deadline - 10, top_k=40)
        update_best(a3)

    # Then Adam
    remaining = phase1_deadline - time.monotonic()
    if remaining > 30:
        a4, obj4 = smooth_adam(best_seq, max_iters=2000, deadline=phase1_deadline - 5,
                                tau_init=8.0, tau_max=80.0, lr_init=0.03)
        update_best(a4)

    # Final CD polish
    if time.monotonic() < phase1_deadline - 10:
        a5 = fast_cd(best_seq, max_sweeps=5000, deadline=phase1_deadline - 2, seed=3)
        update_best(a5)

    # === PHASE 2: Multi-init exploration at parent length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-init exploration at n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_deadline = t_start + 800.0
    inits = generate_inits(n_parent, rng, best_seq)
    n_inits = min(len(inits), 8)
    per_init = (phase2_deadline - time.monotonic()) / max(1, n_inits) - 5

    for idx in range(n_inits):
        rem = phase2_deadline - time.monotonic()
        if rem < 30:
            break
        alloc = min(per_init, rem - 5)
        if alloc < 20:
            break

        a0 = inits[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()

        # Quick pipeline: fast CD + golden CD + LP
        t_c1 = time.monotonic() + 0.3 * alloc
        a_c = fast_cd(a0, max_sweeps=2000, deadline=t_c1, seed=100 + idx)
        update_best(a_c)

        rem = phase2_deadline - time.monotonic()
        if rem > 15:
            t_c2 = time.monotonic() + 0.4 * rem
            a_c = golden_section_cd(a_c, max_sweeps=200, deadline=t_c2, gs_iters=6, seed=200 + idx)
            update_best(a_c)

        rem = phase2_deadline - time.monotonic()
        if rem > 15:
            a_c, obj_c = reduced_sca_lp(a_c, max_iters=50, deadline=phase2_deadline - 5, top_k=25)
            update_best(a_c)

        print(f"  Init {idx}: best J={best_score:.8f}", flush=True)

    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    phase3_deadline = t_start + 1100.0
    candidate_ns = [384, 512, 640, 896, 1024, 1280, 1536]
    candidate_ns = [n for n in candidate_ns if n != n_parent and n <= 2048]
    candidate_ns.sort(key=lambda n: abs(n - n_parent))

    for new_n in candidate_ns[:4]:
        rem = phase3_deadline - time.monotonic()
        if rem < 50:
            break
        alloc = min(80.0, rem - 20)
        if alloc < 30:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        # Quick optimization at new length
        t_c1 = time.monotonic() + 0.4 * alloc
        a_c = fast_cd(a_ext, max_sweeps=2000, deadline=t_c1, seed=300)
        update_best(a_c)

        rem = phase3_deadline - time.monotonic()
        if rem > 15:
            t_c2 = time.monotonic() + 0.4 * rem
            a_c = golden_section_cd(a_c, max_sweeps=150, deadline=t_c2, gs_iters=6, seed=400)
            update_best(a_c)

        rem = phase3_deadline - time.monotonic()
        if rem > 10:
            a_c, _ = reduced_sca_lp(a_c, max_iters=30, deadline=phase3_deadline - 5, top_k=25)
            update_best(a_c)

        print(f"  n={new_n}: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Intensive final optimization on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final intensive optimization (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        remaining = deadline - time.monotonic()

        # Pass 1: Long golden section CD
        if remaining > 60:
            t_p1 = time.monotonic() + 0.2 * remaining
            a_p = golden_section_cd(best_seq, max_sweeps=1000, deadline=t_p1, gs_iters=12, seed=500)
            update_best(a_p)

        # Pass 2: SCA-LP with more constraints
        remaining = deadline - time.monotonic()
        if remaining > 50:
            t_p2 = time.monotonic() + 0.25 * remaining
            a_p, _ = reduced_sca_lp(best_seq, max_iters=100, deadline=t_p2, top_k=50, time_per_lp=8)
            update_best(a_p)

        # Pass 3: Adam with different params
        remaining = deadline - time.monotonic()
        if remaining > 50:
            t_p3 = time.monotonic() + 0.2 * remaining
            a_p, _ = smooth_adam(best_seq, max_iters=5000, deadline=t_p3,
                                  tau_init=5.0, tau_max=60.0, lr_init=0.02)
            update_best(a_p)

        # Pass 4: LP mirror descent
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_p4 = time.monotonic() + 0.2 * remaining
            a_p, _ = lp_mirror_descent(best_seq, max_iters=100, deadline=t_p4, top_k=40)
            update_best(a_p)

        # Pass 5: Perturbation + re-optimize (escape local minima)
        remaining = deadline - time.monotonic()
        if remaining > 40:
            n_pert = 3
            per_pert = remaining / n_pert - 5
            for p_i in range(n_pert):
                rem = deadline - time.monotonic()
                if rem < 20:
                    break
                alloc = min(per_pert, rem - 5)
                if alloc < 15:
                    break
                a_pert = best_seq + rng.normal(0, 0.01, len(best_seq))
                a_pert = project_simplex(a_pert)
                a_opt = fast_cd(a_pert, max_sweeps=2000, deadline=time.monotonic() + 0.5 * alloc, seed=600 + p_i)
                update_best(a_opt)
                rem = deadline - time.monotonic()
                if rem > 10:
                    a_opt = golden_section_cd(a_opt, max_sweeps=100, deadline=deadline - 3, gs_iters=8, seed=700 + p_i)
                    update_best(a_opt)

        # Final polish
        if time.monotonic() < deadline - 10:
            a_final = fast_cd(best_seq, max_sweeps=5000, deadline=deadline - 3, seed=999)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
