import numpy as np
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n-1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_n = a / s
    n = len(a_n)
    b = fast_conv(a_n)
    return float(2.0 * n * b.max())

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def cross_corr_grad(a, w):
    """Compute g[i] = 2 * sum_k w[k] * a[k-i] * 1_{0<=k-i<n} for i=0..n-1."""
    n = len(a)
    w_rev = w[::-1]
    size = 1
    while size < 3*n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)
    return 2.0 * conv[n-1:2*n-1][::-1]

def nesterov_subgrad(a, deadline, tau_init=30.0, tau_max=300.0,
                      lr0=0.05, verbose=False):
    """
    Nesterov accelerated subgradient descent on the simplex.
    Uses log-sum-exp smoothing of the max.
    """
    n = len(a)
    a = np.maximum(a.copy(), 1e-10)
    a = a / a.sum()
    
    y = a.copy()  # Nesterov lookahead
    a_prev = a.copy()
    t = 0
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    
    while time.monotonic() < deadline:
        t += 1
        frac = min(1.0, t / 5000.0)
        tau = tau_init + (tau_max - tau_init) * frac
        
        # Compute smooth gradient at y
        conv = fast_conv(y)
        c_max = conv.max()
        shifted = conv - c_max
        exp_c = np.exp(tau * shifted)
        w = exp_c / exp_c.sum()
        
        g = cross_corr_grad(y, w)
        # Project gradient to tangent space
        g = g - g.mean()
        
        # Nesterov momentum
        theta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        alpha = (1.0 + t) / theta  # learning rate factor
        lr = lr0 / (np.sqrt(t) + 1.0)
        
        a_new = a - lr * alpha * g
        a_new = project_simplex(a_new)
        
        # Update y
        y = a_new + ((t-1) / theta) * (a_new - a_prev)
        y = np.maximum(y, 0)
        if y.sum() > 0:
            y = y / y.sum()
        
        a_prev = a.copy()
        a = a_new
        
        # Check improvement
        obj = compute_J(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart
            a = best_a + np.random.default_rng(t).normal(0, 0.01, n)
            a = project_simplex(a)
            y = a.copy()
            t = 0
            no_improve = 0
        
        if verbose and t % 500 == 0:
            print(f"    Nesterov t={t}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)
    
    return best_a, best_obj

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_n = a / s
    n = len(a_n)
    b = fast_conv(a_n)
    return float(2.0 * n * b.max())


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def cross_corr_grad(a, w):
    """
    Compute g[i] = 2 * sum_k w[k] * a[k-i] * 1_{0<=k-i<n} for i=0..n-1.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    w_rev = w[::-1]
    size = 1
    while size < 3 * n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def smooth_gradient(a, tau):
    """Compute smooth gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = fast_conv(a)
    c_max = conv.max()
    shifted = conv - c_max
    exp_c = np.exp(tau * shifted)
    total = exp_c.sum()
    if total < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_c / total
    g = cross_corr_grad(a, w)
    return g, conv, w


def water_filling_step(a, mu, relaxation=0.7):
    """
    One water-filling step: given mu (weights on lags),
    compute the optimal a via water-filling (soft thresholding).
    """
    n = len(a)
    g = cross_corr_grad(a, mu)
    # Find water level c such that sum(max(0, c - g)) = 1
    g_sorted = np.sort(g)[::-1]
    cssv = np.cumsum(g_sorted)
    # Binary search for the right number of active coordinates
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = g_sorted - (cssv - 1.0) / ind > 0
    k_active = np.where(cond)[0]
    if len(k_active) == 0:
        c = g_sorted[0] + 1.0 / n
    else:
        k = k_active[-1] + 1
        c = (cssv[k - 1] - 1.0) / k
    a_new = np.maximum(c - g, 0.0)
    sn = a_new.sum()
    if sn > 1e-15:
        a_new = a_new / sn
    else:
        a_new = a.copy()
    return (1.0 - relaxation) * a + relaxation * a_new


def nesterov_optimize(a, deadline, tau_init=20.0, tau_max=500.0,

        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        ys = y.sum()

        lr = lr_init / (np.sqrt(t) + 1.0)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):

    inits.append(project_simplex(0.5 * (1 + np.cos(2 * np.pi * t))))

    for width in [0.15, 0.25, 0.4, 0.6]:
        inits.append(project_simplex(np.exp(-((t - 0.5) / width) ** 2) + 0.01))

    # Power law
    for alpha in [0.5, 1.0, 2.0]:

    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')

    # Stage 2: Nesterov

        return best_a, best_obj

    # Stage 4: Directed coordinate descent

    a_dcd, obj_dcd = directed_coord_descent(a, t4, verbose=False)

        update_best(parent)

    # Determine candidate lengths
    n_parent = len(parent)
    candidate_ns = sorted(set([n_parent, 1024, 1536, 2048, 2560, 3072, 4096]))
    # Prioritize: parent first, then nearby, then larger
    candidate_ns.sort(key=lambda n: (abs(n - n_parent) if n != n_parent else -1, n))
    # Make sure we try at least: parent, 2*parent, 4*parent, and some power-of-2
    must_try = [n_parent]
    for target in [2 * n_parent, 4 * n_parent]:
        closest = min(candidate_ns, key=lambda n: abs(n - target))
        if closest not in must_try:
            must_try.append(closest)
    for p2 in [1024, 2048, 4096]:
        if p2 not in must_try:
            must_try.append(p2)
    candidate_ns = [n for n in candidate_ns if n in must_try or n < 5000]
    candidate_ns = sorted(candidate_ns)

    print(f"\n{'=' * 60}", flush=True)
    print(f"Candidate lengths: {candidate_ns}", flush=True)
    print(f"Total budget: {budget_s}s", flush=True)
    print(f"{'=' * 60}\n", flush=True)

    # Allocate time: give more time to larger n values (they're more promising)
    # but ensure parent gets tried first
    total_len = len(candidate_ns)
    # Weights: larger n gets more weight
    weights = []
    for n in candidate_ns:
        if n == n_parent:
            weights.append(0.8)  # Less time for parent (already optimized)
        else:
            weights.append(1.0 + 0.3 * np.log2(max(n / n_parent, 1.0)))
    total_w = sum(weights)
    time_budgets = []
    for w in weights:
        time_budgets.append(max(60.0, w / total_w * (budget_s - 30.0)))
    # Cap individual budgets
    max_budget = 400.0
    for i in range(len(time_budgets)):
        if time_budgets[i] > max_budget:
            time_budgets[i] = max_budget

    # Try each length
    for idx, n in enumerate(candidate_ns):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            print(f"  Time running out, stopping at n={n}", flush=True)
            break

        alloc = min(time_budgets[idx], remaining - 30.0)
        if alloc < 50:
            continue

        print(f"\n--- n={n} (budget {alloc:.0f}s) ---", flush=True)

        # Generate initializations
        inits = generate_inits(n, rng, parent)
        # Limit number of inits based on time budget
        max_inits = min(len(inits), max(2, int(alloc / 100.0)))
        per_init_time = alloc / max_inits - 5

        phase_best_a = None
        phase_best_obj = float('inf')

        for init_idx in range(max_inits):
            if time.monotonic() >= deadline - 30:
                break
            a0 = inits[init_idx].copy()
            a0 = np.maximum(a0, 0)
            if a0.sum() < 1e-10:
                continue
            a0 = a0 / a0.sum()

            init_deadline = time.monotonic() + per_init_time
            if init_deadline > deadline - 30:
                init_deadline = deadline - 30

            a_opt, obj_opt = optimize_at_length(a0, init_deadline, verbose=False,
                                                 seed=seed + idx * 100 + init_idx)
            if obj_opt < phase_best_obj:
                phase_best_obj = obj_opt
                phase_best_a = a_opt.copy()

            print(f"    Init {init_idx}: J={obj_opt:.8f}", flush=True)

        if phase_best_a is not None:
            update_best(phase_best_a)
            # Use phase best as parent for next lengths
            parent = phase_best_a

    # Final intensive polish on the overall best
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        print(f"\n{'=' * 60}", flush=True)
        print(f"FINAL POLISH: remaining {remaining:.0f}s, best J={best_score:.8f} n={best_seq.size}", flush=True)
        print(f"{'=' * 60}\n", flush=True)

        n_best = len(best_seq)
        # Try a few more optimization passes on the best
        for pass_idx in range(3):
            if time.monotonic() >= deadline - 20:
                break
            pass_deadline = min(time.monotonic() + remaining / (4 - pass_idx), deadline - 15)

            # Slight perturbation for each pass
            a_try = best_seq + rng.normal(0, 0.002 * (pass_idx + 1), n_best)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()

            a_opt, obj_opt = optimize_at_length(a_try, pass_deadline, verbose=False,
                                                 seed=seed + 10000 + pass_idx)
            update_best(a_opt)

            # Also try at a slightly larger length
            if time.monotonic() < deadline - 60:
                n_try = min(n_best + 256, 5000)
                if n_try != n_best:
                    a_res = resample_seq(best_seq, n_try)
                    a_res = np.maximum(a_res, 1e-6)
                    a_res = a_res / a_res.sum()
                    res_deadline = min(time.monotonic() + 60.0, deadline - 15)
                    a_opt2, obj_opt2 = optimize_at_length(a_res, res_deadline, verbose=False,
                                                           seed=seed + 20000 + pass_idx)
                    update_best(a_opt2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    # Final evaluation
    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={best_seq.size}", flush=True)
    print(f"Time used: {time.monotonic() - t_start:.1f}s / {budget_s}s", flush=True)
    print(f"{'=' * 60}\n", flush=True)

    return [float(x) for x in best_seq]
