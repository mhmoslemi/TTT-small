"""
Optimization combining smooth log-sum-exp gradient descent with backtracking
line search, adaptive temperature annealing, and LP-based polishing.
Prioritizes deep optimization with periodic diverse restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(a):
    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    n = len(a)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        return a / s
    return np.ones(n) / n


def lse_gradient(a, tau):
    """Compute gradient of 2n*tau*LSE(conv(a,a)/tau) assuming sum(a)=1.
    Returns grad where grad[i] = 4n * sum_j a[j]*w[i+j]."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    exp_shifted = np.exp(np.clip(shifted, -500, 500))
    w = exp_shifted / np.sum(exp_shifted)
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    
    A_fft = rfft(a, fft_size)
    W_fft = rfft(w_padded, fft_size)
    
    T1 = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    
    grad = 4.0 * n * T1
    return grad


def smooth_optimize(a_init, tau_start=0.05, tau_end=0.001, max_iters=5000,
                    time_budget=100.0, lr_init=0.1):
    """
    Smooth optimization using LSE gradient with adaptive temperature
    and backtracking line search.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive temperature
        frac = min(it / max(max_iters, 1), 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        grad = lse_gradient(a, tau)
        
        # Backtracking line search
        for _ in range(20):
            a_try = a - lr * grad
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj + 1e-10:
                a = a_try
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a.copy()
                break
            lr *= 0.5
        
        # Slowly increase lr if stuck
        if it % 100 == 99:
            lr = max(lr, lr_init * 0.1)
    
    return best_a, best_obj

def generate_starts(n, parent=None, num_starts=10, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n)
    center = (n - 1) / 2.0
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Gaussian (various widths)
    for sigma_frac in [0.15, 0.25, 0.35, 0.5]:
        bell = np.exp(-((i - center)**2) / (2 * (n * sigma_frac)**2))
        starts.append(bell / np.sum(bell))
    
    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    starts.append(hann / np.sum(hann))
    
    # Raised cosine
    rc = np.cos(np.pi * (i / (n - 1) - 0.5))
    rc = np.maximum(rc, 0.01)
    starts.append(rc / np.sum(rc))
    
    # Random
    for _ in range(3):
        r = np.abs(rng.normal(0.5, 0.3, n))
        starts.append(r / np.sum(r))
    
    # Parent if available
    if parent is not None:
        starts.append(project_simplex(parent.copy()))
    
    return starts


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):

    s = float(np.sum(a))
