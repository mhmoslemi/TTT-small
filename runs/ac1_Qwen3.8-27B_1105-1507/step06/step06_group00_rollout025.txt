```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Smoothed gradient (log-sum-exp + FISTA) + LP alternating optimization
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-based gradient computation, simplex projection, multi-beta sweeping,
and coordinate descent refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_smoothed_obj_grad(a, beta):
    """
    Compute the smoothed objective and its gradient w.r.t. a.
    f_β(a) = 2n/beta * log(sum exp(beta * conv(a,a)[k]))
    Returns (f, grad) where grad = df/da.
    """
    n = len(a)
    conv = compute_conv(a)
    bc = beta * conv
    max_bc = np.max(bc)
    shifted = np.exp(bc - max_bc)
    lse = max_bc + np.log(np.sum(shifted))
    f = 2.0 * n * lse / beta

    p = shifted / np.sum(shifted)

    # Gradient: df/da[i] = 2n/beta * sum_k p[k]*(a[k-i]+a[i-k])
    a_rev = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    P = rfft(p, size)
    A = rfft(a, size)
    A_rev = rfft(a_rev, size)
    conv_pa = irfft(P * A, size)
    conv_pa_rev = irfft(P * A_rev, size)
    c2 = conv_pa[:n]
    c1 = conv_pa_rev[n - 1:n - 1 + n]
    grad = 2.0 * n / beta * (c1 + c2)

    return f, grad


def fista_smoothed(a, beta, max_iter=300, tol=1e-12, verbose=False):
    """
    FISTA (accelerated projected gradient) on the smoothed objective.
    Works on the simplex {a>=0, sum(a)=1}.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    x = a.copy()  # current iterate
    y = a.copy()  # momentum variable
    t = 1.0
    step = 1.0 / (2.0 * n * beta * 2.0)  # Lipschitz estimate
    min_step = 1e-10

    for it in range(max_iter):
        f_y, grad_y = lse_smoothed_obj_grad(y, beta)
        # Project gradient to tangent space
        grad_y_proj = grad_y - np.mean(grad_y)

        # Gradient step + projection
        x_new = project_simplex(y - step * grad_y_proj)

        # Backtracking if needed
        f_x, _ = lse_smoothed_obj_grad(x_new, beta)
        backtracks = 0
        while f_x > f_y - 0.5 * step * np.sum(grad_y_proj ** 2) + 1e-14 and backtracks < 20:
            step *= 0.5
            x_new = project_simplex(y - step * grad_y_proj)
            f_x, _ = lse_smoothed_obj_grad(x_new, beta)
            backtracks += 1
        if backtracks == 0 and step < 1.0 / (2.0 * n * beta * 2.0):
            step = min(step * 1.2, 1.0 / (2.0 * n * beta * 2.0))

        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = project_simplex(y_new)
        y = y_new
        x = x_new
        t = t_new

        obj = compute_obj(x)
        if obj < best_obj:
            best_obj = obj
            best_a = x.copy()

        if it % 50 == 0 and verbose:
            print(f"      FISTA it{it}: obj={best_obj:.8f}, step={step:.2e}")

        if it > 10 and abs(best_obj - compute_obj(x)) < tol:
            break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Try CVXPY
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass

    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None


def alternating_lp(a, time_budget=60.0, verbose=False):
    """
    LP alternating: iteratively solve min-peak LP and blend with current a.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Adaptive line search
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        candidate_ts = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0 < last_t < 1:
            candidate_ts = sorted(set(candidate_ts + [max(0.05, last_t * 0.5), min(1.0, last_t * 1.5)]))
            candidate_ts = [t for t in candidate_ts if 0 < t <= 1.0]

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and it % 10 == 0:
                print(f"      LP alt it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(hash((it, n, int(time.monotonic()*1000))) % (2**32))
                scale = 0.03 + 0.02 * min(stall // 15, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def coordinate_refine(a, time_budget=20.0):
    """
    Coordinate descent: pairwise adjustments to reduce peak convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from positions contributing to the peak
        step_stride = max(1, n // 100)
        for i in range(0, n, step_stride):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i:
                if a[i] > 1e-8:
                    for frac in [0.01, 0.03, 0.05, 0.1]:
                        delta = frac * a[i]
                        if delta > a[i]:
                            continue
                        a[i] -= delta
                        a[j] += delta
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                            break

        if not improved:
            # Try the other direction
            for i in range(0, n, step_stride):
                if time.monotonic() - start > time_budget:
                    break
                j = k_star - i
                if 0 <= j < n and j != i:
                    if a[j] > 1e-8:
                        for frac in [0.01, 0.03, 0.05]:
                            delta = frac * a[j]
                            a[j] -= delta
                            a[i] += delta
                            obj_try = compute_obj(a)
                            if obj_try < best_obj - 1e-13:
                                best_obj = obj_try
                                best_a = a.copy()
                                improved = True
                                break
                            else:
                                a[j] += delta
                                a[i] -= delta
                                break
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def u_shaped_init(n, curvature=1.0):
    """Generate a U-shaped (convex) sequence."""
    x = np.linspace(0, 1, n)
    a = 1.0 + curvature * (2 * x - 1) ** 2
    a = a / np.sum(a)
    return a


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        a = u_shaped_init(n, curvature=rng.uniform(0.5, 3.0))
    elif mode == 3:
        # Smooth random
        a = np.abs(rng.normal(0, 1, n))
        a = a + 0.1
        a = a / np.sum(a)
    else:
        # Bimodal
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.7, 0.9)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / np.sum(a)
    return a

            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.15, 35.0)
        a_opt = full_optimization_pipeline(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Random restarts ===
    print(f"\n=== Phase 4: Random Restarts ===")
    n_best = len(best_seq)
    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_r = random_init(n_best, rng)
        alloc = min(remaining * 0.12, 25.0)
        a_opt = full_optimization_pipeline(a_r, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_opt:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_final = best_seq.copy()

        # Multiple LP alternating rounds
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.25, 30.0)
            a_new = alternating_lp(a_final, time_budget=alloc, verbose=True)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_final = a_new.copy()
                print(f"  Final LP {rnd}: obj={best_score:.8f}")

        # FISTA with high beta
        remaining = deadline - time.monotonic()
        if remaining > 15:
            for beta in [200, 500, 1000]:
                remaining = deadline - time.monotonic()
                if remaining < 10:
                    break
                a_new = fista_smoothed(a_final, beta, max_iter=100)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_final = a_new.copy()
                    print(f"  Final FISTA beta={beta}: obj={best_score:.8f}")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_new = coordinate_refine(a_final, time_budget=min(remaining * 0.4, 20.0))
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final CD: obj={best_score:.8f}")

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_new = alternating_lp(best_seq, time_budget=min(remaining * 0.5, 20.0))
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final LP2: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
