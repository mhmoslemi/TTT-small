import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_subgrad(a, tau):
    """Compute smooth subgradient of max_k (a*a)[k]. Returns (gradient, max_c, active_weights)."""
    n = len(a)

    max_c = c.max()

    # = 2 * correlate(w, a, 'full')[i] for i=0..n-1
    g = 2.0 * np.correlate(w, a, mode='full')[:n]
    return g, max_c, w

        obj_curr = compute_J(a)
        if obj_curr < best_obj - 1e-12:
            best_obj = obj_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic LP step
        if lp_every > 0 and iter_count % lp_every == 0 and no_improve > 5:
            remaining = deadline - time.monotonic()
            if remaining > lp_time + 2:
                a_lp = lp_direction_step(best_a, time_budget=lp_time)
                if a_lp is not None:
                    obj_lp = compute_J(a_lp)
                    if obj_lp < best_obj:
                        best_obj = obj_lp
                        best_a = a_lp.copy()
                        a = a_lp.copy()
                        no_improve = 0
                        lr = lr_init

        # Reset if stuck too long
        if no_improve > 200:
            a = project_simplex(best_a + 0.01 * np.random.default_rng(iter_count).standard_normal(n))
            lr = lr_init
            no_improve = 0

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t_start
            remaining = deadline - time.monotonic()
            print(f"    SubGrad it={iter_count} tau={tau:.1f} lr={lr:.4f} "
                  f"obj={best_obj:.8f} no_improve={no_improve} "
                  f"elapsed={elapsed:.0f}s remaining={remaining:.0f}s", flush=True)

    return best_a, best_obj


def lp_direction_step(a, time_budget=3.0):
    """
    Solve the SCA-LP direction-finding problem.
    max sum(g) s.t. (a*g)[k] <= c[k], g >= 0
    Return g/sum(g) or None if LP fails.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    c = fast_conv(a)
    M = c.max()

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': max(1, time_budget - 0.5), 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
                return g
    except Exception:
        pass
    return None


def active_set_lp(a, top_m=30, time_budget=2.0):
    """
    Active-set LP: use only the top M most active constraints.
    This is faster and often gives a better direction.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    c = fast_conv(a)
    M_val = c.max()

    # Get top M active constraints
    top_indices = np.argsort(c)[::-1][:top_m]
    active_c = c[top_indices]

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    A = coo_matrix((vals, (rows, cols)), shape=(top_m, n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = active_c
    bounds = [(0, None)] * n

    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': max(0.5, time_budget - 0.5), 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
                return g
    except Exception:
        pass
    return None


def golden_section_1d(f, a, b, tol=1e-12, max_iter=60):
    """Golden section search for 1D minimization."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fb = fd
            c = b - gr * (b - a)
            fd = fc
            fc = f(c)
        else:
            a = c
            c = d
            fa = fd
            d = a + gr * (b - a)
            fc = fd
            fd = f(d)
    return (a + b) / 2.0


def coord_descent_golden(a, deadline, verbose=True, seed=42):
    """
    Coordinate descent with golden section 1D minimization.
    For each coordinate, find the optimal value via golden section search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    best_obj = compute_J(a)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    sweep = 0

    def eval_coord(i, x):
        """Evaluate objective when coordinate i is set to x."""
        nonlocal conv
        old = a[i]
        delta = x - old
        if abs(delta) < 1e-15:
            return compute_J(a)
        lo = max(0, i)
        hi = min(2 * n - 2, i + n - 1)
        a_slice = a[lo - i:hi - i + 1].copy()
        conv[lo:hi + 1] += 2.0 * delta * a_slice
        kq = 2 * i
        if kq <= 2 * n - 2:
            conv[kq] += delta * delta
        new_s = s - old + x
        obj = 2.0 * n * conv.max() / (new_s * new_s) if new_s > 1e-10 else float('inf')
        conv[lo:hi + 1] -= 2.0 * delta * a_slice
        if kq <= 2 * n - 2:
            conv[kq] -= delta * delta
        return obj

    while time.monotonic() < deadline:
        sweep += 1
        order = rng.permutation(n)
        improved_sweep = False

        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Golden section search in [0, 2*old_val + 0.01] or [0, 0.1] if old_val is small
            if old_val > 1e-6:
                lo = 0.0
                hi = max(2.0 * old_val + 0.005, 0.05)
            else:
                lo = 0.0
                hi = 0.05

            # First check if current value is good
            obj_at_lo = eval_coord(i, lo)
            obj_at_hi = eval_coord(i, hi)
            obj_at_old = eval_coord(i, old_val)

            if obj_at_old <= obj_at_lo and obj_at_old <= obj_at_hi:
                # Current value is already good (local min in this interval)
                # Try a wider range
                hi2 = max(3.0 * old_val + 0.01, 0.1)
                obj_at_hi2 = eval_coord(i, hi2)
                if obj_at_hi2 < obj_at_old - 1e-12:
                    new_val = golden_section_1d(lambda x: eval_coord(i, x), old_val, hi2)
                else:
                    continue
            else:
                new_val = golden_section_1d(lambda x: eval_coord(i, x), lo, max(hi, old_val + 0.01))

            if abs(new_val - old_val) > 1e-12:
                # Apply the update
                delta = new_val - old_val
                lo_c = max(0, i)
                hi_c = min(2 * n - 2, i + n - 1)
                a_slice = a[lo_c - i:hi_c - i + 1].copy()
                conv[lo_c:hi_c + 1] += 2.0 * delta * a_slice
                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = new_val
                s = s - old_val + new_val
                improved_sweep = True
                # Renormalize periodically to keep s near 1
                if abs(s - 1.0) > 0.01:
                    a = a / s
                    s = 1.0
                    conv = fast_conv(a)

        current_obj = compute_J(a)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}", flush=True)

        if no_improve > 20:
            break

    return best_a, best_obj


def lp_polish(a, deadline, verbose=True):
    """
    Alternating LP + line search polishing.
    SCA-LP with various active set sizes.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # Try different active set sizes
        for top_m in [20, 50, 100, 200]:
            if time.monotonic() > deadline:
                break
            g = active_set_lp(best_a, top_m=top_m, time_budget=min(3.0, deadline - time.monotonic() - 1))
            if g is None:
                continue

            # Line search: try different blend fractions
            found = False
            for frac in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05]:
                a_try = (1 - frac) * best_a + frac * g
                a_try = np.maximum(a_try, 0)
                if a_try.sum() < 1e-10:
                    continue
                a_try /= a_try.sum()
                obj_try = compute_J(a_try)
                if obj_try < best_obj - 1e-13:
                    best_a = a_try.copy()
                    best_obj = obj_try
                    no_improve = 0
                    found = True
                    break
            if found:
                break

        if not found:
            no_improve += 1

        if verbose and it % 5 == 0:
            print(f"    LP polish it={it}: obj={best_obj:.8f} no_improve={no_improve}", flush=True)

        if no_improve > 30:
            break

    return best_a, best_obj


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences for multi-start."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
        # Asymmetric perturbations
        lin_tilt = np.linspace(-0.05, 0.05, n)
        inits.append(project_simplex(p + lin_tilt))
        inits.append(project_simplex(p - lin_tilt))

    # Standard patterns
    inits.append(np.ones(n) / n)
    for freq in [1, 2, 3, 5, 8]:
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * freq * t)))
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.03))
    # Smooth random
    ksize = max(7, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses: (1) smooth subgradient descent with adaptive tau, (2) LP direction finding,
    (3) golden-section coordinate descent, (4) multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # ============ PHASE 1: Subgradient descent on parent length (0-500s) ============
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Subgradient descent at n={n_parent}", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase1_deadline = t_start + 450.0
    a_curr = parent.copy()
    a_curr = a_curr / a_curr.sum()

    a_opt1, obj1 = subgrad_descent(a_curr, phase1_deadline,
                                    tau_init=20.0, tau_final=400.0,
                                    lr_init=1.0, lp_every=100, lp_time=2.0,
                                    verbose=True)
    update_best(a_opt1)
    print(f"  Phase 1 done: J={obj1:.8f}", flush=True)

    # ============ PHASE 2: LP polish (450-800s) ============
    remaining = deadline - time.monotonic()
    if remaining > 50:
        phase2_deadline = time.monotonic() + min(300.0, remaining * 0.5)
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 2: LP polish (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        a_opt2, obj2 = lp_polish(best_seq, phase2_deadline, verbose=True)
        update_best(a_opt2)
        print(f"  Phase 2 done: J={obj2:.8f}", flush=True)

    # ============ PHASE 3: Coordinate descent golden section (800-1200s) ============
    remaining = deadline - time.monotonic()
    if remaining > 50:
        phase3_deadline = time.monotonic() + min(350.0, remaining * 0.55)
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 3: Coord descent golden (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        a_opt3, obj3 = coord_descent_golden(best_seq, phase3_deadline, seed=123, verbose=True)
        update_best(a_opt3)
        print(f"  Phase 3 done: J={obj3:.8f}", flush=True)

    # ============ PHASE 4: Second round of subgradient + LP (1200-1400s) ============
    remaining = deadline - time.monotonic()
    if remaining > 40:
        phase4_deadline = time.monotonic() + min(150.0, remaining * 0.6)
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 4: Second subgrad round (budget {phase4_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Perturb and re-optimize
        a_pert = project_simplex(best_seq + 0.005 * rng.standard_normal(best_seq.size))
        a_opt4, obj4 = subgrad_descent(a_pert, phase4_deadline,
                                        tau_init=100.0, tau_final=500.0,
                                        lr_init=0.3, lp_every=150, lp_time=1.5,
                                        verbose=True)
        update_best(a_opt4)

        # Also try from the unperturbed best
        remaining4 = deadline - time.monotonic()
        if remaining4 > 30:
            a_opt4b, obj4b = subgrad_descent(best_seq, time.monotonic() + remaining4 * 0.7,
                                              tau_init=200.0, tau_final=600.0,
                                              lr_init=0.2, lp_every=200, lp_time=1.0,
                                              verbose=False)
            update_best(a_opt4b)

    # ============ PHASE 5: Try different lengths (1400-1470s) ============
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 5: Multi-length exploration (budget {remaining:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        candidate_ns = [512, 640, 768, 896, 1024]
        candidate_ns = [n for n in candidate_ns if n != n_parent]
        per_n_time = max(5.0, (remaining - 10) / len(candidate_ns))

        for new_n in candidate_ns[:4]:
            if time.monotonic() > deadline - 10:
                break
            a_ext = resample_seq(best_seq, new_n)
            a_ext = a_ext / a_ext.sum()
            # Quick subgradient run
            a_opt, obj = subgrad_descent(a_ext, time.monotonic() + min(per_n_time, remaining - 15),
                                          tau_init=50.0, tau_final=300.0,
                                          lr_init=0.5, lp_every=200, lp_time=1.0,
                                          verbose=False)
            update_best(a_opt)
            print(f"  n={new_n}: J={obj:.8f}", flush=True)

    # ============ PHASE 6: Final polish ============
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n{'=' * 60}", flush=True)
        print(f"PHASE 6: Final polish (budget {remaining:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Quick CD
        a_final, obj_final = coord_descent_golden(best_seq, time.monotonic() + remaining * 0.4, seed=999, verbose=False)
        update_best(a_final)

        # Quick LP
        remaining6 = deadline - time.monotonic()
        if remaining6 > 10:
            a_lp, obj_lp = lp_polish(best_seq, time.monotonic() + min(remaining6 * 0.5, 20), verbose=False)
            update_best(a_lp)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
