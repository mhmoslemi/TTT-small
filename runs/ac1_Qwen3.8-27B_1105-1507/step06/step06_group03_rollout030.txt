    """Compute J for arbitrary a."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_conv(a)
    return float(2.0 * n * conv.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def lse_smoothed_obj_and_grad(a, tau):
    """
    Compute log-sum-exp smoothed objective and gradient.
    Returns (obj, grad) where obj = 2n * (1/tau) * log(sum exp(tau*conv[k]))
    and grad is the gradient w.r.t. a.
    """
    n = len(a)
    conv = fast_conv(a)
    shifted = tau * (conv - conv.max())
    exp_c = np.exp(shifted)
    log_sum = conv.max() + np.log(exp_c.sum()) / tau
    obj = 2.0 * n * log_sum

    # Softmax weights
    w = exp_c / exp_c.sum()

    # Gradient: grad[m] = 2n * sum_k w[k] * 2 * a[k-m] (for valid k-m)
    # = 4n * sum_k w[k] * a[k-m]
    # This is 4n * (w_reversed * a) evaluated appropriately
    # Using FFT for efficiency:
    # grad = 4n * (w * a_rev) [appropriate slice] + (w_rev * a_rev) [appropriate slice]
    # Actually: sum_k w[k] * a[k-m] for k in [0, 2n-2]
    # = (w convolve a)[m] where we think of it as cross-correlation
    
    # More directly: grad[m] = 4n * sum_{k: valid} w[k] * a[k-m]
    # Let's compute via FFT:
    size = 1
    while size < 3 * n:
        size <<= 1
    a_rev = a[::-1]
    w_flip = w[::-1]
    
    A_fft = np.fft.rfft(a_rev, size)
    W_fft = np.fft.rfft(w, size)
    Wf_fft = np.fft.rfft(w_flip, size)
    
    # sum_k w[k]*a[k-m] = (w * a_rev)[n-1+m] (cross-correlation)
    c1 = np.fft.irfft(W_fft * A_fft, size)
    # sum_k w[k-m]*a[k] ... actually let me think differently
    
    # grad[m] = 4n * sum_{k=0}^{2n-2} w[k] * a[k-m] (where a[j]=0 if j not in [0,n-1])
    # = 4n * (cross-correlation of w and a at lag m)
    # = 4n * (w * flip(a))[n-1+m]
    
    # flip(a) = a[::-1], indexed as a_rev[j] = a[n-1-j]
    # (w * a_rev)[j] = sum_i w[i] * a_rev[j-i] = sum_i w[i] * a[n-1-j+i]
    # We want sum_k w[k] * a[k-m] = (w * a_rev)[n-1+m]... let me verify
    # (w * a_rev)[n-1+m] = sum_i w[i] * a_rev[n-1+m-i] = sum_i w[i] * a[n-1-(n-1+m-i)]
    # = sum_i w[i] * a[i-m] ... that's not quite right either.
    
    # Let me just do it directly for correctness (n is at most ~2000, so O(n^2) is fine for gradient)
    # Actually for n=768, O(n^2) = 590K operations per gradient, which is fine.
    
    grad = np.zeros(n)
    for m in range(n):
        # sum_k w[k] * a[k-m] where k-m in [0,n-1], i.e., k in [m, m+n-1]
        k_lo = max(0, m)
        k_hi = min(2 * n - 2, m + n - 1)
        if k_lo <= k_hi:
            grad[m] = np.dot(w[k_lo:k_hi+1], a[k_lo - m:k_hi - m + 1])
    
    grad *= 4.0 * n
    return obj, grad


def lse_obj_and_grad_fft(a, tau):
    """Faster version using FFT for the gradient computation."""
    n = len(a)
    conv = fast_conv(a)
    shifted = tau * (conv - conv.max())
    exp_c = np.exp(shifted)
    log_sum = conv.max() + np.log(exp_c.sum()) / tau
    obj = 2.0 * n * log_sum
    w = exp_c / exp_c.sum()

    # grad[m] = 4n * sum_k w[k] * a[k-m] for valid indices
    # This equals 4n * (w_rev * a_rev)[2n-2-m + n-1] ... let me use a cleaner approach
    # sum_k w[k]*a[k-m] = (w convolve a)[2*(n-1)-m] ... no
    # Actually: define c[m] = sum_k w[k]*a[k-m]. This is the cross-correlation.
    # c = correlate(w, a) in numpy terms, but with w of length 2n-1 and a of length n.
    # Let's use: c[m] = sum_k w[k]*a[k-m]
    # Pad a to length 2n-1, then c[m] = (w * flip(a_padded))[... ]
    
    # Simplest correct approach with FFT:
    # c[m] = sum_{k} w[k] * a[k-m]
    # Let b = a padded to length 2n-1 (b[j] = a[j] if j<n else 0)
    # c[m] = sum_k w[k] * b[k-m] = (w * flip(b))[2n-2+m]... 
    # Actually c = ifft(fft(w) * conj(fft(b))) gives circular cross-correlation
    # For linear: pad to size >= (2n-1) + n - 1 = 3n-2
    
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    b = np.zeros(size)
    b[:n] = a
    W_fft = np.fft.rfft(w, size)
    B_fft = np.fft.rfft(b, size)
    # Cross-correlation: c[m] = sum_k w[k]*b[k-m]
    # = ifft(fft(w) * conj(fft(b))) but we need the right offset
    # Actually: ifft(fft(w) * fft(b_rev)) where b_rev = b[::-1] gives correlation
    # Let me use: c = ifft(conj(fft(w)) * fft(b)) which gives sum_k w[k-m]*b[k]
    # That's not what we want either.
    
    # Let's just use: c = ifft(fft(w) * conj(fft(b)))
    # This gives c[m] = sum_k w[k] * b[(k-m) mod size] (circular)
    # For our case with sufficient padding, the circular = linear for the range we care about.
    
    C_fft = W_fft * np.conj(B_fft)
    c = np.fft.irfft(C_fft, size)
    # c[m] for m in [0, n-1] gives sum_k w[k]*b[k-m] = sum_k w[k]*a[k-m]
    grad = 4.0 * n * c[:n]
    return obj, grad


def fista_lse(a_init, deadline, tau_start=10.0, tau_end=80.0, verbose=False):
    """FISTA with log-sum-exp smoothed objective, annealed tau."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_J_norm(a)
    
    y = a.copy()
    t_k = 1.0
    max_iters = 20000
    no_improve = 0
    patience = 2000
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Anneal tau
        frac = min(1.0, it / (max_iters * 0.8))
        tau = tau_start + (tau_end - tau_start) * frac
        
        # Compute obj and grad at y
        obj_y, grad_y = lse_obj_and_grad_fft(y, tau)
        
        # Project gradient to tangent space
        grad_y = grad_y - np.mean(grad_y)
        
        # Adaptive step size
        # Lipschitz constant of grad of LSE is bounded by 4n * (some factor)
        # Use a conservative step
        step = 1.0 / (8.0 * n)
        
        a_new = project_simplex(y - step * grad_y)
        
        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        t_k = t_new
        a = a_new
        
        # Track best
        obj_a = compute_J_norm(a)
        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Restart momentum
            y = best_a.copy()
            t_k = 1.0
            no_improve = 0
            # Add small perturbation
            perturb = np.random.default_rng(it).normal(0, 0.01, n)
            a = project_simplex(best_a + perturb)
            y = a.copy()
        
        if verbose and it % 2000 == 0:
            print(f"    FISTA-LSE it {it}: tau={tau:.1f} obj={best_obj:.8f}", flush=True)
    
    return best_a, best_obj


def lbfgs_log_param(a_init, deadline, tau=30.0, verbose=False):
    """L-BFGS optimization with log-parameterization on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    z = np.log(a)  # log-parameterization
    
    best_a = a.copy()
    best_obj = compute_J_norm(a)
    
    def obj_grad(z_vec):
        a_vec = np.exp(z_vec - z_vec.max())  # stable softmax-like
        a_vec = a_vec / a_vec.sum()
        obj, grad_a = lse_obj_and_grad_fft(a_vec, tau)
        # Transform gradient: dz/dz = a_i * (grad_i - sum_j a_j * grad_j)
        grad_z = a_vec * (grad_a - np.dot(a_vec, grad_a))
        return obj, grad_z
    
    # A few L-BFGS restarts with different tau
    for tau_i in [15.0, 30.0, 50.0, 100.0]:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(obj_grad, z, method='L-BFGS-B',
                          options={'maxiter': 500, 'maxfun': 2000,
                                   'gtol': 1e-8, 'iprint': 0})
            if res.success or res.fun < best_obj:
                a_vec = np.exp(res.x - res.x.max())
                a_vec = np.maximum(a_vec, 1e-15)
                a_vec = a_vec / a_vec.sum()
                obj_vec = compute_J_norm(a_vec)
                if obj_vec < best_obj:
                    best_obj = obj_vec
                    best_a = a_vec.copy()
                    z = res.x.copy()
        except Exception:
            continue
    
    return best_a, best_obj


def lp_direction(a, deadline, verbose=False):
    """
    LP-based direction finding.
    Find g minimizing max_k (a*g)[k] subject to sum(g)=1, g>=0.
    Then move towards g.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J_norm(a)
    
    for it in range(50):
        if time.monotonic() > deadline:
            break
        
        # LP: minimize t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0
        # Variables: g[0..n-1], t (index n)
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
        
        A_ub = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        b_ub = np.zeros(2 * n - 1)
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]
        
        bounds = [(0, None)] * n + [(None, None)]
        
        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs',
                       options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                g = g / g.sum()
                
                # Try to move towards g
                found = False
                for frac in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05]:
                    a_try = (1 - frac) * a + frac * g
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    obj_try = compute_J_norm(a_try)
                    if obj_try < best_obj - 1e-12:
                        a = a_try
                        best_obj = obj_try
                        best_a = a_try.copy()
                        no_improve = 0
                        found = True
                        break
                if not found:
                    no_improve = getattr(lp_direction, '_no_improve', 0) + 1
                    lp_direction._no_improve = no_improve
                    if no_improve > 20:
                        break
            else:
                break
        except Exception:
            break
    
    lp_direction._no_improve = 0
    return best_a, best_obj


def sca_lp_step(a, max_iters=50, deadline=None, verbose=False):
    """SCA-LP: linearize quadratic constraint, solve LP."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J_norm(a)
    no_improve = 0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        # LP: minimize t s.t. 2*(a*g)[k] - t <= c[k] for all k
        rows = []
        cols = []
        vals = []
        b_ub = []
        
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])
        
        A_ub = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]
        bounds = [(0, None)] * n + [(None, None)]
        
        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs',
                       options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                if g.sum() > 1e-10:
                    g /= g.sum()
                    found = False
                    for frac in [0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        obj_try = compute_J_norm(a_try)
                        if obj_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = obj_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        if no_improve > 15:
            break
    
    return best_a, best_obj


def coord_descent_fast(a, deadline, verbose=False):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    conv = fast_conv(a)
    best_obj = compute_J_norm(a)
    best_a = a.copy()
    no_improve = 0
    
    for sweep in range(50000):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, 5e-3, 1e-2]
            else:
                candidates = [0.0, 0.3 * old, 0.5 * old, 0.7 * old, 1.3 * old, 1.6 * old]
            
            # Affected range in convolution
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            
            best_local = best_obj
            best_val = old
            
            for x in candidates:
                x = float(max(0.0, x))
                if abs(x - old) < 1e-14:
                    continue
                delta = x - old
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                
                new_obj = 2.0 * n * conv.max()  # sum is still ~1
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                conv[lo:hi+1] -= 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] -= delta * delta
            
            if abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = best_val
                improved = True
        
        current_obj = compute_J_norm(a)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
    
    inits.append(np.ones(n) / n)
    
    #Cosine patterns
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
    
    # Gaussian mixtures
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for w in [0.1, 0.2, 0.35]:
                g = np.exp(-((t - c1) / w)**2) + np.exp(-((t - c2) / w)**2)
                inits.append(project_simplex(g + 0.05))
    
    # Single Gaussian
    for c in [0.3, 0.5, 0.7]:
        for w in [0.1, 0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - c) / w)**2) + 0.05))
    
    # Power law
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        inits.append(project_simplex((t + 0.01)**alpha + 0.05))
        inits.append(project_simplex(((1-t) + 0.01)**alpha + 0.05))
    
    # Smooth random
    ksize = max(7, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: FISTA + L-BFGS + LP + CD, with multi-start and multi-length.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)
    
    n_parent = len(parent)
    
    # === PHASE 1: FISTA with multiple inits on parent length (40% time) ===
    phase1_deadline = t_start + 0.40 * budget_s
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: FISTA optimization at n={n_parent}", flush=True)
    print(f"{'='*60}", flush=True)
    
    inits = generate_inits(n_parent, rng, parent)
    n_trials = min(len(inits), 8)
    per_trial_time = (phase1_deadline - time.monotonic()) / max(1, n_trials) - 10
    
    for idx in range(n_trials):
        if time.monotonic() > phase1_deadline - 10:
            break
        alloc = min(per_trial_time, phase1_deadline - time.monotonic() - 5)
        if alloc < 10:
            break
        
        a0 = inits[idx]
        a0 = np.maximum(a0, 0)
        if a0.sum() < 1e-10:
            continue
        a0 = a0 / a0.sum()
        
        t0 = time.monotonic()
        a_opt, obj_opt = fista_lse(a0, time.monotonic() + alloc,
                                   tau_start=8.0, tau_end=60.0)
        update_best(a_opt)
        print(f"  FISTA init {idx}: J={obj_opt:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)
    
    # === PHASE 2: L-BFGS refinement (15% time) ===
    phase2_deadline = t_start + 0.55 * budget_s
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: L-BFGS refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        a_curr = best_seq.copy()
        a_curr = a_curr / a_curr.sum()
        a_lb, obj_lb = lbfgs_log_param(a_curr, time.monotonic() + (phase2_deadline - time.monotonic()),
                                       tau=25.0)
        update_best(a_lb)
        print(f"  L-BFGS: J={obj_lb:.8f}", flush=True)
    
    # === PHASE 3: LP-based refinement (15% time) ===
    phase3_deadline = t_start + 0.70 * budget_s
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP refinement (SCA + direct)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        a_curr = best_seq.copy() / best_seq.sum()
        
        # SCA-LP
        t_sca = time.monotonic() + 0.5 * (phase3_deadline - time.monotonic())
        a_sca, obj_sca = sca_lp_step(a_curr, max_iters=30, deadline=t_sca)
        update_best(a_sca)
        print(f"  SCA-LP: J={obj_sca:.8f}", flush=True)
        
        # Direct LP direction
        if time.monotonic() < phase3_deadline - 10:
            a_lp, obj_lp = lp_direction(best_seq.copy() / best_seq.sum(),
                                        time.monotonic() + (phase3_deadline - time.monotonic()))
            update_best(a_lp)
            print(f"  LP-dir: J={obj_lp:.8f}", flush=True)
    
    # === PHASE 4: Multi-length exploration (15% time) ===
    phase4_deadline = t_start + 0.85 * budget_s
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    candidate_ns = [384, 512, 640, 768, 896, 1024, 1152, 1280]
    candidate_ns = [n for n in candidate_ns if n != n_parent]
    candidate_ns.sort(key=lambda n: abs(n - n_parent))
    
    for new_n in candidate_ns[:4]:
        if time.monotonic() > phase4_deadline - 30:
            break
        alloc = min(60.0, phase4_deadline - time.monotonic() - 10)
        if alloc < 20:
            break
        
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()
        
        a_opt, obj_opt = fista_lse(a_ext, time.monotonic() + alloc,
                                   tau_start=8.0, tau_end=50.0)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)
    
    # === PHASE 5: Final intensive optimization (15% time) ===
    phase5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final optimization pass", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        a_curr = best_seq.copy() / best_seq.sum()
        remaining = phase5_deadline - time.monotonic()
        
        if remaining > 30:
            # FISTA with refined params
            t_f = time.monotonic() + 0.3 * remaining
            a_f, obj_f = fista_lse(a_curr, t_f, tau_start=5.0, tau_end=40.0)
            update_best(a_f)
            a_curr = best_seq.copy() / best_seq.sum()
        
        if remaining > 30 and time.monotonic() < phase5_deadline - 20:
            # L-BFGS with different tau
            t_l = time.monotonic() + 0.3 * (phase5_deadline - time.monotonic())
            a_l, obj_l = lbfgs_log_param(a_curr, t_l, tau=40.0)
            update_best(a_l)
            a_curr = best_seq.copy() / best_seq.sum()
        
        # Final coordinate descent
        if time.monotonic() < phase5_deadline - 10:
            a_cd = coord_descent_fast(a_curr, phase5_deadline - 5)
            update_best(a_cd)
    
    # Final fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
