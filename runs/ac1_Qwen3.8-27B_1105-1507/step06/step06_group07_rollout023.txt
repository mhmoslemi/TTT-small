import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def coord_1d_obj(x, a, i, old_conv, S):
    """Compute objective when a[i] is changed to x."""
    n = len(a)
    S_new = S - a[i] + x
    if S_new < 1e-10:
        return np.inf
    
    # Compute new conv
    # Only need to update the max
    # new_conv[k] = old_conv[k] - old_contrib(k) + new_contrib(k)
    # We can compute the max efficiently
    
    max_val = 0.0
    ai = a[i]
    for k in range(2*n - 1):
        ki = k - i  # k - i
        if ki == i:  # k = 2*i
            old_c = ai * ai
            new_c = x * x
        elif 0 <= ki < n and ki != i:
            old_c = 2.0 * ai * a[ki]
            new_c = 2.0 * x * a[ki]
        else:
            old_c = 0.0
            new_c = 0.0
        
        new_k = old_conv[k] - old_c + new_c
        if new_k > max_val:
            max_val = new_k
    
    return 2.0 * n * max_val / (S_new * S_new)

One concern: the objective as a function of x might not be unimodal. The max of several functions is convex, and dividing by (S_new)^2 (which is convex and positive) might break unimodality. Let me think...

max_affected(x) is convex in x (max of affine and one quadratic function that's convex).
S_new(x) = S - ai + x is linear.
(S_new(x))^2 is convex and increasing for x > 0 (assuming S - ai > 0, which it should be since a is non-negative and has more than one non-zero element).

So f(x) = max(max_affected(x), max_unaffected) / (S_new(x))^2 * 2n

The numerator is convex (and non-decreasing for large x), the denominator is convex and increasing. The ratio of two positive convex functions is not necessarily unimodal.

However, in practice, since we're doing local optimization (starting from the current value of a[i]), the function should be locally unimodal in a small neighborhood. For the golden section search, we can use a small range around the current value, like [max(0, ai/2), 2*ai].

Actually, let me think about this differently. Since we're doing coordinate descent on a non-convex problem, we should just accept that the 1D subproblem might have multiple local minima. We can handle this by:
1. Starting the golden section from [0, ai] and [ai, 2*ai] separately.
2. Or just using a small range like [max(0, ai - delta), ai + delta] for some delta.

For practical purposes, a small range around the current value should work well, since we're doing local optimization.

Let me also add a "swap" operation: instead of just changing one coordinate, sometimes swap mass between two coordinates. This helps escape local minima.

Now let me also think about the LP refinement more carefully. The previous LP approach solves:
max sum(g) s.t. conv(a, g) <= M, g >= 0

where M = max(conv(a,a)). This finds a direction g that, when blended with a, reduces the peak. The key insight is that this LP is feasible (g=a is always feasible since conv(a,a) <= M), and the solution gives a "better" direction.

"""
Coordinate descent with golden section search, combined with LP refinement 
and multi-n exploration, for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def coord_descent(a, max_sweeps=10, time_budget=60.0, seed=0):

        order = rng.permutation(n)

            if time.monotonic() - start > time_budget:

            ai = a[i]

                continue  # Skip some zero coordinates

            k_min = max(0, i)
            k_max = min(2*n-2, i+n-1)
            
            # Unaffected: k < k_min or k > k_max
            if k_min > 0:
                max_unaff_left = np.max(conv[:k_min])
            else:
                max_unaff_left = -np.inf
            if k_max < 2*n-2:
                max_unaff_right = np.max(conv[k_max+1:])
            else:

                    c2[idx] = 1

                    c1[idx] = 2*a[ki]
                    c2[idx] = 0
                else:

                    c1[idx] = 0
                    c2[idx] = 0

            lower = max(0, ai * 0.1)
            upper = max(ai * 5, 5*np.mean(a) + 0.01)

            b_gs = upper

                S_new = S - ai + x
                if S_new < 1e-10:

                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd

                    fd = eval_x(d_gs)

def build_toeplitz_conv(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j+n, 2*n-1)
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A

    n = len(a)
