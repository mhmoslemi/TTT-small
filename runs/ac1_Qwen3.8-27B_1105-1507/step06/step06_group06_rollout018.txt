    Solve the linearized LP:
    min t
    s.t. sum_i 2*a[k-i] * a[i] <= t + conv[k] for k in active
         sum(a) = 1
         a >= 0
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Build constraint matrix
    # For each active k: sum_i c[k,i] * a[i] - t <= conv[k]
    # where c[k,i] = 2*a[k-i] if 0 <= k-i < n, else 0

    b_ub = np.zeros(m)

            if 0 <= j < n:

        A_ub[idx, n] = -1.0  # coefficient of t
        b_ub[idx] = conv[k]

    c[n] = 1.0  # minimize t

"""
Linearized LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set management with successive linearization of the max-of-quadratics
objective, combined with line search, multi-length exploration, and simulated
annealing for global search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def get_active_set(conv, max_val, frac=0.1):
    """Get indices where conv is within frac of max."""
    threshold = max_val * (1 - frac)
    return np.where(conv >= threshold)[0]

def solve_linearized_lp(a, active_k, time_limit=5.0):
    """
    Solve: min t s.t. 2*(T(a)*a)[k] <= t + conv[k] for k in active_k,
           sum(a) = 1, a >= 0, t >= 0
    """
    n = len(a)
    conv = compute_conv(a)
    m = len(active_k)
    
    if m == 0:
        return None
    
    # Build constraint matrix: A_ub @ [a; t] <= b_ub
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    convvals = conv[active_k]
    
    for idx, k in enumerate(active_k):
        # Row: 2 * sum_i a[k-i] * a[i] - t <= conv[k]
        # i.e., sum_i (2*a[k-i]) * a[i] + (-1)*t <= conv[k]
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_end > i_start:
            indices = np.arange(i_start, i_end)
            a_indices = k_int - indices
            A_ub[idx, indices] = 2.0 * a[a_indices]
        A_ub[idx, n] = -1.0
        b_ub[idx] = convvals[idx]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            a_new = np.maximum(res.x[:n], 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
            return a_new
    except Exception:
        pass
    return None

def lp_descent(a, max_iters=200, time_budget=60.0, active_frac=0.05, 
               line_search=True, verbose=False):
    """
    Main LP-based descent: iteratively solve linearized LP, take step with line search.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Get active set
        active_k = get_active_set(conv, max_conv, frac=active_frac)
        
        # Solve LP
        a_lp = solve_linearized_lp(a, active_k, time_limit=3.0)
        if a_lp is None:
            stall += 1
            if stall > 5:
                break
            # Fallback: gradient step
            # Gradient of max conv
            k_star = int(np.argmax(conv))
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = 2.0 * a[j]
            grad -= np.mean(grad)
            a_new = a - 0.01 * grad
            a = project_nonneg_sum1(a_new)
            continue
        
        # Line search
        if line_search:
            best_t = 0.0
            best_t_obj = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = (1 - t) * a + t * a_lp
                a_try = project_nonneg_sum1(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t
                    best_t_a = a_try.copy()
            
            if best_t > 0:
                a = best_t_a
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a.copy()
                    stall = 0
                    if verbose and it % 20 == 0:
                        print(f"    LP iter {it}: obj={best_obj:.8f}")
                else:
                    stall += 1
            else:
                stall += 1
        else:
            a = project_nonneg_sum1(a_lp)
            obj_new = compute_obj(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        
        if stall > 15:
            break
    
    return best_a, best_obj

def smoothed_max_gradient(a, tau=0.01, top_k=20):
    """
    Compute gradient of smoothed max using softmax over top-k convolution values.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Use top-k values for the softmax
    top_indices = np.argsort(conv)[-top_k:]
    top_vals = conv[top_indices]
    
    # Softmax weights
    shifted = (top_vals - max_conv) / tau
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient: sum over active k of weights[k] * 2*a[k-i]
    grad = np.zeros(n)
    for idx, k in enumerate(top_indices):
        if weights[idx] < 1e-8:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_end > i_start:
            indices = np.arange(i_start, i_end)
            a_indices = k_int - indices
            grad[indices] += weights[idx] * 2.0 * a[a_indices]
    
    return grad, max_conv

def proximal_gradient(a, max_iters=500, time_budget=30.0, tau_start=0.05, tau_end=0.001,
                      lr=0.01, verbose=False):
    """
    Proximal gradient with adaptive tau for smoothed max.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

def simulated_annealing(a, rng, max_iters=200, T_init=0.1, T_min=0.001, time_budget=30.0):
    """Simulated annealing with targeted moves."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        T = T_init * (T_min / T_init) ** (it / max_iters)
        a_trial = cur_a.copy()
        
        move_type = rng.integers(0, 4)
        
        if move_type == 0:
            # Local perturbation
            w = max(5, n // 20)
            pos = rng.integers(0, n - w)
            a_trial[pos:pos+w] += rng.normal(0, 0.1, w)
        elif move_type == 1:
            # Gradient-based move with noise
            grad, _ = smoothed_max_gradient(cur_a, tau=0.02, top_k=10)
            grad -= np.mean(grad)
            a_trial -= 0.05 * grad + rng.normal(0, 0.01, n)
        elif move_type == 2:
            # Swap two windows
            w = max(10, n // 15)
            p1 = rng.integers(0, n - w)
            p2 = rng.integers(0, n - w)
            if abs(p1 - p2) >= w:
                a_trial[p1:p1+w], a_trial[p2:p2+w] = a_trial[p2:p2+w].copy(), a_trial[p1:p1+w].copy()
        else:
            # Spike: increase one position, decrease another
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            delta = 0.05 * a_trial.mean()
            a_trial[i1] += delta
            a_trial[i2] -= delta
        
        a_trial = project_nonneg_sum1(a_trial)
        trial_obj = compute_obj(a_trial)
        
        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Linearized LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set management with successive linearization of the max-of-quadratics
objective, combined with line search, proximal gradient refinement, multi-length
exploration, and simulated annealing for escaping local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_active_set(conv, max_val, frac=0.05, min_count=5, max_count=200):
    """Get indices where conv is within frac of max."""
    threshold = max_val * (1.0 - frac)
    active = np.where(conv >= threshold)[0]
    if len(active) < min_count:
        active = np.argsort(conv)[-min_count:]
    if len(active) > max_count:
        active = np.argsort(conv)[-max_count:]
    return active


def solve_linearized_lp(a, active_k, time_limit=5.0):
    """
    Solve linearized LP:
      min t
      s.t. sum_i (2*a[k-i]) * a[i] - t <= conv[k]  for k in active_k
           sum(a) = 1, a >= 0, t >= 0
    """
    n = len(a)
    conv = compute_conv(a)
    m = len(active_k)

    if m == 0:
        return None, None

    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    convvals = conv[active_k]

    for idx, k in enumerate(active_k):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_end > i_start:
            indices = np.arange(i_start, i_end)
            a_indices = k_int - indices
            A_ub[idx, indices] = 2.0 * a[a_indices]
        A_ub[idx, n] = -1.0
        b_ub[idx] = convvals[idx]

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status in (0, 3):  # optimal or time limit with feasible solution
            a_new = np.maximum(res.x[:n], 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                return a_new, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_descent(a, max_iters=200, time_budget=60.0, active_frac=0.05,
               active_shrink=0.9, verbose=False, max_active=150):
    """
    LP-based descent with adaptive active set and line search.
    """
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    current_frac = active_frac

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        active_k = get_active_set(conv, max_conv, frac=current_frac,
                                   min_count=5, max_count=max_active)

        a_lp, t_lp = solve_linearized_lp(a, active_k, time_limit=3.0)
        if a_lp is None:
            stall += 1
            if stall > 8:
                break
            # Fallback: subgradient step on top peak
            k_star = int(np.argmax(conv))
            grad = np.zeros(n)
            i_start = max(0, k_star - n + 1)
            i_end = min(n, k_star + 1)
            if i_end > i_start:
                indices = np.arange(i_start, i_end)
                a_indices = k_star - indices
                grad[indices] = 2.0 * a[a_indices]
            grad -= np.mean(grad)
            a = project_nonneg_sum1(a - 0.01 * grad)
            continue

        # Line search
        found_improvement = False
        best_t_obj = best_obj
        for t in [0.3, 0.5, 0.7, 1.0, 0.1, 0.2]:
            a_try = (1.0 - t) * a + t * a_lp
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                a = a_try.copy()
                found_improvement = True

        if found_improvement:
            if best_t_obj < best_obj:
                best_obj = best_t_obj
                best_a = a.copy()
                stall = 0
                # Shrink active set as we converge
                current_frac = max(0.01, current_frac * active_shrink)
                if verbose and it % 30 == 0:
                    print(f"      LP iter {it}: obj={best_obj:.8f}, frac={current_frac:.4f}")
            else:
                stall += 1
        else:
            stall += 1
            # Widen active set if stuck
            current_frac = min(0.15, current_frac * 1.3)

        if stall > 20:
            break

    return best_a, best_obj


def smoothed_max_gradient(a, tau=0.02, top_k=20):
    """Compute gradient of smoothed max using softmax over top-k conv values."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    top_indices = np.argsort(conv)[-top_k:]
    top_vals = conv[top_indices]

    shifted = (top_vals - max_conv) / tau
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(top_k)
        weights[0] = 1.0
    else:
        weights /= wsum

    grad = np.zeros(n)
    for idx, k in enumerate(top_indices):
        if weights[idx] < 1e-10:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_end > i_start:
            indices = np.arange(i_start, i_end)
            a_indices = k_int - indices
            grad[indices] += weights[idx] * 2.0 * a[a_indices]

    return grad, max_conv


def proximal_gradient(a, max_iters=500, time_budget=30.0,
                      tau_start=0.05, tau_end=0.001, lr_init=0.02, verbose=False):
    """Proximal gradient with adaptive softmax temperature."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)
        top_k = max(10, min(n // 5, 50))
        grad, _ = smoothed_max_gradient(a, tau=tau, top_k=top_k)
        grad_centered = grad - np.mean(grad)

        a_new = project_nonneg_sum1(a - lr * grad_centered)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.9

    return best_a, best_obj


def simulated_annealing(a, rng, max_iters=200, T_init=0.05, T_min=0.001, time_budget=30.0):
    """Simulated annealing with targeted moves."""
    n = len(a)
    cur_a = project_nonneg_sum1(a.astype(np.float64))
    cur_obj = compute_obj(cur_a)
    best_a = cur_a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (it / max_iters)
        a_trial = cur_a.copy()
        move_type = rng.integers(0, 4)

        if move_type == 0:
            w = max(5, n // 20)
            pos = rng.integers(0, n - w)
            a_trial[pos:pos + w] += rng.normal(0, 0.08, w)
        elif move_type == 1:
            grad, _ = smoothed_max_gradient(cur_a, tau=0.02, top_k=10)
            grad -= np.mean(grad)
            a_trial -= 0.05 * grad + rng.normal(0, 0.008, n)
        elif move_type == 2:
            w = max(10, n // 15)
            p1 = rng.integers(0, n - w)
            p2 = rng.integers(0, n - w)
            if abs(int(p1) - int(p2)) >= w:
                tmp = a_trial[p1:p1 + w].copy()
                a_trial[p1:p1 + w] = a_trial[p2:p2 + w]
                a_trial[p2:p2 + w] = tmp
        else:
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            delta = 0.03 * cur_a.mean()
            a_trial[i1] += delta
            a_trial[i2] -= delta

        a_trial = project_nonneg_sum1(a_trial)
        trial_obj = compute_obj(a_trial)

        delta_obj = trial_obj - cur_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def frequency_compress(a, max_iters=100, time_budget=20.0, alpha=0.8):
    """Frequency domain: compress power spectrum to reduce time-domain peaks."""
    n = len(a)
    a = project_nonneg_sum1(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        ps = np.abs(fa) ** 2
        ps_max = np.max(ps)
        if ps_max < 1e-20:
            break
        ps_norm = np.maximum(ps / ps_max, 1e-15)
        ps_comp = ps_norm ** alpha
        phase = np.angle(fa)
        mag = np.sqrt(ps_comp * ps_max)
        fa_new = mag * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            alpha = max(0.5, alpha * 0.97)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining linearized LP,
    proximal gradient, frequency-domain compression, simulated annealing,
    and multi-length exploration to minimize the self-convolution peak ratio.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    parent = try_load_parent()
    if parent is not None:
        parent = project_nonneg_sum1(parent)
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        best_score = parent_obj
        best_seq = parent.copy()
    else:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=700): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n={n_best}")

    # === Phase 1: LP-based descent (35% budget) ===
    print("\n=== Phase 1: LP Descent ===")
    p1_budget = min(500.0, (deadline - start_time) * 0.35)
    a_work = best_seq.copy()

    for round_idx in range(6):
        if time.monotonic() > start_time + p1_budget:
            break
        remaining = start_time + p1_budget - time.monotonic()
        if remaining < 10:
            break

        af = max(0.02, 0.08 * (0.75 ** round_idx))
        a_new, obj_new = lp_descent(a_work, max_iters=60,
                                     time_budget=min(remaining * 0.85, 70),
                                     active_frac=af,
                                     active_shrink=0.92,
                                     verbose=True, max_active=120)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  LP round {round_idx}: obj={best_score:.8f}")
        a_work = a_new

    # Quick frequency compression
    remaining = start_time + p1_budget - time.monotonic()
    if remaining > 10:
        a_fd, obj_fd = frequency_compress(a_work, max_iters=80,
                                           time_budget=min(remaining * 0.5, 15), alpha=0.85)
        if obj_fd < best_score:
            best_score = obj_fd
            best_seq = a_fd.copy()
            print(f"  Freq compress: obj={best_score:.8f}")
            a_work = a_fd

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Proximal gradient + LP refinement (15% budget) ===
    print("\n=== Phase 2: PG + LP Refinement ===")
    p2_budget = min(200.0, (deadline - start_time) * 0.15)

    a_pg, obj_pg = proximal_gradient(best_seq, max_iters=400,
                                      time_budget=min(p2_budget * 0.4, 30),
                                      tau_start=0.04, tau_end=0.002, lr_init=0.015)
    if obj_pg < best_score:
        best_score = obj_pg
        best_seq = a_pg.copy()
        print(f"  PG: obj={best_score:.8f}")

    a_lp2, obj_lp2 = lp_descent(best_seq, max_iters=50,
                                 time_budget=min(p2_budget * 0.4, 30),
                                 active_frac=0.03, active_shrink=0.95,
                                 verbose=False, max_active=80)
    if obj_lp2 < best_score:
        best_score = obj_lp2
        best_seq = a_lp2.copy()
        print(f"  LP refine: obj={best_score:.8f}")

    # === Phase 3: Multi-length exploration (15% budget) ===
    print("\n=== Phase 3: Multi-length Exploration ===")
    p3_budget = min(220.0, (deadline - start_time) * 0.15)
    n_current = len(best_seq)
    n_tries = []
    for delta in [-100, -50, -30, 30, 50, 100, 150, -200, 200]:
        nt = n_current + delta
        if 100 <= nt <= 3000:
            n_tries.append(nt)
    n_tries = list(dict.fromkeys(n_tries))
    per_n_budget = max(5.0, p3_budget / max(1, len(n_tries)))

    for n_try in n_tries:
        if time.monotonic() > start_time + p3_budget:
            break
        a_res = project_nonneg_sum1(resample_sequence(best_seq, n_try))
        a_opt, obj_opt = lp_descent(a_res, max_iters=30,
                                     time_budget=min(per_n_budget * 0.6, 18),
                                     active_frac=0.05, verbose=False, max_active=100)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: SA + local optimization (15% budget) ===
    print("\n=== Phase 4: SA + Local Opt ===")
    print(f"Before SA: {best_score:.8f}")
    p4_budget = min(200.0, (deadline - start_time) * 0.15)
    n_sa = max(3, int(p4_budget / 25))

    for sa_idx in range(n_sa):
        if time.monotonic() > start_time + p4_budget:
            break
        remaining = start_time + p4_budget - time.monotonic()
        if remaining < 15:
            break

        T_init = 0.04 + 0.015 * sa_idx
        a_sa, obj_sa = simulated_annealing(best_seq, rng, max_iters=120,
                                            T_init=T_init, T_min=0.001,
                                            time_budget=min(remaining * 0.4, 18))
        a_lo, obj_lo = lp_descent(a_sa, max_iters=25,
                                   time_budget=min(remaining * 0.35, 12),
                                   active_frac=0.03, verbose=False, max_active=80)
        if obj_lo < best_score:
            best_score = obj_lo
            best_seq = a_lo.copy()
            print(f"  SA {sa_idx}: NEW BEST {best_score:.8f}")
        elif obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA {sa_idx} (no local): {best_score:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print("\n=== Phase 5: Final Polish ===")
    print(f"Before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    n_final = len(best_seq)

    if remaining5 > 15 and best_seq is not None:
        a_work = best_seq.copy()
        polish_rounds = min(25, int(remaining5 / 8))

        for pr in range(polish_rounds):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            # LP with tight active set
            a_p1, obj_p1 = lp_descent(a_work, max_iters=25,
                                       time_budget=min(rem * 0.25, 15),
                                       active_frac=0.015, active_shrink=0.95,
                                       verbose=False, max_active=60)
            if obj_p1 < best_score:
                best_score = obj_p1
                best_seq = a_p1.copy()
                a_work = a_p1.copy()
                print(f"  Polish LP r{pr}: {best_score:.8f}")

            # PG with small step
            remain = deadline - time.monotonic()
            if remain > 8:
                a_p2, obj_p2 = proximal_gradient(a_work, max_iters=80,
                                                  time_budget=min(remain * 0.2, 10),
                                                  tau_start=0.008, tau_end=0.001,
                                                  lr_init=0.008)
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_work = a_p2.copy()
                    print(f"  Polish PG r{pr}: {best_score:.8f}")

            # Tiny perturbation + re-optimize (escape local min)
            remain = deadline - time.monotonic()
            if remain > 12 and pr % 3 == 2:
                a_pert = a_work * np.exp(rng.normal(0, 0.008, n_final))
                a_pert = project_nonneg_sum1(a_pert)
                a_pert, obj_pert = lp_descent(a_pert, max_iters=20,
                                               time_budget=min(remain * 0.3, 12),
                                               active_frac=0.02, verbose=False, max_active=80)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_work = a_pert.copy()
                    print(f"  Polish pert r{pr}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
