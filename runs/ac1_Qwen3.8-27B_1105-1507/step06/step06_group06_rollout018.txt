"""
Optimized step function search using FFT-based gradient descent with adaptive
softmax weights, strategic LP polishing, and multi-length exploration.
Key innovation: O(n log n) gradient computation via FFT-based Hankel
matrix-vector product, enabling thousands of gradient steps per second.
Combines fast structured gradients (shifted-a differences) with LP for
global improvements, using adaptive tau schedules and momentum.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def hankel_matvec(w, a):
    """
    Compute c[i] = sum_j a[j]*w[i+j] for i=0..n-1 using FFT.
    w has length >= 2n-1, a has length n. O(n log n).
    """
    n = len(a)
    a_rev = a[::-1]
    size = next_fast_len(n + len(w) - 1)
    fa = rfft(a_rev, size)
    fw = rfft(w, size)
    conv = irfft(fa * fw, size)
    return conv[n - 1:2 * n - 1]


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'feasibility_tolerance': 1e-9,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 2000:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    except Exception:
        pass
    return None, None


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

            last_improve_obj = best_obj
            last_improve_iter = it
            stall_count = 0

        # Adaptive: if stuck too long, increase tau temporarily
        if stall_count > 500 and (it - last_improve_iter) > 500:
            lr = lr_start * 0.5
            momentum = 0.7  # Reduce momentum to escape
            stall_count = 0
            # Perturb
            a = a + 0.01 * np.random.standard_normal(n)
            a = project_nonneg_sum1(a)
            v = np.zeros(n)

        # Decay learning rate slowly
        if it % 200 == 0:
            lr = lr_start * (0.95 ** (it // 200))

    return best_a, best_obj


def lp_polish(a, max_iters=20, time_budget=30.0):
    """LP-based polishing: alternate between LP solve and blend."""
    n = len(a)
    if n > 1500:
        return a, compute_obj(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        if it > 0 and time.monotonic() - start > time_budget * 0.8:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=min(12, max(3, time_budget / max_iters)))
        if g is None:
            break

        # Line search over blend parameter
        best_t_obj = best_obj
        best_t_val = 0.0
        best_a_local = best_a.copy()

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
        else:
            # Try min-peak LP as alternative
            g2, M2 = solve_lp_minpeak(a, time_limit=min(8, max(2, time_budget / (max_iters + 1))))
            if g2 is not None:
                for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    a_try = (1.0 - t) * a + t * g2
                    a_try = project_nonneg_sum1(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break
                else:
                    break

    return best_a, best_obj


def mixed_optimize(a, time_budget=120.0, verbose=False):
    """
    Mixed optimization: alternating fast gradient descent and LP polishing.
    This is the core optimization loop.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    grad_budget_frac = 0.6
    lp_budget_frac = 0.4

    n_grad_blocks = max(5, int(time_budget / 15))
    n_lp_blocks = max(2, int(time_budget / 30))

    total_blocks = n_grad_blocks + n_lp_blocks
    block_idx = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Alternate: more gradient blocks than LP blocks
        if block_idx % (n_lp_blocks + 1) < n_lp_blocks:
            # Gradient block
            block_time = min(15.0, remaining * 0.3)
            a_new, obj_new = gradient_descent_fast(
                a, max_iters=5000, time_budget=block_time,
                tau_start=0.12, tau_end=0.015,
                lr_start=0.25, momentum=0.85)
        else:
            # LP block
            block_time = min(25.0, remaining * 0.4)
            if n <= 1500:
                a_new, obj_new = lp_polish(a, max_iters=10, time_budget=block_time)
            else:
                a_new, obj_new = gradient_descent_fast(
                    a, max_iters=5000, time_budget=block_time,
                    tau_start=0.10, tau_end=0.01,
                    lr_start=0.2, momentum=0.88)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [mixed] block {block_idx}: obj={best_obj:.8f} (t={elapsed:.0f}s)")
        else:
            # No improvement - try perturbation
            a_pert = a + 0.02 * np.random.standard_normal(n)
            a_pert = project_nonneg_sum1(a_pert)
            a_new2, obj_new2 = gradient_descent_fast(
                a_pert, max_iters=2000, time_budget=min(10.0, remaining * 0.2),
                tau_start=0.15, tau_end=0.02,
                lr_start=0.3, momentum=0.8)
            if obj_new2 < best_obj:
                best_obj = obj_new2
                best_a = a_new2.copy()
                a = a_new2.copy()

        block_idx += 1

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    x = np.linspace(0, 1, n)
    tri = np.sin(np.pi * x)
    inits.append(tri / tri.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-x ** 2 / 3)
    inits.append(gauss / gauss.sum())

    # Bimodal
    x = np.linspace(0, 1, n)
    bimodal = np.exp(-((x - 0.3) ** 2) / 0.015) + np.exp(-((x - 0.7) ** 2) / 0.015)
    inits.append(bimodal / bimodal.sum())

    # Random smooth
    rnd = rng.random(n)
    # Smooth via convolution
    kernel = np.ones(10) / 10
    rnd_smooth = np.convolve(rnd, kernel, mode='same')
    inits.append(rnd_smooth / rnd_smooth.sum())

    # Cosine window
    x = np.linspace(0, np.pi, n)
    cos_win = np.cos(x / 2) ** 2
    inits.append(cos_win / cos_win.sum())

    # Raised cosine
    x = np.linspace(0, 1, n)
    raised = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(raised / raised.sum())

    if parent is not None:
        # Resampled parent
        inits.append(project_nonneg_sum1(resample_sequence(parent, n)))
        # Perturbed parent
        pert = resample_sequence(parent, n)
        pert = pert * np.exp(rng.normal(0, 0.05, n))
        inits.append(project_nonneg_sum1(pert))

    return inits


def peak_chase_fast(a, n_steps=100, time_budget=10.0):
    """
    Fast peak-chasing: focus gradient on the single maximum peak.
    Very fast since gradient is just a shifted copy of a.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute FFT size for convolution
    N_conv = next_fast_len(2 * n - 1)

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        # Find the peak
        fa = rfft(a, N_conv)
        conv = irfft(fa * fa, N_conv)[:2 * n - 1]
        k_star = int(np.argmax(conv))
        cmax = conv[k_star]

        # Gradient of (a*a)[k_star] is 2*a[k_star - i] (for valid indices)
        # For the objective 2n * (a*a)[k_star] with sum=1:
        # grad_i = 2n * 2 * a[k_star - i] - 2n * cmax * 1 (centered)
        # Simplified: grad = 4n * (a_shifted - cmax)
        a_shifted = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                a_shifted[i] = a[j]

        grad = a_shifted - cmax  # Direction to reduce peak
        grad = grad - np.mean(grad)  # Ensure sum-zero (keeps sum=1)

        # Try several step sizes
        improved = False
        for ss in [0.05, 0.02, 0.01, 0.005, 0.002]:
            a_try = a - ss * n * grad
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and step > 20:
            # Try moving toward flatter distribution
            a_try = a * (1 - 0.01) + 0.01 * np.ones(n) / n
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
            elif step > 50:
                break

    return best_a, best_obj


def multi_length_search(a_best, obj_best, rng, time_budget=120.0):
    """Try resampling to different lengths and optimizing."""
    n_best = len(a_best)
    n_options = [n_best - 50, n_best - 100, n_best + 50, n_best + 100,
                 n_best - 150, n_best + 150, n_best - 200, n_best + 200]
    n_options = sorted(set(n for n in n_options if 100 <= n <= 2500))

    per_n_budget = time_budget / max(1, len(n_options))

    for n_try in n_options:
        if per_n_budget < 3:
            break
        a_resampled = project_nonneg_sum1(resample_sequence(a_best, n_try))
        obj_cur = compute_obj(a_resampled)

        # Quick optimization
        a_opt, obj_opt = gradient_descent_fast(
            a_resampled, max_iters=3000, time_budget=per_n_budget * 0.6,
            tau_start=0.12, tau_end=0.02, lr_start=0.25, momentum=0.85)

        if n_try <= 1200 and per_n_budget > 10:
            a_opt2, obj_opt2 = lp_polish(a_opt, max_iters=5,
                                         time_budget=per_n_budget * 0.3)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < obj_best:
            obj_best = obj_opt
            a_best = a_opt.copy()
            print(f"    Multi-length: n={n_try} improved to {obj_best:.8f}")

    return a_best, obj_best


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses FFT-based gradient descent (O(n log n) per step)
    combined with LP polishing, multi-length exploration, and adaptive
    tau/momentum schedules to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent = project_nonneg_sum1(parent)
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    print(f"Starting: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: Multi-start screening (10% time) ===
    print("\n=== Phase 1: Multi-start Screening ===")
    phase1_budget = min(120.0, (deadline - start_time) * 0.08)
    n_parent = len(best_seq)

    n_options = sorted(set([
        n_parent, max(200, n_parent - 100), max(200, n_parent - 50),
        n_parent + 50, n_parent + 100, min(2000, n_parent + 200),
        max(200, n_parent - 200), min(2500, n_parent + 300)
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]

    candidates = []
    per_init_time = max(2.0, phase1_budget / (len(n_options) * 4))

    for n_try in n_options:
        if time.monotonic() > start_time + phase1_budget:
            break
        inits = generate_initializations(n_try, rng, parent)
        for init in inits[:4]:  # Limit inits per n
            if time.monotonic() > start_time + phase1_budget:
                break
            a_quick, obj_quick = gradient_descent_fast(
                init, max_iters=3000, time_budget=per_init_time,
                tau_start=0.12, tau_end=0.03, lr_start=0.25, momentum=0.85)
            if obj_quick < np.inf:
                candidates.append((obj_quick, a_quick))

    candidates.sort(key=lambda x: x[0])
    if candidates:
        print(f"  Screening: {len(candidates)} candidates, best={candidates[0][0]:.8f}")
        if candidates[0][0] < best_score:
            best_score = candidates[0][0]
            best_seq = candidates[0][1].copy()
            print(f"  New best from screening: {best_score:.8f}")

    # === Phase 2: Intensive mixed optimization (55% time) ===
    print(f"\n=== Phase 2: Intensive Optimization (best={best_score:.8f}) ===")
    phase2_budget = min(800.0, (deadline - start_time) * 0.55)
    phase2_start = time.monotonic()

    # Work on the best candidate with mixed gradient + LP
    a_work = project_nonneg_sum1(best_seq.copy())
    n_work = len(a_work)

    # Divide into sub-phases with progressively tighter tau
    n_sub = max(3, int(phase2_budget / 60))
    sub_budget = phase2_budget / n_sub

    for sub in range(n_sub):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        remaining = phase2_start + phase2_budget - time.monotonic()
        if remaining < 10:
            break

        # Progress-based parameters
        progress = sub / max(1, n_sub - 1)
        tau_s = 0.15 * (1 - progress) + 0.02 * progress
        tau_e = 0.03 * (1 - progress) + 0.005 * progress
        lr_s = 0.3 * (1 - progress) + 0.1 * progress
        mom = 0.85 + 0.05 * progress

        # Gradient phase
        grad_time = min(sub_budget * 0.6, remaining * 0.5)
        a_new, obj_new = gradient_descent_fast(
            a_work, max_iters=10000, time_budget=grad_time,
            tau_start=tau_s, tau_end=tau_e,
            lr_start=lr_s, momentum=mom)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Sub {sub}: gradient improved to {best_score:.8f}")
        a_work = a_new.copy()

        # LP phase (if n is small enough)
        lp_remaining = min(sub_budget * 0.4, remaining * 0.4)
        if n_work <= 1500 and lp_remaining > 5:
            a_lp, obj_lp = lp_polish(a_work, max_iters=8, time_budget=lp_remaining)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Sub {sub}: LP improved to {best_score:.8f}")
            a_work = a_lp.copy()

        # Peak chase (very fast, focused on single peak)
        pc_remaining = min(sub_budget * 0.2, remaining * 0.3)
        if pc_remaining > 3:
            a_pc, obj_pc = peak_chase_fast(a_work, n_steps=200,
                                           time_budget=pc_remaining)
            if obj_pc < best_score:
                best_score = obj_pc
                best_seq = a_pc.copy()
                print(f"  Sub {sub}: peak-chase improved to {best_score:.8f}")
            a_work = a_pc.copy()

    # === Phase 3: Perturbation and escape (15% time) ===
    print(f"\n=== Phase 3: Perturbation & Escape (best={best_score:.8f}) ===")
    phase3_budget = min(200.0, (deadline - start_time) * 0.15)
    phase3_start = time.monotonic()
    n_best = len(best_seq)

    n_pert = max(3, int(phase3_budget / 20))
    per_pert = phase3_budget / n_pert

    for p in range(n_pert):
        if time.monotonic() > phase3_start + phase3_budget:
            break
        if per_pert < 3:
            break

        # Different perturbation strategies
        strategy = p % 4

        if strategy == 0:
            # Gaussian perturbation
            a_pert = best_seq * np.exp(rng.normal(0, 0.03, n_best))
        elif strategy == 1:
            # Window noise
            a_pert = best_seq.copy()
            wsize = max(20, n_best // 10)
            wstart = rng.integers(0, n_best - wsize)
            a_pert[wstart:wstart + wsize] += rng.normal(0, 0.1, wsize) * a_pert.mean()
        elif strategy == 2:
            # Partial randomization
            a_pert = best_seq.copy()
            frac = 0.1 + 0.05 * p
            n_rand = int(n_best * frac)
            idx = rng.choice(n_best, size=n_rand, replace=False)
            a_pert[idx] = rng.dirichlet(np.ones(n_rand)) * a_pert[idx].sum()
        else:
            # Resample to slightly different length and back
            new_n = n_best + rng.integers(-50, 50)
            new_n = max(100, min(3000, new_n))
            a_pert = resample_sequence(best_seq, new_n)
            if len(a_pert) != n_best:
                a_pert = resample_sequence(a_pert, n_best)

        a_pert = project_nonneg_sum1(a_pert)

        # Optimize the perturbed version
        a_opt, obj_opt = gradient_descent_fast(
            a_pert, max_iters=5000, time_budget=per_pert * 0.6,
            tau_start=0.12, tau_end=0.02, lr_start=0.25, momentum=0.85)

        if n_best <= 1200:
            a_opt2, obj_opt2 = lp_polish(a_opt, max_iters=5,
                                         time_budget=per_pert * 0.3)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Pert {p} (strat {strategy}): improved to {best_score:.8f}")

    # === Phase 4: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-length Exploration (best={best_score:.8f}) ===")
    phase4_budget = min(150.0, (deadline - start_time) * 0.10)
    if phase4_budget > 10:
        best_seq, best_score = multi_length_search(
            best_seq, best_score, rng, time_budget=phase4_budget)

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 10 and best_seq is not None:
        n_final = len(best_seq)
        a_final = project_nonneg_sum1(best_seq.copy())

        # Multiple polishing rounds with decreasing tau
        n_rounds = max(3, int(remaining5 / 20))
        for r in range(n_rounds):
            rem = deadline - time.monotonic()
            if rem < 5:
                break

            # Tight gradient
            tau_s = 0.05 + 0.02 * r
            tau_e = 0.005 + 0.002 * r
            a_td, obj_td = gradient_descent_fast(
                a_final, max_iters=10000, time_budget=min(rem * 0.4, 30.0),
                tau_start=tau_s, tau_end=tau_e,
                lr_start=0.15, momentum=0.9)
            if obj_td < best_score:
                best_score = obj_td
                best_seq = a_td.copy()
                a_final = a_td.copy()
                print(f"  Final grad r{r}: {best_score:.8f}")

            # LP if possible
            rem = deadline - time.monotonic()
            if rem > 10 and n_final <= 1500:
                a_lp, obj_lp = lp_polish(a_final, max_iters=5,
                                         time_budget=min(rem * 0.4, 20.0))
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_final = a_lp.copy()
                    print(f"  Final LP r{r}: {best_score:.8f}")

            # Peak chase
            rem = deadline - time.monotonic()
            if rem > 5:
                a_pc, obj_pc = peak_chase_fast(a_final, n_steps=100,
                                               time_budget=min(rem * 0.3, 10.0))
                if obj_pc < best_score:
                    best_score = obj_pc
                    best_seq = a_pc.copy()
                    a_final = a_pc.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
