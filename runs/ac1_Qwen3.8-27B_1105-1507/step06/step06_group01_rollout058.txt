def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """FFT-based linear cross-convolution of a and b (same length)."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho) > cv
    if not cond.any():
        return np.ones(n) / n
    rho_idx = int(np.argmax(~cond))
    if (~cond).any():
        theta = (cv[rho_idx] - 1.0) / (rho_idx + 1.0)
    else:
        theta = (cv[-1] - 1.0) / n
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def run_fista(a_init, tau_start, tau_end, max_iters, deadline, lr=0.3):

    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:

    best_M = fast_conv(a).max()
    
    t_mom = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters * 0.8))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        step = lr / (1.0 + 0.01 * tau)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new if t_mom > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t_mom = t_new
        it += 1
        
        if it % 10 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def lp_best_response_full(a, max_time_per_lp=5.0):

    Returns best response x (normalized) or None.
    """
    n = len(a)
    
    # Build constraint matrix A where A[k,j] = a[k-j] if 0 <= k-j < n
    # Shape: (2n-1, n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            m = k - j
            if 0 <= m < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[m])
    
    from scipy.sparse import csr_matrix
    A_conv = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: min t
    # s.t. A_conv @ x - t <= 0 (2n-1 constraints)
    #      sum(x) = 1
    #      x >= 0
    # Variables: [x_0, ..., x_{n-1}, t]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: A_conv @ x - t <= 0
    # As a row: [A_conv[k,:], -1] @ [x, t] <= 0
    A_ub = np.hstack((A_conv.toarray(), -np.ones((2 * n - 1, 1))))
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # t is free
    
    t_start = time.monotonic()
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': max_time_per_lp, 'presolve': True})
        if time.monotonic() - t_start > max_time_per_lp:
            return None
        if r.success and r.x is not None:
            x = np.maximum(r.x[:n], 0.0)
            s = x.sum()
            if s > 1e-10:
                return x / s
    except Exception:
        pass
    
    return None


def lp_best_response_topk(a, k=50, max_time_per_lp=3.0):
    """LP best response using only top-k constraints (faster for large n)."""
    n = len(a)
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:k]
    
    # Build constraint matrix for only top-k rows
    rows = []
    cols = []
    vals = []
    for idx, k_val in enumerate(top_indices):
        for j in range(max(0, k_val - n + 1), min(n, k_val + 1)):
            m = k_val - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    from scipy.sparse import csr_matrix
    A_conv = csr_matrix((vals, (rows, cols)), shape=(k, n))
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack((A_conv.toarray(), -np.ones((k, 1))))
    b_ub = np.zeros(k)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': max_time_per_lp, 'presolve': True})
        if r.success and r.x is not None:
            x = np.maximum(r.x[:n], 0.0)
            s = x.sum()
            if s > 1e-10:
                return x / s
    except Exception:
        pass
    
    return None


def run_lp_iterations(a_init, max_iters, deadline, use_full=True, k=50,
                      step_values=None):
    """Run LP best-response iterations with adaptive step size."""
    if step_values is None:
        step_values = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
    
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        if use_full and n <= 1024:
            x_star = lp_best_response_full(a, max_time_per_lp=5.0)
        else:
            x_star = lp_best_response_topk(a, k=k, max_time_per_lp=3.0)
        
        if x_star is None:
            no_improve += 2
            if no_improve > 10:
                break
            continue
        
        # Check if x_star is close to a (convergence)
        if np.abs(x_star - a).max() < 1e-6:
            no_improve += 3
            if no_improve > 5:
                break
            continue
        
        # Line search: find best step
        improved = False
        for t in step_values:
            a_trial = (1.0 - t) * a + t * x_star
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break
    
    return best_a


def mirror_descent_iter(a, step):
    """One step of multiplicative weights / mirror descent."""
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    
    # Subgradient at peak: grad[i] = 2*a[k_star - i] if valid
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    
    grad_tan = grad - grad.mean()
    
    # Multiplicative update
    log_w = -step * grad_tan
    log_w -= log_w.max()
    a_new = a * np.exp(log_w)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    
    return a_new


def run_mirror_descent(a_init, max_iters, deadline, step_init=0.3, step_decay=0.999):
    """Mirror descent with decaying step size."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    step = step_init
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a = mirror_descent_iter(a, step)
        step *= step_decay
        
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100 and step < 0.01:
                break
    
    return best_a


def coordinate_descent_1d(a, i, b):
    """Exact 1D optimization for coordinate i on the simplex.
    
    We change a[i] to x, then rescale all other coordinates to maintain sum=1.
    Objective: max of conv after this change.
    """
    n = len(a)
    x_cur = a[i]
    
    # If we set a[i] = x, the other coordinates are scaled by (1-x)/(1-x_cur)
    # to maintain sum = 1 (assuming sum was 1).
    # Actually, let's think of it differently:
    # New a: a_new[i] = x, a_new[j] = a[j] * (1-x) / (1-x_cur) for j != i
    # (This maintains sum = 1 if the old sum was 1.)
    
    # The convolution changes in a complex way because all other coordinates change too.
    # For simplicity, let's use a different approach:
    # Fix all other coordinates, optimize a[i] freely, then renormalize.
    
    # c[k] = sum_j a[j]*a[k-j]
    # After change: a_new[i] = x, others unchanged (before renormalization)
    # Affected k: those where i is a valid index in the sum
    # k in [i, i+n-1] (and also k in [k-i>=0] -> k>=i, k-i<=n-1 -> k<=i+n-1)
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # For each affected k, c_new[k] as function of x:
    # c_new[k] = c_old[k] + delta_k(x)
    # where delta depends on how a[i] appears in c[k]
    
    # Precompute: for each k in [k_lo, k_hi], what's the coefficient structure?
    # c[k] = sum_j a[j]*a[k-j]
    # Terms involving a[i]:
    #   j=i: a[i]*a[k-i] (if k-i valid)
    #   j=k-i: a[k-i]*a[i] (if k-i valid and k-i != i)
    # If k-i == i (k=2i): one term a[i]^2
    # If k-i != i and both valid: two terms, each a[i]*a[k-i]
    
    # So c_new[k](x) = c_base[k] + coef[k] * x + quad[k] * x^2
    # where:
    #   if k == 2i: c_base[k] = c[k] - a[i]^2, coef[k] = 0, quad[k] = 1
    #   elif 0 <= k-i < n: c_base[k] = c[k] - 2*a[i]*a[k-i], coef[k] = 2*a[k-i], quad[k] = 0
    #   else: c_base[k] = c[k], coef[k] = 0, quad[k] = 0
    
    c_base = np.zeros(k_hi - k_lo + 1)
    coef = np.zeros(k_hi - k_lo + 1)
    quad = np.zeros(k_hi - k_lo + 1)
    
    for idx, k in enumerate(range(k_lo, k_hi + 1)):
        c_base[idx] = b[k]
        if k == 2 * i:
            c_base[idx] -= x_cur * x_cur
            quad[idx] = 1.0
        else:
            j = k - i
            if 0 <= j < n:
                c_base[idx] -= 2.0 * x_cur * a[j]
                coef[idx] = 2.0 * a[j]
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    def obj_at(x_test):
        """Max convolution value when a[i] = x_test (before renormalization)."""
        vals = c_base + coef * x_test + quad * x_test * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        return m
    
    # After setting a[i] = x, we renormalize: divide all by S_new = S_old - x_cur + x
    # But since we're on the simplex (S_old = 1), S_new = 1 - x_cur + x
    # The max of conv scales by 1/S_new^2 after normalization.
    # So the actual objective is: obj_at(x) / (1 - x_cur + x)^2
    # (times the constant 2n which we ignore for minimization)
    
    # Actually, since J = 2n * max_conv / sum^2, and we're working on the simplex
    # (sum=1), the objective is just max_conv. But after changing a[i] to x,
    # the sum is no longer 1. So we need to account for renormalization.
    # 
    # Let's just work with the unnormalized version and minimize:
    # f(x) = max_conv_after_change / (sum_after_change)^2
    # = obj_at(x) / (1 - x_cur + x)^2
    # (assuming original sum = 1)
    
    def objective(x_test):
        s_new = 1.0 - x_cur + x_test
        if s_new < 0.01:
            return float('inf')
        return obj_at(x_test) / (s_new * s_new)
    
    # Golden section search on [0, hi]
    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, 0.5, 2.0 / n * 10)
    
    # Make sure hi is large enough
    for _ in range(5):
        if objective(hi) < objective(hi * 0.5) and hi < 10.0:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = objective(a_pt)
    fb = objective(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = objective(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = objective(b_pt)
    
    x_opt = (lo + hi) / 2.0
    return max(0.0, min(x_opt, 1.0 - 1e-10))


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with exact 1D optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b)
            if abs(x_opt - a[i]) > 1e-10:
                # Update a and b
                old_x = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - old_x * old_x)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old_x)
                sweep_improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b = fast_conv(a)  # Recompute to avoid drift
        
        M = b.max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 2
        if no_improve > 20:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        for sigma in [0.01, 0.03, 0.05, 0.1]:
            p_n = np.maximum(p + rng.normal(0, sigma, n), 1e-6)
            inits.append(p_n / p_n.sum())
    
    inits.append(np.ones(n) / n)
    
    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.3 + 0.7 * np.cos(2 * np.pi * freq * t) ** 2
        inits.append(cos_w / cos_w.sum())
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # "Anti-peak" init: try to make convolution flat from the start
    a_anti = np.ones(n) / n
    for _ in range(50):
        b_anti = fast_conv(a_anti)
        k_star = int(np.argmax(b_anti))
        # Reduce a where it contributes to the peak
        corr = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_anti[j]
        a_anti = a_anti - 0.05 * (corr - corr.mean())
        a_anti = np.maximum(a_anti, 0.001)
        a_anti = a_anti / a_anti.sum()
    inits.append(a_anti.copy())
    
    # Random smooth (different frequency)
    ksize2 = max(3, n // 8)
    kernel2 = np.ones(ksize2) / ksize2
    smooth2 = np.convolve(rng.uniform(0, 1, n), kernel2, mode='same')
    smooth2 = np.maximum(smooth2, 0.01)
    inits.append(smooth2 / smooth2.sum())
    
    return inits


def run_combined_optimization(a0, n, deadline, seed=0):
    """Run combined optimization: LP -> mirror descent -> coordinate descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / n
    else:
        a = a / s
    
    t_start = time.monotonic()
    duration = max(1.0, deadline - t_start)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Stage 1: LP iterations (50% of time)
    t1 = t_start + 0.5 * duration
    if time.monotonic() < t1:
        use_full = (n <= 1024)
        a_lp = run_lp_iterations(best_a, 500, t1, use_full=use_full, k=50)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best_a = a_lp.copy()
        print(f"    LP: M={best_M:.10f}", flush=True)
    
    # Stage 2: Mirror descent (25% of time)
    t2 = t_start + 0.75 * duration
    if time.monotonic() < t2:
        a_md = run_mirror_descent(best_a, 10000, t2, step_init=0.2, step_decay=0.9995)
        M_md = fast_conv(a_md).max()
        if M_md < best_M:
            best_M = M_md
            best_a = a_md.copy()
        print(f"    MD: M={best_M:.10f}", flush=True)
    
    # Stage 3: More LP (15% of time)
    t3 = t_start + 0.90 * duration
    if time.monotonic() < t3:
        use_full = (n <= 1024)
        a_lp2 = run_lp_iterations(best_a, 200, t3, use_full=use_full, k=30)
        M_lp2 = fast_conv(a_lp2).max()
        if M_lp2 < best_M:
            best_M = M_lp2
            best_a = a_lp2.copy()
        print(f"    LP2: M={best_M:.10f}", flush=True)
    
    # Stage 4: Coordinate descent (10% of time)
    if time.monotonic() < deadline:
        max_sw = max(3, int((deadline - time.monotonic()) / max(0.2, n / 80.0)))
        a_cd = run_coordinate_descent(best_a, max_sw, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best_a = a_cd.copy()
        print(f"    CD: M={best_M:.10f}", flush=True)
    
    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining full LP best-response, mirror descent,
    and coordinate descent across multiple sequence lengths.
    
    Key improvement: uses ALL convolution constraints in the LP for
    n <= 1024, giving much better best-response directions.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    best_seq = None
    best_score = float('inf')
    t_total_start = time.monotonic()
    
    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_total_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(768)
        update_best(parent)
        print(f"No parent, using uniform n=768, J={best_score:.8f}", flush=True)
    
    n_parent = parent.size
    
    # === Phase 1: Multi-length, multi-init search ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length search", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Determine lengths to try
    lengths = []
    if n_parent >= 256:
        lengths.append(n_parent)
    for candidate_n in [512, 768, 1024, 1536]:
        if candidate_n not in lengths and 256 <= candidate_n <= 2048:
            lengths.append(candidate_n)
    
    # Sort: parent length first, then by size
    lengths.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths = lengths[:5]  # At most 5 lengths
    
    print(f"  Lengths: {lengths}", flush=True)
    
    # Time allocation: 60% for phase 1, 25% for phase 2, 15% for phase 3
    phase1_end = t_total_start + 0.60 * (budget_s - 30)
    phase2_end = t_total_start + 0.85 * (budget_s - 30)
    
    time_per_length = (phase1_end - time.monotonic() - 60) / len(lengths)
    time_per_length = max(time_per_length, 30)
    
    best_per_length = {}
    
    for li, n in enumerate(lengths):
        if time.monotonic() > phase1_end - 60:
            break
        t0 = time.monotonic()
        remaining = phase1_end - 60 - time.monotonic()
        if remaining < 20:
            break
        
        time_alloc = min(time_per_length, remaining / max(1, len(lengths) - li))
        print(f"\n  n={n}, alloc={time_alloc:.0f}s", flush=True)
        
        inits = generate_inits(n, rng, best_seq if best_seq is not None else parent)
        
        best_for_n = None
        best_M_for_n = float('inf')
        
        # Try at most 4 inits per length
        max_inits = min(4, len(inits))
        time_per_init = time_alloc / max_inits
        
        for ii in range(max_inits):
            if time.monotonic() > phase1_end - 60:
                break
            sub_dl = min(t0 + time_alloc, phase1_end - 60)
            if time.monotonic() >= sub_dl - 5:
                break
            
            a0 = np.maximum(inits[ii], 1e-8)
            a0 = a0 / a0.sum()
            
            a_opt = run_combined_optimization(a0, n, sub_dl, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()
            update_best(a_opt)
        
        if best_for_n is not None:
            best_per_length[n] = best_for_n.copy()
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best J={J_val:.8f} (elapsed={time.monotonic()-t0:.0f}s)", flush=True)
    
    update_best(best_seq)
    print(f"\n  After Phase 1: J={best_score:.8f}", flush=True)
    
    # === Phase 2: Intensive optimization on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive optimization on best (n={best_seq.size})", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None and time.monotonic() < phase2_end:
        n_best = best_seq.size
        t2_start = time.monotonic()
        
        # Long LP run
        t_lp_dl = min(phase2_end, time.monotonic() + 0.6 * (phase2_end - t2_start))
        use_full = (n_best <= 1024)
        a_lp = run_lp_iterations(best_seq, 1000, t_lp_dl, use_full=use_full, k=50)
        update_best(a_lp)
        
        # Mirror descent refinement
        if time.monotonic() < phase2_end - 10:
            t_md_dl = min(phase2_end - 5, time.monotonic() + 0.2 * (phase2_end - t2_start))
            a_md = run_mirror_descent(a_lp, 20000, t_md_dl, step_init=0.1, step_decay=0.9999)
            update_best(a_md)
        
        # Coordinate descent
        if time.monotonic() < phase2_end - 5:
            t_cd_dl = min(phase2_end - 5, time.monotonic() + 60)
            a_cd = run_coordinate_descent(a_md, 50, t_cd_dl, seed=12345)
            update_best(a_cd)
        
        print(f"  After Phase 2: J={best_score:.8f} (elapsed={time.monotonic()-t2_start:.0f}s)", flush=True)
    
    # === Phase 3: Final polish + length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Final polish and length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None and time.monotonic() < deadline:
        # Try resampling to slightly different lengths
        n_cur = best_seq.size
        for mult in [1.25, 1.5, 0.75]:
            if time.monotonic() > deadline - 30:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(best_seq, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            
            sub_dl = min(deadline - 10, time.monotonic() + 40)
            a_opt = run_combined_optimization(a_ext, new_n, sub_dl, seed=seed + 999)
            update_best(a_opt)
        
        # Final LP polish
        if time.monotonic() < deadline - 10:
            use_full = (best_seq.size <= 1024)
            a_final = run_lp_iterations(best_seq, 100, deadline - 5,
                                        use_full=use_full, k=40)
            update_best(a_final)
        
        # Final coordinate descent
        if time.monotonic() < deadline - 5:
            a_cd = run_coordinate_descent(a_final, 20, deadline - 2, seed=77777)
            update_best(a_cd)
    
    # Final report
    if best_seq is not None:
        final_J = compute_J(best_seq)
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL: J={final_J:.8f} n={best_seq.size} "
              f"time={time.monotonic()-t_total_start:.0f}s", flush=True)
        print(f"{'='*60}", flush=True)
    
    if best_seq is None:
        best_seq = np.ones(768)
    
    return [float(x) for x in best_seq]
