def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            vals.append(-1.0)

        c_obj = np.zeros(n + 1)

def smooth_gradient_descent(a, max_iters, deadline, tau_init=30.0, tau_end=300.0,
                            verbose=False):
    """
    Smooth gradient descent with log-sum-exp and Polyak step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    # Lower bound: 2n / (2n-1)
    LB = 2.0 * n / (2.0 * n - 1.0)

    no_improve = 0
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_init * (tau_end / tau_init) ** frac

        c = fast_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Gradient: g[j] = 2n * sum_k w[k] * (a[k-j] + a[j-k])
        a_rev = a[::-1]
        w_rev = w[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w_rev, size)

        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = n - 1 + i
            idx2 = 3 * n - 3 - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]
        grad *= 2.0 * n

        # Project to tangent space
        grad_t = grad - np.mean(grad)

        # Polyak step size
        g_norm = np.linalg.norm(grad_t)
        if g_norm < 1e-15:
            break
        lr = (compute_J(a) - LB) / (g_norm * g_norm)
        lr = min(lr, 5.0 / (it + 1) ** 0.5)  # cap it
        lr = max(lr, 1e-6)

        # Adam-style adaptation
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad_t
        v = beta2 * v + (1 - beta2) * grad_t ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        # Use Adam step with Polyak-like scaling
        adam_lr = lr * 0.1
        a_new = a - adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if verbose and it % 200 == 0:
            print(f"    SGD it {it}: tau={tau:.1f} obj={best_obj:.10f} lr={lr:.6f}", flush=True)

        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            a = best_a + rng.normal(0, 0.01, n)
            a = project_simplex(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            no_improve = 0
            if no_improve > 500:
                break

    return best_a, best_obj


def subgradient_descent(a, max_iters, deadline, verbose=False):

    Direct subgradient descent on the simplex with Polyak step.
    Uses the exact (non-smooth) subgradient at the maximizing k.
    """

    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()

    LB = 2.0 * n / (2.0 * n - 1.0)
    no_improve = 0
    avg_obj = best_obj

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()
        k_max = np.argmax(c)

        # Subgradient at k_max: g[j] = 2n * (a[k_max - j] + a[j - k_max])
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            kj = k_max - j
            if 0 <= kj < n:
                val += a[kj]
            jk = j - k_max
            if 0 <= jk < n:
                val += a[jk]
            grad[j] = 2.0 * n * val

        # Project to tangent
        grad_t = grad - np.mean(grad)
        g_norm = np.linalg.norm(grad_t)
        if g_norm < 1e-15:
            break

        # Polyak step
        avg_obj = 0.9 * avg_obj + 0.1 * compute_J(a)
        lr = (avg_obj - LB) / (g_norm * g_norm)
        lr = min(lr, 2.0 / (it + 1) ** 0.5)
        lr = max(lr, 1e-8)

        a_new = a - lr * grad_t
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            a_try = a - lr * 0.1 * grad_t
            a_try = project_simplex(a_try)
            obj_try = compute_J(a_try)
            if obj_try < best_obj:
                a = a_try

def evolutionary_search(a_best, pop_size, generations, deadline, verbose=False):

    for _ in range(pop_size - 1):
        noise_level = rng.uniform(0.02, 0.15)
        candidate = a_best + rng.normal(0, noise_level, n)
        candidate = project_simplex(candidate)
        pop.append(candidate)

    # Evaluate initial population
    pop_obj = [compute_J(p) for p in pop]
    best_idx = int(np.argmin(pop_obj))
    best_a = pop[best_idx].copy()
    best_obj = pop_obj[best_idx]

    for gen in range(generations):
        if time.monotonic() > deadline:
            break

        # Sort by fitness
        order = np.argsort(pop_obj)
        pop = [pop[i] for i in order]
        pop_obj = [pop_obj[i] for i in order]

        # Elitism: keep top half
        elite = pop[:pop_size // 2]

        # Generate new population
        new_pop = list(elite)
        new_obj = list(pop_obj[:pop_size // 2])

        while len(new_pop) < pop_size:
            # Selection: tournament of 3
            candidates = rng.integers(0, pop_size, 3)
            parent1 = pop[candidates[np.argmin([pop_obj[c] for c in candidates])]]
            parent2 = pop[candidates[np.argmin([pop_obj[c] for c in candidates])]]

            # Crossover
            if rng.random() < 0.5:
                child = 0.5 * parent1 + 0.5 * parent2
            else:
                child = parent1.copy()

            # Mutation
            mut_type = rng.integers(0, 4)
            if mut_type == 0:
                # Gaussian noise
                child = child + rng.normal(0, rng.uniform(0.01, 0.05), n)
            elif mut_type == 1:
                # Coordinate scaling
                idx = rng.integers(0, n)
                child[idx] *= rng.uniform(0.3, 2.0)
            elif mut_type == 2:
                # Block perturbation
                blk = rng.integers(5, max(6, n // 20))
                start = rng.integers(0, n - blk)
                child[start:start+blk] *= rng.uniform(0.5, 1.5)
            else:
                # Neighbor swap
                if n > 2:
                    i1 = rng.integers(0, n - 1)
                    i2 = (i1 + 1) % n
                    child[i1], child[i2] = child[i2], child[i1]

            child = project_simplex(child)
            obj = compute_J(child)

            # Compare with worst in population
            worst_idx = np.argmax(new_obj)
            if obj < new_obj[worst_idx]:
                new_pop.append(child)
                new_obj.append(obj)
            else:
                new_pop.append(child)
                new_obj.append(obj)

        pop = new_pop[:pop_size]
        pop_obj = new_obj[:pop_size]

        # Update best
        gen_best = int(np.argmin(pop_obj))
        if pop_obj[gen_best] < best_obj:
            best_obj = pop_obj[gen_best]
            best_a = pop[gen_best].copy()
            if verbose:
                print(f"    EA gen {gen}: best={best_obj:.10f}", flush=True)

    return best_a, best_obj

    print(f"{'='*60}", flush=True)

    if time.monotonic() < phase2_deadline - 30:
        # SCA with shrinking active set
        remaining = phase2_deadline - time.monotonic() - 60
        if remaining > 30:
            a_sca, obj_sca = active_set_sca(best_seq, max_iters=500,
                                            deadline=time.monotonic() + remaining,
                                            active_frac=0.15, verbose=False)
            update_best(a_sca)

    if time.monotonic() < phase2_deadline - 20:
        # Smooth gradient descent
        remaining = phase2_deadline - time.monotonic() - 10
        if remaining > 20:
            a_sgd, obj_sgd = smooth_gradient_descent(best_seq, max_iters=20000,
                                                     deadline=time.monotonic() + remaining,
                                                     tau_init=20.0, tau_end=500.0)
            update_best(a_sgd)

    # === PHASE 3: Multi-length exploration ===
    phase3_deadline = t_start + 1000.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length exploration (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Try lengths around parent
    candidate_ns = []
    for n_try in [n_parent - 64, n_parent - 32, n_parent + 32, n_parent + 64,
                  n_parent + 128, n_parent * 3 // 2, n_parent * 5 // 4]:
        if 128 <= n_try <= 2048 and n_try != n_parent:
            candidate_ns.append(n_try)
    # Add some new lengths
    for n_try in [512, 640, 800, 960, 1024, 1280]:
        if n_try not in candidate_ns and n_try != n_parent and 128 <= n_try <= 2048:
            candidate_ns.append(n_try)

    candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_parent))[:5]

    for new_n in candidate_ns:
        remaining = phase3_deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(80.0, remaining - 15)
        if alloc < 30:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        # Quick CD at new length
        a_opt = coord_descent_refined(a_ext, max_sweeps=20000,
                                      deadline=time.monotonic() + alloc - 5,
                                      seed=777)
        update_best(a_opt)
        print(f"  n={new_n}: J={compute_J(a_opt):.10f}", flush=True)

    # === PHASE 4: Evolutionary search ===
    phase4_deadline = t_start + 1250.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Evolutionary search (budget {phase4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < phase4_deadline - 30:
        remaining = phase4_deadline - time.monotonic() - 10
        if remaining > 30:
            a_ea, obj_ea = evolutionary_search(best_seq, pop_size=10,
                                               generations=10000,
                                               deadline=time.monotonic() + remaining)
            update_best(a_ea)

    # === PHASE 5: Final intensive polish ===
    phase5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final polish (budget {phase5_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a_final = best_seq.copy()

    # Long CD
    remaining = phase5_deadline - time.monotonic()
    if remaining > 30:
        alloc = 0.35 * remaining
        a_final = coord_descent_refined(a_final, max_sweeps=100000,
                                        deadline=time.monotonic() + alloc,
                                        seed=424242)
        update_best(a_final)

    # Subgradient
    remaining = phase5_deadline - time.monotonic()
    if remaining > 20:
        alloc = 0.2 * remaining
        a_sub, obj_sub = subgradient_descent(a_final, max_iters=50000,
                                             deadline=time.monotonic() + alloc)
        update_best(a_sub)

    # SCA with tight active set
    remaining = phase5_deadline - time.monotonic()
    if remaining > 20:
        alloc = 0.15 * remaining
        a_sca2, obj_sca2 = active_set_sca(best_seq, max_iters=300,
                                          deadline=time.monotonic() + alloc,
                                          active_frac=0.05)
        update_best(a_sca2)

    # Final CD on best
    remaining = phase5_deadline - time.monotonic()
    if remaining > 15:
        a_final = coord_descent_refined(best_seq, max_sweeps=50000,
                                        deadline=time.monotonic() + remaining - 5,
                                        seed=999999)
        update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.10f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
