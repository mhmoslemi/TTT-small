import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent_gs(a, max_sweeps, deadline, random_order=True, seed=0,
                     gs_iters=25, hi_factor=3.0):
    """Coordinate descent with golden section search and incremental conv updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 150
    gr = (np.sqrt(5) - 1) / 2

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = a[i]
            lo_k = i
            hi_k = min(2 * n - 2, i + n - 1)
            k_self = 2 * i
            a_slice = a[lo_k - i:hi_k - i + 1].copy()

            # Golden section search range
            lo_x = 0.0
            hi_x = max(hi_factor * old, 0.02 / n)

            c1 = hi_x - gr * (hi_x - lo_x)
            c2 = lo_x + gr * (hi_x - lo_x)

            # Evaluate f(c1)
            saved = conv[lo_k:hi_k + 1].copy()
            d1 = c1 - old
            conv[lo_k:hi_k + 1] += 2.0 * d1 * a_slice
            if 0 <= k_self <= 2 * n - 2:
                conv[k_self] += d1 * d1
            ns1 = s - old + c1
            f1 = 2.0 * n * conv.max() / (ns1 * ns1) if ns1 > 1e-10 else float('inf')
            conv[lo_k:hi_k + 1] = saved

            # Evaluate f(c2)
            saved = conv[lo_k:hi_k + 1].copy()
            d2 = c2 - old
            conv[lo_k:hi_k + 1] += 2.0 * d2 * a_slice
            if 0 <= k_self <= 2 * n - 2:
                conv[k_self] += d2 * d2
            ns2 = s - old + c2
            f2 = 2.0 * n * conv.max() / (ns2 * ns2) if ns2 > 1e-10 else float('inf')
            conv[lo_k:hi_k + 1] = saved

            for _ in range(gs_iters):
                if f1 < f2:
                    hi_x = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi_x - gr * (hi_x - lo_x)
                    saved = conv[lo_k:hi_k + 1].copy()
                    dtemp = c1 - old
                    conv[lo_k:hi_k + 1] += 2.0 * dtemp * a_slice
                    if 0 <= k_self <= 2 * n - 2:

                    nst = s - old + c1
                    f1 = 2.0 * n * conv.max() / (nst * nst) if nst > 1e-10 else float('inf')
                    conv[lo_k:hi_k + 1] = saved
                else:
                    lo_x = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo_x + gr * (hi_x - lo_x)
                    saved = conv[lo_k:hi_k + 1].copy()
                    dtemp = c2 - old

def subgradient_adapt(a, max_iters, deadline, lr_init=0.01):
    """Subgradient descent exploiting the shifted-reversed structure."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[i] = 2*a[k_star - i] if valid
        grad = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]

        # Center for simplex
        grad = grad - np.mean(grad)

        lr = lr_init / (1.0 + 0.05 * it)
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_obj


def sca_lp_step(a, max_iters, deadline, active_frac=0.3):
    """SCA-LP with active set reduction for speed."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break

        c = fast_conv(a)
        M = c.max()

        # Active set: only keep top fraction of constraints
        n_constraints = 2 * n - 1
        n_active = max(n_constraints, int(n_constraints * active_frac))
        top_vals, top_idx = np.sort(c)[::-1][:n_active]

        # Build sparse constraint matrix for active constraints only
        rows = []
        cols = []
        vals = []
        b_ub = []

        for row_i, k in enumerate(top_idx):
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(row_i)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(row_i)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(top_vals[row_i])

        A = coo_matrix((vals, (rows, cols)), shape=(n_active, n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(None, None)]

        A_eq = coo_matrix((np.ones(n), (np.zeros(n), range(n))),
                          shape=(1, n + 1)).tocsr()
        b_eq = [1.0]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': 15, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    J_g = compute_J(g)
                    if J_g < best_obj:
                        a = g.copy()
                        best_obj = J_g
                        best_a = g.copy()
                        no_improve = 0
                    else:
                        found = False
                        for frac in [0.5, 0.3, 0.2, 0.1]:
                            a_try = (1 - frac) * a + frac * g
                            a_try = np.maximum(a_try, 0)
                            a_try = a_try / a_try.sum()
                            J_try = compute_J(a_try)
                            if J_try < best_obj - 1e-13:
                                a = a_try
                                best_obj = J_try
                                best_a = a_try.copy()
                                no_improve = 0
                                found = True
                                break
                        if not found:
                            no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_obj


def block_swap_move(a, rng):
    """Swap two random blocks of equal length."""
    n = len(a)
    block_len = rng.integers(max(2, n // 20), max(3, n // 5))
    i = rng.integers(0, n - block_len)
    j = rng.integers(0, n - block_len)
    if abs(i - j) <= block_len:
        return a.copy()
    a_new = a.copy()
    temp = a_new[i:i + block_len].copy()
    a_new[i:i + block_len] = a_new[j:j + block_len]
    a_new[j:j + block_len] = temp
    return np.maximum(a_new, 0)


def block_reverse_move(a, rng):
    """Reverse a random block."""
    n = len(a)
    block_len = rng.integers(max(2, n // 10), max(3, n // 3))
    i = rng.integers(0, n - block_len)
    a_new = a.copy()
    a_new[i:i + block_len] = a_new[i:i + block_len][::-1]
    return a_new


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences for length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
        # Block perturbation of parent
        p_block = block_swap_move(p, rng)
        if p_block.sum() > 1e-10:
            inits.append(project_simplex(p_block))

    # Structured patterns
    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 4, 5, 7, 11]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    for center in [0.2, 0.35, 0.5, 0.65, 0.8]:
        for width in [0.08, 0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.05))

    # Smooth random
    ksize = max(3, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.03))

    # Two-tone patterns
    for ratio in [0.3, 0.4, 0.5, 0.6]:
        a2 = np.ones(n) * 0.3
        a2[::2] = 1.0
        inits.append(project_simplex(a2 * (1 - ratio) + ratio * rng.uniform(0.5, 1.5, n)))

    return inits


def solve_pipeline(a, deadline, verbose=False):
    """Full optimization pipeline for one sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: CD with golden section (40% time)
    t1 = time.monotonic() + 0.40 * remaining
    a = coord_descent_gs(a, max_sweeps=50000, deadline=t1, random_order=True, seed=42)
    a = try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: SCA-LP (30% time)
    t2 = time.monotonic() + 0.30 * remaining
    a_sca, obj_sca = sca_lp_step(a, max_iters=100, deadline=t2, active_frac=0.4)
    if obj_sca < best_obj:
        best_a = a_sca.copy()
        best_obj = obj_sca
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: CD again on best (20% time)
    t3 = time.monotonic() + 0.20 * remaining
    a = coord_descent_gs(a, max_sweeps=50000, deadline=t3, random_order=True, seed=123)
    a = try_update(a)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Subgradient + final CD (10% time)
    t4 = time.monotonic() + 0.05 * remaining
    a_sg, obj_sg = subgradient_adapt(a, max_iters=200, deadline=t4, lr_init=0.005)
    if obj_sg < best_obj:
        best_a = a_sg.copy()
        best_obj = obj_sg
    a = best_a.copy()

    t5 = deadline - 1
    a = coord_descent_gs(a, max_sweeps=50000, deadline=t5, random_order=True, seed=999)
    a = try_update(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses golden-section CD, SCA-LP, subgradient adaptation,
    block moves, and aggressive multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    # === PHASE 1: Multi-length exploration (350s) ===
    phase1_end = t_start + 350.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-length exploration (budget {phase1_end - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Prioritize lengths near parent and known-good ranges
    n_parent = len(parent)
    candidate_ns = set()
    for delta in range(-128, 129, 16):
        nn = n_parent + delta
        if 256 <= nn <= 2048:
            candidate_ns.add(nn)
    # Add some specific candidates
    for nn in [384, 448, 512, 576, 640, 704, 768, 832, 896, 960, 1024, 1152, 1280, 1408]:
        if 256 <= nn <= 2048:
            candidate_ns.add(nn)

    candidate_ns = sorted(candidate_ns, key=lambda x: abs(x - n_parent))
    candidate_ns = candidate_ns[:12]  # Limit to 12 lengths

    per_length_time = (phase1_end - time.monotonic()) / max(1, len(candidate_ns)) - 5

    for new_n in candidate_ns:
        remaining_check = phase1_end - time.monotonic()
        if remaining_check < 30:
            break
        alloc = min(per_length_time, remaining_check - 10)
        if alloc < 20:
            break

        print(f"  n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_init = resample_seq(parent, new_n)
        a_init = np.maximum(a_init, 1e-8)
        a_init = a_init / a_init.sum()

        # Quick optimization: just CD
        a_opt = coord_descent_gs(a_init, max_sweeps=20000,
                                 deadline=time.monotonic() + alloc * 0.7,
                                 random_order=True, seed=hash(new_n) % 10000)
        update_best(a_opt)

        # Also try a perturbed version
        if remaining_check > 30:
            a_pert = project_simplex(a_init + rng.normal(0, 0.05, new_n))
            a_opt2 = coord_descent_gs(a_pert, max_sweeps=20000,
                                      deadline=time.monotonic() + alloc * 0.25,
                                      random_order=True, seed=hash(new_n + 1) % 10000)
            update_best(a_opt2)

        print(f"    n={new_n} best so far: J={best_score:.8f}", flush=True)

    # === PHASE 2: Intensive optimization at top lengths (750s) ===
    phase2_end = t_start + 1100.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Intensive optimization (budget {phase2_end - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Determine top lengths to focus on
    n_focus = best_seq.size if best_seq is not None else n_parent

    # Full pipeline on best sequence
    remaining = phase2_end - time.monotonic()
    if remaining > 30:
        print(f"  Full pipeline on n={n_focus} (budget {remaining * 0.5:.0f}s)...", flush=True)
        a_full, obj_full = solve_pipeline(best_seq.copy(), time.monotonic() + remaining * 0.5)
        update_best(a_full)

    # Try multiple diverse inits at best length
    remaining = phase2_end - time.monotonic()
    if remaining > 30:
        inits = generate_inits(n_focus, rng, best_seq)
        n_inits = min(len(inits), 8)
        per_init = (remaining - 20) / max(1, n_inits)

        for idx in range(n_inits):
            rem = phase2_end - time.monotonic()
            if rem < 20:
                break
            alloc = min(per_init, rem - 10)
            if alloc < 15:
                break

            a0 = np.maximum(inits[idx], 0)
            if a0.sum() < 1e-10:
                continue
            a0 = a0 / a0.sum()

            # Quick pipeline: CD + maybe LP
            a_opt = coord_descent_gs(a0, max_sweeps=30000,
                                     deadline=time.monotonic() + alloc * 0.6,
                                     random_order=True, seed=1000 + idx)
            update_best(a_opt)

            if rem - alloc * 0.6 > 15:
                a_sca, obj_sca = sca_lp_step(a_opt, max_iters=30,
                                             deadline=time.monotonic() + alloc * 0.3,
                                             active_frac=0.3)
                update_best(a_sca)

            print(f"    Init {idx}: best J={best_score:.8f}", flush=True)

    # Try block moves on best
    remaining = phase2_end - time.monotonic()
    if remaining > 30 and best_seq is not None:
        for mv in range(min(6, int(remaining // 20))):
            rem = phase2_end - time.monotonic()
            if rem < 20:
                break
            a_mv = block_swap_move(best_seq, rng)
            a_mv = a_mv / a_mv.sum()
            a_opt = coord_descent_gs(a_mv, max_sweeps=10000,
                                     deadline=time.monotonic() + rem * 0.4,
                                     random_order=True, seed=5000 + mv)
            update_best(a_opt)

    # === PHASE 3: Final polish (remaining time) ===
    phase3_end = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Final polish (budget {phase3_end - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_end - time.monotonic()
        if remaining > 20:
            # Long CD pass
            t_cd = time.monotonic() + remaining * 0.45
            print(f"  Long CD (budget {t_cd - t_start:.0f}s)...", flush=True)
            a_final = coord_descent_gs(best_seq, max_sweeps=100000,
                                       deadline=t_cd, random_order=True, seed=7777)
            update_best(a_final)

        remaining = phase3_end - time.monotonic()
        if remaining > 15:
            # LP polish
            t_lp = time.monotonic() + remaining * 0.35
            print(f"  LP polish (budget {t_lp - t_start:.0f}s)...", flush=True)
            a_lp, obj_lp = sca_lp_step(best_seq, max_iters=50, deadline=t_lp, active_frac=0.5)
            update_best(a_lp)

        remaining = phase3_end - time.monotonic()
        if remaining > 10:
            # One more CD
            a_final2 = coord_descent_gs(best_seq, max_sweeps=50000,
                                        deadline=phase3_end - 3,
                                        random_order=True, seed=4321)
            update_best(a_final2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
