def fast_conv(a):

        return np.array([])
    size = 1
    while size < 2*n - 1:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):

    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1, dtype=np.float64)
    cond = u - cssv/ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def nesterov_subgradient(a, deadline, verbose=False, eta0=0.5, tau=100):
    """Nesterov accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_J(a)
    
    # Nesterov state
    x = a.copy()  # current point
    y = a.copy()  # extrapolated point
    beta = 1.0
    
    # For averaging
    avg = a.copy()
    T_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() > deadline:
            break
        
        # Compute objective at y
        obj_y = compute_J(y)
        if obj_y < best_obj:
            best_obj = obj_y
            best_a = y.copy()
        
        # Compute subgradient at y
        conv_y = fast_conv(y)
        k_star = np.where(conv_y == conv_y.max())[0]
        S = y.sum()
        
        # Subgradient of max_k conv[k] w.r.t. a[i]:
        # grad_max[i] = 2 * mean over k* of a[k*-i] (if 0<=k*-i<n)
        grad_max = np.zeros(n)
        for k in k_star:
            di = k
            # d(conv[di])/da[i] = 2*a[di-i] if 0<=di-i<n
            lo = max(0, di - n + 1)
            hi = min(n - 1, di)
            grad_max[lo:hi+1] += 2.0 * y[di - lo:di - hi + 1][::-1]
        
        grad_max /= len(k_star)
        
        # Full gradient of J = 2n*max_b/S^2
        # dJ/da[i] = 2n*(grad_max[i]/S - 2*max_b/S^2)
        max_b = conv_y.max()
        grad_J = 2*n * (grad_max/S - 2*max_b/S**2)
        
        # Step size
        eta = eta0 / (1.0 + t / tau)
        
        # Update (don't project yet for Nesterov)
        x = y - eta * grad_J
        
        # Update Nesterov parameter
        beta_new = (1.0 + np.sqrt(1.0 + 4.0*beta*beta)) / 2.0
        y = x + (beta/beta_new - 1.0) * (x - y)
        beta = beta_new
        
        # Project y onto simplex
        y = project_simplex(y)
        x = project_simplex(x)
        
        # Running average
        T_count += 1
        avg = (1.0 - 1.0/T_count) * avg + (1.0/T_count) * y
        
        if verbose and t % 1000 == 0:
            print(f"    NSG t={t}: obj={best_obj:.8f}", flush=True)
    
    # Check the average
    if avg.sum() > 1e-10:
        avg = avg / avg.sum()
        obj_avg = compute_J(avg)
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = avg.copy()
    
    return best_a, best_obj

        M = conv_a.max()
        
        # LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
        # (a*g)[k] = sum_j a[j]*g[k-j]
        # This is linear in g!
        
        rows = []
        cols = []
        vals = []
        for k in range(2*n-1):
            j_min = max(0, k - n + 1)
            j_max = min(n-1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k-j])
        
        if len(rows) == 0:
            break
            
        A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                       options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    # Line search
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1-frac)*a + frac*g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try = a_try / a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        if no_improve > 10:
            break
    
    return best_a, best_obj

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences, biased toward asymmetric ones."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
    
    # Asymmetric inits (key insight: optimal is asymmetric)
    # Concentrated on left
    for width in [0.3, 0.5, 0.7]:
        inits.append(project_simplex(np.exp(-((t / width) ** 2)) + 0.01))
    # Concentrated on right
    for width in [0.3, 0.5, 0.7]:
        inits.append(project_simplex(np.exp(-(((t-1) / width) ** 2)) + 0.01))
    # Ramp
    inits.append(project_simplex(np.linspace(0.1, 1.0, n)))
    inits.append(project_simplex(np.linspace(1.0, 0.1, n)))
    # Quadratic
    inits.append(project_simplex((t**2 + 0.01)))
    inits.append(project_simplex(((1-t)**2 + 0.01)))
    # Sine (asymmetric)
    inits.append(project_simplex(np.abs(np.sin(np.pi * t)) + 0.05))
    inits.append(project_simplex(np.abs(np.cos(np.pi * t)) + 0.05))
    
    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))
    
    # Uniform (baseline)
    inits.append(np.ones(n) / n)
    
    return inits
