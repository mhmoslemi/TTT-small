import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_indices(n):
    """Pre-compute the fixed sparsity structure for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_active(a, M, rows_all, cols_all, active_mask=None, eps=0.3):
    """
    Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
    Only includes constraints where the corresponding autocorrelation is within eps of M.
    """
    n = len(a)
    if active_mask is None:
        # Compute active set
        b = fast_conv(a)
        b_max = b.max()
        # Include constraints where b[k] > (1-eps) * b_max
        # This ensures we don't miss any constraint that could be violated
        active_mask = b > (1.0 - eps) * b_max
    
    rows = rows_all[active_mask]
    cols = cols_all[active_mask]
    vals = a[rows - cols]
    
    m = len(rows)
    if m == 0:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.zeros(2 * n - 1)
    b_ub[rows] = M
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def solve_lp_full(a, M, rows_all, cols_all):
    """Solve the full LP (all constraints)."""
    return solve_lp_active(a, M, rows_all, cols_all, active_mask=None, eps=0.0)


def anderson_mix(history_x, history_Fx, window):
    """Anderson mixing: find convex combination of recent iterates that best reduces residual."""
    m = min(window, len(history_Fx))
    if m < 2:
        return history_Fx[-1].copy()
    
    x_hist = history_x[-m:]
    Fx_hist = history_Fx[-m:]
    
    residuals = np.array([x_hist[i] - Fx_hist[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)
    
    try:
        alpha = np.linalg.solve(G, np.ones(m))
        denom = alpha.sum()
        if abs(denom) < 1e-15:
            alpha = np.ones(m) / m
        else:
            alpha = alpha / denom
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m
    
    alpha = np.clip(alpha, -0.5, 1.5)
    s = alpha.sum()
    if abs(s) < 1e-10:
        alpha = np.ones(m) / m
    else:
        alpha = alpha / s
    
    mixed = np.zeros_like(Fx_hist[0])
    for i in range(m):
        mixed += alpha[i] * Fx_hist[i]
    return np.maximum(mixed, 0)


def run_lp_descent(a, deadline, rows_all, cols_all,

def gs_refine(a, n_iters=200, beta=0.85):
    """Gerchberg-Saxton refinement for flatter autocorrelation."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Flatten the magnitude spectrum
        mag_power = mag ** 0.7
        mag_new = mag_power / (mag_power.max() + 1e-30) * mag.max()
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new /= sn
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def smooth_grad_descent(a, deadline, tau=50.0, lr=0.1, max_iters=2000):
    """Smooth max gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        if b_max < best_M:
            best_M = b_max
            best_a = a.copy()
        
        # Smooth max gradient
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # (a*a)(k) = sum_{j=lo}^{hi} a[j] * a[k-j]
            # d/d a[i] = a[k-i] for i in [lo, hi]
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * p[k] * a[k - i]
        
        a_new = np.maximum(a - lr * grad, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new /= s
        a = a_new
        # Adaptive tau
        tau *= 1.005
    
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_indices(n):

    Returns (rows, cols) arrays where A[rows[i], cols[i]] = a[rows[i] - cols[i]]."""

    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, rows_all, cols_all, eps_active=0.2):
    """
    Solve LP: max sum(g) s.t. conv(a,g)(k) <= M for k in active set, g >= 0.
    Active set: indices where conv(a,a)(k) > (1-eps)*max(conv(a,a)).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    if eps_active >= 1.0:
        active_mask = np.ones(2 * n - 1, dtype=bool)
    else:
        active_mask = b > (1.0 - eps_active) * b_max

    if not np.any(active_mask):
        active_mask = b > b_max * 0.5

    rows = rows_all[active_mask]
    cols = cols_all[active_mask]
    vals = a[rows - cols]

    m_con = len(rows)
    if m_con == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.zeros(2 * n - 1)
    b_ub[rows] = M
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9,
                             'time_limit': 10.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def solve_lp_full(a, M, rows_all, cols_all):
    """Solve the full LP with all constraints."""
    return solve_lp(a, M, rows_all, cols_all, eps_active=1.0)


def anderson_mix(history_x, history_Fx, window):
    """Anderson mixing to accelerate fixed-point iteration convergence."""
    m = min(window, len(history_Fx))
    if m < 2:
        return history_Fx[-1].copy()

    x_hist = history_x[-m:]
    Fx_hist = history_Fx[-m:]

    residuals = np.array([x_hist[i] - Fx_hist[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-14 * np.eye(m)

    try:
        alpha = np.linalg.solve(G, np.ones(m))
        denom = alpha.sum()
        if abs(denom) < 1e-15:
            alpha = np.ones(m) / m
        else:
            alpha = alpha / denom
    except np.linalg.LinAlgError:
        alpha = np.ones(m) / m

    alpha = np.clip(alpha, -0.5, 1.5)
    s = alpha.sum()
    if abs(s) < 1e-10:
        alpha = np.ones(m) / m
    else:
        alpha = alpha / s

    mixed = np.zeros_like(Fx_hist[0])
    for i in range(m):
        mixed += alpha[i] * Fx_hist[i]
    return np.maximum(mixed, 0)


def run_lp_descent(a, deadline, rows_all, cols_all,
                   max_iters=500, eps_start=0.3, eps_end=0.02,
                   anderson_window=5, t_init=0.35, use_symmetry=True):
    """
    LP mirror descent with active set reduction, backtracking, and Anderson acceleration.
    Progressively tightens the active set for better convergence.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    if use_symmetry:
        a = (a + a[::-1]) / 2.0
        a = np.maximum(a, 1e-10)
        a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    hist_x = []
    hist_Fx = []
    no_imp = 0
    t = t_init
    prev_M = best_M

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 60:
                break

        # Adaptive active set: tighten as we converge
        progress = min(1.0, it / (max_iters * 0.5))
        eps = eps_start + (eps_end - eps_start) * progress

        g = solve_lp(a, M, rows_all, cols_all, eps_active=eps)
        if g is None:
            g = solve_lp_full(a, M, rows_all, cols_all)
            if g is None:
                break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        if use_symmetry:
            g = (g + g[::-1]) / 2.0
            g = np.maximum(g, 0)
            gs2 = g.sum()
            if gs2 > 1e-10:
                g /= gs2

        # Check convergence: if g ≈ a, we're at fixed point
        diff_norm = np.linalg.norm(g - a) / (np.linalg.norm(a) + 1e-15)
        if diff_norm < 1e-7:
            break

        # Backtracking line search
        best_t = t
        best_M_new = M + 1e-15
        candidates = sorted(set([t, t * 0.5, t * 1.5, 0.1, 0.2, 0.4, 0.05]),
                            key=lambda x: -abs(x - t))
        for t_try in candidates[:4]:
            t_try = float(np.clip(t_try, 0.005, 1.0))
            a_try = np.maximum((1 - t_try) * a + t_try * g, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            if use_symmetry:
                a_try = (a_try + a_try[::-1]) / 2.0
                a_try = np.maximum(a_try, 0)
                s2 = a_try.sum()
                if s2 < 1e-10:
                    continue
                a_try /= s2
            M_try = fast_conv(a_try).max()
            if M_try < best_M_new:
                best_M_new = M_try
                best_t = t_try
        t = best_t

        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        if use_symmetry:
            a_new = (a_new + a_new[::-1]) / 2.0
            a_new = np.maximum(a_new, 0)
            s2 = a_new.sum()
            if s2 < 1e-10:
                a_new = a
            else:
                a_new /= s2

        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())

        if len(hist_x) > anderson_window + 2:
            hist_x.pop(0)
            hist_Fx.pop(0)

        if len(hist_x) >= 3:
            a_and = anderson_mix(hist_x, hist_Fx, anderson_window)
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
                if use_symmetry:
                    a_and = (a_and + a_and[::-1]) / 2.0
                    a_and = np.maximum(a_and, 0)
                    sa2 = a_and.sum()
                    if sa2 > 1e-10:
                        a_and /= sa2
                M_and = fast_conv(a_and).max()
                M_new = fast_conv(a_new).max()
                if M_and <= M_new * 1.005:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new

        # Decay step size
        t = max(0.01, t * 0.99)

    return best_a


def gs_refine(a, n_iters=300, beta=0.85):
    """Gerchberg-Saxton: flatten power spectrum while maintaining non-negativity."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Compress dynamic range
        mag_norm = mag / (mag.max() + 1e-30)
        # Power compression
        power = 0.5 + 0.4 * beta
        mag_comp = (mag ** power)
        mag_new = mag_comp / (mag_comp.max() + 1e-30) * mag.max()
        # Mixing
        mag_final = (1 - beta) * mag + beta * mag_new
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new /= sn
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def smooth_grad_descent(a, deadline, tau_start=10.0, tau_end=200.0,
                        lr_start=0.08, lr_end=0.002, max_iters=5000):
    """Smooth max gradient descent with adaptive tau and learning rate."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = float(b.max())
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                break

        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_start * (lr_end / lr_start) ** frac

        # Smooth max weights
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] for valid indices
        grad = np.zeros(n)
        active = np.where(p > 1e-9)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # a[k-i] for i in [lo, hi]
            idx = k - np.arange(lo, hi + 1)
            grad[lo:hi + 1] += 2.0 * p[k] * a[idx][::-1]

        a_new = np.maximum(a - lr * grad, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new /= s
        a = a_new

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
