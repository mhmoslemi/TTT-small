    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(1, n+1)) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(a - theta, 0)

        rows = list(self.active_set)

        for k in range(nc):

            j_max = min(n - 1, k)

        # So vals[i] = a[k_i - j_i]
        self.idx = np.array(rows) - np.array(cols)  # k - j for each non-zero
    
    def solve_full(self, a, M):

        vals = a[self.idx]
        A = csr_matrix((vals, (self.rows, self.cols)), shape=(self.nc, n))
        c = -np.ones(n)
        b_ub = np.full(self.nc, M)

        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success:

        return None
    
    def solve_active(self, a, M, active_set):

        a_rows = self.rows[mask]

"""

3. Active-constraint LP mirror descent with cached sparsity and constraint generation

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def fast_conv(a):
    n = len(a)
    size = fft_size(n)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv2(a, b):
    n1 = len(a)
    n2 = len(b)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n1 + n2 - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau, thresh_frac=1e-10):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    thresh = max(thresh_frac, 1.0 / (n * 20))
    active = np.where(p > thresh)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[hi:lo - 1:-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, b, b_max


class ConvLPCache:
    def __init__(self, n):
        self.n = n
        self.nc = 2 * n - 1
        rows = []
        cols = []
        for k in range(self.nc):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.idx = (self.rows - self.cols).astype(np.int32)
        self.nnz = len(rows)
        self.c = -np.ones(n)
        self.bounds = [(0, None)] * n

    def build_matrix(self, a, row_indices=None):
        if row_indices is None:
            mask = np.ones(self.nc, dtype=bool)
            actual_rows = np.arange(self.nc)
        else:
            row_indices = np.asarray(row_indices, dtype=np.int32)
            mask = np.isin(self.rows, row_indices)
            # Remap row indices
            sorted_active = np.sort(row_indices)
            row_map = np.full(self.nc, -1, dtype=np.int32)
            row_map[sorted_active] = np.arange(len(sorted_active))
            actual_rows = row_map[self.rows[mask]]
        vals = a[self.idx[mask]]
        shape = (self.nc if row_indices is None else len(row_indices), self.n)
        A = csr_matrix((vals, (actual_rows, self.cols[mask])), shape=shape)
        return A

    def solve(self, a, M, active_set=None):
        n = self.n
        if active_set is not None and len(active_set) < self.nc:
            active_arr = np.sort(np.asarray(active_set, dtype=np.int32))
            A = self.build_matrix(a, active_arr)
            nc_active = len(active_arr)
        else:
            A = self.build_matrix(a)
            nc_active = self.nc
        b_ub = np.full(nc_active, M)
        r = linprog(self.c, A_ub=A, b_ub=b_ub, bounds=self.bounds,
                    method='highs', options={'presolve': True,
                                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
        return None


def run_adam_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=500.0,
                    lr_init=0.05, lr_end=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(a, tau)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = np.maximum(a - lr * m_hat / (np.sqrt(v_hat) + eps), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        M_new = float(fast_conv(a_new).max())
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()
        a = a_new
    return best_a


def run_fista_smooth(a, deadline, max_iters=2000, tau=50.0, lr=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    t_k = 1.0
    if lr is None:
        lr = 0.5 / (2.0 * tau * n)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, b, b_max = smooth_max_grad(y, tau)
        # Gradient step
        z = np.maximum(y - lr * grad, 0.0)
        sz = z.sum()
        if sz < 1e-10:
            z = y.copy()
        else:
            z = z / sz
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - y)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = z.copy()
        else:
            y = y / sy
        t_k = t_new
        M_cur = float(fast_conv(z).max())
        if M_cur < best_M * (1 - 1e-12):
            best_M = M_cur
            best_a = z.copy()
    return best_a


def gs_flat_init(n, rng, n_iters=200, beta=0.85):
    a = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    a = a / a.sum()
    size = fft_size(n)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        phase = np.angle(F)
        mag = np.abs(F)
        # Compress magnitude
        mag_norm = mag / (mag.max() + 1e-30)
        mag_flat = mag.mean() + 1e-30
        mag_new = (1.0 - beta) * mag + beta * mag_flat
        # Also try power compression
        pw = 0.5 + 0.3 * beta
        mag_pow = (mag ** pw)
        mag_pow = mag_pow / (mag_pow.max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_pow
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        M_new = float(fast_conv(a_new).max())
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new
    return best_a


def flat_spectrum_init(n, rng):
    size = fft_size(n)
    n_freq = size // 2 + 1
    # Flat magnitude with random phase
    mag = np.ones(n_freq)
    phase = rng.uniform(0, 2 * np.pi, n_freq)
    phase[0] = 0.0
    if size % 2 == 0:
        phase[-1] = 0.0
    F = mag * np.exp(1j * phase)
    a = np.fft.irfft(F, size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

def run_active_lp(a, deadline, max_iters=200, lp_cache=None, t_base=0.3, min_t=0.01,
                  active_buffer=30):

    a = a / a.sum()

    if lp_cache is None:
        lp_cache = ConvLPCache(n)
    best_a = a.copy()

    active_set = set(range(2 * n - 1))
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
        # Solve LP
        if len(active_set) < 2 * n - 1:
            g = lp_cache.solve(a, M, active_set)
        else:
            g = lp_cache.solve(a, M)
        if g is None or g.sum() < 1e-12:
            break
        g = g / g.sum()
        # Determine step size
        if it < 3:
            t = t_base
        else:
            t = max(min_t, t_base / (1 + 0.08 * it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        # Update active set
        ag = fast_conv2(a_new, g)
        if len(active_set) > n:
            # Prune: keep only constraints that are close to active
            M_new = float(fast_conv(a_new).max())
            close_active = np.where(ag > M_new - 0.05 * M_new)[0]
            # Also keep recently active
            active_set = set(int(k) for k in close_active)
            # Ensure we have at least active_buffer constraints
            if len(active_set) < active_buffer:
                top_k = np.argsort(ag)[::-1][:active_buffer]
                active_set = set(int(k) for k in top_k)
        a = a_new
    return best_a

        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
        if len(active_set) < 2 * n - 1:
            g = lp_cache.solve(a, M, active_set)
        else:
            g = lp_cache.solve(a, M)
        if g is None or g.sum() < 1e-12:
            break
        g = g / g.sum()
        if it < 5:
            t = t_base
        else:
            t = max(min_t, t_base / (1 + 0.05 * it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        # Anderson acceleration
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        if len(hist_x) >= 3:
            residuals = np.array([hist_x[i] - hist_Fx[i] for i in range(len(hist_x))])
            G = residuals @ residuals.T
            G += 1e-12 * np.eye(len(hist_x))
            ones = np.ones(len(hist_x))
            try:
                G_inv_ones = np.linalg.solve(G, ones)
                denom = ones @ G_inv_ones
                if abs(denom) > 1e-15:
                    alpha = G_inv_ones / denom
                    alpha = np.clip(alpha, -1.0, 1.0)
                    alpha = alpha / alpha.sum()
                    mixed = np.zeros(n)
                    for i in range(len(hist_x)):
                        mixed += alpha[i] * hist_Fx[i]
                    a_anderson = np.maximum(mixed, 0.0)
                    sa = a_anderson.sum()
                    if sa > 1e-10:
                        a_anderson /= sa
                        M_and = float(fast_conv(a_anderson).max())
                        M_new = float(fast_conv(a_new).max())
                        if M_and <= M_new * 1.02:
                            a = a_anderson
                        else:
                            a = a_new
                    else:
                        a = a_new
                else:
                    a = a_new
            except np.linalg.LinAlgError:
                a = a_new
        else:
            a = a_new
        # Update active set
        if it % 5 == 0 and len(active_set) > n:
            ag = fast_conv2(a, g)
            M_cur = float(fast_conv(a).max())
            close_active = np.where(ag > M_cur - 0.05 * M_cur)[0]
            if len(close_active) >= 20:
                active_set = set(int(k) for k in close_active)
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # Determine lengths to explore
    base_n = parent.size
    lengths = sorted(set([256, 384, 512, 640, 768, 896, 1024, 1152, 1280, 1408, 1536, 1792, 2048]))
    # Filter to reasonable range
    lengths = [n for n in lengths if 128 <= n <= 2048]
    print(f"Lengths to try: {lengths}", flush=True)

    # === PHASE 1: Diverse initialization + Adam smooth optimization ===
    p1_budget = 350.0
    p1_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 1: Adam Smooth Optimization ({p1_budget:.0f}s budget)", flush=True)
    print(f"{'=' * 50}", flush=True)

    best_per_length = {}
    n_lengths = len(lengths)
    # Weight time allocation: longer sequences get more time (better bounds)
    # but cost more per iteration
    weights = np.array([1.0 + 0.3 * (n / 1024.0) for n in lengths])
    weights /= weights.sum()
    time_per_length = [p1_budget * w for w in weights]

    for i_n, n in enumerate(lengths):
        t0 = time.monotonic()
        remaining = deadline - 200 - time.monotonic()
        if remaining < 30:
            break
        allot = min(time_per_length[i_n], remaining * 0.4)
        dl = time.monotonic() + allot
        if time.monotonic() >= dl:
            continue

        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 /= a0.sum()
            # Allocate time based on init index
            frac_time = max(0.15, 1.0 / (1.0 + idx * 0.4))
            sub_dl = time.monotonic() + (dl - time.monotonic()) * frac_time

            # Adam optimization
            a = run_adam_smooth(a0, sub_dl, max_iters=2000,
                                tau_start=1.0, tau_end=300.0,
                                lr_init=0.03, lr_end=0.0005)
            M = float(fast_conv(a).max())
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:5d}: M={best_M:.8f} J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: FISTA polish on top candidates ===
    p2_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 2: FISTA Polish", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best_per_length:
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda n: compute_J(best_per_length[n]))
        for n_try in sorted_lengths[:3]:
            if time.monotonic() > deadline - 100:
                break
            a_cand = best_per_length[n_try].copy()
            a_cand = run_fista_smooth(a_cand, min(time.monotonic() + 40, deadline - 80),
                                      max_iters=1000, tau=100.0)
            best.update(a_cand)
            if a_cand.size in best_per_length:
                J_new = compute_J(a_cand)
                J_old = compute_J(best_per_length[a_cand.size])
                if J_new < J_old:
                    best_per_length[a_cand.size] = a_cand.copy()

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Active-constraint LP refinement ===
    p3_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 3: Active-Constraint LP", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        print(f"  Refining n={n_best}", flush=True)
        lp_cache = ConvLPCache(n_best)

        # First: plain active LP
        est_lp_ms = 2.0 * (n_best / 256.0) ** 1.5
        print(f"  Est LP time: {est_lp_ms:.0f}ms", flush=True)
        lp_budget1 = min(400.0, max(50.0, (deadline - 100) - time.monotonic()))
        max_iters1 = max(30, int(lp_budget1 / max(0.005, est_lp_ms / 1000.0)))
        print(f"  LP budget={lp_budget1:.0f}s, max_iters={max_iters1}", flush=True)

        a = run_active_lp(best.seq, min(time.monotonic() + lp_budget1 * 0.5, deadline - 80),
                          max_iters=max_iters1, lp_cache=lp_cache,
                          t_base=0.35, min_t=0.005)
        best.update(a)
        print(f"  After plain LP: J={best.score:.8f}", flush=True)

        # Second: Anderson-accelerated LP
        if time.monotonic() < deadline - 60:
            lp_budget2 = min(300.0, deadline - time.monotonic() - 50)
            max_iters2 = max(20, int(lp_budget2 / max(0.005, est_lp_ms / 1000.0)))
            print(f"  Anderson LP budget={lp_budget2:.0f}s, max_iters={max_iters2}", flush=True)
            a2 = run_anderson_lp(a, min(time.monotonic() + lp_budget2, deadline - 40),
                                 max_iters=max_iters2, lp_cache=lp_cache,
                                 anderson_window=5, t_base=0.25, min_t=0.003)
            best.update(a2)
            print(f"  After Anderson LP: J={best.score:.8f}", flush=True)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Try second-best length with LP ===
    p4_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 4: Second-length LP", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best_per_length and time.monotonic() < deadline - 40:
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda n: compute_J(best_per_length[n]))
        for n_try in sorted_lengths[1:3]:  # Try 2nd and 3rd best
            if time.monotonic() > deadline - 20:
                break
            a_cand = best_per_length[n_try].copy()
            # Quick smooth polish
            a_cand = run_adam_smooth(a_cand, min(time.monotonic() + 20, deadline - 15),
                                     max_iters=500, tau_start=5.0, tau_end=100.0,
                                     lr_init=0.02, lr_end=0.001)
            best.update(a_cand)
            # LP polish
            if time.monotonic() < deadline - 15:
                n_c = a_cand.size
                lp_cache2 = ConvLPCache(n_c)
                est_ms2 = 2.0 * (n_c / 256.0) ** 1.5
                a_lp = run_active_lp(a_cand, deadline - 10,
                                     max_iters=max(10, int(20 / max(0.005, est_ms2 / 1000.0))),
                                     lp_cache=lp_cache2, t_base=0.3, min_t=0.01)
                best.update(a_lp)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Final multi-restart with best structure ===
    p5_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 5: Final Restarts + Polish", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 10:
        n_best = best.seq.size
        # Try a few perturbed restarts
        for restart in range(3):
            if time.monotonic() > deadline - 8:
                break
            a_r = np.maximum(best.seq + rng.normal(0, 0.02 * (restart + 1), n_best), 0.001)
            a_r /= a_r.sum()
            a_r = run_adam_smooth(a_r, min(time.monotonic() + 15, deadline - 5),
                                  max_iters=300, tau_start=10.0, tau_end=200.0,
                                  lr_init=0.01, lr_end=0.001)
            best.update(a_r)
        # Final LP polish
        if time.monotonic() < deadline - 3:
            lp_cache_final = ConvLPCache(best.seq.size)
            a_final = run_anderson_lp(best.seq, deadline,
                                      max_iters=30, lp_cache=lp_cache_final,
                                      anderson_window=5, t_base=0.15, min_t=0.002)
            best.update(a_final)

    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Length exploration around best ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        n_best = best.seq.size
        for delta in [-64, -32, 32, 64, 128]:
            if time.monotonic() > deadline - 3:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()
            a_ext = run_adam_smooth(a_ext, min(time.monotonic() + 10, deadline - 2),
                                    max_iters=200, tau_start=5.0, tau_end=100.0,
                                    lr_init=0.02, lr_end=0.005)
            best.update(a_ext)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'=' * 50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'=' * 50}", flush=True)
    return [float(x) for x in best.seq]
