    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def peak_grad(a, m_star):
    """Compute gradient of conv(a,a)[m_star] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        k = m_star - j
        if 0 <= k < n:
            grad[j] = 2.0 * a[k]
    return grad


def peak_directed_refine(a, max_iters=500, time_budget=30.0):

            obj_try = compute_obj(a_try)

def adam_log_opt(a, max_iters=20000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):

    v_adam = np.zeros(n)

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def coordinate_refine(a, time_budget=20.0, seed=0):
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()

    rng = np.random.default_rng(seed)

            break

        for p in peaks:
            i_min = max(0, p - (n - 1))

            for i in range(i_min, i_max + 1):
                if time.monotonic() - start > time_budget:

                if j < 0 or j >= n or j == i:

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def multi_peak_grad(a, peak_indices, peak_weights=None):
    """Compute weighted gradient of sum of top peaks of conv(a,a)."""
    n = len(a)
    if peak_weights is None:
        peak_weights = np.ones(len(peak_indices)) / len(peak_indices)
    grad = np.zeros(n)
    for idx, m in enumerate(peak_indices):
        w = peak_weights[idx]
        for j in range(n):
            k = m - j
            if 0 <= k < n:
                grad[j] += w * 2.0 * a[k]
    return grad
