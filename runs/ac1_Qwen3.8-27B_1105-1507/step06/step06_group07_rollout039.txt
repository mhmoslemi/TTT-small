import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):

    if n == 0:

    if S < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)

        if it % 100 == 0:
            x = x - np.max(x)

        # Check true objective every 200 iterations or every 2 seconds
        if it % 200 == 0 or (time.monotonic() - last_check > 2.0):
            last_check = time.monotonic()
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()

    # Final check
    x_c = x - np.max(x)
    a_final = np.exp(x_c)
    a_final = a_final / np.sum(a_final)
    true_obj = compute_obj(a_final)
    if true_obj < best_true_obj:
        best_true_obj = true_obj
        best_a = a_final.copy()

    return best_a


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0,...,2n-2."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget=30.0):
    """LP-based refinement: find direction reducing peak, then blend."""
    n = len(a)
    start = time.monotonic()
    S = np.sum(a)
    if S < 1e-10:
        return a
    a_norm = a / S

    conv = compute_conv(a_norm)
    M = float(np.max(conv))

    remaining = time_budget - (time.monotonic() - start)
    if remaining < 5:
        return a

    g = solve_lp(a_norm, M, time_limit=max(1, remaining * 0.5))
    if g is None:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        if time.monotonic() - start > time_budget:
            break
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0)
        S_try = np.sum(a_try)
        if S_try < 1e-10:
            continue
        a_try = a_try / S_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def peak_focused_opt(a, time_budget=20.0):
    """Optimize by focusing on reducing the peak position of the convolution."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    S = np.sum(best_a)
    if S < 1e-10:
        return a

    for outer in range(50):
        if time.monotonic() - start > time_budget:
            break

        a_norm = best_a / np.sum(best_a)
        conv = compute_conv(a_norm)
        peak_pos = int(np.argmax(conv))
        peak_val = conv[peak_pos]

        # Find elements contributing to the peak
        i_min = max(0, peak_pos - (n - 1))
        i_max = min(peak_pos, n - 1)

        # Try reducing contributions to the peak
        improved = False
        for i in range(i_min, i_max + 1):
            j = peak_pos - i
            if j < 0 or j >= n:
                continue
            # Try moving mass from the larger to reduce the product
            if best_a[i] > best_a[j]:
                src, dst = i, j
            else:
                src, dst = j, i

            for frac in [0.01, 0.03, 0.05]:
                delta = frac * best_a[src]
                a_try = best_a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                if a_try[src] < 0:
                    continue
                S_try = np.sum(a_try)
                if S_try < 1e-10:
                    continue
                a_try = a_try / S_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break

        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(outer)
            for _ in range(5):
                idx = rng.integers(0, n)
                delta = 0.02 * np.mean(best_a)
                a_try = best_a.copy()
                a_try[idx] += delta
                if a_try[idx] < 0:
                    a_try[idx] = 0
                S_try = np.sum(a_try)
                if S_try < 1e-10:
                    continue
                a_try = a_try / S_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break

        if not improved and outer > 10:
            break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            S = np.sum(parent)
            if S >= 0.01:
                obj = compute_obj(parent)
                print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
                return parent.copy(), obj
    except NameError:
        pass
    except Exception as e:
        print(f"Failed to load parent: {e}")
    return None, np.inf


def propose_candidate(seed=42):
    """
    Main search function combining multiple optimization strategies to minimize
    2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
    """
    start_time = time.monotonic()
    budget = 1500.0
    deadline = start_time + budget - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget:.0f}s")

    # === Phase 1: Intensive Adam on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Adam log-space optimization on parent (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        S = np.sum(a)
        if S > 1e-10:
            a = a / S

        # Progressive beta schedule
        rounds = [
            (5.0, 40.0, 0.06, 0.99997, 80.0),
            (10.0, 80.0, 0.05, 0.99998, 70.0),
            (20.0, 150.0, 0.04, 0.99998, 60.0),
            (50.0, 300.0, 0.03, 0.99999, 50.0),
            (100.0, 500.0, 0.02, 0.99999, 40.0),
            (200.0, 1000.0, 0.01, 0.99999, 30.0),
            (500.0, 2000.0, 0.005, 0.99999, 25.0),
        ]

        for r_idx, (b_s, b_e, lr0, lr_d, alloc) in enumerate(rounds):
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc_actual = min(alloc, remaining * 0.15)
            if alloc_actual < 5:
                break

            a_new = adam_log_opt(a, max_iters=30000, time_budget=alloc_actual,
                                 beta_start=b_s, beta_end=b_e,
                                 lr_start=lr0, lr_decay=lr_d)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  Round {r_idx} (beta {b_s:.0f}->{b_e:.0f}): {best_score:.8f} ***")
            else:
                print(f"  Round {r_idx} (beta {b_s:.0f}->{b_e:.0f}): {obj:.8f} (best={best_score:.8f})")

    # === Phase 2: Diverse initial conditions ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse initial conditions")
    print(f"{'='*60}")

    if best_seq is not None:
        n_base = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)

        initial_conditions = []

        # 1. Raised cosine
        rc = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n_base) / (n_base - 1)))
        rc = np.maximum(rc, 0.01)
        initial_conditions.append(('cosine', rc / np.sum(rc)))

        # 2. Triangular
        tri = np.concatenate([np.linspace(0, 1, n_base // 2 + 1),
                              np.linspace(1, 0, n_base - n_base // 2 - 1)])
        tri = np.maximum(tri, 0.001)
        initial_conditions.append(('triangular', tri / np.sum(tri)))

        # 3. Gaussian centered
        sigma = n_base * 0.2
        centers = np.arange(n_base) - n_base / 2
        gauss = np.exp(-centers ** 2 / (2 * sigma ** 2))
        initial_conditions.append(('gaussian', gauss / np.sum(gauss)))

        # 4. Bathtub (low center, high edges)
        bat = 1.0 - 0.5 * np.cos(2 * np.pi * np.arange(n_base) / (n_base - 1))
        bat = np.maximum(bat, 0.01)
        initial_conditions.append(('bathtub', bat / np.sum(bat)))

        # 5. Random positive
        rand = np.abs(rng.normal(1.0, 0.5, n_base))
        initial_conditions.append(('random', rand / np.sum(rand)))

        # 6. Parent with perturbation
        pert = a_norm * np.exp(rng.normal(0, 0.1, n_base))
        initial_conditions.append(('parent_pert', pert / np.sum(pert)))

        # 7. Two-peak structure
        peak1 = int(n_base * 0.25)
        peak2 = int(n_base * 0.75)
        twopk = np.zeros(n_base)
        for i in range(n_base):
            twopk[i] = np.exp(-(i - peak1) ** 2 / (2 * (n_base * 0.1) ** 2)) + \
                       np.exp(-(i - peak2) ** 2 / (2 * (n_base * 0.1) ** 2))
        twopk = np.maximum(twopk, 0.001)
        initial_conditions.append(('two_peak', twopk / np.sum(twopk)))

        for name, a_init in initial_conditions:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.1, 30.0)
            a_opt = adam_log_opt(a_init, max_iters=20000, time_budget=alloc,
                                 beta_start=10.0, beta_end=500.0,
                                 lr_start=0.04, lr_decay=0.99998)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  {name}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  {name}: {obj:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 200), max(200, n_best - 100),
            n_best,
            n_best + 100, n_best + 200, n_best + 400,
            500, 800, 1000, 1200, 1500
        ]))
        n_options = [n for n in n_options if 200 <= n <= 2000]

        for n_try in n_options:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_seq(best_seq, n_try)
            S = np.sum(a_try)
            if S > 1e-10:
                a_try = a_try / S

            alloc = min(remaining * 0.12, 40.0)
            a_opt = adam_log_opt(a_try, max_iters=20000, time_budget=alloc,
                                 beta_start=10.0, beta_end=400.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 4: LP refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 4: LP refinement")
    print(f"{'='*60}")

    if best_seq is not None and len(best_seq) <= 900:
        for r_idx in range(4):
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.15, 35.0)
            a_lp = lp_refine(best_seq, time_budget=alloc)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp.copy()
                print(f"  LP {r_idx}: {best_score:.8f} ***")
            else:
                print(f"  LP {r_idx}: {obj:.8f}")

            # Adam after LP
            if time.monotonic() < deadline - 30:
                remaining = deadline - time.monotonic()
                a_post = adam_log_opt(a_lp, max_iters=15000,
                                      time_budget=min(remaining * 0.1, 20.0),
                                      beta_start=50.0, beta_end=800.0,
                                      lr_start=0.02, lr_decay=0.99999)
                obj_post = compute_obj(a_post)
                if obj_post < best_score:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    print(f"  LP+Adam {r_idx}: {best_score:.8f} ***")

    # === Phase 5: Peak-focused optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Peak-focused optimization")
    print(f"{'='*60}")

    if best_seq is not None:
        if time.monotonic() < deadline - 40:
            remaining = deadline - time.monotonic()
            a_pf = peak_focused_opt(best_seq, time_budget=min(remaining * 0.2, 30.0))
            obj_pf = compute_obj(a_pf)
            if obj_pf < best_score:
                best_score = obj_pf
                best_seq = a_pf.copy()
                print(f"  Peak-focused: {best_score:.8f} ***")
            else:
                print(f"  Peak-focused: {obj_pf:.8f}")

    # === Phase 6: Random restarts with diverse strategies ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Random restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)

        for restart in range(10):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            strategy = restart % 4
            if strategy == 0:
                # Multiplicative noise
                scale = 0.02 + 0.01 * restart
                a_pert = a_norm * np.exp(rng.normal(0, scale, n_best))
            elif strategy == 1:
                # Blend with cosine
                alpha = 0.1 + 0.05 * (restart % 3)
                cos = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n_best) / (n_best - 1)))
                cos = np.maximum(cos, 0.01) / np.sum(np.maximum(cos, 0.01))
                a_pert = (1 - alpha) * a_norm + alpha * cos
            elif strategy == 2:
                # Blend with bathtub
                alpha = 0.1 + 0.05 * (restart % 3)
                bat = 1.0 - 0.5 * np.cos(2 * np.pi * np.arange(n_best) / (n_best - 1))
                bat = np.maximum(bat, 0.01) / np.sum(np.maximum(bat, 0.01))
                a_pert = (1 - alpha) * a_norm + alpha * bat
            else:
                # Sparse perturbation
                a_pert = a_norm.copy()
                num_spikes = max(3, n_best // 30)
                for _ in range(num_spikes):
                    idx = rng.integers(0, n_best)
                    a_pert[idx] += rng.uniform(0.05, 0.2) * np.mean(a_norm)

            a_pert = np.maximum(a_pert, 1e-10)
            S_p = np.sum(a_pert)
            if S_p < 1e-10:
                continue
            a_pert = a_pert / S_p

            alloc = min(remaining * 0.1, 20.0)
            a_opt = adam_log_opt(a_pert, max_iters=15000, time_budget=alloc,
                                 beta_start=10.0, beta_end=600.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart} (strat={strategy}): {best_score:.8f} ***")
            else:
                print(f"  Restart {restart} (strat={strategy}): {obj:.8f}")

    # === Phase 7: Final intensive polish with very high beta ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        # Multiple rounds of high-beta Adam
        for polish_round in range(3):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            beta_s = 200.0 * (polish_round + 1)
            beta_e = 1000.0 * (polish_round + 1)
            lr = 0.003 / (polish_round + 1)

            a_final = adam_log_opt(best_seq, max_iters=15000,
                                   time_budget=min(remaining * 0.25, 25.0),
                                   beta_start=beta_s, beta_end=beta_e,
                                   lr_start=lr, lr_decay=0.99999)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Polish {polish_round} (beta {beta_s:.0f}-{beta_e:.0f}): {best_score:.8f} ***")
            else:
                print(f"  Polish {polish_round} (beta {beta_s:.0f}-{beta_e:.0f}): {obj_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
