"""

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_max_gradient(a, tau):
    """
    Compute smooth gradient of max(conv(a,a)) using log-sum-exp.
    Returns (grad, smooth_max_value).
    The gradient of conv[k] w.r.t. a[i] is 2*a[k-i] if 0<=k-i<n else 0.
    So the smoothed gradient is: sum_k w[k] * 2*a[k-i] (with bounds).
    """
    n = len(a)
    c = fast_conv(a)
    c_shifted = c - c.max()
    exp_c = np.exp(tau * c_shifted)
    w = exp_c / exp_c.sum()
    smooth_val = c.max() + np.log(exp_c.sum()) / tau

    # Gradient: grad[i] = sum_k w[k] * 2*a[k-i] where 0<=k-i<n
    # This is equivalent to: grad = 2 * (w * a_rev)[n-1+i] for appropriate shifts
    # Let's compute via convolution:
    # grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
    # Let j = k-i, so k = j+i, and 0<=j<n, 0<=j+i<=2n-2
    # grad[i] = 2 * sum_{j=max(0,-i)}^{min(n-1, 2n-2-i)} w[j+i] * a[j]
    # This is a cross-correlation of w and a.
    
    # Using FFT: grad[i] = 2 * (w * flip(a))[n-1+i] ... need to be careful
    # Actually: sum_j w[j+i]*a[j] is the (i-th lag) correlation of w and a.
    # corr[i] = sum_j w[j+i]*a[j] for i = -(n-1), ..., n-1
    # We want corr for i = 0, ..., n-1 (since k-i must be in [0,n-1] and k in [0,2n-2])
    
    # Simpler approach using FFT:
    # Let w_ext = w (length 2n-1), a_ext = a (length n)
    # conv(w, a)[n-1+i] = sum_j w[n-1+i-j]*a[j]
    # We want sum_j w[j+i]*a[j] = conv(flipped_w, a)[?]
    
    # Let's just do it directly with a different FFT trick:
    # reversed_a = a[::-1] (length n)
    # conv(w, reversed_a)[n-1+i] = sum_j w[n-1+i-j] * reversed_a[j] = sum_j w[n-1+i-j] * a[n-1-j]
    # Let j' = n-1-j: sum_{j'} w[i+j'] * a[j']  (where j' = n-1-j, so j = n-1-j')
    # = sum_{j': 0<=j'<n} w[i+j'] * a[j']
    # We need i+j' in [0, 2n-2], which is always true for i in [0,n-1] and j' in [0,n-1].
    
    size = 1
    while size < 3 * n:
        size <<= 1
    w_fft = np.fft.rfft(w, size)
    ra_fft = np.fft.rfft(a[::-1], size)
    conv_w_ra = np.fft.irfft(w_fft * ra_fft, size)
    
    grad = np.zeros(n)
    for i in range(n):
        idx = n - 1 + i
        if 0 <= idx < len(conv_w_ra):
            grad[i] = 2.0 * conv_w_ra[idx]
    
    return grad, smooth_val, w


def nesterov_smoothed_descent(a, max_iters, deadline, tau_init=50.0, tau_end=500.0,
                               lr_init=0.01, momentum_init=0.9):

    Nesterov-accelerated smoothed gradient descent on the simplex.

    m = 0.9  # momentum
    
    no_improve = 0
    t_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t_count += 1
        frac = it / max(1, max_iters)
        tau = tau_init + (tau_end - tau_init) * frac
        lr = lr_init / (1.0 + 0.05 * t_count ** 0.5)
        momentum = min(0.999, momentum_init + 0.0999 * frac)
        
        # Nesterov lookahead
        y = x + (momentum - 1.0) * (x - x_prev)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-10:
            y = best_a.copy()
        else:
            y = y / ys
        
        # Compute smoothed gradient at y
        grad, smooth_val, w = smooth_max_gradient(y, tau)
        
        # Project gradient to simplex tangent
        grad = grad - np.mean(grad)
        
        # Update
        x_prev = x.copy()
        x_new = y - lr * grad
        x_new = project_simplex(x_new)
        
        obj_new = compute_J(x_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Perturb and restart

                x = project_simplex(best_a + rng.normal(0, 0.01, n))
                x_prev = x.copy()
                no_improve = 0
        
        x = x_new
    
    return best_a, best_obj

def build_lp_matrix(n, a, c):
    """Build the LP constraint matrix efficiently for SCA-LP."""
    # Variables: g[0..n-1], t (index n)
    # Constraints: for k=0..2n-2: sum_j 2*a[k-j]*g[j] - t <= c[k]
    # where the sum is over j such that 0<=k-j<n
    
    rows = []
    cols = []
    vals = []
    b_ub = []
    
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
        b_ub.append(c[k])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1)).tocsr()
    return A, b_ub


def sca_lp_polish(a, max_iters, deadline, time_per_lp=10.0):
    """SCA-LP polish: iterative linearization with LP solves."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        remaining_lp = time.monotonic() + time_per_lp
        if remaining_lp > deadline:
            remaining_lp = deadline

        c = fast_conv(a)
        M = c.max()

        # Build LP
        try:
            A, b_ub = build_lp_matrix(n, a, c)
            c_obj = np.zeros(n + 1)
            c_obj[n] = 1.0

            A_eq = coo_matrix((np.ones(n), ([0] * n, list(range(n)))),
                             shape=(1, n + 1)).tocsr()
            b_eq = [1.0]
            bounds = [(0, None)] * n + [(-np.inf, np.inf)]

            r = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs',
                       options={'time_limit': max(1, int(deadline - time.monotonic())),
                                'presolve': True})
            
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    found = False
                    for frac in [1.0, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try = a_try / a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_obj


def multi_start_inits(n, rng, parent=None):
    """Generate diverse initial sequences for multi-start."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.02, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))
        # Symmetrized versions
        sym_p = 0.5 * (p + p[::-1])
        sym_p = project_simplex(sym_p)
        inits.append(sym_p)
        inits.append(project_simplex(sym_p + rng.normal(0, 0.03, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns (these tend to flatten the auto-convolution)
    for freq in [1, 2, 3, 4, 5, 7]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t + np.pi / 3)))

    # Quadratic/cubic shapes
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(project_simplex((1 - np.abs(2 * t - 1)) ** alpha))

    # Double bump
    for center in [0.25, 0.5, 0.75]:
        for width in [0.05, 0.1, 0.2]:
            inits.append(project_simplex(
                np.exp(-((t - center) / width) ** 2) +
                np.exp(-((t - (1 - center)) / width) ** 2) + 0.01))

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    # "Ideal" flat-top approximation: try to equalize the convolution
    # Start with uniform and add corrections
    if parent is not None:
        p = resample_seq(parent, n)
        p = p / max(p.sum(), 1e-10)
        # Add small oscillations at various frequencies
        for freq in [2, 3, 5, 8, 13]:
            for amp in [0.05, 0.1, 0.15]:
                inits.append(project_simplex(p + amp * np.sin(2 * np.pi * freq * t)))

    return inits


def optimize_single(a, deadline, verbose=False):
    """
    Full optimization pipeline for a single sequence:
    1. Coordinate descent
    2. Nesterov smoothed gradient
    3. LP polish
    4. Final CD polish
    """

    if a.sum() < 1e-10:
        a = np.ones(n) / n
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def check_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 1: Coordinate descent (35% of time)
    t1 = time.monotonic() + 0.35 * remaining
    if verbose:
        print(f"    CD stage (budget {t1 - time.monotonic():.0f}s)", flush=True)
    a_cd = coord_descent_fast(a, max_sweeps=50000, deadline=t1, seed=42)
    check_update(a_cd)
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 2: Nesterov smoothed gradient (30% of time)
    t2 = time.monotonic() + 0.30 * remaining
    if verbose:
        print(f"    Nesterov stage (budget {t2 - time.monotonic():.0f}s)", flush=True)
    a_nest, obj_nest = nesterov_smoothed_descent(a, max_iters=20000, deadline=t2,
                                                  tau_init=30.0, tau_end=300.0, lr_init=0.005)
    if obj_nest < best_obj:
        best_a = a_nest.copy()
        best_obj = obj_nest
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 3: LP polish (25% of time)
    t3 = time.monotonic() + 0.25 * remaining
    if verbose:
        print(f"    LP stage (budget {t3 - time.monotonic():.0f}s)", flush=True)
    a_lp, obj_lp = sca_lp_polish(a, max_iters=100, deadline=t3, time_per_lp=15.0)
    if obj_lp < best_obj:
        best_a = a_lp.copy()
        best_obj = obj_lp
    a = best_a.copy()

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Stage 4: Final CD polish (10% of time)
    t4 = time.monotonic() + 0.10 * remaining
    a_final = coord_descent_fast(a, max_sweeps=50000, deadline=t4, seed=123)
    check_update(a_final)

    return best_a, best_obj

        print(f"  Parent optimized: J={obj_opt:.8f}", flush=True)

    # Try diverse inits at same length
    remaining_p1 = phase1_deadline - time.monotonic()
    if remaining_p1 > 60:
        inits = multi_start_inits(n_parent, rng, parent)
        # Skip inits that are too similar to parent
        p_norm = parent / max(parent.sum(), 1e-10)
        filtered_inits = []
        for init in inits:
            init_s = init.sum()
            if init_s < 1e-10:
                continue
            init_n = init / init_s
            dist = np.linalg.norm(init_n - p_norm) / np.sqrt(n_parent)
            if dist > 0.005:  # Only keep sufficiently different inits
                filtered_inits.append(init)
        
        n_to_try = min(len(filtered_inits), 6)
        per_init_time = (remaining_p1 - 20) / max(1, n_to_try)
        
        for idx in range(n_to_try):
            rem = phase1_deadline - time.monotonic()
            if rem < 30:
                break
            alloc = min(per_init_time, rem - 10)
            if alloc < 25:
                break
            
            a0 = filtered_inits[idx]
            a0 = np.maximum(a0, 0)
            if a0.sum() < 1e-10:
                continue
            a0 = a0 / a0.sum()
            
            print(f"  Init {idx} (budget {alloc:.0f}s)...", flush=True)
            a_opt, obj_opt = optimize_single(a0, time.monotonic() + alloc, verbose=False)
            update_best(a_opt)
            print(f"    -> J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Try nearby lengths ===
    phase2_deadline = t_start + 1000.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try lengths that are close to parent and "nice"
    candidate_ns = []
    for delta in [-64, -32, -16, -8, -4, -2, -1, 1, 2, 4, 8, 16, 32, 64]:
        nn = n_parent + delta
        if 128 <= nn <= 3000 and nn != n_parent:
            candidate_ns.append(nn)
    # Also try some round numbers
    for nn in [512, 640, 768, 896, 1024, 1152, 1280]:
        if 128 <= nn <= 3000 and nn != n_parent and nn not in candidate_ns:
            candidate_ns.append(nn)
    
    candidate_ns = list(set(candidate_ns))
    candidate_ns.sort(key=lambda x: abs(x - n_parent))
    
    for new_n in candidate_ns[:4]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 80:
            break
        alloc = min(120.0, remaining_check - 30)
        if alloc < 50:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = optimize_single(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"    -> J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Final intensive polish ===
    phase3_deadline = deadline
    remaining = phase3_deadline - time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Final intensive polish (budget {remaining:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining > 30:
        a_curr = best_seq.copy()
        
        # Multiple alternating passes
        n_passes = min(5, int(remaining // 60))
        for p in range(n_passes):
            rem = phase3_deadline - time.monotonic()
            if rem < 30:
                break
            
            pass_budget = rem / (n_passes - p)
            sub_deadline = time.monotonic() + pass_budget - 5
            
            # Alternate between Nesterov and CD
            if p % 2 == 0:
                print(f"  Pass {p}: Nesterov (budget {pass_budget:.0f}s)", flush=True)
                a_nest, obj_nest = nesterov_smoothed_descent(
                    a_curr, max_iters=50000, deadline=sub_deadline,
                    tau_init=20.0 + 10 * p, tau_end=200.0 + 50 * p, lr_init=0.003)
                if obj_nest < best_score:
                    best_a = a_nest.copy()
                    best_obj = obj_nest
                    # Verify with evaluate_sequence
                    score = evaluate_sequence(best_a.tolist())
                    if np.isfinite(score) and score < best_score:
                        best_seq = best_a.copy()
                        best_score = score
                        print(f"    Improved: J={score:.8f}", flush=True)
                a_curr = best_seq.copy()
            else:
                print(f"  Pass {p}: CD (budget {pass_budget:.0f}s)", flush=True)
                a_cd = coord_descent_fast(a_curr, max_sweeps=100000,
                                          deadline=sub_deadline, seed=100 + p)
                score = evaluate_sequence(a_cd.tolist())
                if np.isfinite(score) and score < best_score:
                    best_seq = a_cd.copy()
                    best_score = score
                    print(f"    Improved: J={score:.8f}", flush=True)
                a_curr = best_seq.copy()

        # Final LP if time permits
        rem = phase3_deadline - time.monotonic()
        if rem > 60:
            print(f"  Final LP (budget {rem:.0f}s)", flush=True)
            a_lp, obj_lp = sca_lp_polish(best_seq, max_iters=50, deadline=phase3_deadline - 10,
                                          time_per_lp=15.0)
            score = evaluate_sequence(a_lp.tolist())
            if np.isfinite(score) and score < best_score:
                best_seq = a_lp.copy()
                best_score = score
                print(f"    LP improved: J={score:.8f}", flush=True)
            
            # One more CD after LP
            rem = phase3_deadline - time.monotonic()
            if rem > 30:
                a_final = coord_descent_fast(best_seq, max_sweeps=50000,
                                             deadline=phase3_deadline - 5, seed=999)
                score = evaluate_sequence(a_final.tolist())
                if np.isfinite(score) and score < best_score:
                    best_seq = a_final.copy()
                    best_score = score
                    print(f"    Final CD improved: J={score:.8f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
