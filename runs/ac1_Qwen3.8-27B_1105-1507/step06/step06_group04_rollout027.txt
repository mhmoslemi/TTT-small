Peak-reduction search using block coordinate descent, improved LP alternating,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

    s = np.sum(a)

def game_theoretic_optimize(a_init, max_iters=200, time_budget=60):

        # Actually, a^T T_k a = sum_i sum_j a[i] * T_k[i,j] * a[j]

            return 2 * Q @ x

    for it in range(max_iters):

            break
        
        # Compute FFT and power spectrum
        A = np.fft.fft(a)
        P = np.abs(A)**2  # power spectrum, length n

        P_shift = P / tau
        P_shift -= P_shift.max()
        weights = np.exp(P_shift)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(n)
            weights[np.argmax(P)] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: d/d a[j] [sum_k weights[k] * P[k]]
        # = sum_k weights[k] * 2 * Re(A[k] * exp(2*pi*i*k*j/n))
        # = 2 * Re(ifft(weights * A)) ... let me verify
        # ifft(X)[j] = (1/n) * sum_k X[k] * exp(2*pi*i*k*j/n)
        # So n * ifft(weights * A)[j] = sum_k weights[k] * A[k] * exp(2*pi*i*k*j/n)
        # And Re of that = sum_k weights[k] * Re(A[k] * exp(2*pi*i*k*j/n))
        # = sum_k weights[k] * Re(A[k]) * cos(2*pi*k*j/n) + Im(A[k]) * sin(2*pi*k*j/n)

        # So d P_smooth / d a[j] = sum_k weights[k] * 2 * Re(A[k] * exp(2*pi*i*k*j/n))
        # = 2 * n * Re(ifft(weights * A))
        
        X = weights * A
        grad = 2.0 * n * np.real(np.fft.ifft(X))
        
        # Center the gradient (project onto sum=const)
        grad -= np.mean(grad)
        
        # Momentum step
        momentum = mu * momentum + grad
        a_new = a - lr * momentum
        a_new = normalize_simplex(a_new)
        
        # Evaluate
        obj = compute_obj(a_new)
        if obj < best_obj:

            best_a = a_new.copy()
            a = a_new

            lr *= 0.99
        else:
            lr = min(lr * 1.01, 0.1)
        
        # Slowly decrease temperature for sharper focus
        tau *= 0.9998
        tau = max(tau, 0.005)
    
    return best_a, best_obj

"""
Peak-reduction search combining frequency-domain smoothed optimization,
Hankel-matrix gradient descent, LP alternating, and adaptive multi-peak
subgradient methods. Uses game-theoretic minimax perspective with
Hankel matrix structure for efficient gradient computation.
Explores sequence lengths around the parent and uses diverse
initializations to escape local optima.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_maxsum_lp(fixed_seq, M, time_limit=10):
    """Solve: max sum(g) s.t. (fixed_seq * g)[k] <= M, g >= 0."""
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_minpeak_lp(fixed_seq, time_limit=10):
    """Solve: min t s.t. (fixed_seq * g)[k] <= t, sum(g)=1, g>=0."""
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def hankel_gradient_optimize(a, max_iters=5000, time_budget=60.0):
    """
    Uses the Hankel matrix structure Q[i,j] = p[i+j] where p is the
    softmax-weighted autocorrelation. Computes gradient of sum_k p[k]*c[k]
    efficiently using the fact that it's a correlation.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = 0.02
    lr = 0.04
    mu = 0.82
    momentum = np.zeros(n)

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (valid indices)
        # = 2 * cross_correlation(weights, a)[i]
        # Using FFT: ifft(fft(a, N) * conj(fft(weights, N)))[i] = sum_j a[j]*weights[i+j]
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        A_ext = rfft(a, fft_size)
        W_ext = rfft(w_ext, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]

        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        momentum = mu * momentum + grad / grad_norm
        a_new = a - lr * momentum
        a_new = normalize_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.003, 0.1)
        else:
            lr *= 0.994
            a = best_a.copy()

        tau *= 0.9998
        if tau < 0.002:
            tau = 0.002

    return best_a, best_obj


def multi_peak_subgradient(a, num_peaks=5, max_iters=3000, time_budget=50.0):
    """
    Subgradient using top-K peaks with adaptive weighting.
    More aggressive peak reduction by focusing on multiple peaks simultaneously.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    mu = 0.8

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)

        threshold = cmax - 0.05 * cmax

        weights = np.zeros(len(conv))

        if w_sum < 1e-15:

            weights /= w_sum

        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        A_ext = rfft(a, fft_size)
        W_ext = rfft(w_ext, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)

def block_coordinate_refine(a, block_size=7, max_sweeps=20, time_budget=40.0):
    """
    Block coordinate descent: sweep over blocks of consecutive elements,
    optimizing each block via local gradient descent while keeping others fixed.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for block_start in range(0, n, block_size):
            if time.monotonic() - start > time_budget - 2:
                break

            block_end = min(block_start + block_size, n)
            block_len = block_end - block_start

            # Local optimization of this block via a few gradient steps
            for _ in range(50):
                conv = compute_conv(a)
                cmax = float(np.max(conv))
                # Compute gradient for the block only
                # grad[i] = 2 * sum_k (conv_k / cmax_soft) * a[k-i]
                soft_w = (conv / (0.01 * cmax)) ** 4
                sw_sum = soft_w.sum()
                if sw_sum < 1e-15:
                    break
                soft_w /= sw_sum

                # Compute gradient via correlation
                fft_size = 1
                while fft_size < 2 * n - 1:
                    fft_size <<= 1
                w_ext = np.zeros(fft_size)
                w_ext[:len(soft_w)] = soft_w
                A_ext = rfft(a, fft_size)
                W_ext = rfft(w_ext, fft_size)
                corr = irfft(A_ext * np.conj(W_ext), fft_size)
                grad = 2.0 * corr[:n]

                # Only update the block
                block_grad = grad[block_start:block_end]
                # Center the block gradient w.r.t. the block sum
                a[block_start:block_end] -= 0.01 * block_grad
                a[block_start:block_end] = np.maximum(a[block_start:block_end], 0)

            # Renormalize
            a = normalize_simplex(a)
            obj = compute_obj(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                improved = True

        if not improved and sweep > 3:
            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def generate_diverse_starts(n, seed=42, num_starts=10):

        freq = 1 + rng.integers(0, 6)
        amp = 0.3 + 0.4 * rng.random()
        phase = 2 * np.pi * rng.random()
        base = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        s = base * mod
        s = np.maximum(s, 0.001)
        starts.append(s / np.sum(s))

    return starts


def polish_sequence(a, time_budget=60.0):
    """
    Full polishing pipeline alternating between frequency-domain optimization,
    Hankel gradient, multi-peak subgradient, LP alternating, and block refinement.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)

    methods = [freq_smooth_optimize, hankel_gradient_optimize, multi_peak_subgradient]
    method_names = ['freq', 'hankel', 'multi-peak']
    idx = 0
    stall_global = 0

    while time.monotonic() - start < time_budget - 3:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # Allocate time: gradient methods get more time, LP gets less
        if idx % len(methods) == 0 and n <= 900 and remaining > 30:
            # Use LP every 3 rounds
            alloc = min(remaining * 0.3, 40.0)
            a_new, obj_new = lp_alternate_enhanced(best_a, max_iters=25, time_budget=alloc)
        else:
            alloc = min(remaining * 0.35, 35.0)
            method = methods[idx % len(methods)]
            a_new, obj_new = method(best_a, time_budget=alloc)

        idx += 1

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_global = 0
        else:
            stall_global += 1
            if stall_global > 3:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 2**31)
                scale = 0.01 + 0.01 * (stall_global % 4)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize_simplex(a_pert)
                # Quick optimize
                a_pert, obj_pert = freq_smooth_optimize(a_pert, max_iters=500, time_budget=5.0)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                stall_global = 0

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Loaded parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines frequency-domain smoothed optimization,
    Hankel-matrix gradient descent, multi-peak subgradient, LP alternating,
    and block coordinate refinement with diverse initialization and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # Phase 1: Quick screening of lengths
    print("\n=== Phase 1: Length screening ===")
    n_values = sorted(set([n_parent, n_parent - 100, n_parent - 50,
                           n_parent + 50, n_parent + 100,
                           n_parent - 200, n_parent + 200]))
    n_values = [x for x in n_values if 400 <= x <= 1200]

    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Generate starts
        if n == n_parent and best_seq is not None:
            starts = [normalize_simplex(best_seq.copy())]
            starts.extend(generate_diverse_starts(n, seed=seed, num_starts=5))
        else:
            if best_seq is not None:
                starts = [normalize_simplex(resample_sequence(best_seq, n))]
            starts.extend(generate_diverse_starts(n, seed=seed + n, num_starts=5))

        alloc_per_start = min(remaining / len(starts), 15.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 250:
                break
            t_alloc = min(alloc_per_start, 12.0)
            if t_alloc < 3:
                break

            # Quick optimization with frequency method
            a_opt, obj_opt = freq_smooth_optimize(s, max_iters=int(t_alloc * 8), time_budget=t_alloc * 0.5)

            # Quick multi-peak
            a_opt2, obj_opt2 = multi_peak_subgradient(a_opt, num_peaks=4, max_iters=int(t_alloc * 4), time_budget=t_alloc * 0.3)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")

            candidates.append((obj_opt, a_opt.copy()))

    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive optimization of top candidates
    print("\n=== Phase 2: Intensive optimization ===")
    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = candidates[ci][1].copy()
        n_cand = len(a_cand)
        alloc = min(remaining * 0.35, 120.0)

        print(f"  Candidate {ci} (n={n_cand}, obj={candidates[ci][0]:.8f}): polishing with {alloc:.0f}s")
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"  No improvement (got {obj_pol:.8f})")

    # Phase 3: Block coordinate refinement on the best
    print(f"\n=== Phase 3: Block refinement (best: {best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.25, 50.0)
        n_best = len(best_seq)
        a_br, obj_br = block_coordinate_refine(best_seq, block_size=5,
                                               max_sweeps=int(alloc / 2), time_budget=alloc)
        if obj_br < best_score:
            best_score = obj_br
            best_seq = a_br.copy()
            print(f"  Block refine: {best_score:.8f}")
        else:
            print(f"  Block refine: {obj_br:.8f} (no improvement)")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.5, 100.0)
        a_final, obj_final = polish_sequence(best_seq, time_budget=alloc)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish: {best_score:.8f}")
        else:
            print(f"  Final polish: {obj_final:.8f}")

    # Phase 5: Perturbation search on the final best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_final = len(best_seq)
        rng = np.random.default_rng(seed + 77777)

        for perturb_idx in range(15):
            if time.monotonic() > deadline - 10:
                break
            scale = 0.005 + 0.005 * (perturb_idx % 5)
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_pert = normalize_simplex(a_pert)

            t_alloc = min(remaining * 0.1, 8.0)
            if t_alloc < 2:
                break

            a_pert, obj_pert = freq_smooth_optimize(a_pert, max_iters=int(t_alloc * 10), time_budget=t_alloc)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {perturb_idx} (s={scale:.3f}): ** {best_score:.8f} **")

    # Phase 6: Final LP on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        if n_best <= 800 and remaining > 20:
            a_lp, obj_lp = lp_alternate_enhanced(best_seq, max_iters=20, time_budget=min(remaining * 0.5, 30.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
