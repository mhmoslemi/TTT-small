```python
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=20):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s  # normalized to sum=1
    except Exception:
        pass
    return None

def smooth_obj_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    
    return obj, grad_obj

def adam_log_opt(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_centered = x - np.max(x)
        a_cur = np.exp(x_centered)
        a_cur = a_cur / np.sum(a_cur)
        
        obj_sm, grad_a = smooth_obj_grad(a_cur, beta)
        grad_x = grad_a * a_cur
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    
    return best_a
```

```python
"""
Iterative LP-based peak refinement with Adam log-space fine-tuning for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Key improvements over previous: aggressive LP iteration,
adaptive M shrinking, dual-gap monitoring, multi-n exploration, and subgradient fallback.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=15):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 2000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_dual(a, M, time_limit=15):
    """Solve dual: min M*sum(lambda) s.t. A^T @ lambda >= 1, lambda >= 0."""
    n = len(a)
    if n > 2000:
        return None
    A = build_toeplitz(a)
    c = M * np.ones(2 * n - 1)
    A_ub = -A.T
    b_ub = -np.ones(n)
    bounds = [(0.0, None)] * (2 * n - 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return float(res.fun)
    except Exception:
        pass
    return None


def smooth_obj_grad(a, beta):
    """Compute smoothed objective (logsumexp of beta*conv) and its gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj


def adam_log_opt(a, max_iters=10000, time_budget=60.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    x = np.log(a)
    x -= np.mean(x)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        x_centered = x - np.max(x)
        a_cur = np.exp(x_centered)
        a_cur = a_cur / np.sum(a_cur)
        obj_sm, grad_a = smooth_obj_grad(a_cur, beta)
        grad_x = grad_a * a_cur
        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        if it % 200 == 0:
            x -= np.mean(x)
        if it % 80 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_check.copy()
    return best_a


def subgradient_step(a, step_size):
    """Frank-Wolfe style: move towards the vertex that reduces the peak."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    # Subgradient: s[i] = 2*a[k_star - i] for valid i
    s = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            s[i] = 2.0 * a[j]
    # Frank-Wolfe: move towards argmin(s)
    j_min = int(np.argmin(s))
    a_new = a.copy()
    a_new -= step_size * a
    a_new[j_min] += step_size
    a_new = np.maximum(a_new, 0)
    s_new = np.sum(a_new)
    if s_new < 1e-10:
        return a
    return a_new / s_new


def iterative_lp_refine(a, time_budget=200.0, verbose=True):
    """
    Main LP refinement: iteratively solve LP and take steps.
    Uses adaptive M shrinking and line search.
    """
    n = len(a)
    if n > 1500:
        return a
    start = time.monotonic()
    best_a = a.copy().astype(np.float64)
    s = np.sum(best_a)
    if s < 1e-10:
        return a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    best_conv = compute_conv(best_a)
    best_M = float(np.max(best_conv))
    no_improve_count = 0
    shrink_factor = 1.0

    for iteration in range(200):
        if time.monotonic() - start > time_budget:
            break
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        # Compute current peak
        conv = compute_conv(best_a)
        M = float(np.max(conv))

        # Adaptive M: start at M, gradually shrink
        M_lp = M * shrink_factor

        # Solve LP
        lp_time = min(10.0, remaining * 0.3)
        g = solve_lp(best_a, M_lp, time_limit=lp_time)
        if g is None:
            # Try with slightly larger M
            g = solve_lp(best_a, M * 1.01, time_limit=lp_time)
            if g is None:
                no_improve_count += 1
                if no_improve_count > 3:
                    break
                continue

        # Line search: try multiple blend factors
        found = False
        t_values = [0.3, 0.5, 0.7, 0.8, 1.0, 0.2, 0.4, 0.6, 0.1, 0.05, 0.02, 0.01]
        for t in t_values:
            a_try = (1 - t) * best_a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                best_conv = compute_conv(a_try)
                new_M = float(np.max(best_conv))
                no_improve_count = 0
                # If M decreased significantly, reset shrink
                if new_M < M * 0.95:
                    shrink_factor = 1.0
                found = True
                break

        if not found:
            # Try smaller steps
            for t in [0.005, 0.002, 0.001]:
                a_try = (1 - t) * best_a + t * g
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    no_improve_count = 0
                    found = True
                    break

        if not found:
            no_improve_count += 1
            # Gradually shrink M to force more aggressive direction
            shrink_factor = max(0.85, shrink_factor * 0.98)
            if no_improve_count > 8:
                # Reset and try subgradient
                shrink_factor = 1.0
                for _ in range(5):
                    a_sg = subgradient_step(best_a, 0.1)
                    obj_sg = compute_obj(a_sg)
                    if obj_sg < best_obj - 1e-12:
                        best_obj = obj_sg
                        best_a = a_sg
                if no_improve_count > 12:
                    break

        if verbose and iteration % 10 == 0:
            print(f"    LP iter {iteration}: obj={best_obj:.10f}, M={M:.6f}, shrink={shrink_factor:.3f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def coordinate_refine(a, time_budget=20.0, seed=0):
    """Fine-tune by adjusting pairs of coordinates that contribute to peaks."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(best_a)
        # Find top 3 peaks
        top_peaks = np.argsort(conv)[::-1][:3]
        improved = False
        for p in top_peaks:
            i_min = max(0, p - (n - 1))
            i_max = min(p, n - 1)
            for i in range(i_min, i_max + 1):
                j = p - i
                if j < 0 or j >= n or j == i:
                    continue
                for delta_frac in [0.002, 0.005, 0.01, 0.02, -0.005, -0.01]:
                    delta = delta_frac * max(best_a[i], best_a[j])
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
            if improved:
                break
        if not improved and it > 30:
            break
    return best_a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: aggressive iterative LP refinement combined with Adam log-space
    optimization, multi-n exploration, coordinate descent, and structured restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n={n_parent}, budget={budget_s}s")

    # === Phase 1: Aggressive LP refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Aggressive LP refinement (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # First: moderate LP refinement
        lp_budget1 = min(300.0, (deadline - time.monotonic()) * 0.35)
        a_lp = iterative_lp_refine(a, time_budget=lp_budget1, verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  After LP round 1: {best_score:.10f} ***")
        else:
            print(f"  After LP round 1: {obj:.10f} (no improvement)")

        # Adam fine-tune after LP
        if time.monotonic() < deadline - 60:
            rem = deadline - time.monotonic()
            a_adam = adam_log_opt(best_seq, max_iters=20000, time_budget=min(rem * 0.15, 60.0),
                                  beta_start=10.0, beta_end=200.0,
                                  lr_start=0.03, lr_decay=0.99998)
            obj_adam = compute_obj(a_adam)
            if obj_adam < best_score:
                best_score = obj_adam
                best_seq = a_adam.copy()
                print(f"  After Adam: {best_score:.10f} ***")

            # Second LP round after Adam
            if time.monotonic() < deadline - 50:
                rem = deadline - time.monotonic()
                a_lp2 = iterative_lp_refine(best_seq, time_budget=min(rem * 0.25, 150.0),
                                            verbose=False)
                obj_lp2 = compute_obj(a_lp2)
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  After LP round 2: {best_score:.10f} ***")
                else:
                    print(f"  After LP round 2: {obj_lp2:.10f} (no improvement)")

                # Final Adam after second LP
                if time.monotonic() < deadline - 40:
                    rem = deadline - time.monotonic()
                    a_final1 = adam_log_opt(best_seq, max_iters=15000,
                                            time_budget=min(rem * 0.15, 40.0),
                                            beta_start=50.0, beta_end=500.0,
                                            lr_start=0.02, lr_decay=0.99999)
                    obj_f1 = compute_obj(a_final1)
                    if obj_f1 < best_score:
                        best_score = obj_f1
                        best_seq = a_final1.copy()
                        print(f"  After final Adam: {best_score:.10f} ***")

    # === Phase 2: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 100), n_best - 50, n_best,
            n_best + 50, n_best + 100, n_best + 200
        ]))
        n_options = [n for n in n_options if 200 <= n <= 1500 and n != n_best]

        for n_try in n_options:
            if time.monotonic() > deadline - 60:
                break
            rem = deadline - time.monotonic()
            if rem < 20:
                break

            a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

            # Quick LP on resampled
            lp_alloc = min(rem * 0.15, 40.0)
            a_lp_res = iterative_lp_refine(a_try, time_budget=lp_alloc, verbose=False)
            obj_res = compute_obj(a_lp_res)

            # Quick Adam on best of original/LP
            a_base = a_lp_res if obj_res < compute_obj(a_try) else a_try
            adam_alloc = min(rem * 0.2, 30.0)
            a_opt = adam_log_opt(a_base, max_iters=10000, time_budget=adam_alloc,
                                 beta_start=10.0, beta_end=300.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj_opt = compute_obj(a_opt)

            final_obj = min(obj_res, obj_opt)
            final_seq = a_lp_res if obj_res < obj_opt else a_opt

            if final_obj < best_score:
                best_score = final_obj
                best_seq = final_seq.copy()
                print(f"  n={n_try}: {best_score:.10f} *** UPDATE")
            else:
                print(f"  n={n_try}: {final_obj:.10f}")

    # === Phase 3: Third LP round on the global best ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Final LP polish (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 40:
        rem = deadline - time.monotonic()
        n_cur = len(best_seq)
        if n_cur <= 1200:
            a_polish = iterative_lp_refine(best_seq, time_budget=min(rem * 0.3, 100.0),
                                           verbose=False)
            obj_polish = compute_obj(a_polish)
            if obj_polish < best_score:
                best_score = obj_polish
                best_seq = a_polish.copy()
                print(f"  LP polish: {best_score:.10f} ***")
            else:
                print(f"  LP polish: {obj_polish:.10f} (no improvement)")

    # === Phase 4: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Coordinate descent")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 25:
        rem = deadline - time.monotonic()
        a_cd = coordinate_refine(best_seq, time_budget=min(rem * 0.2, 30.0), seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord descent: {best_score:.10f} ***")
        else:
            print(f"  Coord descent: {obj_cd:.10f} (no improvement)")

    # === Phase 5: Structured restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Structured restarts")
    print(f"{'='*60}")

    if best_seq is not None:
        rng = np.random.default_rng(seed)
        n_best = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)

        restarts = 6
        for r in range(restarts):
            if time.monotonic() > deadline - 20:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break

            if r == 0:
                # Blend with Hann window
                hann = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n_best) / max(n_best - 1, 1)))
                hann = hann / np.sum(hann)
                a_pert = 0.8 * a_norm + 0.2 * hann
            elif r == 1:
                # Blend with triangular
                tri = np.ones(n_best)
                mid = n_best // 2
                for i in range(n_best):
                    tri[i] = 1.0 - abs(i - mid) / (mid + 1)
                tri = np.maximum(tri, 0)
                tri = tri / np.sum(tri)
                a_pert = 0.8 * a_norm + 0.2 * tri
            elif r == 2:
                # Multiplicative noise
                scale = 0.05
                a_pert = a_norm * np.exp(rng.normal(0, scale, n_best))
            elif r == 3:
                # Sparse perturbation
                a_pert = a_norm.copy()
                for _ in range(n_best // 30):
                    idx = int(rng.integers(0, n_best))
                    a_pert[idx] += 0.05 * np.mean(a_norm)
            elif r == 4:
                # Gaussian window
                sigma = n_best / 4.0
                x = np.arange(n_best) - n_best / 2.0
                gauss = np.exp(-x**2 / (2 * sigma**2))
                gauss = gauss / np.sum(gauss)
                a_pert = 0.7 * a_norm + 0.3 * gauss
            else:
                # Blackman window
                blackman = np.blackman(n_best)
                blackman = blackman / np.sum(blackman)
                a_pert = 0.75 * a_norm + 0.25 * blackman

            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / np.sum(a_pert)

            # Quick optimization
            alloc = min(rem * 0.12, 20.0)
            a_opt = adam_log_opt(a_pert, max_iters=8000, time_budget=alloc,
                                 beta_start=10.0, beta_end=400.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {r}: {best_score:.10f} *** UPDATE")
            else:
                print(f"  Restart {r}: {obj:.10f}")

    # === Final: one more LP + Adam cycle ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        rem = deadline - time.monotonic()
        n_cur = len(best_seq)
        if n_cur <= 1200:
            a_final = iterative_lp_refine(best_seq, time_budget=min(rem * 0.3, 30.0),
                                          verbose=False)
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Final LP: {best_score:.10f} ***")

            if time.monotonic() < deadline - 8:
                rem2 = deadline - time.monotonic()
                a_last = adam_log_opt(best_seq, max_iters=5000,
                                      time_budget=min(rem2 * 0.4, 15.0),
                                      beta_start=100.0, beta_end=800.0,
                                      lr_start=0.01, lr_decay=0.99999)
                obj_last = compute_obj(a_last)
                if obj_last < best_score:
                    best_score = obj_last
                    best_seq = a_last.copy()
                    print(f"  Final Adam: {best_score:.10f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
```
