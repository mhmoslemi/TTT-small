    # This is the (m-th) element of the "anti-convolution" or equivalently:
    # If we define v as the time-reversal of a: v[j] = a[n-1-j]
    # Then cc[m] = sum_j v[n-1-j] * w[j+m] = conv(v, w)[n-1+m]
    # 
    # Let me just compute it directly for clarity (n=700 is small enough)
    # Actually for efficiency, let's use the FFT approach properly.
    
    # The cross-correlation (a ⋆ w)[m] = sum_j a[j] * conj(w[j-m])
    # We want sum_j a[j] * w[j+m] = (a ⋆ w_rev)[m] where w_rev[k] = w[-k]
    # 
    # In FFT: ifft(conj(FFT(a)) * FFT(w))[m] = sum_j conj(a[j]) * w[m-j] (circular)
    # 
    # Let me just be explicit:
    # dC_da[m] = 2 * C^(1-p) * sum_{j=0}^{n-1} a[j] * w[j+m]
    # where the sum is over j such that j+m < 2n-1 (which is always true for j < n, m < n, since j+m < 2n)
    
    # Use FFT: define the linear cross-correlation properly
    # sum_j a[j] * w[j+m] for m = 0, 1, ..., n-1
    # Let b = reverse of a: b[j] = a[n-1-j]
    # Then sum_j a[j] * w[j+m] = sum_j b[n-1-j] * w[j+m]
    # Let i = n-1-j: sum_i b[i] * w[n-1-i+m] = conv(b, w)[n-1+m]
    
    b = a[::-1]
    size2 = 1
    while size2 < n + (2 * n - 1) - 1:
        size2 <<= 1
    B = rfft(b, size2)
    W2 = rfft(w, size2)
    conv_bw = irfft(B * W2, size2)
    # conv_bw[n-1+m] = sum_i b[i] * w[n-1-i+m] = sum_j a[j] * w[j+m]
    cross_corr = conv_bw[n - 1: n - 1 + n]

    C_factor = 2.0 * C ** (1.0 - p)
    grad_C = C_factor * cross_corr

    # dobj/da = (2n/S^2) * (grad_C - 2*C*a/S)
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_C - 2.0 * C * a / sum_a)

    return obj, grad_obj


def log_space_lbfgs(a_init, p=50, max_iters=500, time_budget=60.0):
    """
    L-BFGS optimization in log-space with power smoothing.
    x = log(a) / centering normalization since objective is scale-invariant.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a_init
    a = a / s

    # Work in log-space, centered
    x0 = np.log(a)
    x0 -= np.mean(x0)

    def objective_and_grad(x):
        # Convert to a
        xc = x - np.max(x)
        a_local = np.exp(xc)
        a_local = a_local / np.sum(a_local)

        obj, grad_a = compute_grad_smooth(a_local, p)
        if not np.isfinite(obj):
            return 1e10, np.zeros(n)

        # Chain rule: grad_x = grad_a * a
        grad_x = grad_a * a_local
        return obj, grad_x

    start = time.monotonic()
    best_x = x0.copy()
    best_obj = compute_obj(a)

    try:
        result = minimize(
            objective_and_grad, x0, method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iters,
                'ftol': 1e-15,
                'gtol': 1e-12,
                'maxls': 50,
                'maxfun': max_iters * 5,
            }
        )
        if result.success or result.fun < best_obj:
            x_final = result.x
            xc = x_final - np.max(x_final)
            a_final = np.exp(xc)
            a_final = a_final / np.sum(a_final)
            true_obj = compute_obj(a_final)
            if true_obj < best_obj:
                best_obj = true_obj
                best_x = x_final.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    # Check sub-iterations via callback wasn't available, so do periodic check
    # Convert final
    xc = best_x - np.max(best_x)
    a_result = np.exp(xc)
    a_result = a_result / np.sum(a_result)
    return a_result


def subgradient_opt(a_init, max_iters=2000, time_budget=30.0, lr=0.002, momentum=0.9):
    """
    Subgradient optimization with momentum for the unsmoothed max objective.
    Works in log-space to maintain positivity.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    if s < 1e-10:
        return a_init
    a = a / s

    x = np.log(a)
    x -= np.mean(x)
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        xc = x - np.max(x)
        a_local = np.exp(xc)
        a_local = a_local / np.sum(a_local)

        conv = compute_conv(a_local)
        max_b = float(np.max(conv))
        S = float(np.sum(a_local))
        obj = 2.0 * n * max_b / (S * S)

        if obj < best_obj:
            best_obj = obj
            best_a = a_local.copy()

        # Subgradient of max(conv): find all near-max positions
        threshold = max_b * 0.999
        peak_indices = np.where(conv >= threshold)[0]

        # Average subgradient over near-peak positions (smoothes the subgradient)
        grad = np.zeros(n)
        for k in peak_indices:
            # d conv[k] / d a[m] = 2 * a[k-m] if 0 <= k-m < n
            m_range = np.arange(max(0, k - n + 1), min(k + 1, n))
            j_range = k - m_range
            grad[m_range] += 2.0 * a_local[j_range]
        grad /= len(peak_indices)

        # Full gradient of objective
        grad_obj = (2.0 * n / (S * S)) * (grad - 2.0 * max_b * a_local / S)

        # Chain rule for log-space
        grad_x = grad_obj * a_local

        # Momentum update
        v = momentum * v - lr * grad_x
        x = x + v

        # Periodic centering
        if it % 100 == 0:
            x -= np.mean(x)

        # Adaptive lr decay
        if it % 200 == 0 and it > 0:
            lr *= 0.8

    return best_a


def reweighting_opt(a_init, max_iters=500, time_budget=30.0):
    """
    Iterative reweighting: focus optimization on current peak positions.
    At each iteration, weight the convolution elements by their proximity to max,
    then do a gradient step on the weighted sum of squares.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    lr = 0.01
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_b = float(np.max(conv))
        S = float(np.sum(a))
        obj = 2.0 * n * max_b / (S * S)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Create weights peaked at max positions
        # w[k] = exp(-alpha * (max_b - conv[k]) / max_b)
        alpha = 5.0 + 5.0 * (it / max_iters)  # increasing focus
        w = np.exp(-alpha * (max_b - conv) / (max_b + 1e-30))
        w = w / np.sum(w)

        # Gradient of sum(w[k] * conv[k]^2) w.r.t. a
        # d sum_k w[k]*conv_k^2 / d a[m] = 2 * sum_k w[k] * conv[k] * 2 * a[k-m]
        # = 4 * sum_k w[k] * conv[k] * a[k-m] (for valid k-m)
        wconv = 2.0 * w * conv  # absorb the factor of 2
        # cross_corr[m] = sum_j a[j] * wconv[j+m]
        b = a[::-1]
        size = 1
        while size < n + len(wconv) - 1:
            size <<= 1
        B = rfft(b, size)
        WC = rfft(wconv, size)
        cc = irfft(B * WC, size)
        cross_corr = cc[n - 1: n - 1 + n]

        grad = 4.0 * cross_corr

        # Also add normalization penalty gradient
        grad = grad - 2.0 * 4.0 * float(np.sum(wconv)) * a / S  # approx

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm

        # Update in log-space
        x = np.log(np.maximum(a, 1e-15))
        x = x - lr * grad * a
        x -= np.max(x)
        a_new = np.exp(x)
        a_new = a_new / np.sum(a_new)

        # Accept if better or with annealing
        obj_new = compute_obj(a_new)
        if obj_new < obj:
            a = a_new
        elif it < max_iters * 0.3:
            # Allow some uphill moves early
            if np.random.random() < 0.3:
                a = a_new

        # Decay lr
        lr *= 0.999

    return best_a


def symmetric_project(a):
    """Project to symmetric: a[i] = (a[i] + a[n-1-i]) / 2."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    return a_sym


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    a_new = np.maximum(a_new, 0)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def multi_p_lbfgs(a_init, time_budget=120.0):
    """Try L-BFGS with multiple power values, keep the best."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    p_values = [20, 50, 100, 200, 500]

    for p in p_values:
        if time.monotonic() - start > time_budget - 10:
            break
        alloc = min((time_budget - (time.monotonic() - start)) * 0.4, 30.0)
        a_try = log_space_lbfgs(best_a, p=p, max_iters=200, time_budget=alloc)
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_a = a_try.copy()
            print(f"    p={p}: {obj:.8f} ***")
        else:
            print(f"    p={p}: {obj:.8f}")

    return best_a


def basin_hopping(a_init, time_budget=60.0, n_restarts=5, seed=42):
    """Basin hopping: perturb and re-optimize."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)

    start = time.monotonic()
    for r in range(n_restarts):
        if time.monotonic() - start > time_budget - 5:
            break
        alloc = (time_budget - (time.monotonic() - start)) / (n_restarts - r)
        alloc = min(alloc, 20.0)

        # Perturb
        strategy = r % 3
        if strategy == 0:
            # Gaussian noise in log-space
            scale = 0.05 + 0.03 * r
            x = np.log(np.maximum(best_a, 1e-15))
            x += rng.normal(0, scale, n)
            x -= np.max(x)
            a_pert = np.exp(x)
            a_pert = a_pert / np.sum(a_pert)
        elif strategy == 1:
            # Random coordinate swaps
            a_pert = best_a.copy()
            for _ in range(n // 20):
                i, j = rng.integers(0, n, 2)
                delta = rng.uniform(-0.05, 0.05) * np.mean(best_a)
                a_pert[i] = max(0, best_a[i] - delta)
                a_pert[j] = max(0, best_a[j] + delta)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s
            else:
                a_pert = best_a.copy()
        else:
            # Blend with structured sequence
            alpha = rng.uniform(0.05, 0.2)
            # Try different structures
            struct_choice = rng.integers(0, 4)
            if struct_choice == 0:
                # Triangular
                structured = np.abs(1 - 2 * np.arange(n) / (n - 1))
            elif struct_choice == 1:
                # Cosine bump
                structured = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
            elif struct_choice == 2:
                # Two bumps
                structured = np.exp(-((np.arange(n) - n * 0.25) ** 2) / (n * 0.05)) + \
                             np.exp(-((np.arange(n) - n * 0.75) ** 2) / (n * 0.05))
            else:
                # Flat top
                structured = np.ones(n)
                structured[:n // 4] = 0.5
                structured[-n // 4:] = 0.5

            structured = np.maximum(structured, 0.01)
            structured = structured / np.sum(structured)
            a_pert = (1 - alpha) * (best_a / np.sum(best_a)) + alpha * structured
            a_pert = a_pert / np.sum(a_pert)

        # Quick optimization
        a_opt = log_space_lbfgs(a_pert, p=100, max_iters=100, time_budget=alloc * 0.6)
        a_opt2 = subgradient_opt(a_opt, max_iters=500, time_budget=alloc * 0.4)

        obj = compute_obj(a_opt2)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt2.copy()
            print(f"    BH restart {r}: {obj:.8f} ***")
        else:
            print(f"    BH restart {r}: {obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization combining L-BFGS with
    power smoothing, subgradient methods, reweighting, basin hopping,
    and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        # Normalize
        s = np.sum(best_seq)
        if s > 1e-10:
            best_seq = best_seq / s

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n={n_parent}, budget={budget_s}s")
    print(f"Deadline in {deadline - start_time:.1f}s")

    # === Phase 1: L-BFGS with multiple p values on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-p L-BFGS (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a = best_seq.copy()
        remaining = deadline - time.monotonic()
        a_new = multi_p_lbfgs(a, time_budget=min(remaining * 0.2, 120.0))
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 1 result: {best_score:.8f} ***")
        else:
            print(f"  Phase 1 result: {obj:.8f} (no improvement)")

    # === Phase 2: Subgradient optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Subgradient + momentum (n={len(best_seq) if best_seq is not None else '?'})")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        a_sub = subgradient_opt(best_seq, max_iters=3000,
                                time_budget=min(remaining * 0.1, 40.0),
                                lr=0.003, momentum=0.92)
        obj = compute_obj(a_sub)
        if obj < best_score:
            best_score = obj
            best_seq = a_sub.copy()
            print(f"  Subgradient: {best_score:.8f} ***")
        else:
            print(f"  Subgradient: {obj:.8f}")

    # === Phase 3: Reweighting optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Iterative reweighting")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        a_rw = reweighting_opt(best_seq, max_iters=1000,
                               time_budget=min(remaining * 0.08, 30.0))
        obj = compute_obj(a_rw)
        if obj < best_score:
            best_score = obj
            best_seq = a_rw.copy()
            print(f"  Reweighting: {best_score:.8f} ***")
        else:
            print(f"  Reweighting: {obj:.8f}")

    # === Phase 4: Symmetric optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Symmetric sequence optimization")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        # Try symmetric version
        a_sym = symmetric_project(best_seq)
        a_sym = a_sym / np.sum(a_sym)
        obj_sym = compute_obj(a_sym)
        print(f"  Symmetric projection: {obj_sym:.8f}")

        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()

        # Optimize symmetric version
        a_sym_opt = log_space_lbfgs(a_sym, p=100, max_iters=200,
                                    time_budget=min(remaining * 0.08, 30.0))
        # Re-symmetrize after optimization
        a_sym_opt = symmetric_project(a_sym_opt)
        a_sym_opt = a_sym_opt / np.sum(a_sym_opt)
        obj_sym_opt = compute_obj(a_sym_opt)
        print(f"  Symmetric optimized: {obj_sym_opt:.8f}")

        if obj_sym_opt < best_score:
            best_score = obj_sym_opt
            best_seq = a_sym_opt.copy()

    # === Phase 5: Multi-n exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        # Try a range of n values
        n_options = []
        for dn in [-100, -50, -20, 0, 20, 50, 100, 150, 200]:
            n_try = n_best + dn
            if 300 <= n_try <= 1500:
                n_options.append(n_try)
        n_options = sorted(set(n_options))

        for n_try in n_options:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
                a_try = np.maximum(a_try, 0)
                s = np.sum(a_try)
                if s > 1e-10:
                    a_try = a_try / s
                else:
                    continue

            alloc = min(remaining * 0.1, 30.0)
            a_opt = log_space_lbfgs(a_try, p=50, max_iters=150, time_budget=alloc)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 6: Basin hopping ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Basin hopping")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        a_bh = basin_hopping(best_seq, time_budget=min(remaining * 0.2, 120.0),
                             n_restarts=8, seed=seed)
        obj = compute_obj(a_bh)
        if obj < best_score:
            best_score = obj
            best_seq = a_bh.copy()
            print(f"  Basin hopping: {best_score:.8f} ***")
        else:
            print(f"  Basin hopping: {obj:.8f}")

    # === Phase 7: Final intensive L-BFGS with high p ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final high-p L-BFGS polish")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        for p in [200, 500, 1000]:
            if time.monotonic() > deadline - 10:
                break
            alloc = min(remaining * 0.15, 25.0)
            a_pol = log_space_lbfgs(best_seq, p=p, max_iters=300, time_budget=alloc)
            obj = compute_obj(a_pol)
            if obj < best_score:
                best_score = obj
                best_seq = a_pol.copy()
                print(f"  p={p}: {best_score:.8f} ***")
            else:
                print(f"  p={p}: {obj:.8f}")

    # === Phase 8: Subgradient final polish ===
    print(f"\n{'='*60}")
    print(f"Phase 8: Final subgradient polish")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        a_final = subgradient_opt(best_seq, max_iters=5000,
                                  time_budget=min(remaining * 0.3, 60.0),
                                  lr=0.001, momentum=0.95)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final subgrad: {best_score:.8f} ***")
        else:
            print(f"  Final subgrad: {obj:.8f}")

    # === Phase 9: One last L-BFGS ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        a_last = log_space_lbfgs(best_seq, p=300, max_iters=200,
                                 time_budget=min(remaining * 0.5, 20.0))
        obj = compute_obj(a_last)
        if obj < best_score:
            best_score = obj
            best_seq = a_last.copy()
            print(f"  Last L-BFGS: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    s = np.sum(best_seq)
    if s < 0.01:
        best_seq = best_seq / s * 0.01

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
