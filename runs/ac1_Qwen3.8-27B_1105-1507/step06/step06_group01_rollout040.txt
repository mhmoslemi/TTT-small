"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse initialization with multiple sequence lengths
2. FISTA on log-sum-exp smooth max approximation
3. Cutting-plane LP optimization (main workhorse)

5. Final polish with adaptive strategies

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_max = np.max(rho[cond])
    else:
        rho_max = 1
    theta = (cssv[rho_max - 1] - 1) / rho_max
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def lse_max_and_grad(a, tau):
    """Compute log-sum-exp smooth max of conv(a,a) and its gradient."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_b = np.exp(shifted)
    s = exp_b.sum()
    p = exp_b / s
    smooth_val = b_max + np.log(s) / tau
    
    # Gradient: grad[i] = 2 * sum_{k} p[k] * a[k-i] (where valid)
    n = len(a)
    grad = np.zeros(n)
    conv_len = 2 * n - 1
    for i in range(n):
        # p[k] * a[k-i] for k where 0 <= k-i < n
        # k ranges from i to i+n-1
        lo = i
        hi = min(i + n, conv_len)
        if lo < hi:
            grad[i] = 2.0 * np.dot(p[lo:hi], a[:hi-lo])
    return smooth_val, grad


def fista_smooth(a0, tau, max_iters, deadline, lr=None):
    """FISTA on smooth max approximation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr is None:
        lr = 1.0 / (4.0 * tau)  # Lipschitz-based step
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        _, grad_y = lse_max_and_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Adaptive step with backtracking
        step = lr
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        if (it + 1) % 10 == 0:
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    M = fast_conv(a).max()
    if M < best_M:
        best_a = a.copy()
    
    return best_a


def build_lp_matrix(a, active_indices):
    """Build sparse constraint matrix for LP: (a*g)[k] <= M for k in active_indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(active_indices):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(ii)
                cols.append(i)
                vals.append(a[j])
    if not rows:
        return None, None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A, np.array(active_indices)


def solve_lp(a, M, active_indices):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    matrix, _ = build_lp_matrix(a, active_indices)
    if matrix is None or matrix.nnz == 0:
        return None
    
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=matrix, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_optimize(a0, deadline, n_active_init=50, max_iters=500):

    a = np.maximum(a0.astype(np.float64), 0.0)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Initial active set
    b = fast_conv(a)
    M = b.max()
    threshold = M * 0.95
    active = np.where(b >= threshold)[0].tolist()

        if time.monotonic() > deadline:
            break
        
        # Solve LP

                no_improve += 1
                if no_improve > 5:
                    break
                continue
        
        g_sum = g.sum()
        g_norm = g / g_sum
        
        # Line search
        improved = False
        for t in [t_base, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:

            if s_t < 1e-10:

                best_a = a.copy()

            active = np.argsort(b)[::-1][:min(len(active) * 2 + 10, conv_len)].tolist()

def subgradient_optimize(a0, deadline, max_iters=10000):
    """Nesterov accelerated subgradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Nesterov's dual averaging
    a_avg = a.copy()
    a_curr = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at current point
        b = fast_conv(a_curr)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a_curr[j]
        grad_tan = grad - grad.mean()
        
        # Step size
        step = 1.0 / np.sqrt(it + 1)
        
        # Update
        a_curr = a_curr - step * grad_tan
        a_curr = proj_simplex(a_curr)
        
        # Running average
        a_avg = ((it) * a_avg + a_curr) / (it + 1)
        
        # Check improvement
        if (it + 1) % 20 == 0:
            M_check = fast_conv(a_avg).max()
            if M_check < best_M:
                best_M = M_check
                best_a = a_avg.copy()
            M_curr = fast_conv(a_curr).max()
            if M_curr < best_M:
                best_M = M_curr
                best_a = a_curr.copy()
    
    return best_a


def coordinate_descent_1d(a, i, b):
    """Optimize a[i] via golden section."""
    n = len(a)
    x_cur = a[i]
    
    # Affected k range
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c_vals.append(b[k] - x_cur * x_cur)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c_vals.append(b[k] - 2.0 * x_cur * a[j])
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.5)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def coordinate_descent_full(a0, deadline, max_sweeps=100, seed=0):
    """Full coordinate descent."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = np.arange(n)
        rng.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b)
            diff = x_opt - a[i]
            if abs(diff) > 1e-10:
                a[i] = x_opt
                # Update b
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - (x_opt - diff) * (x_opt - diff))
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * diff
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved or no_improve > 10:
            break
    
    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = np.interp(t, np.linspace(0, 1, len(parent)), parent)
        p = np.maximum(p, 1e-6)
        inits.append(p / p.sum())
        p2 = np.maximum(p + rng.normal(0, 0.05, n), 1e-6)
        inits.append(p2 / p2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various shapes
    shapes = [
        (t * (1 - t) + 0.1),
        (1.0 / (1.0 + 10.0 * (t - 0.5) ** 2)),
        (0.5 + 0.5 * np.cos(2 * np.pi * t)),
        (0.5 + 0.3 * np.cos(4 * np.pi * t) + 0.2 * np.cos(6 * np.pi * t)),
        np.abs(t - 0.5) ** 0.5 + 0.1,
        1.0 / (np.abs(t - 0.5) + 0.1),
    ]
    for sh in shapes:
        sh = np.maximum(sh, 0.01)
        inits.append(sh / sh.sum())
    
    # Random smooth
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    Combines FISTA, cutting-plane LP, subgradient, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)
    
    # Track best
    best_seq = None
    best_J = float('inf')

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. FISTA warmup with log-sum-exp smooth max
2. Cutting-plane LP optimization with full/adaptive constraint sets (main workhorse)
3. Nesterov subgradient method
4. Coordinate descent with golden section search
5. Multi-length exploration and final polish
Key improvements: full-constraint LP for tighter convergence, better active set
management, more aggressive line search, and multi-start strategy.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if cond.any():
        rho_max = int(np.max(rho[cond]))
    else:
        rho_max = 1
    theta = (cssv[rho_max - 1] - 1.0) / rho_max
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def lse_max_and_grad(a, tau):
    """Compute log-sum-exp smooth max of conv(a,a) and gradient."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_b = np.exp(shifted)
    s = exp_b.sum()
    p = exp_b / s
    smooth_val = b_max + np.log(s) / tau

    n = len(a)
    grad = np.zeros(n)
    conv_len = 2 * n - 1
    for i in range(n):
        lo = i
        hi = min(i + n, conv_len)
        if lo < hi:
            grad[i] = 2.0 * np.dot(p[lo:hi], a[:hi - lo])
    return smooth_val, grad


def fista_smooth(a0, tau, max_iters, deadline, lr=None):
    """FISTA on smooth max approximation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    if lr is None:
        lr = 0.5 / tau

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t = 1.0
    y = a.copy()
    a_prev = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        _, grad_y = lse_max_and_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()

        step = lr / (1.0 + 0.01 * it)
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t = t_new

        if (it + 1) % 20 == 0:
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()

    M = fast_conv(a).max()
    if M < best_M:
        best_a = a.copy()

    return best_a


def solve_lp_full(a, M, n_active_limit=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for selected k, g >= 0.
    
    If n_active_limit is None, use all constraints.
    Returns g (unnormalized) or None.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Determine active set
    b = fast_conv(a)
    if n_active_limit is None or n_active_limit >= conv_len:
        active_idx = list(range(conv_len))
    else:
        # Use top n_active_limit by convolution value
        top_idx = np.argsort(b)[::-1][:n_active_limit]
        active_idx = top_idx.tolist()

    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(active_idx):
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            if 0 <= j < n:
                rows.append(ii)
                cols.append(i)
                vals.append(a[j])

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))

    c = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def solve_lp_active(a, M, active_indices):
    """Solve LP with specific active indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(active_indices):
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            if 0 <= j < n:
                rows.append(ii)
                cols.append(i)
                vals.append(a[j])

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))

    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_optimize(a0, deadline, use_full=True, max_iters=1000):
    """LP-based optimization with cutting plane / adaptive active set.
    
    At each iteration:
    1. Solve LP to find best response g to current a
    2. Check if g violates any non-active constraints
    3. If violated, add those constraints and re-solve
    4. Line search to update a
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    best_a = a.copy()
    best_M = fast_conv(a).max()

    if time.monotonic() > deadline:
        return best_a, best_M

    # Initial active set
    b = fast_conv(a)
    if use_full:
        active = list(range(conv_len))
    else:
        threshold = best_M * 0.9
        active = np.where(b >= threshold)[0].tolist()
        if len(active) < 10:
            active = np.argsort(b)[::-1][:30].tolist()

    no_improve = 0
    dual_gap = float('inf')

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP
        if use_full:
            g = solve_lp_full(a, best_M, n_active_limit=len(active))
        else:
            g = solve_lp_active(a, best_M, active)

        if g is None:
            # Fallback: use full LP
            g = solve_lp_full(a, best_M)
            if g is None:
                no_improve += 1
                if no_improve > 10:
                    break
                continue

        g_sum = g.sum()
        g_norm = g / g_sum

        # The dual gives a lower bound: if g_sum (before normalization) <= 1,
        # then a is optimal (up to tolerance)
        # Actually: the LP max is sum(g). If max sum(g) <= 1, a is optimal.
        # But we normalized a to sum=1, so the feasible set includes a itself.
        # Thus max sum(g) >= 1 always. If max sum(g) = 1, a is optimal.
        dual_gap = g_sum - 1.0

        # Line search to update
        improved = False
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_trial = (1.0 - t) * a + t * g_norm
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break

        if improved:
            # Update active set if not using full
            if not use_full:
                b = fast_conv(a)
                # Add newly active constraints
                threshold = best_M * 0.85
                new_active = set(np.where(b >= threshold)[0].tolist())
                # Also add all previously active
                new_active.update(active)
                # Also add top-K
                top_k = np.argsort(b)[::-1][:50].tolist()
                new_active.update(top_k)
                active = list(new_active)
                if len(active) > conv_len:
                    active = list(range(conv_len))
                    use_full = True
        else:
            no_improve += 1
            if not use_full:
                # Expand active set
                b = fast_conv(a)
                more_active = np.argsort(b)[::-1][:min(len(active) * 2, conv_len)].tolist()
                active = list(set(active + more_active))
                if len(active) >= conv_len * 0.5:
                    use_full = True
                    active = list(range(conv_len))
            if no_improve > 15 or dual_gap < 1e-8:
                break

    return best_a, best_M


def subgradient_nesterov(a0, deadline, max_iters=5000):
    """Nesterov accelerated subgradient method for minimizing max(conv)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    a_d = a.copy()  # Nesterov's dual variable
    best_a_avg = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a_d)
        k_star = int(np.argmax(b))

        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a_d[j]
        grad_tan = grad - grad.mean()

        step = 1.0 / (np.sqrt(2.0) * np.sqrt(it + 1))

        a_d = a_d - step * grad_tan
        a_d = proj_simplex(a_d)

        # Running average (Nesterov's trick)
        a_avg = (it * best_a_avg + a_d) / (it + 1)

        if (it + 1) % 50 == 0:
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:

                best_a = a_avg.copy()
            M_cur = fast_conv(a_d).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a_d.copy()
            best_a_avg = best_a.copy()

    return best_a


def coordinate_1d(a, i, b):
    """Find optimal a[i] keeping others fixed. Golden section on 1D convex function."""
    n = len(a)
    x_cur = a[i]

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c_vals.append(b[k] - x_cur * x_cur)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c_vals.append(b[k] - 2.0 * x_cur * a[j])
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.3)

    # Make sure hi is large enough
    for _ in range(15):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 100.0:
            hi *= 2.0
        else:
            break

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(45):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt) + 1e-15:
        x_opt = 0.0

    return max(0.0, x_opt)


def coordinate_descent(a0, deadline, max_sweeps=50, seed=0):
    """Coordinate descent with random ordering."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = np.arange(n)
        rng.shuffle(order)

        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            x_opt = coordinate_1d(a, i, b)
            diff = x_opt - a[i]
            if abs(diff) > 1e-9:
                old = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = b[k] - old * old + x_opt * x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] = b[k] + 2.0 * a[j] * (x_opt - old)
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved or no_improve > 8:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        1.0 / (1.0 + 8.0 * (t - 0.5) ** 2),
        0.5 + 0.5 * np.cos(2.0 * np.pi * t),
        0.5 + 0.3 * np.cos(4.0 * np.pi * t) + 0.15 * np.cos(6.0 * np.pi * t),
        np.abs(t - 0.5) ** 0.7 + 0.05,
        1.0 / (np.abs(t - 0.5) + 0.15),
        0.3 + 0.7 * np.sin(np.pi * t) ** 2,
    ]
    for sh in shapes:
        sh = np.maximum(sh, 0.01)
        inits.append(sh / sh.sum())

    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: FISTA -> LP -> Subgradient -> Coordinate Descent.
    Tries multiple sequence lengths and initialization strategies.
    """
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_J = float('inf')

    def update_best(a):
        nonlocal best_seq, best_J
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            return
        if np.isfinite(score) and score < best_J:
            best_J = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_J:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(768)
        update_best(parent)
        print(f"No parent, using uniform n=768, J={best_J:.8f}", flush=True)

    n_parent = parent.size

    # === PHASE 1: FISTA warmup (20% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: FISTA warmup (budget: {0.20*total_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_end = t_start + 0.20 * (total_budget - 15)

    # Focus on a few key lengths
    key_lengths = sorted(set([n_parent, 512, 640, 768, 896, 1024]))
    key_lengths = [l for l in key_lengths if 256 <= l <= 2048]

    for n in key_lengths:
        if time.monotonic() > phase1_end - 20:
            break
        inits = generate_inits(n, rng, parent)
        best_M_n = float('inf')
        best_a_n = None

        time_per_init = max(3.0, (phase1_end - time.monotonic() - 10) / max(1, len(inits) * len(key_lengths)))

        for a0 in inits:
            if time.monotonic() > phase1_end - 10:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            # Multi-tau FISTA
            for tau in [80, 200]:
                if time.monotonic() > phase1_end - 10:
                    break
                max_it = max(50, int(time_per_init / 0.03))
                a_opt = fista_smooth(a0, tau, max_it, min(phase1_end - 10, time.monotonic() + time_per_init))
                M_opt = fast_conv(a_opt).max()
                if M_opt < best_M_n:
                    best_M_n = M_opt
                    best_a_n = a_opt.copy()

        if best_a_n is not None:
            update_best(best_a_n)
            print(f"  n={n}: FISTA best M={best_M_n:.10f}", flush=True)

    print(f"Phase 1 complete. Best J={best_J:.8f}", flush=True)

    # === PHASE 2: LP optimization (50% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP optimization (main)", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_end = t_start + 0.70 * (total_budget - 15)

    # Work on the best sequence found so far
    if best_seq is not None:
        a_work = best_seq.copy()
        n_work = best_seq.size
    else:
        a_work = parent.copy()
        n_work = n_parent

    # Ensure right length
    a_work = np.maximum(a_work, 1e-6)
    a_work = a_work / a_work.sum()

    # Run LP with full constraints (most rigorous)
    print(f"  Full-constraint LP on n={n_work}...", flush=True)
    a_lp, M_lp = lp_optimize(a_work, phase2_end - 60, use_full=True, max_iters=500)
    update_best(a_lp)
    print(f"  After full LP: M={M_lp:.10f}", flush=True)

    # Try a different length from best
    if time.monotonic() < phase2_end - 60:
        alt_n = n_work + (128 if n_work < 1024 else -128)
        alt_n = max(256, min(2048, alt_n))
        if alt_n != n_work:
            a_alt = resample_seq(a_lp, alt_n)
            a_alt = np.maximum(a_alt, 1e-6)
            a_alt = a_alt / a_alt.sum()
            t_sub = min(time.monotonic() + 60, phase2_end - 60)
            a_alt_opt, M_alt = lp_optimize(a_alt, t_sub, use_full=True, max_iters=200)
            update_best(a_alt_opt)
            print(f"  Alt length n={alt_n}: M={M_alt:.10f}", flush=True)

    # === PHASE 3: Subgradient refinement (10% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Subgradient refinement", flush=True)
    print(f"{'='*60}", flush=True)

    phase3_end = min(t_start + 0.80 * (total_budget - 15), phase2_end + 0.10 * (total_budget - 15))

    if best_seq is not None:
        a_sub = subgradient_nesterov(best_seq, phase3_end, max_iters=10000)
        update_best(a_sub)
        print(f"  After subgradient: M={fast_conv(a_sub).max():.10f}", flush=True)

    # === PHASE 4: Coordinate descent polish (10% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)

    phase4_end = min(deadline - 60, t_start + 0.90 * (total_budget - 15))

    if best_seq is not None:
        a_cd = coordinate_descent(best_seq, phase4_end, max_sweeps=30, seed=seed + 1)
        update_best(a_cd)
        print(f"  After CD: M={fast_conv(a_cd).max():.10f}", flush=True)

        # One more LP pass on CD result
        if time.monotonic() < phase4_end - 10:
            a_lp2, M_lp2 = lp_optimize(a_cd, phase4_end - 5, use_full=True, max_iters=100)
            update_best(a_lp2)

    # === PHASE 5: Final intensive optimization (remaining time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final intensive LP + CD", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and time.monotonic() < deadline - 10:
        # Alternate LP and CD
        a_final = best_seq.copy()
        n_iter = 0
        while time.monotonic() < deadline - 5 and n_iter < 20:
            # LP step
            t_lp_deadline = min(deadline - 5, time.monotonic() + 30)
            a_final, M_final = lp_optimize(a_final, t_lp_deadline, use_full=True, max_iters=50)
            update_best(a_final)

            # CD step
            t_cd_deadline = min(deadline - 5, time.monotonic() + 20)
            a_final = coordinate_descent(a_final, t_cd_deadline, max_sweeps=5,
                                         seed=seed + 100 + n_iter)
            update_best(a_final)
            n_iter += 1

        print(f"  After {n_iter} LP+CD cycles: M={fast_conv(a_final).max():.10f}", flush=True)

    # Final validation
    if best_seq is not None:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL RESULT: J={final_score:.8f} n={best_seq.size}", flush=True)
        print(f"Time used: {time.monotonic()-t_start:.1f}s / {total_budget}s", flush=True)
        print(f"{'='*60}", flush=True)
    else:
        # Fallback
        best_seq = np.ones(768)
        best_J = compute_J(best_seq)
        print(f"Fallback: J={best_J:.8f} n=768")

    return [float(x) for x in best_seq]
