Multi-strategy optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Spectral equalization via DFT manipulation, (2) Frank-Wolfe with 
away steps using LP dual weights, (3) Adaptive coordinate descent, (4) Multi-scale 
upsampling from optimized shorter sequences. Length 768.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    """Compute max of conv(a,a)."""
    return fast_conv(a).max()

def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = fast_conv(a).max()
    return float(2 * len(a) * M / (s * s))

def spectral_equalize(a, n_iters=10, flatness_target=0.1):
    """Iteratively equalize the power spectrum of a."""
    n = len(a)
    # Use zero-padded DFT for linear convolution
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    A = np.fft.rfft(a, L)
    # |A|^2 is related to the circular autocorrelation of the zero-padded sequence
    # For the unpadded sequence, we want the power spectrum to be flat
    # Use the DFT of the unpadded sequence
    A_unpadded = np.fft.fft(a)
    power = np.abs(A_unpadded)**2
    
    for it in range(n_iters):
        # Target: make power spectrum flatter
        # Use a soft equalization: power_new = power^alpha for alpha < 1
        # Or: redistribute energy from high-power bins to low-power bins
        avg_power = power.mean()
        # Simply clip and redistribute
        cap = avg_power * (1 + flatness_target)
        excess = np.maximum(power - cap, 0).sum()
        deficit_mask = power < cap
        if deficit_mask.sum() == 0:
            break
        deficit_total = (cap - power[deficit_mask]).sum()
        if deficit_total < 1e-10:
            break
        scale = min(1.0, excess / deficit_total)
        power_new = power.copy()
        power_new[power > cap] = cap
        power_new[deficit_mask] += scale * (cap - power[deficit_mask])
        
        # Now reconstruct a from the modified power spectrum
        # We need to maintain the phase
        phases = np.angle(A_unpadded)
        magnitude = np.sqrt(power_new)
        A_new = magnitude * np.exp(1j * phases)
        # Enforce Hermitian symmetry for real signal
        A_new[0] = np.abs(A_new[0])  # DC is real
        if n % 2 == 0:
            A_new[n//2] = np.abs(A_new[n//2])  # Nyquist is real
        # Reconstruct
        a_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        A_unpadded = np.fft.fft(a_new)
        power = np.abs(A_unpadded)**2
    return a_new

"""
Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Spectral equalization with symmetry constraint for initialization,
(2) LP dual-weighted Frank-Wolfe with away steps, (3) Fast LP best-response
with dual formulation, (4) Multi-scale upsample-refine, (5) Exact coordinate
descent polish. Targets length 768 with J < 1.5027.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = fast_conv(a).max()
    return 2.0 * len(a) * M / (s * s)


def proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def spectral_equalize_symmetric(n, n_iters=30, beta=0.3):
    """Create symmetric sequence with flattened power spectrum.
    
    Uses the fact that for symmetric a, the linear convolution peak
    is closely related to the circular autocorrelation, which is
    IDFT(|DFT|^2)/n. Flattening |DFT|^2 flattens the autocorrelation.
    """
    # Start with a symmetric window (raised cosine)
    t = np.arange(n) / float(n)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    # Make it symmetric
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    
    for it in range(n_iters):
        A = np.fft.fft(a)
        power = np.abs(A) ** 2
        # Target: flat power spectrum (except DC)
        target = power[0] / n  # average of non-DC bins
        # Soft equalization: blend current and target magnitude
        mag = np.abs(A)
        target_mag = np.sqrt(target)
        # Don't touch DC
        new_mag = mag.copy()
        for m in range(1, n):
            # Geometric interpolation towards target
            new_mag[m] = mag[m] ** (1 - beta) * target_mag ** beta
        # Reconstruct with original phases
        phases = np.angle(A)
        A_new = new_mag * np.exp(1j * phases)
        # Enforce Hermitian symmetry for real output
        A_new[0] = np.abs(A_new[0])
        if n % 2 == 0:
            A_new[n // 2] = np.abs(A_new[n // 2])
        # Reconstruct and enforce symmetry
        a_new = np.fft.ifft(A_new).real
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def solve_lp_dual(n, a, active_K, M):
    """Solve the dual LP to get constraint weights.
    
    Dual: min M * sum(y)
    s.t. sum_{k in K} y[k] * a[k-j] >= 1 for all j
    y >= 0
    
    Returns (y, optimal_value) or (None, None) if infeasible.
    """
    K = list(active_K)
    n_K = len(K)
    if n_K == 0:
        return None, None
    
    # Build constraint matrix: A[j, idx] = -a[K[idx] - j] if valid
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(K):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(j)
                cols.append(idx)
                vals.append(-a[m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(n, n_K))
    b = -np.ones(n)
    c = M * np.ones(n_K)
    bounds = [(0.0, None)] * n_K
    
    try:
        r = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            y = np.maximum(r.x, 0.0)
            return y, float(r.fun)
    except Exception:
        pass
    return None, None


def lp_best_response(a, M, n_active=30):
    """Solve LP best response using active constraints.
    
    Primal: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0
    Returns (g_normalized, dual_weights) or (None, None).
    """
    n = len(a)
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()
    
    n_K = len(active)
    if n_K == 0:
        return None, None
    
    # Build primal constraint matrix: A[idx, j] = a[k-j]
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(n_K, n))
    if A.nnz == 0:
        return None, None
    
    c = -np.ones(n)
    b_ub = np.full(n_K, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g_norm = g / g_sum
                # Get dual variables (constraints)
                if r.ineqfun is not None:
                    y = np.maximum(-r.ineqfun, 0.0)
                else:
                    y = np.ones(n_K) / n_K
                return g_norm, y
    except Exception:
        pass
    return None, None


def fft_subgradient(a, y, active_K):
    """Compute weighted subgradient: g[j] = 2 * sum_k y[k] * a[k-j].
    
    Uses vectorized slice operations for speed.
    """
    n = len(a)
    g = np.zeros(n)
    for idx, k in enumerate(active_K):
        w = y[idx]
        if w < 1e-12:
            continue
        # g[j] += w * a[k-j] for valid j
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        m_lo = k - j_hi
        m_hi = k - j_lo
        g[j_lo:j_hi + 1] += w * a[m_lo:m_hi + 1]
    return 2.0 * g

def run_lp_iterative(a_init, deadline, n_active_start=20, n_active_max=80):

    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    n_active = n_active_start
    no_improve = 0
    
    while time.monotonic() < deadline:
        g_resp, y_dual = lp_best_response(a, best_M, n_active)
        if g_resp is None:
            no_improve += 1
            n_active = min(n_active * 2, n_active_max)
            if no_improve > 10:
                break
            continue
        
        # Line search for optimal blend
        improved = False
        base_t = 0.3
        for t in [base_t, base_t * 0.7, base_t * 0.5, base_t * 0.3,
                  0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g_resp
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            n_active = min(int(n_active * 1.3), n_active_max)
            if no_improve > 20:
                break
    
    return best_a


def run_mirror_descent_dual(a_init, deadline, n_active=30):
    """Mirror descent (multiplicative weights) with LP dual-weighted subgradient."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    it = 0
    while time.monotonic() < deadline:
        # Get subgradient from LP dual
        b = fast_conv(a)
        M = b.max()
        top_idx = np.argsort(b)[::-1][:n_active]
        
        # Quick dual solve
        y, _ = solve_lp_dual(n, a, top_idx.tolist(), M)
        if y is None:
            y = np.ones(n_active) / n_active
        
        g_sub = fft_subgradient(a, y, top_idx.tolist())
        g_tan = g_sub - g_sub.mean()
        
        # Multiplicative weights update
        step = 0.5 / (1.0 + 0.01 * it)
        log_w = -step * g_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
        it += 1
        
        M_cur = fast_conv(a).max()
        if M_cur < best_M * (1 - 1e-12):
            best_M = M_cur
            best_a = a.copy()
    
    return best_a


def coordinate_descent_exact(a, i, b, n):
    """Find optimal a[i] given all others fixed. Golden section search."""
    x_cur = a[i]
    
    # Affected convolution indices: k in [i, i+n-1] (clamped to [0, 2n-2])
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute: c[k] = base + slope * x + (quad if k == 2*i)
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c_vals.append(b[k] - x_cur * x_cur)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c_vals.append(b[k] - 2.0 * x_cur * a[j])
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x):
        vals = c_arr + s_arr * x
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x * x
            if qv > m:
                m = qv
        return m
    
    # Golden section search
    lo = 0.0
    hi = max(2.0 * x_cur, 0.5, np.sqrt(max(b.max(), 0.0)) + 0.5)
    
    for _ in range(5):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 50.0:
            hi *= 2.0
        else:
            break
    
    if hi <= lo:
        hi = lo + 1.0
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, seed=0):
    """Exact coordinate descent with smart ordering."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = fast_conv(a)
        sweep_improved = False
        
        # Process in order of subgradient magnitude (most important first)
        top_idx = np.argsort(b)[::-1][:min(50, len(b))]
        order = list(top_idx)
        remaining = [i for i in range(n) if i not in set(order)]
        rng_cd.shuffle(remaining)
        order.extend(remaining)
        
        for i in order:
            x_opt = coordinate_descent_exact(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                # Update affected b values
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - old * old
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old)
                sweep_improved = True
        
        a = a / a.sum()
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved or no_improve > 15:
            break
    
    return best_a


def run_multiscale_start(n_target, deadline, rng):
    """Multi-scale: optimize at n/4, upsample, refine at n/2, upsample, refine at n."""
    # Start at n/4
    n_small = max(32, n_target // 4)
    a_small = spectral_equalize_symmetric(n_small, n_iters=30, beta=0.4)
    
    # Quick LP at small scale
    t_small = min(time.monotonic() + 30, deadline - 100)
    a_small = run_lp_iterative(a_small, t_small, n_active_start=10, n_active_max=30)
    
    # Upsample to n/2
    n_mid = max(64, n_target // 2)
    a_mid = resample_seq(a_small, n_mid)
    a_mid = np.maximum(a_mid, 1e-6)
    a_mid = a_mid / a_mid.sum()
    
    t_mid = min(time.monotonic() + 60, deadline - 50)
    a_mid = run_lp_iterative(a_mid, t_mid, n_active_start=15, n_active_max=40)
    
    # Upsample to n_target
    a_full = resample_seq(a_mid, n_target)
    a_full = np.maximum(a_full, 1e-6)
    a_full = a_full / a_full.sum()
    
    return a_full


def run_combined_optimization(a0, n, deadline, seed=0):
    """Run the full optimization pipeline on a single initialization."""
    best = np.maximum(a0, 0.0)
    s = best.sum()
    if s < 1e-10:
        best = np.ones(n) / n
    else:
        best = best / s
    best_M = fast_conv(best).max()
    
    t_start = time.monotonic()
    remaining = deadline - t_start
    
    # Stage 1: Mirror descent with dual weights (30% time)
    t1 = t_start + 0.3 * remaining
    if time.monotonic() < t1:
        a_md = run_mirror_descent_dual(best, t1, n_active=20)
        M_md = fast_conv(a_md).max()
        if M_md < best_M:
            best_M = M_md
            best = a_md
        print(f"    MD: M={best_M:.12f}", flush=True)
    
    # Stage 2: LP iterative (40% time)
    t2 = t_start + 0.7 * remaining
    if time.monotonic() < t2:
        a_lp = run_lp_iterative(best, t2, n_active_start=20, n_active_max=60)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp
        print(f"    LP: M={best_M:.12f}", flush=True)
    
    # Stage 3: FW with away steps (15% time)
    t3 = t_start + 0.85 * remaining
    if time.monotonic() < t3:
        a_fw = run_fw_dual_weighted(best, t3, n_active_start=15, n_active_max=40)
        M_fw = fast_conv(a_fw).max()
        if M_fw < best_M:
            best_M = M_fw
            best = a_fw
        print(f"    FW: M={best_M:.12f}", flush=True)
    
    # Stage 4: Coordinate descent (15% time)
    if time.monotonic() < deadline:
        a_cd = run_coordinate_descent(best, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd
        print(f"    CD: M={best_M:.12f}", flush=True)
    
    return best, best_M


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Multi-phase, multi-start optimization for
    minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2 with n=768.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    
    best_seq = None
    best_score = float('inf')
    
    def track(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - (deadline - float(budget_s) + 30.0)
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    t0 = time.monotonic()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            track(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    # === Phase 0: Multi-scale initialization (60s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 0: Multi-scale initialization", flush=True)
    print(f"{'='*60}", flush=True)
    
    n_target = 768
    if parent is not None and parent.size == n_target:
        n_target = parent.size
    
    t_phase0 = min(time.monotonic() + 60, deadline - 200)
    a_multiscale = run_multiscale_start(n_target, t_phase0, rng)
    track(a_multiscale)
    print(f"  Multi-scale init done: J={best_score:.8f}", flush=True)
    
    # === Phase 1: Multi-start optimization (70% of remaining time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-start optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Generate diverse initializations
    inits = generate_diverse_inits(n_target, rng, parent)
    # Add multi-scale init
    inits.append(a_multiscale)
    # Add parent with perturbation
    if parent is not None:
        for sigma in [0.01, 0.03, 0.05]:
            p_pert = np.maximum(parent + rng.normal(0, sigma, n_target), 1e-6)
            inits.append(p_pert / p_pert.sum())
    
    # Remove duplicates (by J value similarity)
    # Sort by initial J
    init_scores = []
    for a0 in inits:
        M0 = fast_conv(a0).max()
        s0 = a0.sum()
        J0 = 2.0 * n_target * M0 / (s0 * s0) if s0 > 0.01 else float('inf')
        init_scores.append(J0)
    
    # Sort by score, take top diverse set
    order = np.argsort(init_scores)
    # Take top 8 diverse inits
    selected_indices = []
    for idx in order:
        if len(selected_indices) >= 8:
            break
        selected_indices.append(idx)
    
    # Allocate time
    t_phase1_end = deadline - 120  # reserve 120s for final polish
    time_per_init = (t_phase1_end - time.monotonic()) / len(selected_indices)
    
    best_for_phase1 = None
    best_M_phase1 = float('inf')
    
    for si, idx in enumerate(selected_indices):
        if time.monotonic() > t_phase1_end:
            break
        a0 = inits[idx]
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()
        
        print(f"\n  Init {si+1}/{len(selected_indices)} (J0={init_scores[idx]:.6f}), "
              f"time_alloc={time_per_init:.0f}s", flush=True)
        
        sub_deadline = min(t_phase1_end, time.monotonic() + time_per_init)
        a_opt, M_opt = run_combined_optimization(a0, n_target, sub_deadline, seed=seed + si)
        
        if M_opt < best_M_phase1:
            best_M_phase1 = M_opt
            best_for_phase1 = a_opt.copy()
        
        track(a_opt)
    
    tracker_best = best_seq.copy() if best_seq is not None else a_multiscale
    print(f"\n  After Phase 1: best J={best_score:.8f}", flush=True)
    
    # === Phase 2: Intensive LP + CD on best (80s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive polish on best", flush=True)
    print(f"{'='*60}", flush=True)
    
    t_phase2 = min(deadline - 40, time.monotonic() + 80)
    if time.monotonic() < t_phase2:
        # Long LP with growing active set
        a_lp = run_lp_iterative(tracker_best, t_phase2, n_active_start=30, n_active_max=100)
        track(a_lp)
        tracker_best = best_seq.copy() if best_seq is not None else a_lp
    
    # === Phase 3: Final coordinate descent + LP (40s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Final coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    t_phase3 = min(deadline - 10, time.monotonic() + 40)
    if time.monotonic() < t_phase3:
        a_cd = run_coordinate_descent(tracker_best, t_phase3, seed=99999)
        track(a_cd)
        
        # One more LP pass
        if time.monotonic() < deadline - 5:
            a_final = run_lp_iterative(a_cd, deadline - 2, n_active_start=25, n_active_max=70)
            track(a_final)
    
    # === Phase 4: Try slightly different lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if time.monotonic() < deadline - 20:
        n_cur = tracker_best.size
        for delta in [-64, 64, -128, 128]:
            new_n = n_cur + delta
            if new_n < 128 or new_n > 2048:
                continue
            if (new_n // 64) * 64 != new_n:
                continue
            if time.monotonic() > deadline - 15:
                break
            a_try = resample_seq(tracker_best, new_n)
            a_try = np.maximum(a_try, 1e-6)
            a_try = a_try / a_try.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 20)
            a_opt, _ = run_combined_optimization(a_try, new_n, sub_dl, seed=seed + 500)
            track(a_opt)
    
    # === Final ===
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_J(best_seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
