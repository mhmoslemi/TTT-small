Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines log-sum-exp smooth optimization, subgradient descent with momentum,
LP alternating, and coordinate refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)

One more idea: **Frank-Wolfe (Conditional Gradient) method**.

The Frank-Wolfe algorithm for minimizing a convex function over a polytope:
1. At each iteration, compute the subgradient g at the current point.
2. Solve the "linear oracle": find `v = argmin_{v in P} <g, v>` where P is the feasible polytope.
3. Update: `a_new = (1-t)*a + t*v` where `t` is chosen by line search.

In our case:
- The objective `f(a) = max_k conv(a,a)[k]` is convex (max of linear functions).
- The feasible set is the simplex `{a >= 0, sum(a) = 1}`.
- The linear oracle: `v = argmin_{a >= 0, sum(a)=1} <g, a> = e_j` where `j = argmin(g)`.

So the Frank-Wolfe step is:
1. Compute subgradient g (e.g., `g[j] = 2*a[k*-j]` for k* = argmax(conv)).
2. Set `v = e_j` where `j = argmin(g)` (the coordinate with smallest gradient).
3. Blend: `a_new = (1-t)*a + t*e_j`.

This is extremely simple! At each step, we're moving mass toward the coordinate that has the least "responsibility" for the peak.

The downside is that Frank-Wolfe converges slowly (1/t rate). But it's very cheap per iteration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na = len(a)
    nb = len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    ex = np.exp(x)
    return ex / np.sum(ex)


def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1.0 + u * np.arange(1, n + 1) - cssv
    rho = np.argmax(cond > 0)
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_value(conv, L):
    """Compute log-sum-exp approximation of max: (1/L)*log(sum(exp(L*x)))."""
    Lx = L * conv
    m = np.max(Lx)
    if m > 400:
        return m / L
    return (m + np.log(np.sum(np.exp(Lx - m)))) / L


def soft_max_gradient(a, L):
    """
    Gradient of (1/L)*log(sum(exp(L*conv(a,a)))) w.r.t. a.
    Returns array of same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)
    Lconv = L * conv
    m = np.max(Lconv)
    if m > 400:
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for j in range(n):
            k = k_star - j
            if 0 <= k < n:
                grad[j] = 2.0 * a[k]
        return grad
    p = np.exp(Lconv - m)
    p /= np.sum(p)
    # grad[j] = 2 * sum_k p[k]*a[k-j] (with bounds)
    # = 2 * conv_full(a, p_rev)[2n-2-j] where p_rev = p[::-1]
    p_rev = p[::-1]
    cross = compute_conv_cross(a, p_rev)
    grad = 2.0 * cross[2 * n - 2::-1][:n]
    return grad


def smooth_objective_and_grad(x, L, n):
    """Objective and gradient for the softmax-parametrized smooth optimization."""
    a = softmax(x)
    conv = compute_conv(a)
    Lconv = L * conv
    m = np.max(Lconv)
    if m > 400:
        obj = 2.0 * n * m / L
        k_star = int(np.argmax(conv))
        grad_a = np.zeros(n)
        for j in range(n):
            k = k_star - j
            if 0 <= k < n:
                grad_a[j] = 2.0 * a[k]
    else:
        obj = 2.0 * n * (m + np.log(np.sum(np.exp(Lconv - m)))) / L
        p = np.exp(Lconv - m)

        p_rev = p[::-1]
        cross = compute_conv_cross(a, p_rev)
        grad_a = 2.0 * cross[2 * n - 2::-1][:n]
    # Chain rule for softmax parametrization
    mean_grad = float(np.dot(a, grad_a))
    grad_x = a * (grad_a - mean_grad)
    return obj, grad_x

    Continuation method: smooth optimization with gradually increasing L.
    """

            return o

            return g

def subgradient_optimize(a_init, time_budget=60.0, momentum_decay=0.9):
    """
    Subgradient descent with momentum and simplex projection.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    step_size = 0.05
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for j in range(n):
            k = k_star - j
            if 0 <= k < n:
                grad[j] = 2.0 * a[k]
        # Center gradient (projection onto sum=0 hyperplane)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        grad = grad / grad_norm

        # Heavy ball momentum
        momentum = momentum_decay * momentum - step_size * grad
        a_new = a + momentum
        a_new = project_simplex(a_new)
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            # Decay step size slowly
            step_size *= (1.0 - 0.001)
        else:
            # Revert and reduce step size
            step_size *= 0.9
            momentum *= 0.5
            if step_size < 1e-6:
                break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    if n > 2000:
        return None, None
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CBC, cp.ECOS, cp.SCS, cp.MOSEK, cp.CLARABEL]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

    n = len(a)

def frank_wolfe_optimize(a_init, time_budget=30.0):
    """Frank-Wolfe (conditional gradient) method."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        t = 2.0 / (it + 2.0)  # FW step size schedule
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for j in range(n):
            k = k_star - j
            if 0 <= k < n:
                grad[j] = 2.0 * a[k]

        # Linear oracle: min <grad, v> s.t. v in simplex => v = e_{argmin(grad)}
        j_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[j_min] = 1.0

        a_new = (1.0 - t) * a + t * v
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def coordinate_refine(a, time_budget=20.0):
    """Coordinate descent: try moving mass between pairs to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        k_star = int(np.argmax(conv))
        improved = False

        for i in range(0, n, 2):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i:
                # Try moving mass from i to j (or vice versa)
                for direction in [-1, 1]:
                    for frac in [0.01, 0.03, 0.1]:
                        if a[i] < 1e-12:
                            continue
                        delta = direction * frac * min(a[i], a[j] + frac)
                        if delta <= 0 and direction > 0:
                            continue
                        a[i] -= delta
                        a[j] += delta
                        if a[i] < 0:
                            a[i] += delta
                            a[j] -= delta
                            continue
                        a[j] += 0  # ensure no negative
                        if a[j] < 0:
                            a[i] += delta
                            a[j] -= delta
                            continue
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                            break
                if a[i] < 0:
                    a[i] = 0
                if a[j] < 0:
                    a[j] = 0

        if not improved:
            break

    best_a = best_a / np.sum(best_a)
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = np.sum(a_new)
    if s > 0:
        a_new = a_new / s
    return np.maximum(a_new, 0)

                print(f"  LP round {r}: obj={obj_lp:.8f} (no improvement)")
        except Exception as e:
            print(f"  LP round {r} failed: {e}")
            break

    # ============ Phase 4: Frank-Wolfe ============
    print(f"\n=== Phase 4: Frank-Wolfe ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        phase4_budget = min(remaining * 0.1, 40.0)
        try:
            a_fw = frank_wolfe_optimize(best_seq, time_budget=phase4_budget)
            obj_fw = compute_obj_normalized(a_fw)
            if obj_fw < best_score:
                best_score = obj_fw
                best_seq = a_fw.copy()
                print(f"  Frank-Wolfe: obj={best_score:.8f}")
            else:
                print(f"  Frank-Wolfe: obj={obj_fw:.8f} (no improvement)")
        except Exception as e:
            print(f"  Frank-Wolfe failed: {e}")

    # ============ Phase 5: Multi-n Exploration ============
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50), n_best,
        n_best + 50, n_best + 100, n_best + 200,
        500, 700, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]

    for n_try in n_options:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(rem * 0.2, 40.0)
        if alloc < 5:
            continue

        # Quick smooth optimization at this n
        L_sched = [5, 20, 50]
        try:
            a_opt = smooth_optimize(a_try, L_sched, max_iter_per_L=100,
                                    time_budget=alloc * 0.6)
            obj_try = compute_obj_normalized(a_opt)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                n_best = len(best_seq)
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")
        except Exception:
            pass

    # ============ Phase 6: Final Polish ============
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        # Smooth optimization with higher L
        L_final = [50, 100, 200, 500]
        alloc = min(remaining * 0.3, 60.0)
        try:
            a_fin = smooth_optimize(best_seq, L_final, max_iter_per_L=200,
                                    time_budget=alloc)
            obj_fin = compute_obj_normalized(a_fin)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Final smooth: obj={best_score:.8f}")
        except Exception:
            pass

    if remaining > 30 or (deadline - time.monotonic()) > 15:
        # Subgradient refinement
        rem = deadline - time.monotonic()
        if rem > 10:
            alloc = min(rem * 0.2, 30.0)
            try:
                a_sg2 = subgradient_optimize(best_seq, time_budget=alloc)
                obj_sg2 = compute_obj_normalized(a_sg2)
                if obj_sg2 < best_score:
                    best_score = obj_sg2
                    best_seq = a_sg2.copy()
                    print(f"  Final subgrad: obj={best_score:.8f}")
            except Exception:
                pass

    if (deadline - time.monotonic()) > 15:
        # LP one more time
        rem = deadline - time.monotonic()
        alloc = min(rem * 0.3, 40.0)
        try:
            a_lp2 = lp_alternating(best_seq, time_budget=alloc, max_iters=100)
            obj_lp2 = compute_obj_normalized(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  Final LP: obj={best_score:.8f}")
        except Exception:
            pass

    if (deadline - time.monotonic()) > 10:
        # Coordinate refinement
        rem = deadline - time.monotonic()
        alloc = min(rem * 0.3, 20.0)
        try:
            a_cd = coordinate_refine(best_seq, time_budget=alloc)
            obj_cd = compute_obj_normalized(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Coord refine: obj={best_score:.8f}")
        except Exception:
            pass

    # ============ Phase 7: Random Restarts ============
    print(f"\n=== Phase 7: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_restarts = 6
    for r in range(n_restarts):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        alloc = min(rem * 0.15, 40.0)
        if alloc < 5:
            break

        # Different initialization
        init_type = r % 3
        if init_type == 0:
            # Perturbed best
            a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif init_type == 1:
            # Different n with perturbation
            n_r = n_best + int(rng.integers(-50, 50))
            n_r = max(200, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.03, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            # Use smooth opt from scratch with random init
            a_r = rng.uniform(0.1, 1.0, n_best)
            a_r = a_r / np.sum(a_r)

        try:
            L_s = [5, 20, 50, 100]
            a_opt = smooth_optimize(a_r, L_s, max_iter_per_L=100,
                                    time_budget=alloc * 0.7)
            obj_r = compute_obj_normalized(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r} (type={init_type}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (type={init_type}): obj={obj_r:.8f}")
        except Exception:
            pass

    # ============ Final Validation ============
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    try:
        val = evaluate_sequence(result)
        if val < np.inf and val < best_score:
            best_score = val
        elif val >= np.inf:
            result = [1.0] * 100
            best_score = 2.0
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
