```
    cmax = float(np.max(c))
    L=len(c)
```

```
def smooth_step(a,c,V,tau_factor=0.1,q=2.0,eta_base=1.0):
   n=len(a); L=len(c)
   w=active_weights(c,tau_factor,q)
   grad = w @ V
```

```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a)) / sum(a)**2.

The main local search is a fast exact pairwise coordinate descent:
move mass from one coordinate to another, where the objective along that
one-dimensional move is exactly the maximum of convex quadratics and is
minimized by golden-section search.  Smooth active-peak gradient steps,
LP-based alternating directions, multi-length resampling, and random
restarts are used to escape local minima.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:need]
    np.maximum(conv, 0.0, out=conv)
    return conv


def score_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = compute_conv(a)
    if c.size == 0:
        return np.inf
    mb = float(np.max(c))
    if mb < 0.0:
        mb = 0.0
    return float(2.0 * n * mb / (s * s))


def normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    np.maximum(a, 0.0, out=a)
    s = float(np.sum(a))
    if s <= 1e-15:
        n = len(a)
        if n == 0:
            return np.array([1.0])
        return np.ones(n, dtype=np.float64) / float(n)
    a /= s
    return a


def resample_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def build_V(a):
    """
    V[i,k] = d convolution coefficient k / d a_i = 2*a[k-i]
    when 0 <= k-i < n, and 0 otherwise.
    """
    n = len(a)
    L = 2 * n - 1
    V = np.zeros((n, L), dtype=np.float64)
    two_a = 2.0 * np.asarray(a, dtype=np.float64)
    for i in range(n):
        V[i, i:i + n] = two_a
    return V


def get_diag_indices(n):
    """
    Flat indices in V.ravel() for V[m, m+p], over valid m.
    """
    inds = []
    step = 2 * n
    for p in range(n):
        length = n - p
        inds.append(p + np.arange(length, dtype=np.int64) * step)
    return inds


def eval_pair_delta(c, d, i, j, delta, temp):
    """
    Objective max coefficient after moving delta from i to j, given old c
    and d = V[j] - V[i].
    """
    np.add(c, delta * d, out=temp)
    delta2 = delta * delta
    temp[2 * i] += delta2
    temp[2 * j] += delta2
    temp[i + j] += 2.0 * delta2
    return float(np.max(temp))


def golden_search_pair(c, d, i, j, lo, hi, temp, max_iter=20):
    """Golden-section minimum of the convex pairwise move objective."""
    if hi <= lo:
        f0 = eval_pair_delta(c, d, i, j, lo, temp)
        return lo, f0

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    f_lo = eval_pair_delta(c, d, i, j, lo, temp)
    f_hi = eval_pair_delta(c, d, i, j, hi, temp)

    best_delta = lo
    best_val = f_lo
    if f_hi < best_val:
        best_delta = hi
        best_val = f_hi

    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = eval_pair_delta(c, d, i, j, x1, temp)
    f2 = eval_pair_delta(c, d, i, j, x2, temp)

    if f1 < best_val:
        best_delta, best_val = x1, f1
    if f2 < best_val:
        best_delta, best_val = x2, f2

    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = eval_pair_delta(c, d, i, j, x1, temp)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = eval_pair_delta(c, d, i, j, x2, temp)

        if f1 < best_val:
            best_delta, best_val = x1, f1
        if f2 < best_val:
            best_delta, best_val = x2, f2

    return best_delta, best_val


def apply_pair_move(a, c, Vview, diag_inds, i, j, d, delta):
    """Apply a pairwise mass transfer in-place."""
    if delta == 0.0:
        return
    c += delta * d
    delta2 = delta * delta
    c[2 * i] += delta2
    c[2 * j] += delta2
    c[i + j] += 2.0 * delta2

    ni = a[i] - delta
    if ni < 0.0:
        ni = 0.0
    a[i] = ni
    a[j] = a[j] + delta

    dv = 2.0 * delta
    Vview[diag_inds[i]] -= dv
    Vview[diag_inds[j]] += dv


def active_weights(c, tau_factor=0.1, q=2.0):
    """Weights emphasizing near-maximum convolution coefficients."""
    L = len(c)
    cmax = float(np.max(c))
    if L == 0 or cmax <= 0.0:
        return np.ones(L, dtype=np.float64) / max(1, L)

    threshold = cmax * (1.0 - tau_factor)
    above = c > threshold
    count = int(np.sum(above))
    if count == 0:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w

    vals = c[above] - threshold
    scale = cmax - threshold
    if scale <= 0.0:
        w = np.zeros(L, dtype=np.float64)
        w[above] = 1.0
    else:
        w = np.zeros(L, dtype=np.float64)
        w[above] = (vals / scale) ** q

    s = float(np.sum(w))
    if s > 1e-300:
        w /= s
    else:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    return w


def smooth_step(a, c, V, tau_factor=0.1, q=2.0, eta_base=1.0):
    """
    One normalized steepest-descent step on an active-peak smooth surrogate.
    Returns a better normalized sequence, or None.
    """
    n = len(a)
    if n < 2:
        return None
    L = len(c)
    w = active_weights(c, tau_factor=tau_factor, q=q)
    grad = np.dot(w, V)
    grad -= np.mean(grad)
    scale = float(np.max(np.abs(grad)))
    if scale <= 1e-300:
        return None

    direction = -grad / scale
    current_obj = 2.0 * n * float(np.max(c))
    best_obj = current_obj
    best_a = None

    etas = (0.25 * eta_base, 0.5 * eta_base, 1.0 * eta_base,
            2.0 * eta_base, 4.0 * eta_base)

    for eta in etas:
        step = (float(eta) / float(n)) * direction
        trial = a + step
        np.maximum(trial, 0.0, out=trial)
        s = float(np.sum(trial))
        if s <= 1e-15:
            continue
        trial /= s
        c2 = compute_conv(trial)
        obj = 2.0 * n * float(np.max(c2))
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = trial.copy()

    return best_a


def try_lp_step(a, M, V, time_limit=20.0):
    """
    Given current a, solve
        max sum(q) s.t. conv(a,q) <= max(conv(a,a)), q >= 0.
    Returns normalized q or None.
    """
    n = len(a)
    L = 2 * n - 1
    if M <= 1e-300:
        return None

    A = (0.5 * V.T).copy()
    A /= M
    b = np.ones(L, dtype=np.float64)
    cvec = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(
            cvec,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method='highs',
            options={
                'presolve': True,
                'time_limit': float(time_limit),
                'dual_feasibility_tolerance': 1e-9,
                'primal_feasibility_tolerance': 1e-9,
            }
        )
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None

    q = np.asarray(res.x, dtype=np.float64)
    np.maximum(q, 0.0, out=q)
    s = float(np.sum(q))
    if s <= 1e-12:
        return None
    return q / s


def lp_blend_line_search(a, q, current_obj):
    """Line search over blends of a and LP direction q."""
    n = len(a)
    best_obj = float(current_obj)
    best_a = None
    for t in (0.1, 0.25, 0.5, 0.75, 0.9, 1.0):
        at = (1.0 - t) * a + t * q
        np.maximum(at, 0.0, out=at)
        s = float(np.sum(at))
        if s <= 1e-15:
            continue
        at /= s
        c = compute_conv(at)
        obj = 2.0 * n * float(np.max(c))
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = at.copy()
    return best_a


def pair_descent(a, time_budget=30.0, rng=None, allow_lp=False):
    """
    Main local optimizer.  Maintains a, c=conv(a,a), and V=derivative matrix.
    Pairwise moves are exact in convex 1D subproblems.
    """
    if rng is None:
        rng = np.random.default_rng(123456789)

    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    if n < 2:
        return a

    start = time.monotonic()
    c = compute_conv(a)
    V = build_V(a)
    Vview = V.ravel()
    diag_inds = get_diag_indices(n)
    L = len(c)

    temp = np.empty(L, dtype=np.float64)
    d = np.empty(L, dtype=np.float64)

    best_max = float(np.max(c))
    best_obj = 2.0 * n * best_max
    best_a = a.copy()

    no_improve = 0
    move_count = 0
    K = min(20, L)

    while time.monotonic() - start < time_budget:
        if move_count > 0 and move_count % 250 == 0:
            a = normalize(a)
            c = compute_conv(a)
            V = build_V(a)
            Vview = V.ravel()
            diag_inds = get_diag_indices(n)

        M = float(np.max(c))
        if M < best_max - 1e-15:
            best_max = M
            best_obj = 2.0 * n * best_max
            best_a = a.copy()

        if L <= K:
            top_all = np.arange(L, dtype=int)
        else:
            top_all = np.argpartition(-c, K)[:K]
        top_sorted = top_all[np.argsort(c[top_all])[::-1]]

        VA = V[:, top_all]
        risk = np.max(VA, axis=1)

        improved = False
        n_peaks = min(5, len(top_sorted))

        for pk_idx in range(n_peaks):
            kp = int(top_sorted[pk_idx])
            col = V[:, kp]

            src_count = min(8, max(1, n - 1))
            tgt_count = min(20, max(1, n - 1))

            src_score = col * a + 0.1 * risk * a
            srcs = np.argpartition(-src_score, src_count)[:src_count]

            tgt_score = risk + 0.25 * col
            tgts = np.argpartition(tgt_score, tgt_count)[:tgt_count]

            for ii in range(len(srcs)):
                i = int(srcs[ii])
                if a[i] <= 1e-14:
                    continue
                ai = a[i]
                for jj in range(len(tgts)):
                    j = int(tgts[jj])
                    if i == j:
                        continue
                    if col[i] <= col[j] + 1e-12:
                        continue
                    hi = ai * (1.0 - 1e-15)
                    if hi <= 1e-15:
                        continue
                    if time.monotonic() - start >= time_budget:
                        break

                    np.subtract(V[j], V[i], out=d)
                    delta, val = golden_search_pair(c, d, i, j, 0.0, hi, temp, max_iter=18)

                    if val < M - 1e-12:
                        apply_pair_move(a, c, Vview, diag_inds, i, j, d, delta)
                        newM = float(np.max(c))
                        if newM < best_max:
                            best_max = newM
                            best_obj = 2.0 * n * newM
                            best_a = a.copy()
                        improved = True
                        no_improve = 0
                        move_count += 1
                        break
                if improved or time.monotonic() - start >= time_budget:
                    break
            if improved:
                break

        if improved:
            continue

        no_improve += 1

        # Try a few random pairwise moves.
        current_M = float(np.max(c))
        random_tries = 0
        while not improved and random_tries < 20 and time.monotonic() - start < time_budget:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            random_tries += 1
            if i == j:
                continue
            if a[i] > a[j]:
                src = i
                tgt = j
                hi = a[i] * (1.0 - 1e-15)
            else:
                src = j
                tgt = i
                hi = a[j] * (1.0 - 1e-15)
            if hi <= 1e-14:
                continue

            np.subtract(V[tgt], V[src], out=d)
            delta, val = golden_search_pair(c, d, src, tgt, 0.0, hi, temp, max_iter=16)
            if val < current_M - 1e-12:
                apply_pair_move(a, c, Vview, diag_inds, src, tgt, d, delta)
                newM = float(np.max(c))
                if newM < best_max:
                    best_max = newM
                    best_obj = 2.0 * n * newM
                    best_a = a.copy()
                improved = True
                no_improve = 0
                move_count += 1

        if improved:
            continue

        remaining = time_budget - (time.monotonic() - start)

        # Try simple symmetrization.
        a_sym = 0.5 * (a + a[::-1])
        c_sym = compute_conv(a_sym)
        M_sym = float(np.max(c_sym))
        if M_sym < current_M - 1e-12:
            a = normalize(a_sym)
            c = compute_conv(a)
            V = build_V(a)
            Vview = V.ravel()
            diag_inds = get_diag_indices(n)
            if float(np.max(c)) < best_max:
                best_max = float(np.max(c))
                best_obj = 2.0 * n * best_max
                best_a = a.copy()
            no_improve = 0
            continue

        # Occasionally use LP alternating if cheap enough.
        if allow_lp and n <= 900 and remaining > 60.0:
            a = normalize(a)
            c = compute_conv(a)
            V = build_V(a)
            Vview = V.ravel()
            diag_inds = get_diag_indices(n)
            Mcur = float(np.max(c))
            q = try_lp_step(a, Mcur, V, min(30.0, max(5.0, 0.3 * remaining)))
            if q is not None:
                a_new = lp_blend_line_search(a, q, 2.0 * n * Mcur)
                if a_new is not None:
                    a = a_new
                    c = compute_conv(a)
                    V = build_V(a)
                    Vview = V.ravel()
                    diag_inds = get_diag_indices(n)
                    curM = float(np.max(c))
                    if curM < best_max:
                        best_max = curM
                        best_obj = 2.0 * n * curM
                        best_a = a.copy()
                    no_improve = 0
                    continue

        # Smooth active-peak gradient step.
        a_new = smooth_step(a, c, V, tau_factor=0.15, q=2.0, eta_base=1.0)
        if a_new is not None:
            a = a_new
            c = compute_conv(a)
            V = build_V(a)
            Vview = V.ravel()
            diag_inds = get_diag_indices(n)
            curM = float(np.max(c))
            if curM < best_max:
                best_max = curM
                best_obj = 2.0 * n * curM
                best_a = a.copy()
            no_improve = 0
            continue

        # Random multiplicative perturbation.
        sigma = 0.05 + 0.02 * min(5, no_improve // 10)
        if no_improve > 60:
            a = best_a.copy()
        a = normalize(a * np.exp(rng.normal(0.0, sigma, n)))
        c = compute_conv(a)
        V = build_V(a)
        Vview = V.ravel()
        diag_inds = get_diag_indices(n)
        no_improve = 0

    return normalize(best_a)


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = parent[np.isfinite(parent)]
        np.clip(parent, 0.0, 1000.0, out=parent)
        if parent.size == 0:
            return None, np.inf
        sc = score_sequence(parent)
        if np.isfinite(sc):
            return parent, sc
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    start_global = time.monotonic()
    deadline = start_global + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    parent, parent_score = try_load_parent()
    if parent is not None and len(parent) > 0 and np.isfinite(parent_score):
        best_a = normalize(parent.astype(np.float64).copy())
        best_score = parent_score
    else:
        best_a = np.ones(700, dtype=np.float64) / 700.0
        best_score = score_sequence(best_a)

    print(f"Initial: n={len(best_a)}, score={best_score:.10f}")

    # ---------------- Phase 1: diversify nearby lengths ----------------
    base_n = len(best_a)
    phase1_end = min(deadline, start_global + 320.0)

    n_cand = [base_n]
    for delta in (-300, -200, -100, -50, -25, 25, 50, 100, 200, 300, 500):
        nn = base_n + delta
        if 200 <= nn <= 1200:
            n_cand.append(nn)
    for nn in (500, 700, 900, 1000):
        n_cand.append(nn)

    seen = set()
    n_cand_clean = []
    for nn in n_cand:
        if nn not in seen:
            seen.add(nn)
            n_cand_clean.append(nn)
    n_cand_clean = sorted(n_cand_clean)[:8]

    for n_try in n_cand_clean:
        if time.monotonic() >= phase1_end - 25.0:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 20.0:
            break

        if len(best_a) == n_try:
            a0 = best_a.copy()
        else:
            a0 = normalize(resample_sequence(best_a, n_try))

        # Add a little noise so resampling does not exactly duplicate a local point.
        a0 = normalize(a0 * np.exp(rng.normal(0.0, 0.01, len(a0))))

        budget = min(45.0, max(8.0, 0.12 * remaining))
        a1 = pair_descent(a0, time_budget=budget, rng=rng, allow_lp=False)
        s1 = score_sequence(a1)
        if s1 < best_score - 1e-12:
            best_a = a1
            best_score = s1
            print(f"Phase1 length {len(a1)}: {best_score:.10f}")
        else:
            print(f"Phase1 length {len(a1)}: {s1:.10f}")

    # Two simple synthetic starts at the current best length.
    syn_budget = min(70.0, max(10.0, deadline - time.monotonic() - 700.0))
    if syn_budget > 20.0:
        n_syn = len(best_a)
        x = np.linspace(0.0, 1.0, n_syn)

        a_u = 1.0 + 1.0 * np.abs(2.0 * x - 1.0)
        a_u = normalize(a_u)
        if time.monotonic() < deadline - 100.0:
            a_try = pair_descent(a_u, time_budget=max(10.0, 0.5 * syn_budget), rng=rng, allow_lp=False)
            s_try = score_sequence(a_try)
            if s_try < best_score - 1e-12:
                best_a = a_try
                best_score = s_try
                print(f"Synthetic U: {best_score:.10f}")
            else:
                print(f"Synthetic U: {s_try:.10f}")

        a_rand = normalize(rng.dirichlet(np.ones(n_syn) * 0.15))
        if time.monotonic() < deadline - 80.0:
            a_try = pair_descent(a_rand, time_budget=max(8.0, 0.4 * syn_budget), rng=rng, allow_lp=False)
            s_try = score_sequence(a_try)
            if s_try < best_score - 1e-12:
                best_a = a_try
                best_score = s_try
                print(f"Synthetic Dirichlet: {best_score:.10f}")
            else:
                print(f"Synthetic Dirichlet: {s_try:.10f}")

    # ---------------- Phase 2: main intensive search ----------------
    main_end = deadline - 200.0
    no_improve_global = 0

    while time.monotonic() < main_end:
        remaining = main_end - time.monotonic()
        if remaining < 40.0:
            break

        budget = min(120.0, max(30.0, 0.30 * remaining))
        start_a = best_a.copy()

        if no_improve_global > 0:
            sigma = 0.03 + 0.025 * min(4, no_improve_global)
            if no_improve_global % 3 == 0:
                n_alt = len(start_a) + int(rng.integers(-60, 61))
                if 200 <= n_alt <= 1200 and n_alt != len(start_a):
                    start_a = normalize(resample_sequence(start_a, n_alt))
            start_a = normalize(start_a * np.exp(rng.normal(0.0, sigma, len(start_a))))

        a_new = pair_descent(start_a, time_budget=budget, rng=rng, allow_lp=True)
        s_new = score_sequence(a_new)

        if s_new < best_score - 1e-12:
            best_a = a_new
            best_score = s_new
            no_improve_global = 0
            print(f"Main: n={len(best_a)}, score={best_score:.10f}")
        else:
            no_improve_global += 1
            print(f"Main no improve (attempt {no_improve_global}): {s_new:.10f}")

    # ---------------- Phase 3: final polish ----------------
    final_end = deadline - 5.0
    last_best = best_score

    while time.monotonic() < final_end:
        remaining = final_end - time.monotonic()
        if remaining < 20.0:
            break

        budget = min(60.0, max(10.0, 0.30 * remaining))
        a_new = pair_descent(best_a, time_budget=budget, rng=rng, allow_lp=False)
        s_new = score_sequence(a_new)

        if s_new < best_score - 1e-12:
            best_a = a_new
            best_score = s_new
            last_best = best_score
            print(f"Final polish: n={len(best_a)}, score={best_score:.10f}")
            if time.monotonic() >= final_end - 15.0:
                break
            continue

        # Very mild perturbation, then polish.
        a_try = normalize(best_a * np.exp(rng.normal(0.0, 0.025, len(best_a))))
        budget2 = min(35.0, max(8.0, 0.35 * remaining))
        a_new2 = pair_descent(a_try, time_budget=budget2, rng=rng, allow_lp=remaining > 120.0)
        s_new2 = score_sequence(a_new2)

        if s_new2 < best_score - 1e-12:
            best_a = a_new2
            best_score = s_new2
            print(f"Final polish perturbed: n={len(best_a)}, score={best_score:.10f}")

    # ---------------- Final validation and return ----------------
    best_a = normalize(best_a)
    result = [float(min(1000.0, max(0.0, x))) for x in best_a]

    # Guarantee the public evaluator accepts the result.
    try:
        val = evaluate_sequence(result)
    except Exception:
        val = np.inf

    if not np.isfinite(val):
        fallback = np.ones(700, dtype=np.float64) / 700.0
        result = [float(x) for x in fallback]
        try:
            val = evaluate_sequence(result)
        except Exception:
            val = np.inf

    elapsed = time.monotonic() - start_global
    print(f"FINAL: n={len(result)}, score={val:.10f}, time={elapsed:.1f}s")

    return result
```
