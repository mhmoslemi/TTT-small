"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FISTA with soft-max subgradient, mirror descent, LP alternating, peak deflation,
and multi-n exploration. Key improvements over prior approach:
- FISTA with momentum for faster gradient convergence
- Mirror descent (multiplicative weights) for better simplex geometry
- Targeted peak deflation via direct element manipulation
- Analytic shape initializations (power law, exponential)
- More aggressive multi-n search with per-n optimization

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1)
    cond = u * rho_arr > (cv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_subgrad(a, tau):
    """Compute soft-max subgradient of max_k (a*a)[k]. Returns (grad, conv, cmax)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    w = np.exp(shifted)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= w_sum
    # s[i] = 2 * sum_k w[k] * a_ext[k-i]
    # Cross-correlation: s[i] = 2 * sum_m a[m] * w[m+i]
    size = 1
    while size < 3 * n:
        size <<= 1
    A = fft(a, size)
    W = fft(w, size)
    corr = np.real(ifft(np.conj(A) * W, size))[:n]
    grad = 2.0 * corr
    return grad, conv, cmax


def hard_subgrad(a):
    """Compute hard subgradient at the argmax of (a*a). Returns (grad, conv, cmax, k_star)."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    cmax = float(conv[k_star])
    grad = np.zeros(n)
    for i in range(n):
        k_minus_i = k_star - i
        if 0 <= k_minus_i < n:
            grad[i] = 2.0 * a[k_minus_i]
    return grad, conv, cmax, k_star


def fista_softmax(a, tau_init, lr_init, max_steps, time_budget):

        t_fista = t_new

    eps = 1e-15
    theta = np.log(a + eps)

def build_toeplitz(a):
    """Build Toeplitz matrix where (T@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section(f, lo, hi, max_iter=20):
    """Golden section search for minimum."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 0.005:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf
