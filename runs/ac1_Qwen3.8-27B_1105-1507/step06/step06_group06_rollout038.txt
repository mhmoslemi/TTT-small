    fa = rfft(a, size)

        best_t_val = 0.0
        best_a_local = best_a.copy()
        
        # Try a range of step sizes
        t_candidates = np.logspace(np.log10(trust_min), np.log10(trust), 10)
        t_candidates = np.unique(np.clip(t_candidates, 0.01, 1.0))
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            improvement = best_obj - best_t_obj
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            # Update trust region
            if best_t_val > 0.5 * trust:
                trust = min(1.0, trust * 1.5)
            elif best_t_val < 0.2 * trust:
                trust = max(trust_min, trust * 0.7)
            stall = 0
        else:
            stall += 1
            trust = max(trust_min, trust * 0.7)
            if stall > 20:
                break
    
    return best_a, best_obj

def spectral_project(a, n_iters=100, time_budget=30.0, flat_target='uniform'):
    """
    Iterative projection: alternate between time domain (non-negative, sum=1)
    and frequency domain (flat power spectrum).
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Frequency domain step: flatten the power spectrum
        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        
        # Target: uniform power spectrum (except DC)
        if flat_target == 'uniform':
            target_power = np.ones(N)
            target_power[0] = power[0]  # Keep DC (sum constraint)
            # Actually, we want the IFFT of power to be flat
            # So we want power to be uniform
        elif flat_target == 'tapered':
            # Taper the edges to reduce Gibbs artifacts
            x = np.linspace(0, 1, N)
            target_power = (np.sin(np.pi * x) ** 2 + 0.1)
            target_power[0] = power[0]
        
        # Interpolate between current power and target
        alpha = 0.3 + 0.5 * (it / max(n_iters, 1))  # Gradually increase toward target
        alpha = min(alpha, 0.8)
        
        # Normalize target to have same total power
        # Total power = sum(power) = (1/N) * sum(conv) = (1/N) * (sum(a))^2 = 1/N (with sum=1)
        # Actually, by Parseval: sum(|fa|^2) = N * sum(a^2)
        # We want to preserve sum(a) = 1, so aa[0] = |fa[0]|^2 = 1
        total_target = np.sum(target_power[1:])  # exclude DC
        total_current = np.sum(power[1:])
        if total_target > 0 and total_current > 0:
            target_power[1:] = target_power[1:] * (total_current / total_target)
            target_power[0] = power[0]
        
        power_new = (1 - alpha) * power + alpha * target_power
        power_new = np.maximum(power_new, 0)
        
        # Reconstruct: keep phase, update magnitude
        phase = np.angle(fa)
        magnitude = np.sqrt(power_new)
        fa_new = magnitude * np.exp(1j * phase)
        
        # Time domain
        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def logsumexp_descent(a, max_iters=200, time_budget=30.0, tau_init=0.1, tau_end=0.01):
    """
    Gradient descent on the log-sum-exp smoothed objective.
    Objective: 2*n*tau*log(sum(exp(conv/tau)))/(sum(a))^2
    As tau -> 0, this approaches 2*n*max(conv)/(sum(a))^2
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step_size = 0.05
    v = np.zeros(n)  # momentum
    mu = 0.9
    
    N_conv = 2 * n - 1
    N_fft = 1
    while N_fft < N_conv + n - 1:
        N_fft <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        tau = tau_init * (tau_end / tau_init) ** (it / max_iters)
        
        conv = compute_conv(a)
        # Numerical stability
        conv_shift = conv - np.max(conv)
        lse = tau * np.log(np.sum(np.exp(conv_shift / tau))) + np.max(conv)
        
        # Gradient of lse wrt a:
        # d lse / d a[i] = (1/tau) * sum_k (conv[k]-max)/sum(exp) * d conv[k]/d a[i]
        #                   + (d max / d a[i]) ... this is complicated
        # Simpler: use the softmax weights
        weights = np.exp((conv - np.max(conv)) / tau)
        w_sum = np.sum(weights)
        weights /= w_sum
        
        # Gradient of the weighted sum: sum_k weights[k] * conv[k]
        # d/d a[i] of conv[k] = a[k-i] + a[i-(k-n+1)] (for valid indices)
        # More precisely: conv[k] = sum_j a[j]*a[k-j]
        # d/d a[i] = a[k-i] (if 0<=k-i<n) + a[i] * (d/d a[i] of a[k-i] when j=k-i)
        # Wait: d/d a[i] of sum_j a[j]*a[k-j]
        # = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
        # = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i<n, i.e., j=k-i is valid)
        # = (a[k-i] + a[k-i]) if 0<=k-i<n... no
        # = a[k-i] (first term, j=i) + a[i] (second term, j=k-i)
        # But a[k-i] and a[i] are different in general!
        # So d conv[k]/d a[i] = a[k-i] + (a[i] if k-i is a valid index for j)
        # Wait, I'm confusing myself. Let me be more careful.
        
        # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d/d a[i] = sum_j [delta(j,i) * a[k-j] + a[j] * delta(k-j, i)]
        # = a[k-i] (if 0<=i<n and 0<=k-i<n) + a[i] (if 0<=k-i<n and 0<=i<n... wait)
        # First term: j=i, contributes a[k-i] if 0<=k-i<n
        # Second term: k-j=i, so j=k-i, contributes a[k-i] if 0<=k-i<n
        # Wait no: second term is a[j] where j=k-i, so it's a[k-i]
        # So d conv[k]/d a[i] = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n
        # Unless i = k-i, i.e., 2i=k, in which case it's just a[i] (one term)
        
        # Actually wait. Let me re-derive.
        # conv[k] = sum_j a[j] * a[k-j]
        # This is a product of a[j] and a[k-j].
        # d/d a[i] of a[j]*a[k-j] = delta(j,i)*a[k-j] + a[j]*delta(k-j,i)
        # Sum over j:
        #   Term 1: j=i gives a[k-i] (if 0<=k-i<n)
        #   Term 2: k-j=i, i.e., j=k-i, gives a[k-i] (if 0<=k-i<n)
        # So total: 2*a[k-i] if 0<=k-i<n, except when i = k-i (i.e., k=2i), 
        #   in which case both terms give a[i] and we get 2*a[i] (same thing).
        # Wait, when k=2i, term 1 gives a[2i-i]=a[i] and term 2 gives a[2i-i]=a[i], total 2*a[i].
        # So in all cases: d conv[k]/d a[i] = 2*a[k-i] if 0<=k-i<n, else 0.
        
        # Hmm wait, that's not right either. When i = k-i (k even, i=k/2):
        # conv[k] contains the term a[i]*a[i] = a[i]^2
        # d/d a[i] of a[i]^2 = 2*a[i] ✓
        # And my formula gives 2*a[k-i] = 2*a[i] ✓
        # Good.
        
        # So: grad[i] = sum_k weights[k] * 2 * a[k-i] (if 0<=k-i<n)
        # = 2 * sum_{k: 0<=k-i<n} weights[k] * a[k-i]
        # = 2 * sum_{j: 0<=j<n} weights[i+j] * a[j]  (substituting j=k-i)
        # = 2 * (weights[n-1::-1] * a)... no, it's a correlation
        
        # Let me compute it via FFT
        # grad[i] = 2 * sum_j a[j] * weights[i+j] (for 0<=i+j < 2n-1)
        # This is a cross-correlation of a and weights
        
        # Using FFT: 
        # Define w_rev[m] = weights[2n-2-m] (reversed weights)
        # Then grad[i]/2 = sum_j a[j] * weights[i+j]
        # = (a * w_rev)[n-1+i] (convolution)
        # Hmm, let me just compute it directly for clarity (n is at most ~1000)
        
        grad = np.zeros(n)
        # grad[i] = 2 * sum_j a[j] * weights[i+j] for 0 <= i+j <= 2n-2
        for i in range(n):
            # j ranges from 0 to n-1, and i+j ranges from i to i+n-1
            # weights[i+j] is valid if i+j <= 2n-2, i.e., j <= 2n-2-i
            # Since j <= n-1 and 2n-2-i >= n-1 when i <= n-1, always valid
            js = np.arange(n)
            ks = i + js
            grad[i] = 2.0 * np.sum(a * weights[ks])
        
        # Actually this is O(n^2) which is fine for n <= 1000
        # But let's use FFT for efficiency
        # grad[i]/2 = sum_j a[j] * weights[i+j]
        # Let's define the cross-correlation: (a ⋆ weights)[i] = sum_j a[j] * weights[i+j]
        # This is the "anti-convolution" or we can compute via FFT
        # (a ⋆ w)[m] = sum_j a[j] * w[m+j]
        # = sum_j a[j] * w_rev[2n-2-(m+j)] where w_rev[k] = w[2n-2-k]
        # = (a * w_rev)[2n-2-m]
        
        w_rev = weights[::-1]
        # (a * w_rev)[m] = sum_j a[j] * w_rev[m-j]
        # We want: sum_j a[j] * weights[i+j] = sum_j a[j] * w_rev[2n-2-i-j]
        # Let m = 2n-2-i: sum_j a[j] * w_rev[m-j] = (a * w_rev)[m]
        
        N_use = 1
        while N_use < 2 * n:
            N_use <<= 1
        fa_fft = rfft(a, N_use)
        fw_fft = rfft(w_rev, N_use)
        conv_aw = irfft(fa_fft * fw_fft, N_use)
        # grad[i] = 2 * conv_aw[2n-2-i]
        grad = 2.0 * conv_aw[2*n-2-n:2*n-2+1][::-1]  # indices from n-1 to 2n-2, reversed
        # Wait: grad[i] = 2 * conv_aw[2n-2-i] for i=0,...,n-1
        # So grad = 2 * conv_aw[2n-2-n+1 : 2n-2+1][::-1] = 2 * conv_aw[n-1:2n-1][::-1]
        # Let me verify: for i=0, grad[0] = 2*conv_aw[2n-2]
        # for i=n-1, grad[n-1] = 2*conv_aw[n-1]
        # So grad[i] = 2*conv_aw[2n-2-i], i=0..n-1
        # = 2 * conv_aw[2n-2-n+1 : 2n-2+1][::-1]... 
        # indices 2n-2, 2n-3, ..., n-1
        # = 2 * conv_aw[n-1 : 2n-1][::-1]
        
        # Hmm, let me just do it explicitly to avoid bugs
        grad = np.zeros(n)
        for i in range(n):
            idx = 2*n-2-i
            if idx < len(conv_aw):
                grad[i] = 2.0 * conv_aw[idx]
        
        # Normalize gradient (the objective also has 1/(sum(a))^2 factor)
        # Since we're on the simplex (sum=1), we can ignore the sum factor
        # and just minimize the numerator: 2*n*lse
        # But we need to project back to simplex after each step
        
        # Center the gradient (to maintain sum=1)
        grad_centered = grad - np.mean(grad)
        
        # Momentum step
        v = mu * v - step_size * grad_centered
        a_new = a + v
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        step_size *= 0.999
    
    return best_a, best_obj

def chase_peak_improved(a, n_steps=100, time_budget=20.0):
    """
    Improved chase-the-peak: use top-k peaks for the subgradient.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N_fft = 1
    while N_fft < 2 * n - 1 + n - 1:
        N_fft <<= 1
    
    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Use top-k peaks with exponential weighting
        k_top = max(5, n // 10)
        sorted_idx = np.argsort(conv)[::-1][:k_top]
        weights = np.zeros(2*n-1)
        for rank, idx in enumerate(sorted_idx):
            weights[idx] = np.exp(-rank * 0.3)
        
        # Compute gradient
        w_rev = weights[::-1]
        fa_fft = rfft(a, N_fft)
        fw_fft = rfft(w_rev, N_fft)
        conv_aw = irfft(fa_fft * fw_fft, N_fft)
        grad = np.zeros(n)
        for i in range(n):
            idx = 2*n-2-i
            if idx < len(conv_aw):
                grad[i] = conv_aw[idx]
        
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        
        # Try multiple step sizes
        for ss in [0.02, 0.05, 0.1, 0.2]:
            a_try = a - ss * grad_centered / (grad_norm / n)
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def diverse_spectral_init(n, rng):
    """
    Generate diverse initializations using spectral design.
    """
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n) / n)
    
    # 2. Flat spectrum with random phase
    N = 1
    while N < 2 * n:
        N <<= 1
    for trial in range(3):
        phase = rng.uniform(0, 2 * np.pi, N)
        power = np.ones(N)
        power[0] = n  # DC component (sum = 1, so |FFT[0]| = 1, |FFT[0]|^2 = 1... 
                      # Actually FFT(a)[0] = sum(a) = 1, so power[0] = 1)
        power[0] = 1.0
        fa = np.sqrt(power) * np.exp(1j * phase)
        a_init = np.fft.ifft(fa).real[:n]
        a_init = project_nonneg_sum1(a_init)
        inits.append(a_init)
    
    # 3. Triangle
    x = np.linspace(0, 1, n)
    tri = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    tri = tri + 0.01
    inits.append(tri / tri.sum())
    
    # 4. Sine
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())
    
    # 5. Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())
    
    # 6. Two-peak
    peak = np.zeros(n)
    p1, p2 = int(n * 0.25), int(n * 0.75)
    sigma = n * 0.1
    peak = np.exp(-((x - 0.25)**2) / (2*sigma**2/n)) + np.exp(-((x - 0.75)**2) / (2*sigma**2/n))
    inits.append(peak / peak.sum())
    
    return inits

def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy optimization combining LP alternating with trust region,
    spectral domain projection, log-sum-exp smoothed gradient descent,
    and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()
    
    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
    
    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")
    
    # === Phase 1: Quick screening with diverse inits (5% time) ===
    print("\n=== Phase 1: Diverse Initialization ===")
    phase1_budget = max(30.0, (deadline - start_time) * 0.05)
    n_options = [n_parent, max(200, n_parent - 100), n_parent + 50, n_parent + 100,
                 max(200, n_parent - 200), n_parent + 200]
    n_options = list(dict.fromkeys([n for n in n_options if 100 <= n <= 2000]))
    
    candidates = []
    per_init_budget = max(2.0, phase1_budget / (len(n_options) * 3))
    
    for n_try in n_options:
        if time.monotonic() > start_time + phase1_budget:
            break
        inits = diverse_spectral_init(n_try, rng)
        if parent is not None:
            inits.append(project_nonneg_sum1(resample_sequence(parent, n_try)))
        
        for init in inits:
            if time.monotonic() > start_time + phase1_budget:
                break
            a_quick, obj_quick = chase_peak_improved(init, n_steps=30, time_budget=per_init_budget)
            if obj_quick < np.inf and obj_quick < best_score:
                candidates.append((obj_quick, a_quick))
    
    candidates.sort(key=lambda x: x[0])
    if candidates:
        print(f"  Found {len(candidates)} improving candidates, best: {candidates[0][0]:.8f}")
        if candidates[0][0] < best_score:
            best_score = candidates[0][0]
            best_seq = candidates[0][1].copy()
    
    # === Phase 2: Intensive optimization (60% time) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"Best so far: {best_score:.8f}")
    phase2_budget = (deadline - start_time) * 0.60
    phase2_start = time.monotonic()
    
    a_curr = project_nonneg_sum1(best_seq.copy())
    n_curr = len(a_curr)
    
    # Cycle through strategies
    strategy_cycle = [
        ('lp_trust', lp_alternate_trust_region, {'max_iters': 50, 'time_budget': 25.0}),
        ('spectral', spectral_project, {'n_iters': 100, 'time_budget': 20.0}),
        ('lse', logsumexp_descent, {'max_iters': 100, 'time_budget': 20.0, 'tau_init': 0.05, 'tau_end': 0.005}),
        ('chase', chase_peak_improved, {'n_steps': 100, 'time_budget': 15.0}),
    ]
    
    cycle = 0
    while time.monotonic() < phase2_start + phase2_budget:
        remaining = phase2_start + phase2_budget - time.monotonic()
        if remaining < 10:
            break
        
        name, func, params = strategy_cycle[cycle % len(strategy_cycle)]
        budget = min(params.get('time_budget', 20.0), remaining * 0.3)
        params['time_budget'] = budget
        
        print(f"  [{name}] remaining={remaining:.0f}s, budget={budget:.0f}s")
        
        try:
            if name == 'lp_trust':
                a_new, obj_new = func(a_curr, time_budget=budget)
            elif name == 'spectral':
                a_new, obj_new = func(a_curr, time_budget=budget)
            elif name == 'lse':
                a_new, obj_new = func(a_curr, time_budget=budget)
            elif name == 'chase':
                a_new, obj_new = func(a_curr, time_budget=budget)
            else:
                a_new, obj_new = a_curr, compute_obj(a_curr)
        except Exception as e:
            print(f"    Error in {name}: {e}")
            a_new, obj_new = a_curr, compute_obj(a_curr)
        
        if obj_new < best_score - 1e-10:
            best_score = obj_new
            best_seq = a_new.copy()

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Trust-region LP alternating with adaptive diversification (spectral perturbation to escape local minima)
2. Log-sum-exp smoothed gradient descent with FFT-based gradients

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_alternate_trust(a, max_iters=100, time_budget=30.0, trust_init=0.4, verbose=False):
    """
    LP alternating with trust region (SQP subproblem).
    Solves: max sum(g) s.t. (a*g)[k] <= M, g >= 0
    Then does line search with trust region control.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    trust = trust_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining = time_budget - (time.monotonic() - start)

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': min(8.0, remaining * 0.4)})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s < 1e-10:
                break
            g = g / s
        except Exception:
            break

        # Line search within trust region
        best_t_obj = np.inf
        best_t_val = 0.0
        best_a_local = best_a.copy()

                best_t_val = t

def spectral_flatten(a, n_iters=80, time_budget=25.0, blend_rate=0.5):
    """
    Iterative spectral flattening: alternate between time domain (non-neg + sum=1)
    and frequency domain (flatten power spectrum).
    Key insight: flat |FFT(a)|^2 means flat convolution (by Parseval).
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2

        # Target: flat power spectrum (DC preserved for sum constraint)
        # Use geometric mean as a soft target
        non_dc = power[1:]
        non_dc = np.maximum(non_dc, 1e-20)
        log_target = np.log(non_dc).mean()
        target_power = np.ones(N)
        target_power[0] = power[0]
        target_power[1:] = np.exp(log_target)

        # Adaptive blend: start gentle, get more aggressive
        alpha = blend_rate * (0.3 + 0.7 * it / max(n_iters, 1))
        alpha = min(alpha, 0.85)

        # Interpolate in log domain for better behavior
        power_interp = power.copy()
        power_interp[1:] = np.exp((1 - alpha) * np.log(np.maximum(power[1:], 1e-20)) +
                                   alpha * np.log(target_power[1:]))
        power_interp[1:] = np.maximum(power_interp[1:], 1e-20)

        # Reconstruct with updated magnitude, preserved phase
        phase = np.angle(fa)
        magnitude = np.sqrt(power_interp)
        fa_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.ifft(fa_new).real[:n]
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def lse_gradient_descent(a, max_iters=150, time_budget=25.0, tau_start=0.05, tau_end=0.008):
    """
    Gradient descent on log-sum-exp smoothed max(conv).
    Uses FFT-based gradient computation for efficiency.
    As tau -> 0, converges to true max.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1

    step = 0.08
    momentum = 0.85
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)

        conv = compute_conv(a)
        cmax = float(npmax(conv)) if False else float(np.max(conv))
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # FFT-based gradient: grad[i] = 2 * sum_j a[j] * weights[i+j]
        w_rev = weights[::-1]
        fa_fft = rfft(a, N_fft)
        fw_fft = rfft(w_rev, N_fft)
        conv_aw = irfft(fa_fft * fw_fft, N_fft)

        grad = np.zeros(n)
        for i in range(n):
            idx = 2 * n - 2 - i
            if 0 <= idx < len(conv_aw):
                grad[i] = 2.0 * conv_aw[idx]

        # Center to maintain sum constraint
        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-15:
            break

        # Nesterov-like momentum step
        v = momentum * v - step * grad_centered / (gnorm / n + 1e-10)
        a_new = a + v
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        step *= 0.997

    return best_a, best_obj


def chase_peak_topk(a, n_steps=80, time_budget=15.0, k_frac=0.1):
    """
    Chase-the-peak using top-k convolution peaks for subgradient.
    More robust than single peak tracing.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1

    k_top = max(3, int((2 * n - 1) * k_frac))

    for _ in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1][:k_top]
        weights = np.zeros(2 * n - 1)
        for rank, idx in enumerate(sorted_idx):
            weights[idx] = np.exp(-rank * 0.4)

        w_rev = weights[::-1]
        fa_fft = rfft(a, N_fft)
        fw_fft = rfft(w_rev, N_fft)
        conv_aw = irfft(fa_fft * fw_fft, N_fft)

        grad = np.zeros(n)
        for i in range(n):
            idx = 2 * n - 2 - i
            if 0 <= idx < len(conv_aw):
                grad[i] = conv_aw[idx]

        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-15:
            break

        improved = False
        for ss in [0.03, 0.07, 0.12, 0.2]:
            a_try = a - ss * grad_centered / (gnorm / n + 1e-10)
            a_try = project_nonneg_sum1(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_obj


def spectral_perturb(a, rng, strength=0.1):
    """
    Apply a spectral perturbation to escape local minima.
    Randomly modify the power spectrum, then project back.
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1

    fa = np.fft.fft(a, N)
    power = np.abs(fa) ** 2

    # Random spectral perturbation
    pert = np.ones(N)
    n_pert = max(5, N // 10)
    pert_idx = rng.choice(N, size=n_pert, replace=False)
    pert[pert_idx] = rng.uniform(1 - strength, 1 + strength, n_pert)
    pert[0] = 1.0  # Preserve DC

    fa_pert = fa * np.sqrt(pert)
    a_pert = np.fft.ifft(fa_pert).real[:n]
    a_pert = project_nonneg_sum1(a_pert)
    return a_pert


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangle
    tri = np.maximum(0, 1 - 2 * np.abs(x - 0.5)) + 0.01
    inits.append(tri / tri.sum())

    # Sine
    sine = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Gaussian
    gauss = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.01
    inits.append(gauss / gauss.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Two-peak
    two = np.exp(-((x - 0.3) ** 2) / 0.03) + np.exp(-((x - 0.7) ** 2) / 0.03)
    inits.append(two / two.sum())

    # Flat spectrum random phase
    N = 1
    while N < 2 * n:
        N <<= 1
    for _ in range(2):
        phase = rng.uniform(0, 2 * np.pi, N)
        fa = np.ones(N) * np.exp(1j * phase)
        fa[0] = 1.0
        a_flat = np.fft.ifft(fa).real[:n]
        a_flat = project_nonneg_sum1(a_flat)
        inits.append(a_flat)

    if parent is not None:
        inits.append(project_nonneg_sum1(resample_sequence(parent, n)))
        pert = project_nonneg_sum1(resample_sequence(parent, n) *
                                    np.exp(rng.normal(0, 0.15, n)))
        inits.append(pert)

    return inits


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-strategy optimization with adaptive diversification.
    Strategy:
    1. Quick screen diverse initializations (10% time)
    2. Intensive LP alternating with periodic spectral diversification (50% time)
    3. Multi-length exploration (15% time)
    4. Final intensive polish with LSE smoothing (25% time)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # === Phase 1: Diverse initialization screening (8% time) ===
    print("\n=== Phase 1: Diverse Initialization Screening ===")
    phase1_end = start_time + max(40.0, (deadline - start_time) * 0.08)
    n_options = list(dict.fromkeys([
        n_parent, max(200, n_parent - 100), n_parent + 50,
        n_parent + 100, max(200, n_parent - 200), n_parent + 200
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2000]

    candidates = []
    for n_try in n_options:
        if time.monotonic() > phase1_end:
            break
        inits = diverse_inits(n_try, rng, parent)
        for init in inits:
            if time.monotonic() > phase1_end:
                break
            a_q, obj_q = chase_peak_topk(init, n_steps=30, time_budget=3.0)
            if obj_q < np.inf:
                candidates.append((obj_q, a_q))

    candidates.sort(key=lambda x: x[0])
    if candidates:
        print(f"  Screened {len(candidates)}, best: {candidates[0][0]:.8f}")
        if candidates[0][0] < best_score:
            best_score = candidates[0][0]
            best_seq = candidates[0][1].copy()

    # === Phase 2: Intensive optimization with diversification (50% time) ===
    print(f"\n=== Phase 2: Intensive Optimization (best={best_score:.8f}) ===")
    phase2_end = start_time + (deadline - start_time) * 0.58
    a_curr = project_nonneg_sum1(best_seq.copy())
    n_curr = len(a_curr)
    diversify_counter = 0
    last_improve = time.monotonic()

    while time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        if remaining < 15:
            break

        # Determine strategy based on stagnation
        stagnant = (time.monotonic() - last_improve) > 30.0

        if stagnant and diversify_counter < 8:
            # Diversify: spectral perturbation + quick re-optimization
            a_pert = spectral_perturb(a_curr, rng, strength=0.15 + 0.05 * diversify_counter)
            a_pert, obj_pert = chase_peak_topk(a_pert, n_steps=20, time_budget=5.0)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_curr = a_pert.copy()
                last_improve = time.monotonic()
                print(f"  Diversify {diversify_counter}: {best_score:.8f}")
            else:
                a_curr = spectral_perturb(a_curr, rng, strength=0.1)
                a_curr, _ = chase_peak_topk(a_curr, n_steps=15, time_budget=4.0)
            diversify_counter += 1
            continue

        # Main optimization: cycle through strategies
        budget = min(remaining * 0.3, 25.0)
        strat = int((time.monotonic() - start_time) / 20) % 4

        if strat == 0:
            a_new, obj_new = lp_alternate_trust(a_curr, max_iters=40, time_budget=budget)
        elif strat == 1:
            a_new, obj_new = spectral_flatten(a_curr, n_iters=60, time_budget=budget)
        elif strat == 2:
            a_new, obj_new = lse_gradient_descent(a_curr, max_iters=80, time_budget=budget)
        else:
            a_new, obj_new = chase_peak_topk(a_curr, n_steps=50, time_budget=budget)

        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a_curr = a_new.copy()
            last_improve = time.monotonic()
            print(f"  Update: {best_score:.8f} (strat={strat})")
        elif obj_new < compute_obj(a_curr):
            a_curr = a_new.copy()

    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    phase3_end = time.monotonic() + max(40.0, (deadline - time.monotonic()) * 0.15)
    n_best = len(best_seq)
    n_tries = list(dict.fromkeys([
        n_best - 50, n_best - 100, n_best + 50, n_best + 100,
        n_best + 150, n_best - 150, n_best + 250
    ]))
    n_tries = [n for n in n_tries if 100 <= n <= 2500]
    per_n_budget = max(10.0, (phase3_end - time.monotonic()) / max(len(n_tries), 1))

    for n_try in n_tries:
        if time.monotonic() > phase3_end:
            break
        a_res = project_nonneg_sum1(resample_sequence(best_seq, n_try))
        a_opt, obj_opt = chase_peak_topk(a_res, n_steps=40, time_budget=per_n_budget * 0.3)
        a_opt2, obj_opt2 = lp_alternate_trust(a_opt, max_iters=30, time_budget=per_n_budget * 0.5)
        if obj_opt2 < obj_opt:
            a_opt, obj_opt = a_opt2, obj_opt2
        a_opt3, obj_opt3 = spectral_flatten(a_opt, n_iters=40, time_budget=per_n_budget * 0.2)
        if obj_opt3 < obj_opt:
            a_opt, obj_opt = a_opt3, obj_opt3

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={n_try}: {obj_opt:.8f}")

    # === Phase 4: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)

        while time.monotonic() < deadline - 10:
            rem = deadline - time.monotonic()
            if rem < 15:
                break

            # LP alternating (primary)
            a_p, obj_p = lp_alternate_trust(a_final, max_iters=30, time_budget=min(rem * 0.35, 25.0))
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_final = a_p.copy()
                print(f"  LP polish: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 10:
                break

            # LSE smoothing (secondary - helps with fine-tuning)
            a_p2, obj_p2 = lse_gradient_descent(a_final, max_iters=60, time_budget=min(rem * 0.3, 15.0),
                                                tau_start=0.03, tau_end=0.005)
            if obj_p2 < best_score:
                best_score = obj_p2
                best_seq = a_p2.copy()
                a_final = a_p2.copy()
                print(f"  LSE polish: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 10:
                break

            # Small perturbation + re-optimize (escape mechanism)
            a_pert = a_final * np.exp(rng.normal(0, 0.008, n_final))
            a_pert = project_nonneg_sum1(a_pert)
            a_pert, obj_pert = lp_alternate_trust(a_pert, max_iters=20, time_budget=min(rem * 0.3, 15.0))
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_final = a_pert.copy()
                print(f"  Perturb+LP: {best_score:.8f}")
            elif obj_pert < compute_obj(a_final):
                a_final = a_pert.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
