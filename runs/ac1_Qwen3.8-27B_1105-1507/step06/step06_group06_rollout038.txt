LP-alternating optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Core method: iterate a <- argmin_g max_k (a*g)[k] (solved as LP).

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_from_conv(a, conv):
    """Compute objective from precomputed convolution."""
    n = len(a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    conv = compute_conv_fft(a)
    return compute_obj_from_conv(a, conv)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_rhs(a):
    """
    Build the constraint matrix for the LP.
    Returns A_ub (shape (2n-1, n+1)) and b_ub (shape (2n-1,)).
    The constraints are: (a*g)[k] - t <= 0 for k=0,...,2n-2.
    Variable ordering: (g_0, ..., g_{n-1}, t)
    """
    n = len(a)
    n_con = 2 * n - 1
    A_ub = np.zeros((n_con, n + 1))
    # Toeplitz part
    for j in range(n):
        # (a*g)[k] includes a[k-j]*g[j] for valid k-j
        k_start = j
        k_end = min(j + n, n_con)
        A_ub[k_start:k_end, j] = a[:k_end - k_start]
    # -t column
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_con)
    return A_ub, b_ub


def solve_minpeak_lp(a, max_time=10.0):
    """
    Solve: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.

    """
    n = len(a)
    if n > 3000:
        return None, None
    
    A_ub, b_ub = build_toeplitz_rhs(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',

                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:

            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                t_opt = float(res.x[n])
                return g, t_opt
    except Exception:
        pass
    return None, None

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def spectral_precondition(a, n_iters=20, gamma=0.5):
    """
    Spectral pre-conditioning: iteratively flatten the power spectrum.
    Reduces the dynamic range of |A_m| to help LP converge faster.
    """
    n = len(a)
    a = normalize(a.copy())
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for _ in range(n_iters):
        A = np.fft.fft(a, N)
        # Compress dynamic range of |A|
        mag = np.abs(A)
        phase = np.angle(A)
        mag_max = mag.max()
        if mag_max < 1e-15:
            break
        mag_norm = mag / mag_max
        # Power compression
        mag_new = mag_norm ** gamma * mag_max
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a = normalize(a_new)
    
    return a

"""
LP-alternating optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Core method: iterate a <- argmin_g max_k (a*g)[k] solved as a linear program.
Enhanced with: spectral pre-conditioning, multi-n exploration, diverse initializations,
perturbation restarts, and subgradient-assisted escapes from local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_from_conv(a, conv):
    """Compute objective from precomputed convolution."""
    n = len(a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    conv = compute_conv_fft(a)
    return compute_obj_from_conv(a, conv)


def normalize(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_constraints(a):
    """
    Build A_ub for the min-peak LP.
    Constraints: (a*g)[k] - t <= 0, k=0,...,2n-2.
    Variable ordering: (g_0,...,g_{n-1}, t).
    Returns A_ub (2n-1 x n+1).
    """
    n = len(a)
    n_con = 2 * n - 1
    A_ub = np.zeros((n_con, n + 1))
    for j in range(n):
        k_start = j
        k_end = min(j + n, n_con)
        A_ub[k_start:k_end, j] = a[:k_end - k_start]
    A_ub[:, n] = -1.0
    return A_ub


def solve_minpeak_lp(a, max_time=10.0):

    Returns (g_normalized, t_opt) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    A_ub = build_toeplitz_constraints(a)
    b_ub = np.zeros(2 * n - 1)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * (n + 1)

        if res.status == 0:

def subgradient_step(a, n_steps=30, step_init=0.05, decay=0.97):
    """Chase-the-peak subgradient method."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    step = step_init

    for _ in range(n_steps):
        conv = compute_conv_fft(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad_centered = grad - np.mean(grad)
        a_try = normalize(a - step * grad_centered)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        step *= decay

    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def diverse_init(n, mode, rng=None):
    """Generate diverse initializations."""
    if mode == 'uniform':
        a = np.ones(n)
    elif mode == 'sinc':
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0.01)
    elif mode == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2 / 3.0)
    elif mode == 'raised_cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif mode == 'flat_top':
        x = np.linspace(0, 1, n)
        a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * x) +
             0.277263158 * np.cos(4 * np.pi * x) - 0.083578947 * np.cos(6 * np.pi * x) +
             0.006947368 * np.cos(8 * np.pi * x))
        a = np.maximum(a, 0.001)
    elif mode == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01)
    elif mode == 'truncated_sinc':
        x = np.linspace(-2, 2, n)
        a = np.sinc(x / 2.0) ** 2
        a = np.maximum(a, 0.005)
    elif mode == 'random':
        if rng is None:
            rng = np.random.default_rng(99)
        a = rng.random(n) + 0.1
    else:
        a = np.ones(n)

    return normalize(a)


def perturbation_search(a, rng, n_tries=8, sigma=0.04, time_budget=40.0):
    """Try random perturbations followed by LP optimization."""
    best_a = a.copy()
    best_obj = compute_obj(a)
    n = len(a)
    start = time.monotonic()

    for i in range(n_tries):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        per_try = min(8.0, (time_budget - elapsed) / max(1, n_tries - i))
        if per_try < 2.0:
            break

        # Different perturbation strategies
        if i % 3 == 0:
            a_pert = a * np.exp(rng.normal(0, sigma, n))
        elif i % 3 == 1:
            # Window perturbation
            wsize = max(5, n // 20)
            wstart = rng.integers(0, n - wsize)
            a_pert = a.copy()
            a_pert[wstart:wstart + wsize] *= np.exp(rng.normal(0, sigma * 2, wsize))
        else:
            # Anti-peak perturbation: reduce at high-conv positions, increase elsewhere
            conv = compute_conv_fft(a)
            k_peak = int(np.argmax(conv))
            a_pert = a.copy()
            # Reduce elements that contribute to the peak
            for i_idx in range(n):
                j = k_peak - i_idx
                if 0 <= j < n:
                    a_pert[i_idx] *= (1.0 - sigma)
            a_pert = normalize(a_pert)

        a_pert = normalize(a_pert)
        a_opt, obj_opt = lp_alternate(a_pert, max_iters=150, tol=1e-11,
                                      time_budget=per_try, full_step=True)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a, best_obj


def lp_with_subgradient_mix(a, time_budget=60.0, lp_iters=100, sub_interval=20, sub_steps=10):
    """
    LP alternating with periodic subgradient steps to escape local minima.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(lp_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        remaining = time_budget - elapsed
        if remaining < 2.0:
            break

        # Periodically insert subgradient steps
        if it > 0 and it % sub_interval == 0:
            a_sub, obj_sub = subgradient_step(a, n_steps=sub_steps, step_init=0.03)
            if obj_sub < best_obj:
                best_obj = obj_sub
                best_a = a_sub.copy()
            a = normalize((best_a + a) / 2.0)

        g, t_opt = solve_minpeak_lp(a, max_time=min(15.0, remaining * 0.8))
        if g is None:
            break

        a_new = normalize(g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None
