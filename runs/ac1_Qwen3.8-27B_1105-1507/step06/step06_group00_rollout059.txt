import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def mirror_descent_step(a, eta, k_star, top_k=1):
    """One mirror descent step. Returns new a."""
    n = len(a)
    # Get top k peaks
    conv = compute_conv(a)
    if top_k > 1:
        # Use weighted subgradient from top peaks
        top_indices = np.argsort(conv)[-top_k:][::-1]
        weights = conv[top_indices]
        weights = weights / weights.sum()
        g = np.zeros(n)
        for w, k in zip(weights, top_indices):
            for j in range(n):
                j2 = int(k) - j
                if 0 <= j2 < n:
                    g[j] += w * 2 * a[j2]
    else:
        g = np.zeros(n)
        k = int(k_star)
        for j in range(n):
            j2 = k - j
            if 0 <= j2 < n:
                g[j] = 2 * a[j2]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 1e-300)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    a_new = a_new / s
    return a_new

Uses multiplicative weights with adaptive step size, momentum, and periodic LP restarts.
Exploits symmetry and uses multi-scale approach.
"""

from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""

        g[idx] = 2.0 * a[k - idx]

def soft_subgradient(a, T=50.0):
    """Compute smoothed subgradient using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    # Numerically stable log-sum-exp
    conv_max = np.max(conv)
    shifted = conv - conv_max
    exp_shifted = np.exp(T * shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Weighted subgradient
    g = np.zeros(n)
    # Only process significant weights
    significant = weights > 1e-10
    for k in np.where(significant)[0]:
        w = weights[k]
        j_min = max(0, int(k) - (n - 1))
        j_max = min(n - 1, int(k))
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            g[idx] += w * 2.0 * a[int(k) - idx]
    return g, conv

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def get_subgradient(a, k_star, n=None):

def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def proj_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n+1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

One more thought: **should I try to exploit the structure of the problem more directly?**

The key observation is that the subgradient at peak k is `g[j] = 2*a[k-j]`. This is a "reflected" version of `a`. So the mirror descent step is:
```
a_new[j] ∝ a[j] * exp(-eta * 2*a[k-j])
```

This is like a "mutual inhibition" between `a[j]` and `a[k-j]`. If both are large, they both get reduced. This is similar to competitive dynamics in ecology.

The fixed point of this dynamics (when it converges) satisfies:
```
a[j] * exp(-eta * 2*a[k*-j]) = constant (for all j in the support)
```
This means `log(a[j]) - eta * 2*a[k*-j] = constant`, or:
```
a[j] = C * exp(eta * 2*a[k*-j])
```

This is a self-consistent equation. Solving it analytically is hard, but it suggests that the optimal solution has a specific structure: the value at position j is an exponential of the value at the "mirror" position.

If the peak is at k*=n-1 (center, which is common for symmetric-ish distributions), then:
```
a[j] = C * exp(eta * 2*a[n-1-j])
```

This is a "dual" relationship between positions. If `a` is symmetric (a[j] = a[n-1-j]), this becomes:
```
a[j] = C * exp(eta * 2*a[j])
```
Which is a fixed-point equation for each j independently. The solution is `a[j] = W(2C*eta)/eta` where W is the Lambert W function. This gives a constant solution (all a[j] equal), which is the uniform. So the only symmetric fixed point is the uniform, confirming my earlier analysis.

For asymmetric solutions, the peak is not at the center, and the fixed-point equation couples different positions in a non-trivial way.

This analysis suggests that the optimal solution has a "self-similar" structure where each position's value is determined by its mirror position's value. This is an interesting structural property that could guide the optimization.

    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    return 2.0 * n * float(np.max(conv))


def get_subgradient(a, k_star):

        g[idx] = 2.0 * a[k - idx]
    return g
