def fast_conv(a):
    """FFT-based linear self-convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:

    if fft_size is None:
        fft_size = 1
        while fft_size < (2 * n - 1) + n - 1:
            fft_size <<= 1

    if lr is None:
        # Lipschitz constant of 2*Q*a is 4*||Q||_2 <= 4 * max_i sum_j |q[i+j]|
        row_max = 0.0
        for i in range(0, n, max(1, n // 50)):
            row_max = max(row_max, q[i:i + n].sum())
        L = 2.0 * row_max
        lr = 1.0 / max(L, 1e-10)

    a_prev = a.copy()
    for it in range(max_iters):
        grad = 2.0 * hankel_mv(q, a, fft_size)
        a_new = a - lr * grad
        a_new = proj_simplex(a_new)

        # Check convergence
        if it > 10 and np.max(np.abs(a_new - a)) < 1e-8:
            break
        a = a_new

    return a


def dual_ascent(a_init, time_budget, alpha0=1.0, inner_iters=200, verbose=True):
    """Dual ascent (multiplicative weights) with QP primal solves.
    
    The problem min max_k (a*a)[k] s.t. sum(a)=1, a>=0 is convex.
    Dual: max_{lam>=0, sum(lam)=1} min_{a in simplex} sum_k lam_k (a*a)[k]
    Dual update: lam_k <- lam_k * exp(alpha * (a*a)[k]) / Z
    """
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    # FFT size precomputation
    fft_size_q = 1
    while fft_size_q < conv_len + n - 1:
        fft_size_q <<= 1
    fft_size_c = 1
    while fft_size_c < conv_len:
        fft_size_c <<= 1

    # Initialize dual variables (uniform)
    lam = np.ones(conv_len) / conv_len

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t_start = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t_start < time_budget:
        # Solve QP primal (warm-start from current a)
        a = solve_qp_projected_grad(a, lam, max_iters=inner_iters, fft_size=fft_size_q)

        # Compute autocorrelation
        c = fast_conv(a)
        M = c.max()

        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Dual update (multiplicative weights)
        alpha = alpha0 / (1.0 + 0.005 * it)
        # Clip for numerical stability
        c_clipped = np.clip(c, 0, None)
        log_lam = np.log(lam + 1e-30) + alpha * c_clipped
        log_lam -= log_lam.max()
        lam = np.exp(log_lam)
        lam = lam / lam.sum()

        it += 1

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    DA it={it} M={best_M:.12f} J={2*n*best_M:.6f} "
                  f"alpha={alpha:.4f} no_improve={no_improve} t={elapsed:.0f}s", flush=True)

        if no_improve > 200:
            break

    return best_a


def lp_best_response(a, M, n_active=50, time_limit=20):
    """Solve LP best response: max sum(g) s.t. (a*g)[k] <= M for top n_active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])

    if len(rows) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_polish(a_init, max_iters, deadline, n_active=40):
    """LP mirror descent polish."""
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    cur_active = n_active

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_best_response(a, best_M, cur_active, time_limit=15)
        if g is None:
            cur_active = min(cur_active * 2, 2 * n - 1)
            no_improve += 1
            if no_improve > 10:
                break
            continue

        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 0.01:
                continue
            a_trial = a_trial / st
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            if cur_active > 15:
                cur_active = max(15, int(cur_active * 0.9))
        else:
            no_improve += 1
            cur_active = min(int(cur_active * 1.5), 2 * n - 1)
            if no_improve > 20:
                break

    return best_a


def coordinate_descent_1d(a, i, b, n):
    """Find optimal value for a[i] using golden section search."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur, 0.1)

    for _ in range(8):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 10.0:
            hi *= 2.0
        else:
            break

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Run coordinate descent sweeps with golden section."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)

        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                old = a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - old * old)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - old)
                sweep_improved = True

        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p_n / p_n.sum())

    inits.append(np.ones(n) / n)

    # Triangular (peak at center)
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Cosine
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    # Gaussian
    gauss = np.exp(-6.0 * (t - 0.5) ** 2) + 0.02
    inits.append(gauss / gauss.sum())

    # Asymmetric (front-loaded)
    asym = np.exp(-3.0 * t) + 0.05
    inits.append(asym / asym.sum())

    # Asymmetric (back-loaded)
    asym2 = np.exp(-3.0 * (1 - t)) + 0.05
    inits.append(asym2 / asym2.sum())

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_combined_optimization(a0, n, deadline, seed=0):
    """Run the full optimization pipeline: dual ascent -> LP -> coordinate descent."""
    best = np.maximum(a0, 0.0).copy()
    s = best.sum()
    if s < 1e-15:
        best = np.ones(n) / n
    else:
        best = best / s

    best_M = fast_conv(best).max()
    t_start = time.monotonic()
    total_time = deadline - t_start

    # Stage 1: Dual ascent (50% of time)
    t1 = t_start + 0.5 * total_time
    if time.monotonic() < t1:
        a_da = dual_ascent(best, t1 - time.monotonic(), alpha0=1.0,
                           inner_iters=150, verbose=True)
        M_da = fast_conv(a_da).max()
        if M_da < best_M:
            best_M = M_da
            best = a_da.copy()
        print(f"    After DA: M={best_M:.12f} J={2*n*best_M:.6f}", flush=True)

    # Stage 2: LP polish (25% of time)
    t2 = t_start + 0.75 * total_time
    if time.monotonic() < t2:
        max_lp = max(10, int((t2 - time.monotonic()) / 2.0))
        a_lp = run_lp_polish(best, max_lp, t2, n_active=40)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
        print(f"    After LP: M={best_M:.12f} J={2*n*best_M:.6f}", flush=True)

    # Stage 3: Second dual ascent pass (15% of time)
    t3 = t_start + 0.90 * total_time
    if time.monotonic() < t3:
        a_da2 = dual_ascent(best, max(5, t3 - time.monotonic()), alpha0=0.5,
                            inner_iters=100, verbose=False)
        M_da2 = fast_conv(a_da2).max()
        if M_da2 < best_M:
            best_M = M_da2
            best = a_da2.copy()
        print(f"    After DA2: M={best_M:.12f} J={2*n*best_M:.6f}", flush=True)

    # Stage 4: Coordinate descent (10% of time)
    if time.monotonic() < deadline:
        max_sw = max(3, int((deadline - time.monotonic()) / max(1.0, n / 50.0)))
        a_cd = run_coordinate_descent(best, max_sw, deadline, seed=seed)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
        print(f"    After CD: M={best_M:.12f} J={2*n*best_M:.6f}", flush=True)

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search using dual ascent (multiplicative weights with QP primal),
    LP best-response polish, and coordinate descent across multiple sequence lengths.
    Exploits convexity of the minimax autocorrelation problem.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512", flush=True)

    n_parent = parent.size

    # Determine lengths to try
    lengths_to_try = [n_parent]
    for delta in [-128, -64, 64, 128, 256, -256]:
        nl = n_parent + delta
        if 256 <= nl <= 2048 and nl not in lengths_to_try:
            lengths_to_try.append(nl)
    if 768 not in lengths_to_try:
        lengths_to_try.append(768)
    if 512 not in lengths_to_try:
        lengths_to_try.append(512)

    # Sort: parent first, then by closeness
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths_to_try = lengths_to_try[:5]

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length dual ascent search on {lengths_to_try}", flush=True)
    print(f"{'='*60}", flush=True)

    # Allocate time: 70% to the best-priority lengths, 30% reserved
    reserve_time = 300.0
    available = deadline - time.monotonic() - reserve_time
    n_lengths = len(lengths_to_try)
    # Weight: give more time to parent length
    weights = []
    for nl in lengths_to_try:
        if nl == n_parent:
            weights.append(3.0)
        elif abs(nl - n_parent) <= 128:
            weights.append(2.0)
        else:
            weights.append(1.0)
    total_w = sum(weights)
    time_per_length = [available * w / total_w for w in weights]

    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - reserve_time:
            break
        t0 = time.monotonic()
        time_alloc = min(time_per_length[li], deadline - reserve_time - time.monotonic())
        if time_alloc < 30:
            break

        print(f"\n  Length n={n}, time_alloc={time_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)

        best_for_n = None
        best_M_for_n = float('inf')

        # Time per init (use at most 3 inits for longer sequences)
        max_inits = min(len(inits), 4 if n <= 1024 else 2)
        time_per_init = time_alloc / max_inits

        for ii in range(max_inits):
            if time.monotonic() > deadline - reserve_time:
                break
            sub_deadline = min(t0 + (ii + 1) * time_per_init, deadline - reserve_time)
            if time.monotonic() >= sub_deadline:
                break

            a0 = np.maximum(inits[ii], 1e-8)
            a0 = a0 / a0.sum()

            print(f"    Init {ii}/{max_inits-1}, sub_deadline in {sub_deadline-time.monotonic():.0f}s", flush=True)
            a_opt = run_combined_optimization(a0, n, sub_deadline, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()

            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best M={best_M_for_n:.12f} J={J_val:.6f} "
                  f"(elapsed={time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("After Phase 1")

    # === Phase 2: Intensive LP polish on global best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive LP polish on global best", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 60:
        t_lp = min(deadline - 30, time.monotonic() + 150)
        a_lp = run_lp_polish(tracker.best, 100, t_lp, n_active=50)
        tracker.update(a_lp)

        # Second LP pass with different active set size
        if time.monotonic() < deadline - 30:
            t_lp2 = min(deadline - 20, time.monotonic() + 80)
            a_lp2 = run_lp_polish(a_lp, 50, t_lp2, n_active=20)
            tracker.update(a_lp2)

    tracker.report("After Phase 2")

    # === Phase 3: High-precision dual ascent refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Fine-tuning dual ascent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 40:
        t_da = min(deadline - 20, time.monotonic() + 120)
        a_da = dual_ascent(tracker.best, t_da - time.monotonic(),
                           alpha0=0.3, inner_iters=300, verbose=True)
        tracker.update(a_da)

    tracker.report("After Phase 3")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        cd_deadline = min(deadline - 5, time.monotonic() + 60)
        max_sw = max(5, int((cd_deadline - time.monotonic()) / max(1.0, tracker.best.size / 80.0)))
        a_cd = run_coordinate_descent(tracker.best, max_sw, cd_deadline, seed=99999)
        tracker.update(a_cd)

        # Final LP pass
        if time.monotonic() < deadline - 5:
            a_final = run_lp_polish(a_cd, 20, deadline, n_active=30)
            tracker.update(a_final)

    tracker.report("After Phase 4")

    # === Phase 5: Try resampling to nearby lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_cur = tracker.best.size
        for mult in [1.25, 0.8, 1.5]:
            if time.monotonic() > deadline - 15:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            if new_n in lengths_to_try:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 10, time.monotonic() + 25)
            a_opt = run_combined_optimization(a_ext, new_n, sub_dl, seed=seed + 777)
            tracker.update(a_opt)

    tracker.report("After Phase 5")

    # Final safety check
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
