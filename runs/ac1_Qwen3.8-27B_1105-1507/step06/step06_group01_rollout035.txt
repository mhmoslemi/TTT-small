    term1 = conv_af_p[n - 1: 2 * n - 1]

    # term2[m] = conv(p, a)[m] for m=0..n-1
    conv_p_a = np.fft.irfft(F_p * F_a, L)
    term2 = conv_p_a[0: n]

    grad = term1 + term2
    return smooth_val, grad


def exact_max_and_grad(a):
    """Compute exact max of conv(a,a) and a subgradient (average of active grads)."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    # Active set: all k where b[k] is close to max
    active = np.where(b >= M * (1 - 1e-9))[0]
    
    # Subgradient: average of gradients at active points
    # grad_b_k[i] = a[k-i] + a[i-k] (with a[j]=0 outside [0,n-1])
    grad = np.zeros(n)
    p = np.ones(len(active)) / len(active)
    
    # Use same FFT trick as smooth_max_and_grad
    L = 1
    while L < 3 * n:
        L <<= 1
    a_flip = a[::-1]
    F_af = np.fft.rfft(a_flip, L)
    F_p = np.fft.rfft(p, L)
    F_a = np.fft.rfft(a, L)
    
    conv_af_p = np.fft.irfft(F_af * F_p, L)
    term1 = conv_af_p[n - 1: 2 * n - 1]
    conv_p_a = np.fft.irfft(F_p * F_a, L)
    term2 = conv_p_a[0: n]
    
    grad = term1 + term2
    return M, grad


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_nonneg(v):
    """Project onto non-negative orthant."""
    return np.maximum(v, 0.0)


def normalize(a):
    """Normalize to sum=1, keeping non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def run_subgradient(a_init, deadline, tau=50, step_init=0.05):
    """Subgradient method with adaptive step size on smooth approximation."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    step = step_init
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        _, grad = smooth_max_and_grad(a, tau)
        # Project gradient to tangent of simplex (remove mean)
        grad_tan = grad - grad.mean()
        
        a_new = a - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase step if improving
            step = min(step * 1.01, 0.2)
        else:
            no_improve += 1
            step *= 0.95
        
        a = a_new
        
        if no_improve > 200:
            break
        # Adapt tau
        if it % 5000 == 0 and it > 0:
            tau *= 1.5
    
    return best_a


def run_fista(a_init, deadline, tau_start=30, tau_end=500, max_iters=20000):
    """FISTA (Nesterov accelerated) on smooth max approximation."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    # Lipschitz constant estimate
    L = tau_start * 2 * n
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_and_grad(y, tau)
        grad_tan = grad_y - grad_y.mean()
        
        # Backtracking line search for step size
        step = 1.0 / (tau * 2.5)
        a_new = y - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / s_y
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        
        # Track best
        if it % 10 == 0:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
    
    return best_a


def run_frank_wolfe(a_init, deadline, tau=50, max_iters=10000):
    """Frank-Wolfe (Conditional Gradient) on smooth approximation over simplex."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    step = 2.0 / 3.0  # FW step size
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        _, grad = smooth_max_and_grad(a, tau)
        grad_tan = grad - grad.mean()
        
        # Linear minimization oracle: pick argmin of gradient
        vertex = np.zeros(n)
        vertex[np.argmin(grad_tan)] = 1.0
        
        # FW step
        gamma = step / (it + 1) ** 0.5  # diminishing step
        a_new = (1 - gamma) * a + gamma * vertex
        a_new = normalize(a_new)
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Periodic check
        if it % 100 == 0:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    return best_a


def run_mirror_descent(a_init, deadline, step_init=0.3, max_iters=20000):
    """Mirror descent (multiplicative weights) on non-smooth max."""
    a = np.maximum(a_init.copy(), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    step = step_init
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        # Active set
        active_thresh = M * (1 - 1e-6)
        active = np.where(b >= active_thresh)[0]
        
        # Subgradient from active set
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 0.5 * a[j]
        grad /= max(1, len(active))
        grad_tan = grad - grad.mean()
        
        # Exponential tilting
        log_weights = -step * grad_tan
        log_weights -= log_weights.max()
        a_new = a * np.exp(log_weights)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        step *= 0.9995
        
        if no_improve > 500:
            break
    
    return best_a


def lp_best_response(a, n_active=30, time_limit=10):
    """Solve LP best response using top-n_active convolution coefficients.
    
    Primal: max sum(g) s.t. (a*g)[k] <= 1 for k in active, g >= 0
    (Normalize a to sum=1 first, so we're bounding conv by 1)
    """
    n = len(a)
    a_norm = normalize(a)
    b = fast_conv(a_norm)
    M = b.max()
    
    # Active set: top n_active coefficients
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()
    
    # Build constraint matrix: (a_norm * g)[k] = sum_j a_norm[k-j] * g[j] <= 1
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a_norm[m])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.ones(len(active))
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_polish(a_init, deadline, n_active_start=20, n_active_max=100):
    """LP mirror descent with adaptive active set."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    current_active = n_active_start
    
    for it in range(200):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response(a, current_active, time_limit=8)
        if g is None:
            current_active = min(current_active * 2, n_active_max)
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        # Line search
        improved = False
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            a_trial = normalize((1.0 - t) * a + t * g)
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            current_active = max(10, int(current_active * 0.8))
        else:
            no_improve += 1
            current_active = min(current_active + 10, n_active_max)
            if no_improve > 10:
                break
    
    return best_a


def coordinate_descent_step(a, i):
    """Optimize a single coordinate a[i] to minimize max convolution."""
    n = len(a)
    b = fast_conv(a)
    x_cur = a[i]
    
    # Affected k range: [i, min(2n-2, i+n-1)]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute
    c_vals = []
    slopes = []
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    # Unaffected max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    base_max = b[mask].max() if mask.any() else 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            m = max(m, vals.max())
        if quad_pos >= 0:
            m = max(m, c_arr[quad_pos] + x_test * x_test)
        return m
    
    # Golden section search
    lo = 0.0
    hi = max(3.0 * x_cur, np.sqrt(max(b.max(), 1e-10)) + 0.5, 1.0)
    
    for _ in range(5):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 10.0:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, deadline, max_sweeps=50, seed=0):
    """Run coordinate descent focusing on active coordinates."""
    rng = np.random.default_rng(seed)
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        order = np.arange(n)
        rng.shuffle(order)
        
        sweep_improved = False
        for i in order:
            x_opt = coordinate_descent_step(a, i)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 5
        if no_improve > 15:
            break
    
    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        # Resample parent
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            p = np.interp(t, t_old, parent)
        else:
            p = parent.copy()
        p = np.maximum(p, 0)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            p_n = np.maximum(p / p.sum() + rng.normal(0, sigma, n), 1e-4)
            inits.append(p_n / p_n.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Bell (concave)
    bell = 1.0 - (2 * t - 1) ** 2
    inits.append(bell / bell.sum())
    
    # Sine bell
    bell2 = np.sin(np.pi * t)
    inits.append(bell2 / bell2.sum())
    
    # U-shaped (convex)
    ushape = (2 * t - 1) ** 2 + 0.1
    inits.append(ushape / ushape.sum())
    
    # Cosine windows
    for freq in [1, 2, 3, 4]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    for width in [0.2, 0.3, 0.5]:
        gauss = np.exp(-0.5 * ((t - 0.5) / width) ** 2) + 0.01
        inits.append(gauss / gauss.sum())
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        smooth = np.maximum(smooth, 0.01)
        inits.append(smooth / smooth.sum())
    
    # "Optimal" shape: a[i] ~ 1/sqrt(2i+1) * 1/sqrt(2(n-1-i)+1) (edge-enhanced)
    edge = 1.0 / np.sqrt((2 * np.arange(n) + 1) * (2 * (n - 1 - np.arange(n)) + 1) + 1)
    inits.append(edge / edge.sum())
    
    # Power law
    for alpha in [0.5, 1.0, 1.5]:
        pl = (t + 0.01) ** alpha * ((1 - t) + 0.01) ** alpha
        inits.append(pl / pl.sum())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_optimization_pipeline(a0, n, deadline, seed=0):
    """Run full optimization pipeline: FISTA -> Mirror -> LP -> CD."""
    best = a0.copy()
    best_M = fast_conv(a0).max()
    
    elapsed_total = deadline - time.monotonic()
    if elapsed_total < 5:
        return best
    
    # Stage 1: FISTA (35% of time)
    t1 = time.monotonic() + 0.30 * elapsed_total
    a_f = run_fista(best, t1, tau_start=30, tau_end=300, max_iters=15000)
    if fast_conv(a_f).max() < best_M:
        best_M = fast_conv(a_f).max()
        best = a_f.copy()
    
    # Stage 2: Mirror descent (15%)
    t2 = time.monotonic() + 0.15 * elapsed_total
    if time.monotonic() < t2:
        a_md = run_mirror_descent(best, t2, step_init=0.3, max_iters=15000)
        if fast_conv(a_md).max() < best_M:
            best_M = fast_conv(a_md).max()
            best = a_md.copy()
    
    # Stage 3: LP polish (30%)
    t3 = time.monotonic() + 0.30 * elapsed_total
    if time.monotonic() < t3:
        a_lp = run_lp_polish(best, t3, n_active_start=20, n_active_max=80)
        if fast_conv(a_lp).max() < best_M:
            best_M = fast_conv(a_lp).max()
            best = a_lp.copy()
    
    # Stage 4: Coordinate descent (20%)
    if time.monotonic() < deadline:
        max_sw = max(5, int((deadline - time.monotonic()) / max(0.5, n / 80.0)))
        a_cd = run_coordinate_descent(best, deadline, max_sweeps=max_sw, seed=seed)
        if fast_conv(a_cd).max() < best_M:
            best_M = fast_conv(a_cd).max()
            best = a_cd.copy()
    
    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search to minimize J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FISTA, mirror descent, LP best-response, and coordinate descent
    with diverse initializations at n=768.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    n_target = 768
    if parent is not None and parent.size > 0:
        n_target = parent.size
    
    if parent is None:
        parent = np.ones(n_target)
        tracker.update(parent)
        print(f"No parent, using uniform n={n_target}, J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'='*60}", flush=True)
    print(f"OPTIMIZATION PHASE 1: Multi-init at n={n_target}", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Phase 1: Try multiple initializations
    inits = generate_inits(n_target, rng, parent)
    print(f"  {len(inits)} initializations generated", flush=True)
    
    # Time allocation: give more time to promising inits
    time_per_init = (deadline - time.monotonic() - 400) / max(1, len(inits))
    time_per_init = max(time_per_init, 20)
    
    best_for_n = tracker.best.copy() if tracker.best is not None else parent.copy()
    best_M = fast_conv(best_for_n).max()
    
    for ii, a0 in enumerate(inits):
        if time.monotonic() > deadline - 400:
            break
        sub_deadline = min(time.monotonic() + time_per_init, deadline - 400)
        if time.monotonic() >= sub_deadline:
            break
        
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        
        print(f"  Init {ii+1}/{len(inits)}: M_init={fast_conv(a0).max():.8f}", flush=True)
        
        a_opt = run_optimization_pipeline(a0, n_target, sub_deadline, seed=seed + ii)
        M_opt = fast_conv(a_opt).max()
        
        if M_opt < best_M:
            best_M = M_opt
            best_for_n = a_opt.copy()
            J_val = compute_J(best_for_n)
            print(f"    Improved! M={M_opt:.10f} J={J_val:.6f}", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("After Phase 1")
    
    # Phase 2: Intensive LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"OPTIMIZATION PHASE 2: Intensive LP polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t_lp = min(deadline - 200, time.monotonic() + 200)
        a_lp = run_lp_polish(tracker.best, t_lp, n_active_start=30, n_active_max=120)
        tracker.update(a_lp)
        
        # Second pass with tighter active set
        if time.monotonic() < deadline - 100:
            a_lp2 = run_lp_polish(a_lp, min(deadline - 100, time.monotonic() + 60),
                                  n_active_start=15, n_active_max=40)
            tracker.update(a_lp2)
    
    tracker.report("After Phase 2")
    
    # Phase 3: High-tau FISTA refinement
    print(f"\n{'='*60}", flush=True)
    print(f"OPTIMIZATION PHASE 3: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t3 = min(deadline - 80, time.monotonic() + 120)
        for tau in [100, 300, 800]:
            if time.monotonic() > t3:
                break
            a_f = run_fista(tracker.best, min(t3, time.monotonic() + 40),
                           tau_start=tau, tau_end=tau * 3, max_iters=8000)
            tracker.update(a_f)
    
    tracker.report("After Phase 3")
    
    # Phase 4: Coordinate descent polish
    print(f"\n{'='*60}", flush=True)
    print(f"OPTIMIZATION PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t_cd = min(deadline - 30, time.monotonic() + 100)
        max_sw = max(5, int((t_cd - time.monotonic()) / max(0.5, n_target / 60.0)))
        a_cd = run_coordinate_descent(tracker.best, t_cd, max_sweeps=max_sw, seed=99999)
        tracker.update(a_cd)
    
    tracker.report("After Phase 4")
    
    # Phase 5: Perturbation restart from best
    print(f"\n{'='*60}", flush=True)
    print(f"OPTIMIZATION PHASE 5: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_restarts = 5
        t_per_restart = (deadline - 30 - time.monotonic()) / n_restarts
        t_per_restart = max(t_per_restart, 10)
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            # Perturb the best solution
            sigma = 0.05 * (1 + 0.5 * r)
            a_pert = tracker.best + rng.normal(0, sigma, tracker.best.size)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            
            sub_dl = min(deadline - 30, time.monotonic() + t_per_restart)
            a_opt = run_optimization_pipeline(a_pert, n_target, sub_dl, seed=seed + 500 + r)
            tracker.update(a_opt)
            print(f"  Restart {r+1}: J={tracker.best_score:.8f}", flush=True)
    
    tracker.report("After Phase 5")
    
    # Phase 6: Final LP + CD polish
    print(f"\n{'='*60}", flush=True)
    print(f"OPTIMIZATION PHASE 6: Final polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 5:
        if time.monotonic() < deadline - 20:
            a_final_lp = run_lp_polish(tracker.best, min(deadline - 20, time.monotonic() + 30),
                                       n_active_start=25, n_active_max=60)
            tracker.update(a_final_lp)
        
        if time.monotonic() < deadline - 5:
            a_final_cd = run_coordinate_descent(tracker.best, deadline - 2,
                                                max_sweeps=20, seed=12345)
            tracker.update(a_final_cd)
    
    tracker.report("FINAL")
    
    # Ensure valid return
    if tracker.best is None:
        tracker.best = np.ones(n_target)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
