"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Quick multi-start smooth optimization for initialization
2. LP mirror descent with constraint reduction and adaptive step size
3. Local coordinate search for final polish
Key improvements over previous: faster LP via constraint pruning,
local search phase, and better time allocation favoring LP phase.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-11)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.35, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def lp_best_response_reduced(a, M, epsilon=0.01):
    """LP best response with constraint reduction: only include constraints near the max."""
    n = len(a)
    b = fast_conv(a)
    # Only include constraints where b[k] >= M * (1 - epsilon)

        active_k = np.where(b >= M * 0.9)[0]

        active_k = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for k in active_k:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k - active_k[0])  # remap to sequential
            cols.append(j)
            vals.append(a[k - j])
    
    # Actually let's keep original indices for clarity
    rows = []
    cols = []
    vals = []

        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))

        pass
    return None, active_k

            g = lp_best_response_full(a, M)
            active_k = None

        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Adaptive step size: try multiple values

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n))
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
    # Random smooth
    k = max(3, n // 20)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Symmetric
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    return inits

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))

        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def lp_solve_reduced(a, M, frac_constraints=0.4, timeout=15.0):
    """LP best response with constraint reduction.
    Only includes the top frac_constraints fraction of convolution values."""
    n = len(a)
    b = fast_conv(a)
    num_constraints = max(10, int((2 * n - 1) * frac_constraints))
    top_k = np.argpartition(b, -(num_constraints))[-num_constraints:]
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': timeout,
                             'dual_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_solve_full(a, M, timeout=30.0):
    """Full LP best response with all constraints."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': timeout,
                             'dual_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror_v2(a, deadline, use_full=False, frac_cons=0.4, max_iters=500):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    t_current = 0.3
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                # Try a smaller step
                t_current *= 0.5
                if t_current < 0.002:
                    break
        
        if use_full:
            g = lp_solve_full(a, M, timeout=20.0)
        else:
            g = lp_solve_reduced(a, M, frac_constraints=frac_cons, timeout=15.0)
        
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Backtracking line search
        improved = False
        step_candidates = [t_current, t_current * 0.6, t_current * 0.3,
                          t_current * 0.15, t_current * 1.5]
        step_candidates = sorted(set(min(max(s, 0.003), 0.9) for s in step_candidates))
        
        best_step_M = M
        best_step_a = None
        best_step_t = t_current
        
        for t_try in step_candidates:
            a_new = np.maximum((1.0 - t_try) * a + t_try * g, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new /= s
            M_new = float(fast_conv(a_new).max())
            if M_new < best_step_M:
                best_step_M = M_new
                best_step_a = a_new
                best_step_t = t_try
        
        if best_step_a is not None and best_step_M < M - 1e-13:
            a = best_step_a
            t_current = best_step_t
            if best_step_M < best_M:
                best_M = best_step_M
                best_a = a.copy()
                no_imp = 0
        else:
            no_imp += 1
            t_current *= 0.7
            if t_current < 0.002:
                break
    
    return best_a


def local_search(a, deadline, max_rounds=500, seed=99):
    """Local coordinate search with adaptive step sizes."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_M = float(fast_conv(a).max())
    rng = np.random.default_rng(seed)
    stall_count = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Try a batch of random coordinates
        batch_size = max(5, n // 20)
        indices = rng.integers(0, n, batch_size)
        any_improved = False
        
        for i in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-6:
                # Try making it positive
                for new_val in [1e-4, 1e-3, 0.001, 0.01]:
                    a_test = a.copy()
                    a_test[i] = new_val
                    s = a_test.sum()
                    a_test /= s
                    M_test = float(fast_conv(a_test).max())
                    if M_test < best_M - 1e-13:
                        a = a_test
                        best_M = M_test
                        any_improved = True
                        break
            else:
                # Try multiplicative perturbations
                for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05,
                             0.1, -0.1, 0.15, -0.15]:
                    new_val = old_val * (1.0 + delta)
                    if new_val < 0:
                        new_val = 0.0
                    a_test = a.copy()
                    a_test[i] = new_val
                    s = a_test.sum()
                    if s < 1e-10:
                        continue
                    a_test /= s
                    M_test = float(fast_conv(a_test).max())
                    if M_test < best_M - 1e-13:
                        a = a_test
                        best_M = M_test
                        any_improved = True
                        break
        
        if any_improved:
            stall_count = 0
        else:
            stall_count += 1
            # Try pair perturbations when stalling
            if stall_count > 3:
                found = False
                for _ in range(10):
                    i1, i2 = rng.integers(0, n, 2)
                    for d1 in [0.05, -0.05, 0.1, -0.1]:
                        for d2 in [0.05, -0.05, 0.1, -0.1]:
                            a_test = a.copy()
                            a_test[i1] = max(0.0, a_test[i1] * (1 + d1))
                            a_test[i2] = max(0.0, a_test[i2] * (1 + d2))
                            s = a_test.sum()
                            if s < 1e-10:
                                continue
                            a_test /= s
                            M_test = float(fast_conv(a_test).max())
                            if M_test < best_M - 1e-13:
                                a = a_test
                                best_M = M_test
                                found = True
                                break
                        if found:
                            break
                    if found:
                        break
                if found:
                    stall_count = 0
                elif stall_count > 20:
                    break
    
    return a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
