def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def golden_section_1d(func, a, b, tol=1e-10, max_iter=50):
    """Minimize func(x) for x in [a, b] using golden section search."""
    ratio = (np.sqrt(5) - 1) / 2
    c = b - ratio * (b - a)
    d = a + ratio * (b - a)
    fc = func(c)
    fd = func(d)

            break
        if fc < fd:

            d = c
            fd = fc
            c = b - ratio * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + ratio * (b - a)

                    g /= g.sum()

                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj

def annealing_moves(a, deadline, seed=0, verbose=False):
    """
    Simulated annealing with structured moves:
    - Swap mass between two positions
    - Split/merge adjacent positions
    - Perturb a random block
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    
    T = 0.01
    T_min = 0.0001
    cooling = 0.995
    no_improve = 0
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        move_type = rng.integers(0, 3)
        a_try = a.copy()
        
        if move_type == 0:
            # Swap mass between two random positions
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            amt = min(a[i], a[j]) * rng.uniform(0.1, 0.5)
            dir = 1 if a[i] > a[j] else -1
            a_try[i] -= dir * amt
            a_try[j] += dir * amt
            if a_try[i] < 0 or a_try[j] < 0:
                continue
        elif move_type == 1:
            # Perturb a random block
            block_size = rng.integers(3, min(20, n // 4))
            start = rng.integers(0, n - block_size)
            perturbation = rng.normal(0, 0.02, block_size)
            a_try[start:start+block_size] += perturbation
            a_try = np.maximum(a_try, 0)
            if a_try.sum() < 0.01:
                continue
        else:
            # Targeted: reduce contribution to peak
            peak_k = np.argmax(conv)
            lo = max(0, peak_k - n + 1)
            hi = min(n - 1, peak_k)
            i = rng.integers(lo, hi + 1)
            # Reduce a[i] a bit and add to a random position far from the peak pair
            reduce_amt = a[i] * rng.uniform(0.05, 0.3)
            a_try[i] -= reduce_amt
            # Add to a position that doesn't contribute to the peak
            candidates = [x for x in range(n) if abs(x - i) > n // 4]
            if candidates:
                j = candidates[rng.integers(0, len(candidates))]
                a_try[j] += reduce_amt
            else:
                a_try[0] += reduce_amt
        
        new_s = a_try.sum()
        if new_s < 1e-10:
            continue
        a_try = a_try / new_s
        
        # Compute new conv efficiently (full recompute for simplicity in annealing)
        new_conv = fast_conv(a_try)
        new_obj = 2.0 * n * new_conv.max()
        
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = a_try.copy()
            no_improve = 0
        else:
            delta_obj = new_obj - best_obj
            if rng.random() < np.exp(-delta_obj / T):
                no_improve += 1
            else:
                no_improve += 1
        
        a = a_try
        conv = new_conv
        T *= cooling
        if T < T_min:
            T = T_min
        
        if no_improve > 2000:
            # Restart from best with higher temperature
            a = best_a.copy()
            conv = fast_conv(a)
            T = 0.005
            no_improve = 0
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a

def generate_structured_inits(n, rng, parent=None):
    """Generate diverse initial sequences with structure."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))
    
    inits.append(np.ones(n) / n)
    
    # Number-theoretic patterns
    # Thue-Morse-like
    tm = np.zeros(n)
    idx = 0
    for i in range(n):
        bit_count = bin(i).count('1')
        tm[i] = 1.0 if bit_count % 2 == 0 else 0.3
    inits.append(project_simplex(tm))
    
    # Irrotational patterns using golden ratio
    phi = (1 + np.sqrt(5)) / 2
    golden = np.array([1.0 if (i * phi) % 1.0 < 0.618 else 0.2 for i in range(n)])
    inits.append(project_simplex(golden))
    
    # Multi-scale: combination of frequencies
    for f1, f2, amp in [(1, 7, 0.3), (2, 5, 0.4), (3, 8, 0.5)]:
        pattern = 1.0 + amp * np.cos(2 * np.pi * f1 * t) + 0.3 * amp * np.cos(2 * np.pi * f2 * t)
        inits.append(project_simplex(np.maximum(pattern, 0.01)))
    
    # Two-peak patterns (asymmetric)
    for c1, c2, w1, w2 in [(0.3, 0.7, 0.15, 0.15), (0.25, 0.75, 0.1, 0.2), (0.4, 0.8, 0.2, 0.1)]:
        pattern = np.exp(-((t - c1) / w1)**2) + 0.7 * np.exp(-((t - c2) / w2)**2) + 0.05
        inits.append(project_simplex(pattern))
    
    return inits

def solve_full(a, deadline, verbose=False):

    n = len(a)

    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_J(a)
    
    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new
    
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj
    
    # Stage 1: Golden-section CD (35% of time)
    t1 = time.monotonic() + 0.35 * remaining
    a = golden_coord_descent(a, max_sweeps=5000, deadline=t1, random_order=True, seed=42, peak_focus=True)
    a = try_update(a)
    
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj
    
    # Stage 2: Peak reduction (15% of time)
    t2 = time.monotonic() + 0.15 * remaining
    a_pr = peak_reduction(best_a, deadline=t2)
    a = try_update(a_pr)
    
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj
    
    # Stage 3: SCA-LP (25% of time)
    t3 = time.monotonic() + 0.25 * remaining
    a_sca, obj_sca = sca_lp_step(a, max_iters=100, deadline=t3)
    if obj_sca < best_obj:

        best_obj = obj_sca
    a = best_a.copy()

    if remaining < 10:

    # Stage 4: Annealing (15% of time)
