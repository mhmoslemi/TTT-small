Fast FFT-based gradient descent with adaptive soft-max for minimizing
2*n*max(conv(a,a))/(sum(a))^2, combined with LP polishing.
The gradient of the soft-max smoothed objective is a cross-correlation
computable in O(n log n) via FFT, enabling thousands of fast iterations.

import numpy as np
from scipy.fft import fft, ifft
import time
import cvxpy as cp


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N <<= 1
    return np.real(ifft(fft(a, N) * fft(b, N)))[:n]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def softmax_weights(conv, beta):
    """Compute softmax(beta * conv) with numerical stability."""
    c_max = np.max(conv)
    exp_vals = np.exp(beta * (conv - c_max))
    s = np.sum(exp_vals)
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / s


def compute_grad(a, beta):
    """
    Compute gradient of soft-max smoothed objective.
    grad[j] = 2 * sum_{l=0}^{n-1} p[j+l] * a[l]
    computed via FFT cross-correlation.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    p = softmax_weights(conv, beta)
    a_rev = a[::-1]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    cross = np.real(ifft(fft(p, N) * fft(a_rev, N)))
    grad = 2.0 * cross[n - 1:2 * n - 1]
    return grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    rho = np.where(1.0 - u / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        rho = np.array([n - 1])
    r = rho[-1]
    tau = (cumsum[r] - 1.0) / (r + 1.0)
    return np.maximum(v - tau, 0.0)

def build_conv_matrix(a):
    """Build (2n-1 x n) matrix such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_budget=10.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_objective(a)
    start = time.monotonic()

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a, time_budget=5.0)
        if g_lp is None:
            break

        # Line search
        best_t = 0.5
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_objective(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                best_t = t
                break

        if verbose and it % 5 == 0:
            print(f"    {tag} LP it{it}: obj={best_obj:.8f}")

    return best_a


def fejer_init(n):
    """Fejér kernel shape (raised cosine squared)."""
    x = np.linspace(0, 1, n)
    a = np.sinc(x - 0.5) ** 2
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def cosine_init(n):
    """Raised cosine window."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))
    a = a + 0.1
    a = a / np.sum(a)
    return a


def smoothed_top_hat(n, width_frac=0.7):
    """Top-hat with smooth edges."""
    a = np.zeros(n)
    center = n // 2
    half_width = int(n * width_frac / 2)
    lo = max(0, center - half_width)
    hi = min(n - 1, center + half_width)
    a[lo:hi+1] = 1.0
    # Smooth the edges
    edge = max(1, n // 20)
    for i in range(edge):
        frac = (i + 1) / (edge + 1)
        if lo - 1 - i >= 0:
            a[lo - 1 - i] = frac
        if hi + 1 + i < n:
            a[hi + 1 + i] = frac
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    return a


def symmetric_bump(n, exponent=2.0):
    """Symmetric bump: (1 - (2i/n - 1)^2)^exponent"""
    x = 2.0 * np.arange(n) / n - 1.0
    a = np.maximum(1.0 - x ** 2, 0) ** exponent
    a = a + 0.01
    a = a / np.sum(a)
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: fast gradient descent + LP polish, multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_objective(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = len(parent)
                print(f"Parent (n={n_parent}): obj={obj:.8f}")
    except Exception:
        parent = None
        print("No parent available, using default init")

    if best_seq is None:
        best_seq = fejer_init(800)
        best_score = compute_objective(best_seq)
        print(f"Default Fejer init (n=800): obj={best_score:.8f}")

    # === Phase 1: Intensive gradient descent on parent ===
    print(f"\n=== Phase 1: Gradient Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 400.0)
        a_opt = gradient_descent(best_seq, time_budget=alloc,
                                  beta_start=3.0, beta_end=80.0,
                                  lr_base=0.005, verbose=True, tag="P1")
        obj = compute_objective(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** P1 improved: obj={best_score:.8f} **")
        else:
            print(f"  P1: obj={obj:.8f} (no improvement)")

    # === Phase 2: LP polish ===
    print(f"\n=== Phase 2: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.15, 60.0)
        a_lp = lp_alternate(best_seq, time_budget=alloc, verbose=True, tag="P2")
        obj = compute_objective(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  ** P2 LP improved: obj={best_score:.8f} **")
        else:
            print(f"  P2 LP: obj={obj:.8f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_options = sorted(set([
        max(200, n_parent - 100), max(200, n_parent - 50),
        n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        500, 600, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc_grad = min(remaining * 0.2, 80.0)
        a_opt = gradient_descent(a_try, time_budget=alloc_grad,
                                  beta_start=3.0, beta_end=60.0,
                                  lr_base=0.004, verbose=False, tag=f"n{n_try}")
        obj = compute_objective(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 4: Different initializations ===
    print(f"\n=== Phase 4: Diverse Initializations ===")
    n_best = len(best_seq)
    inits = [
        ("Fejer", fejer_init(n_best)),
        ("Cosine", cosine_init(n_best)),
        ("TopHat", smoothed_top_hat(n_best)),
        ("Bump2", symmetric_bump(n_best, 2.0)),
        ("Bump3", symmetric_bump(n_best, 3.0)),
        ("Perturb", None),  # Will be filled
    ]
    # Add perturbed best
    inits[5] = ("PerturbBest", best_seq * np.exp(rng.normal(0, 0.05, n_best)))
    inits[5] = ("PerturbBest", np.maximum(inits[5][1], 0) / np.sum(np.maximum(inits[5][1], 0)))

    for name, a_init in inits:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_init = np.maximum(a_init, 0)
        s = np.sum(a_init)
        if s < 1e-10:
            continue
        a_init = a_init / s

        alloc = min(remaining * 0.15, 60.0)
        a_opt = gradient_descent(a_init, time_budget=alloc,
                                  beta_start=5.0, beta_end=80.0,
                                  lr_base=0.004, verbose=False, tag=name)
        obj = compute_objective(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** {name}: obj={best_score:.8f} **")
        else:
            print(f"  {name}: obj={obj:.8f}")

    # === Phase 5: Different n with diverse inits ===
    print(f"\n=== Phase 5: Diverse n + Inits ===")
    diverse_ns = [500, 700, 900, 1100, 1300]
    for n_try in diverse_ns:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Resample best and also try a fresh Fejer init
        a_try1 = resample_sequence(best_seq, n_try)
        a_try2 = fejer_init(n_try)

        alloc = min(remaining * 0.1, 50.0) / 2
        a_opt1 = gradient_descent(a_try1, time_budget=alloc,
                                   beta_start=3.0, beta_end=60.0,
                                   lr_base=0.004, verbose=False, tag=f"{n_try}r")
        obj1 = compute_objective(a_opt1)

        a_opt2 = gradient_descent(a_try2, time_budget=alloc,
                                   beta_start=3.0, beta_end=60.0,
                                   lr_base=0.004, verbose=False, tag=f"{n_try}f")
        obj2 = compute_objective(a_opt2)

        best_local = min(obj1, obj2)
        best_local_seq = a_opt1 if obj1 <= obj2 else a_opt2
        if best_local < best_score:
            best_score = best_local
            best_seq = best_local_seq.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: best={best_local:.8f}")

    # === Phase 6: Final intensive gradient + LP ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Long gradient run
        alloc_grad = min(remaining * 0.4, 120.0)
        a_final = gradient_descent(best_seq, time_budget=alloc_grad,
                                    beta_start=10.0, beta_end=100.0,
                                    lr_base=0.003, verbose=True, tag="Fin")
        obj = compute_objective(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  ** Final grad: obj={best_score:.8f} **")
        else:
            print(f"  Final grad: obj={obj:.8f}")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_lp = min(remaining * 0.3, 40.0)
        a_lp = lp_alternate(best_seq, time_budget=alloc_lp, verbose=True, tag="FinLP")
        obj = compute_objective(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj:.8f}")

    # One more short gradient pass after LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_last = gradient_descent(best_seq, time_budget=min(remaining * 0.5, 30.0),
                                   beta_start=20.0, beta_end=100.0,
                                   lr_base=0.002, verbose=False, tag="Last")
        obj = compute_objective(a_last)
        if obj < best_score:
            best_score = obj
            best_seq = a_last.copy()
            print(f"  ** Last grad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_objective(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
