"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def spectral_compress(a, power=0.5, n_iters=200, lmbda=0.5):
    """
    Iteratively compress the magnitude spectrum to flatten the autocorrelation.
    Works in the frequency domain directly.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Adaptive compression: different powers for different frequency bands
        # Low frequencies get more compression (they correspond to large-scale structure)
        freq_idx = np.arange(len(mag))
        freq_norm = freq_idx / (len(mag) - 1) if len(mag) > 1 else np.array([0])
        
        # Compression strength varies with frequency
        # More compression at low frequencies
        local_power = power * (1.0 + 2.0 * (1.0 - freq_norm))
        
        mag_normed = mag / (mag.max() + 1e-30)
        mag_compressed = mag_normed ** local_power
        mag_compressed *= mag.max()
        
        # Mix with original
        mag_new = (1 - lmbda) * mag + lmbda * mag_compressed
        
        # Preserve DC component (sum of a)
        mag_new[0] = mag[0]
        
        F_new = np.exp(1j * phase) * mag_new
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        
    return best_a

def smooth_max_grad(a, tau):
    """Compute gradient of smooth approximation to max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Shift to avoid overflow
    shifted = b - b_max
    exp_b = np.exp(tau * shifted)
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * p[k] * a[k-hi:k-lo+1][::-1]
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * p[k] * a[k - j]
    
    return grad, p, b, b_max


def run_nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,

        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):

        else:
            no_imp += 1
            if no_imp > 200:
                break
        
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta * (it + 1) / (it + 1))) / 2.0
        # Actually use standard Nesterov schedule
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0 if it > 1 else 1.0
        t_prev = t_k if it > 1 else 1.0
        # Simplified: use fixed momentum
        momentum = (t_curr - 1.0) / (t_curr + 1.0)
        
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:

        y = a_new + momentum * (a_new - a)

def build_A_sparse(n, a):
    """Build sparse matrix for LP: (a*g)(k) = sum_j a[k-j]*g[j] <= M for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_mix(history_X, history_FX, window=5):
    """Anderson acceleration mixing."""
    m = min(window, len(history_X))
    if m < 2:
        return history_FX[-1].copy()
    
    X = np.array(history_X[-m:])
    FX = np.array(history_FX[-m:])
    R = X - FX  # residuals, shape (m, n)
    
    G = R @ R.T + 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        alpha = np.linalg.solve(G, ones)
        alpha_sum = alpha.sum()
        if abs(alpha_sum) < 1e-15:
            alpha = ones / m
        else:
            alpha = alpha / alpha_sum
    except np.linalg.LinAlgError:
        alpha = ones / m
    
    alpha = np.clip(alpha, -2.0, 2.0)
    alpha = alpha / alpha.sum()
    
    mixed = np.zeros_like(X[0])
    for i in range(m):
        mixed += alpha[i] * FX[i]
    return np.maximum(mixed, 0)


def run_lp_anderson(a, deadline, max_iters=100, t_base=0.3, t_min=0.005, window=5):
    """LP mirror descent with Anderson acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    hist_X = []
    hist_FX = []
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        t = max(t_min, t_base * (0.95 ** it))
        a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        hist_X.append(a.copy())
        hist_FX.append(a_new.copy())
        if len(hist_X) > window + 2:
            hist_X.pop(0)
            hist_FX.pop(0)
        
        if len(hist_X) >= 3:
            a_and = anderson_mix(hist_X, hist_FX, window)
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
            M_and = fast_conv(a_and).max()
            if M_and <= fast_conv(a_new).max() * 1.02:
                a = a_and
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def make_asymmetric_inits(n, rng, parent=None):
    """Generate diverse asymmetric initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Linear ramp (strongly asymmetric)
    inits.append(np.maximum(0.1 + 0.9 * t, 0.01))
    inits.append(np.maximum(0.1 + 0.9 * (1 - t), 0.01))
    
    # Quadratic ramp
    inits.append(np.maximum(0.05 + 0.95 * t**2, 0.01))
    inits.append(np.maximum(0.05 + 0.95 * (1-t)**2, 0.01))
    
    # Exponential decay
    for rate in [2, 4, 6, 8]:
        inits.append(np.maximum(np.exp(-rate * t) + 0.01, 0.005))
        inits.append(np.maximum(np.exp(-rate * (1-t)) + 0.01, 0.005))
    
    # Two bump (asymmetric positions)
    for pos1, pos2, w1, w2 in [(0.2, 0.7, 0.1, 0.15), (0.15, 0.6, 0.08, 0.12),
                                (0.3, 0.8, 0.1, 0.1), (0.1, 0.5, 0.15, 0.1)]:
        bump = np.exp(-(t - pos1)**2 / (2*w1**2)) + 0.7 * np.exp(-(t - pos2)**2 / (2*w2**2))
        inits.append(np.maximum(bump + 0.02, 0.01))
    
    # Sawtooth-like
    for period in [4, 8, 16]:
        x = (np.arange(n) % period) / period
        inits.append(np.maximum(0.3 + 0.7 * x, 0.05))
    
    # Cosine with phase shift (asymmetric)
    for f in [1, 2, 3]:
        for phase in [0.3, 0.7, 1.2]:
            inits.append(np.maximum(0.4 + 0.5 * np.cos(2 * np.pi * f * t + phase), 0.01))
    
    # Product of ramp and oscillation
    inits.append(np.maximum(t * (0.5 + 0.5 * np.cos(2*np.pi*3*t)), 0.01))
    inits.append(np.maximum((1-t) * (0.5 + 0.5 * np.cos(2*np.pi*5*t)), 0.01))
    
    # Random smooth (asymmetric)
    kernel_len = max(3, n // 15)
    kern = np.ones(kernel_len) / kernel_len
    for _ in range(3):
        inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.02, 0.01))
    
    # If parent available, perturb it
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.2 * rng.normal(0, 1, n)), 0.01))
        # Shift parent
        shift = rng.integers(1, n // 4)
        p_shifted = np.roll(p, shift)
        inits.append(np.maximum(p_shifted, 0.01))
    
    # GS flattened random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(spectral_compress(a_rand, power=0.5, n_iters=50, lmbda=0.7))
    
    return inits


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:

            return

            self.seq = a.copy()

            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

def coarse_fine_optimize(a_coarse, n_fine, deadline):
