Actually, let me reconsider. The subgradient of max(conv) at the peak is just the indicator of the peak, which corresponds to the gradient of conv[k*] at the peak. This is a very sparse gradient (only points to the specific peak), which makes it noisy but potentially effective.

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def lse_smoothed_grad(a, beta):
    """Compute logsumexp-smoothed objective gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    
    ls = cmax + np.log(e_sum) / beta
    obj = 2.0 * n * ls / (sum_a * sum_a)
    
    # Gradient: 2 * cross_correlation(w, a)
    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad_ls = 2.0 * cc[:n]
    
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)
    return obj, grad_obj

def adam_lse_opt(a, time_budget=100.0, beta_start=5.0, beta_end=500.0, lr_start=0.05, lr_decay=0.99998, max_iters=50000):
    """Adam in log-space with LSE smoothing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_c = x - np.max(x)
        a = np.exp(x_c)
        a = a / np.sum(a)
        
        obj_sm, grad_a = lse_smoothed_grad(a, beta)
        grad_x = grad_a * a
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            x -= np.mean(x)

            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()
    
    return best_a

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2*n - 1)
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A

def lp_refine(a, time_budget=30.0):

    n = len(a)
    if n > 1000:
        return a
    
    start = time.monotonic()

            sg = float(np.sum(g))
