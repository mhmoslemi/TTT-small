import numpy as np

def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

    n = len(a)
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_lo = max(0, k - (n - 1))
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_descent(a_init, max_iters, deadline, n_active_start=40, n_active_end=10):
    """LP mirror descent with adaptive active set."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    cur_active = n_active_start

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        mine = max(5, cur_active)
        g = lp_best_response(a, best_M, mine, time_limit=8)
        if g is None:
            no_improve += 1
            if no_improve > 8:
                break
            cur_active = min(int(cur_active * 1.5), 2 * n - 1)
            continue

        improved = False
        for t in [0.35, 0.25, 0.15, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            cur_active = max(8, int(cur_active * 0.85))
        else:
            no_improve += 1
            cur_active = min(int(cur_active * 1.4), 2 * n - 1)
            if no_improve > 12:
                break

    return best_a


def run_mirror_descent_subgrad(a_init, max_iters, deadline, avg_k=3):
    """Mirror descent using average of top-k subgradients."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        # Use average of top k subgradients
        top_k = np.argsort(b)[::-1][:min(avg_k, len(b))]
        grad = np.zeros(n)
        for k_star in top_k:
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(top_k)

        grad_tan = grad - grad.mean()
        step = 0.15 / (1.0 + 0.01 * it)
        log_w = -step * grad_tan
        log_w -= log_w.max()
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()
        a = a_new

        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()

    return best_a


def local_search(a_init, deadline, seed, n_perturb=20, perturb_scale=0.05):
    """Local search: add small random perturbations and re-optimize each."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    time_per_perturb = (deadline - time.monotonic()) / max(1, n_perturb)
    time_per_perturb = min(time_per_perturb, 20.0)

    for p in range(n_perturb):
        if time.monotonic() > deadline:
            break
        # Random perturbation
        noise = rng.normal(0, perturb_scale, n)
        a_pert = np.maximum(a + noise, 0.0)
        s_p = a_pert.sum()
        if s_p < 0.01:
            continue
        a_pert = a_pert / s_p

        # Quick coordinate descent on perturbed
        sub_dl = min(deadline, time.monotonic() + time_per_perturb * 0.6)
        max_sw = max(3, int(time_per_perturb * 0.6 / max(0.1, n / 200.0)))
        a_cd = run_coordinate_descent_fast(a_pert, max_sw, sub_dl, seed=seed + p)
        M_cd = fast_conv(a_cd).max()

        if M_cd < best_M:
            best_M = M_cd
            best_a = a_cd.copy()
            print(f"    Local search p={p}: M={best_M:.10f}", flush=True)
            # Further polish from new best
            sub_dl2 = min(deadline, time.monotonic() + time_per_perturb * 0.4)
            max_sw2 = max(3, int(time_per_perturb * 0.4 / max(0.1, n / 200.0)))
            a_cd2 = run_coordinate_descent_fast(a_cd, max_sw2, sub_dl2, seed=seed + 1000 + p)
            M_cd2 = fast_conv(a_cd2).max()
            if M_cd2 < best_M:
                best_M = M_cd2
                best_a = a_cd2.copy()

    return best_a


def gaussian_mixture_init(n, K, rng):
    """Initialize using a mixture of Gaussian bumps."""
    mus = np.sort(rng.uniform(0.1, 0.9, K))
    sigmas = rng.uniform(0.02, 0.15, K)
    amps = rng.uniform(0.5, 2.0, K)
    base = rng.uniform(0.05, 0.2)

    t = np.arange(n) / n
    a = np.zeros(n)
    for j in range(K):
        a += amps[j] * np.exp(-0.5 * ((t - mus[j]) / sigmas[j]) ** 2)
    a += base
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def cosine_init(n, rng):
    """Initialize using cosine basis with random coefficients."""
    K = min(5, n // 20)
    freqs = rng.choice(np.arange(1, max(2, n // 10) + 1), size=K, replace=False)
    phases = rng.uniform(0, 2 * np.pi, K)
    amps = rng.uniform(0.3, 1.0, K)
    base = rng.uniform(0.5, 1.5)

    t = np.arange(n) / n
    a = np.full(n, base)
    for j in range(K):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t + phases[j])
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def multi_scale_cd(a_init, deadline, seed, scales=[1.0, 0.7, 0.5, 0.3, 0.2, 0.1]):
    """Multi-scale coordinate descent: optimize at reduced resolution then refine."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for scale in scales:
        if time.monotonic() > deadline:
            break
        red_n = max(32, int(n * scale))
        # Downsample
        if red_n < n:
            t_old = np.linspace(0, 1, n)
            t_new = np.linspace(0, 1, red_n)
            a_red = np.maximum(np.interp(t_new, t_old, a), 0.0)
        else:
            a_red = a.copy()
        a_red = a_red / a_red.sum()

        # Optimize at reduced scale
        sub_dl = min(deadline, time.monotonic() + max(5, (deadline - time.monotonic()) / len(scales)))
        max_sw = max(5, int(sub_dl / max(0.05, red_n / 500.0)))
        a_red_opt = run_coordinate_descent_fast(a_red, max_sw, sub_dl, seed=seed)
        M_red = fast_conv(a_red_opt).max()

        # Upsample back
        t_old = np.linspace(0, 1, red_n)
        t_new = np.linspace(0, 1, n)
        a_up = np.maximum(np.interp(t_new, t_old, a_red_opt), 0.0)
        a_up = a_up / a_up.sum()

        # Refine at full scale
        sub_dl2 = min(deadline, time.monotonic() + max(5, (deadline - time.monotonic()) / len(scales)))
        max_sw2 = max(5, int(sub_dl2 / max(0.1, n / 300.0)))
        a_full = run_coordinate_descent_fast(a_up, max_sw2, sub_dl2, seed=seed + 1)
        M_full = fast_conv(a_full).max()

        if M_full < best_M:
            best_M = M_full
            best_a = a_full.copy()
            a = a_full.copy()

    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)
