```python
def balanced_subgradient(a, S):
    """Compute the balanced subgradient for the active set S.
    
    Returns (direction, lambda_weights) where direction is the negated
    balanced subgradient (descent direction) and lambda_weights are the
    weights on the active constraints.
    """
    n = len(a)
    m = len(S)
    
    # Build G: G[i, k_idx] = 2 * a[S[k_idx] - i] (with bounds)
    G = np.zeros((n, m))
    for idx, k in enumerate(S):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                G[i, idx] = 2.0 * a[j]
    
    # Project G onto tangent space: G_tan = G - mean(G, axis=0)
    G_tan = G - G.mean(axis=0)
    
    # Solve QP: min ||G_tan^T w||^2 s.t. w >= 0, sum(w) = 1
    # This is: min w^T (G_tan G_tan^T) w s.t. w >= 0, sum(w) = 1
    Q = G_tan @ G_tan.T  # m x m
    
    # Solve with CVXPY or manually
    # For small m, use CVXPY
    import cvxpy as cp
    w = cp.Variable(m)
    obj = cp.Minimize(cp.quad_form(w, cp.psd_wrap(Q)))
    cons = [w >= 0, cp.sum(w) == 1]
    prob = cp.Problem(obj, cons)
    prob.solve(solver=cp.CLARABEL)
    
    if w.value is None:
        # Fallback: uniform weights
        w_val = np.ones(m) / m
    else:
        w_val = np.maximum(w.value, 0)
        w_val /= w_val.sum()
    
    # Compute balanced subgradient
    grad_bal = G.T @ w_val  # n-vector
    grad_bal_tan = grad_bal - grad_bal.mean()
    
    return -grad_bal_tan, w_val
```

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L2 autocorrelation warmup (exploits Toeplitz structure)
2. FISTA with log-sum-exp smoothing
3. Balanced subgradient active-set refinement (exact convex subgradient)
4. KKT-based optimality verification
5. Coordinate descent polish
Key improvement: Uses the exact (balanced) subgradient of the max function,
which is the optimal convex combination of active constraint gradients,
giving a much better descent direction than the LP best-response approach.
"""

import numpy as np

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (1.0 + u * rho_arr) > cv
    if not cond.any():
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        if not (~cond).any():
            rho = n - 1
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def get_active_set(b, threshold_frac=0.999):
    """Get active set: indices where b_k >= threshold_frac * b_max."""
    b_max = b.max()
    threshold = threshold_frac * b_max
    S = np.where(b >= threshold)[0]
    return S

def build_gradient_matrix(a, S):
    """Build G where G[i, idx] = 2*a[S[idx]-i] (with bounds)."""
    n = len(a)
    m = len(S)
    G = np.zeros((n, m))
    for idx, k in enumerate(S):
        # G[i, idx] = 2 * a[k-i] if 0 <= k-i < n
        i_vals = np.arange(n)
        j_vals = k - i_vals
        valid = (j_vals >= 0) & (j_vals < n)
        G[i_vals[valid], idx] = 2.0 * a[j_vals[valid]]
    return G

def solve_balanced_qp(G, max_iters=500):
    """Solve min ||G.T @ w||^2 s.t. w >= 0, sum(w) = 1 using Frank-Wolfe.
    
    G is n x m. The objective is w^T (G.G.T) w = ||G^T w||^2.
    We work in the tangent space: subtract the mean of each column of G.
    """
    n, m = G.shape
    # Work with tangent-space version
    G_tan = G - G.mean(axis=0, keepdims=True)
    Q = G_tan.T @ G_tan  # m x m
    
    # Frank-Wolfe
    w = np.ones(m) / m
    best_obj = float('inf')
    
    for it in range(max_iters):
        grad = 2.0 * Q @ w
        # Find s = e_{argmin grad}
        s_idx = int(np.argmin(grad))
        s = np.zeros(m)
        s[s_idx] = 1.0
        
        # Line search
        best_t = 0.0
        best_val = float('inf')
        for t in [2.0/(it+3), 1.0/(it+2), 0.5, 0.25, 0.1]:
            w_trial = (1.0 - t) * w + t * s
            val = w_trial @ Q @ w_trial
            if val < best_val:
                best_val = val
                best_t = t
                best_obj = min(best_obj, val)
        w = (1.0 - best_t) * w + best_t * s
    
    return np.maximum(w, 0.0)

def balanced_subgradient(a, S):
    """Compute the balanced subgradient direction."""
    n = len(a)
    m = len(S)
    
    if m == 0:
        return np.zeros(n)
    if m == 1:
        k = S[0]
        g = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]
        return -(g - g.mean())
    
    G = build_gradient_matrix(a, S)
    w = solve_balanced_qp(G)
    
    grad = G @ w  # n-vector
    direction = -(grad - grad.mean())  # negative of balanced subgradient (descent direction)
    return direction

def run_balanced_subgradient_method(a0, deadline, max_iters=100000, n_active_start=20, n_active_end=200):
    """Main optimization loop using balanced subgradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M_cur = b.max()
        
        # Adaptive active set size
        frac = it / max(1, min(it, 1000))
        n_active_target = int(n_active_start + (n_active_end - n_active_start) * frac)
        n_active_target = min(n_active_target, 2 * n - 1)
        
        # Get active set
        b_max = M_cur
        # Use a threshold that gives approximately n_active_target elements
        # Start with top n_active_target
        S = np.argsort(b)[::-1][:n_active_target]
        
        # Also include all elements above 99.9% of max
        threshold = 0.999 * b_max
        S_extra = np.where(b >= threshold)[0]
        S = np.unique(np.concatenate([S, S_extra]))
        S = S[:min(len(S), 2*n-1)]
        
        # Compute balanced subgradient
        direction = balanced_subgradient(a, S)
        
        # Check if direction is zero (KKT satisfied)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-12:
            break
        
        # Line search
        improved = False
        best_step = 0.0
        best_M_trial = M_cur
        
        # Try various step sizes
        max_step = 0.1 / (1.0 + 0.01 * it)
        for t in np.concatenate([
            np.linspace(0.01, max_step, 10),
            [0.5, 0.25, 0.1, 0.05]
        ]):
            a_trial = a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            a_trial_sum = a_trial.sum()
            if a_trial_sum < 0.01:
                continue
            a_trial = a_trial / a_trial_sum
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_step = t
        
        if best_step > 0 and best_M_trial < best_M - 1e-15:
            a = a + best_step * direction
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_M = best_M_trial
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_a, best_M

def run_fista_smooth(a_init, tau_start, tau_end, max_iters, deadline, lr=None):
    """FISTA on smooth max approximation."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr is None:
        lr = 0.5
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient
        grad = np.zeros(n)
        for i in range(n):
            j_start = max(0, i)
            j_end = min(n, 2 * n - 1 - i)
            # grad[i] = 2 * sum_k p[k] * a[k-i]
            # k-i is in [0,n-1] means k is in [i, i+n-1]
            lo = max(0, i)
            hi = min(2 * n - 1, i + n)
            if hi > lo:
                grad[i] = 2.0 * np.dot(p[lo:hi], a[lo-i:hi-i])
        
        grad_tan = grad - grad.mean()
        step = lr / (1.0 + 0.05 * tau * it)
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1
        
        if it % 10 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
    
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()
    
    return best_a

def coordinate_descent_1d(a, i, b, n):
    """Find optimal value for a[i] given all others."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.5)
    
    for _ in range(10):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 100.0:
            hi *= 2.0
        else:
            break
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12:
                delta = x_opt - a[i]
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - a[i] * a[i])
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * delta
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()

def l2_warmup(n, n_iters=1000):
    """Solve min a^T C a s.t. sum(a)=1, a>=0 where C[i,j] = n - |i-j|.
    
    Uses projected gradient with the known eigendecomposition of C.
    """
    # C is Toeplitz with C[i,j] = n - |i-j|
    # Its eigenvectors are the DFT basis (approximately, for large n)
    # Actually, for the specific matrix C[i,j] = n-|i-j|, the eigenvectors
    # are not exactly the DFT basis, but they're close.
    # Let's just use projected gradient with the matrix-vector product.
    
    def C_MATVEC(x):
        """Compute C @ x where C[i,j] = n - |i-j|."""
        # C @ x = sum_j (n - |i-j|) x_j
        # = n * sum_j x_j - sum_j |i-j| x_j
        # sum_j x_j is just x.sum()
        # sum_j |i-j| x_j can be computed in O(n) using prefix sums
        total = x.sum()
        # For each i: sum_j |i-j| x_j
        # = sum_{j<i} (i-j) x_j + sum_{j>i} (j-i) x_j
        # = i * sum_{j<i} x_j - sum_{j<i} j*x_j + sum_{j>i} j*x_j - i * sum_{j>i} x_j
        prefix_sum = np.cumsum(x)
        prefix_jx = np.cumsum(x * np.arange(n))
        
        result = np.zeros(n)
        for i in range(n):
            if i == 0:
                sum_jlt = 0.0
                sum_jx_lt = 0.0
            else:
                sum_jlt = prefix_sum[i-1]
                sum_jx_lt = prefix_jx[i-1]
            
            if i == n-1:
                sum_jgt = 0.0
                sum_jx_gt = 0.0
            else:
                sum_jgt = total - prefix_sum[i]
                sum_jx_gt = prefix_jx[-1] - prefix_jx[i]
            
            result[i] = n * total - (i * sum_jlt - sum_jx_lt + sum_jx_gt - i * sum_jgt)
        
        return result
    
    a = np.ones(n) / n
    best_a = a.copy()
    best_obj = a @ C_MATVEC(a)
    
    lr = 1.0  # Will use backtracking
    
    for it in range(n_iters):
        grad = C_MATVEC(a)
        grad_tan = grad - grad.mean()
        
        # Backtracking line search
        t = 1.0
        obj_cur = a @ C_MATVEC(a)
        for _ in range(30):
            a_trial = a - t * grad_tan
            a_trial = proj_simplex(a_trial)
            obj_trial = a_trial @ C_MATVEC(a_trial)
            if obj_trial <= obj_cur - 1e-4 * t * np.dot(grad_tan, grad_tan):
                break
            t *= 0.5
        
        a = a_trial
        obj_cur = a @ C_MATVEC(a)
        if obj_cur < best_obj:
            best_obj = obj_cur
            best_a = a.copy()
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p_n / p_n.sum())
    
    inits.append(np.ones(n) / n)
    
    # L2 optimal (warmup)
    a_l2 = l2_warmup(n, 500)
    inits.append(a_l2)
    
    # Various smooth inits
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        poly = t**alpha + (1-t)**alpha + 0.01
        inits.append(poly / poly.sum())
    
    for freq in [1, 2, 3, 4]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    return inits

def kkt_check(a, S, tol=1e-8):
    """Check if KKT conditions are approximately satisfied.
    
    Returns True if the balanced subgradient is approximately zero.
    """
    n = len(a)
    m = len(S)
    if m == 0:
        return True
    if m == 1:
        # For single active constraint, KKT requires grad to be constant
        k = S[0]
        g = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]
        return np.std(g) < tol
    return False

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)
