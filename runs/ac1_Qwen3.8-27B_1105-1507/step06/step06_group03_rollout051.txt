def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def smooth_subgradient_descent(a, max_iters, deadline, tau=50.0, lr=0.01):
    """
    Subgradient method with log-sum-exp smoothing.
    Works on the probability simplex.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_shift = c - c.max()
        exp_c = np.exp(tau * c_shift)
        w = exp_c / exp_c.sum()

        # Gradient of smooth approximation: grad_i = sum_k w[k] * d(c_k)/d(a_i)
        # d(c_k)/d(a_i) = a[k-i] + a[i-k] (with zero padding)
        # Efficient computation via convolution:
        # grad = convolve(w, a_rev) extracted appropriately
        a_rev = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1
        A_fft = np.fft.rfft(a_rev, size)
        W_fft = np.fft.rfft(w, size)
        Wrev_fft = np.fft.rfft(w[::-1], size)

        conv1 = np.fft.irfft(W_fft * A_fft, size)
        conv2 = np.fft.irfft(Wrev_fft * A_fft, size)

        grad = np.zeros(n)
        for i in range(n):
            idx1 = (n - 1) + i
            idx2 = (3 * n - 3) - i
            if 0 <= idx1 < len(conv1):
                grad[i] += conv1[idx1]
            if 0 <= idx2 < len(conv2):
                grad[i] += conv2[idx2]

        # Project gradient onto simplex tangent
        grad = grad - np.mean(grad)

        # Step size decay
        step = lr / (1.0 + 0.05 * it)

        a_new = a - step * grad
        a_new = project_simplex(a_new)

        obj_new = compute_J(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve > 200:
                break

    return best_a, best_obj


def generate_inits(n, rng, parent=None):

    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

        ps = p.sum()
        if ps > 1e-10:

        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 5, 8]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Gaussian
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25, 0.4]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.02))

    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.01))

    return inits

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Intensive optimization at parent length ===
    phase1_deadline = t_start + 500.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Optimization at n={n_parent} (budget {phase1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Start with parent
    a_start = parent.copy()
    a_start = np.maximum(a_start, 0)
    a_start = a_start / a_start.sum()

    a_opt, obj_opt = solve_single(a_start, phase1_deadline, verbose=True)
    update_best(a_opt)
    print(f"  Parent optimized: J={obj_opt:.10f}", flush=True)

    # Try a perturbed version
    remaining_p1 = phase1_deadline - time.monotonic()
    if remaining_p1 > 60:
        a_pert = best_seq + rng.normal(0, 0.01, best_seq.size)
        a_pert = project_simplex(a_pert)
        a_opt2, obj_opt2 = solve_single(a_pert, phase1_deadline - 5, verbose=False)
        update_best(a_opt2)
        print(f"  Perturbed: J={obj_opt2:.10f}", flush=True)

    # === PHASE 2: Multi-length exploration ===
    phase2_deadline = t_start + 1000.0
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = []
    for n_try in [384, 512, 640, 768, 896, 1024, 1280]:
        if n_try != n_parent:
            candidate_ns.append(n_try)
    candidate_ns.sort(key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:4]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 80:
            break
        alloc = min(120.0, remaining_check - 30)
        if alloc < 50:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_single(a_ext, time.monotonic() + alloc, verbose=False)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.10f}", flush=True)

    # === PHASE 3: Final intensive optimization ===
    phase3_deadline = deadline
    remaining = phase3_deadline - time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Final intensive optimization (budget {remaining:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and remaining > 30:
        # Long CD pass
        t_pass1 = time.monotonic() + 0.30 * remaining
        print(f"  Long CD pass ({t_pass1 - time.monotonic():.0f}s)...", flush=True)
        a_cd = precise_coord_descent(best_seq, max_sweeps=100000, deadline=t_pass1, seed=99)
        update_best(a_cd)

        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # LP pass
            t_pass2 = time.monotonic() + 0.35 * remaining
            print(f"  LP pass ({t_pass2 - time.monotonic():.0f}s)...", flush=True)
            a_lp, obj_lp = reduced_lp_step(best_seq, max_iters=500, deadline=t_pass2, top_k_frac=0.5)
            update_best(a_lp)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 20:
                # Another CD
                t_pass3 = time.monotonic() + 0.30 * remaining
                print(f"  CD polish ({t_pass3 - time.monotonic():.0f}s)...", flush=True)
                a_cd2 = precise_coord_descent(best_seq, max_sweeps=50000, deadline=t_pass3, seed=200)
                update_best(a_cd2)

                remaining = phase3_deadline - time.monotonic()
                if remaining > 15:
                    # Final LP
                    t_pass4 = deadline - 5
                    print(f"  Final LP ({t_pass4 - time.monotonic():.0f}s)...", flush=True)
                    a_lp2, obj_lp2 = reduced_lp_step(best_seq, max_iters=200, deadline=t_pass4, top_k_frac=0.6)
                    update_best(a_lp2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.10f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.10f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
