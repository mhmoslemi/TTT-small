"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

   - Line search over multiple step sizes

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-11)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=2000, tau_start=2.0, tau_end=150.0,
                       lr_init=0.3, lr_end=0.005):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)

        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for (a*g)(k) <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M, use_reduced=False, threshold=0.85):
    """Solve the best-response LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    
    if use_reduced:
        # Only keep constraints that are close to tight based on a*a
        b_aa = fast_conv(a)
        keep = b_aa > threshold * M
        if keep.sum() < n // 4:
            keep = b_aa > 0.5 * M
        if keep.sum() < n // 4:
            keep = np.ones(2 * n - 1, dtype=bool)
        A = A[keep, :]
        b_ub = np.full(A.shape[0], M)
    else:
        b_ub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            g = np.maximum(r.x, 0)
            # Verify constraints are satisfied
            if not use_reduced:
                check = A @ g
                if check.max() > M * (1 + 1e-6):
                    # Slight violation, scale down
                    g *= M / check.max()
            return g
    except Exception:
        pass
    return None


def anderson_mix(history_x, history_Fx, m):
    """Anderson mixing: find convex combination of recent iterates."""
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -0.5, 2.0)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha /= s
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                continue
            if n_try % 2 != 0:
                continue
            
            a_try = resample_seq(best_a_lp, n_try)
            a_try = np.maximum(a_try, 1e-8)
            a_try /= a_try.sum()
            
            # Quick smooth optimization
            a_try = run_smooth_descent(a_try, min(time.monotonic() + 8, p4_end - 5),
                                       max_iters=300, tau_start=5, tau_end=80,
                                       lr_init=0.15, lr_end=0.01)
            track(a_try)
            
            # Quick LP
            if time.monotonic() < p4_end - 5:
                a_lp = a_try.copy()
                a_lp = np.maximum(a_lp, 1e-10)
                a_lp /= a_lp.sum()
                M_best_try = fast_conv(a_lp).max()
                for lit in range(20):
                    if time.monotonic() > p4_end - 5:
                        break
                    b = fast_conv(a_lp)
                    M_c = float(b.max())
                    g = solve_lp(a_lp, M_c)
                    if g is None:
                        break
                    gs = g.sum()
                    if gs < 1e-10:
                        break
                    g /= gs
                    improved = False
                    for t in [0.01, 0.03, 0.05, 0.1, 0.2]:
                        a_t = np.maximum((1 - t) * a_lp + t * g, 0)
                        st = a_t.sum()
                        if st < 1e-10:
                            continue
                        a_t /= st
                        Mt = fast_conv(a_t).max()
                        if Mt < M_best_try - 1e-14:
                            a_lp = a_t
                            M_best_try = Mt
                            improved = True
                            break
                    if not improved:
                        break
                track(a_lp)
    
    # === PHASE 5: Final polish on absolute best (20s) ===
    if best_seq is not None and time.monotonic() < deadline:
        print(f"\nPhase 5: Final polish", flush=True)
        a_fin = best_seq.copy()
        a_fin = np.maximum(a_fin, 1e-10)
        a_fin /= a_fin.sum()
        M_fin = fast_conv(a_fin).max()
        n_fin = len(a_fin)
        
        for fit in range(100):
            if time.monotonic() > deadline:
                break
            b = fast_conv(a_fin)
            M_c = float(b.max())
            g = solve_lp(a_fin, M_c)
            if g is None:
                break
            gs = g.sum()
            if gs < 1e-10:
                break
            g /= gs
            improved = False
            for t in [0.005, 0.01, 0.02, 0.03, 0.05]:
                a_t = np.maximum((1 - t) * a_fin + t * g, 0)
                st = a_t.sum()
                if st < 1e-10:
                    continue
                a_t /= st
                Mt = fast_conv(a_t).max()
                if Mt < M_fin - 1e-15:
                    a_fin = a_t
                    M_fin = Mt
                    improved = True
                    break
            if not improved:
                break
        track(a_fin)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = compute_J(best_seq)
    
    total_time = time.monotonic() - (deadline - budget_s)
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={total_time:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    
    return [float(x) for x in best_seq]
