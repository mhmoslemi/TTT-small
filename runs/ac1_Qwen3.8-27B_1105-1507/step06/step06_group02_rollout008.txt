import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def sqp_optimize(a, deadline, max_iters=100, top_k=20):
    """Active set SQP for minimizing max_k (a*a)(k) subject to sum(a)=1, a>=0."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = float(b.max())
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        # Identify active set: top lags
        # Include all lags where b_k > b_max * (1 - 0.05) or top top_k
        active_set = np.where(b > b_max * 0.95)[0]
        if len(active_set) < top_k:
            top_indices = np.argsort(b)[-top_k:]
            active_set = np.unique(np.concatenate([active_set, top_indices]))
        active_set = sorted(active_set.tolist())
        
        # Build LP for SQP step
        # Variables: d[0..n-1], t
        # Minimize t
        # Subject to:
        #   For k in active_set: sum_i (2 * a[k-i] * d[i]) - t <= -b[k]
        #     where the sum is over valid i
        #   sum(d) = 0
        #   d[i] >= -a[i]
        
        # Actually, let me use the gradient form:
        # grad_k[i] = 2*a[k-i] if k-i is in [0,n-1] and i is in valid range, else 0
        # Constraint: grad_k @ d <= t - b[k]
        
        m = len(active_set)
        # A_ub is m + n rows by n+1 cols
        # First m rows: active set constraints
        # Next n rows: non-negativity (d[i] >= -a[i] => -d[i] <= a[i])
        
        A_ub = np.zeros((m + n, n + 1))
        b_ub = np.zeros(m + n)
        
        for idx, k in enumerate(active_set):
            # grad_k[i] = 2 * a[k-i] if 0 <= k-i < n and i in valid range
            # Valid range for i: max(0, k-n+1) <= i <= min(k, n-1)
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            for i in range(i_min, i_max + 1):
                A_ub[idx, i] = 2.0 * a[k - i]
            A_ub[idx, n] = -1.0  # coefficient of t
            b_ub[idx] = -b[k]
        
        # Non-negativity: d[i] >= -a[i] => -d[i] <= a[i]
        for i in range(n):
            A_ub[m + i, i] = -1.0
            b_ub[m + i] = a[i]
        
        # Equality: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Objective: minimize t
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                            bounds=[(None, None)] * n + [(None, None)],
                            method='highs')
        except Exception:
            break
        
        if not result.success or result.x is None:
            break
        
        d = result.x[:n]
        t_val = result.x[n]
        
        # The direction is d, and the "predicted" t is t_val
        # Take a damped step
        # The actual new M might be different from t_val due to linearization error
        # Use backtracking line search
        
        best_step = 0
        best_new_M = b_max
        step_sizes = [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]
        for alpha in step_sizes:
            a_new = a + alpha * d
            if np.any(a_new < -1e-10):
                continue
            a_new = np.maximum(a_new, 0)
            # Re-normalize
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new /= s
            new_M = fast_conv(a_new).max()
            if new_M < best_new_M:
                best_new_M = new_M
                best_step = alpha
                best_a_new = a_new.copy()
        
        if best_step == 0:
            break
        
        a = best_a_new
        new_b = fast_conv(a)
        if new_b.max() < best_M * (1 - 1e-12):
            best_M = new_b.max()
            best_a = a.copy()
        
        # Check convergence
        if best_new_M > b_max * (1 - 1e-10):
            # No improvement possible with current active set
            # Expand active set
            if len(active_set) < min(2 * n - 1, top_k + 20):
                active_set = np.where(fast_conv(a) > fast_conv(a).max() * 0.8)[0]
                active_set = np.unique(np.concatenate([active_set, np.argsort(fast_conv(a))[-top_k:]]))
            else:
                break
    return best_a
