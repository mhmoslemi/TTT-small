def fft_conv(a):

def solve_lp(a, M, row_idx, col_idx):

    except Exception:
        pass
    return None


def anderson_mix(hist_x, hist_Fx, m, max_alpha=2.0):
    """Anderson acceleration: find alpha minimizing ||sum alpha_i * r_i||^2 s.t. sum alpha = 1."""
    m = min(m, len(hist_x))
    if m < 2:
        return hist_Fx[-1].copy()
    xs = hist_x[-m:]
    Fxs = hist_Fx[-m:]
    residuals = np.array([xs[i] - Fxs[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-10 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -max_alpha, max_alpha)
    s = alpha.sum()
    if abs(s) < 1e-15:
        alpha = ones / m
    else:
        alpha /= s
    mixed = np.zeros_like(Fxs[0])
    for i in range(m):
        mixed += alpha[i] * Fxs[i]
    return np.maximum(mixed, 0)
