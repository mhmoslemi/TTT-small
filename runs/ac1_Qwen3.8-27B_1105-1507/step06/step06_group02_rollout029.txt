import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def active_set_lp(a, M, active_indices):
    """Solve max sum(g) s.t. (a*g)_k <= M for k in active_indices, g >= 0.
    Returns optimal g."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None and r.x.sum() > 1e-15:
        return np.maximum(r.x, 0)
    return None

1. Active-set LP mirror descent (fast per-iteration, allowing many more steps)
2. Symmetric sequence exploitation (halves problem size)
3. Spectral initialization with diverse frequency targets
4. Adaptive backtracking step size
5. Multi-length exploration with resampling
6. Final full-constraint LP polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_conv_asymmetric(a, g):
    """Compute a * g (not a * a) using FFT."""
    n = len(a)
    m = len(g)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:n + m - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def make_symmetric(half, n):
    """Create a full symmetric sequence from its first half."""
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, half[-2::-1]])

def get_half(a, n):
    """Extract the first half of a symmetric sequence."""
    if n % 2 == 0:
        return a[:n//2]
    else:
        return a[:(n+1)//2]

def spectral_init(n, rng, target='flat'):
    """Create initial sequence via spectral method."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    freqs = np.arange(size // 2 + 1)
    
    if target == 'flat':
        mag = np.ones(size // 2 + 1)
    elif target == 'raised_cosine':
        mag = np.cos(np.pi * freqs / (size // 2)) ** 2
        mag = np.maximum(mag, 0.1)
    elif target == 'tapered':
        falloff = np.exp(-2.0 * (freqs / (size // 2)) ** 2)
        mag = 0.5 + 0.5 * falloff
    else:  # random phases
        mag = np.ones(size // 2 + 1)
    
    # Random phases with conjugate symmetry
    phases = rng.uniform(-np.pi, np.pi, size // 2 + 1)
    phases[0] = 0
    if size % 2 == 0:
        phases[-1] = 0
    
    F = mag * np.exp(1j * phases)
    a = np.fft.irfft(F, size)[:n]
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a

def gs_init(n, n_iters=200, beta=0.9):
    """Gerchberg-Saxton initialization for flat autocorrelation."""
    a = np.abs(np.random.randn(n)) + 0.1
    a /= a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_M = fast_conv(a).max()
    best_a = a.copy()
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        # Target: constant magnitude (flat spectrum = flat autocorrelation)
        target_mag = np.ones_like(mag) * mag.mean()
        new_mag = (1 - beta) * mag + beta * target_mag
        F_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0.001)
        a_new /= a_new.sum()
        M = fast_conv(a_new).max()
        if M < best_M:
            best_M = M
            best_a = a_new.copy()
        a = a_new
    return best_a

def active_set_lp_full(a, M, active_indices):
    """Solve max sum(g) s.t. (a*g)_k <= M for k in active_indices, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def full_lp(a, M):
    """Solve max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def run_active_set_mirror(a, deadline, max_iters=5000, max_active=40, 
                          t_init=0.3, t_min=0.005, t_decay=0.999):
    """Active-set LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Initial active set
    b = fast_conv(a)
    top_k = min(max_active, 2 * n - 1)
    active_indices = np.argsort(b)[-top_k:][::-1].tolist()
    
    t = t_init
    no_imp = 0
    stalled = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
            stalled = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                stalled += 1
                if stalled > 3:
                    break
        
        # Solve LP with active set
        g = active_set_lp_full(a, M, active_indices)
        if g is None:
            # Try with more constraints
            if len(active_indices) < 2 * (2 * n - 1):
                b2 = fast_conv(a)
                active_indices = np.argsort(b2)[-min(max_active * 2, 2*n-1):][::-1].tolist()
                g = active_set_lp_full(a, M, active_indices)
            if g is None:
                break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Check and update active set
        ag = fast_conv_asymmetric(a, g)
        if len(ag) >= 2 * n - 1:
            viol_mask = ag[:2*n-1] > M * 1.01
            if np.any(viol_mask):
                viol_idx = np.where(viol_mask)[0]
                for vi in viol_idx[:5]:  # Add at most 5 new constraints
                    vi_int = int(vi)
                    if vi_int not in active_indices:
                        active_indices.append(vi_int)
                # Trim to max size
                if len(active_indices) > max_active + 20:
                    # Keep the ones with highest (a*g)_k values
                    ag_at_active = np.array([ag[:2*n-1][i] if i < len(ag) else 0 for i in active_indices])
                    sorted_order = np.argsort(ag_at_active)[::-1][:max_active]
                    active_indices = [active_indices[i] for i in sorted_order]
        
        # Step size with backtracking
        best_t = 0.0
        best_M_t = M
        for try_t in [t, t * 0.7, t * 0.5, t * 0.3, t * 0.2, 0.05, 0.02, 0.01, 0.005]:
            a_try = np.maximum((1 - try_t) * a + try_t * g, 0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try /= st
            M_try = fast_conv(a_try).max()
            if M_try < best_M_t:
                best_M_t = M_try
                best_t = try_t
        if best_t == 0:
            best_t = 0.005
        
        a = np.maximum((1 - best_t) * a + best_t * g, 0)
        sa = a.sum()
        if sa < 1e-10:
            a = best_a.copy()
        else:
            a /= sa
        
        # Decay step size slowly
        t = max(t_min, t * t_decay)
        
        # Periodically refresh active set
        if it % 20 == 0:
            b = fast_conv(a)
            top_k = min(max_active, 2 * n - 1)
            active_indices = np.argsort(b)[-top_k:][::-1].tolist()
    
    return best_a

def run_full_lp_mirror(a, deadline, max_iters=200, t_init=0.2):
    """Full LP mirror descent for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = t_init
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
        
        g = full_lp(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Backtracking line search
        best_t = 0.0
        best_M_t = M
        for try_t in [t, t*0.8, t*0.6, t*0.4, t*0.2, 0.1, 0.05, 0.02, 0.01]:
            a_try = np.maximum((1 - try_t) * a + try_t * g, 0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try /= st
            M_try = fast_conv(a_try).max()
            if M_try < best_M_t:
                best_M_t = M_try
                best_t = try_t
        if best_t == 0:
            best_t = 0.01
        
        a = np.maximum((1 - best_t) * a + best_t * g, 0)
        sa = a.sum()
        a /= sa
        t = max(0.005, t * 0.95)
    
    return best_a

def smooth_max_optimize(a, deadline, max_iters=2000, tau=50.0, lr_init=0.3, lr_end=0.001):
    """Smooth max gradient descent as a quick optimizer."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        lr = lr_init * (lr_end / lr_init) ** frac
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of smooth max of (a*a)_k
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-10:
                continue
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                pass  # This is O(n^2) which is too slow for large n
        
        # Use FFT-based gradient instead
        # grad_i = 2 * sum_k p_k * a_{k-i} (approximately)
        # This is a convolution of p with a
        p_full = np.zeros(2 * n - 1)
        p_full[:2*n-1] = p
        # (p * a)_i = sum_k p_k * a_{k-i}
        # This is correlation of p and a
        # grad = 2 * correlate(p, a)
        # Using FFT:
        size = 1
        while size < 3 * n:
            size <<= 1
        P = np.fft.rfft(p_full, size)
        A = np.fft.rfft(a, size)
        # We want sum_k p_k * a_{k-i} for i = 0,...,n-1
        # This is (p flipped) * a evaluated at appropriate positions
        p_flip = p_full[::-1]
        PF = np.fft.rfft(p_flip, size)
        conv_result = np.fft.irfft(PF * A, size)
        # conv_result[k] = sum_j p_flip[j] * a[k-j] = sum_j p_full[2n-2-j] * a[k-j]
        # We want sum_k p_k * a_{k-i}, which is different
        # Actually, let's just compute it directly for the active set
        
        grad = np.zeros(n)
        active = np.where(p > 1e-8)[0]
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            # d(a*a)_k / da_i = a_{k-i} (with the symmetry factor)
            # For simplicity, use: grad_i += p[k] * a[k-i] for valid k-i
            for i in range(j_min, j_max + 1):
                grad[i] += 2.0 * p[k] * a[k - i]
        
        # Projected gradient step
        a_new = np.maximum(a - lr * grad / n, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def local_perturbation(a, rng, n_trials=200, deadline=None):
    """Try small perturbations to improve the sequence."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for trial in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Random perturbation
        noise = rng.normal(0, 0.01 / np.sqrt(n), n)
        a_pert = np.maximum(a + noise, 0)
        sp = a_pert.sum()
        if sp < 1e-10:
            continue
        a_pert /= sp
        M_pert = fast_conv(a_pert).max()
        if M_pert < best_M * (1 - 1e-10):
            best_M = M_pert
            best_a = a_pert.copy()
            # Try to refine
            for _ in range(50):
                noise2 = rng.normal(0, 0.005 / np.sqrt(n), n)
                a_pert2 = np.maximum(best_a + noise2, 0)
                sp2 = a_pert2.sum()
                if sp2 < 1e-10:
                    break
                a_pert2 /= sp2
                M_pert2 = fast_conv(a_pert2).max()
                if M_pert2 < best_M * (1 - 1e-10):
                    best_M = M_pert2
                    best_a = a_pert2.copy()
                else:
                    break
    
    return best_a

class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set LP mirror descent (30-80 constraints vs full 2n-1, enabling 10x+ more iterations)
2. Adaptive backtracking step size for aggressive progress

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, g):
    """Compute a * g (cross-convolution) using FFT."""
    n = len(a)
    m = len(g)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_init(n, n_iters=150, beta=0.9, seed=None):
    """Gerchberg-Saxton initialization for flat autocorrelation."""
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n)) + 0.1
    a /= a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_M = fast_conv(a).max()
    best_a = a.copy()
    for _ in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        target_mag = np.ones_like(mag) * mag.mean()
        new_mag = (1 - beta) * mag + beta * target_mag
        F_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0.001)
        a_new /= a_new.sum()
        M = fast_conv(a_new).max()
        if M < best_M:
            best_M = M
            best_a = a_new.copy()
        a = a_new
    return best_a


def spectral_init(n, rng, mode='flat'):
    """Create initial sequence via spectral method with different targets."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    freqs = np.arange(size // 2 + 1)
    max_f = size // 2

    if mode == 'flat':
        mag = np.ones(max_f + 1)
    elif mode == 'raised':
        mag = np.cos(np.pi * freqs / (2 * max_f)) ** 2
        mag = np.maximum(mag, 0.05)
    elif mode == 'tapered':
        mag = np.exp(-1.5 * (freqs / max_f) ** 2) + 0.1
    elif mode == 'bump':
        center = max_f * 0.3
        mag = np.exp(-3.0 * ((freqs - center) / max_f) ** 2) + 0.2
    else:
        mag = np.ones(max_f + 1)

    phases = rng.uniform(-np.pi, np.pi, max_f + 1)
    phases[0] = 0.0
    if size % 2 == 0:
        phases[-1] = 0.0

    F = mag * np.exp(1j * phases)
    a = np.fft.irfft(F, size)[:n]
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def active_set_lp(a, M, active_indices):
    """Solve max sum(g) s.t. (a*g)_k <= M for k in active_indices, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def full_lp(a, M):
    """Solve max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_active_set_mirror(a, deadline, max_iters=20000, max_active=40,
                          t_init=0.25, t_min=0.002, t_decay=0.9997,
                          refresh_interval=15, check_violations=True):
    """
    Active-set LP mirror descent with adaptive step size.
    Key: uses only top max_active constraints in LP for speed,
    allowing many more iterations than full LP.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    conv_len = 2 * n - 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    b = fast_conv(a)
    top_k = min(max_active, conv_len)
    active_indices = np.argsort(b)[-top_k:][::-1].tolist()

    t = t_init
    no_imp = 0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_imp = 0
            stall_count = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                stall_count += 1
                if stall_count > 5:
                    break

        # Solve LP with active set
        g = active_set_lp(a, M, active_indices)
        if g is None:
            # Expand active set and retry
            b_tmp = fast_conv(a)
            expanded = np.argsort(b_tmp)[-min(max_active * 3, conv_len):][::-1].tolist()
            g = active_set_lp(a, M, expanded)
            if g is None:
                break
            active_indices = expanded

        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Check for violated constraints and update active set
        if check_violations and it % 5 == 0:
            ag = fast_cross_conv(a, g)
            ag = ag[:conv_len]
            viol_mask = (ag > M * 1.02)
            if np.any(viol_mask):
                viol_idx = np.where(viol_mask)[0]
                added = 0
                for vi in viol_idx:
                    vi_int = int(vi)
                    if vi_int not in active_indices and added < 10:
                        active_indices.append(vi_int)
                        added += 1
                # Trim if too large
                if len(active_indices) > max_active + 30:
                    ag_vals = np.array([ag[i] if i < len(ag) else 0.0 for i in active_indices])
                    keep = np.argsort(ag_vals)[::-1][:max_active]
                    active_indices = [active_indices[i] for i in keep]

        # Backtracking line search
        best_t = 0.0
        best_M_t = M
        t_candidates = [t, t * 0.8, t * 0.6, t * 0.4, t * 0.25, t * 0.15, 0.08, 0.04, 0.02, 0.01]
        for try_t in t_candidates:
            if try_t <= 0:
                continue
            a_try = np.maximum((1.0 - try_t) * a + try_t * g, 0.0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try /= st
            M_try = float(fast_conv(a_try).max())
            if M_try < best_M_t:
                best_M_t = M_try
                best_t = try_t

        if best_t == 0.0:
            best_t = 0.005

        a = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
        sa = a.sum()
        if sa < 1e-10:
            a = best_a.copy()
        else:
            a /= sa

        # Decay step size
        t = max(t_min, t * t_decay)

        # Periodically refresh active set based on current a*a
        if it % refresh_interval == 0:
            b = fast_conv(a)
            top_k = min(max_active, conv_len)
            active_indices = np.argsort(b)[-top_k:][::-1].tolist()

    return best_a

def smooth_max_optimize(a, deadline, max_iters=1000, tau_start=10.0, tau_end=100.0,
                        lr_init=0.3, lr_end=0.005):
    """Smooth max gradient descent with Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    conv_len = 2 * n - 1
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    beta_nv = 0.9
    prev_t = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        b = fast_conv(y)
        b_max = float(b.max())
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient: grad_i = 2 * sum_k p_k * a_{k-i}
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_max < j_min:
                continue
            # For each i in [j_min, j_max], grad[i] += 2*p[k]*a[k - i]
            idx = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += 2.0 * p[k] * a[idx]

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_nv ** 2)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta_nv * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

        M_cur = float(fast_conv(a).max())
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    best.report("Phase 1")

    # === PHASE 2: Active-Set LP Mirror Descent (Main) ===
    p2_dl = min(time.monotonic() + 700, deadline - 120)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Active-Set LP Mirror Descent (main)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        print(f"  Starting from n={n_best}, J={best.score:.8f}", flush=True)

        # First pass: broad active set, moderate step
        a1 = run_active_set_mirror(best.seq, min(time.monotonic() + 300, p2_dl),
                                   max_iters=50000, max_active=35,
                                   t_init=0.25, t_min=0.003, t_decay=0.9997,
                                   refresh_interval=15)
        best.update(a1)

        # Second pass: wider active set, smaller step
        if time.monotonic() < p2_dl - 60:
            a2 = run_active_set_mirror(a1, min(time.monotonic() + 300, p2_dl),
                                       max_iters=50000, max_active=60,
                                       t_init=0.15, t_min=0.002, t_decay=0.9998,
                                       refresh_interval=20)
            best.update(a2)

        # Third pass: narrow active set, focused
        if time.monotonic() < p2_dl - 30:
            a3 = run_active_set_mirror(a2, p2_dl,
                                       max_iters=50000, max_active=20,
                                       t_init=0.1, t_min=0.001, t_decay=0.9999,
                                       refresh_interval=10)
            best.update(a3)

    best.report("Phase 2")

    # === PHASE 3: Full LP Polish ===
    p3_dl = min(time.monotonic() + 250, deadline - 40)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Full LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        a_full = run_full_lp_mirror(best.seq, p3_dl, max_iters=150, t_init=0.12)
        best.update(a_full)

    best.report("Phase 3")

    # === PHASE 4: Multi-length exploration ===
    p4_dl = min(time.monotonic() + 120, deadline - 30)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_curr = best.seq.size
        for n_try in lengths:
            if time.monotonic() > p4_dl:
                break
            if n_try == n_curr:
                continue
            a_cand = resample_seq(best.seq, n_try)
            a_cand = np.maximum(a_cand, 1e-6)
            a_cand /= a_cand.sum()
            # Quick active-set optimization
            sub_dl = min(time.monotonic() + 50, p4_dl - 10)
            a_opt = run_active_set_mirror(a_cand, sub_dl,
                                          max_iters=10000, max_active=35,
                                          t_init=0.2, t_min=0.005, t_decay=0.9995,
                                          refresh_interval=12)
            best.update(a_opt)
            if best.seq is not None and best.seq.size == n_try:
                print(f"  n={n_try} improved to J={best.score:.8f}", flush=True)

    best.report("Phase 4")

    # === PHASE 5: Local Perturbation + Final Polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Local Perturbation + Final", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        # Local perturbation
        a_pert = local_perturbation(best.seq, rng, n_trials=500, deadline=deadline - 30)
        best.update(a_pert)

        # Final active-set pass
        if time.monotonic() < deadline - 20:
            a_fin = run_active_set_mirror(a_pert, min(time.monotonic() + 60, deadline - 10),
                                          max_iters=30000, max_active=30,
                                          t_init=0.12, t_min=0.001, t_decay=0.9998,
                                          refresh_interval=10)
            best.update(a_fin)

        # Final full LP if time allows
        if time.monotonic() < deadline - 5:
            a_final = run_full_lp_mirror(a_fin, deadline, max_iters=50, t_init=0.08)
            best.update(a_final)

    best.report("Phase 5")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
