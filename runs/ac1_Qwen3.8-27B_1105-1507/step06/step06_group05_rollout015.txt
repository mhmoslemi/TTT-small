"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) power-spectrum-based peak reduction exploiting the duality between
autocorrelation and power spectrum, (2) phase-retrieval-style projection for
non-negativity, (3) adaptive subgradient descent with multi-peak soft-max,
(4) LP polishing for small sequences, (5) multi-n exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def subgradient_multi_peak(a, tau, step_size, max_steps, time_budget):
    """
    Subgradient descent with soft-max multi-peak targeting.
    Uses adaptive step size via backtracking.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fft(a)
        cmax = float(np.max(conv))
        
        # Soft-max weights emphasizing peaks
        tau_eff = max(tau, cmax * 0.01)
        shifted = (conv - cmax) / tau_eff
        weights = np.exp(np.clip(shifted, -500, 0))
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient: g_i = sum_k w_k * a_{k-i} = cross-correlation of a with weighted peaks
        # This is the correlation of a with the weighted peak pattern
        a_flip = a[::-1]
        Wf = rfft(weights, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Center gradient to preserve sum
        grad = grad - np.mean(grad)
        
        # Backtracking line search
        obj_current = best_obj
        step_eff = step_size
        improved = False
        for _ in range(10):
            a_try = normalize(a - step_eff * grad)
            obj_try = compute_obj(a_try)
            if obj_try < obj_current - 1e-12:
                a = a_try
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
                break
            step_eff *= 0.5
            if step_eff < 1e-10:
                break
        
        if not improved:
            # Try smaller step or perturb
            if step_eff < 1e-10:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))
                step_size = max(step_size * 0.8, 0.001)
            else:
                step_size = max(step_size * 0.7, 0.001)
        
        # Cool temperature

def spectral_phase_retrieval(a, max_iters, time_budget):
    """
    Alternates between:
    1. Power spectrum shaping (reduce autocorrelation peak by modifying |A|)
    2. Phase retrieval (find phase making time-domain non-negative)
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 4 * n:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # --- Phase Retrieval Step ---
        # Given current |A|, find phase that makes IFFT non-negative
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # GS iteration for phase
        for _ in range(3):
            a_temp = np.real(ifft(mag * np.exp(1j * phase), N))[:n]
            a_proj = np.maximum(a_temp, 0.0)
            s = float(np.sum(a_proj))
            if s < 1e-10:
                break
            a_proj = a_proj / s
            A_new = fft(a_proj, N)
            phase = np.angle(A_new)
        
        # --- Power Shaping Step ---
        # Modify |A| to reduce autocorrelation peak
        A = mag * np.exp(1j * phase)
        a_curr = np.real(ifft(A, N))[:n]
        a_curr = np.maximum(a_curr, 0.0)
        s = float(np.sum(a_curr))
        if s < 1e-10:
            break
        a_curr = a_curr / s

        conv = compute_conv_fft(a_curr)
        cmax = float(np.max(conv))
        
        # Gradient of cmax w.r.t. |A|^2 in frequency domain
        # r(k) = IFFT(|A|^2)[k], so dr(k)/d|A(f)|^2 = (1/N) * exp(2*pi*i*k*f/N)
        # To reduce r(k*), decrease |A(f)|^2 where Re(exp(2*pi*i*k*f/N)) > 0
        k_star = int(np.argmax(conv))
        f_idx = np.arange(N // 2 + 1)  # for rfft
        # Actually let's work with full fft
        f_full = np.arange(N)
        grad_mag = np.real(np.exp(2j * np.pi * k_star * f_full / N))
        
        # Apply gradient step to magnitude
        alpha = 0.1 / (1 + it * 0.1)
        mag_new = mag - alpha * grad_mag * mag
        mag_new = np.maximum(mag_new, 0.0)
        
        # Ensure Hermitian symmetry for real signal
        mag_new[N - np.arange(N // 2 + 1)] = mag_new[np.arange(N // 2 + 1)]
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
    
    return best_a, best_obj


def lp_polish(a, max_iters, time_budget):
    """LP-based polishing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv_fft(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.7)))
        if g is None:
            break

        # Blend
        t = 0.3 + 0.4 * (it / max_iters)
        a_new = normalize((1 - t) * a + t * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller blend
            for t_try in [0.1, 0.05, 0.2]:
                a_try = normalize((1 - t_try) * a + t_try * g)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break

    return best_a, best_obj


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

    return None, np.inf
