            print(f"Error in scoring mutated sequence: {e}")
            continue
        
        # Update best if improvement detected
        if current_best_score < best_score:
            best_score = current_best_score
            best_seq = current_best_seq.copy()
            improvement_counter = 0  # Reset improvement counter
        else:
            improvement_counter += 1
        
        # Check for stability (no improvement over many generations)
        if improvement_counter > 100:
            print("No recent improvements, terminating optimization.")
            break
        
        # Record the current epoch's candidate
        population.append(current_best_seq.copy())
        
        # Report progress periodically
        if epoch % 30 == 0:
            current_time = time.time()
            if current_time - t0 >= 60:  # Every minute, report the current best
                remaining = time_limit - current_time
                print(f"\nEpoch {epoch}: Best score so far {best_score:.8f}, time used {current_time-t0:.1f}s, time remaining {remaining:.1f}s")
            
    # If we haven't found a better sequence, keep the current best
    if best_score < current_best_score:
        current_best_seq = best_seq.copy()
    else:
        current_best_seq = best_candidates.pop()
        
    # Final validation with the best_sequence
    final_score = evaluate_sequence(current_best_seq)
    if final_score < np.inf:
        best_score = final_score
        best_seq = current_best_seq.copy()
    else:
        print("Validation failed for final sequence, use floating minimal sequence instead.")
        fallback_seq = np.ones(int(generation_length/2)) / len(fallback_seq)
        best_score = evaluate_sequence(fallback_seq)
        best_seq = fallback_seq.copy()
