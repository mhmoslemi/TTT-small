    a = np.maximum(a, 1e-8).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    gamma = 1.0  # step size parameter
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        b_max = float(b.max())
        if b_max < best_M:
            best_M = b_max
            best_a = a.copy()
        
        # Subgradient from the max peak
        k_star = int(np.argmax(b))
        
        # Compute subgradient g[m] = a[k_star - m] + a[m - k_star]
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)

        # k_star - m ranges from k_star - hi to k_star - lo

        # a[m - k_star] for m in [lo, hi]:

import cvxpy as cp

def solve_direct(n):
    a = cp.Variable(n)
    # Autocorrelation constraints
    b = [sum(a[j] * a[k-j] for j in range(max(0, k-n+1), min(n-1, k)+1)) for k in range(2*n-1)]

    obj = cp.Minimize(t)

    prob.solve(solver=cp.ECOS, verbose=True)
    return a.value

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gs_flat_spectral(init, n_iters=200, beta=0.85):
    a = np.maximum(init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    best_a = a.copy()
    best_M = fast_conv(a).max()
    for it in range(n_iters):
        F = np.fft.rfft(a, size)
        mag = np.abs(F)
        phase = np.angle(F)
        mag_norm = mag / (mag.max() + 1e-30)
        mag_flat = mag.mean() + 1e-30
        mag_new = (1.0 - beta) * mag + beta * mag_flat
        power = 0.5 + 0.5 * beta
        mag_pow = (mag ** power) / ((mag ** power).max() + 1e-30) * mag.max()
        mag_final = 0.5 * mag_new + 0.5 * mag_pow
        F_new = np.exp(1j * phase) * mag_final
        a_new = np.fft.irfft(F_new, size)[:n]
        a_candidate = np.maximum(a_new, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            break
        a_candidate /= sc
        M_c = fast_conv(a_candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = a_candidate.copy()
        a = a_candidate
    return best_a


def smooth_max_grad_fft(a, tau):
    """Compute smooth max gradient using FFT-based approach."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    # Compute gradient: g[m] = sum_k p[k] * (a[k-m] + a[m-k])
    # Using FFT: g = cross_corr(p, a) + conv(a, p) (truncated to n elements)
    size = 1
    while size < 3 * n:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    # cross_corr[p, a][m] = sum_n p[n+m] * a[n] = ifft(conj(P) * A)[m + offset]
    cross = np.fft.irfft(np.conj(P) * A, size)
    # conv[a, p][m] = sum_n a[n] * p[m-n] = ifft(A * P)[m]
    conv = np.fft.irfft(A * P, size)
    
    # Determine correct offsets by checking with direct computation for small case
    # cross[m] should equal sum_n p[n+m]*a[n]
    # conv[m] should equal sum_n a[n]*p[m-n]
    
    # The cross-correlation via FFT: ifft(conj(P)*A) gives x where
    # x[k] = sum_n conj(p[n]) * a[n+j] for some offset j
    # For real signals: x[k] = sum_n p[n] * a[n+k] (with wraparound)
    # We want g1[m] = sum_n p[n+m] * a[n] (no wraparound)
    # This equals ifft(conj(P)*A)[m + len(p) - 1]... let me just use direct for the gradient
    
    # Actually, let me compute it more carefully.
    # ifft(conj(P)*A)[k] = (1/N) * sum_j conj(P[j])*A[j]*exp(2*pi*i*j*k/N)
    # = sum_n sum_m conj(p[n])*a[m]*exp(2*pi*i*(m-n)*k/N) ... this is the circular cross-correlation
    
    # For the linear cross-correlation (no wrap), we need zero-padding.
    # With size >= len(p) + len(a) - 1 = (2n-1) + n - 1 = 3n-2, we avoid wraparound.
    
    # ifft(conj(P)*A) gives the cross-correlation c[m] = sum_n p[n] * a[n+m]
    # (circular, but with our padding it's the same as linear for m in valid range)
    
    # Let me verify: c[0] = sum_n p[n]*a[n] (first n terms valid)
    # c[m] = sum_n p[n]*a[n+m]
    # We want g1[m] = sum_n p[n+m]*a[n] = sum_i p[i]*a[i-m] (substituting i=n+m)
    # This is c[-m] in circular sense = c[size-m]
    
    # So g1[m] = cross[size-m] for m=0,...,n-1 (with cross being the full ifft result)
    # Actually for m=0: g1[0] = sum_n p[n]*a[n] = cross[0]. Good.
    # For m=1: g1[1] = sum_n p[n+1]*a[n] = sum_i p[i]*a[i-1] 
    # cross[size-1] = sum_n p[n]*a[n+size-1]... with zero padding, a[n+size-1] = 0 for n < size-n
    # Hmm this is getting confusing. Let me just use the direct O(n^2) computation.
    # For n up to 2048, O(n^2) = 4M ops per gradient, which is fine for a few thousand iterations.
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # a[k-m] for m in [lo, hi]
        seg1 = a[k - hi:k - lo + 1][::-1]
        g[lo:hi + 1] += p[k] * seg1
        # a[m-k] for m in [lo, hi] (valid when 0 <= m-k < n)
        lo2 = max(lo, k)
        hi2 = min(hi, k + n - 1)
        if lo2 <= hi2:
            seg2 = a[lo2 - k:hi2 - k + 1]
            g[lo2:hi2 + 1] += p[k] * seg2
    return g, p, b, b_max


def smooth_max_grad_topk(a, tau, top_k=5):
    """Faster gradient using only top-k peaks."""
    n = len(a)

    b_max = float(b.max())

        top_k = 2 * n - 1
    idx = np.argsort(b)[::-1][:top_k]
    exp_b = np.exp(tau * (b[idx] - b_max))
    p_top = exp_b / exp_b.sum()
    
    g = np.zeros(n)

def frank_wolfe_subgrad(a, deadline, max_iters=200000, mode='ash'):
    """Frank-Wolfe with hard subgradient (just the max peak). Very fast per iteration."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    last_imp = time.monotonic()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = float(b.max())
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
            last_imp = time.monotonic()
        
        # Check for stagnation
        if time.monotonic() - last_imp > 30.0:
            break
        
        k_star = int(np.argmax(b))
        # Subgradient: g[m] = a[k_star - m] + a[m - k_star]
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        # a[k_star - m] for m in [lo, hi]
        g[lo:hi + 1] += a[k_star - hi:k_star - lo + 1][::-1]
        # a[m - k_star] for m in [lo, hi] (valid when k_star <= m <= k_star+n-1)
        lo2 = max(lo, k_star)
        hi2 = min(hi, k_star + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] += a[lo2 - k_star:hi2 - k_star + 1]
        
        s_idx = int(np.argmin(g))
        
        if mode == 'ash':
            # Arve-Schmalheiser: t = 2/(it+2)
            t = 2.0 / (it + 2.0)
        elif mode == 'backtrack':
            # Backtracking line search
            t = 1.0
            f_curr = b_max
            while t > 1e-6:
                e = np.zeros(n)
                e[s_idx] = 1.0
                a_trial = (1.0 - t) * a + t * e
                sa = a_trial.sum()
                if sa < 1e-10:
                    t *= 0.5
                    continue
                a_trial /= sa
                b_trial = fast_conv(a_trial)
                if b_trial.max() < f_curr - 1e-4 * t * (g[s_idx] - g.max()) + 1e-12:
                    a = a_trial
                    break
                t *= 0.5
            else:
                e = np.zeros(n)
                e[s_idx] = 1.0
                a = (1.0 - t) * a + t * e
                sa = a.sum()
                if sa > 1e-10:
                    a /= sa
                continue
        else:
            t = 1.0 / (it ** 0.5)
        
        e = np.zeros(n)
        e[s_idx] = 1.0
        a = (1.0 - t) * a + t * e
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s
    
    return best_a, best_M

def anderson_mixed(history_x, history_Fx, m, clip_range=0.5):
    """Anderson mixing."""
    if m < 2:
        return history_Fx[-1].copy()
    residuals = np.array([history_x[i] - history_Fx[i] for i in range(m)])
    G = residuals @ residuals.T
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    alpha = np.clip(alpha, -clip_range, 1.0 + clip_range)
    alpha_sum = alpha.sum()
    if abs(alpha_sum) < 1e-15:
        alpha = ones / m
    else:
        alpha = alpha / alpha_sum
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

            a_new /= s
        
        hist_x.append(a.copy())
        hist_Fx.append(a_new.copy())
        if len(hist_x) > anderson_window:
            hist_x.pop(0)
            hist_Fx.pop(0)
        
        if len(hist_x) >= 3:
            a_and = anderson_mixed(hist_x, hist_Fx, len(hist_x))
            sa = a_and.sum()
            if sa > 1e-10:
                a_and /= sa
                M_and = fast_conv(a_and).max()
                M_new = fast_conv(a_new).max()
                if M_and <= M_new * 1.02:
                    a = a_and
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new
    
    return best_a


def run_lp_with_backtrack(a, deadline, max_iters=200, t_values=None):
    """LP mirror descent with backtracking line search."""
    if t_values is None:
        t_values = [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break
        
        g = lp_best_response_active(a, M, active_thresh=0.5)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        improved = False
        for t in t_values:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if not improved:
            no_imp += 1
            if no_imp > 25:
                break
    
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def make_inits_diverse(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1.0 + 0.15 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n))
    for alpha in [0.7, 1.0, 1.5, 2.0]:
        inits.append(np.maximum(0.3 + 0.7 * (t**alpha + (1 - t)**alpha) / 2.0, 0.01))
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-5 * (t - 0.5)**2) + 0.08)
    k = max(3, n // 15)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0.2, 1.0, n), kern, 'same') + 0.03, 0.01))
    if parent is not None:
        inits.append(gs_flat_spectral(resample_seq(parent, n), n_iters=100, beta=0.9))
    inits.append(gs_flat_spectral(np.maximum(rng.normal(0.5, 0.2, n), 0.01), n_iters=80, beta=0.85))
    return inits


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25.0
    best = BestTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent loaded: n={parent.size}, J={best.score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass
    
    if parent is None:
        parent = np.ones(1024)
        best.update(parent)
    
    n_parent = best.seq.size
    
    # === PHASE 1: Quick smooth optimization on parent + nearby lengths ===
    p1_start = time.monotonic()
    p1_budget = 250.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Optimization ({p1_budget:.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)
    
    lengths_to_try = sorted(set([n_parent, n_parent + 128, n_parent - 128, 
                                  n_parent + 256, n_parent - 256]))
    lengths_to_try = [n for n in lengths_to_try if 256 <= n <= 2560]
    
    best_per_length = {}
    n_lengths = len(lengths_to_try)
    time_per_length = p1_budget / n_lengths
    
    for n in lengths_to_try:
        if time.monotonic() > p1_start + p1_budget:
            break
        sub_dl = min(time.monotonic() + time_per_length, deadline - 400)
        if time.monotonic() >= sub_dl:
            continue
        
        inits = make_inits_diverse(n, rng, best.seq)
        best_n = None
        best_M = float('inf')
        
        n_iters_base = max(300, int(6000 * (512.0 / n)))
        for idx, a0 in enumerate(inits):
            if time.monotonic() > sub_dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            frac_time = 1.0 / (1.0 + idx * 0.5)
            sub_sub_dl = time.monotonic() + (sub_dl - time.monotonic()) * frac_time
            a = run_nesterov_smooth(a0, sub_sub_dl, max_iters=n_iters_base,
                                    tau_start=1.0, tau_end=150.0,
                                    lr_init=0.3, lr_end=0.003, top_k=3)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()
        
        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: M={best_M:.8f} J={J:.6f}", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()
    
    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)
    
    # === PHASE 2: Frank-Wolfe fast refinement ===
    p2_start = time.monotonic()
    p2_budget = 200.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Frank-Wolfe Refinement ({p2_budget:.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)
    
    if best.seq is not None:
        fw_dl = min(time.monotonic() + p2_budget, deadline - 300)
        a_fw = frank_wolfe_subgrad(best.seq, fw_dl, max_iters=500000, mode='ash')
        best.update(a_fw)
    
    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)
    
    # === PHASE 3: Active-set LP mirror descent (main workhorse) ===
    p3_start = time.monotonic()
    p3_budget = 700.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Active-Set LP Mirror Descent ({p3_budget:.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)
    
    if best.seq is not None:
        n = best.seq.size
        lp_dl = min(time.monotonic() + p3_budget, deadline - 100)
        max_iters = 300

                                   max_iters=max_iters, anderson_window=5,
                                   t_base=0.4, min_t=0.01, active_thresh=0.5)

        if time.monotonic() < deadline - 100:

                                        max_iters=int(max_iters * 0.8), anderson_window=6,
                                        t_base=0.2, min_t=0.003, active_thresh=0.3)

    # === PHASE 5: Multi-length LP exploration ===
    p5_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Multi-length Exploration", flush=True)
    print(f"{'='*50}", flush=True)
    
    if best_per_length and time.monotonic() < deadline - 40:
        for n_try in sorted(best_per_length.keys()):
            if time.monotonic() > deadline - 30:
                break
            if n_try == best.seq.size:
                continue
            a_cand = best_per_length[n_try]
            # Quick smooth
            a_cand = run_nesterov_smooth(a_cand, min(time.monotonic() + 20, deadline - 40),
                                         max_iters=200, tau_start=5, tau_end=80,
                                         lr_init=0.15, lr_end=0.01, top_k=3)
            best.update(a_cand)
            # LP polish
            if time.monotonic() < deadline - 40:
                a_lp = run_lp_anderson_active(a_cand, min(time.monotonic() + 40, deadline - 25),
                                              max_iters=30, anderson_window=4,
                                              t_base=0.3, min_t=0.01, active_thresh=0.5)

        for noise_scale in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 40:

            a_pert = np.maximum(best.seq + rng.normal(0, noise_scale, n), 1e-6)
            a_pert /= a_pert.sum()
            # Quick smooth
            a_pert = run_nesterov_smooth(a_pert, min(time.monotonic() + 15, deadline - 35),
                                         max_iters=150, tau_start=5, tau_end=50,
                                         lr_init=0.1, lr_end=0.01, top_k=3)

                                              max_iters=20, anderson_window=4,
                                              t_base=0.3, min_t=0.01, active_thresh=0.5)

    if best.seq is not None and time.monotonic() < deadline - 5:

        print(f"PHASE 7: Final Polish", flush=True)

    # Fallback
    if best.seq is None:

        best.score = compute_J(best.seq)
    
    total_time = time.monotonic() - best.t0
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={total_time:.0f}s", flush=True)

    return [float(x) for x in best.seq]
