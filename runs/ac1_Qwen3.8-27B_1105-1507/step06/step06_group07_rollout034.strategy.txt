def plateau_residual(n, frac=0.5):
    # Calculate absolute plateau length
    n_plateau = int(round(frac * n))

    plateau = n_plateau

    peak_start = (n - plateau) // 2
    peak_end = peak_start + plateau - 1
    # Value for the plateau
    plateau_value = 1.0 / plateau
    # Separate the ramp from the plateau
    left_ramp_indices = list(range(0, peak_start)) if peak_start > 0 else []

    # Build candidate
