"""
FISTA-based optimization with log-sum-exp smoothing for minimum peak
autocorrelation design. Combines accelerated gradient methods, adaptive
temperature annealing, multi-length search, and LP polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # We want to find the closest point to v in {x >= 0, sum(x) = target_sum}
    # Solution: x_i = max(v_i - threshold, 0) where threshold is chosen so sum = target_sum
    # Use bisection on threshold
    lo = float(np.min(v)) - target_sum
    hi = float(np.max(v))
    
    if target_sum <= 0:
        return np.zeros(n)
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.maximum(v - mid, 0.0)
        s = float(np.sum(x))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    
    x = np.maximum(v - (lo + hi) / 2.0, 0.0)
    s = float(np.sum(x))
    if s < 1e-15:
        return np.full(n, target_sum / n)
    return x / s * target_sum

def compute_gradient_smoothed(a, tau, fft_size=None):
    """
    Compute gradient of 2n * tau * log(sum_k exp(r[k]/tau)) where r = conv(a,a).
    Returns the gradient vector of length n.
    """
    n = len(a)
    if fft_size is None:
        fft_size = 1
        while fft_size < 2 * n - 1:
            fft_size <<= 1
    
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip(conv / tau - cmax / tau, -500.0, 500.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # grad[i] = 2n * sum_k weights[k] * 2 * a[k-i] (for 0 <= k-i < n)
    # = 4n * sum_k weights[k] * a[k-i]
    # This is 4n * (cross-correlation of weights and a) at each lag i
    # Using FFT: corr[i] = sum_j a[j] * weights[i+j] (for valid i+j)
    
    w_ext = np.zeros(fft_size)

"""
Improved minimum-peak autocorrelation design combining LP-alternating with
adaptive line search, FISTA with log-sum-exp smoothing and temperature
annealing, multi-length exploration, and active-set polishing.
Targets improvement below 1.5027 by: (1) searching over n more aggressively,
(2) using LP dual weights for focused subgradient directions, (3) adaptive
blending in the LP alternating loop, and (4) FISTA for escaping local optima.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum} via bisection."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if np.all(v <= 0):
        return np.full(n, target_sum / n)
    lo = float(np.min(v)) - target_sum
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    thr = 0.5 * (lo + hi)
    x = np.maximum(v - thr, 0.0)
    s = float(np.sum(x))
    if s < 1e-15:
        return np.full(n, target_sum / n)
    return x * (target_sum / s)


def normalize_simplex(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20.0):
    """
    Solve: min t s.t. (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g_normalized, t_opt, dual_weights) or (None, None, None).
    """
    n = len(fixed_seq)
    A = build_toeplitz_mat(fixed_seq)
    m = 2 * n - 1
    # Variables: g[0..n-1], t => total n+1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                # Extract dual weights from the inequality constraints
                # res.ineqlin.marginals gives duals for A_ub @ x <= b_ub
                duals = None
                if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                    if hasattr(res.ineqlin, 'marginals') and res.ineqlin.marginals is not None:
                        duals = np.abs(res.ineqlin.marginals)
                return g, float(res.x[n]), duals
    except Exception:
        pass
    return None, None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=20.0):
    """
    Solve: max sum(g) s.t. (fixed_seq * g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None.
    """

    A = build_toeplitz_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:

        lp_time = min(25.0, max(5.0, t_rem * 0.3))

        alphas = sorted(set(alphas))

def fista_smoothed(a, tau_start=0.03, tau_end=0.002, max_iters=5000,
                   time_budget=60.0, mu=0.9):
    """
    FISTA (accelerated projected gradient) with log-sum-exp smoothing
    and decreasing temperature.
    """
    n = len(a)
    a = normalize_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # FFT size
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    y = a.copy()
    t_k = 1.0
    v = np.zeros(n)
    step = 0.1  # initial step size (will be adapted)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive temperature
        frac = min(1.0, it / (max_iters * 0.7))
        tau = tau_start * np.exp(-4.0 * frac) + (tau_end - tau_start * np.exp(-4.0)) * frac
        tau = max(tau, tau_end)

        # Compute smoothed gradient
        conv = compute_conv(y)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -400.0, 400.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # grad[i] = 4n * sum_k weights[k] * a[k-i] (zero-padded)
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        Y_ext = rfft(y, fft_size)
        corr = irfft(Y_ext * np.conj(W_ext), fft_size)
        grad = 4.0 * n * corr[:n]

        # Center gradient (for sum constraint)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Adaptive step: use gradient norm to scale
        # Lipschitz constant ~ 4n * ||weights||_2 (rough)
        L_est = 4.0 * n * float(np.linalg.norm(weights))
        step = 1.0 / max(L_est, 1e-10)
        # Cap step to avoid too-large moves
        step = min(step, 0.05 / max(grad_norm / n, 1e-10))

        # Nesterov momentum update
        v = mu * v - step * grad_centered
        a_new = project_simplex(y - v)

        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (t_k - 1.0) / (t_new + 1.0)
        y = a_new + theta * (a_new - a)
        y = np.maximum(y, 0.0)
        y = normalize_simplex(y)
        t_k = t_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def fista_with_lp_guide(a, max_iters=3000, time_budget=60.0,
                        lp_interval=100, lp_time=10.0):
    """
    FISTA that periodically uses LP dual weights to guide the gradient.
    When LP is available, use its dual weights instead of softmax.
    """
    n = len(a)
    a = normalize_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    y = a.copy()
    t_k = 1.0
    v = np.zeros(n)

    # Initialize weights with uniform
    weights = np.ones(2 * n - 1) / (2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Periodically update weights via LP
        if it % lp_interval == 0 and n <= 900:
            g, t_opt, duals = solve_minpeak_lp(a, time_limit=lp_time)
            if duals is not None and duals.size == 2 * n - 1:
                w_sum = float(np.sum(duals))
                if w_sum > 1e-10:
                    weights = duals / w_sum
                elif g is not None:
                    # Fallback: use the conv peaks
                    conv = compute_conv(a)
                    idx = int(np.argmax(conv))
                    weights = np.zeros(2 * n - 1)
                    weights[idx] = 1.0
            elif g is not None:
                # No duals but have solution; use sharp softmax on conv
                conv = compute_conv(a)
                cmax = float(np.max(conv))
                shifted = (conv - cmax) / 0.01
                shifted = np.clip(shifted, -400, 400)
                weights = np.exp(shifted)
                ws = float(np.sum(weights))
                if ws > 1e-300:
                    weights /= ws

        # Compute gradient using current weights
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        Y_ext = rfft(y, fft_size)
        corr = irfft(Y_ext * np.conj(W_ext), fft_size)
        grad = 4.0 * n * corr[:n]
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Step size
        L_est = 4.0 * n * float(np.linalg.norm(weights))
        step = min(1.0 / max(L_est, 1e-10), 0.03)

        v = 0.9 * v - step * grad_centered
        a_new = project_simplex(y - v)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (t_k - 1.0) / (t_new + 1.0)
        y = a_new + theta * (a_new - a)
        y = normalize_simplex(np.maximum(y, 0.0))
        t_k = t_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def generate_starts(n, seed=42, num_starts=5):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # 1. Uniform
    starts.append(np.ones(n) / n)

    # 2. Triangular
    tri = 1.0 - np.abs(i - center) / (center + 0.1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # 3. Gaussian (narrow)
    bell = np.exp(-((i - center) ** 2) / (2.0 * (n / 5.0) ** 2))
    starts.append(bell / np.sum(bell))

    # 4. Cosine window
    cos_w = 0.5 * (1.0 + np.cos(np.pi * i / (n - 1)))
    cos_w = np.maximum(cos_w, 0.01)
    starts.append(cos_w / np.sum(cos_w))

    # 5. Random smooth
    noise = rng.normal(0, 0.3, n)
    # Smooth via cumulative sum trick
    from scipy.ndimage import gaussian_filter1d
    smooth = gaussian_filter1d(noise, sigma=n / 10.0)
    a_rand = np.abs(smooth) + 0.1
    starts.append(a_rand / np.sum(a_rand))

    return starts[:num_starts]


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_candidate(a, time_budget=120.0):
    """
    Multi-method polishing: alternate between LP, FISTA, and FISTA+LP.
    """
    n = len(a)
    a = normalize_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Cycle through methods
    methods = []
    if n <= 900:
        methods.append(('lp', None))
    methods.append(('fista', None))
    if n <= 800:
        methods.append(('fista_lp', None))
    methods.append(('fista', None))
    if n <= 900:
        methods.append(('lp', None))

    method_idx = 0
    stall = 0

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        alloc = min(remaining * 0.3, 40.0)
        if alloc < 5:
            break

        method = methods[method_idx % len(methods)][0]
        method_idx += 1

        if method == 'lp':
            a_new, obj_new = lp_alternate_adaptive(best_a, max_iters=50, time_budget=alloc)
        elif method == 'fista':
            a_new, obj_new = fista_smoothed(best_a, max_iters=int(alloc * 30),
                                            time_budget=alloc * 0.9)
        else:  # fista_lp
            a_new, obj_new = fista_with_lp_guide(best_a, max_iters=int(alloc * 20),
                                                 time_budget=alloc * 0.9,
                                                 lp_interval=50,
                                                 lp_time=min(8.0, alloc * 0.2))

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                scale = 0.02 + 0.01 * (stall % 5)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize_simplex(a_pert)
                # Quick FISTA on perturbation
                a_pert, obj_pert = fista_smoothed(a_pert, max_iters=200, time_budget=5.0)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                stall = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multi-length exploration, LP-alternating
    with adaptive blending, FISTA with temperature annealing, and LP-guided
    FISTA to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting from parent: n={parent_obj and len(best_seq)}, obj={best_score:.8f}")

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}")

    # Phase 1: Multi-length exploration
    print("\n=== Phase 1: Multi-length exploration ===")
    n_values = [500, 600, 650, n_parent, 750, 800, 900, 1000, 1200]
    n_values = sorted(set([n for n in n_values if 300 <= n <= 1500]))

    candidates = []  # (obj, seq)

    for n in n_values:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Generate starts
        if n == n_parent and best_seq is not None:
            starts = [normalize_simplex(best_seq.copy().astype(np.float64))]
            starts.extend(generate_starts(n, seed=seed, num_starts=3))
        else:
            # If we have a good sequence, resample it as an additional start
            starts = []
            if best_seq is not None:
                starts.append(normalize_simplex(resample_sequence(best_seq, n)))
            starts.extend(generate_starts(n, seed=seed + n * 31, num_starts=3))

        starts = starts[:4]
        alloc_per_start = min((remaining - 20) / len(starts), 35.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 250:
                break
            t_alloc = min(alloc_per_start, 30.0)
            if t_alloc < 8:
                break

            # Method 1: LP alternating (if n allows)
            if n <= 900:
                a_opt, obj_opt = lp_alternate_adaptive(s, max_iters=30, time_budget=t_alloc * 0.5)
            else:
                a_opt, obj_opt = fista_smoothed(s, max_iters=int(t_alloc * 15), time_budget=t_alloc * 0.5)

            # Method 2: FISTA polish
            t_rem = t_alloc - (time.monotonic() - (start_time + (deadline - remaining)))
            fista_budget = min(t_alloc * 0.4, 15.0)
            a_opt2, obj_opt2 = fista_smoothed(a_opt, max_iters=int(fista_budget * 20),
                                              time_budget=fista_budget)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")

            candidates.append((obj_opt, a_opt.copy()))

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    top_objs = [f'{c[0]:.6f}' for c in candidates[:8]]
    print(f"\nTop candidates: {top_objs}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive optimization of top candidates
    print(f"\n=== Phase 2: Intensive optimization (best: {best_score:.8f}) ===")

    top_k = min(5, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]
        n_cand = len(a_cand)
        alloc = min(remaining * 0.35, 180.0)

        print(f"  Candidate {ci} (n={n_cand}, obj={obj_cand:.8f}): {alloc:.0f}s budget")

        a_pol, obj_pol = polish_candidate(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"  No improvement (got {obj_pol:.8f})")

    # Phase 3: Length perturbation on best
    print(f"\n=== Phase 3: Length perturbation (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = []
        for dn in [-30, -20, -10, 10, 20, 30, 50, -50, 70, -70, 100]:
            nv = n_best + dn
            if 400 <= nv <= 1500:
                test_lengths.append(nv)
        test_lengths = sorted(set(test_lengths))[:8]

        for n_test in test_lengths:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_resampled = normalize_simplex(resample_sequence(best_seq, n_test))
            alloc = min(remaining * 0.2, 40.0)
            a_pol, obj_pol = polish_candidate(a_resampled, time_budget=alloc)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** {obj_pol:.8f} **")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_final, obj_final = polish_candidate(best_seq, time_budget=remaining * 0.6)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: {best_score:.8f}")

        # Random perturbations
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_final = len(best_seq)
            rng = np.random.default_rng(seed + 12345)
            for pi in range(8):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.01 + 0.008 * pi
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
                a_pert = normalize_simplex(a_pert)
                t_alloc = min(remaining * 0.1, 12.0)
                if t_alloc < 5:
                    break
                a_pert, obj_pert = fista_smoothed(a_pert, max_iters=int(t_alloc * 20),
                                                  time_budget=t_alloc)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {pi} (s={scale:.4f}): ** {best_score:.8f} **")

    # Phase 5: One final LP if time and n allow
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        if n_best <= 850 and remaining > 25:
            print(f"\n=== Phase 5: Final LP (n={n_best}, {remaining:.0f}s) ===")
            a_lp, obj_lp = lp_alternate_adaptive(best_seq, max_iters=60,
                                                 time_budget=min(remaining * 0.8, 60.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
