"""
Optimized peak-autocorrelation sequence search using smooth subgradient with
adaptive temperature, FFT-accelerated gradients, mirror descent, peak chasing,
and alternating LP refinement. Targets improvement below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s > 1e-15:
        return a / s
    return np.ones(len(a)) / len(a)


def compute_grad_fft(a, w, fft_size=None):
    """
    Compute subgradient: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    using FFT. w has length 2n-1, zero-padded as needed.
    """
    n = len(a)
    ar = a[::-1]
    if fft_size is None:
        fft_size = 1
        while fft_size < 3 * n - 2:
            fft_size <<= 1
    fa = rfft(ar, fft_size)
    fw = rfft(w, fft_size)
    conv_full = irfft(fa * fw, fft_size)
    return 2.0 * conv_full[n - 1:2 * n - 1]


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (a * g)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=30):
    """
    Solve: min t s.t. (fixed_seq * g)[k] <= t, sum(g)=1, g>=0.
    Returns (g_normalized, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def parametric_init(n, style, rng):
    """Generate parametric initial sequences."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if style == 'uniform':
        a = np.ones(n)
    elif style == 'gaussian':
        sigma = n * (0.2 + 0.3 * rng.random())
        a = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
    elif style == 'tilted_gaussian':
        sigma = n * (0.3 + 0.2 * rng.random())
        slope = (0.5 + 1.5 * rng.random()) * (1 if rng.random() < 0.5 else -1)
        a = np.exp(-((i - center) ** 2) / (2 * sigma ** 2) + slope * (i - center) / n)
    elif style == 'sigmoid':
        c = n * (0.3 + 0.4 * rng.random())
        beta = 2.0 + 8.0 * rng.random()
        a = 1.0 / (1.0 + np.exp(-beta * (i - c) / n))
    elif style == 'power':
        alpha = 0.3 + 2.0 * rng.random()
        if rng.random() < 0.5:
            a = (i + 1) ** (-alpha)
        else:
            a = (n - i) ** (-alpha)
    elif style == 'cosine_tilt':
        width = n * (0.4 + 0.3 * rng.random())
        x = (i - center) / width
        a = np.where(np.abs(x) < 1, np.cos(np.pi * x / 2) ** 2, 0)
        slope = (0.5 + 1.0 * rng.random()) * (1 if rng.random() < 0.5 else -1)
        a = a * np.exp(slope * (i - center) / n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

        lp_time = min(remaining * 0.4, 30.0)
        g, t_opt = solve_minpeak_lp(a, time_limit=lp_time)
        if g is None:
            stall += 1
            if stall > 8:
                break
            continue

        # Try different blend factors
        best_blend_obj = best_obj
        best_blend_a = best_a.copy()
        for t_blend in [0.3, 0.5, 0.7, 0.8, 1.0]:
            a_try = normalize_simplex((1.0 - t_blend) * a + t_blend * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try.copy()

        if best_blend_obj < best_obj - 1e-14:
            best_obj = best_blend_obj
            best_a = best_blend_a
            a = best_blend_a
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb and retry
                rng = np.random.default_rng(it * 7919 + 3)
                a = normalize_simplex(best_a * np.exp(rng.normal(0, 0.03, n)))
                stall = 0

    return best_a, best_obj


def euclidean_subgradient_opt(a, max_iters=50000, time_budget=200.0,
                              tau=0.005, momentum=0.9, lr_init=0.05):
    """Euclidean subgradient with fixed temperature (for fine polishing)."""
    n = len(a)
    a = normalize_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    conv_fft_size = 1
    while conv_fft_size < 2 * n - 1:
        conv_fft_size <<= 1
    grad_fft_size = 1
    while grad_fft_size < 3 * n - 2:
        grad_fft_size <<= 1

    v = np.zeros(n)
    lr = lr_init
    start_time = time.monotonic()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        fa = rfft(a, conv_fft_size)
        conv = irfft(fa * fa, conv_fft_size)[:2 * n - 1]

        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            top_k = np.argsort(conv)[-max(5, n // 100):]
            weights = np.zeros(2 * n - 1)
            weights[top_k] = 1.0 / len(top_k)
        else:
            weights /= wsum

        ar = a[::-1]
        fa2 = rfft(ar, grad_fft_size)
        fw = rfft(weights, grad_fft_size)
        conv_grad = irfft(fa2 * fw, grad_fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        grad -= np.mean(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        v = momentum * v - lr * grad / gnorm
        a_new = a + v
        a_new = normalize_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.7
                v = 0.5 * v
                no_improve = 0

        if it % 500 == 499:
            lr *= 0.998

    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def polish_pipeline(a, time_budget=200.0):
    """Multi-method polishing pipeline."""
    n = len(a)
    a = normalize_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Stage 1: Fine smooth subgradient (mirror)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t1 = min(remaining * 0.35, 80.0)
        a1, obj1 = smooth_subgradient_opt(best_a, tau_start=0.01, tau_end=0.001,
                                          max_iters=50000, time_budget=t1,
                                          mirror=True, lr_init=0.06)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1

    # Stage 2: Peak chasing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t2 = min(remaining * 0.25, 50.0)
        a2, obj2 = peak_chase_opt(best_a, num_peaks=5, max_iters=30000,
                                  time_budget=t2, lr_init=0.02)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2

    # Stage 3: LP (if n allows)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15 and n <= 700:
        t3 = min(remaining * 0.3, 60.0)
        a3, obj3 = lp_alternate_opt(best_a, max_iters=30, time_budget=t3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3

    # Stage 4: Final Euclidean subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t4 = min(remaining * 0.35, 40.0)
        a4, obj4 = euclidean_subgradient_opt(best_a, max_iters=30000,
                                             time_budget=t4, tau=0.003,
                                             momentum=0.88, lr_init=0.03)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization combining smooth subgradient,
    peak chasing, LP refinement, and parametric initialization to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Try to load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        parent = parent[np.isfinite(parent)]
        if parent.size > 50:
            parent = normalize_simplex(parent)
            parent_obj = compute_obj(parent)
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
                print(f"Parent loaded: n={parent.size}, obj={parent_obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Generate initial sequence
        n_init = 700
        rng = np.random.default_rng(seed)
        best_seq = parametric_init(n_init, 'tilted_gaussian', rng)
        best_seq = normalize_simplex(best_seq)
        best_score = compute_obj(best_seq)
        print(f"Generated initial: n={len(best_seq)}, obj={best_score:.8f}")

    # Phase 1: Try multiple initializations and lengths (quick screen)
    print("\n=== Phase 1: Multi-start exploration ===")
    candidate_list = []  # (obj, seq)

    n_candidates = [len(best_seq)]
    for dn in [-100, -200, 100, 200, 300, 500]:
        nv = len(best_seq) + dn
        if 400 <= nv <= 1500:
            n_candidates.append(nv)
    n_candidates = sorted(set(n_candidates))

    styles = ['tilted_gaussian', 'sigmoid', 'cosine_tilt', 'gaussian', 'power']
    rng = np.random.default_rng(seed)

    # Add parent as a candidate
    candidate_list.append((best_score, best_seq.copy()))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 400:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        for style in styles[:3]:  # Try 3 styles per length
            a_init = parametric_init(n_try, style, rng)
            a_init = normalize_simplex(a_init)
            obj_init = compute_obj(a_init)

            # Quick optimization
            t_quick = min(remaining * 0.05, 15.0)
            if t_quick < 5:
                break
            a_opt, obj_opt = smooth_subgradient_opt(
                a_init, tau_start=0.05, tau_end=0.005,
                max_iters=20000, time_budget=t_quick,
                mirror=True, lr_init=0.08)

            candidate_list.append((obj_opt, a_opt.copy()))
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}, {style}: {obj_opt:.8f} ** NEW BEST **")

    # Sort and keep top candidates
    candidate_list.sort(key=lambda x: x[0])
    print(f"Top candidates: {[f'{c[0]:.6f}' for c in candidate_list[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive optimization of top candidates
    print("\n=== Phase 2: Intensive optimization ===")
    top_k = min(4, len(candidate_list))
    for ci in range(top_k):
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_cand = candidate_list[ci][1].copy()
        obj_cand = candidate_list[ci][0]
        alloc = min(remaining * 0.25, 120.0)
        print(f"  Candidate {ci} (n={len(a_cand)}, obj={obj_cand:.8f}): {alloc:.0f}s budget")

        a_pol, obj_pol = polish_pipeline(a_cand, time_budget=alloc)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")

    # Phase 3: Length exploration around best
    print(f"\n=== Phase 3: Length exploration (best: {best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_ns = sorted(set([n_best - 100, n_best - 50, n_best + 50,
                              n_best + 100, n_best + 200]))
        test_ns = [x for x in test_ns if 400 <= x <= 1500]

        for n_test in test_ns:
            if time.monotonic() > deadline - 150:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            a_res = normalize_simplex(resample_seq(best_seq, n_test))
            obj_res = compute_obj(a_res)
            alloc = min(remaining * 0.2, 50.0)
            a_pol, obj_pol = polish_pipeline(a_res, time_budget=alloc)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** UPDATE to {obj_pol:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f}")

    # Phase 4: Heavy polish on the best sequence
    print(f"\n=== Phase 4: Heavy polish (best: {best_score:.8f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            alloc = min(remaining * 0.5, 200.0)
            a_final, obj_final = polish_pipeline(best_seq, time_budget=alloc)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  After polish: {best_score:.8f}")

        # Phase 5: Perturbation and re-optimize
        remaining = deadline - time.monotonic()
        if remaining > 30 and best_seq is not None:
            n_final = len(best_seq)
            rng2 = np.random.default_rng(seed + 12345)
            for pert_i in range(5):
                if time.monotonic() > deadline - 20:
                    break
                scale = 0.015 + 0.015 * pert_i
                a_pert = best_seq * np.exp(rng2.normal(0, scale, n_final))
                a_pert = normalize_simplex(a_pert)
                t_alloc = min(remaining * 0.15, 30.0)
                if t_alloc < 10:
                    break
                a_pert, obj_pert = polish_pipeline(a_pert, time_budget=t_alloc)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {pert_i} (scale={scale:.3f}): ** {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
