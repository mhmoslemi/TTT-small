"""
Hybrid multi-temperature subgradient with FFT-accelerated gradients, LP polishing,
adaptive backtracking, and diverse multi-start search to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key innovations over previous approach:
- Multi-temperature softmax for composite peak/shape gradients
- Backtracking line search for robust step selection
- Log-parameterization option for automatic non-negativity
- Coarse-to-fine step schedule
- Orthogonal perturbation for escaping local minima

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def compute_subgradient_fft(a, weights, fft_size):
    """Compute subgradient g[j] = 2*sum_k weights[k]*a[k-j] using FFT.
    weights has length 2n-1, a has length n.
    """
    W = rfft(weights, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    return 2.0 * corr[:len(a)]


def fft_size_for(n):
    """Get FFT size needed for subgradient computation (N >= 3n-2)."""
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    return size


def multi_temp_subgradient(a, tau_list, fft_size):
    """Compute combined multi-temperature subgradient."""
    n = len(a)
    c = compute_conv(a)
    cmax = float(np.max(c))
    g = np.zeros(n)
    for tau in tau_list:
        shifted = (c - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        w = np.exp(shifted)
        ws = np.sum(w)
        if ws < 1e-300:
            idx = int(np.argmax(c))
            w = np.zeros(len(c))
            w[idx] = 1.0
        else:
            w /= ws
        g += compute_subgradient_fft(a, w, fft_size)
    g /= len(tau_list)
    return g, c, cmax

            stall += 1

def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=20):

    Solve: min t s.t. (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_maxsum_lp(fixed_seq, M, time_limit=20):
    """
    Solve: max sum(g) s.t. (fixed_seq * g)[k] <= M for all k, g >= 0.
    Returns g normalized to sum=1, or None.
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate_one_step(a, mode='minpeak'):
    """One step of LP alternating. Returns new_a or None."""
    n = len(a)
    if mode == 'minpeak':
        g = solve_minpeak_lp(a, time_limit=15)
    else:
        c = compute_conv(a)
        M = float(np.max(c))
        g = solve_maxsum_lp(a, M, time_limit=15)
    if g is None:
        return None
    # Try several blending factors
    best_a = a.copy()
    best_obj = compute_obj(a)
    for t in [0.3, 0.5, 0.7, 1.0]:
        a_try = normalize_simplex((1.0 - t) * a + t * g)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try
    return best_a


def polish_sequence(a, time_budget=60.0):
    """
    Intensive polishing: alternate between subgradient and LP steps.
    """
    n = len(a)
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    # Phase A: Subgradient (coarse)
    alloc_a = min(time_budget * 0.4, 25.0)
    a_s, obj_s = subgradient_backtrack(best_a, max_iters=2000, time_budget=alloc_a,
                                        tau_list=[0.005, 0.015, 0.04],
                                        init_step=0.12, min_step=0.005)
    if obj_s < best_obj:
        best_obj = obj_s
        best_a = a_s

    # Phase B: LP alternating
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 900:
        for step_i in range(6):
            if time.monotonic() - start > time_budget - 5:
                break
            remaining_step = time_budget - (time.monotonic() - start)
            if remaining_step < 8:
                break
            a_try = lp_alternate_one_step(best_a, mode='minpeak' if step_i % 2 == 0 else 'maxsum')
            if a_try is not None:
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try

    # Phase C: Fine subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_f, obj_f = subgradient_backtrack(best_a, max_iters=3000, time_budget=remaining * 0.6,
                                            tau_list=[0.002, 0.005, 0.01],
                                            init_step=0.05, min_step=0.002, momentum=0.9)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f

    return best_a, best_obj


def generate_start_uniform(n):
    return np.ones(n) / n


def generate_start_bell(n, width_frac=0.4):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n * width_frac / 2.0
    a = np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))
    return a / np.sum(a)


def generate_start_chirp(n, freq_param=3.0):
    """Generate a chirp-like envelope (non-negative)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (i - center) / center
    # Chirp-like: amplitude modulated by a smooth oscillation
    a = np.exp(-1.5 * x ** 2) * (1.0 + 0.3 * np.cos(freq_param * np.pi * x))
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def generate_start_dual_lobe(n, separation=0.3):
    """Two Gaussian lobes separated by 'separation' fraction of n."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n * 0.15
    c1 = center - separation * n / 2
    c2 = center + separation * n / 2
    a = np.exp(-((i - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((i - c2) ** 2) / (2 * sigma ** 2))
    a = np.maximum(a, 0)
    return a / np.sum(a)


def generate_start_random_smooth(n, seed=42, smoothness=3):
    """Random sequence smoothed by repeated averaging."""
    rng = np.random.default_rng(seed)
    a = np.abs(rng.normal(0, 1, n)) + 0.1
    for _ in range(smoothness):
        a = (a + np.roll(a, 1)) / 2.0
        a = (a + np.roll(a, -1)) / 2.0
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def generate_diverse_starts(n, seed=42):
    """Generate a diverse set of initial sequences."""
    starts = [
        generate_start_uniform(n),
        generate_start_bell(n, 0.3),
        generate_start_bell(n, 0.5),
        generate_start_bell(n, 0.7),
        generate_start_chirp(n, 2.0),
        generate_start_chirp(n, 4.0),
        generate_start_chirp(n, 6.0),
        generate_start_dual_lobe(n, 0.2),
        generate_start_dual_lobe(n, 0.4),
        generate_start_random_smooth(n, seed=seed, smoothness=2),
        generate_start_random_smooth(n, seed=seed + 1, smoothness=5),
        generate_start_random_smooth(n, seed=seed + 2, smoothness=8),
    ]
    # Add some parametric variations
    for k in range(4):
        rng = np.random.default_rng(seed + 100 + k)
        sigma = n * (0.2 + 0.4 * rng.random())
        freq = 1.0 + 5.0 * rng.random()
        amp = 0.1 + 0.5 * rng.random()
        phase = 2 * np.pi * rng.random()
        i = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        bell = np.exp(-((i - center) ** 2) / (2 * sigma ** 2))
        mod = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = bell * mod
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-15:
            starts.append(a / s)
    return starts


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded: n={parent.size}, obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start diverse exploration with multi-temperature
    subgradient (backtracking), LP alternating polishing, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # === PHASE 1: Quick subgradient on parent ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            alloc = min(remaining * 0.2, 120.0)
            print(f"\n[Phase 1] Subgradient on parent (n={n_parent}), budget={alloc:.0f}s")
            a_opt, obj_opt = subgradient_backtrack(
                best_seq, max_iters=5000, time_budget=alloc,
                tau_list=[0.003, 0.01, 0.03],
                init_step=0.12, min_step=0.003, momentum=0.85)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Improved: {best_score:.8f}")

    # === PHASE 2: Multi-start diverse exploration ===
    print(f"\n[Phase 2] Multi-start exploration (best: {best_score:.8f})")

    # Try several lengths
    n_candidates = sorted(set([
        n_parent,
        n_parent - 100, n_parent + 100,
        n_parent - 200, n_parent + 200,
        n_parent - 50, n_parent + 50,
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1200]

    all_candidates = []  # (obj, seq)

    if best_seq is not None:
        all_candidates.append((best_score, best_seq.copy()))

    for n in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        starts = generate_diverse_starts(n, seed=seed + n)
        # Include resampled best if available
        if best_seq is not None and len(best_seq) != n:
            starts.insert(0, normalize_simplex(resample_sequence(best_seq, n)))

        # Quick screen each start
        num_starts = len(starts)
        alloc_per = min(remaining / num_starts, 20.0)

        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(alloc_per, 15.0)
            if t_alloc < 3:
                break

            a_opt, obj_opt = subgradient_backtrack(
                s, max_iters=2000, time_budget=t_alloc,
                tau_list=[0.005, 0.015, 0.04],
                init_step=0.15, min_step=0.005)

            all_candidates.append((obj_opt, a_opt.copy()))
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {best_score:.8f} ** NEW BEST **")

    # Sort and keep top candidates
    all_candidates.sort(key=lambda x: x[0])
    print(f"  Top 5: {[f'{c[0]:.6f}' for c in all_candidates[:5]]}")

    # === PHASE 3: Intensive polishing of top candidates ===
    print(f"\n[Phase 3] Intensive polishing (best: {best_score:.8f})")

    top_k = min(5, len(all_candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_cand = all_candidates[ci][1].copy()
        n_cand = len(a_cand)
        alloc = min(remaining * 0.25, 80.0)

        print(f"  Candidate {ci} (n={n_cand}, obj={all_candidates[ci][0]:.8f}): polishing {alloc:.0f}s")
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=alloc)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"    ** UPDATE: {best_score:.8f} **")
        else:
            print(f"    No improvement ({obj_pol:.8f})")

    # === PHASE 4: Final intensive optimization on the best ===
    print(f"\n[Phase 4] Final intensive (best: {best_score:.8f})")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Long subgradient with very fine temperatures
            alloc = min(remaining * 0.5, 120.0)
            a_final, obj_final = subgradient_backtrack(
                best_seq, max_iters=10000, time_budget=alloc,
                tau_list=[0.001, 0.003, 0.008],
                init_step=0.05, min_step=0.001, momentum=0.9)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Fine subgradient: {best_score:.8f} **")

            remaining = deadline - time.monotonic()
            if remaining > 20 and len(best_seq) <= 900:
                # LP polishing
                a_lp, obj_lp = polish_sequence(best_seq, time_budget=min(remaining * 0.6, 60.0))
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  LP polish: {best_score:.8f} **")

    # === PHASE 5: Perturbation restarts ===
    print(f"\n[Phase 5] Perturbation restarts (best: {best_score:.8f})")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 77777)

        for pi in range(8):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # Vary perturbation scale
            scale = 0.015 + 0.015 * pi
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_pert = normalize_simplex(a_pert)

            t_alloc = min(remaining * 0.3, 25.0)
            if t_alloc < 5:
                break

            a_opt, obj_opt = subgradient_backtrack(
                a_pert, max_iters=2000, time_budget=t_alloc,
                tau_list=[0.002, 0.008, 0.02],
                init_step=0.08, min_step=0.003)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Perturb {pi} (scale={scale:.3f}): {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0 / 100] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
