"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint LP best-response alternation (all 2n-1 constraints)
2. Nesterov-accelerated projected subgradient with backtracking
3. Exact 1D coordinate descent via golden section
4. Multi-length, multi-init search with perturbation/restart
Key improvement: full LP (not top-k subset) gives stronger best-response steps.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0) if v[0] > 0 else 1e-15])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (target_sum + u * rho_arr) > cv
    if not np.any(cond):
        theta = 0.0
    else:
        rho = int(np.argmax(~cond))
        theta = (cv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.full(n, target_sum / n)
    return w * (target_sum / s)


def build_full_lp_matrix(n, a):
    """Build full sparse constraint matrix A where A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response_full(a, M):
    """Solve full-constraint LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_full_lp_matrix(n, a)
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_best_response_topk(a, M, k_active=50):
    """Solve LP with only top-k_active convolution constraints."""
    n = len(a)
    b = fast_conv(a)
    top_idx = np.argsort(b)[::-1][:min(k_active, 2 * n - 1)]
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(top_idx):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(ii)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    if A.nnz == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def nesterov_subgradient(a_init, max_iters, deadline):
    """Nesterov-accelerated projected subgradient with backtracking."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    eta = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        by = fast_conv(y)
        k_star = int(np.argmax(by))

        # Subgradient of max_k conv(y,y)[k] at the maximizer
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * y[j]

        # Project to tangent space
        grad_tan = grad - grad.mean()

        # Nesterov step
        a_new = y - eta * grad_tan
        a_new = proj_simplex(a_new)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)

        a_prev = a
        a = a_new
        y = y_new
        t = t_new

        # Decay step size
        eta = 0.3 / (1.0 + 0.01 * it)

        # Track best
        if it % 3 == 0:
            M_a = fast_conv(a).max()
            if M_a < best_M:
                best_M = M_a
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
        best_M = M_final
    return best_a


def coordinate_descent_1d(a, i, b, n):
    """Find optimal a[i] given all others fixed. Golden section on convex 1D f."""
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    c_vals = []
    slopes = []
    quad_flag = False
    quad_idx = -1

    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            quad_flag = True
            quad_idx = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)

    def obj_at(x_test):
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_flag:
            qv = c_arr[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m

    lo = 0.0
    hi = max(2.0 * x_cur + 0.1, np.sqrt(max(fast_conv(a).max(), 0.0)) * 1.5, 0.3)
    for _ in range(15):
        if obj_at(hi) > obj_at(hi * 0.7) and hi < 50.0:
            hi *= 1.5
        else:
            break

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Cyclic/random coordinate descent with exact 1D optimization."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = np.arange(n)
        rng_cd.shuffle(order)

        b = fast_conv(a)
        any_change = False
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-12 * max(x_opt, 1.0):
                delta = x_opt - a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += x_opt * x_opt - (x_opt - delta) * (x_opt - delta)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * delta
                any_change = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            break

    return best_a


def lp_alternation(a_init, max_iters, deadline, use_full=True):

    s = a.sum()
    if s < 1e-10:

    else:
        a = a / s
    n = len(a)

    best_M = fast_conv(a).max()
    no_improve = 0

        if use_full:

        else:
            g = lp_best_response_topk(a, M_cur, k_active=min(50, 2 * n - 1))

        if g is None:
            # Fallback: try with top-k
            g = lp_best_response_topk(a, M_cur, k_active=30)

        if g is None:
            no_improve += 2
            if no_improve > 10:
                break
            continue

        # Geometric mixing
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 0.01:
                continue
            a_trial = a_trial / st
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_a


def spectral_init(n, rng):
    """Initialize by trying to flatten the power spectrum."""
    a = np.ones(n) / n
    for _ in range(50):
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
        grad_tan = grad - grad.mean()
        a = a - 0.15 * grad_tan
        a = np.maximum(a, 1e-4)
        a = a / a.sum()
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        # Resample parent
        if parent.size != n:
            old_t = np.linspace(0, 1, parent.size)
            new_vals = np.interp(t, old_t, parent)
        else:
            new_vals = parent.copy()
        p = np.maximum(new_vals, 0.0)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed parent
        p2 = np.maximum(p + rng.normal(0, 0.05, n), 1e-4)
        inits.append(p2 / p2.sum())
        p3 = np.maximum(p + rng.normal(0, 0.15, n), 1e-4)
        inits.append(p3 / p3.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Spectral
    inits.append(spectral_init(n, rng))

    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    # Gaussian
    gauss = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    sr = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sr = np.maximum(sr, 0.01)
    inits.append(sr / sr.sum())

    # Two-lobe
    two = 0.3 + 0.7 * np.abs(np.sin(2 * np.pi * t))
    inits.append(two / two.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample sequence via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return np.maximum(src, 0.0).copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def optimize_single(a0, n, deadline, seed=0, use_full_lp=True):
    """Full optimization pipeline on a single sequence."""
    best = a0.copy()
    best_M = fast_conv(a0).max()

    # Stage 1: Nesterov subgradient (30% of time)
    t1 = time.monotonic() + 0.3 * (deadline - time.monotonic())
    a_ns = nesterov_subgradient(best, 5000, t1)
    M_ns = fast_conv(a_ns).max()
    if M_ns < best_M:
        best_M = M_ns
        best = a_ns.copy()
    print(f"    After Nesterov: M={best_M:.10f}", flush=True)

    # Stage 2: LP alternation (40% of time)
    t2 = time.monotonic() + 0.4 * (deadline - time.monotonic())
    est_lp_time = 3.0 if use_full_lp else 1.5
    max_lp = max(5, int((t2 - time.monotonic()) / est_lp_time))
    a_lp = lp_alternation(best, max_lp, t2, use_full=use_full_lp)
    M_lp = fast_conv(a_lp).max()
    if M_lp < best_M:
        best_M = M_lp
        best = a_lp.copy()
    print(f"    After LP: M={best_M:.10f}", flush=True)

    # Stage 3: Coordinate descent (20% of time)
    t3 = time.monotonic() + 0.2 * (deadline - time.monotonic())
    max_sw = max(5, int((t3 - time.monotonic()) / max(0.3, n / 80.0)))
    a_cd = run_coordinate_descent(best, max_sw, t3, seed=seed)
    M_cd = fast_conv(a_cd).max()
    if M_cd < best_M:
        best_M = M_cd
        best = a_cd.copy()
    print(f"    After CD: M={best_M:.10f}", flush=True)

    # Stage 4: Final LP polish (10% of time)
    if time.monotonic() < deadline - 2:
        a_fin = lp_alternation(best, 20, deadline, use_full=True)
        M_fin = fast_conv(a_fin).max()
        if M_fin < best_M:
            best_M = M_fin
            best = a_fin.copy()
        print(f"    After Final LP: M={best_M:.10f}", flush=True)

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: Nesterov subgradient + full-constraint LP alternation
    + exact coordinate descent, across multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # Determine lengths to try
    lengths = [n_parent]
    for delta in [-128, -64, 64, 128, 256, -256]:
        nl = n_parent + delta
        if 256 <= nl <= 1536 and nl not in lengths:
            lengths.append(nl)
    if 768 not in lengths:
        lengths.append(768)
    if 1024 not in lengths:
        lengths.append(1024)
    # Sort by priority
    lengths.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))
    lengths = lengths[:5]

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-length optimization. Lengths: {lengths}", flush=True)
    print(f"{'='*60}", flush=True)

    # Reserve time for later phases
    reserve_late = 300
    avail = deadline - time.monotonic() - reserve_late
    time_per_len = max(60, avail / len(lengths))

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - reserve_late:
            break
        remaining = deadline - reserve_late - time.monotonic()
        if remaining < 30:
            break

        t_alloc = min(time_per_len, remaining / max(1, len(lengths) - li))
        print(f"\n  n={n}, alloc={t_alloc:.0f}s", flush=True)

        inits = generate_inits(n, rng, tracker.best if tracker.best is not None else parent)

        best_for_n = None
        best_M_n = float('inf')

        # Time per init (allow 2-4 inits)
        n_inits_to_try = min(len(inits), max(2, int(t_alloc / 80)))
        t_per_init = t_alloc / n_inits_to_try

        for ii in range(n_inits_to_try):
            if time.monotonic() > deadline - reserve_late:
                break
            a0 = np.maximum(inits[ii], 1e-6)
            a0 = a0 / a0.sum()
            sub_dl = min(time.monotonic() + t_per_init, deadline - reserve_late)
            if time.monotonic() >= sub_dl:
                break

            a_opt = optimize_single(a0, n, sub_dl, seed=seed + li * 100 + ii,
                                    use_full_lp=(n <= 1024))
            M_opt = fast_conv(a_opt).max()
            if M_opt < best_M_n:
                best_M_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n}: best J={J_val:.6f} (elapsed remainder)", flush=True)
            tracker.update(best_for_n)

    tracker.report("After Phase 1")

    # Phase 2: Intensive LP polish on global best
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Intensive full-LP polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 60:
        t_lp = min(deadline - 40, time.monotonic() + 180)
        a_lp = lp_alternation(tracker.best, 100, t_lp, use_full=True)
        tracker.update(a_lp)
        print(f"  After LP polish: J={tracker.best_score:.8f}", flush=True)

    tracker.report("After Phase 2")

    # Phase 3: Coordinate descent + Nesterov refinement
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: CD + Nesterov refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        t_cd = min(deadline - 20, time.monotonic() + 120)
        max_sw = max(20, int((t_cd - time.monotonic()) / max(0.3, n_best / 100.0)))
        a_cd = run_coordinate_descent(tracker.best, max_sw, t_cd, seed=12345)
        tracker.update(a_cd)

        if time.monotonic() < deadline - 20:
            t_ns = min(deadline - 10, time.monotonic() + 60)
            a_ns = nesterov_subgradient(tracker.best, 3000, t_ns)
            tracker.update(a_ns)

    tracker.report("After Phase 3")

    # Phase 4: Perturbation + re-optimization (escape local optima)
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        n_restarts = 3
        t_per_restart = (deadline - 30 - time.monotonic()) / n_restarts
        t_per_restart = max(t_per_restart, 20)

        for r in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            # Different perturbation scales
            scale = [0.03, 0.08, 0.15][r % 3]
            perturbed = np.maximum(tracker.best + rng.normal(0, scale, n_best), 1e-5)
            perturbed = perturbed / perturbed.sum()
            sub_dl = min(deadline - 20, time.monotonic() + t_per_restart)
            a_opt = optimize_single(perturbed, n_best, sub_dl, seed=77777 + r,
                                    use_full_lp=(n_best <= 1024))
            tracker.update(a_opt)

    tracker.report("After Phase 4")

    # Phase 5: Try a different length from the best
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 40:
        n_cur = tracker.best.size
        for mult in [1.25, 0.8]:
            if time.monotonic() > deadline - 30:
                break
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 1536)
            new_n = (new_n // 32) * 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 20, time.monotonic() + 60)
            a_opt = optimize_single(a_ext, new_n, sub_dl, seed=55555,
                                    use_full_lp=(new_n <= 1024))
            tracker.update(a_opt)

    tracker.report("After Phase 5")

    # Final safety
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
