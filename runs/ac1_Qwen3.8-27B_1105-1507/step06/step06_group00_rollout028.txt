import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def subgradient_max_conv(a):
    """Compute subgradient of max_k (a*a)_k with respect to a."""

                grad[i] = 2.0 * a[j]
    return grad, k_star, conv

        grad_centered = grad - np.mean(grad)

        momentum = momentum_beta * momentum + (1 - momentum_beta) * grad_centered
        
        # Multiplicative update
        a_new = a * np.exp(-eta * momentum)
        a_new = np.maximum(a_new, 1e-30)  # Avoid log(0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                # Decrease eta slowly
                eta *= 0.999
            else:
                # Try smaller step
                a_try = a * np.exp(-eta * 0.1 * momentum)
                a_try = np.maximum(a_try, 1e-30)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                    else:
                        eta *= 0.7
                        momentum *= 0.5
        else:
            eta *= 0.5
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog for speed (no CVXPY overhead).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # LP: min M
    # s.t. A @ g - M <= 0  (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', 
                      options={'time_limit': timeout, 'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def solve_lp_minpeak_cvxpy(a, timeout=10.0):
    """CVXPY version of min-peak LP."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def geometric_blend(a, g, t):
    """Geometric mean blend: a^(1-t) * g^t, normalized."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_g = np.log(np.maximum(g, 1e-30))
    log_blended = (1 - t) * log_a + t * log_g
    blended = np.exp(log_blended)
    blended = np.maximum(blended, 0)
    s = float(np.sum(blended))
    if s > 1e-10:
        blended = blended / s
    else:
        blended = (a + g) / 2
        blended = blended / np.sum(blended)
    return blended

def peak_rebalance(a, n_rounds=50, time_budget=20.0):
    """
    Direct peak reduction: identify the peak convolution bin and
    redistribute mass to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find the "partners" that contribute to the peak
        # (a*a)[k_star] = sum_i a[i] * a[k_star - i]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Move mass from high-contributing positions to low ones
        # that don't contribute to the peak
        sorted_idx = np.argsort(contributions)[::-1]
        
        improved_this_round = False
        for i in sorted_idx[:10]:  # Try top 10 contributors
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-6:
                continue
            
            # Find a target position that doesn't contribute to this peak
            # and has low value
            for target in range(n):
                target_j = k_star - target
                contributes = (0 <= target_j < n)
                if contributes:
                    continue
                
                # Try moving a small amount
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * a[i]
                    if delta < 1e-8:
                        break
                    a[i] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved_this_round = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
            
            if improved_this_round:
                break
        
        if not improved_this_round:
            break

    return best_a


def smooth_descent(a, n_steps=200, time_budget=30.0):
    """
    Gradient descent on a smoothed version of the objective using log-sum-exp.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = 0.01 * float(np.max(compute_conv(a)))
    step = 0.01
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        
        # Smoothed max via log-sum-exp
        # grad of log-sum-exp(conv / tau) w.r.t. a_i
        weights = np.exp((conv - conv.max()) / tau)
        weights = weights / np.sum(weights)
        
        # Gradient of sum_k weights_k * conv_k w.r.t. a
        grad = np.zeros(n)
        for k in range(len(conv)):
            if weights[k] < 1e-10:
                continue
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] += weights[k] * a[j]
                    else:
                        grad[i] += weights[k] * 2.0 * a[j]
        
        # Center for sum constraint
        grad_centered = grad - np.mean(grad)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad_centered
        
        # Update
        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step = min(step * 1.05, 0.1)
            else:
                step *= 0.8
                momentum *= 0.5
        
        # Adaptively decrease tau
        tau *= 0.99
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def smooth_gradient(a, w):
    """Compute gradient of sum_k w_k * (a*a)_k with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-12:
            continue

        valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))

        grad[valid_i] += wk * contrib
    return grad

"""
Hybrid optimization combining multiplicative weight updates, LP refinement,
geometric blending, and smoothed gradient descent for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Key innovations over prior LP-only approach:
- Geometric mean blending escapes different local minima
- Exponentiated gradient with momentum for non-Euclidean exploration
- Log-sum-exp smoothed descent for handling max non-differentiability
- Stochastic subgradients from near-active set for robustness
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def subgradient_at_peak(a):
    """Compute subgradient of max_k (a*a)_k. Returns (grad, k_star, conv)."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if i == j:
                grad[i] = a[j]
            else:
                grad[i] = 2.0 * a[j]
    return grad, k_star, conv


def stochastic_subgradient(a, num_samples=10):
    """Sample subgradients from near-active set for robustness."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Find all bins within 1% of max
    active = np.where(conv >= max_val * 0.99)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    grad_sum = np.zeros(n)
    for _ in range(num_samples):
        k = active[np.random.randint(len(active))]
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if i == j:
                    grad_sum[i] += a[j]
                else:
                    grad_sum[i] += 2.0 * a[j]
    grad_sum /= num_samples
    return grad_sum, max_val


def multiplicative_update(a, eta_init=0.05, n_steps=150, time_budget=25.0):
    """
    Exponentiated gradient (multiplicative weight) with momentum.
    Naturally keeps values non-negative.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    momentum = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        grad, k_star, conv = subgradient_at_peak(a)
        grad_centered = grad - np.mean(grad)
        momentum = beta * momentum + (1.0 - beta) * grad_centered

        a_new = a * np.exp(-eta * momentum)
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1

                    eta *= 0.8

                            best_a = a_try.copy()
                            a = a_try
                            stall = 0

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak_scipy(a, timeout=8.0):
    """Solve LP min M s.t. (a@g)_k <= M, sum(g)=1, g>=0 using scipy HiGHS."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_minpeak_cvxpy(a, timeout=10.0):
    """CVXPY fallback for min-peak LP."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp(a):
    """Try scipy first, then cvxpy."""
    result = solve_lp_minpeak_scipy(a, timeout=6.0)
    if result[0] is not None:
        return result
    return solve_lp_minpeak_cvxpy(a, timeout=8.0)


def geometric_blend(a, g, t):
    """Geometric mean blend: a^(1-t) * g^t, normalized."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_g = np.log(np.maximum(g, 1e-30))
    blended = np.exp((1.0 - t) * log_a + t * log_g)
    blended = np.maximum(blended, 0)
    s = float(np.sum(blended))
    if s > 1e-10:
        return blended / s
    # Fallback to linear
    lin = (1.0 - t) * a + t * g
    lin = np.maximum(lin, 0)
    sl = float(np.sum(lin))
    if sl > 1e-10:
        return lin / sl
    return a.copy()

def smoothed_descent(a, n_steps=150, time_budget=25.0):
    """Gradient descent on log-sum-exp smoothed objective with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv_init = compute_conv(a)
    tau = 0.005 * float(np.max(conv_init))
    step = 0.005
    momentum = np.zeros(n)
    beta_m = 0.9

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))

        # Log-sum-exp weights
        shifted = (conv - max_c) / tau
        shifted = np.clip(shifted, -50, 50)
        w = np.exp(shifted)
        w = w / np.sum(w)

        # Gradient of weighted sum of conv k
        grad = np.zeros(n)
        for k in range(len(conv)):
            wk = w[k]
            if wk < 1e-10:
                continue
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                if i == j:
                    grad[i] += wk * a[j]
                else:
                    grad[i] += 2.0 * wk * a[j]

        # Project to sum=const
        grad_c = grad - np.mean(grad)
        momentum = beta_m * momentum + (1.0 - beta_m) * grad_c

        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step = min(step * 1.02, 0.02)
            else:
                step *= 0.85
                momentum *= 0.5

        # Slowly decrease tau (annealing)
        tau *= 0.995

    return best_a


def peak_direct_reduce(a, n_rounds=40, time_budget=15.0):
    """
    Directly reduce the peak convolution bin by moving mass
    from contributing positions to non-contributing ones.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Compute contributions to peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    contrib[i] = a[i] * a[j]
                else:
                    contrib[i] = a[i] * a[j]

        # Sort by contribution (highest first)
        order = np.argsort(contrib)[::-1]

        # Find positions that DON'T contribute to this peak
        non_contrib = []
        for pos in range(n):
            j = k_star - pos
            if not (0 <= j < n):
                non_contrib.append(pos)

        if not non_contrib:
            break

        # Try moving from top contributors to non-contributing positions
        improved = False
        for idx in order[:8]:
            if time.monotonic() - start > time_budget:
                break
            if a[idx] < 1e-6:
                continue

            # Pick a random non-contributing target
            target = int(np.random.randint(len(non_contrib)))
            target = non_contrib[target]

            for frac in [0.01, 0.02, 0.05, 0.1]:
                delta = frac * a[idx]
                if delta < 1e-8:
                    break
                a[idx] -= delta
                a[target] += delta
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation on [0,1]."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_pipeline(a, time_budget=60.0, verbose=False):
    """

    cycle = 0
    while time.monotonic() - start < time_budget:
        m_name, m_func = methods[cycle % len(methods)]
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:

        alloc = min(remaining * 0.3, 20.0)

            else:
                if verbose:
                    print(f"      [{m_name}] no improvement (obj={obj_new:.8f})")
        cycle += 1

    return best_a


def diverse_restart(n, rng, mode):
    """Generate diverse initializations."""
    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Smooth random
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amps = rng.uniform(0.1, 0.5, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif mode == 2:
        # Triangular
        mid = n // 2
        a = 1.0 - np.abs(np.arange(n) - mid) / mid
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif mode == 3:
        # Quadratic bump
        x = np.linspace(-1, 1, n)
        a = (1 - x ** 2) ** 2 + 0.05
        return a / np.sum(a)
    elif mode == 4:
        # Exponential decay
        a = np.exp(-np.arange(n) / (n * 0.3))
        return a / np.sum(a)
    else:
        # Random smooth (low-pass filtered noise)
        a = rng.uniform(0, 1, n)
        # Simple moving average
        k = max(3, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / np.sum(a)

        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r % 6
        if r == 0:
            # Perturbed best (small)
            a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif r == 1:
            # Perturbed best (medium)
            n_r = n_best + int(rng.integers(-30, 30))
            n_r = max(200, min(1500, n_r))
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.08, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            n_r = n_best
            a_r = diverse_restart(n_r, rng, mode - 2)

        alloc = min(remaining * 0.1, 25.0)
        a_opt = optimize_pipeline(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (n={len(a_opt)}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={len(a_opt)}): obj={obj_r:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    for polish_round in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Alternate between methods with increasing focus on LP and multiplicative
        alloc = min(remaining * 0.25, 35.0)

        if polish_round % 2 == 0:
            a_p = lp_alternate_blend(best_seq, time_budget=alloc)
        else:
            a_p = multiplicative_update(best_seq, eta_init=0.02, n_steps=100, time_budget=alloc)
        
        # Follow up with smoothed descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_p = smoothed_descent(a_p, n_steps=80, time_budget=min(remaining * 0.3, 15.0))

        a_p = np.maximum(a_p, 0)
        s_p = float(np.sum(a_p))
        if s_p > 1e-10:
            a_p = a_p / s_p
            obj_p = compute_obj(a_p)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Polish {polish_round}: obj={best_score:.8f}")
            else:
                print(f"  Polish {polish_round}: obj={obj_p:.8f} (stalled)")
                if polish_round >= 2:
                    break

    # === Phase 5: Final micro-refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Phase 5: Micro-Refinement ===")
        # Quick coordinate descent on the peak
        n = len(best_seq)
        a = best_seq.copy()
        for _ in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            a微 = peak_direct_reduce(a, n_rounds=20, time_budget=min(remaining * 0.3, 10.0))
            obj_微 = compute_obj(a微)
            if obj_微 < best_score:
                best_score = obj_微
                best_seq = a微.copy()
                a = a微
                print(f"  Micro: obj={best_score:.8f}")
            else:
                break

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
