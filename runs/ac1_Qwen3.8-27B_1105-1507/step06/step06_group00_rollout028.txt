def fast_lp_step(a):
    """Solve the min-peak LP using scipy HiGHS."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    
    # Build A matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        start = j
        length = min(n, 2*n-1-j)
        A[start:start+length, j] = a[:length]
    
    # Objective: [0, 0, ..., 0, 1] (minimize M which is the last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

        g = res.x[:n]
        g = np.maximum(g, 0)
        s = np.sum(g)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j+length, j] = a[:length]
    return A

def solve_lp_minpeak_fast(a):
    """Solve min-peak LP using HiGHS. Returns g (normalized) or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    
    if res.status == 0:
        g = res.x[:n]
        g = np.maximum(g, 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None

def golden_section_line_search(a, direction, max_iter=30):
    """Find best t in [0,1] to blend a with direction."""
    def phi(t):
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        return compute_obj(a_try / s)
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    a_gs = 0.0
    b_gs = 1.0
    c_gs = b_gs - gr * (b_gs - a_gs)
    d_gs = a_gs + gr * (b_gs - a_gs)
    
    fc = phi(c_gs)
    fd = phi(d_gs)
    
    for _ in range(max_iter):
        if fc < fd:
            b_gs = d_gs
            d_gs = c_gs
            fd = fc
            c_gs = b_gs - gr * (b_gs - a_gs)
            fc = phi(c_gs)
        else:
            a_gs = c_gs
            c_gs = d_gs
            fc = fd
            d_gs = a_gs + gr * (b_gs - a_gs)
            fd = phi(d_gs)
        if abs(b_gs - a_gs) < 1e-4:
            break
    
    t_opt = (a_gs + b_gs) / 2
    a_result = (1.0 - t_opt) * a + t_opt * direction
    a_result = np.maximum(a_result, 0)
    s = float(np.sum(a_result))
    if s < 1e-10:
        return a, np.inf
    return a_result / s, phi(t_opt)

def alternating_lp_optimize(a, time_budget=60.0, verbose=False):
    """LP alternating with golden section line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp = solve_lp_minpeak_fast(a)
        if g_lp is None:
            break
        
        # Line search
        a_new, obj_new = golden_section_line_search(a, g_lp, max_iter=20)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 50 == 0:
                print(f"    it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 30:
                # Random perturbation
                rng = np.random.default_rng(hash((it, n, best_obj)) % (2**32))
                scale = 0.05
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0
    
    return best_a

    exp_conv = np.exp(alpha * conv_shifted)
    p = exp_conv / np.sum(exp_conv)

"""
Hybrid optimization combining soft-max gradient descent with LP alternating,
using scipy HiGHS for fast LP solves, Nesterov momentum, adaptive alpha schedule,
multi-n exploration, and golden-section line search.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_normalized(a):
    """Compute 2*n*max(conv) assuming sum(a) is already 1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build sparse-ish matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns normalized g or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None


def softmax_obj_grad(a, alpha):
    """
    Compute soft-max approximation of 2n*max(conv(a,a)) and its gradient.
    Assumes sum(a) = 1.
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    conv_shifted = conv - max_conv
    exp_conv = np.exp(alpha * conv_shifted)
    sum_exp = float(np.sum(exp_conv))

    obj = 2.0 * n * (max_conv + (1.0 / alpha) * np.log(sum_exp))

    p = exp_conv / sum_exp

    # Gradient: d/da[i] [(1/alpha)*log(sum exp(alpha*c_k))]
    # = (1/alpha) * sum_k p[k] * 2*a[k-i] (if 0<=k-i<n)
    # = (2/alpha) * f[i] where f[i] = sum_k p[k]*a[k-i]
    # Full objective gradient: 2n * (2/alpha) * f[i] = (4n/alpha) * f[i]

    # f[i] = sum_k p[k] * a[k-i] = cross-correlation
    # f[i] = conv(p_rev, a)[2n-2-i] where p_rev[k] = p[2n-2-k]
    p_rev = p[::-1].copy()
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    fp = rfft(p_rev, size)
    fa = rfft(a, size)
    conv_result = irfft(fp * fa, size)[: (2 * n - 1) + n - 1]

    # f[i] = conv_result[2n-2-i] for i=0..n-1
    f = conv_result[2 * n - 2 - np.arange(n)][::-1].copy()
    # Wait, let me re-derive. f[i] = sum_k p[k]*a[k-i]
    # p_rev[m] = p[2n-2-m]
    # conv(p_rev, a)[m] = sum_j p_rev[j]*a[m-j] = sum_j p[2n-2-j]*a[m-j]
    # Let k = 2n-2-j, so j = 2n-2-k:
    # = sum_k p[k]*a[m-2n+2+k]
    # We want f[i] = sum_k p[k]*a[k-i]
    # So: m - 2n + 2 + k = k - i => m = k - i - k + 2n - 2 = 2n - 2 - i
    # So f[i] = conv(p_rev, a)[2n-2-i]

    f = np.zeros(n)
    for i in range(n):
        idx = 2 * n - 2 - i
        if 0 <= idx < len(conv_result):
            f[i] = conv_result[idx]

    grad = (4.0 * n / alpha) * f
    return obj, grad


def topk_softmax_obj_grad(a, alpha, K=5):
    """
    Use only top-K peaks in the soft-max for a "harder" gradient.
    Assumes sum(a) = 1.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    # Find top K peaks
    if K >= len(conv):
        topk_idx = np.arange(len(conv))
    else:
        topk_idx = np.argpartition(conv, -K)[-K:]

    conv_topk = conv[topk_idx]
    conv_topk_shifted = conv_topk - np.max(conv_topk)
    exp_topk = np.exp(alpha * conv_topk_shifted)
    sum_exp = float(np.sum(exp_topk))
    p_topk = exp_topk / sum_exp

    obj = 2.0 * n * (np.max(conv_topk) + (1.0 / alpha) * np.log(sum_exp))

    # Gradient using only top-K peaks (uniform weighting for simplicity)
    # grad[i] = (4n/alpha) * sum_{k in topk} p_topk[k]*a[k-i]
    grad = np.zeros(n)
    for idx_k in range(len(topk_idx)):
        k = topk_idx[idx_k]
        w = p_topk[idx_k]
        # a[k-i] is valid for i in [k-n+1, k] ∩ [0, n-1]
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:
            j_vals = k - np.arange(i_min, i_max + 1)  # j = k - i
            grad[i_min:i_max + 1] += w * a[j_vals]

    grad *= (4.0 * n / alpha)
    return obj, grad


def xn_project(v, z):
    """Project v onto simplex {x >= 0, sum(x) = z}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - z)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, z / n)
    rho = idx[-1]
    theta = (cssv[rho] - z) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_gradient_descent(a, time_budget=60.0, alpha_start=30.0, alpha_end=1000.0,
                              n_max_steps=5000, use_topk=False, topk_K=5):
    """
    Soft-max gradient descent with Nesterov momentum and adaptive alpha.
    Assumes input a has sum=1.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    step_scale = 0.5

    for step in range(n_max_steps):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive alpha (geometric progression)
        frac = step / max(n_max_steps - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        if use_topk:
            obj_cur, grad = topk_softmax_obj_grad(y, alpha, topk_K)
        else:
            obj_cur, grad = softmax_obj_grad(y, alpha)

        # Nesterov update
        # y_{k+1} = x_k + beta*(x_k - x_{k-1}) = (1+beta)*x_k - beta*x_{k-1}
        # x_{k+1} = project(y_{k+1} - step_size * grad)

        # Adaptive step size
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-12:
            break
        eta = step_scale / grad_norm * 0.1

        # Cap step size to prevent oscillation
        eta = min(eta, 0.5)

        # Nesterov: y = a + beta * (a - a_prev)
        # Simplified: just use momentum on gradient
        v = beta * v + (1 - beta) * grad
        v_norm = float(np.linalg.norm(v))
        if v_norm < 1e-12:
            break
        eta_v = min(0.5 / v_norm, 0.3)

        a_new = xn_project(a - eta_v * v, 1.0)
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj:
            # Nesterov: update y
            y = a_new + beta * (a_new - a)
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step_scale = min(step_scale * 1.02, 2.0)
        else:
            step_scale *= 0.8

    return best_a


def alternating_lp(a, time_budget=60.0, verbose=False):
    """LP alternating with golden-section line search using HiGHS."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Golden section line search
        def phi(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            return compute_obj_normalized(a_try / s_try)

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = phi(c_pt)
        fd = phi(d_pt)

        for _ in range(25):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = phi(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = phi(d_pt)
            if hi - lo < 1e-4:
                break

        t_opt = (lo + hi) / 2.0
        a_new = (1.0 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            stall += 1
            continue
        a_new = a_new / s_new
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 100 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng((it * 7919 + n) % (2**32))
                scale = 0.03 + 0.02 * min(stall // 20, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj_normalized(a)
                stall = 0

    return best_a


def chase_peak_refine(a, time_budget=30.0, n_steps=500):
    """Chase-the-peak with adaptive step size and multi-peak awareness."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    step_size = 0.02

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Find all peaks above 90% of max
        threshold = 0.9 * max_val
        peak_indices = np.where(conv >= threshold)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(conv))])

        # Compute weighted gradient from multiple peaks
        grad = np.zeros(n)
        weights = conv[peak_indices] - 0.85 * max_val
        weights = np.maximum(weights, 0.01)
        weights = weights / np.sum(weights)

        for w, k in zip(weights, peak_indices):
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                j_vals = k - np.arange(i_min, i_max + 1)
                grad[i_min:i_max + 1] += w * 2.0 * a[j_vals]

        # Center gradient (project onto sum=0 subspace)
        grad_centered = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad_centered))
        if grad_norm < 1e-12:
            break

        # Adaptive step
        eta = step_size / grad_norm
        eta = min(eta, 0.1)

        a_new = a - eta * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            step_size *= 0.5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj_normalized(a_new)

            best_obj = obj_new
            best_a = a_new.copy()

            step_size = min(step_size * 1.05, 0.05)

def coordinate_descent_refine(a, time_budget=20.0):
    """Pairwise coordinate descent targeting peak reduction."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from high contributors to low contributors
        step = max(1, n // 200)
        for i in range(0, n, step):
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 0.02 * a[j] + 1e-6:
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if delta > a[i] * 0.5:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    obj_try = 2.0 * n * float(np.max(new_conv))
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            # Try inverse: move mass TO peak contributors from non-contributors
            for i in range(0, n, step * 2):
                if time.monotonic() - start > time_budget:
                    break
                j = k_star - i
                if 0 <= j < n and j != i:
                    for frac in [0.03, 0.05]:
                        delta = frac * a[i]
                        a[i] -= delta
                        a[j] += delta
                        new_conv = compute_conv(a)
                        obj_try = 2.0 * n * float(np.max(new_conv))
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a.copy()
                            conv = new_conv
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                            break
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def random_init(n, rng):
    """Generate random non-negative sequence with sum=1."""
    mode = rng.integers(0, 5)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / np.sum(a)
    elif mode == 2:
        # Smooth: cumulative of random
        a = np.cumsum(rng.uniform(0, 1, n))
        a = a / a[-1]
        a = a / np.sum(a)
    elif mode == 3:
        # Bimodal
        c1 = n * 0.3
        c2 = n * 0.7
        w = n * 0.15
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a += 0.1
        a = a / np.sum(a)
    else:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.3
        a = a / np.sum(a)
    return a

            break
        alloc = p2_budget / (n_rounds - round_idx) * 0.7
        alloc = min(alloc, remaining * 0.7)

        a_opt = alternating_lp(a, time_budget=alloc, verbose=True)
        obj = compute_obj_normalized(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = len(a)
            print(f"  LP round {round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {round_idx}: obj={obj:.8f}")

        # Chase peak between LP rounds
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cp = chase_peak_refine(a, time_budget=min(30.0, remaining * 0.1))
            obj_cp = compute_obj_normalized(a_cp)
            if obj_cp < best_score:
                best_score = obj_cp
                best_seq = a_cp.copy()
                a = a_cp.copy()
                print(f"  Chase peak: obj={best_score:.8f} **")
            else:
                print(f"  Chase peak: obj={obj_cp:.8f}")

    # Phase 3: Multi-n exploration
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best,
        min(1500, n_best + 50),
        min(1500, n_best + 100),
        600, 800, 1000, 1200
    ]))
    n_options = [n for n in n_options if n_best // 2 <= n <= min(2000, n_best * 2)]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: soft-max then LP
        alloc_sm = min(20.0, remaining * 0.1)
        alloc_lp = min(40.0, remaining * 0.2)

        a_sm = softmax_gradient_descent(a_try, time_budget=alloc_sm,
                                        alpha_start=50.0, alpha_end=500.0,
                                        n_max_steps=1000)
        a_opt = alternating_lp(a_sm, time_budget=alloc_lp)
        obj_try = compute_obj_normalized(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = len(a)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 4: Random restarts
    print(f"\n=== Phase 4: Random Restarts ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        n_r = n_best + int(rng.integers(-50, 50))
        n_r = max(300, min(1500, n_r))

        if r % 3 == 0:
            a_r = random_init(n_r, rng)
        elif r % 3 == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.08, len(best_seq)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        else:
            a_r = resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)

        alloc = min(50.0, remaining * 0.25)
        a_sm = softmax_gradient_descent(a_r, time_budget=alloc * 0.4,
                                        alpha_start=30.0, alpha_end=500.0,
                                        n_max_steps=1000)
        a_opt = alternating_lp(a_sm, time_budget=alloc * 0.5)
        obj_r = compute_obj_normalized(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = len(a)
            print(f"  Restart {r} (n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}): obj={obj_r:.8f}")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a = best_seq.copy()

        # LP
        alloc = min(100.0, remaining * 0.3)
        a_opt = alternating_lp(a, time_budget=alloc, verbose=True)
        obj = compute_obj_normalized(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj:.8f}")

        # Chase peak
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cp = chase_peak_refine(a, time_budget=min(40.0, remaining * 0.25))
            obj_cp = compute_obj_normalized(a_cp)
            if obj_cp < best_score:
                best_score = obj_cp
                best_seq = a_cp.copy()
                a = a_cp.copy()
                print(f"  Final chase: obj={best_score:.8f} **")
            else:
                print(f"  Final chase: obj={obj_cp:.8f}")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_cd = coordinate_descent_refine(a, time_budget=min(30.0, remaining * 0.3))
            obj_cd = compute_obj_normalized(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f} **")
            else:
                print(f"  Final CD: obj={obj_cd:.8f}")

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_opt = alternating_lp(a, time_budget=min(40.0, remaining * 0.5))
            obj = compute_obj_normalized(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final LP2: obj={best_score:.8f} **")
            else:
                print(f"  Final LP2: obj={obj:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0 / 100.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
