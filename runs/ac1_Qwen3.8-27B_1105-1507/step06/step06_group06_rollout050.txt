"""
Adaptive spectral-flattening and targeted subgradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines: (1) adaptive power-spectrum flattening, (2) top-peak-weighted subgradient

refinement, (5) structured diversification via spectral perturbation.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_fft(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(a_new)


def build_toeplitz_conv(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_direction(a, M_scale=1.0, time_limit=15):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M*M_scale, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    if n > 1500:
        return None
    conv = compute_conv_fft(a)
    M = float(np.max(conv)) * M_scale

    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def spectral_perturb(a, rng, strength=0.1):
    """
    Perturb in the frequency domain for diversity:
    rotate phases slightly and add structured noise to magnitudes.
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    F = np.fft.fft(a, N)
    mag = np.abs(F)

    # Add small phase rotation
    phase += rng.normal(0, strength * 0.5, N)
    # Modify magnitudes with structured noise

    noise = np.maximum(noise, 0.1)
    mag *= noise

    F_new = mag * np.exp(1j * phase)
    a_new = np.fft.ifft(F_new).real[:n]
    a_new = project_simplex(a_new)
    return a_new

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian centered
    x = np.linspace(-2.5, 2.5, n)
    g = np.exp(-x ** 2)
    inits.append(g / g.sum())

    # Sinc-like (good for flat spectrum)
    x = np.linspace(-4, 4, n)
    sinc = np.sinc(x / 4)
    sinc = np.abs(sinc) + 0.01
    inits.append(sinc / sinc.sum())

    # Raised cosine
    x = np.linspace(0, 2 * np.pi, n)
    rc = 0.5 * (1 + np.cos(x))
    rc = rc + 0.05
    inits.append(rc / rc.sum())

    # Random
    r = rng.random(n) + 0.01
    inits.append(r / r.sum())

    # Two-hump
    x = np.linspace(0, 1, n)
    two = np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01)
    inits.append(two / two.sum())

    # Cosine window
    x = np.linspace(0, 1, n)
    cw = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    inits.append(cw / cw.sum())

    # Parent resampled
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        p = p * np.exp(rng.normal(0, 0.15, n))
        inits.append(project_simplex(p))

    return inits


def multi_resolution_optimize(a, time_budget=100.0):
    """
    Coarse-to-fine optimization: downsample, optimize, upsample, refine.
    """
    n = len(a)
    best_a = project_simplex(a.copy())
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    # Downsample to coarser resolution
    n_coarse = max(100, n // 4)
    a_coarse = resample_seq(best_a, n_coarse)

    # Optimize at coarse level
    remain = time_budget - (time.monotonic() - start)
    a_c, obj_c = adaptive_subgradient(a_coarse, time_budget=remain * 0.3)
    a_c2, obj_c2 = spectral_flatten(a_c, alpha_init=0.6, time_budget=remain * 0.2)
    if obj_c2 < obj_c:
        a_c, obj_c = a_c2, obj_c2

    # Upsample back
    a_up = resample_seq(a_c, n)
    a_up, obj_up = adaptive_subgradient(a_up, time_budget=time_budget * 0.3)
    if obj_up < best_obj:
        best_a, best_obj = a_up, obj_up

    # Final spectral flatten at full resolution
    remain = time_budget - (time.monotonic() - start)
    if remain > 3:
        a_f, obj_f = spectral_flatten(best_a, alpha_init=0.75, time_budget=remain * 0.5)
        if obj_f < best_obj:
            best_a, best_obj = a_f, obj_f

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def hybrid_optimize(a, time_budget=100.0, n_max_lp=1000):
    """
    Run multiple optimization strategies in sequence, keeping the best.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    strategies = []

    # Spectral flatten
    remain = time_budget - (time.monotonic() - start)
    if remain > 5:
        a1, obj1 = spectral_flatten(best_a, alpha_init=0.65, n_iter=80,
                                     time_budget=min(remain * 0.3, 25.0))
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
        strategies.append(("flatten", best_obj))

    # Adaptive subgradient
    remain = time_budget - (time.monotonic() - start)
    if remain > 5:
        a2, obj2 = adaptive_subgradient(best_a, time_budget=min(remain * 0.3, 30.0))
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
        strategies.append(("subgrad", best_obj))

    # Spectral flatten v2
    remain = time_budget - (time.monotonic() - start)
    if remain > 5:
        a3, obj3 = spectral_flatten_v2(best_a, n_iter=80,
                                        time_budget=min(remain * 0.25, 20.0))
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
        strategies.append(("flatten2", best_obj))

    # LP alternating (if n is small enough)
    if n <= n_max_lp:
        remain = time_budget - (time.monotonic() - start)
        if remain > 10:
            a4, obj4 = lp_alternate(best_a, max_iters=20,
                                     time_budget=min(remain * 0.4, 35.0))
            if obj4 < best_obj:
                best_obj = obj4
                best_a = a4.copy()
            strategies.append(("lp", best_obj))

    # Second round of subgradient (finer)
    remain = time_budget - (time.monotonic() - start)
    if remain > 5:
        a5, obj5 = top_peak_subgradient(best_a, n_steps=200, time_budget=min(remain * 0.5, 25.0),
                                         n_top=5, step_init=0.015, momentum=0.8, decay=0.9995)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()
        strategies.append(("subgrad_fine", best_obj))

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining spectral flattening, subgradient, LP,
    and multi-resolution optimization for minimizing the self-convolution peak.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent loaded: n={parent.size}, obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()
        print(f"No parent. Starting with uniform n=500, obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # === Phase 1: Multi-length exploration with diverse inits (15% time) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    phase1_budget = min(150.0, (deadline - start_time) * 0.15)
    phase1_start = time.monotonic()

    # Try various lengths around the parent
    n_list = sorted(set([
        max(200, n_parent - 100),
        max(200, n_parent - 50),
        n_parent,
        n_parent + 50,
        n_parent + 100,
        n_parent + 200,
        n_parent + 400,
        min(3000, n_parent + 800),
    ]))

    candidates = []  # (obj, a, n)
    per_n_budget = max(8.0, phase1_budget / len(n_list))

    for n_try in n_list:
        if time.monotonic() > phase1_start + phase1_budget:
            break

        inits = generate_inits(n_try, rng, parent)
        per_init_time = max(2.0, per_n_budget / len(inits) * 0.6)

        for init in inits:
            if time.monotonic() > phase1_start + phase1_budget:
                break
            # Quick spectral flatten + subgradient
            a_quick, obj_quick = spectral_flatten(init, alpha_init=0.7,
                                                   n_iter=30, time_budget=per_init_time * 0.5)
            a_quick2, obj_quick2 = top_peak_subgradient(a_quick, n_steps=50,
                                                         time_budget=per_init_time * 0.5,
                                                         n_top=2, step_init=0.05)
            if obj_quick2 < obj_quick:
                a_quick, obj_quick = a_quick2, obj_quick2

            if obj_quick < np.inf:
                candidates.append((obj_quick, a_quick, n_try))

    candidates.sort(key=lambda x: x[0])
    print(f"  Candidates: {len(candidates)}, best: {candidates[0][0] if candidates else 'N/A'}")

    for obj_c, a_c, n_c in candidates[:5]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  New best from Phase 1: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive optimization on top candidates (45% time) ===
    print("\n=== Phase 2: Intensive Optimization ===")
    phase2_budget = min((deadline - start_time) * 0.45, 700.0)
    phase2_start = time.monotonic()
    print(f"  Budget: {phase2_budget:.0f}s, best so far: {best_score:.8f}")

    # Focus on best candidate and a few others
    top_list = [(best_score, best_seq.copy(), len(best_seq))]
    for obj_c, a_c, n_c in candidates[:4]:
        if abs(obj_c - best_score) > 1e-6 or n_c != len(best_seq):
            top_list.append((obj_c, a_c.copy(), n_c))

    per_candidate_budget = phase2_budget / max(1, len(top_list))

    for ci, (obj_init, a_init, n_init) in enumerate(top_list):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        alloc = min(per_candidate_budget, phase2_start + phase2_budget - time.monotonic())
        if alloc < 10:
            break

        print(f"  Candidate {ci}: n={n_init}, init_obj={obj_init:.8f}, budget={alloc:.0f}s")
        a_opt, obj_opt = hybrid_optimize(a_init, time_budget=alloc, n_max_lp=1000)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n_init}, obj={best_score:.8f} **")
        else:
            print(f"  Candidate {ci} final: obj={obj_opt:.8f} (no improvement)")

    # === Phase 3: Diversification and re-optimization (20% time) ===
    print(f"\n=== Phase 3: Diversification ===")
    print(f"Best so far: {best_score:.8f}")
    phase3_budget = min((deadline - time.monotonic()) * 0.20, 300.0)
    phase3_start = time.monotonic()
    n_best = len(best_seq)

    n_diversify = min(8, max(3, int(phase3_budget / 25)))
    per_div_budget = max(10.0, phase3_budget / n_diversify)

    for di in range(n_diversify):
        if time.monotonic() > phase3_start + phase3_budget:
            break
        allocate = min(per_div_budget, phase3_start + phase3_budget - time.monotonic())
        if allocate < 8:
            break

        # Create diverse candidate via spectral perturbation
        strength = 0.1 + 0.1 * (di / n_diversify)
        a_div = spectral_perturb(best_seq, rng, strength=strength)

        # Also try different length
        if di % 2 == 1:
            n_alt = n_best + rng.integers(-50, 100)
            n_alt = max(200, min(2500, n_alt))
            a_div = resample_seq(a_div, n_alt)

        # Optimize
        a_div_opt, obj_div = hybrid_optimize(a_div, time_budget=allocate * 0.8, n_max_lp=1000)

        if obj_div < best_score:
            best_score = obj_div
            best_seq = a_div_opt.copy()
            print(f"  Diversify {di}: NEW BEST obj={best_score:.8f}")
        else:
            print(f"  Diversify {di}: obj={obj_div:.8f}")

    # === Phase 4: Multi-resolution refinement (10% time) ===
    print(f"\n=== Phase 4: Multi-resolution Refinement ===")
    print(f"Best so far: {best_score:.8f}")
    phase4_budget = min((deadline - time.monotonic()) * 0.10, 120.0)
    phase4_start = time.monotonic()

    if phase4_budget > 10:
        a_mr, obj_mr = multi_resolution_optimize(best_seq, time_budget=phase4_budget * 0.6)
        if obj_mr < best_score:
            best_score = obj_mr
            best_seq = a_mr.copy()
            print(f"  Multi-res: NEW BEST obj={best_score:.8f}")

        # Also try from a fresh start at the best length
        remain4 = phase4_start + phase4_budget - time.monotonic()
        if remain4 > 10:
            n_best_now = len(best_seq)
            a_fresh = generate_inits(n_best_now, rng, best_seq)[0]  # uniform
            a_fresh_opt, obj_fresh = hybrid_optimize(a_fresh, time_budget=remain4 * 0.5, n_max_lp=1000)
            if obj_fresh < best_score:
                best_score = obj_fresh
                best_seq = a_fresh_opt.copy()
                print(f"  Fresh start: NEW BEST obj={best_score:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    n_final = len(best_seq)

    if remaining5 > 10:
        # Repeated rounds of the most effective strategies
        polish_rounds = max(1, int(remaining5 / 20))
        for pr in range(polish_rounds):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break

            # Fine subgradient
            a_p1, obj_p1 = top_peak_subgradient(best_seq, n_steps=min(200, int(rem)),
                                                 time_budget=rem * 0.25,
                                                 n_top=3, step_init=0.01,
                                                 momentum=0.85, decay=0.9995)
            if obj_p1 < best_score:
                best_score = obj_p1
                best_seq = a_p1.copy()
                print(f"  Polish r{pr} subgrad: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 5:
                break

            # Spectral flatten (mild)
            a_p2, obj_p2 = spectral_flatten(best_seq, alpha_init=0.85,
                                             n_iter=50, time_budget=rem * 0.2)
            if obj_p2 < best_score:
                best_score = obj_p2
                best_seq = a_p2.copy()
                print(f"  Polish r{pr} spectral: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 10:
                break

            # LP (if applicable)
            if n_final <= 1200:
                a_p3, obj_p3 = lp_alternate(best_seq, max_iters=10,
                                             time_budget=min(rem * 0.3, 20.0))
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    n_final = len(best_seq)
                    print(f"  Polish r{pr} LP: {best_score:.8f}")

            rem = deadline - time.monotonic()
            if rem < 10:
                break

            # Small perturbation + re-optimize
            a_pert = spectral_perturb(best_seq, rng, strength=0.03)
            a_pert_opt, obj_pert = hybrid_optimize(a_pert, time_budget=rem * 0.3, n_max_lp=1000)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert_opt.copy()
                n_final = len(best_seq)
                print(f"  Polish r{pr} perturb: {best_score:.8f}")

    # Final validation using the actual evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        print(f"\nValidation: evaluate_sequence gives {val:.8f}")
        if val > best_score:
            # Mismatch (shouldn't happen but be safe)
            best_score = val
    else:
        print("Validation failed, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
