def compute_conv(a):
    """FFT-based linear convolution."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):

    n = len(a)
    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Palindromic and free Adam in log-space with adaptive temperature
2. Exact subgradient polishing
3. (1+1)-ES evolutionary exploration
4. Multi-n search with resampling
5. Coordinate descent refinement

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_grad(a, beta):
    """
    Compute smoothed objective (logsumexp of beta*conv / sum_a^2 * 2n)
    and its gradient w.r.t. a using FFT.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    # Gradient computation via FFT
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    # T1 = cross-correlation: ifft(FFT(a) * conj(FFT(w)))
    T1 = irfft(A * np.conj(W), size)
    # T2 = convolution: ifft(FFT(w) * FFT(a))
    T2 = irfft(W * A, size)
    grad_ls = (T1[:n] + T2[:n]) / beta

    # Full gradient of obj = 2n*ls/S^2
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def subgradient_step(a, lr):
    """
    One subgradient step using exact subgradient at the argmax of conv.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    max_b = float(np.max(conv))
    # Find all indices where conv is at max (within tolerance)
    tol = 1e-10
    max_indices = np.where(np.abs(conv - max_b) < tol * max_b)[0]
    # Subgradient: w is uniform over max indices
    w = np.zeros(2 * n - 1)
    w[max_indices] = 1.0 / len(max_indices)

    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    T1 = irfft(A * np.conj(W), size)
    T2 = irfft(W * A, size)
    grad_ls = T1[:n] + T2[:n]  # without 1/beta since we use exact subgradient

    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - max_b / sum_a)

    a_new = a - lr * grad_obj
    return np.maximum(a_new, 1e-15)

    Optionally enforces palindromic symmetry.

    a = a.copy().astype(np.float64)

        return a

        def half_to_full(x):

            x_c = x - np.max(x)
            ah = np.exp(x_c)
            ah = ah / np.sum(ah)
            # Mirror to full

            # Handle odd case

            return full

        x = np.log(a)
        x -= np.mean(x)

        def half_to_full(x):

def es_explore(a, time_budget=30.0, seed=0):
    """
    (1+1) Evolution Strategy with adaptive step size.
    Explores the neighborhood of a using random perturbations.
    """

    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    sigma = 0.05  # initial step size

    log_a = np.log(np.maximum(a, 1e-15))

    while time.monotonic() - start < time_budget:
        # Generate mutant
        noise = rng.normal(0, sigma, n)
        log_m = log_a + noise
        m = np.exp(log_m - np.max(log_m))
        m = m / np.sum(m)

        obj_m = compute_obj(m)

        if obj_m < best_obj:
            # Success: increase or maintain sigma
            best_a = m.copy()
            best_obj = obj_m
            log_a = log_m.copy()
            sigma *= 1.05
            print(f"    ES: obj={best_obj:.8f}, sigma={sigma:.4f}")
        else:
            # Failure: decrease sigma
            sigma *= 0.95

        if sigma < 0.001:
            break

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def palindromize(a):
    """Project a sequence to its palindromic part."""
    n = len(a)
    half = (n + 1) // 2
    p = np.zeros(n)
    for i in range(half):
        p[i] = (a[i] + a[n - 1 - i]) / 2.0
        p[n - 1 - i] = p[i]
    s = np.sum(p)
    if s > 1e-10:
        p = p / s
    return p

                cos_seq = 0.5 * (1 + np.cos(np.pi * np.arange(n_best) / (n_best - 1)))
                cos_seq = cos_seq / np.sum(cos_seq)
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * cos_seq
            else:
                # Blend with flat + noise
                alpha = rng.uniform(0.05, 0.15)
                noise = 1.0 + rng.normal(0, 0.1, n_best)
                noise = np.maximum(noise, 0.1)
                noise = noise / np.sum(noise)
                a_pert = (1 - alpha) * (best_seq / np.sum(best_seq)) + alpha * noise

            a_pert = np.maximum(a_pert, 1e-12)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            alloc = min(time_left() * 0.12, 25.0)
            a_opt = adam_log_opt(a_pert, max_iters=10000, time_budget=alloc,
                                 beta_start=20.0, beta_end=500.0,
                                 lr_start=0.025, lr_decay=0.99999)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n{'='*50}")
    print(f"Phase 8: Final polish")
    print(f"{'='*50}")

    if best_seq is not None and time_left() > 10:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # High-beta Adam
        alloc = min(time_left() * 0.3, 30.0)
        a_final = adam_log_opt(a, max_iters=20000, time_budget=alloc,
                               beta_start=500.0, beta_end=2000.0,
                               lr_start=0.005, lr_decay=0.99999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final high-beta Adam: {best_score:.8f} ***")

        # Subgradient on the final result
        if time_left() > 10:
            a_sg = subgradient_step(best_seq, 0.05 / len(best_seq))
            a_sg = np.maximum(a_sg, 1e-15)
            s_sg = np.sum(a_sg)
            if s_sg > 1e-10:
                a_sg = a_sg / s_sg
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    print(f"  Final SG: {best_score:.8f} ***")

        # One more coordinate descent
        if time_left() > 5:
            a_cd = coordinate_refine(best_seq, time_budget=min(time_left() * 0.4, 15.0),
                                     seed=seed + 7777)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: {best_score:.8f} ***")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
