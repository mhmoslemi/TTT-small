            continue
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            gv = np.maximum(res.x[:n], 0)
            s = float(np.sum(gv))
            if s > 1e-10:
                return gv / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M_target):
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_target]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                s = float(np.sum(gv))
                if s > 1e-10:
                    return gv / s
        except Exception:
            continue
    return None


def solve_lp_weighted(a, weights):
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1, g >= 0]
    obj = cp.sum(cp.multiply(weights, A @ g))
    prob = cp.Problem(cp.Minimize(obj), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                s = float(np.sum(gv))
                if s > 1e-10:
                    return gv / s
        except Exception:
            continue
    return None


def line_search(a, g, current_obj, max_t=1.0):
    best_t = 0.0
    best_obj = current_obj
    best_a = a.copy()
    ts = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()
    if best_t == 0.0:
        a_try = (1.0 - 0.05) * a + 0.05 * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = 0.05
                best_a = a_try.copy()
    return best_a, best_obj, best_t


def alternating_lp(a, time_budget=300.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 1.0

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fast(a)
        M_cur = float(np.max(conv))

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is not None:
            pass
        else:
            g_max = solve_lp_maxsum(a, M_cur)
            if g_max is None:
                break
            g_lp = g_max

        a_new, obj_new, t_used = line_search(a, g_lp, best_obj)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            last_t = t_used
            stall = 0
            if verbose and it % 50 == 0:
                print(f"    LP-alt it{it}: obj={best_obj:.8f}, t={t_used:.3f}")
        else:
            stall += 1
            if stall > 15:
                # Try weighted LP
                conv = compute_conv_fast(a)
                k_peaks = np.argsort(conv)[-10:]
                weights = np.zeros(2 * n - 1)
                for k in k_peaks:
                    weights[int(k)] = 1.0
                g_w = solve_lp_weighted(a, weights)
                if g_w is not None:
                    a_new, obj_new, t_used = line_search(a, g_w, best_obj)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        stall = 0
                        continue

                # Perturb
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                scale = 0.03 + 0.03 * min(stall // 15, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0
            elif stall > 5:
                # Try smaller step
                a_try = (1.0 - 0.05) * a + 0.05 * g_lp
                a_try = np.maximum(a_try, 0)
                s = float(np.sum(a_try))
                if s > 1e-10:
                    a_try = a_try / s
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0

    return best_a


def subgradient_refine(a, time_budget=60.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.05

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv_fast(a)
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
        grad -= np.mean(grad)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                eta *= 0.95
        else:
            eta *= 0.5

    return best_a


def symmetric_refine(a, time_budget=60.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a_sym = (a + a[::-1]) / 2.0
    s = float(np.sum(a_sym))
    if s < 1e-10:
        return a
    a_sym = a_sym / s
    best_a = a_sym.copy()
    best_obj = compute_obj(a_sym)
    start = time.monotonic()

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        g_lp, _ = solve_lp_minpeak(a_sym)
        if g_lp is None:
            break
        g_sym = (g_lp + g_lp[::-1]) / 2.0
        gs = float(np.sum(g_sym))
        if gs < 1e-10:
            break
        g_sym = g_sym / gs
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a_sym + t * g_sym
            a_try = (a_try + a_try[::-1]) / 2.0
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a_sym = a_try
                break

    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    old_t = np.linspace(0, 1, n)
    new_t = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(new_t, old_t, a), 0)


def random_init(n, rng, mode=0):
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        return a / np.sum(a)
    elif mode == 2:
        center = n // 2
        sigma = n / 6.0
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        return a / np.sum(a)
    elif mode == 3:
        a = np.random.uniform(0, 1, n // 2)
        a = np.concatenate([a, a[::-1]])
        if len(a) < n:
            a = np.pad(a, (0, n - len(a)))
        elif len(a) > n:
            a = a[:n]
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    else:
        a = rng.exponential(1.0, n)
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_seq = parent.copy()
                best_score = compute_obj(parent)
                print(f"Parent n={parent.size}: obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_best = len(best_seq)

    # Phase 1: Intensive LP alternating on parent
    print(f"\n=== Phase 1: LP Alternating (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 450.0)

    a = best_seq.copy()
    for rnd in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p1_budget / (4 - rnd), remaining * 0.7)
        a_new = alternating_lp(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Round {rnd}: obj={best_score:.8f}")
        else:
            print(f"  Round {rnd}: obj={obj:.8f} (no improvement)")

    # Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sg = subgradient_refine(a, time_budget=min(remaining * 0.08, 30.0))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f}")

    # Phase 2: Symmetric optimization
    print(f"\n=== Phase 2: Symmetric ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_sym = symmetric_refine(a, time_budget=min(remaining * 0.12, 90.0))
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            a = a_sym.copy()
            print(f"  Symmetric: obj={best_score:.8f}")
        else:
            print(f"  Symmetric: obj={obj_sym:.8f}")

    # Phase 3: Multi-n exploration
    print(f"\n=== Phase 3: Multi-n ===")
    n_opts = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200,
        600, 800, 1000, 1200, 1500
    ]))
    n_opts = [x for x in n_opts if 200 <= x <= 2000]

    n_visited = set()
    for n_try in n_opts:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try in n_visited:
            continue
        n_visited.add(n_try)

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample(best_seq, n_try)

        alloc = min(remaining * 0.15, 60.0)
        a_opt = alternating_lp(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = len(a_opt)
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 4: Random restarts
    print(f"\n=== Phase 4: Restarts ===")
    remaining = deadline - time.monotonic()
    restart_n = len(best_seq)

    for r in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        mode = r % 5
        if mode == 0:
            a_r = random_init(restart_n, rng, 1)
        elif mode == 1:
            a_r = best_seq * np.exp(rng.normal(0, 0.08, len(best_seq)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif mode == 2:
            n_r = int(restart_n + rng.integers(-150, 150))
            n_r = max(300, min(2000, n_r))
            a_r = resample(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / np.sum(a_r)
        elif mode == 3:
            a_r = random_init(restart_n, rng, 3)
        else:
            a_r = random_init(restart_n, rng, 2)

        alloc = min(remaining * 0.12, 50.0)
        a_opt = alternating_lp(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (m={mode}, n={len(a_opt)}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (m={mode}): obj={obj_r:.8f}")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    a = best_seq.copy()

    if remaining > 30:
        alloc = min(remaining * 0.35, 200.0)
        a1 = alternating_lp(a, time_budget=alloc, verbose=True)
        o1 = compute_obj(a1)
        if o1 < best_score:
            best_score = o1
            best_seq = a1.copy()
            a = a1
            print(f"  Final LP: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a2 = subgradient_refine(a, time_budget=min(remaining * 0.2, 40.0))
        o2 = compute_obj(a2)
        if o2 < best_score:
            best_score = o2
            best_seq = a2.copy()
            a = a2
            print(f"  Final Subgrad: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a3 = symmetric_refine(a, time_budget=min(remaining * 0.25, 40.0))
        o3 = compute_obj(a3)
        if o3 < best_score:
            best_score = o3
            best_seq = a3.copy()
            print(f"  Final Sym: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a4 = alternating_lp(best_seq.copy(), time_budget=min(remaining * 0.5, 60.0))
        o4 = compute_obj(a4)
        if o4 < best_score:
            best_score = o4
            best_seq = a4.copy()
            print(f"  Final LP2: obj={best_score:.8f}")

    # Validate
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result

4. **More aggressive multi-n exploration** (up to 2000)
5. **More restarts** (12 vs 8) with diverse initialization
6. **Better time allocation** across 5 phases with adaptive budgets
7. **Subgradient refinement** interleaved with LP phases for local polish
