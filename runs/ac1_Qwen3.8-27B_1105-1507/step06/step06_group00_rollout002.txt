OK so now, the new objective when a_i is set to x (with everything else fixed, and we'll normalize at the end):

new_sum = 1 - a_i + x (assuming sum was 1)
new_conv_k = conv_old_k + change_k(x)
new_obj = 2n * max_k new_conv_k / new_sum^2

This is a piecewise function of x (piecewise because the argmax can change). For a fixed argmax k*, it's a quadratic in x:
new_obj(x) = 2n * (conv_old_{k*} + c*{k*}*x + d*{k*}*x^2) / (1-a_i+x)^2

where c*{k*} and d*{k*} are the linear and quadratic coefficients.

To minimize, we can:
1. Find the current argmax k*.
2. Find the minimum of new_obj(x) assuming argmax stays at k* (solve the 1D quadratic/linear fractional optimization).
3. Check if the argmax actually stays at k* for the optimal x.
4. If not, add that k to the active set and redo.

This is essentially the same as the LP approach but in 1D! It should be fast.

Let me implement this coordinate descent approach. It should be a good complement to the LP.

Now let me also think about the LP approach. The LP in the previous program solves:
min M subject to A@g <= M, sum(g) = 1, g >= 0

where A is the convolution matrix (A[j, i] = a_{j-i} if valid, 0 otherwise... actually A @ g = conv(a, g)).

This is correct: conv(a, g)_k = sum_j a_j * g_{k-j} = sum_i (sum_j a_j * delta_{k-j, i}) * g_i.

Hmm, let me re-derive. conv(a, g)_k = sum_j a_j * g_{k-j} (j, k-j in range).
= sum_i g_i * a_{k-i} (letting i = k-j, so j = k-i).

So A[k, i] = a_{k-i} if 0 <= k-i < n, else 0.

The LP is: min M s.t. A@g <= M (componentwise), 1^T g = 1, g >= 0.

This finds the g that minimizes the peak of conv(a, g), which is the best "response" to a. This is correct for the alternating method.

One improvement: instead of minimizing the peak, we could minimize a weighted sum of the top-k peaks, which might give a better direction.

Another improvement: solve the LP with a barrier or use the dual to get information about which constraints are active.

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating, coordinate descent with 1D optimization,
soft subgradient, and population-based search.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import itertools

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    # conv(a,g)_k = sum_i a_{k-i} * g_i where 0 <= k-i < n
    # So A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # a[k-i] is nonzero when 0 <= k-i < n, i.e., i <= k < i+n
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = a[:k_end - k_start]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Try CVXPY first
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except:
                continue
    except:
        pass
    
    # Fallback to scipy linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    # A_ub @ [g, M] <= b_ub: conv(a,g) - M <= 0 => A@g - M <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    
    return None, None

def solve_lp_weighted(a, weights):
    """Solve LP: min sum(weights * conv(a,g)) s.t. sum(g)=1, g>=0, conv(a,g) <= M_cur."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    conv_a = compute_conv(a)
    M_cur = float(np.max(conv_a))
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        obj = weights @ (A @ g)  # weighted sum of conv values
        # Actually minimize the max with weighting... let's just do min M with a soft constraint
        # Or: minimize weighted_sum subject to conv <= 1.5 * M_cur
        constraints = [A @ g <= 1.1 * M_cur, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(weights, A @ g))), constraints)
        for solver in [cp.MOSEK, cp.SCS]:

    except:
        pass
    return None

def coordinate_descent_1d(a, time_budget=30.0, verbose=False):

def soft_subgradient_descent(a, time_budget=30.0, temperature=0.01, lr_init=0.01, verbose=False):
    """
    Soft subgradient descent using softmax weighting of peaks.
    Instead of targeting just the max, uses a softmax over all peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    
    max_iters = 2000
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        if max_conv < 1e-15:
            break
        
        # Softmax weights over the convolution values
        # w_k = exp(conv_k / (T * max_conv)) / sum(exp(conv_j / (T * max_conv)))
        scaled = conv / (temperature * max_conv)
        # Subtract max for numerical stability
        scaled = scaled - np.max(scaled)
        weights = np.exp(scaled)
        weights = weights / np.sum(weights)
        
        # Gradient: dw_k/da_i = 2*weights_k * a_{k-i} (for 0<=k-i<n) + softmax correction
        # Simplified: just use the weighted subgradient
        grad = np.zeros(n)
        for i in range(n):
            # conv_k = sum_j a_j * a_{k-j}
            # d(conv_k)/da_i = 2*a_{k-i} (if 0<=k-i<n)
            k_start = i
            k_end = min(i + n, 2*n-1)
            grad[i] = 2.0 * np.dot(weights[k_start:k_end], a[:k_end-k_start])
        
        # Project gradient to be orthogonal to 1 (maintain sum)
        grad = grad - np.mean(grad)
        
        # Adaptive step size
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                lr = min(lr * 1.05, 0.05)
            else:
                stall += 1
                if stall > 20:
                    lr *= 0.7
                    stall = 0
                if lr < 1e-6:
                    break
        else:
            lr *= 0.5
    
    return best_a

def multi_peak_lp(a, time_budget=30.0, verbose=False):
    """
    LP that minimizes a weighted combination of top peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Create weights: emphasize top peaks
        sorted_idx = np.argsort(conv)[::-1]
        top_k = min(20, 2*n-1)
        weights = np.zeros(2*n-1)
        for rank, idx in enumerate(sorted_idx[:top_k]):
            weights[idx] = 1.0 / (rank + 1)
        
        # Solve LP: min weighted_sum(conv(a,g)) s.t. conv(a,g) <= 1.05*max_conv, sum(g)=1, g>=0
        g = solve_lp_weighted_2(a, weights, max_conv)
        if g is None:
            break
        
        # Blend
        best_imp = False
        for t in [1.0, 0.7, 0.5, 0.3]:
            a_try = (1-t)*a + t*g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                best_imp = True
                stall = 0
                break
        
        if not best_imp:
            stall += 1
            if stall > 20:
                break
    
    return best_a

def solve_lp_weighted_2(a, weights, M_bound):
    """Solve LP: min weights^T (A@g) s.t. A@g <= M_bound, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M_bound, cp.sum(g) == 1]
        obj = cp.sum(cp.multiply(weights, A @ g))
        prob = cp.Problem(cp.Minimize(obj), constraints)
        for solver in [cp.MOSEK, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except:
                continue
    except:
        pass
    
    # Fallback: scipy
    c = weights @ A  # objective: c^T g
    A_ub = A.copy()
    b_ub = np.full(2*n-1, M_bound)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    
    return None

def block_coordinate_descent(a, time_budget=30.0, block_size=4):
    """
    Block coordinate descent: optimize a block of consecutive coordinates at once.
    Uses GridSearch within each block.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for round_idx in range(30):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for start_idx in range(0, n, block_size):
            if time.monotonic() - start > time_budget:
                break
            end_idx = min(start_idx + block_size, n)
            block = a[start_idx:end_idx].copy()
            block_sum = float(np.sum(block))
            
            # Try redistributing mass within the block
            # Keep the total sum of the block roughly constant
            rng = np.random.default_rng(hash((round_idx, start_idx, n)) % (2**32))
            
            best_block = block
            best_block_obj = best_obj
            
            for trial in range(5):
                # Random perturbation of the block
                perturb = np.exp(rng.normal(0, 0.1, block.size))
                new_block = block * perturb
                # Renormalize block to same sum
                s_new_block = float(np.sum(new_block))
                if s_new_block > 1e-10:
                    new_block = new_block * block_sum / s_new_block
                else:
                    continue
                
                a_trial = a.copy()
                a_trial[start_idx:end_idx] = new_block
                s_total = float(np.sum(a_trial))
                if s_total < 0.01:
                    continue
                a_trial = a_trial / s_total
                obj_trial = compute_obj(a_trial)
                
                if obj_trial < best_block_obj - 1e-12:
                    best_block_obj = obj_trial
                    best_block = new_block.copy()
                    best_block_a = a_trial.copy()
            
            if float(best_block_obj) < best_obj - 1e-12:
                a = best_block_a
                best_obj = float(best_block_obj)
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, rng):
    """Generate a random non-negative sequence."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = np.sum(a)
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    elif mode == 2:
        # Low-frequency dominant (smooth)
        a = np.ones(n) / n
        for freq in range(1, 5):
            a += 0.3 * rng.uniform(-1, 1) * np.cos(2*np.pi*freq*np.arange(n)/n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
    elif mode == 3:
        # Symmetric random
        half = (n+1)//2
        h = rng.uniform(0, 1, half)
        a = np.zeros(n)
        a[:half] = h
        a[half:] = h[:n-half][::-1]
        s = a.sum()
        if s > 0:
            a = a / s
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x/2) ** 2
        a = a / a.sum()
    else:
        # Two-peak
        c1 = n // 3
        c2 = 2*n // 3
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n)-c1)**2)/(2*w**2)) + np.exp(-((np.arange(n)-c2)**2)/(2*w**2))
        a = a / a.sum()
    return a
