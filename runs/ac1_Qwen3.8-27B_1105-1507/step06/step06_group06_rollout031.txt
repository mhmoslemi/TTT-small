"""
Optimized step function search using spectral shaping, adaptive subgradient,
and multi-peak water-filling to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize to sum 1 and non-negative."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def spectral_shaping(a, max_iters=200, time_budget=30, alpha_schedule=None):
    """
    Spectral shaping: iteratively modify the power spectrum to reduce
    the peak of the time-domain convolution.
    
    Key idea: compress the dynamic range of |A(f)|^2, which flattens
    the time-domain convolution.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        if alpha_schedule is not None:
            alpha = alpha_schedule(it, max_iters)
        else:
            alpha = 0.5 + 0.3 * it / max_iters
        
        A = fft(a, N)
        power = np.abs(A) ** 2
        p_max = np.max(power)
        if p_max < 1e-30:
            break
        
        # Compress the power spectrum
        power_norm = power / p_max
        power_comp = power_norm ** alpha
        
        # Reconstruct
        phase = np.angle(A)
        mag = np.sqrt(power_comp * p_max)
        A_new = mag * np.exp(1j * phase)
        
        a_new = ifft(A_new).real[:n]
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def spectral_shaping_v2(a, max_iters=300, time_budget=40):
    """
    Improved spectral shaping with adaptive alpha and peak-targeting.
    Instead of uniform compression, target the specific frequency bins
    that contribute to the time-domain peaks.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        c_max = np.max(conv)
        k_peak = int(np.argmax(conv))
        
        A = fft(a, N)
        power = np.abs(A) ** 2
        p_max = np.max(power)
        if p_max < 1e-30:
            break
        
        # The time-domain peak at k_peak corresponds to the integral of
        # the power spectrum with a phase. We want to reduce the
        # "sidelobe" content.
        
        # Strategy: reduce power at frequencies that are NOT DC
        # The DC bin is index 0 in FFT
        power_new = power.copy()
        
        # Find the main lobe width (where most of the DC energy is)
        total_power = np.sum(power)
        dc_power = power[0]
        
        # Target: make the power spectrum more like a delta at DC
        # But we can't eliminate all sidelobes because of non-negativity
        # Use a soft threshold
        alpha = 0.4 + 0.3 * min(1.0, it / 100)
        power_norm = power / p_max
        power_soft = power_norm ** alpha * p_max
        
        # Blend
        blend = 0.3 + 0.4 * it / max_iters
        power_new = (1 - blend) * power + blend * power_soft
        
        phase = np.angle(A)
        mag = np.sqrt(np.maximum(power_new, 0))
        A_new = mag * np.exp(1j * phase)
        
        a_new = ifft(A_new).real[:n]
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def subgradient_optimize(a, max_iters=500, time_budget=40, tau=1.0):
    """
    Subgradient optimization with softened max (log-sum-exp) and
    adaptive step size.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.1
    momentum_vec = np.zeros(n)
    momentum = 0.9
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        
        # Soft max weights
        tau_eff = tau + 0.5 * it / max_iters
        shifted = (conv - c_max) / tau_eff
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        
        # Gradient computation
        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            if weights[k_idx] < 1e-10:
                continue
            wk = k_idx
            # d conv[wk] / d a[i] = 2*a[wk-i] if 0 <= wk-i < n
            i_start = max(0, wk - n + 1)
            i_end = min(n, wk + 1)
            if i_end > i_start:
                indices = np.arange(i_start, i_end)
                a_idx = wk - indices
                grad[indices] += 2.0 * weights[k_idx] * a[a_idx]
        
        # Center the gradient (doesn't change the objective since sum is preserved)
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step size
        l2 = np.sum(grad_centered ** 2)
        if l2 > 1e-15:
            step = min(0.5, 0.1 / np.sqrt(l2))
        
        momentum_vec = momentum * momentum_vec - step * grad_centered
        a_new = a + momentum_vec
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Rejected step: reduce momentum
            momentum_vec *= 0.5
    
    return best_a, best_obj


def multi_peak_water_filling(a, max_iters=300, time_budget=40):
    """
    Multi-peak water filling: identify all active peaks and reduce them
    simultaneously using a projected subgradient approach.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 0.05
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        
        # Find all peaks within 5% of the max
        threshold = c_max * 0.95
        active_idx = np.where(conv >= threshold)[0]
        
        # Compute combined gradient
        grad = np.zeros(n)
        for k_idx in active_idx:
            i_start = max(0, k_idx - n + 1)
            i_end = min(n, k_idx + 1)
            if i_end > i_start:
                indices = np.arange(i_start, i_end)
                a_idx = k_idx - indices
                grad[indices] += 2.0 * a[a_idx]
        
        # Normalize gradient by number of active peaks
        grad /= len(active_idx)
        grad_centered = grad - np.mean(grad)
        
        # Line search
        best_local_obj = best_obj
        best_local_a = a.copy()
        for ss in [step, step * 0.5, step * 2.0, step * 0.25]:
            a_try = a - ss * grad_centered
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_local_a.copy()
            step = min(0.5, step * 1.05)
        else:
            step *= 0.8
    
    return best_a, best_obj


def peak_directed_search(a, n_iters=200, time_budget=30):
    """
    For each peak position, compute the exact direction that reduces it,

    """

        # Gradient for this peak
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        if i_end > i_start:
            indices = np.arange(i_start, i_end)
            a_idx = k_star - indices

        improved = False

            a_pert = a * np.exp(np.random.default_rng(it).normal(0, 0.01, n))

    return best_a, best_obj


def coordinate_descent(a, n_sweeps=10, time_budget=30):
    """
    Coordinate descent: optimize one element at a time.
    For each element a[i], find the value that minimizes the objective
    (given all other elements fixed).
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            # Try a few values for a[i]
            orig = a[i]
            candidates = [orig * 0.5, orig * 0.75, orig * 1.25, orig * 1.5,
                         orig + 0.01, orig - 0.01, 0.0]
            candidates = [max(0, c) for c in candidates]
            
            best_val = orig
            best_local_obj = compute_obj(a)  # current value
            
            for val in candidates:
                a[i] = val
                a_norm = normalize(a)
                obj = compute_obj(a_norm)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_val = val
            
            a[i] = best_val
            a = normalize(a)
            
            if compute_obj(a) < best_obj:
                best_obj = compute_obj(a)
                best_a = a.copy()
    
    return best_a, best_obj


def iterative_soft_threshold(a, max_iters=200, time_budget=30):
    """
    Iterative soft thresholding on the convolution:
    1. Compute conv
    2. Soft-threshold the peaks (reduce values above a threshold)
    3. Find a new a whose convolution matches the modified one
       (via spectral factorization: a = isqrt(conv) in frequency domain)
    4. Project to non-negative
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        c_mean = float(np.mean(conv))
        
        # Target: bring peaks down toward the mean
        # Use a soft threshold
        threshold = c_mean * 3.0 + c_max * 0.3
        conv_mod = conv.copy()
        above = conv_mod > threshold
        conv_mod[above] = threshold + (conv_mod[above] - threshold) * 0.5
        conv_mod = np.maximum(conv_mod, 0)
        
        # Spectral factorization: find a such that a*a ≈ conv_mod
        # In frequency domain: |A|^2 = FFT(conv_mod), so A = sqrt(FFT(conv_mod))
        C = fft(conv_mod, N)
        # Due to numerical issues, C should be real and non-negative if conv_mod is a valid autocorrelation
        # But it might not be perfectly, so take complex sqrt
        A_new = np.sqrt(C + 0j)
        a_new = ifft(A_new).real[:n]
        
        # Project
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def subgradient_optimize(a, max_iters=400, time_budget=30.0, tau_init=0.5):
    """
    Subgradient optimization with Nesterov-style momentum on softmax-smoothed peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    momentum = 0.85
    step = 0.1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        c_max = float(np.max(conv))

        tau = tau_init + 1.0 * it / max_iters
        shifted = np.clip((conv - c_max) / tau, -300.0, 300.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] for valid k-i
        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            if weights[k_idx] < 1e-12:
                continue
            wk = k_idx
            i_start = max(0, wk - n + 1)
            i_end = min(n, wk + 1)
            if i_end > i_start:
                indices = np.arange(i_start, i_end)
                a_idx = wk - indices
                grad[indices] += 2.0 * weights[k_idx] * a[a_idx]

        grad_centered = grad - np.mean(grad)

        # Adaptive step
        l2 = np.sum(grad_centered ** 2)
        if l2 > 1e-15:
            step = min(0.5, 0.05 / max(np.sqrt(l2), 0.01))

        v = momentum * v - step * grad_centered
        a_new = a + v
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            v *= 0.5

    return best_a, best_obj

            a_idx = k_star - indices
            grad[indices] = 2.0 * a[a_idx]

        grad_centered = grad - np.mean(grad)

        improved = False
        for ss in [step, step * 0.5, step * 1.5, step * 0.25]:
            a_try = a - ss * grad_centered
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                step = min(0.3, step * 1.05)
                improved = True
                break

        if not improved:
            step *= 0.85
            if step < 0.001:
                break

    return best_a, best_obj


def convolution_shaping(a, max_iters=200, time_budget=30.0):
    """
    Direct convolution shaping: modify the convolution directly, then
    factor back via spectral methods.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        c_mean = float(np.mean(conv))

        # Soft threshold peaks
        threshold = c_mean + 0.5 * (c_max - c_mean)
        conv_mod = conv.copy()
        above = conv_mod > threshold
        # Compress above-threshold values
        conv_mod[above] = threshold + (conv_mod[above] - threshold) * 0.3
        conv_mod = np.maximum(conv_mod, 0)

        # Ensure the sum is preserved (approximately)
        s_orig = np.sum(conv)
        s_mod = np.sum(conv_mod)
        if s_mod > 1e-15:
            conv_mod = conv_mod * (s_orig / s_mod)

        # Spectral factorization
        C = fft(conv_mod, N)
        A_new = np.sqrt(C + 0j)
        a_new = ifft(A_new).real[:n]
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def try_load_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy optimization:
    1. GS iteration (principled frequency-domain Newton steps targeting flat conv)
    2. Spectral shaping (power spectrum compression)
    3. Subgradient with momentum
    4. Multi-peak reduction
    5. Peak-directed search
    6. Convolution shaping (direct conv modification + spectral factorization)
    7. Multi-length exploration and perturbation restarts
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    parent = try_load_parent()
    if parent is not None:
        parent = normalize(parent.astype(np.float64))
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700) / 700.0
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # === Phase 1: Quick GS screening of lengths ===
    print("\n=== Phase 1: Length Screening (GS) ===")
    phase1_end = start_time + 80.0

    n_options = sorted(set([
        n_parent,
        max(100, n_parent - 100), max(100, n_parent - 50),
        n_parent + 50, n_parent + 100, n_parent + 200,
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2000]

    candidates = []
    for n_try in n_options:
        if time.monotonic() > phase1_end:
            break
        a = normalize(resample(best_seq, n_try))
        a_opt, obj_opt = gs_iteration(a, max_iters=50, time_budget=4.0)
        if obj_opt < np.inf:
            candidates.append((obj_opt, a_opt, n_try))
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

    candidates.sort(key=lambda x: x[0])
    if candidates:
        print(f"  Screened {len(candidates)} lengths, best: {candidates[0][0]:.8f} (n={candidates[0][2]})")

    # === Phase 2: Intensive multi-strategy optimization ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    print(f"Best before: {best_score:.8f}")
    phase2_end = start_time + 800.0

    top = [(best_score, best_seq.copy(), len(best_seq))]
    for obj_c, a_c, n_c in candidates[:2]:
        if n_c != len(best_seq):
            top.append((obj_c, a_c.copy(), n_c))

    for ci, (obj_init, a_init, n_init) in enumerate(top):
        if time.monotonic() > phase2_end:
            break

        a = normalize(a_init.copy())
        local_obj = compute_obj(a)
        print(f"  Candidate {ci} (n={n_init}): obj={local_obj:.8f}")

        # Cycle through strategies
        strategies = [
            ("gs1", lambda: gs_iteration(a, max_iters=150, time_budget=20)),
            ("gs2", lambda: gs_iteration_v2(a, max_iters=200, time_budget=20)),
            ("spec1", lambda: spectral_shaping(a, max_iters=150, time_budget=20)),
            ("spec2", lambda: spectral_shaping_aggressive(a, max_iters=150, time_budget=20)),
            ("subgrad", lambda: subgradient_optimize(a, max_iters=200, time_budget=20)),
            ("multipk", lambda: multi_peak_reduce(a, max_iters=150, time_budget=20)),
            ("peakdir", lambda: peak_directed(a, n_iters=150, time_budget=15)),
            ("convshape", lambda: convolution_shaping(a, max_iters=100, time_budget=20)),
        ]

        cycle = 0
        while time.monotonic() < phase2_end:
            remaining = phase2_end - time.monotonic()
            if remaining < 20:
                break

            name, fn = strategies[cycle % len(strategies)]
            a_new, obj_new = fn()

            if obj_new < local_obj - 1e-12:
                local_obj = obj_new
                a = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = a.copy()
                    print(f"    ** {name} cycle {cycle}: {best_score:.8f} **")
            cycle += 1

        print(f"  Candidate {ci} final: obj={local_obj:.8f}")

    # === Phase 3: Perturbation and re-optimization ===
    print(f"\n=== Phase 3: Perturbation ===")
    print(f"Best: {best_score:.8f}")
    phase3_end = start_time + 1150.0
    n_best = len(best_seq)

    for attempt in range(30):
        if time.monotonic() > phase3_end:
            break

        noise_scale = 0.005 + 0.02 * rng.random()
        a_pert = best_seq * np.exp(rng.normal(0, noise_scale, n_best))
        a_pert = normalize(a_pert)

        # Try a quick GS + peak directed
        a_opt, obj_opt = gs_iteration(a_pert, max_iters=50, time_budget=8)
        a_opt2, obj_opt2 = peak_directed(a_opt, n_iters=50, time_budget=8)

        if obj_opt2 < best_score:
            best_score = obj_opt2
            best_seq = a_opt2.copy()
            print(f"  Perturb {attempt}: {best_score:.8f}")
        elif obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Perturb GS {attempt}: {best_score:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best: {best_score:.8f}")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        a_final = normalize(best_seq.copy())

        polish_rounds = min(15, int(remaining / 15))
        for pr in range(polish_rounds):
            if time.monotonic() > deadline - 10:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break

            # Alternate between GS and spectral shaping with longer budgets
            if pr % 2 == 0:
                a_new, obj_new = gs_iteration_v2(a_final, max_iters=200, time_budget=min(rem * 0.4, 25))
            else:
                a_new, obj_new = spectral_shaping(a_final, max_iters=200, time_budget=min(rem * 0.4, 25))

            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a_final = a_new.copy()
                print(f"  Polish {pr}: {best_score:.8f}")

            # Quick peak-directed between rounds
            rem2 = deadline - time.monotonic()
            if rem2 > 10:
                a_pd, obj_pd = peak_directed(a_final, n_iters=50, time_budget=min(rem2 * 0.3, 10))
                if obj_pd < best_score:
                    best_score = obj_pd
                    best_seq = a_pd.copy()
                    a_final = a_pd.copy()
                    print(f"  Polish-PD {pr}: {best_score:.8f}")

    # === Phase 5: Try nearby lengths ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_cur = len(best_seq)
        for n_try in [n_cur - 10, n_cur + 10, n_cur - 20, n_cur + 20, n_cur + 50]:
            if time.monotonic() > deadline - 10:
                break
            if n_try < 100 or n_try > 2500:
                continue
            a_r = normalize(resample(best_seq, n_try))
            a_opt, obj_opt = gs_iteration(a_r, max_iters=80, time_budget=8)
            a_opt2, obj_opt2 = peak_directed(a_opt, n_iters=50, time_budget=5)
            if obj_opt2 < best_score:
                best_score = obj_opt2
                best_seq = a_opt2.copy()
                print(f"  Length try n={n_try}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
