```python
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = (conv(a, g))[k]"""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        for k in range(j, min(j + n, m)):
            A[k, j] = a[k - j]
    return A

def solve_lp(a, M=None, time_limit=30):
    """
    Solve: min M s.t. (conv(a,g))[k] <= M for all k, sum(g)=1, g>=0
    Returns (g, M_opt)
    """
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)
    
    # Variables: g[0..n-1], M
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ g - M <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except:
        pass
    return None, None

def lp_alternate(a, max_iters=100, time_budget=100, verbose=True):
    """
    Alternating LP: fix a, solve for g; fix g, solve for a. Iterate.
    """
    n = len(a)
    a = project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        tl = min(remaining * 0.4, 20)
        
        # Solve for g given a
        g, M_g = solve_lp(a, time_limit=tl)
        if g is None:
            break
        
        # Solve for a given g
        a_new, M_a = solve_lp(g, time_limit=tl)
        if a_new is None:
            break
        
        # Blending for stability
        a = 0.5 * a + 0.5 * a_new
        a = project(a)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    LP alt it{it}: obj={best_obj:.8f}, M_g={M_g:.6f}, M_a={M_a:.6f}")
        
        # Check convergence
        if it > 5 and best_obj > obj * (1 - 1e-10):  # No improvement
            pass  # Continue for a few more iterations
    
    return best_a, best_obj
```

    a = rng.random(n) + 0.1  # Start with random positive values
    a = a / a.sum()

    while N < 2 * n - 1:
        N <<= 1

        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2
        
        # Modify power spectrum to be flatter
        # Replace with a "flattened" version
        power_flat = np.maximum(power, np.mean(power) * 0.5)  # Floor the spectrum

"""
Optimized step function sequence search using LP alternating, mirror descent,
and frequency-domain initialization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary strategy: LP block-coordinate descent with adaptive blending,
supplemented by multiplicative (mirror) descent polishing and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(fixed, n):
    """Build matrix M where (M @ x)[k] = conv(fixed, x)[k]."""
    m = 2 * n - 1
    M = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, m - 1)
        M[k_start:k_end + 1, j] = fixed[:k_end - k_start + 1]
    return M


def solve_lp_minpeak(fixed, time_limit=20):
    """
    Solve: min M s.t. conv(fixed, x)[k] <= M for all k, sum(x)=1, x>=0.
    Returns (x_best, M_opt) or (None, None).
    """
    n = len(fixed)
    m = 2 * n - 1
    A = build_conv_matrix(fixed, n)

    # Variables: x[0..n-1], M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: A @ x - M <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            x = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(x))
            if s > 1e-10:
                x = x / s
                M_opt = float(res.x[n])
                return x, M_opt
    except Exception:
        pass
    return None, None


def lp_alternate(a, max_iters=200, time_budget=100.0, verbose=False):
    """
    LP block-coordinate descent: alternate solving for a given g and g given a.
    Uses adaptive blending for stability.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_improvement = 0.0

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        remaining = time_budget - elapsed
        if remaining < 4.0:
            break

        tl = min(remaining * 0.35, 15.0)

        # Solve for g given a
        g, M_g = solve_lp_minpeak(a, time_limit=tl)
        if g is None:
            break

        # Solve for a given g
        a_new, M_a = solve_lp_minpeak(g, time_limit=tl)
        if a_new is None:
            break

        # Try different blend factors
        best_blend_obj = best_obj
        best_blend_a = best_a.copy()

        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * a_new
            a_try = project(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try.copy()

        if best_blend_obj < best_obj - 1e-12:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            a = best_blend_a.copy()
            stall_count = 0
            last_improvement = elapsed
            if verbose and it % 10 == 0:
                print(f"    LP alt it{it}: obj={best_obj:.8f}")
        else:
            stall_count += 1
            # Try a small perturbation to escape stagnation
            if stall_count % 10 == 0:
                rng_local = np.random.default_rng(it)
                a_pert = a * np.exp(rng_local.normal(0, 0.05, n))
                a = project(a_pert)

        if stall_count > 50:
            break

    return best_a, best_obj


def mirror_descent(a, max_iters=500, time_budget=60.0, tau_init=0.01, eta_init=0.1):
    """
    Mirror descent (multiplicative updates) with softmax-subgradient.
    a_new[i] = a[i] * exp(-eta * grad[i]), then project.
    """
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        if elapsed < 0.5:
            if (it + 1) % 50 != 0:
                continue  # Skip some early iterations for speed

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights
        tau = tau_init + 0.5 * it / max_iters
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -50.0, 50.0)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
        # Use FFT for efficiency
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fw = rfft(w, size)
        fa = rfft(a, size)
        # conv(w_rev, a) gives sum_j w_rev[j]*a[k-j] = sum_j w[len(w)-1-j]*a[k-j]
        # We want sum_k w[k]*a[k-i] = (w * a_rev)[n-1-i] ... 
        # Actually: let's just compute the correlation directly
        # grad[i] = 2 * sum_{k=0}^{2n-2} w[k] * a[k-i] (where 0 <= k-i < n)
        # = 2 * (w * a)[i] reversed... 
        # Let me use the identity: sum_k w[k]*a[k-i] = (w_rev * a)[2n-2-i]... 
        # Simpler: just do it with the Toeplitz structure
        # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]  (where i+j < 2n-1)
        # This is a cross-correlation
        w_rev = w[::-1]
        N_fft = 1
        while N_fft < len(w) + n - 1:
            N_fft <<= 1
        # correlation: sum_j w[i+j]*a[j] for valid j
        # = ifft(fft(w) * conj(fft(a))) shifted appropriately
        fw_full = np.fft.fft(w, N_fft)
        fa_full = np.fft.fft(a, N_fft)
        corr = np.fft.ifft(fw_full * np.conj(fa_full)).real
        # corr[m] = sum_j w[j]*conj(a[j-m])... for real: sum_j w[j]*a[j-m]
        # We want sum_j w[i+j]*a[j]. Let l = i+j, j = l-i:
        # = sum_l w[l]*a[l-i] = corr_full where corr_full[m] = sum_l w[l]*a[l-m]
        # So grad[i] = 2 * corr[i] (with appropriate indexing)
        # corr from ifft(fft(w)*conj(fft(a))) gives: corr[m] = sum_l w[(l-m)%N]*a[l]
        # Hmm, let me just compute it directly for correctness
        grad = np.zeros(n)
        for i in range(n):
            # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] where i+j < 2n-1
            valid_j = np.arange(n)
            idx = i + valid_j
            valid = idx < len(w)
            if np.any(valid):
                grad[i] = 2.0 * np.sum(w[idx[valid]] * a[valid_j[valid]])

        # Normalize gradient to have unit mean (for centering)
        grad_mean = np.mean(grad)

        # Mirror descent update
        a_new = a * np.exp(-eta * (grad - grad_mean))
        a_new = project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Reduce step size
            eta *= 0.95

    return best_a, best_obj


def flat_spectrum_init(n, rng, iterations=50):
    """Generate initialization with approximately flat autocorrelation via FFT."""
    a = np.ones(n) / n  # Start uniform

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(iterations):
        fa = np.fft.fft(a, N)
        power = np.abs(fa) ** 2

        # Flatten the power spectrum (reduce peaks)
        p_max = np.max(power)
        p_min = np.min(power[power > 0]) if np.any(power > 0) else p_max * 0.1
        # Apply power compression
        alpha = 0.6 + 0.3 * (it / iterations)  # Start aggressive, end gentle
        power_compressed = (power / p_max) ** alpha * p_max
        power_compressed = np.maximum(power_compressed, p_min * 0.5)

        # Reconstruct
        magnitude = np.sqrt(power_compressed)
        phase = np.angle(fa)
        fa_new = magnitude * np.exp(1j * phase)
        a_new = np.fft.ifft(fa_new).real[:n]
        a = project(a_new)

    return a


def hann_like_init(n):
    """Hann-window-like initialization."""
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = 0.5 * (1 + np.cos(x))
    return project(a)


def cosine_perturb_init(n, rng):
    """Initialization with cosine perturbations of uniform."""
    a = np.ones(n) / n
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    # Add a few cosine modes
    for k in range(1, 4):
        amp = rng.uniform(-0.3, 0.3)
        a += amp * np.cos(k * x)
    return project(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def subgradient_polish(a, max_iters=200, time_budget=30.0):
    """Quick subgradient polishing with adaptive step size."""
    n = len(a)
    a = project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Subgradient: chase the peak
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
            j2 = i + (2 * n - 2) - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad = grad - np.mean(grad)

        # Try a few step sizes
        improved = False
        for ss in [eta, eta * 0.5, eta * 2.0]:
            a_try = a - ss * grad
            a_try = project(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                eta = ss
                improved = True
                break

        if not improved:
            eta *= 0.9

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and np.sum(parent) > 0.01:
            return parent.copy()
    except (NameError, Exception):
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-stage optimization combining LP alternating,
    mirror descent, and frequency-domain initialization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_norm = project(parent)
        parent_obj = compute_obj(parent_norm)
        print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
        best_score = parent_obj
        best_seq = parent_norm.copy()
    else:
        # Default initialization
        n_default = 700
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"No parent found. Starting uniform n={n_default}: obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting best: {best_score:.8f} (n={n_best})")

    # === Phase 1: Generate and screen initializations (60s) ===
    print("\n=== Phase 1: Initialization Screening ===")
    p1_budget = min(60.0, (deadline - start_time) * 0.04)
    p1_start = time.monotonic()

    n_options = [n_best]
    # Add nearby lengths
    for delta in [-50, -25, 25, 50, 75, 100]:
        nn = n_best + delta
        if 100 <= nn <= 2000:
            n_options.append(nn)
    n_options = list(dict.fromkeys(n_options))

    candidates = []  # (obj, a)

    for n_try in n_options:
        if time.monotonic() - p1_start > p1_budget:
            break
        per_init_time = max(2.0, p1_budget / (len(n_options) * 4))

        # Generate several initializations at this length
        inits = []
        if parent is not None:
            inits.append(resample_sequence(parent, n_try))
        inits.append(np.ones(n_try) / n_try)
        inits.append(hann_like_init(n_try))
        inits.append(flat_spectrum_init(n_try, rng, iterations=30))
        inits.append(cosine_perturb_init(n_try, rng))
        inits.append(project(np.abs(rng.standard_normal(n_try)) + 0.1))

        for init in inits:
            if time.monotonic() - p1_start > p1_budget:
                break
            a_init = project(init)
            # Quick subgradient polish
            a_quick, obj_quick = subgradient_polish(a_init, max_iters=30, time_budget=per_init_time)
            if obj_quick < np.inf:
                candidates.append((obj_quick, a_quick))
                if obj_quick < best_score:
                    best_score = obj_quick
                    best_seq = a_quick.copy()
                    print(f"  New best from init: n={n_try}, obj={best_score:.8f}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Screened {len(candidates)} candidates. Best: {candidates[0][0] if candidates else 'N/A':.8f}")

    # === Phase 2: LP Alternating on top candidates (50% of time) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    print(f"Best so far: {best_score:.8f}")
    p2_budget = min(700.0, (deadline - time.monotonic()) * 0.50)
    p2_start = time.monotonic()

    # Focus on top 3 candidates
    top_cands = []
    seen = set()
    for obj_c, a_c in candidates[:6]:
        n_c = len(a_c)
        if n_c not in seen:
            top_cands.append((obj_c, a_c))
            seen.add(n_c)
        if len(top_cands) >= 3:
            break

    # Always include the current best
    best_n = len(best_seq)
    if best_n not in seen:
        top_cands.append((best_score, best_seq.copy()))

    for ci, (obj_init, a_init) in enumerate(top_cands):
        if time.monotonic() - p2_start > p2_budget:
            break
        remaining = p2_budget - (time.monotonic() - p2_start)
        if remaining < 20:
            break
        alloc = min(remaining, 200.0)
        n_c = len(a_init)
        print(f"  Candidate {ci} (n={n_c}): obj={obj_init:.8f}, budget={alloc:.0f}s")

        a_work = project(a_init.copy())
        a_res, obj_res = lp_alternate(a_work, max_iters=300, time_budget=alloc, verbose=True)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  ** LP improved: obj={best_score:.8f} **")

    # === Phase 3: Mirror descent polishing (20% of time) ===
    print(f"\n=== Phase 3: Mirror Descent Polish ===")
    print(f"Best so far: {best_score:.8f}")
    p3_budget = min(250.0, (deadline - time.monotonic()) * 0.20)
    p3_start = time.monotonic()

    a_work = project(best_seq.copy())
    n_best = len(a_work)

    # Multiple rounds with different parameters
    for round_i in range(3):
        if time.monotonic() - p3_start > p3_budget:
            break
        remaining = p3_budget - (time.monotonic() - p3_start)
        if remaining < 10:
            break
        alloc = min(remaining / (3 - round_i), 80.0)

        tau_val = [0.005, 0.01, 0.02][round_i]
        eta_val = [0.05, 0.1, 0.15][round_i]

        a_md, obj_md = mirror_descent(a_work, max_iters=300, time_budget=alloc,
                                      tau_init=tau_val, eta_init=eta_val)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            a_work = a_md.copy()
            print(f"  MD round {round_i}: obj={best_score:.8f}")
        else:
            a_work = project(a_work)

    # === Phase 4: More LP on the best (20% of time) ===
    print(f"\n=== Phase 4: Additional LP Polish ===")
    print(f"Best so far: {best_score:.8f}")
    p4_budget = min(250.0, (deadline - time.monotonic()) * 0.20)
    p4_start = time.monotonic()

    a_work = project(best_seq.copy())
    a_res, obj_res = lp_alternate(a_work, max_iters=500, time_budget=p4_budget, verbose=True)
    if obj_res < best_score:
        best_score = obj_res
        best_seq = a_res.copy()
        print(f"  LP polish: obj={best_score:.8f}")

    # === Phase 5: Length exploration (15% of time) ===
    print(f"\n=== Phase 5: Length Exploration ===")
    print(f"Best so far: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 30:
        p5_budget = min(200.0, remaining5 * 0.35)
        p5_start = time.monotonic()

        n_current = len(best_seq)
        n_tries = []
        for delta in [-100, -75, -50, -25, 25, 50, 75, 100, 150, 200]:
            nn = n_current + delta
            if 150 <= nn <= 2500 and nn != n_current:
                n_tries.append(nn)
        n_tries = list(dict.fromkeys(n_tries))[:6]

        per_n_budget = max(5.0, p5_budget / max(1, len(n_tries)))

        for n_try in n_tries:
            if time.monotonic() - p5_start > p5_budget:
                break
            rem_n = p5_budget - (time.monotonic() - p5_start)
            if rem_n < 5:
                break

            a_resampled = resample_sequence(best_seq, n_try)
            a_resampled = project(a_resampled)

            # Quick LP
            a_opt, obj_opt = lp_alternate(a_resampled, max_iters=50,
                                          time_budget=min(rem_n * 0.6, per_n_budget),
                                          verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    print(f"Best before final: {best_score:.8f}")
    remaining6 = deadline - time.monotonic()
    if remaining6 > 10 and best_seq is not None:
        a_final = project(best_seq.copy())
        n_final = len(a_final)

        # Cycle through methods until time runs out
        method_cycle = 0
        while time.monotonic() < deadline - 5:
            rem = deadline - time.monotonic()
            if rem < 5:
                break

            if method_cycle % 3 == 0:
                # LP
                if n_final <= 1500:
                    a_p, obj_p = lp_alternate(a_final, max_iters=50,
                                              time_budget=min(rem * 0.3, 40.0), verbose=False)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        a_final = a_p.copy()
                        print(f"  Final LP: {best_score:.8f}")
            elif method_cycle % 3 == 1:
                # Mirror descent
                a_p, obj_p = mirror_descent(a_final, max_iters=100,
                                            time_budget=min(rem * 0.2, 20.0),
                                            tau_init=0.005, eta_init=0.05)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Final MD: {best_score:.8f}")
            else:
                # Subgradient + perturbation
                a_p, obj_p = subgradient_polish(a_final, max_iters=50,
                                                time_budget=min(rem * 0.1, 10.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                # Small perturbation and re-optimize
                if time.monotonic() < deadline - 10:
                    a_pert = a_final * np.exp(rng.normal(0, 0.02, n_final))
                    a_pert = project(a_pert)
                    a_pert, obj_pert = subgradient_polish(a_pert, max_iters=30, time_budget=5.0)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_final = a_pert.copy()
                        print(f"  Final pert: {best_score:.8f}")

            method_cycle += 1

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"\n  evaluate_sequence says: {val:.8f}")
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
