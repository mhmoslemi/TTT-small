For k=0 to n-1: k+1 terms → sum = n(n+1)/2
For k=n to 2n-2: 2n-1-k terms → sum = n(n-1)/2
Total = n(n+1)/2 + n(n-1)/2 = n^2

Plus (2n-1) for the t variable column. Total ≈ n^2 + 2n ≈ n^2.

For n=768: about 590K non-zeros. This should be fine for HiGHS.

The LP solve time for this size is probably 1-10 seconds depending on the solver. So we can do maybe 50-200 LP iterations in 1500 seconds (including other overhead).

"""
Optimized search for low-autocorrelation non-negative sequences.
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based convolution and exact coordinate descent
2. SCA-LP (Successive Convex Approximation with LP subproblems)
3. Log-sum-exp smoothed Adam optimization
4. Multi-length exploration with restarts
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix

def fft_conv(a):
    """Full self-convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)

def resample(src, n):
    """Resample 1D array to length n."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    if src.size == 0:
        return np.ones(n) / n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coord_descent(a, deadline, n_sweeps=5000, seed=42):
    """Coordinate descent with candidate values including 0."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    conv = fft_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n) if sweep % 2 == 0 else range(n)
        improved = False
        
        for i in order:
            old = a[i]
            
            # Build candidate values
            candidates = [0.0]
            if old > 1e-10:
                candidates += [0.1*old, 0.3*old, 0.5*old, 0.7*old,
                               1.3*old, 1.5*old, 1.7*old, 2.0*old]
            else:
                candidates += [1e-4, 1e-3, 1e-2]
            
            # Compute affected range
            lo = max(0, i)
            hi = min(2*n-2, i + n - 1)
            kq = 2 * i
            a_slice = a[lo - i:hi - i + 1].copy()
            
            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)
            
            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                
                # Update conv
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if 0 <= kq < 2*n-1:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    conv[lo:hi+1] -= 2.0 * delta * a_slice
                    if 0 <= kq < 2*n-1:
                        conv[kq] -= delta * delta
                    continue
                
                new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                # Undo
                conv[lo:hi+1] -= 2.0 * delta * a_slice
                if 0 <= kq < 2*n-1:
                    conv[kq] -= delta * delta
            
            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if 0 <= kq < 2*n-1:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s) if s > 1e-10 else float('inf')
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a

def sca_lp(a, deadline, max_iters=100, verbose=False):
    """SCA-LP: Solve LP subproblem at each iteration."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fft_conv(a)
        M = c.max()
        
        # LP: minimize t
        # s.t. (a*g)[k] - t <= c[k]  for all k
        #      sum(g) = 1
        #      g >= 0
        
        rows = []
        cols = []
        vals = []
        b_ub_list = []
        
        for k in range(2*n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub_list.append(c[k])
        
        A_ub = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n+1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = [1.0]
        
        bounds = [(0, None)] * n + [(None, None)]
        
        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b_ub_list),
                       A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                       method='highs', options={'time_limit': 15})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Try different step sizes
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-14:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        if verbose and it % 10 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f}", flush=True)
        
        if no_improve > 15:
            break
    
    return best_a, best_obj

"""
Optimized search for low-autocorrelation non-negative sequences minimizing
J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses: SCA-LP with constraint reduction, FFT coordinate descent,
smoothed Adam, multi-length exploration, and restart strategies.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fft_conv(a):
    """Full self-convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, n_dest):
    """Resample 1D array to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_dest:
        return src.copy()
    if src.size == 0:
        return np.ones(n_dest) / n_dest
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_dest)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent(a, deadline, max_sweeps=10000, seed=42):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fft_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n) if sweep % 3 == 0 else range(n)
        improved = False

        for i in order:
            old = a[i]

            if old < 1e-12:
                candidates = [0.0, 0.5e-3, 1e-3, 5e-3, 1e-2]
            else:
                candidates = [0.0, 0.1*old, 0.3*old, 0.5*old, 0.7*old,
                              1.0*old, 1.4*old, 1.8*old]

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            a_slice = a[lo - i:hi - i + 1].copy()

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if 0 <= kq < 2 * n - 1:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if 0 <= kq < 2 * n - 1:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if 0 <= kq < 2 * n - 1:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        if s > 1e-10:
            current_obj = 2.0 * n * conv.max() / (s * s)
        else:
            current_obj = float('inf')
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def sca_lp(a, deadline, max_iters=200, constraint_frac=1.0, verbose=False):
    """
    SCA-LP with optional constraint reduction.
    At each step, linearize and solve LP for best direction.
    constraint_frac: fraction of constraints to keep (top C_k values).
    """
    n = len(a)

    best_a = a.copy()

    for it in range(max_iters):

            break

        # Select important constraints (top fraction)
        n_constraints = 2 * n - 1
        if constraint_frac < 1.0:
            n_keep = max(n, int(n_constraints * constraint_frac))
            # Keep constraints where C_k is in the top n_keep

            mask = c >= thresholds
        else:
            mask = np.ones(n_constraints, dtype=bool)

        active_idx = np.where(mask)[0]

        rows = []
        cols = []
        vals = []
        b_ub_list = []

        for idx in active_idx:
            k = int(idx)

            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub_list))
                cols.append(j)
                vals.append(a[k - j])
            rows.append(len(b_ub_list))
            cols.append(n)
            vals.append(-1.0)
            b_ub_list.append(c[k])

        A_ub = coo_matrix((vals, (rows, cols)),
                          shape=(len(b_ub_list), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)

        A_eq = np.zeros((1, n + 1))

        try:

                    if not found:

                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 5 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f}", flush=True)

            break

def smooth_adam(a, deadline, max_iters=8000, seed=0, verbose=False):
    """Adam optimization with log-sum-exp smoothed max(conv) gradient."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)

    t_count = 0
    no_improve = 0
    rng = np.random.default_rng(seed)

    size2 = 1
    while size2 < 2 * n - 1:
        size2 <<= 1
    A_fft_cache = np.fft.rfft(a, size2)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = 15.0 + 60.0 * frac

        c = fft_conv(a)
        c_shifted = c - c.max()
        exp_c = np.exp(tau * c_shifted)
        w = exp_c / exp_c.sum()

        # Compute gradient using FFT
        W_fft = np.fft.rfft(w, size2)
        wa = np.fft.irfft(W_fft * A_fft_cache, size2)[:2 * n - 1]

        w_rev = w[::-1]
        Wr_fft = np.fft.rfft(w_rev, size2)
        awr = np.fft.irfft(A_fft_cache * Wr_fft, size2)[:2 * n - 1]

        grad = np.zeros(n)
        for i in range(n):
            grad[i] = wa[i] + awr[n - 1 + i]

        # Project to simplex tangent space
        grad = grad - np.mean(grad)

        # Adam update
        t_count += 1
        lr = 0.04 / (1.0 + 0.03 * t_count ** 0.5)
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * grad ** 2
        m_hat = m_vec / (1.0 - beta1 ** t_count)
        v_hat = v_vec / (1.0 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        A_fft_cache = np.fft.rfft(a_new, size2)

        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                a_new = best_a + rng.normal(0, 0.015, n)
                a_new = project_simplex(a_new)
                m_vec = np.zeros(n)
                v_vec = np.zeros(n)
                t_count = 0
                no_improve = 0

        a = a_new

    return best_a, best_obj

def generate_inits(n, rng, parent=None):

        if ps > 1e-10:

        inits.append(p.copy())

        inits.append(project_simplex(p + rng.normal(0, 0.07, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.12, n)))

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 5, 7]:

            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    for center in [0.25, 0.5, 0.75]:
        for width in [0.1, 0.2, 0.3]:
            inits.append(project_simplex(np.exp(-((t - center) / width) ** 2) + 0.05))

    ksize = max(5, n // 12)

    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Anti-symmetric pattern (high at edges, low center)
    inits.append(project_simplex(1.0 + 0.4 * np.cos(np.pi * t * 2)))
    # Power of cosine
    inits.append(project_simplex(np.abs(np.cos(np.pi * t)) ** 2 + 0.1))

    a_opt, obj_opt = optimize_sequence(parent, deadline=phase1_end - 60)
    update_best(a_opt)
    print(f"  Parent optimized: J={obj_opt:.8f}", flush=True)

    # Try a few diverse inits at same n
    remaining_p1 = phase1_end - time.monotonic()
    if remaining_p1 > 30:
        inits = generate_inits(n_parent, rng, parent)
        n_try = min(4, len(inits))
        per_init = remaining_p1 / (n_try + 1)
        for idx in range(1, n_try + 1):
            if time.monotonic() > phase1_end - 20:
                break
            a0 = inits[idx]
            a0 = np.maximum(a0, 0)
            if a0.sum() < 1e-10:
                continue
            a0 = a0 / a0.sum()
            alloc = min(per_init - 5, phase1_end - time.monotonic() - 10)
            if alloc < 20:
                break
            a_opt, obj_opt = optimize_sequence(a0, deadline=time.monotonic() + alloc)
            update_best(a_opt)
            print(f"  Init {idx}: J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration (25% of total time) ===
    phase2_end = t_start + 0.70 * (budget_s - 30)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration (until {phase2_end - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidate_ns = [384, 512, 640, 896, 1024, 1280, 1536]
    candidate_ns = [n for n in candidate_ns if n != n_parent and n <= 2048 and n >= 256]
    candidate_ns.sort(key=lambda x: abs(x - n_parent))

    for new_n in candidate_ns[:4]:
        if time.monotonic() > phase2_end - 40:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 50:
            break
        alloc = min(120.0, remaining - 15)
        if alloc < 40:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = optimize_sequence(a_ext, deadline=time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive optimization of best (30% of total time) ===
    phase3_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive optimization (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        remaining = phase3_deadline - time.monotonic()
        if remaining > 30:
            # Pass 1: SCA-LP with reduced constraints (faster iterations)
            t_p1 = time.monotonic() + 0.25 * remaining
            print(f"  SCA-LP reduced (budget {t_p1 - time.monotonic():.0f}s)...", flush=True)
            a_sca, obj_sca = sca_lp(best_seq, deadline=t_p1, max_iters=150,
                                    constraint_frac=0.6)
            update_best(a_sca)

            remaining = phase3_deadline - time.monotonic()
            if remaining > 30:
                # Pass 2: Coord descent (long)
                t_p2 = time.monotonic() + 0.25 * remaining
                print(f"  Coord descent (budget {t_p2 - time.monotonic():.0f}s)...", flush=True)
                a_cd = coord_descent(best_seq, deadline=t_p2, max_sweeps=20000, seed=777)
                update_best(a_cd)

                remaining = phase3_deadline - time.monotonic()
                if remaining > 30:
                    # Pass 3: Smooth Adam with different params
                    t_p3 = time.monotonic() + 0.20 * remaining
                    print(f"  Adam (budget {t_p3 - time.monotonic():.0f}s)...", flush=True)
                    a_adam, obj_adam = smooth_adam(best_seq, deadline=t_p3,
                                                   max_iters=8000, seed=42)
                    update_best(a_adam)

                    remaining = phase3_deadline - time.monotonic()
                    if remaining > 30:
                        # Pass 4: LP mirror descent
                        t_p4 = time.monotonic() + 0.20 * remaining
                        print(f"  LP mirror (budget {t_p4 - time.monotonic():.0f}s)...", flush=True)
                        a_lp, obj_lp = lp_mirror(best_seq, deadline=t_p4, max_iters=80)
                        update_best(a_lp)

                        remaining = phase3_deadline - time.monotonic()
                        if remaining > 20:
                            # Final CD polish
                            t_p5 = phase3_deadline - 5
                            print(f"  Final CD (budget {t_p5 - time.monotonic():.0f}s)...", flush=True)
                            a_final = coord_descent(best_seq, deadline=t_p5,
                                                    max_sweeps=10000, seed=999)
                            update_best(a_final)

        # Perturbation restarts if time remains
        remaining = phase3_deadline - time.monotonic()
        if remaining > 60:
            print(f"\n  Perturbation restarts ({remaining:.0f}s remaining)...", flush=True)
            n_restarts = min(3, int(remaining // 80))
            for ri in range(n_restarts):
                if time.monotonic() > phase3_deadline - 30:
                    break
                pert_scale = 0.05 + 0.03 * ri
                a_pert = best_seq + rng.normal(0, pert_scale, best_seq.size)
                a_pert = project_simplex(a_pert)
                alloc = min(60.0, phase3_deadline - time.monotonic() - 20)
                if alloc < 30:
                    break
                a_opt, obj_opt = optimize_sequence(a_pert,
                                                   deadline=time.monotonic() + alloc)
                update_best(a_opt)
                print(f"  Restart {ri}: J={obj_opt:.8f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
