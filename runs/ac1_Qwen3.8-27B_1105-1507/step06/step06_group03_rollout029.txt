def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def compute_smooth_grad(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a using FFT cross-correlation."""
    n = len(a)
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    # This is 2 * (a cross-correlated with w)[j]
    # Using FFT: cross_corr = IFFT(FFT(a) * conj(FFT(w_padded)))
    size = 1
    while size < n + (2 * n - 1):
        size <<= 1
    a_pad = np.zeros(size)
    a_pad[:n] = a
    w_pad = np.zeros(size)
    w_pad[:2 * n - 1] = w
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)
    # cross-correlation: IFFT(A * conj(W))
    cc = np.fft.irfft(A * np.conj(W), size)
    grad = 2.0 * cc[:n]
    return grad

    v_arr = np.zeros(n)

    no_improve = 0

        if time.monotonic() > deadline:
            break

        grad = grad - np.mean(grad)

                t_count = 0

        c = fast_conv(a)
        k_star = np.argmax(c)

        # Subgradient: grad[j] = 2*a[k_star - j] if 0 <= k_star - j < n
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        grad = np.zeros(n)
        idx = np.arange(j_min, j_max + 1)
        grad[idx] = 2.0 * a[k_star - idx]
        grad = grad - np.mean(grad)

        lr = lr_init / (1.0 + 0.05 * np.sqrt(it + 1))
        a_new = project_simplex(a - lr * grad)
        obj_new = compute_J(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def active_set_sca(a, max_iters, deadline, top_k=150, verbose=False):
    """SCA-LP using only top-K most violated constraints for speed."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    active_set = None

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = c.max()

        # Select top-K lags (adaptively grow the set)
        k_sel = min(top_k + it // 10, 2 * n - 1)
        k_indices = np.argsort(c)[::-1][:k_sel]

        rows = []
        cols = []
        vals = []
        b_ub = []

        for idx, k in enumerate(k_indices):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
            rows.append(idx)
            cols.append(n)
            vals.append(-1.0)
            b_ub.append(c[k])

        if len(rows) == 0:
            break

        A = coo_matrix((vals, (rows, cols)), shape=(len(k_indices), n + 1)).tocsr()
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A, b_ub=np.array(b_ub), A_eq=A_eq, b_eq=[1.0],
                        bounds=bounds, method='highs',
                        options={'time_limit': 8, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    found = False
                    for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        atry_sum = a_try.sum()
                        if atry_sum < 1e-10:
                            continue
                        a_try /= atry_sum
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-10:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"    SCA it {it}: obj={best_obj:.8f} k_sel={k_sel}", flush=True)

        if no_improve > 25:
            break

    return best_a, best_obj


def coord_descent_fast(a, max_sweeps, deadline, seed=42):
    """Coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a).copy()
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n) if sweep % 3 == 0 else range(n)
        improved = False

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            best_val = old
            best_local = 2.0 * n * conv.max()

            # Try several candidate values
            candidates = np.array([0.0, 0.2 * old, 0.4 * old, 0.6 * old,
                                   0.8 * old, 1.2 * old, 1.5 * old, 2.0 * old])
            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                new_obj = 2.0 * n * conv.max()
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                improved = True

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences, emphasizing skewed patterns."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.03, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.08, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.15, n)))

    # Skewed patterns (key for this problem)
    for alpha, beta in [(3, 1), (5, 1), (2, 1), (4, 1.5), (6, 1), (3, 2), (5, 2), (7, 1.5)]:
        # Beta-like: t^(alpha-1) * (1-t)^(beta-1)
        shape = np.maximum(t + 0.001, 1e-6) ** (alpha - 1) * np.maximum(1 - t + 0.001, 1e-6) ** (beta - 1)
        inits.append(project_simplex(shape))

    # Exponential decay
    for rate in [1, 2, 3, 5]:
        inits.append(project_simplex(np.exp(-rate * t)))
        inits.append(project_simplex(np.exp(-rate * (1 - t))))

    # Power-law
    for pwr in [0.5, 1, 2, 3]:
        inits.append(project_simplex((1 - t + 0.01) ** pwr))
        inits.append(project_simplex((t + 0.01) ** pwr))

    # Cosine/sine
    for freq in [1, 2, 3]:
        inits.append(project_simplex(1.0 + 0.5 * np.cos(2 * np.pi * freq * t)))

    # Uniform
    inits.append(np.ones(n) / n)

    # Random smooth
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Skewed Gaussian
    for center in [0.3, 0.4, 0.6]:
        for width in [0.15, 0.25, 0.4]:
            skew_shape = np.exp(-((t - center) / width) ** 2 / (1 + 0.5 * (t - center)))
            inits.append(project_simplex(skew_shape + 0.02))

    return inits


def solve_full_one(a0, deadline, verbose=False):
    """Full pipeline for one initial sequence."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return a
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        return a_new

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 1: Adam smooth (30%)
    t1 = time.monotonic() + 0.30 * remaining
    a_adam, obj_adam = adam_smooth(a, max_iters=5000, deadline=t1,
                                   tau_init=30.0, tau_end=300.0, lr_init=0.03)
    a = try_update(a_adam)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 2: Active-set SCA-LP (30%)
    t2 = time.monotonic() + 0.30 * remaining
    a_sca, obj_sca = active_set_sca(a, max_iters=100, deadline=t2, top_k=100)
    a = try_update(a_sca)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 3: Coordinate descent (20%)
    t3 = time.monotonic() + 0.20 * remaining
    a_cd = coord_descent_fast(a, max_sweeps=5000, deadline=t3, seed=123)
    a = try_update(a_cd)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Stage 4: Final Adam polish (20%)
    t4 = time.monotonic() + 0.20 * remaining
    a_adam2, obj_adam2 = adam_smooth(a, max_iters=3000, deadline=t4,
                                     tau_init=100.0, tau_end=1000.0, lr_init=0.01)
    a = try_update(a_adam2)

    return best_a, best_obj


def solve_polish(a, deadline):
    """Intensive local polish on a single sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_J(a)

    def try_update(a_new):
        nonlocal best_a, best_obj
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            return
        a_new = a_new / a_new.sum()
        obj = compute_J(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # CD
    t1 = time.monotonic() + 0.25 * remaining
    a_cd = coord_descent_fast(a, max_sweeps=10000, deadline=t1, seed=777)
    try_update(a_cd)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # SCA-LP
    t2 = time.monotonic() + 0.35 * remaining
    a_sca, obj_sca = active_set_sca(best_a, max_iters=150, deadline=t2, top_k=200)
    try_update(a_sca)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_obj

    # Adam with high tau
    t3 = time.monotonic() + 0.30 * remaining
    a_adam, obj_adam = adam_smooth(best_a, max_iters=5000, deadline=t3,
                                   tau_init=200.0, tau_end=2000.0, lr_init=0.005)
    try_update(a_adam)

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a, best_obj

    # Final CD
    t4 = deadline - 3
    a_cd2 = coord_descent_fast(best_a, max_sweeps=5000, deadline=t4, seed=999)
    try_update(a_cd2)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Strategy:
    - Phase 1: Optimize parent and diverse inits at parent length
    - Phase 2: Explore nearby lengths
    - Phase 3: Intensive polish on best found
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    n_parent = len(parent)

    # === PHASE 1: Optimize at parent length with diverse inits ===
    phase1_deadline = t_start + min(500.0, budget_s * 0.35)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-init at n={n_parent} (deadline t={phase1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    # Sort inits: parent first, then by initial J
    init_scores = []
    for i, init in enumerate(inits):
        init_s = np.maximum(init, 0)
        if init_s.sum() > 1e-10:
            init_s = init_s / init_s.sum()
            init_scores.append((compute_J(init_s), i, init_s))
        else:
            init_scores.append((float('inf'), i, init))
    init_scores.sort(key=lambda x: x[0])

    n_to_try = min(len(init_scores), 10)
    per_init_time = (phase1_deadline - time.monotonic()) / max(1, n_to_try) - 3

    for rank in range(n_to_try):
        remaining_check = phase1_deadline - time.monotonic()
        if remaining_check < 20:
            break
        alloc = min(per_init_time, remaining_check - 3)
        if alloc < 10:
            break

        _, idx, a0 = init_scores[rank]
        J_init = compute_J(a0)
        print(f"  Init {rank} (idx={idx}): J_init={J_init:.6f} (budget {alloc:.0f}s)", flush=True)

        a_opt, obj_opt = solve_full_one(a0, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"    -> J={obj_opt:.8f}", flush=True)

    # === PHASE 2: Explore nearby lengths ===
    phase2_deadline = t_start + min(900.0, budget_s * 0.60)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Multi-length exploration (deadline t={phase2_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try lengths near parent and some round numbers
    nearby_ns = sorted(set([
        n_parent - 32, n_parent - 16, n_parent + 16, n_parent + 32,
        n_parent - 64, n_parent + 64,
        512, 640, 768, 896, 1024, 1152, 1280
    ]))
    nearby_ns = [n for n in nearby_ns if 128 <= n <= 2048 and n != n_parent]
    nearby_ns.sort(key=lambda x: abs(x - n_parent))

    for new_n in nearby_ns[:5]:
        remaining_check = phase2_deadline - time.monotonic()
        if remaining_check < 50:
            break
        alloc = min(80.0, remaining_check - 10)
        if alloc < 30:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt, obj_opt = solve_full_one(a_ext, time.monotonic() + alloc)
        update_best(a_opt)
        print(f"  n={new_n}: J={obj_opt:.8f}", flush=True)

    # === PHASE 3: Intensive polish on best ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Intensive polish (remaining {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and deadline - time.monotonic() > 30:
        # Multiple polish passes with different seeds/strategies
        a_curr = best_seq.copy()

        # Pass A: Full polish
        t_passA = time.monotonic() + min(200.0, (deadline - time.monotonic()) * 0.4)
        a_polished, obj_polished = solve_polish(a_curr, t_passA)
        update_best(a_polished)
        print(f"  Pass A: J={obj_polished:.8f}", flush=True)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Pass B: Perturb and re-optimize
            a_perturbed = project_simplex(a_curr + rng.normal(0, 0.01, len(a_curr)))
            t_passB = time.monotonic() + min(150.0, remaining * 0.4)
            a_polished2, obj_polished2 = solve_polish(a_perturbed, t_passB)
            update_best(a_polished2)
            print(f"  Pass B: J={obj_polished2:.8f}", flush=True)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Pass C: Try a slightly different length
            if best_seq is not None:
                n_best = len(best_seq)
                for dn in [16, -16, 32, -32]:
                    new_n = n_best + dn
                    if new_n < 64:
                        continue
                    a_try = resample_seq(best_seq, new_n)
                    a_try = np.maximum(a_try, 1e-6) / a_try.sum()
                    t_passC = time.monotonic() + min(60.0, remaining * 0.3)
                    a_res, obj_res = solve_full_one(a_try, t_passC)
                    update_best(a_res)
                    remaining = deadline - time.monotonic()
                    if remaining < 20:
                        break

        # Final: one more CD polish on absolute best
        if best_seq is not None and deadline - time.monotonic() > 15:
            a_final_cd = coord_descent_fast(best_seq, max_sweeps=5000,
                                            deadline=deadline - 5, seed=31415)
            update_best(a_final_cd)
            print(f"  Final CD: J={compute_J(a_final_cd):.8f}", flush=True)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
