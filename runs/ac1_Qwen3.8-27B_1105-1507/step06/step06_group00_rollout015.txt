import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        length = min(n, 2 * n - 1 - j)
        A[start:start + length, j] = a[:length]
    return A

def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS. Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    num_constraints = 2 * n - 1
    
    # Decision variables: g_0, ..., g_{n-1}, M
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A@g - M <= 0 for each row
    A_ub = np.zeros((num_constraints, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_constraints)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def solve_lp_topk(a, k=5):
    """
    Solve LP targeting top-k peaks more aggressively.
    Min M s.t. conv(a,g)[k] <= M for all k, but with tighter bounds on top-k.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    num_constraints = 2 * n - 1
    
    # First compute current peaks
    # Use uniform g as initial estimate
    g_uniform = np.ones(n) / n
    conv_current = A @ g_uniform
    peak_vals = np.sort(conv_current)[::-1][:k]
    target = float(np.min(peak_vals)) * 0.95  # Try to reduce top peaks by 5%
    
    # Modified LP: min M s.t. conv(a,g)[k] <= M for all k
    # But add extra constraints: conv(a,g)[top_k] <= target
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((num_constraints + k, n + 1))
    A_ub[:num_constraints, :n] = A
    A_ub[:num_constraints, n] = -1.0
    b_ub = np.zeros(num_constraints + k)
    
    # Add tighter constraints on top-k
    top_indices = np.argsort(conv_current)[::-1][:k]
    A_ub[num_constraints:num_constraints+k, :n] = A[top_indices]
    A_ub[num_constraints:num_constraints+k, n] = 0  # These are just <= target, not <= M
    b_ub[num_constraints:num_constraints+k] = target
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None

def subgradient_momentum(a, time_budget=60.0, momentum=0.9):
    """Subgradient descent with momentum targeting top peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    velocity = np.zeros(n)
    
    step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005]
    current_ss_idx = 0
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Target top-k peaks
        k_top = 3
        top_indices = np.argsort(conv)[::-1][:k_top]
        
        # Compute subgradient (average over top-k)
        grad = np.zeros(n)
        for k_star in top_indices:
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= k_top
        
        # Center (project onto sum=const)
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        alpha = step_sizes[current_ss_idx]
        velocity = momentum * velocity - alpha * grad_centered
        a_new = a + velocity
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                a = best_a.copy()
                velocity *= 0.5
                if current_ss_idx < len(step_sizes) - 1:
                    current_ss_idx += 1
        else:
            a = best_a.copy()
            velocity *= 0.5
            if current_ss_idx < len(step_sizes) - 1:
                current_ss_idx += 1
    
    return best_a

            if 0 <= j < n:
                peak_contributors.append((i, j, a[i] * a[j]))
        
        peak_contributors.sort(key=lambda x: -x[2])

                    continue
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[dest] += delta
                    new_conv = compute_conv(a)

                    else:
                        # Revert
                        a_backup = a.copy()
                        a[i] += delta
                        a[dest] -= delta
                        a[:] = a_backup
                        break
        
        if not improved:
            # Try adding small mass to underutilized peaks
            for _ in range(10):
                if time.monotonic() - start > time_budget:

                i = np.random.randint(0, n)

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, rng):
    """Generate diverse random initializations."""
    mode = rng.integers(0, 6)
    x = np.arange(n)
    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Smooth random
        freq = rng.uniform(0, 10, 5)
        phase = rng.uniform(0, 2*np.pi, 5)
        a = np.ones(n)
        for f, p in zip(freq, phase):
            a += np.cos(2*np.pi*f*x/n + p)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    elif mode == 2:
        # Gaussian bump
        center = rng.uniform(0.2*n, 0.8*n)
        width = rng.uniform(0.05*n, 0.3*n)
        a = np.exp(-((x - center)**2) / (2*width**2))
        a = a / np.sum(a)
    elif mode == 3:
        # Periodic
        period = rng.integers(max(2, n//20), n//3 + 1)
        a = 1.0 + 0.5 * np.cos(2*np.pi*x/period)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 4:
        # m-like sequence (binary smoothed)
        a = rng.choice([0.3, 1.0], n).astype(float)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = a / np.sum(a)
    else:
        # Golden ratio spacing
        golden = (1 + np.sqrt(5)) / 2
        a = np.sin(np.pi * (x * golden) % 1.0) ** 2 + 0.1
        a = a / np.sum(a)
    return a

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating with top-k peak targeting, momentum subgradient,
structured perturbations, and aggressive multi-n exploration.
Key improvements over prior approach:
- Top-k peak LP targeting for more aggressive peak suppression
- Momentum-based subgradient with adaptive step sizing

- More careful time allocation across phases

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2, normalized."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        length = min(n, 2 * n - 1 - j)
        A[start:start + length, j] = a[:length]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    num_cons = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((num_cons, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_cons)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_topk(a, k=5, reduction=0.97):
    """
    LP that additionally enforces tighter bounds on the top-k peaks.
    Returns g_normalized or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    num_cons = 2 * n - 1

    # Compute current peak values with uniform g
    g_uniform = np.ones(n) / n
    conv_current = A @ g_uniform
    peak_vals = np.sort(conv_current)[::-1]
    target = float(peak_vals[k - 1]) * reduction

    c = np.zeros(n + 1)
    c[-1] = 1.0

    top_indices = np.argsort(conv_current)[::-1][:k]
    total_cons = num_cons + k

    A_ub = np.zeros((total_cons, n + 1))
    A_ub[:num_cons, :n] = A
    A_ub[:num_cons, n] = -1.0
    b_ub = np.zeros(total_cons)

    A_ub[num_cons:num_cons + k, :n] = A[top_indices]
    b_ub[num_cons:num_cons + k] = target

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))

        k_star = int(np.argmax(conv))
        improved = False

        # Find top contributors to the peak
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((i, j, a[i] * a[j]))
        contribs.sort(key=lambda x: -x[2])

        for (i, j, _) in contribs[:15]:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-12:
                continue

            # Move mass from i to various destinations
            destinations = [i // 2, (i + 1) % n, n - 1 - i, (i + n // 3) % n]
            for dest in destinations:
                if dest == i or dest == j:
                    continue
                save = a.copy()
                for frac in [0.05, 0.1, 0.15]:
                    delta = frac * a[i]
                    if delta > a[i]:
                        continue
                    a[i] -= delta
                    a[dest] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[:] = save
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, mode, rng):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Smooth random spectrum
        freqs = rng.uniform(0, 20, 7)
        phases = rng.uniform(0, 2 * np.pi, 7)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 2:
        # Centered Gaussian
        center = rng.uniform(0.25 * n, 0.75 * n)
        width = rng.uniform(0.05 * n, 0.25 * n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = a / np.sum(a)
    elif mode == 3:
        # Periodic with random period
        period = rng.integers(max(3, n // 20), max(4, n // 4))
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x / period)
        a = np.maximum(a, 0.05)
        a = a / np.sum(a)
    elif mode == 4:
        # Flat spectrum (random phases)
        n_freq = n // 2 + 1
        phases = rng.uniform(0, 2 * np.pi, n_freq)
        spec = np.zeros(n_freq, dtype=complex)
        spec[1:-1] = np.exp(1j * phases[1:-1])
        spec[0] = 1.0
        a = irfft(spec, n).real
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif mode == 5:
        # Double bump
        c1, c2 = n // 3, 2 * n // 3
        w = n // 8
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a + 0.05
        a = a / np.sum(a)
    elif mode == 6:
        # Golden ratio modulation
        golden = (1 + np.sqrt(5)) / 2
        a = 0.8 + 0.4 * np.cos(2 * np.pi * (x / n) * golden * 10)
        a = np.maximum(a, 0.05)
        a = a / np.sum(a)
    else:
        a = rng.uniform(0.2, 1.0, n)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining LP alternating,
    subgradient with momentum, coordinate descent, and diverse explorations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Intensive LP on parent ===
    print(f"\n=== Phase 1: LP Alternating on Parent (n={len(best_seq)}) ===")
    a = best_seq.copy()
    n_parent = len(best_seq)

    for rd in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(70.0, remaining * 0.25)
        a_new = lp_alternate(a, time_budget=alloc, use_topk=True, topk_k=5)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP rd{rd}: obj={best_score:.8f} ***")
        else:
            print(f"  LP rd{rd}: obj={obj:.8f}")

    # Subgradient complement
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_sg = subgradient_momentum(a, time_budget=min(40.0, remaining * 0.12))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} ***")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # Coordinate refine
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd = coordinate_refine(a, time_budget=min(25.0, remaining * 0.1))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  Coord: obj={best_score:.8f} ***")
        else:
            print(f"  Coord: obj={obj_cd:.8f}")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    n_options = sorted(set([
        max(200, n_parent - 100), max(300, n_parent - 50),
        n_parent,
        n_parent + 50, n_parent + 100, n_parent + 150,
        500, 600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(50.0, remaining * 0.2)

        # LP then subgradient
        a_opt = lp_alternate(a_try, time_budget=alloc * 0.6, use_topk=True)
        obj_lp = compute_obj(a_opt)

        a_opt2 = subgradient_momentum(a_opt, time_budget=alloc * 0.4)
        obj_sg = compute_obj(a_opt2)

        best_local = a_opt if obj_lp <= obj_sg else a_opt2
        best_local_obj = min(obj_lp, obj_sg)

        if best_local_obj < best_score:
            best_score = best_local_obj
            best_seq = best_local.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} ***")
        else:
            print(f"  n={n_try}: obj={best_local_obj:.8f}")

    # === Phase 3: Diverse restarts ===
    print(f"\n=== Phase 3: Diverse Restarts ===")
    n_best = len(best_seq)

    for r in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        # Choose initialization mode
        if r < 4:
            # Perturbations of current best
            if r == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            elif r == 1:
                a_r = best_seq * (1.0 + 0.05 * np.cos(2 * np.pi * np.arange(n_best) / n_best * rng.integers(3, 20)))
            elif r == 2:
                a_r = best_seq.copy()
                for _ in range(20):
                    i, j = rng.integers(0, n_best, 2)
                    if i != j:
                        d = 0.05 * a_r[i]
                        a_r[i] -= d
                        a_r[j] += d
            else:
                # Different n
                n_r = max(300, min(1500, n_best + rng.integers(-150, 150)))
                a_r = resample_sequence(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.05, len(a_r)))
        else:
            # Truly diverse
            n_r = max(300, min(1500, n_best + rng.integers(-200, 300)))
            mode = (r - 4) % 7
            a_r = diverse_init(n_r, mode, rng)

        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r
        else:
            continue

        alloc = min(40.0, remaining * 0.2)
        a_opt = lp_alternate(a_r, time_budget=alloc * 0.55, use_topk=(r % 2 == 0))
        obj_opt = compute_obj(a_opt)

        # If close, try subgradient too
        if obj_opt < best_score * 1.001:
            a_opt2 = subgradient_momentum(a_opt, time_budget=alloc * 0.45)
            obj_opt2 = compute_obj(a_opt2)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  R{r} (n={len(a_opt)}, mode={r}): obj={best_score:.8f} ***")
        else:
            print(f"  R{r} (n={len(a_opt)}): obj={obj_opt:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        # Extended LP
        alloc = min(80.0, remaining * 0.3)
        a_lp = lp_alternate(a_final, time_budget=alloc, use_topk=True, topk_k=7)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} ***")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sg = subgradient_momentum(a_final, time_budget=min(30.0, remaining * 0.2))
            obj = compute_obj(a_sg)
            if obj < best_score:
                best_score = obj
                best_seq = a_sg.copy()
                a_final = a_sg.copy()
                print(f"  Final SG: obj={best_score:.8f} ***")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_cd = coordinate_refine(a_final, time_budget=min(20.0, remaining * 0.2))
            obj = compute_obj(a_cd)
            if obj < best_score:
                best_score = obj
                best_seq = a_cd.copy()
                a_final = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f} ***")

        # One final LP round
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_last = lp_alternate(a_final, time_budget=min(25.0, remaining * 0.3), use_topk=True)
            obj = compute_obj(a_last)
            if obj < best_score:
                best_score = obj
                best_seq = a_last.copy()
                print(f"  Final LP2: obj={best_score:.8f} ***")

    # Final validation and output
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
