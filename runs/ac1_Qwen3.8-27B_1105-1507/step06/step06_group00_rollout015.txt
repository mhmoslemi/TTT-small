def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

"""
Smoothed gradient (log-sum-exp) with mirror descent, combined with LP refinement.
Uses temperature annealing to converge to the true max-peak minimizer.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_smoothed_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*conv(a,a))) w.r.t. a.
    The gradient is proportional to 2 * cross_correlation(a, p) where p is softmax.
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    
    # Softmax with temperature T
    scaled = T * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()  # length 2n-1
    
    # Cross-correlation: (a ⋆ p)[i] = sum_j a[j] * p[j+i]
    # Using FFT
    N = 1
    while N < 3*n:
        N <<= 1
    A = rfft(a, N)
    P = rfft(p, N)
    cross_corr = irfft(np.conj(A) * P, N)[:n].real
    
    # Gradient = 2 * cross_corr (the factor 2 comes from d(conv[k])/d(a[i]) = 2*a[k-i])
    grad = 2.0 * cross_corr
    return grad, p, conv

def mirror_descent(a, time_budget=60.0, T_start=1.0, T_end=100.0, n_phases=10, steps_per_phase=100, lr=1.0, verbose=False):
    """
    Mirror descent (multiplicative updates) with log-sum-exp smoothing.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    total_steps = n_phases * steps_per_phase
    for step in range(total_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature (exponential schedule)
        frac = step / total_steps
        T = T_start * (T_end / T_start) ** frac
        
        grad, p, conv = compute_smoothed_grad(a, T)
        
        # Center gradient (to work on simplex)
        grad_centered = grad - np.mean(grad)
        
        # Adaptive learning rate
        grad_norm = np.abs(grad_centered).max()
        if grad_norm > 1e-15:
            lr_eff = lr / (grad_norm + lr)  # ensures lr * grad_norm <= lr
        else:
            lr_eff = lr
        
        # Mirror descent update: a * exp(-lr_eff * grad_centered)
        log_mult = -lr_eff * grad_centered
        log_mult = np.clip(log_mult, -20, 20)
        
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = a * np.exp(0.5 * log_mult)
            a_try = np.maximum(a_try, 1e-30)
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
            # If no improvement, keep a as is (or try to reduce lr)
        
        if verbose and step % 50 == 0:
            print(f"    MD step {step}: T={T:.2f}, obj={best_obj:.8f}")
    
    return best_a
