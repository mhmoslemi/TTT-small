def compute_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):

    sum_a = float(np.sum(a))

    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_refine(a, time_budget=30.0):

    n = len(a)
    if n > 900:
        return a
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    conv = compute_conv(a_norm)
    M = float(np.max(conv))
    
    A = build_toeplitz(a_norm)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': min(time_budget * 0.5, 20)})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            sum_g = float(np.sum(g))
            if sum_g > 1e-10:
                g = g / sum_g
                # Try blending
                best_a = a.copy()
                best_obj = compute_obj(a)
                for t in np.linspace(0.05, 1.0, 20):
                    if time.monotonic() - start > time_budget:
                        break
                    a_try = (1 - t) * a_norm + t * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a
    except Exception:
        pass
    return a


def adam_smooth(a, max_iters=5000, time_budget=100.0, beta_start=5.0,
                beta_end=500.0, lr_start=0.05, lr_decay=0.99995):
    """Adam optimization with logsumexp smoothing in log-space."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_start
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_c = x - np.max(x)
        a = np.exp(x_c)
        a = a / np.sum(a)
        
        # Smoothed objective and gradient
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip(beta * (conv - cmax), -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta
        
        size = 1
        while size < 2 * len(conv):
            size <<= 1
        W = rfft(w, size)
        A = rfft(a, size)
        cc = irfft(W * np.conj(A), size)
        grad_ls = 2.0 * cc[:n]
        
        S = np.sum(a)
        grad_a = (2.0 * n / (S * S)) * (grad_ls - ls / S)
        grad_x = grad_a * a
        
        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % 200 == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            x_cc = x - np.max(x)
            a_check = np.exp(x_cc)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def coordinate_refine(a, time_budget=30.0, seed=0):
    """Coordinate descent refinement focusing on peak-contributing elements."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    
    for outer in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        peak_idx = int(np.argmax(conv))
        
        # Find elements contributing most to the peak
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        contribs = []
        for m in range(lo, hi + 1):
            contribs.append((absolute_val := abs(best_a[m] * best_a[peak_idx - m]), m))
        contribs.sort(reverse=True)
        
        improved = False
        for _, m in contribs[:10]:
            if time.monotonic() - start > time_budget:
                break
            # Try moving mass from m to a random other position
            for trial in range(5):
                m2 = rng.integers(0, n)
                if m2 == m:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    delta = frac * best_a[m]
                    if delta > best_a[m] * 0.5:
                        continue
                    a_try = best_a.copy()
                    a_try[m] -= delta
                    a_try[m2] += delta
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved and outer > 50:
            break
    
    return best_a

Hybrid optimization combining high-beta Adam in log-space, exact subgradient
with Nesterov momentum, LP refinement with tighter constraints, water-filling,
coordinate descent, and multi-n exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def peak_gradient(a, conv, top_k=5, beta_sel=30.0):
    """
    Compute gradient of the peak (soft max) of conv w.r.t. a.
    Uses soft selection over top_k peaks.
    """
    n = len(a)
    cmax = float(np.max(conv))
    # Soft weights over peak positions
    shifted = beta_sel * (conv - cmax)
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / w_sum

    grad = np.zeros(n)
    # For each peak position k with weight w[k], gradient at m is 2*a[k-m] if valid
    # Efficient: convolve w with a
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    # cross-correlate: sum_k w[k] * a[k-m] for each m
    # This is: ifft(FFT(w) * conj(FFT(a))) gives sum_k w[k] * a[k+m]... 
    # Actually we want: for each m, sum_k w[k] * a[k-m]
    # = (w cross-correlate a)[m]
    # = ifft(conj(FFT(w)) * FFT(a)) shifted... let me just do it via the formula
    # grad[m] = 2 * sum_k w[k] * a[k-m] (where 0<=k-m<n)
    # This is 2 * (Conv(w, a)) where Conv is cross-correlation
    # cross_corr(w, a)[m] = sum_k w[k] * a[k+m] ... hmm
    
    # Let me just do it directly for clarity (n is at most ~2000)
    for k in range(len(conv)):
        if w[k] < 1e-10:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for m in range(lo, hi + 1):
            grad[m] += w[k] * 2.0 * a[k - m]
    
    return grad


def peak_gradient_fast(a, conv, top_k=5, beta_sel=30.0):
    """Faster version using FFT for the weighted sum."""
    n = len(a)
    cmax = float(np.max(conv))
    shifted = np.clip(beta_sel * (conv - cmax), -100, 100)
    w = np.exp(shifted)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        # Just use the single peak
        pk = int(np.argmax(conv))
        lo = max(0, pk - (n - 1))
        hi = min(n - 1, pk)
        grad = np.zeros(n)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[pk - m]
        return grad
    w = w / w_sum

    # grad[m] = 2 * sum_k w[k] * a[k-m] for valid k-m
    # = 2 * full cross-correlation of w and a, evaluated at appropriate lags
    # Using FFT: let's pad both to size >= len(conv) + n - 1
    size = 1
    while size < len(conv) + n - 1:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    # cross = ifft(conj(W) * A) gives cross-correlation
    # cross[m] = sum_k w[k] * a[k + m - (size-1)] ... this is getting confusing
    # Let me just use the direct approach but vectorized
    grad = np.zeros(n)
    # For each m in [0, n-1], grad[m] = 2 * sum_k w[k] * a[k-m] where 0 <= k-m < n
    # = 2 * sum_{j=0}^{n-1} w[m+j] * a[j] (where m+j < len(conv))
    for m in range(n):
        j_max = min(n, len(conv) - m)
        if j_max <= 0:
            continue
        j = np.arange(j_max)
        grad[m] = 2.0 * np.dot(w[m:m + j_max], a[:j_max])
    
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def nesterov_subgradient(a, max_iters=10000, time_budget=80.0, 
                          lr_start=0.02, momentum=0.9, beta_sel=30.0):
    """
    Nesterov-accelerated projected subgradient method.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # Velocity
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Decreasing step size
        lr = lr_start / (1.0 + 0.05 * it)
        
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute gradient at y
        conv_y = compute_conv(y)
        grad = peak_gradient_fast(y, conv_y, top_k=5, beta_sel=beta_sel)
        
        # Projected gradient step
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Check for improvement periodically
        if it % 200 == 0 or it == max_iters - 1:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def adam_logspace(a, max_iters=10000, time_budget=100.0, beta_start=10.0,
                  beta_end=5000.0, lr_start=0.04, lr_decay=0.99997,
                  center_every=200):
    """
    Adam optimization in log-space with logsumexp smoothing.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    x = np.log(a)
    x -= np.mean(x)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        x_c = x - np.max(x)
        a = np.exp(x_c)
        a = a / np.sum(a)
        
        # Smoothed objective and gradient via logsumexp
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip(beta * (conv - cmax), -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        w = e / e_sum
        ls = cmax + np.log(e_sum) / beta
        
        # Gradient via FFT cross-correlation
        size = 1
        while size < 2 * len(conv):
            size <<= 1
        W = rfft(w, size)
        A = rfft(a, size)
        cc = irfft(W * np.conj(A), size)
        grad_ls = 2.0 * cc[:n]
        
        S = np.sum(a)
        grad_a = (2.0 * n / (S * S)) * (grad_ls - ls / S)
        grad_x = grad_a * a
        
        t += 1
        lr_t = lr_start * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if it % center_every == 0:
            x -= np.mean(x)
        
        if it % 100 == 0 or it == max_iters - 1:
            x_cc = x - np.max(x)
            a_check = np.exp(x_cc)
            a_check = a_check / np.sum(a_check)
            true_obj = compute_obj(a_check)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_check.copy()
    
    return best_a


def mirror_descent(a, max_iters=20000, time_budget=80.0, 
                   T_start=0.5, T_end=0.05, beta_sel=30.0, lr_start=0.005):
    """
    Mirror descent (multiplicative weights) with entropic regularization.
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        progress = it / max(max_iters - 1, 1)
        T = T_start * (T_end / T_start) ** progress
        
        conv = compute_conv(a)
        grad = peak_gradient_fast(a, conv, top_k=3, beta_sel=beta_sel)
        
        # Entropic regularization
        a_safe = np.maximum(a, 1e-15)
        reg = (1.0 / T) * (1.0 + np.log(a_safe))
        
        total_grad = grad + reg
        step = lr_start / (1.0 + 0.02 * it)
        
        a_new = a * np.exp(-step * total_grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        a = a_new
        
        if it % 200 == 0 or it == max_iters - 1:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a


def build_toeplitz(a):
    """Build matrix A where (A @ g) = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_refine(a, time_budget=30.0, tightness=1.0):
    """
    LP refinement: find g maximizing sum(g) s.t. conv(a,g) <= M * tightness, g >= 0.
    If tightness < 1, forces the new peak below the current one.
    """
    n = len(a)
    if n > 1000:
        return a
    start = time.monotonic()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a_norm = a / s
    
    conv = compute_conv(a_norm)
    M = float(np.max(conv)) * tightness
    
    A = build_toeplitz(a_norm)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': min(time_budget * 0.5, 20)})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            sum_g = float(np.sum(g))
            if sum_g > 1e-10:
                g = g / sum_g
                best_a = a.copy()
                best_obj = compute_obj(a)
                for t in np.linspace(0.05, 1.0, 15):
                    if time.monotonic() - start > time_budget:
                        break
                    a_try = (1 - t) * a_norm + t * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a
    except Exception:
        pass
    return a


def coordinate_refine(a, time_budget=40.0, seed=0):
    """
    Coordinate descent: move mass from peak-contributing positions to non-contributing ones.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    
    stagnation = 0
    for outer in range(500):
        if time.monotonic() - start > time_budget:
            break
        if stagnation > 100:
            break
        
        conv = compute_conv(best_a)
        peak_idx = int(np.argmax(conv))
        
        # Find top contributors to the peak
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for m in range(lo, hi + 1):
            contrib[m] = best_a[m] * best_a[peak_idx - m]
        
        top_idx = np.argsort(contrib)[::-1][:min(20, n // 10)]
        
        improved = False
        for m in top_idx:
            if time.monotonic() - start > time_budget:
                break
            
            # Try moving mass to positions far from the peak contribution
            for trial in range(8):
                # Choose destination: prefer positions with low contribution to peak
                dest_candidates = np.where(contrib < np.median(contrib))[0]
                if len(dest_candidates) == 0:
                    dest_candidates = np.arange(n)
                m2 = dest_candidates[rng.integers(0, len(dest_candidates))]
                if m2 == m:
                    continue
                
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    delta = frac * best_a[m]
                    if delta >= best_a[m] * 0.3:
                        continue
                    a_try = best_a.copy()
                    a_try[m] -= delta
                    a_try[m2] += delta
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stagnation = 0
        else:
            stagnation += 1
    
    return best_a


def water_fill(a, n_iter=100, time_budget=30.0, move_frac=0.03):
    """
    Water-filling: move mass from peak contributors to valley positions.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for m in range(lo, hi + 1):
            contrib[m] = a[m] * a[peak_idx - m]
        
        # Top contributors (source)
        top_n = max(5, n // 10)
        sources = np.argsort(contrib)[::-1][:top_n]
        # Low contributors (destination)
        dests = np.argsort(contrib)[:top_n]
        
        # Move mass
        total_move = 0.0
        for m in sources:
            delta = move_frac * a[m]
            a[m] -= delta
            total_move += delta
        
        # Distribute to destinations proportional to their current value
        dest_sum = np.sum(a[dests])
        if dest_sum > 1e-10 and total_move > 0:
            for m in dests:
                a[m] += total_move * (a[m] / dest_sum)
        
        a = np.maximum(a, 0)
        s_new = np.sum(a)
        if s_new < 1e-10:
            break
        a = a / s_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase hybrid optimizer:
    1. High-beta Adam log-space (global)
    2. Nesterov subgradient (exact peak targeting)
    3. Mirror descent (multiplicative updates)
    4. LP refinement with tightness sweep
    5. Water-filling + coordinate descent (local)
    6. Multi-n exploration
    7. Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    if best_seq is None:
        n = 700
        best_seq = np.ones(n) / n
        best_score = compute_obj(best_seq)
        print(f"Uniform init: n={n}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")
    
    # === Phase 1: Adam log-space with very high beta ===
    print(f"\n{'='*50}")
    print(f"Phase 1: Adam logspace high-beta (n={n_best})")
    print(f"{'='*50}")
    
    a = best_seq.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    
    adam_configs = [
        # (beta_start, beta_end, lr, lr_decay, label)
        (5.0, 100.0, 0.05, 0.99996, "low"),
        (50.0, 500.0, 0.03, 0.99998, "mid"),
        (200.0, 2000.0, 0.015, 0.99999, "high"),
        (500.0, 5000.0, 0.008, 0.99999, "vhigh"),
        (1000.0, 10000.0, 0.004, 0.99999, "max"),
    ]
    
    for b_s, b_e, lr0, lr_d, label in adam_configs:
        if time.monotonic() > deadline - 250:
            break
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.12, 50.0)
        a_new = adam_logspace(a, max_iters=30000, time_budget=alloc,
                              beta_start=b_s, beta_end=b_e,
                              lr_start=lr0, lr_decay=lr_d)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Adam [{label}]: {best_score:.8f} ***")
        else:
            print(f"  Adam [{label}]: {obj:.8f}")
    
    # === Phase 2: Nesterov subgradient ===
    print(f"\n{'='*50}")
    print(f"Phase 2: Nesterov subgradient (n={len(best_seq)})")
    print(f"{'='*50}")
    
    if time.monotonic() < deadline - 200:
        remaining = deadline - time.monotonic()
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        
        ns_configs = [
            (0.01, 0.9, "v1"),
            (0.005, 0.95, "v2"),
            (0.02, 0.85, "v3"),
        ]
        
        for lr0, mom, label in ns_configs:
            if time.monotonic() > deadline - 150:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.15, 40.0)
            a_new = nesterov_subgradient(a, max_iters=50000, time_budget=alloc,
                                         lr_start=lr0, momentum=mom)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  Nesterov [{label}]: {best_score:.8f} ***")
            else:
                print(f"  Nesterov [{label}]: {obj:.8f}")
    
    # === Phase 3: Mirror descent ===
    print(f"\n{'='*50}")
    print(f"Phase 3: Mirror descent (n={len(best_seq)})")
    print(f"{'='*50}")
    
    if time.monotonic() < deadline - 150:
        remaining = deadline - time.monotonic()
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        
        for T_s, T_e, lr0, label in [(0.5, 0.1, 0.005, "T1"), (0.3, 0.05, 0.003, "T2"), (0.2, 0.03, 0.002, "T3")]:
            if time.monotonic() > deadline - 120:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.15, 35.0)
            a_new = mirror_descent(a, max_iters=30000, time_budget=alloc,
                                   T_start=T_s, T_end=T_e, lr_start=lr0)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  MD [{label}]: {best_score:.8f} ***")
            else:
                print(f"  MD [{label}]: {obj:.8f}")
    
    # === Phase 4: LP refinement with tightness sweep ===
    print(f"\n{'='*50}")
    print(f"Phase 4: LP refinement (n={len(best_seq)})")
    print(f"{'='*50}")
    
    if len(best_seq) <= 900 and time.monotonic() < deadline - 100:
        for tight in [1.0, 0.99, 0.98, 0.97, 0.95]:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.2, 30.0)
            a_new = lp_refine(best_seq, time_budget=alloc, tightness=tight)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  LP tight={tight}: {best_score:.8f} ***")
            else:
                print(f"  LP tight={tight}: {obj:.8f}")
            
            # Follow-up Adam
            if time.monotonic() < deadline - 30:
                remaining = deadline - time.monotonic()
                a_post = adam_logspace(a_new, max_iters=10000,
                                       time_budget=min(remaining * 0.1, 15.0),
                                       beta_start=200.0, beta_end=3000.0,
                                       lr_start=0.01, lr_decay=0.99999)
                obj_post = compute_obj(a_post)
                if obj_post < best_score:
                    best_score = obj_post
                    best_seq = a_post.copy()
                    print(f"  LP+Adam tight={tight}: {best_score:.8f} ***")
    
    # === Phase 5: Water-filling + Coordinate descent ===
    print(f"\n{'='*50}")
    print(f"Phase 5: Water-fill + Coord descent (n={len(best_seq)})")
    print(f"{'='*50}")
    
    if time.monotonic() < deadline - 80:
        remaining = deadline - time.monotonic()
        
        # Water-filling
        a_wf = water_fill(best_seq, n_iter=200, time_budget=min(remaining * 0.15, 25.0))
        obj_wf = compute_obj(a_wf)
        if obj_wf < best_score:
            best_score = obj_wf
            best_seq = a_wf.copy()
            print(f"  Water-fill: {best_score:.8f} ***")
        else:
            print(f"  Water-fill: {obj_wf:.8f}")
        
        # Coordinate descent
        if time.monotonic() < deadline - 40:
            remaining = deadline - time.monotonic()
            a_cd = coordinate_refine(best_seq, time_budget=min(remaining * 0.25, 35.0), seed=seed)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Coord: {best_score:.8f} ***")
            else:
                print(f"  Coord: {obj_cd:.8f}")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n{'='*50}")
    print(f"Phase 6: Multi-n exploration")
    print(f"{'='*50}")
    
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, n_best + 400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_best]
    
    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_seq)
        a_try = a_try / np.sum(a_try)
        
        alloc = min(remaining * 0.2, 25.0)
        a_opt = adam_logspace(a_try, max_iters=15000, time_budget=alloc,
                              beta_start=20.0, beta_end=2000.0,
                              lr_start=0.03, lr_decay=0.99998)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
        else:
            print(f"  n={n_try}: {obj:.8f}")
    
    # === Phase 7: Final intensive polish ===
    print(f"\n{'='*50}")
    print(f"Phase 7: Final polish (n={len(best_seq)})")
    print(f"{'='*50}")
    
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        
        # Very high beta Adam
        a_f1 = adam_logspace(a, max_iters=15000, time_budget=min(remaining * 0.35, 25.0),
                             beta_start=2000.0, beta_end=20000.0,
                             lr_start=0.003, lr_decay=0.99999)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final Adam: {best_score:.8f} ***")
        else:
            print(f"  Final Adam: {obj_f1:.8f}")
        
        # One more LP if applicable
        if len(best_seq) <= 900 and time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_lp = lp_refine(best_seq, time_budget=min(remaining * 0.4, 20.0), tightness=0.99)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.8f} ***")
        
        # Quick coordinate descent
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            a_cd = coordinate_refine(best_seq, time_budget=min(remaining * 0.5, 10.0), seed=seed + 1)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: {best_score:.8f} ***")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")
    
    return result
