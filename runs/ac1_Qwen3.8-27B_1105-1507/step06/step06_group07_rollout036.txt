def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def simplex_project(v):
    """Project v onto the simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_step(a, lr, momentum, top_k=5):

def heavy_ball_subgradient(a0, max_iters=5000, time_budget=60.0,

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lp_refine_multi(a, time_budget=60.0, n_passes=3):
    """
    Multi-pass LP refinement. Each pass:
    1. Solve LP to find direction g that maximizes sum s.t. conv(a,g) <= M
    2. Blend a with g
    3. Update M to the new (hopefully lower) peak
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for pass_idx in range(n_passes):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try slightly tighter M for aggressive reduction
        for M_factor in [1.0, 0.99, 0.98]:
            M_try = M * M_factor
            g = solve_lp(a, M_try, time_limit=min(remaining * 0.3, 15))
            if g is None:
                continue
            
            # Try different blend factors
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
            
            if g is not None:
                break
    
    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def fista_subgrad(a0, max_iters=10000, time_budget=60.0, L=2.0, top_k=3, lr_scale=1.0):
    """
    FISTA (Nesterov accelerated) for minimizing max(conv(a,a)) on the simplex.
    """
    n = len(a0)
    a = a0.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a0
    a = a / s
    
    best_a = a.copy()
    best_val = float(np.max(compute_conv(a)))
    
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient at y
        conv_y = compute_conv(y)
        # Find top-k peaks
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        top_vals = conv_y[top_idx]
        
        # Weight: exponential of (val - max) * top_k to focus on the max
        if top_k > 1:
            w = np.exp((top_vals - top_vals[0]) * top_k)
            w = w / w.sum()
        else:
            w = np.array([1.0])
        
        # Compute gradient
        grad = np.zeros(n)
        for idx, k in enumerate(top_idx):
            m_min = max(0, k - (n - 1))
            m_max = min(k, n - 1)
            if m_max >= m_min:
                j = k - np.arange(m_min, m_max + 1)
                grad[m_min:m_max + 1] += 2.0 * w[idx] * y[j]
        
        # FISTA step
        lr = lr_scale / (L * t)
        a_new = simplex_project(y - lr * grad)
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
        
        # Restart if objective increased significantly
        if it % 50 == 0:
            val_y = float(np.max(compute_conv(y_new)))
            if val_y > best_val * 1.1:
                y_new = a_new.copy()
                t_new = 1.0

"""
Hybrid optimizer combining Nesterov-accelerated subgradient descent (FISTA),
Adam in log-space with high-temperature logsumexp, aggressive multi-pass LP
refinement with decreasing peak constraints, and systematic n-exploration.
Key improvements over previous approach:
- FISTA with restart for O(1/t^2) rate on the smoothed problem
- More aggressive LP with geometrically decreasing M targets
- Two-phase strategy: exploration (moderate beta) then exploitation (very high beta)
- Better time management and adaptive hyperparameters
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def smooth_obj_grad(a, beta):
    """
    Compute smoothed objective (logsumexp of beta*conv) and its gradient w.r.t. a.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    cmax = float(np.max(conv))
    shifted = beta * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = float(np.sum(e))
    w = e / e_sum
    ls = cmax + np.log(e_sum + 1e-300) / beta

    obj = 2.0 * n * ls / (sum_a * sum_a)

    size = 1
    while size < 2 * len(conv):
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)

    grad_ls = 2.0 * cc[:n]
    grad_obj = (2.0 * n / (sum_a * sum_a)) * (grad_ls - ls / sum_a)

    return obj, grad_obj


def adam_log_opt(a, max_iters=5000, time_budget=60.0, beta_start=5.0,
                 beta_end=200.0, lr_start=0.05, lr_decay=0.99995):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    x = np.log(a)
    x -= np.mean(x)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start

    best_a = a.copy()
    best_true_obj = compute_obj(a)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        x_centered = x - np.max(x)
        a = np.exp(x_centered)
        a_s = np.sum(a)
        if a_s < 1e-300:
            break
        a = a / a_s

        obj_sm, grad_a = smooth_obj_grad(a, beta)
        grad_x = grad_a * a

        t += 1
        lr_t = lr * (lr_decay ** it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if it % 200 == 0:
            x -= np.mean(x)

        if it % 50 == 0 or it == max_iters - 1:
            x_c = x - np.max(x)
            a_check = np.exp(x_c)
            a_cs = np.sum(a_check)
            if a_cs > 1e-300:
                a_check = a_check / a_cs
                true_obj = compute_obj(a_check)
                if true_obj < best_true_obj:
                    best_true_obj = true_obj
                    best_a = a_check.copy()

    return best_a


def fista_smooth(a0, max_iters=10000, time_budget=60.0, beta=100.0,
                 lr_scale=0.5, restart_interval=50):
    """
    FISTA (Nesterov accelerated gradient) on the smoothed logsumexp objective,
    with periodic restarts. Works in the simplex directly.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = np.sum(a)
    if s < 1e-10:
        return a0
    a = a / s

    best_a = a.copy()
    best_val = float(np.max(compute_conv(a)))

    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    lr = lr_scale

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute smoothed gradient at y
        conv_y = compute_conv(y)
        sum_y = float(np.sum(y))
        cmax = float(np.max(conv_y))
        shifted = beta * (conv_y - cmax)
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        e_sum = float(np.sum(e))
        w = e / (e_sum + 1e-300)
        ls = cmax + np.log(e_sum + 1e-300) / beta

        size = 1
        while size < 2 * len(conv_y):
            size <<= 1
        W = rfft(w, size)
        A = rfft(y, size)
        cc = irfft(W * np.conj(A), size)
        grad_ls = 2.0 * cc[:n]
        grad = grad_ls - ls / sum_y

        # FISTA update
        a_new = simplex_project(y - lr * grad)

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y_new = a_new + momentum * (a_new - a_prev)

        # Restart check
        if (it + 1) % restart_interval == 0:
            val_new = float(np.max(compute_conv(a_new)))
            if val_new > best_val * 1.01:
                y_new = a_new.copy()
                t_new = 1.0
                momentum = 0.0
                y_new = a_new.copy()

        if (it + 1) % 20 == 0:
            val = float(np.max(compute_conv(a_new)))
            if val < best_val:
                best_val = val
                best_a = a_new.copy()

        a_prev = a_new
        a = a_new
        y = y_new
        t = t_new

    val_final = float(np.max(compute_conv(a)))
    if val_final < best_val:
        best_val = val_final
        best_a = a.copy()

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine_aggressive(a, time_budget=60.0, n_passes=5):
    """
    Aggressive multi-pass LP refinement with geometrically decreasing M.
    Each pass tries to reduce the peak by solving LP with M * factor.
    """
    n = len(a)
    start = time.monotonic()
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for pass_idx in range(n_passes):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Try decreasing M factors
        for M_factor in [1.0, 0.995, 0.99, 0.98, 0.97]:
            M_try = M * M_factor
            g = solve_lp(a, M_try, time_limit=min(remaining * 0.25, 12))
            if g is None:
                continue

            # Try multiple blend factors
            improvements = []
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                improvements.append((obj_try, t, a_try))

            if improvements:
                improvements.sort(key=lambda x: x[0])
                best_obj_try, best_t, best_a_try = improvements[0]
                if best_obj_try < best_obj:
                    best_obj = best_obj_try
                    best_a = best_a_try.copy()
                    a = best_a_try.copy()
            break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

            if remaining < 10:

            alloc = min(remaining * 0.4, budget_frac)
            a_new = adam_log_opt(a, max_iters=15000, time_budget=alloc,
                                 beta_start=b_start, beta_end=b_end,
                                 lr_start=lr0, lr_decay=lr_dec)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  HighBeta {h_idx} ({b_start:.0f}-{b_end:.0f}): {best_score:.8f} ***")
            else:
                print(f"  HighBeta {h_idx} ({b_start:.0f}-{b_end:.0f}): {obj:.8f}")

            # FISTA at high beta
            if time.monotonic() < deadline * 0.95 - 10:
                a_f = fista_smooth(a, max_iters=5000,
                                   time_budget=min(remaining * 0.2, 12.0),
                                   beta=b_end, lr_scale=0.1)
                obj_f = compute_obj(a_f)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    a = a_f.copy()
                    print(f"  FISTA high {h_idx}: {best_score:.8f} ***")

    # === Phase 4: N exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: N exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        n_options = [n for n in [500, 600, 700, 800, 900, 1000, 1100, 1200]
                     if n != n_best and 400 <= n <= 1200]
        n_options.sort(key=lambda x: abs(x - n_best))

        for n_try in n_options[:4]:
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_try = resample(best_seq, n_try)
            a_try = a_try / np.sum(a_try)

            # Quick Adam
            alloc = min(remaining * 0.3, 20.0)
            a_opt = adam_log_opt(a_try, max_iters=8000, time_budget=alloc,
                                 beta_start=20.0, beta_end=300.0,
                                 lr_start=0.03, lr_decay=0.99998)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  n={n_try}: {obj:.8f}")

    # === Phase 5: Final polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        n_final = len(best_seq)
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # Coordinate descent (if fast enough)
        if n_final <= 1000 and time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_cd = coordinate_refine(a, time_budget=min(remaining * 0.2, 20.0), seed=seed)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a = a_cd.copy()
                print(f"  Coord descent: {best_score:.8f} ***")
            else:
                print(f"  Coord descent: {obj_cd:.8f}")

        # One final very high beta Adam
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_final = adam_log_opt(a, max_iters=10000,
                                   time_budget=min(remaining * 0.5, 15.0),
                                   beta_start=500.0, beta_end=5000.0,
                                   lr_start=0.004, lr_decay=0.99999)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final Adam: {best_score:.8f} ***")

        # Final LP if n allows
        if n_final <= 800 and time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            a_lp = lp_refine_aggressive(best_seq, time_budget=min(remaining * 0.6, 10.0), n_passes=2)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.8f} ***")

    # === Final: Random restarts if time permits ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 1)
        remaining = deadline - time.monotonic()

        for restart in range(4):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            # Different perturbation
            if restart % 2 == 0:
                scale = 0.02 + 0.005 * restart
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
            else:
                alpha = rng.uniform(0.05, 0.15)
                # Raise-cosine window perturbation
                t_arr = np.arange(n_best) / max(n_best - 1, 1)
                window = np.cos(np.pi * (t_arr - 0.5)) ** 2
                window = np.maximum(window, 1e-8)
                window = window / window.sum()
                base = best_seq / best_seq.sum()
                a_pert = (1 - alpha) * base + alpha * window

            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()

            alloc = min(rem * 0.4, 10.0)
            a_opt = adam_log_opt(a_pert, max_iters=5000, time_budget=alloc,
                                 beta_start=50.0, beta_end=500.0,
                                 lr_start=0.02, lr_decay=0.99999)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Restart {restart}: {best_score:.8f} *** UPDATE")
            else:
                print(f"  Restart {restart}: {obj:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
