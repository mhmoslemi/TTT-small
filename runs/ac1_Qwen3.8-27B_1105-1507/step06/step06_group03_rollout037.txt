"""

import numpy as np
import time

from scipy.sparse import coo_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    lam = cssv[k[-1]] / (k[-1] + 1)
    return np.maximum(v - lam, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent_exact(a, deadline, max_sweeps=50000, random_order=True, seed=42, verbose=False):
    """
    Coordinate descent with exact 1D optimal step computation.
    For each coordinate i, analytically compute the best delta.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 50

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            old = a[i]
            if old < 1e-15:
                # Try small positive values
                for trial in [1e-6, 1e-5, 1e-4]:
                    delta = trial
                    new_s = s + delta
                    if new_s < 1e-10:
                        continue

                    lo = i
                    hi = min(2 * n - 2, i + n - 1)
                    a_slice = a[lo - i:hi - i + 1].copy()

                        a[i] = trial
                        s = new_s
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                continue

            # Find the best delta for this coordinate
            # The max of conv is achieved at some position k*
            # If k* is affected by a[i], we can compute optimal delta
            k_star = np.argmax(conv)
            delta_opt = None

            # Check if k_star is affected by coordinate i
            if i <= k_star <= i + n - 1:
                paired = k_star - i
                if 0 <= paired <= n - 1 and a[paired] > 1e-15:
                    # Optimal delta for reducing conv[k_star]

            # Find second max

            max_val = conv_sorted[0]

                # Clamp to valid range
                delta_clamped = max(-old, min(delta_opt, old * 5))
                candidates.append(delta_clamped)
                # Also try fraction of optimal
                candidates.append(max(-old, delta_clamped * 0.5))
                candidates.append(max(-old, delta_clamped * 0.7))

            # Also try setting to equalize top peaks

            if i <= k_star <= i + n - 1:
                paired = k_star - i
                if 0 <= paired <= n - 1 and a[paired] > 1e-15:
                    delta_eq = (second_max - conv[k_star]) / (2.0 * a[paired])
                    delta_eq = max(-old, min(delta_eq, old * 3))
                    candidates.append(delta_eq)
                    candidates.append(delta_eq * 0.5)
                    candidates.append(delta_eq * 0.8)

            # Evaluate candidates
            best_local_obj = best_obj
            best_delta = 0.0
            lo = i
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()

            for delta in candidates:

                if new_s < 1e-10:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice

                new_max = conv.max()
                new_obj = 2.0 * n * new_max / (new_s * new_s)
                conv[lo:hi + 1] -= 2.0 * delta * a_slice

                    conv[2 * i] -= delta * delta

                if new_obj < best_local_obj - 1e-14:
                    best_local_obj = new_obj
                    best_delta = delta

            if abs(best_delta) > 1e-14:
                conv[lo:hi + 1] += 2.0 * best_delta * a_slice
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += best_delta * best_delta
                a[i] = max(0.0, a[i] + best_delta)
                s = s + best_delta
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and sweep % 100 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
    return a


def smooth_gradient_descent_momentum(a, deadline, max_iters=20000, verbose=False):

    Momentum-accelerated gradient descent on smoothed max objective.
    Adaptive temperature based on peak distribution.

    beta = 0.9
    no_improve = 0
    lr = 0.05

            break

        # Nesterov extrapolation
        beta_new = (1 + np.sqrt(1 + 4 * beta * beta)) / 2

        beta = beta_new

        # Compute smoothed max
        c = fast_conv(y)
        c_max = c.max()

        tau = 1.0 / gap * 3.0

def active_set_lp(a, deadline, max_iters=100, verbose=False):
    """
    Active-set LP: identify top K peaks, solve LP to minimize max of those peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    K = min(20, 2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        # Find top K peak positions
        top_indices = np.argsort(c)[::-1][:K]
        M = c[top_indices[0]]

        # LP: minimize t
        # s.t. (a*g)[k] <= M * (t) for k in top_indices  (scaled)
        #      sum(g) = 1
        #      g >= 0
        # Actually: minimize max_{k in top} (a*g)[k] subject to sum(g)=1, g>=0
        # Linearize: min t s.t. (a*g)[k] <= t for k in top, sum(g)=1, g>=0
        
        rows = []
        cols = []
        vals = []
        b_ub = []

        for k in top_indices:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(len(b_ub))  # constraint index
                cols.append(j)
                vals.append(a[k - j])
            rows.append(len(b_ub))
            cols.append(n)  # t variable
            vals.append(-1.0)
            b_ub.append(float('inf'))  # will set to M initially, then optimize

        # Actually, let's set up: min t s.t. (a*g)[k] - t <= 0 for k in top, sum(g)=1, g>=0
        # This is: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0
        # With scipy linprog: min c_obj^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
        
        n_constraints = len(top_indices)
        A_ub_rows = []
        A_ub_cols = []
        A_ub_vals = []
        for idx, k in enumerate(top_indices):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                A_ub_rows.append(idx)
                A_ub_cols.append(j)
                A_ub_vals.append(a[k - j])
            A_ub_rows.append(idx)
            A_ub_cols.append(n)
            A_ub_vals.append(-1.0)

        A_ub = coo_matrix((A_ub_vals, (A_ub_rows, A_ub_cols)), shape=(n_constraints, n + 1)).tocsr()
        b_ub_vec = np.zeros(n_constraints)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), list(range(n)))), shape=(1, n + 1)).tocsr()
        b_eq_vec = [1.0]
        bounds = [(0, None)] * n + [(None, None)]

        try:
            r = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_vec, A_eq=A_eq, b_eq=b_eq_vec,
                        bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x[:n], 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    # Line search
                    found = False
                    for frac in [0.5, 0.7, 0.3, 0.9, 0.1]:
                        a_try = (1 - frac) * a + frac * g
                        a_try = np.maximum(a_try, 0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try /= a_try.sum()
                        J_try = compute_J(a_try)
                        if J_try < best_obj - 1e-14:
                            a = a_try
                            best_obj = J_try
                            best_a = a_try.copy()
                            no_improve = 0
                            found = True
                            break
                    if not found:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj

    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1]:
            inits.append(project_simplex(p + rng.normal(0, sigma, n)))

    inits.append(np.ones(n) / n)

    # Cosine with edge emphasis
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(project_simplex(1.0 + amp * np.cos(np.pi * t)))
            inits.append(project_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t)))

    # Two-lobe patterns
    for ratio in [0.3, 0.4, 0.6]:
        inits.append(project_simplex((t < ratio) * 1.0 + (t >= ratio) * 0.3))
        inits.append(project_simplex(np.exp(-((t - ratio) / 0.15)**2) + np.exp(-((t - (1-ratio)) / 0.15)**2) + 0.05))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for best sequence minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses exact coordinate descent, momentum gradient, active-set LP, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    n_parent = parent.size if parent is not None else 512

    if parent is None:
        parent = np.ones(512) / 512.0
        n_parent = 512
        update_best(parent)

    # === PHASE 1: Quick optimization at parent length (300s) ===
    phase1_deadline = t_start + 300.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Optimize at n={n_parent} (budget {phase1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a_curr = parent.astype(np.float64).copy()
    a_curr = np.maximum(a_curr, 1e-10)
    a_curr = a_curr / a_curr.sum()

    # Step 1: Coordinate descent
    t_cd = time.monotonic() + 0.4 * (phase1_deadline - time.monotonic())
    a_curr = coord_descent_exact(a_curr, deadline=t_cd, max_sweeps=30000, seed=42)
    update_best(a_curr)

    # Step 2: Smooth gradient with momentum
    remaining = phase1_deadline - time.monotonic()
    if remaining > 30:
        t_grad = time.monotonic() + 0.5 * remaining
        a_g, obj_g = smooth_gradient_descent_momentum(a_curr, deadline=t_grad, max_iters=15000)
        update_best(a_g)
        a_curr = best_seq.copy() if best_seq is not None else a_curr

    # Step 3: Active set LP
    remaining = phase1_deadline - time.monotonic()
    if remaining > 20:
        a_lp, obj_lp = active_set_lp(a_curr, deadline=time.monotonic() + remaining * 0.8, max_iters=50)
        update_best(a_lp)
        a_curr = best_seq.copy() if best_seq is not None else a_curr

    # === PHASE 2: Multi-length exploration (400s) ===
    phase2_deadline = t_start + 700.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length exploration (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    candidate_ns = []
    for n in [n_parent - 64, n_parent - 32, n_parent + 32, n_parent + 64,
              n_parent + 128, n_parent + 256, n_parent + 512]:
        if 64 <= n <= 4096 and n != n_parent:
            candidate_ns.append(n)
    # Add some "round" numbers
    for n in [512, 640, 768, 896, 1024, 1280, 1536]:
        if 64 <= n <= 4096 and n not in candidate_ns and n != n_parent:
            candidate_ns.append(n)
    candidate_ns = sorted(set(candidate_ns))[:6]

    for new_n in candidate_ns:
        remaining = phase2_deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(120.0, remaining - 20)
        if alloc < 40:
            break

        print(f"  Trying n={new_n} (budget {alloc:.0f}s)...", flush=True)
        a_ext = resample_seq(best_seq if best_seq is not None else parent, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        # Quick optimization: CD + gradient
        t_cd2 = time.monotonic() + 0.4 * alloc
        a_ext = coord_descent_exact(a_ext, deadline=t_cd2, max_sweeps=20000, seed=new_n)
        update_best(a_ext)

        remaining = alloc - (time.monotonic() - (time.monotonic() - alloc))
        if time.monotonic() < time.monotonic() + 0.5 * alloc:
            a_g2, _ = smooth_gradient_descent_momentum(a_ext, deadline=time.monotonic() + 0.4 * alloc, max_iters=8000)
            update_best(a_g2)
            a_best_local = best_seq.copy() if best_seq is not None else a_ext

        if time.monotonic() < phase2_deadline - 20:
            a_lp2, _ = active_set_lp(a_best_local, deadline=min(time.monotonic() + 30, phase2_deadline - 10), max_iters=30)
            update_best(a_lp2)

    # === PHASE 3: Intensive local optimization (remaining time) ===
    phase3_deadline = deadline
    remaining = phase3_deadline - time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Intensive optimization (budget {remaining:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and remaining > 30:
        a_curr = best_seq.astype(np.float64).copy()
        a_curr = np.maximum(a_curr, 0)
        if a_curr.sum() < 1e-10:
            a_curr = np.ones_like(a_curr)
        a_curr = a_curr / a_curr.sum()

        # Alternate between methods
        n_passes = 0
        while time.monotonic() < phase3_deadline - 20:
            n_passes += 1
            remaining = phase3_deadline - time.monotonic()
            if remaining < 15:
                break

            # Pass A: Coordinate descent
            t_pass = time.monotonic() + min(remaining * 0.3, 100.0)
            a_curr = coord_descent_exact(a_curr, deadline=t_pass, max_sweeps=50000, seed=1000 + n_passes)
            update_best(a_curr)

            if time.monotonic() >= phase3_deadline - 20:
                break

            # Pass B: Gradient with momentum
            remaining = phase3_deadline - time.monotonic()
            if remaining < 15:
                break
            t_pass = time.monotonic() + min(remaining * 0.3, 80.0)
            a_g, _ = smooth_gradient_descent_momentum(a_curr, deadline=t_pass, max_iters=10000)
            update_best(a_g)
            a_curr = best_seq.astype(np.float64).copy() if best_seq is not None else a_curr
            a_curr = np.maximum(a_curr, 0)
            if a_curr.sum() < 1e-10:
                a_curr = np.ones_like(a_curr)
            a_curr = a_curr / a_curr.sum()

            if time.monotonic() >= phase3_deadline - 20:
                break

            # Pass C: Active set LP
            remaining = phase3_deadline - time.monotonic()
            if remaining < 15:
                break
            t_pass = time.monotonic() + min(remaining * 0.25, 60.0)
            a_lp, _ = active_set_lp(a_curr, deadline=t_pass, max_iters=40)
            update_best(a_lp)
            a_curr = best_seq.astype(np.float64).copy() if best_seq is not None else a_curr
            a_curr = np.maximum(a_curr, 0)
            if a_curr.sum() < 1e-10:
                a_curr = np.ones_like(a_curr)
            a_curr = a_curr / a_curr.sum()

    # Final fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.10f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
