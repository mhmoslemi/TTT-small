```python
"""
Multi-strategy optimization using L-BFGS with log-sum-exp smoothing,
LP polishing, and diverse restarts. Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax(b):
    b = b - np.max(b)
    e = np.exp(b)
    return e / np.sum(e)

def smooth_obj(b, tau):
    a = softmax(b)
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    val = 2.0 * n * (cmax + tau * np.log(np.sum(np.exp((conv - cmax) / tau))))
    return val

def smooth_grad(b, tau):
    a = softmax(b)
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    shifted = (conv - cmax) / tau
    weights = np.exp(shifted)
    wsum = weights.sum()
    weights /= wsum
    
    # Compute grad_a = 2 * correlate(weights, a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    Wf = rfft(weights, size)
    Af = rfft(a[::-1], size)
    corr = irfft(Wf * Af, size)
    grad_a = 2.0 * corr[n-1:2*n-1]
    
    # Chain rule through softmax
    grad_b = a * (grad_a - np.dot(grad_a, a))
    grad_b *= 2.0 * n
    return grad_b

def smooth_obj_grad(b, tau):
    a = softmax(b)
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    shifted = (conv - cmax) / tau
    weights = np.exp(shifted)
    wsum = weights.sum()
    weights /= wsum
    
    val = 2.0 * n * (cmax + tau * np.log(wsum))
    
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    Wf = rfft(weights, size)
    Af = rfft(a[::-1], size)
    corr = irfft(Wf * Af, size)
    grad_a = 2.0 * corr[n-1:2*n-1]
    
    grad_b = a * (grad_a - np.dot(grad_a, a))
    grad_b *= 2.0 * n
    return val, grad_b
```

```python
def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except:
        pass
    return None

def golden_section(f, lo, hi, tol=0.01, max_iter=20):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo); fc = f(c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo); fd = f(d)
    return (lo + hi) / 2.0

def lp_alternate(a, max_iters=30, time_budget=60.0):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break
        
        def obj_at_t(t):
            at = (1.0 - t) * a + t * g
            at = normalize(at)
            return compute_obj(at)
        
        t_opt = golden_section(obj_at_t, 0.0, 1.0, tol=0.02, max_iter=12)
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a
```

```python
"""
```

```python
momentum-based subgradient, diverse structured restarts, and adaptive n-sweep.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_fn(b):
    b = b - np.max(b)
    e = np.exp(b)
    return e / np.sum(e)


def smooth_obj_grad(b, tau):
    a = softmax_fn(b)
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    shifted = (conv - cmax) / tau
    weights = np.exp(shifted)
    wsum = float(np.sum(weights))
    weights /= wsum
    val = 2.0 * n * (cmax + tau * np.log(wsum))
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    Wf = rfft(weights, size)
    Af = rfft(a[::-1], size)
    corr = irfft(Wf * Af, size)
    grad_a = 2.0 * corr[n - 1:2 * n - 1]
    grad_b = a * (grad_a - np.dot(grad_a, a))
    grad_b *= 2.0 * n
    return val, grad_b

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_for_g(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section_search(f, lo, hi, tol=0.01, max_iter=20):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def obj_at_blend(base, target, t):

def momentum_subgradient(a, steps=100, time_budget=30.0):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    velocity = np.zeros(n)
    beta = 0.9
    start = time.monotonic()
    lr = 0.02

    for step in range(steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            k2 = k_star - i
            if 0 <= k2 < n:
                grad[i] = 2.0 * a[k2]

        grad_centered = grad - np.mean(grad)
        velocity = beta * velocity + (1 - beta) * grad_centered
        a_new = normalize(a - lr * velocity)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.8

        lr = max(lr, 0.001) * 0.995

    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

        a_new = lp_alternate(a_work, max_iters=15, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_work = a_new
            print(f"  LP cycle {cycle}: {best_score:.8f} **")
        else:
            a_work = a_new
            print(f"  LP cycle {cycle}: {obj:.8f}")

    # Phase 3: Explore neighboring n with LP
    print(f"\n{'='*60}")
    print(f"Phase 3: n-exploration with LP")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    for dn in [-75, -50, -25, 25, 50, 75, -100, 100]:
        if time.monotonic() > deadline - 150:
            break
        n_try = n_cur + dn
        if n_try < 100 or n_try > 1200:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.1, 40.0)
        if alloc < 8:
            continue

        a_try = normalize(resample(best_seq, n_try))
        a_try = lp_alternate(a_try, max_iters=10, time_budget=alloc)
        obj = compute_obj(a_try)
        if obj < best_score:
            best_score = obj
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj:.8f}")

    # Phase 4: Momentum subgradient + L-BFGS refinement
    print(f"\n{'='*60}")
    print(f"Phase 4: Momentum + L-BFGS refinement")
    print(f"{'='*60}")

    for rnd in range(4):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        a_work = normalize(best_seq.copy())

        # Momentum
        a_mom = momentum_subgradient(a_work, steps=80, time_budget=min(20, remaining * 0.2))
        obj_m = compute_obj(a_mom)
        if obj_m < best_score:
            best_score = obj_m
            best_seq = a_mom.copy()
            a_work = a_mom
        else:
            a_work = 0.7 * normalize(a_work) + 0.3 * normalize(a_mom)

        # L-BFGS fine
        a_lb = lbfgs_optimize(a_work, [0.01, 0.005, 0.002], max_iter_per_tau=200,
                              time_budget=min(30, remaining * 0.3))
        obj_lb = compute_obj(a_lb)
        if obj_lb < best_score:
            best_score = obj_lb
            best_seq = a_lb.copy()
            a_work = a_lb

        # LP short
        if remaining > 20 and len(best_seq) <= 800:
            a_lp = lp_alternate(a_work, max_iters=5, time_budget=min(15, remaining * 0.2))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_work = a_lp

        print(f"  Round {rnd}: {best_score:.8f}")

    # Phase 5: Diverse restarts
    print(f"\n{'='*60}")
    print(f"Phase 5: Diverse restarts")
    print(f"{'='*60}")

    n_cur = len(best_seq)
    for restart in range(12):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break
        alloc = min(remaining * 0.08, 30.0)
        if alloc < 8:
            continue

        strategy = restart % 6
        if strategy == 0:
            a_try = best_seq * np.exp(rng.normal(0, 0.04, n_cur))
        elif strategy == 1:
            freq = rng.uniform(1.0, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.08 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            a_try = best_seq * mod
        elif strategy == 2:
            a_struct = structured_init(n_cur, int(rng.integers(0, 6)), rng)
            mix = rng.uniform(0.3, 0.6)
            a_try = (1 - mix) * normalize(best_seq) + mix * a_struct
        elif strategy == 3:
            n_new = n_cur + int(rng.integers(-100, 100))
            n_new = max(100, min(1200, n_new))
            a_try = resample(best_seq, n_new)
        elif strategy == 4:
            a_try = best_seq.copy()
            blk = int(rng.integers(n_cur // 5, n_cur // 3))
            st = int(rng.integers(0, n_cur - blk))
            a_try[st:st + blk] = a_try[st:st + blk][::-1]
        else:
            a_try = best_seq * (1.0 + 0.05 * rng.standard_t(3, n_cur))

        a_try = normalize(a_try)
        n_try = len(a_try)

        # Quick optimization
        a_try = lbfgs_optimize(a_try, [0.02, 0.01, 0.005], max_iter_per_tau=80,
                               time_budget=alloc * 0.5)
        if n_try <= 700 and alloc > 10:
            a_try = lp_alternate(a_try, max_iters=4, time_budget=alloc * 0.4)

        obj = compute_obj(a_try)
        if obj < best_score:
            best_score = obj
            best_seq = a_try.copy()
            print(f"  Restart {restart} (n={n_try}): {best_score:.8f} **")
        else:
            print(f"  Restart {restart} (n={n_try}): {obj:.8f}")

    # Phase 6: Final intensive LP polish
    print(f"\n{'='*60}")
    print(f"Phase 6: Final LP polish")
    print(f"{'='*60}")

    a_work = normalize(best_seq.copy())
    for final in range(5):
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining * 0.2, 45.0)
        if alloc < 8:
            break

        if len(a_work) <= 900:
            a_new = lp_alternate(a_work, max_iters=10, time_budget=alloc)
        else:
            a_new = lbfgs_optimize(a_work, [0.005, 0.002], max_iter_per_tau=300,
                                   time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_work = a_new
            print(f"  Final {final}: {best_score:.8f} **")
        else:
            a_work = a_new
            print(f"  Final {final}: {obj:.8f}")

    # Result
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
```
