import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s*s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_gradient(a, tau):
    """Compute gradient of 2n*tau*log(sum exp(q_k/tau)) at a.
    Returns (gradient, conv, p) where p is softmax weights."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    exp_vals = np.exp((conv - max_c) / tau)
    p = exp_vals / np.sum(exp_vals)
    
    # Gradient: 4n * (a_rev * p)[n-1 : 2n-1]
    a_rev = a[::-1]
    size = 1
    while size < 3*n - 1:
        size <<= 1
    ar_f = rfft(a_rev, size)
    p_f = rfft(p, size)
    conv_res = irfft(ar_f * p_f, size)
    grad = 4.0 * n * conv_res[n-1:2*n-1]
    return grad, conv, p

def nesterov_lse(a, tau, step_size, n_iters, time_budget):
    """Nesterov-accelerated gradient descent on log-sum-exp objective."""
    n = len(a)
    a = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = 0.0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov lookahead
        y = a + momentum * (a - a_prev)
        
        # Gradient at y
        grad, conv, p = lse_gradient(y, tau)
        
        # Step
        a_new = y - step_size * grad
        a_new = project_simplex(a_new)
        
        # Update momentum (Nesterov)
        t_cur = (1 + np.sqrt(1 + 4*momentum**2)) / 2
        new_momentum = (t_cur - 1) / t_cur
        # Actually, standard Nesterov: momentum = (1 - t_cur) / (1 + t_cur) ... 
        # Let me use the standard form: y = a + (t_{k-1} - 1)/t_k * (a - a_prev)
        # where t_k = (1 + sqrt(1 + 4*t_{k-1}^2))/2
        
        a_prev = a
        a = a_new
        momentum = new_momentum
        
        # Check improvement
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

"""
Optimized step function using log-sum-exp gradient descent with L-BFGS tau-annealing,
LP polishing, multi-n exploration, and diverse restart strategies.
Key innovation: fast LSE gradient via FFT cross-correlation, enabling many more
optimization steps than the LP-only approach of the previous state of the art.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_func_and_grad_x(x, tau, n):
    """
    LSE objective and gradient w.r.t. x (softmax parameterization).
    a = softmax(x), objective = 2n*tau*log(sum exp(q_k/tau))
    """
    x = np.asarray(x, dtype=np.float64)
    # Stable softmax
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()

    # Convolution
    conv = compute_conv(a)

    # Log-sum-exp
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    log_sum_exp = max_c / tau + np.log(sum_exp)
    obj = 2.0 * n * tau * log_sum_exp

    # Softmax weights p_k
    p = exp_shifted / sum_exp

    # Gradient w.r.t. a: 4n * cross_corr(a, p)
    a_rev = a[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    ar_f = rfft(a_rev, size)
    p_f = rfft(p, size)
    conv_res = irfft(ar_f * p_f, size)
    grad_a = 4.0 * n * conv_res[n - 1:2 * n - 1]

    # Gradient w.r.t. x: a * (grad_a - dot(grad_a, a))
    dot_ga = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - dot_ga)

    return obj, grad_x


def lse_optimize_bfgs(a, tau, max_iter=80, time_budget=30.0):
    """Optimize LSE objective using L-BFGS-B with softmax parameterization."""
    n = len(a)
    x0 = np.log(a + 1e-300)

    start = time.monotonic()
    state = {'count': 0}

    def fun_wrapped(x):
        state['count'] += 1
        if time.monotonic() - start > time_budget:
            return 1e10, np.zeros(n)
        obj, grad = lse_func_and_grad_x(x, tau, n)
        return obj, grad

    try:
        res = minimize(fun_wrapped, x0, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                'gtol': 1e-8, 'ftol': 1e-12})
        if res.x is not None:
            x_opt = res.x
            x_shifted = x_opt - np.max(x_opt)
            a_opt = np.exp(x_shifted)
            a_opt = a_opt / a_opt.sum()
            a_opt = np.maximum(a_opt, 0.0)
            return normalize(a_opt)
    except Exception:
        pass
    return a


def lse_anneal(a, time_budget=300.0, verbose=True):
    """
    Main LSE optimizer with geometric tau annealing.
    Starts with large tau (smooth) and decreases to small tau (sharp).
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initial peak for tau scale
    conv0 = compute_conv(a)
    M0 = float(np.max(conv0))

    # Tau schedule: geometric from M0/4 to M0/2000
    tau_max = M0 / 4.0
    tau_min = M0 / 2000.0
    n_levels = 12
    tau_schedule = tau_max * (tau_min / tau_max) ** (np.arange(n_levels) / (n_levels - 1))

    for level, tau in enumerate(tau_schedule):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break

        remaining = time_budget - elapsed
        alloc = min(remaining * 0.3, 40.0)

        a_opt = lse_optimize_bfgs(a, tau, max_iter=60, time_budget=alloc)
        obj_opt = compute_obj(a_opt)

        if verbose:
            print(f"    LSE level {level}: tau={tau:.6f}, obj={obj_opt:.8f}")

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
        a = a_opt

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, m)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section(f, lo, hi, max_iter=20):
    """Golden section search for minimum on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 0.01:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_polish(a, max_iters=20, time_budget=60.0):
    """LP-based polishing with golden section line search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit=max(3, int(remaining * 0.5)))
        if g is None:
            break

        def obj_at_t(t):
            a_try = normalize((1.0 - t) * a + t * g)
            return compute_obj(a_try)

        t_opt = golden_section(obj_at_t, 0.0, 1.0, max_iter=15)
        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Small perturbation to escape
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def spectral_shape(a, max_iters=40, time_budget=30.0):
    """Spectral shaping: flatten the power spectrum to reduce convolution peaks."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        alpha = 0.4 + 0.4 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = normalize(0.6 * a + 0.4 * a_new)

    return best_a


def nesterov_projected(a, tau, n_iters=500, time_budget=30.0):
    """Nesterov-accelerated projected subgradient for the LSE objective."""
    n = len(a)
    a = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_val = 1.0
    step_base = 1.0 / (4.0 * n * np.max(compute_conv(a)))

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        t_val = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum = (t_val - 1.0) / t_val
        t_inv = 1.0 / t_val

        # Nesterov lookahead
        y = a + momentum * (a - a_prev)

        # Gradient at y
        conv_y = compute_conv(y)
        max_c = float(np.max(conv_y))
        shifted = (conv_y - max_c) / tau
        exp_s = np.exp(shifted)
        p = exp_s / exp_s.sum()

        a_rev = y[::-1]
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        ar_f = rfft(a_rev, size)
        p_f = rfft(p, size)
        conv_res = irfft(ar_f * p_f, size)
        grad = 4.0 * n * conv_res[n - 1:2 * n - 1]

        # Step with projection
        a_new = project_simplex(y - t_inv * step_base * grad)
        a_prev = a
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)

def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    method = rng.integers(0, 6)
    if method == 0:
        # Uniform
        return np.ones(n) / n
    elif method == 1:
        # Gaussian-like (centered)
        x = np.arange(n) - n / 2.0
        a = np.exp(-x ** 2 / (2 * (n / 4) ** 2))
        return a / a.sum()
    elif method == 2:
        # Two bumps
        x = np.arange(n)
        a = np.exp(-((x - n * 0.25)) ** 2 / (2 * (n / 8) ** 2)) + \
            np.exp(-((x - n * 0.75)) ** 2 / (2 * (n / 8) ** 2))
        return a / a.sum()
    elif method == 3:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        return a / a.sum()
    elif method == 4:
        # Random Dirichlet
        return rng.dirichlet(np.ones(n) * 0.5)
    else:
        # Triangular
        a = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
        a += 1e-6
        return a / a.sum()

    print(f"Phase 6: Final polish")
    print(f"{'='*60}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final = normalize(best_seq.copy())

            # Check peak structure for targeted optimization
            conv_f = compute_conv(a_final)
            n_f = len(a_final)
            M_f = float(np.max(conv_f))
            peak_indices = np.where(conv_f > 0.95 * M_f)[0]
            print(f"  Active peaks: {len(peak_indices)} at positions {peak_indices[:10]}")

            # Final LSE with very small tau
            tau_very_small = M_f / 5000.0
            a_lse_final = lse_optimize_bfgs(a_final, tau_very_small, max_iter=50,
                                            time_budget=min(20, remaining * 0.3))
            obj_f1 = compute_obj(a_lse_final)
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_lse_final.copy()
            a_final = a_lse_final

            # LP polish
            if len(a_final) <= 900 and remaining > 30:
                a_lp_final = lp_polish(a_final, max_iters=8,
                                       time_budget=min(30, remaining * 0.4))
                obj_f2 = compute_obj(a_lp_final)
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_lp_final.copy()

            # Nesterov with tiny tau
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_nest = nesterov_projected(best_seq, M_f / 10000.0,
                                            n_iters=500, time_budget=min(15, remaining2 * 0.5))
                obj_f3 = compute_obj(a_nest)
                if obj_f3 < best_score:
                    best_score = obj_f3
                    best_seq = a_nest.copy()

            print(f"  Final: {best_score:.8f}")

    # =========================================================
    # Final validation and return
    # =========================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
