def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

    rho = rho[-1]

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative updates), LP alternating, and subgradient methods.
Exploits top-K peak targeting, adaptive learning rates, and multi-scale exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_topK(a, K=3, temperature=2.0):
    """Compute subgradient using top-K peaks with softmax weighting."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights over top peaks
    top_idx = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n), int(top_idx[0])
    weights = np.exp((peak_vals - max_val) * temperature / max_val)
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
    return g, int(top_idx[0])


def subgradient_single(a, k):
    """Compute subgradient for a single peak position k."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def mirror_descent_optimize(a, time_budget=200.0, topK=3, verbose=False):
    """
    Mirror descent (multiplicative updates) on the probability simplex.
    Minimizes max_k conv(a,a)[k] using top-K peak subgradients.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive learning rate
    lr = 1.0 / np.sqrt(n) * 2.0
    min_lr = 1e-4 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)
    stall_count = 0
    patience = 200

    it = 0
    while time.monotonic() - start < time_budget:
        if it % 100 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

        g, k_star = subgradient_topK(a, K=topK)

        # Momentum update
        v = momentum * v + g
        a_try = a - lr * v
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try
        else:
            a_try = best_a.copy()

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            stall_count = 0
            # Slowly decay learning rate
            lr = max(min_lr, lr * 0.999)
            # Increase momentum gradually
            momentum = min(0.9, momentum + 0.001)
        else:
            stall_count += 1
            if stall_count > patience:
                # Perturb to escape
                rng = np.random.default_rng(it * 7919 + n)
                scale = min(0.5, 0.05 + 0.02 * (stall_count // patience))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0
                lr = max(min_lr, lr * 0.5)
                stall_count = 0
            else:
                lr = max(min_lr, lr * 0.95)

        it += 1

    return best_a


def multiplicative_update_optimize(a, time_budget=200.0, verbose=False):
    """
    Multiplicative (entropy mirror) descent.
    a_new[i] = a[i] * exp(-lr * w[i]) / Z
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    # Ensure no exact zeros (for log)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    stall = 0
    K = 3

    it = 0
    while time.monotonic() - start < time_budget:
        if it % 200 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    MU it{it}: obj={best_obj:.8f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

        g, k_star = subgradient_topK(a, K=K)
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * g
        # Normalize (softmax)
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        obj_new = compute_obj(a_new)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def solve_lp_max_sum(a, M_target):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target for all k, g >= 0.
    Returns g_normalized or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_target]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue

    return None

def chase_peak_refine(a, time_budget=60.0, verbose=False):
    """
    Chase-the-peak: iterative subgradient step targeting the maximum convolution bin.
    Uses Barzilai-Borwein step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = a.copy()
    prev_g = None
    step_size = 0.05 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        g = subgradient_single(a, k_star)

        # Barzilai-Borwein step size
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-15:
                bb_step = ss / abs(sy)
                step_size = min(max(step_size * 0.5, 1e-6), bb_step, 1.0)
        prev_a = a.copy()
        prev_g = g.copy()

        # Take step with projection
        a_new = a - step_size * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try smaller step
            a_new2 = a - step_size * 0.3 * g
            a_new2 = project_simplex(a_new2)
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_obj - 1e-12:
                best_obj = obj_new2
                best_a = a_new2.copy()
                a = a_new2.copy()
            else:
                step_size *= 0.7
                it += 1
                if it % 100 == 0 and verbose:
                    print(f"    CP it{it}: obj={best_obj:.8f}, step={step_size:.8f}")

        it += 1

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """
    Paired coordinate descent: move mass between positions to reduce peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from peak-contributing positions to non-contributing ones
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                contributors.append((contrib, i, j))

        contributors.sort(reverse=True)

        # Non-contributors (positions i where k_star - i is out of range)
        non_contrib = set(range(n))
        for _, i, j in contributors:
            non_contrib.discard(i)
            non_contrib.discard(j)
        non_contrib_list = list(non_contrib)

        if not non_contrib_list:
            break

        for contrib, i, j in contributors[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            for target in non_contrib_list[:5]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Random
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangular (inverted)
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like (U-shaped)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 5:
        # Smooth random (low frequency)
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Tail-heavy
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2 + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            print(f"  Mirror descent: obj={best_score:.8f} **")
        else:
            print(f"  Mirror descent: obj={obj_md:.8f} (no improvement)")

    # Multiplicative update
    remaining = deadline - time.monotonic()
    if remaining > 30:
        mu_budget = min(remaining * 0.15, 150.0)
        a_mu = multiplicative_update_optimize(best_seq, time_budget=mu_budget, verbose=True)
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            print(f"  Multiplicative: obj={best_score:.8f} **")
        else:
            print(f"  Multiplicative: obj={obj_mu:.8f} (no improvement)")

    # Phase 2: LP Alternating (main workhorse)
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    phase2_budget = min(remaining * 0.4, 500.0)

    if phase2_budget > 30:
        # Run multiple rounds
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            alloc = min(phase2_budget / (3 - rnd), remaining * 0.5)
            if alloc < 20:
                break
            a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # Chase peak refinement
    remaining = deadline - time.monotonic()
    if remaining > 30:
        cp_budget = min(remaining * 0.1, 60.0)
        a_cp = chase_peak_refine(best_seq, time_budget=cp_budget, verbose=True)
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  Chase peak: obj={best_score:.8f} **")
        else:
            print(f"  Chase peak: obj={obj_cp:.8f} (no improvement)")

    # Phase 3: Multi-n exploration
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 50, n_current - 100,
        n_current + 50, n_current + 100, n_current + 200,
        500, 600, 700, 800, 900, 1000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1200]

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick mirror descent then LP
        alloc = min(remaining * 0.2, 80.0)
        half_alloc = alloc / 2

        if half_alloc > 10:
            a_md = mirror_descent_optimize(a_try, time_budget=half_alloc, topK=3)
            obj_md = compute_obj(a_md)
            if obj_md < compute_obj(a_try):
                a_try = a_md

        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.15, 60.0)
        if lp_alloc > 15:
            a_opt = lp_alternating(a_try, time_budget=lp_alloc)
            obj_opt = compute_obj(a_opt)
        else:
            a_opt = a_try
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 4: Diverse restarts with LP
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_restarts = 4

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)

        # Blend with best
        blend = 0.3
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.2, 80.0)
        a_opt = lp_alternating(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        # Mirror descent polish
        md_alloc = min(remaining * 0.2, 60.0)
        a_md = mirror_descent_optimize(best_seq, time_budget=md_alloc, topK=5)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP polish
        lp_alloc = min(remaining * 0.3, 100.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Chase peak
        cp_alloc = min(remaining * 0.3, 40.0)
        a_cp = chase_peak_refine(best_seq, time_budget=cp_alloc)
        obj_cp = compute_obj(a_cp)
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  Final CP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        # Coordinate refinement
        cd_alloc = min(remaining * 0.4, 30.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # One more LP round
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 40.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
