def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b


def build_conv_matrix_sparse(a, rows=None):
    """Build sparse matrix A (m x n) such that A[i] @ g = conv(a,g)[rows[i]]."""
    from scipy.sparse import lil_matrix
    n = len(a)
    if rows is None:
        rows = list(range(2 * n - 1))
    m = len(rows)
    A = lil_matrix((m, n), dtype=np.float64)
    for idx, k in enumerate(rows):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[idx, i] = a[j]
    return A.tocsr()


def solve_lp_active(a, top_k=30, time_limit=5.0):

        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                return g_val, float(res.x[n])
    except Exception:
        pass

    return None, None


def golden_section_line_search(a, g, obj_current, n_trials=15):
    """
    Golden section search along direction from a to g.
    Finds best t in [0,1] minimizing obj((1-t)*a + t*g).
    """
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    inv_phi = 1.0 / phi
    inv_phi2 = 1.0 / (phi * phi)

    lo = 0.0
    hi = 1.0
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)

    fc = compute_obj(np.maximum((1 - c) * a + c * g, 0))
    fd = compute_obj(np.maximum((1 - d) * a + d * g, 0))

    for _ in range(n_trials):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = compute_obj(np.maximum((1 - c) * a + c * g, 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = compute_obj(np.maximum((1 - d) * a + d * g, 0))

    t_best = (lo + hi) / 2.0
    a_best = np.maximum((1 - t_best) * a + t_best * g, 0)
    s = float(np.sum(a_best))
    if s < 1e-10:
        return a, obj_current, 0.0
    a_best = a_best / s
    obj_best = compute_obj(a_best)
    return a_best, obj_best, t_best


def alternating_optimize_v2(a, time_budget=100.0, top_k=30, verbose=False):
    """
    Improved LP alternating with active-set LP, golden section search,
    and dual-guided perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    avg_a = a.copy()  # Nesterov averaging
    avg_count = 0

    max_iters = 300
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve active-set LP
        g_lp, M_lp, duals, top_pos = solve_lp_active(a, top_k=top_k, time_limit=5.0)

        if g_lp is None:
            # Fall back to full LP
            g_lp, M_lp = solve_lp_full(a, time_limit=10.0)
            if g_lp is None:
                break

        # Golden section line search
        a_new, obj_new, t_best = golden_section_line_search(a, g_lp, best_obj, n_trials=12)

        if obj_new < best_obj - 1e-12:
            # Update
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            last_t = t_best
            stall = 0
            # Nesterov averaging
            avg_count += 1
            avg_a = (avg_a * (avg_count - 1) + a) / avg_count
            if verbose and it % 10 == 0:
                print(f"    alt_v2 it{it}: obj={best_obj:.8f}, t={t_best:.3f}, M_lp={M_lp:.6f}")
        else:
            stall += 1
            if stall > 15:
                # Dual-guided perturbation
                if duals is not None and len(duals) > 0:
                    # Use dual variables to guide perturbation
                    duals = np.array(duals, dtype=np.float64)
                    duals = duals / (np.sum(duals) + 1e-10)
                    # Create perturbation direction from duals
                    pert = np.zeros(n)
                    for idx, k in enumerate(top_pos):
                        w = duals[idx] if idx < len(duals) else 0.01
                        for i in range(n):
                            j = k - i
                            if 0 <= j < n:
                                pert[i] += w * a[j]
                    # Move away from peak-contributing positions
                    rng = np.random.default_rng(hash((it, n, stall)) % (2**31))
                    scale = 0.01 + 0.02 * min(stall // 15, 5)
                    a = best_a * np.exp(rng.normal(0, scale, n) - 0.1 * pert / (np.max(pert) + 1e-10))
                else:
                    rng = np.random.default_rng(hash((it, n, stall)) % (2**31))
                    scale = 0.02 + 0.02 * min(stall // 15, 5)
                    a = best_a * np.exp(rng.normal(0, scale, n))

                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    avg_a = a.copy()
                    avg_count = 1
                    stall = 0

    # Check averaged solution
    s_avg = float(np.sum(avg_a))
    if s_avg > 1e-10:
        avg_a = avg_a / s_avg
        obj_avg = compute_obj(avg_a)
        if obj_avg < best_obj:
            best_a = avg_a
            best_obj = obj_avg

    return best_a


def peak_balance_refine(a, time_budget=30.0, n_iter=50):
    """
    Refine by targeting the top peaks specifically.
    For each top peak position, solve a mini-LP to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top 5 peaks
        top_idx = np.argsort(conv)[-5:]
        if len(top_idx) == 0:
            break

        # For each top peak, try to reduce it via coordinate moves
        improved = False
        for k_star in top_idx:
            if time.monotonic() - start > time_budget:
                break
            # Find which pairs (i, k_star - i) contribute most to this peak
            contribs = []
            for i in range(n):
                j = int(k_star) - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)

            # Try moving mass from high-contribution pairs
            for _, i, j in contribs[:10]:
                if i == j:
                    continue
                # Move small amount from max(a[i], a[j]) to a random other position
                src = i if a[i] > a[j] else j
                dst = np.random.randint(0, n)
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * a[src]
                    if a[src] - delta < 0:
                        continue
                    a[src] -= delta
                    a[dst] += delta
                    s = float(np.sum(a))
                    if s < 1e-10:
                        a[src] += delta
                        a[dst] -= delta
                        continue
                    a_norm = a / s
                    new_obj = 2.0 * n * float(np.max(compute_conv(a_norm)))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[dst] -= delta
                        break

        if not improved:
            break

    return best_a


def subgradient_step(a, step_size=0.02, n_subgrad=5):
    """
    One subgradient step for the non-convex objective.
    Uses random subgradients from top peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    top_idx = np.argsort(conv)[-n_subgrad:]

    # Average subgradient over top peaks
    grad = np.zeros(n)
    for k_star in top_idx:
        for i in range(n):
            j = int(k_star) - i
            if 0 <= j < n:
                grad[i] += a[j]
    grad /= len(top_idx)

    # Projected subgradient step (project onto {a >= 0, sum(a) = 1})
    a_new = a - step_size * (grad - np.mean(grad))
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif mode == 2:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif mode == 3:
        # Periodic-like
        period = max(2, n // 7)
        a = np.ones(n)
        for i in range(n):
            if (i % period) < period // 2:
                a[i] = 1.0
            else:
                a[i] = 0.3
        return a / np.sum(a)
    elif mode == 4:
        # Random smooth
        a = np.ones(n)
        for _ in range(3):
            kernel_size = max(3, n // 20 | 1)
            if kernel_size % 2 == 0:
                kernel_size += 1
            kernel = np.ones(kernel_size) / kernel_size
            a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif mode == 5:
        # Triangle
        mid = n // 2
        a = 1.0 - np.abs(np.arange(n) - mid) / mid
        a = np.maximum(a, 0.1)
        return a / np.sum(a)
    elif mode == 6:
        # Inverse cosine (peaked at edges)
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.abs(np.cos(x))
        return a / np.sum(a)
    else:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        return a / np.sum(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining active-set LP alternating,
    peak balancing, subgradient methods, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                n_parent = len(best_seq)
                print(f"Parent (n={n_parent}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Intensive LP alternating with active set ===
    print(f"\n=== Phase 1: Active-Set LP Alternating (n={len(best_seq)}) ===")
    phase1_budget = min(450.0, (deadline - start_time) * 0.35)

    # Run multiple rounds with increasing top_k
    for round_idx, tk in enumerate([20, 30, 50, 80]):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = phase1_budget / (4 - round_idx)
        alloc = min(alloc, remaining * 0.7)

        a = best_seq.copy()
        a_new = alternating_optimize_v2(a, time_budget=alloc, top_k=tk, verbose=(round_idx == 0))
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase1 round{round_idx} (top_k={tk}): obj={best_score:.8f} **")
        else:
            print(f"  Phase1 round{round_idx} (top_k={tk}): obj={obj:.8f}")

    # === Phase 2: Peak balance refinement ===
    print(f"\n=== Phase 2: Peak Balance Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_pb = peak_balance_refine(best_seq, time_budget=min(remaining * 0.15, 40.0))
        obj_pb = compute_obj(a_pb)
        if obj_pb < best_score:
            best_score = obj_pb
            best_seq = a_pb.copy()
            print(f"  Peak balance: obj={best_score:.8f} **")
        else:
            print(f"  Peak balance: obj={obj_pb:.8f}")

    # === Phase 3: Subgradient + LP hybrid ===
    print(f"\n=== Phase 3: Subgradient + LP Hybrid ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a = best_seq.copy()
        best_a_phase3 = a.copy()
        best_obj_phase3 = compute_obj(a)
        sub_start = time.monotonic()
        step_size = 0.03

        for sg_iter in range(200):
            if time.monotonic() - sub_start > min(remaining * 0.2, 60.0):
                break
            a = subgradient_step(a, step_size=step_size, n_subgrad=3)
            obj_sg = compute_obj(a)
            if obj_sg < best_obj_phase3:
                best_obj_phase3 = obj_sg
                best_a_phase3 = a.copy()
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a.copy()
                    print(f"  Subgrad it{sg_iter}: obj={best_score:.8f} **")
            else:
                step_size *= 0.95

        # Follow up with LP on best subgradient result
        remaining = deadline - time.monotonic()
        if remaining > 30 and best_obj_phase3 < best_score + 1e-6:
            a_lp = alternating_optimize_v2(best_a_phase3, time_budget=min(remaining * 0.3, 50.0), top_k=40)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Post-subgrad LP: obj={best_score:.8f} **")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_options = sorted(set([
        max(200, n_parent - 100), max(200, n_parent - 50), n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.15, 40.0)
        a_opt = alternating_optimize_v2(a_try, time_budget=alloc, top_k=30)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse random restarts ===
    print(f"\n=== Phase 5: Diverse Random Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    restart_configs = [
        (n_best, 0), (n_best, 1), (n_best, 2), (n_best, 3),
        (n_best, 4), (n_best, 5), (n_best, 6),
        (min(n_best + 100, 1200), 0), (max(n_best - 100, 300), 2),
    ]

    for r, (n_r, mode) in enumerate(restart_configs):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(remaining * 0.12, 30.0)

        if r < 3:
            # Use diverse initializations
            a_r = diverse_init(n_r, rng, mode=mode)
        else:
            # Perturbed best
            a_r = best_seq[:n_r] if n_r == n_best else resample_sequence(best_seq, n_r)
            a_r = a_r * np.exp(rng.normal(0, 0.15, len(a_r)))
            a_r = np.maximum(a_r, 1e-6)
            a_r = a_r / np.sum(a_r)

        a_opt = alternating_optimize_v2(a_r, time_budget=alloc, top_k=25)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (n={n_r}, mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (n={n_r}, mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        # LP with large top_k
        alloc = min(remaining * 0.35, 80.0)
        a_final = alternating_optimize_v2(best_seq, time_budget=alloc, top_k=60, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP (top_k=60): obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    if remaining > deadline - time.monotonic() + 30:
        pass  # timing check

    # One more peak balance
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_pb2 = peak_balance_refine(best_seq, time_budget=min(remaining * 0.3, 30.0))
        obj_pb2 = compute_obj(a_pb2)
        if obj_pb2 < best_score:
            best_score = obj_pb2
            best_seq = a_pb2.copy()
            print(f"  Final peak balance: obj={best_score:.8f} **")

    # Final subgradient sweep
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a = best_seq.copy()
        best_a_sf = a.copy()
        best_obj_sf = compute_obj(a)
        step_size = 0.01
        for _ in range(int(remaining * 10)):
            if time.monotonic() > deadline - 10:
                break
            a = subgradient_step(a, step_size=step_size, n_subgrad=2)
            obj_sf = compute_obj(a)
            if obj_sf < best_obj_sf:
                best_obj_sf = obj_sf
                best_a_sf = a.copy()
            else:
                step_size *= 0.9
                if step_size < 1e-5:
                    break
        if best_obj_sf < best_score:
            best_score = best_obj_sf
            best_seq = best_a_sf.copy()
            print(f"  Final subgrad: obj={best_score:.8f} **")

    # Ensure valid output
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
